# AITeam 黑盒测试策略与实施方案

## 📋 概述

本文档描述了在 AITeam 多 Agent 协作系统中，如何利用测试 Agent 实现全流程黑盒测试，保证代码交付质量。

---

## 🎯 核心目标

1. **全流程覆盖**: 从需求分析到代码交付，每个环节都有测试验证
2. **自动化执行**: 测试 Agent 自动执行测试用例，生成报告
3. **质量保证**: 建立多维度质量评分体系
4. **持续改进**: 基于测试反馈持续优化代码质量

```python
class CriticalErrorDetector:
    """严重错误检测器（🔴 必须修复，代码无法运行）"""
    
    def check_empty_methods(self, code: str) -> List[Issue]:
        """1. 空方法体检测"""
        issues = []
        pattern = r'(function|method)\s+\w+\s*\([^)]*\)\s*\{\s*(//|/\*|\}'
        
        for match in re.finditer(pattern, code):
            issues.append({
                "type": "empty_method",
                "severity": "critical",
                "message": "方法体为空，没有实际执行代码",
                "location": match.group(),
                "suggestion": "实现方法的具体逻辑"
            })
        
        return issues
    
    def check_duplicate_definitions(self, code: str) -> List[Issue]:
        """2. 重复定义检测"""
        issues = []
        
        # 检测重复的 class 定义
        classes = re.findall(r'class\s+(\w+)', code)
        duplicates = [cls for cls in classes if classes.count(cls) > 1]
        
        if duplicates:
            issues.append({
                "type": "duplicate_class",
                "severity": "critical",
                "message": f"重复定义的类: {set(duplicates)}",
                "suggestion": "删除重复定义，保留一个版本"
            })
        
        return issues
    
    def check_missing_game_loop(self, code: str) -> List[Issue]:
        """3. 缺少游戏循环检测"""
        issues = []
        
        has_game_class = 'class Game' in code or 'class.*Game' in code
        has_loop = 'requestAnimationFrame' in code or 'setInterval' in code
        
        if has_game_class and not has_loop:
            issues.append({
                "type": "missing_game_loop",
                "severity": "critical",
                "message": "游戏类定义了但没有游戏循环",
                "suggestion": "添加 gameLoop() { requestAnimationFrame(() => this.gameLoop()); }"
            })
        
        return issues
    
    def check_undefined_variables(self, code: str) -> List[Issue]:
        """5. 未定义变量检测"""
        issues = []
        
        # 常见的 Canvas/WebGL 变量
        common_vars = ['ctx', 'canvas', 'gl', 'game']
        
        for var in common_vars:
            # 检查是否使用但未定义
            used = re.search(rf'\b{var}\b', code)
            defined = re.search(rf'(const|let|var)\s+{var}\s*=', code) or \
                      re.search(rf'this\.{var}\s*=', code)
            
            if used and not defined:
                issues.append({
                    "type": "undefined_variable",
                    "severity": "critical",
                    "message": f"使用了未定义的变量: {var}",
                    "suggestion": f"在构造函数中初始化: this.{var} = ..."
                })
        
        return issues
```

```python
class MediumIssueDetector:
    """中等问题检测器（🟡 应该修复，影响代码质量）"""
    
    def check_boundary_checks(self, code: str) -> List[Issue]:
        """边界检查缺失检测"""
        issues = []
        
        # 检测数组访问没有边界检查
        array_access = re.findall(r'\w+\[(\w+)\]', code)
        
        for index_var in array_access:
            # 检查是否有边界检查
            has_check = re.search(rf'if\s*\(\s*{index_var}\s*[<>=]', code)
            
            if not has_check:
                issues.append({
                    "type": "missing_boundary_check",
                    "severity": "medium",
                    "message": f"数组访问缺少边界检查: {index_var}",
                    "suggestion": f"添加: if ({index_var} >= 0 && {index_var} < array.length)"
                })
        
        return issues
    
    def check_error_handling(self, code: str) -> List[Issue]:
        """错误处理缺失检测"""
        issues = []
        
        # 检测可能抛出异常的代码没有 try-catch
        risky_operations = [
            (r'JSON\.parse\(', 'JSON.parse 缺少错误处理'),
            (r'fetch\(', 'fetch 缺少错误处理'),
            (r'localStorage\.', 'localStorage 操作缺少错误处理'),
        ]
        
        for pattern, message in risky_operations:
            matches = re.finditer(pattern, code)
            
            for match in matches:
                # 检查周围是否有 try-catch
                line_num = code[:match.start()].count('\n')
                lines = code.split('\n')
                context = '\n'.join(lines[max(0, line_num-5):line_num+5])
                
                if 'try' not in context:
                    issues.append({
                        "type": "missing_error_handling",
                        "severity": "medium",
                        "message": message,
                        "location": match.group(),
                        "suggestion": "添加 try-catch 错误处理"
                    })
        
        return issues
```

```python
class AutomatedTestExecutor:
    """自动化测试执行器"""
    
    async def execute_functional_tests(
        self, 
        test_url: str,
        test_cases: List[TestCase]
    ) -> TestReport:
        """
        执行功能测试
        
        Args:
            test_url: 测试页面 URL
            test_cases: 测试用例列表
        
        Returns:
            TestReport: 测试报告
        """
        results = []
        
        for test_case in test_cases:
            result = await self.run_single_test(test_url, test_case)
            results.append(result)
        
        # 生成报告
        report = self.generate_report(results)
        
        return report
    
    async def run_single_test(
        self, 
        url: str, 
        test_case: TestCase
    ) -> TestResult:
        """执行单个测试用例"""
        
        # 1. 使用 Playwright/Puppeteer 打开页面
        browser = await self.launch_browser()
        page = await browser.new_page()
        
        try:
            # 2. 导航到测试 URL
            await page.goto(url)
            
            # 3. 执行测试步骤
            for step in test_case.steps:
                await self.execute_step(page, step)
            
            # 4. 验证结果
            passed = await self.verify_result(page, test_case.expected)
            
            return TestResult(
                test_case=test_case.name,
                passed=passed,
                screenshot=await page.screenshot(),
                logs=await self.get_console_logs(page)
            )
        
        finally:
            await browser.close()
    
    async def execute_step(self, page, step: TestStep):
        """执行测试步骤"""
        
        if step.action == "click":
            await page.click(step.selector)
        
        elif step.action == "type":
            await page.type(step.selector, step.value)
        
        elif step.action == "wait":
            await page.wait_for_timeout(step.duration)
        
        elif step.action == "verify":
            element = await page.query_selector(step.selector)
            text = await element.inner_text()
            assert step.expected in text
```

```python
class RegressionTestManager:
    """回归测试管理器"""
    
    def __init__(self):
        self.test_history = []
        self.baseline_results = {}
    
    async def run_regression_tests(
        self,
        test_url: str,
        changed_files: List[str]
    ) -> RegressionReport:
        """
        执行回归测试
        
        Args:
            test_url: 测试 URL
            changed_files: 变更的文件列表
        
        Returns:
            RegressionReport: 回归测试报告
        """
        # 1. 识别受影响的测试用例
        affected_tests = self.identify_affected_tests(changed_files)
        
        # 2. 执行受影响的测试
        results = []
        for test_case in affected_tests:
            result = await self.execute_test(test_url, test_case)
            results.append(result)
        
        # 3. 对比基线结果
        comparison = self.compare_with_baseline(results)
        
        # 4. 生成报告
        report = RegressionReport(
            total_tests=len(affected_tests),
            passed=comparison['passed'],
            failed=comparison['failed'],
            new_bugs=comparison['new_bugs'],
            fixed_bugs=comparison['fixed_bugs']
        )
        
        return report
    
    def identify_affected_tests(self, changed_files: List[str]) -> List[TestCase]:
        """识别受影响的测试用例"""
        affected = []
        
        for file in changed_files:
            # 根据文件类型识别相关测试
            if file.endswith('.js') or file.endswith('.ts'):
                affected.extend(self.get_js_tests(file))
            elif file.endswith('.css'):
                affected.extend(self.get_ui_tests(file))
        
        return affected
```

### 多维度质量评估

```python
class QualityScorer:
    """质量评分器"""
    
    def calculate_overall_score(self, test_results: TestResults) -> QualityScore:
        """
        计算综合质量分数
        
        维度：
        1. 正确性 (40%): 核心功能是否正确
        2. 完整性 (30%): 功能是否完整
        3. 健壮性 (20%): 错误处理是否完善
        4. 性能 (10%): 性能是否达标
        """
        
        # 正确性评分
        correctness = self.score_correctness(test_results)
        
        # 完整性评分
        completeness = self.score_completeness(test_results)
        
        # 健壮性评分
        robustness = self.score_robustness(test_results)
        
        # 性能评分
        performance = self.score_performance(test_results)
        
        # 加权平均
        total = (
            correctness * 0.4 +
            completeness * 0.3 +
            robustness * 0.2 +
            performance * 0.1
        )
        
        return QualityScore(
            total=total,
            breakdown={
                "correctness": correctness,
                "completeness": completeness,
                "robustness": robustness,
                "performance": performance
            },
            passed=total >= 60,  # 60 分及格线
            grade=self.get_grade(total)
        )
    
    def get_grade(self, score: float) -> str:
        """获取等级"""
        if score >= 90:
            return "A+ (优秀)"
        elif score >= 80:
            return "A (良好)"
        elif score >= 70:
            return "B (合格)"
        elif score >= 60:
            return "C (及格)"
        else:
            return "D (不及格)"
```

```python
        test_cases = await self.generate_test_cases(code_output)
        
        # 阶段 3: 执行功能测试
        functional_results = await self.execute_functional_tests(
            test_url, 
            test_cases
        )
        
        # 阶段 4: 质量评分
        quality_score = self.calculate_quality_score(
            static_analysis,
            functional_results
        )
        
        # 阶段 5: 生成报告
        report = self.generate_final_report(
            static_analysis,
            functional_results,
            quality_score
        )
        
        return report
```

### 2. 与其他 Agent 的协作

```python
# 在 Coordinator 中的集成

class Coordinator:
    async def execute_pipeline(self, pipeline: Pipeline):
        """执行 Pipeline 流程"""
        
        for stage in pipeline.stages:
            if stage.agent_type == "coder":
                # Coder 完成编码
                code_output = await self.assign_to_coder(stage.task)
            
            elif stage.agent_type == "tester":
                # Tester 执行测试
                test_result = await self.assign_to_tester(
                    code_output=code_output,
                    test_url=pipeline.test_url
                )
                
                # 如果测试不通过，返回给 Coder 修复
                if not test_result.passed:
                    await self.request_fix(coder, test_result.issues)
                    # 重新执行 Pipeline
                    return await self.execute_pipeline(pipeline)
```

```python
class TesterEvolution:
    """测试 Agent 进化系统"""
    
    def learn_from_feedback(self, feedback: TestFeedback):
        """从反馈中学习"""
        
        # 记录误报
        if feedback.false_positive:
            self.false_positive_patterns.add(feedback.pattern)
        
        # 记录漏报
        if feedback.missed_bug:
            self.missed_bug_patterns.add(feedback.pattern)
        
        # 更新检测规则
        self.update_detection_rules()
    
    def update_detection_rules(self):
        """更新检测规则"""
        
        # 减少误报
        for pattern in self.false_positive_patterns:
            self.rules.exclude_pattern(pattern)
        
        # 增加漏报检测
        for pattern in self.missed_bug_patterns:
            self.rules.add_pattern(pattern)
```
