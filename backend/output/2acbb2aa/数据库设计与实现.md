        class DatabaseManager {
            constructor() {
                this.db = null;
                this.initDB();
            }
            
            initDB() {
                const request = indexedDB.open('TestCaseDB', 1);
                
                request.onupgradeneeded = (event) => {
                    const db = event.target.result;
                    
                    // 创建测试用例表
                    if (!db.objectStoreNames.contains('testCases')) {
                        const testCaseStore = db.createObjectStore('testCases', { keyPath: 'id', autoIncrement: true });
                        testCaseStore.createIndex('name', 'name', { unique: false });
                        testCaseStore.createIndex('category', 'category', { unique: false });
                        testCaseStore.createIndex('status', 'status', { unique: false });
                    }
                    
                    // 创建测试历史表
                    if (!db.objectStoreNames.contains('testHistory')) {
                        const historyStore = db.createObjectStore('testHistory', { keyPath: 'id', autoIncrement: true });
                        historyStore.createIndex('testCaseId', 'testCaseId', { unique: false });
                        historyStore.createIndex('status', 'status', { unique: false });
                        historyStore.createIndex('executionTime', 'executionTime', { unique: false });
                    }
                    
                    // 创建配置表

                        const configStore = db.createObjectStore('config', { keyPath: 'key' });
