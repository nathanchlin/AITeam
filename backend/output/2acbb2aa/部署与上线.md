        // 部署模拟功能
        function initializeDeploymentSimulation() {
            const startBtn = document.getElementById('startDeployment');
            const resetBtn = document.getElementById('resetDeployment');
            const statusMessage = document.getElementById('statusMessage');
            const mainProgressBar = document.getElementById('mainProgressBar');
            const flowProgress = document.getElementById('flowProgress');
            
            let currentStep = 1;
            let stepProgress = [0, 0, 0, 0];
            let simulationInterval;
            
            startBtn.addEventListener('click', startDeploymentSimulation);
            resetBtn.addEventListener('click', resetDeploymentSimulation);
            
            function startDeploymentSimulation() {
                startBtn.disabled = true;
                statusMessage.style.display = 'block';
                statusMessage.className = 'status-message status-info';
                statusMessage.textContent = '开始部署流程...';
                
                simulationInterval = setInterval(updateDeploymentStep, 500);
            }
            
            function updateDeploymentStep() {
                if (currentStep <= 4) {
                    if (stepProgress[currentStep - 1] < 100) {
                        stepProgress[currentStep - 1] += 10;
                        
                        // 更新步骤进度
                        document.getElementById(`step${currentStep}Progress`).textContent = `${stepProgress[currentStep - 1]}%`;
                        
                        // 更新主进度条
                        const totalProgress = stepProgress.reduce((a, b) => a + b, 0) / 4;
                        mainProgressBar.style.width = `${totalProgress}%`;
                        mainProgressBar.textContent = `${Math.round(totalProgress)}%`;
                        
                        // 更新流程线进度
                        flowProgress.style.width = `${(currentStep - 1) * 25 + stepProgress[currentStep - 1] / 4}%`;
                        
                        // 更新状态信息
                        updateStatusMessage(currentStep, stepProgress[currentStep - 1]);
                        
                        // 更新步骤样式
                        updateStepStyles();
                    } else {
                        currentStep++;
                    }
                } else {
                    clearInterval(simulationInterval);
                    statusMessage.className = 'status-message status-success';
                    statusMessage.textContent = '部署完成！应用已成功上线。';
                    startBtn.disabled = false;
                }
            }
            
            function updateStatusMessage(step, progress) {
                const messages = [
                    { step: 1, messages: ['正在准备云环境...', '配置网络和安全组...', '安装运行环境...'] },
                    { step: 2, messages: ['打包应用代码...', '配置CI/CD流水线...', '部署应用到服务器...'] },
                    { step: 3, messages: ['设置监控系统...', '配置日志收集...', '配置告警规则...'] },
                    { step: 4, messages: ['执行最终检查...', '启动应用服务...', '验证系统运行...'] }
                ];
                
                const stepInfo = messages[step - 1];
                const messageIndex = Math.floor(progress / 33.33);
                statusMessage.textContent = stepInfo.messages[Math.min(messageIndex, stepInfo.messages.length - 1)];
            }
            
            function updateStepStyles() {
                for (let i = 1; i <= 4; i++) {
                    const stepElement = document.getElementById(`step${i}`);
                    if (i < currentStep) {
                        stepElement.classList.add('completed');
                        stepElement.classList.remove('active');
                    } else if (i === currentStep) {
                        stepElement.classList.add('active');
                        stepElement.classList.remove('completed');
                    } else {
                        stepElement.classList.remove('active', 'completed');
                    }
                }
            }

                clearInterval(simulationInterval);
