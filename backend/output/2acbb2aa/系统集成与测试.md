        // API测试

            log('开始API测试...');
            
            // 测试1: 基本连接

            // 测试2: 数据获取

        }

        // 添加测试
        function addTest(testName) {
            testResults.push({
                name: testName,
                status: 'pending',
                message: '',
                timestamp: new Date().toLocaleTimeString()
            });
            totalTests++;
        }
        
        // 测试通过
        function passTest(testName) {
            const testIndex = testResults.findIndex(t => t.name === testName);
            if (testIndex !== -1) {
                testResults[testIndex].status = 'success';
