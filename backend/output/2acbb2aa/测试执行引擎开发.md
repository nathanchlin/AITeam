    <script>
        // 测试执行引擎核心代码
        document.addEventListener('DOMContentLoaded', function() {
            // 全局变量
            const testEngine = {
                tests: [],
                currentTestIndex: 0,
                isRunning: false,
                status: 'inactive',
                testResults: {
                    total: 0,
                    passed: 0,
                    failed: 0,
                    pending: 0
                },
                logs: []
            };
            
            // DOM元素
            const elements = {
                statusDot: document.getElementById('statusDot'),
                statusText: document.getElementById('statusText'),
                testResults: document.getElementById('testResults'),
                codeEditor: document.getElementById('codeEditor'),
                logContainer: document.getElementById('logContainer'),
                progressBar: document.getElementById('progressBar'),
                totalTests: document.getElementById('totalTests'),
                passedTests: document.getElementById('passedTests'),
                failedTests: document.getElementById('failedTests'),
                pendingTests: document.getElementById('pendingTests'),
                testCanvas: document.getElementById('testCanvas')
            };
            
            // 初始化测试用例
            function initializeTests() {
                // 添加默认测试用例
                testEngine.tests = [
                    { id: 1, name: '斐波那契数列测试', code: 'testFibonacci()', status: 'pending' },
                    { id: 2, name: '排序算法测试', code: 'testSort()', status: 'pending' },
                    { id: 3, name: '字符串处理测试', code: 'testString()', status: 'pending' }
                ];
                updateTestResults();
                updateStats();
            }
            
            // 更新状态指示器
            function updateStatus(status, text) {
                testEngine.status = status;
                elements.statusText.textContent = text;
                
                if (status === 'active') {
                    elements.statusDot.classList.remove('inactive');
                } else {
                    elements.statusDot.classList.add('inactive');
                }
            }
            
            // 添加日志
            function addLog(message, type = 'info') {
                const timestamp = new Date().toLocaleTimeString();
                const logEntry = {
                    message,
                    type,
                    timestamp
                };
                
                testEngine.logs.push(logEntry);
                renderLogs();
            }
            
            // 渲染日志
            function renderLogs() {
                elements.logContainer.innerHTML = '';
                
                testEngine.logs.forEach(log => {
                    const logDiv = document.createElement('div');
                    logDiv.className = `log-entry log-${log.type}`;
                    logDiv.textContent = `[${log.timestamp}] ${log.message}`;
                    elements.logContainer.appendChild(logDiv);
                });
                
                // 滚动到底部
                elements.logContainer.scrollTop = elements.logContainer.scrollHeight;
            }
            
            // 更新测试结果显示
            function updateTestResults() {
                elements.testResults.innerHTML = '';
                
                testEngine.tests.forEach(test => {
                    const testItem = document.createElement('div');
                    testItem.className = `test-item ${test.status}`;
                    
                    testItem.innerHTML = `
                        <div class="test-name">${test.name}</div>
                        <div class="test-status status-${test.status}">
                            ${test.status === 'passed' ? '通过' : test.status === 'failed' ? '失败' : '待执行'}
                        </div>
                    `;
                    
                    elements.testResults.appendChild(testItem);
                });
            }
            
            // 更新统计信息
            function updateStats() {
                testEngine.testResults = {
                    total: testEngine.tests.length,
                    passed: testEngine.tests.filter(t => t.status === 'passed').length,
                    failed: testEngine.tests.filter(t => t.status === 'failed').length,
                    pending: testEngine.tests.filter(t => t.status === 'pending').length
                };
                
                elements.totalTests.textContent = testEngine.testResults.total;
                elements.passedTests.textContent = testEngine.testResults.passed;
                elements.failedTests.textContent = testEngine.testResults.failed;
                elements.pendingTests.textContent = testEngine.testResults.pending;
                
                // 更新进度条
                const progress = testEngine.testResults.total > 0 
                    ? (testEngine.testResults.passed + testEngine.testResults.failed) / testEngine.testResults.total * 100 
                    : 0;
                elements.progressBar.style.width = `${progress}%`;
            }
            
            // 运行测试
            async function runTests() {
                if (testEngine.isRunning) {
                    addLog('测试正在运行中，请稍候...', 'warning');
                    return;
                }
                
                if (testEngine.tests.length === 0) {
                    addLog('没有可执行的测试用例', 'warning');
                    return;
                }
                
                testEngine.isRunning = true;
                updateStatus('active', '测试运行中');
                addLog('开始执行测试用例', 'info');
                
                // 重置测试状态
                testEngine.tests.forEach(test => {
                    test.status = 'pending';
                });
                testEngine.currentTestIndex = 0;
                updateTestResults();
                updateStats();
                
                // 逐个执行测试
                for (let i = 0; i < testEngine.tests.length; i++) {
                    const test = testEngine.tests[i];
                    testEngine.currentTestIndex = i;
                    
                    addLog(`正在执行测试: ${test.name}`, 'info');
                    updateTestResults();
                    
                    try {
                        // 模拟测试执行
                        await executeTest(test);
                        
                        // 检查测试结果
                        const testResult = await evaluateTest(test);
                        
                        if (testResult) {
                            test.status = 'passed';
                            addLog(`测试通过: ${test.name}`, 'info');
                        } else {
                            test.status = 'failed';
                            addLog(`测试失败: ${test.name}`, 'error');
                        }
                    } catch (error) {
                        test.status = 'failed';
                        addLog(`测试出错: ${test.name} - ${error.message}`, 'error');
                    }
                    
                    updateTestResults();
                    updateStats();
                    
                    // 延迟以便观察进度
                    await new Promise(resolve => setTimeout(resolve, 500));
                }
                
                testEngine.isRunning = false;
                updateStatus('inactive', '测试完成');
                addLog('所有测试用例执行完毕', 'info');
                
                // 绘制测试结果图表
