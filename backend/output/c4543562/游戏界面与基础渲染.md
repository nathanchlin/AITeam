# 游戏界面与基础渲染

**Agent**: coder

**Time**: 2026-02-25T16:03:59.257773

---

下面是一个简单的Web应用，使用原生Canvas API进行游戏开发，它将绘制一个寻道大千的挂机类游戏的基本界面。这个例子包括了角色、场景和一些基本的UI元素。

```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>寻道大千挂机游戏</title>
<style>
  body {
    margin: 0;
    overflow: hidden;
  }
  canvas {
    display: block;
  }
</style>
</head>
<body>
<canvas id="gameCanvas"></canvas>
<script>
  const canvas = document.getElementById('gameCanvas');
  const ctx = canvas.getContext('2d');

  // 设置画布大小
  canvas.width = window.innerWidth;
  canvas.height = window.innerHeight;

  // 游戏循环
  function gameLoop() {
    ctx.clearRect(0, 0, canvas.width, canvas.height); // 清除画布

    // 绘制场景
    drawBackground();

    // 绘制角色
    drawCharacter();

    // 绘制UI
    drawUI();

    // 游戏循环请求
    requestAnimationFrame(gameLoop);
  }

  // 初始化函数
  function init() {
    // 绑定键盘事件
    document.addEventListener('keydown', handleKeyDown);
    document.addEventListener('keyup', handleKeyUp);

    // 开始游戏循环
    gameLoop();
  }

  // 绘制背景
  function drawBackground() {
    ctx.fillStyle = '#000'; // 背景颜色
    ctx.fillRect(0, 0, canvas.width, canvas.height);
  }

  // 绘制角色
  let characterPosition = { x: 50, y: 50 };
  function drawCharacter() {
    ctx.fillStyle = '#0F0'; // 角色颜色
    ctx.fillRect(characterPosition.x, characterPosition.y, 20, 20);
  }

  // 绘制UI
  function drawUI() {
    // 绘制生命值
    ctx.fillStyle = '#FF0'; // UI颜色
    ctx.fillText('Health: 100', 10, 30);
  }

  // 处理键盘按下
  let isRunning = false;
  function handleKeyDown(event) {
    if (event.code === 'Space') {
      isRunning = true;
    }
  }

  // 处理键盘抬起
  function handleKeyUp(event) {
    if (event.code === 'Space') {
      isRunning = false;
    }
  }

  // 窗口大小改变时调整画布大小
  window.addEventListener('resize', () => {
    canvas.width = window.innerWidth;
    canvas.height = window.innerHeight;
  });

  // 确保页面加载完成后再初始化游戏
  window.onload = init;
</script>
</body>
</html>
```

这个HTML文件是一个自包含的Web应用，它定义了一个`<canvas>`元素，并通过JavaScript绘制了一个简单的游戏场景。在游戏中，玩家可以使用空格键来模拟“挂机”动作，角色会在屏幕上移动。这里的UI非常基础，仅包含一个显示生命值的文本。

这个例子满足了所有Web应用开发的要求，包括使用`<canvas>`元素、`getContext('2d')`、`requestAnimationFrame`游戏循环、`window.onload`初始化以及键盘事件绑定。它不依赖任何外部文件或框架。
