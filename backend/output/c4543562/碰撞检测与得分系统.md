# 碰撞检测与得分系统

**Agent**: coder

**Time**: 2026-02-25T16:05:31.152074

---

为了实现碰撞检测和得分系统，我们需要在现有的代码基础上添加以下功能：

1. 碰撞检测：定义一个函数来检测角色是否与游戏中的障碍物发生了碰撞。
2. 得分系统：定义一个变量来追踪得分，并在角色与障碍物碰撞时增加得分。

以下是实现这些功能的代码示例：

```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>寻道大千挂机游戏</title>
<style>
  body {
    margin: 0;
    overflow: hidden;
  }
  canvas {
    display: block;
  }
</style>
</head>
<body>
<canvas id="gameCanvas"></canvas>
<script>
  const canvas = document.getElementById('gameCanvas');
  const ctx = canvas.getContext('2d');

  // 设置画布大小
  canvas.width = window.innerWidth;
  canvas.height = window.innerHeight;

  // 角色移动速度
  const speed = 5;

  // 得分
  let score = 0;

  // 游戏循环
  function gameLoop() {
    ctx.clearRect(0, 0, canvas.width, canvas.height); // 清除画布

    // 绘制背景
    drawBackground();

    // 更新并绘制角色
    updateCharacterPosition();
    drawCharacter();

    // 绘制UI
    drawUI();

    // 游戏循环请求
    requestAnimationFrame(gameLoop);
  }

  // 初始化函数
  function init() {
    // 绑定键盘事件
    document.addEventListener('keydown', handleKeyDown);
    document.addEventListener('keyup', handleKeyUp);

    // 开始游戏循环
    gameLoop();
  }

  // 绘制背景
  function drawBackground() {
    ctx.fillStyle = '#000'; // 背景颜色
    ctx.fillRect(0, 0, canvas.width, canvas.height);
  }

  // 绘制角色
  let characterPosition = { x: 50, y: 50 };
  function drawCharacter() {
    ctx.fillStyle = '#0F0'; // 角色颜色
    ctx.fillRect(characterPosition.x, characterPosition.y, 20, 20);
  }

  // 绘制UI
  function drawUI() {
    // 绘制生命值和得分
    ctx.fillStyle = '#FF0'; // UI颜色
    ctx.fillText('Health: 100', 10, 30);
    ctx.fillText('Score: ' + score, 10, 50);
  }

  // 处理键盘按下
  let isUp = false;
  let isDown = false;
  let isLeft = false;
  let isRight = false;
  function handleKeyDown(event) {
    switch (event.code) {
      case 'ArrowUp':
        isUp = true;
        break;
      case 'ArrowDown':
        isDown = true;
        break;
      case 'ArrowLeft':
        isLeft = true;
        break;
      case 'ArrowRight':
        isRight = true;
        break;
    }
  }

  // 处理键盘抬起
  function handleKeyUp(event) {
    switch (event.code) {
      case 'ArrowUp':
        isUp = false;
        break;
      case 'ArrowDown':
        isDown = false;
        break;
      case 'ArrowLeft':
        isLeft = false;
        break;
      case 'ArrowRight':
        isRight = false;
        break;
    }
  }

  // 更新角色位置
  function updateCharacterPosition() {
    if (isUp) {
      characterPosition.y -= speed;
    }
    if (isDown) {
      characterPosition.y += speed;
    }
    if (isLeft) {
      characterPosition.x -= speed;
    }
    if (isRight) {
      characterPosition.x += speed;
    }
    // 确保角色不会移出画布
    characterPosition.x = Math.max(0, Math.min(characterPosition.x, canvas.width - 20));
    characterPosition.y = Math.max(0, Math.min(characterPosition.y, canvas.height - 20));
  }

  // 窗口大小改变时调整画布大小
  window.addEventListener('resize', () => {
    canvas.width = window.innerWidth;
    canvas.height = window.innerHeight;
  });

  // 确保页面加载完成后再初始化游戏
  window.onload = init;
</script>
</body>
</html>
```

在这个代码中，我们添加了一个名为 `score` 的变量来追踪得分，并在 `drawUI` 函数中显示得分。目前，我们没有具体的障碍物来检测碰撞，因此碰撞检测函数是空的。在实际游戏中，你需要定义障碍物的位置和形状，并在 `updateCharacterPosition` 函数中添加碰撞检测逻辑。

以下是添加碰撞检测的示例代码：

```javascript
// 假设障碍物是一个矩形，位置在 (obstacleX, obstacleY) 和 (obstacleX + obstacleWidth, obstacleY + obstacleHeight)
const obstacleX = 100;
const obstacleY = 100;
const obstacleWidth = 50;
const obstacleHeight = 50;

function updateCharacterPosition() {
  // ...之前的代码

  // 检测碰撞
  if (characterPosition.x < obstacleX + obstacleWidth &&
      characterPosition.x + 20 > obstacleX &&
      characterPosition.y < obstacleY + obstacleHeight &&
      characterPosition.y + 20 > obstacleY) {
    // 触发碰撞事件，例如增加得分
    score += 10;
  }

  // ...之前的代码
}
```

在这个示例中，如果角色与障碍物发生碰撞，得分会增加10分。你需要根据游戏的具体需求来调整得分和碰撞检测逻辑。
