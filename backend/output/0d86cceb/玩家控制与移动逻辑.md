# 玩家控制与移动逻辑

```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>寻道大千 - 挂机类游戏</title>
<style>
  body {
    margin: 0;
    overflow: hidden;
  }
  canvas {
    display: block;
  }
</style>
</head>
<body>
<canvas id="gameCanvas"></canvas>
<script>
  const canvas = document.getElementById('gameCanvas');
  const ctx = canvas.getContext('2d');
  canvas.width = window.innerWidth;
  canvas.height = window.innerHeight;

  let player = {
    x: canvas.width / 2,
    y: canvas.height / 2,
    width: 50,
    height: 50,
    color: 'blue'
  };

  function drawRect(x, y, width, height, color) {
    ctx.fillStyle = color;
    ctx.fillRect(x, y, width, height);
  }

  function drawPlayer() {
    drawRect(player.x, player.y, player.width, player.height, player.color);
  }

  function drawMap() {
    drawRect(0, 0, canvas.width, canvas.height, 'green');
  }

  function gameLoop() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    drawMap();
    drawPlayer();
    requestAnimationFrame(gameLoop);
  }

  function handleKeyPress(event) {
    const { keyCode } = event;
    const step = 10;

    switch (keyCode) {
      case 37: // Left arrow
        player.x = Math.max(0, player.x - step);
        break;
      case 38: // Up arrow
        player.y = Math.max(0, player.y - step);
        break;
      case 39: // Right arrow
        player.x = Math.min(canvas.width - player.width, player.x + step);
        break;
      case 40: // Down arrow
        player.y = Math.min(canvas.height - player.height, player.y + step);
        break;
    }
  }

  window.onload = function() {
    document.addEventListener('keydown', handleKeyPress);
    gameLoop();
  };
</script>
</body>
</html>
```

这段代码是一个简单的挂机类游戏界面，使用了Canvas API进行基础渲染。游戏包含一个玩家角色和一个地图。玩家可以通过键盘的箭头键在画布上移动。我已经对原始代码进行了一些优化：

1. 移除了`map`对象中的`x`和`y`属性，因为地图始终覆盖整个画布。
2. 创建了`handleKeyPress`函数来处理键盘事件，并在`window.onload`事件中将其添加为事件监听器，以便在页面加载完成后立即启用。
3. 对移动逻辑进行了边界检查，以确保玩家不会移动出画布的边界。
