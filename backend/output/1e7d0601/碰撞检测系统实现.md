        // 游戏常量
        const GAME_STATES = {
            START: 'start',
            PLAYING: 'playing',
            PAUSED: 'paused',
            GAME_OVER: 'game_over'
        };

        // 碰撞检测类型
        const COLLISION_TYPES = {
            RECTANGLE: 'rectangle',
            CIRCLE: 'circle'
        };

        // 游戏配置
        const GAME_CONFIG = {
            PLAYER_SPEED: 5,
            ENEMY_SPEED: 2,
            BULLET_SPEED: 8,
            ENEMY_BULLET_SPEED: 4,
            PLAYER_FIRE_RATE: 10, // 帧数
            ENEMY_FIRE_RATE: 60, // 帧数
            MAX_ENEMIES: 5,
            MAX_BULLETS: 10,
            INITIAL_LIVES: 3,
            SCORE_PER_ENEMY: 100,
            BACKGROUND_SCROLL_SPEED: 2
        };

        // 游戏对象类
        class GameObject {
            constructor(x, y, width, height, type = COLLISION_TYPES.RECTANGLE) {
                this.x = x;
                this.y = y;
                this.width = width;
                this.height = height;
                this.type = type;
                this.active = true;
            }

            // 获取碰撞边界
            getBounds() {
                return {
                    left: this.x,
                    right: this.x + this.width,
                    top: this.y,
                    bottom: this.y + this.height
                };
            }

            // 检查是否在画布内
            isWithinCanvas(canvasWidth, canvasHeight) {
                return this.x >= -this.width && 
                       this.x <= canvasWidth && 
                       this.y >= -this.height && 
                       this.y <= canvasHeight;
            }

            // 更新对象位置
            update() {
                // 子类实现
            }

            // 渲染对象
            render(ctx) {
                // 子类实现
            }

            // 检查与其他对象的碰撞
            collidesWith(other) {
                if (!this.active || !other.active) return false;
                
                if (this.type === COLLISION_TYPES.RECTANGLE && other.type === COLLISION_TYPES.RECTANGLE) {
                    return this.rectRectCollision(other);
                } else if (this.type === COLLISION_TYPES.CIRCLE && other.type === COLLISION_TYPES.CIRCLE) {
                    return this.circleCircleCollision(other);
                } else {
                    // 矩形和圆形碰撞检测
                    const rect = this.type === COLLISION_TYPES.RECTANGLE ? this : other;
                    const circle = this.type === COLLISION_TYPES.CIRCLE ? this : other;
                    return this.rectCircleCollision(rect, circle);
                }
            }

            // 矩形与矩形碰撞检测
            rectRectCollision(other) {
                const bounds1 = this.getBounds();
                const bounds2 = other.getBounds();
                
                return bounds1.left < bounds2.right &&
                       bounds1.right > bounds2.left &&
                       bounds1.top < bounds2.bottom &&
                       bounds1.bottom > bounds2.top;
            }

            // 圆形与圆形碰撞检测
            circleCircleCollision(other) {
                const dx = this.x - other.x;
                const dy = this.y - other.y;
                const distance = Math.sqrt(dx * dx + dy * dy);
                
                return distance < (this.width / 2 + other.width / 2);
            }

            // 矩形与圆形碰撞检测
            rectCircleCollision(rect, circle) {
                // 找到矩形上距离圆心最近的点
                const closestX = Math.max(rect.x, Math.min(circle.x, rect.x + rect.width));
                const closestY = Math.max(rect.y, Math.min(circle.y, rect.y + rect.height));
                
                // 计算最近点到圆心的距离
                const dx = circle.x - closestX;
                const dy = circle.y - closestY;
                const distance = Math.sqrt(dx * dx + dy * dy);
                
                return distance < (circle.width / 2);
            }
        }

        // 玩家飞机类
        class Player extends GameObject {
            constructor(x, y) {
                super(x, y, 40, 40, COLLISION_TYPES.RECTANGLE);
                this.speed = GAME_CONFIG.PLAYER_SPEED;
                this.fireRate = GAME_CONFIG.PLAYER_FIRE_RATE;
                this.fireCounter = 0;
                this.canFire = true;
            }

            update(canvasWidth, canvasHeight) {
                // 边界检查
                this.x = Math.max(0, Math.min(canvasWidth - this.width, this.x));
                this.y = Math.max(0, Math.min(canvasHeight - this.height, this.y));
                
                // 更新射击冷却
                if (this.fireCounter > 0) {
                    this.fireCounter--;
                    if (this.fireCounter === 0) {
                        this.canFire = true;
                    }
                }
            }

            render(ctx) {
                ctx.fillStyle = '#3498db';
                ctx.beginPath();
                ctx.moveTo(this.x + this.width / 2, this.y);
                ctx.lineTo(this.x + this.width, this.y + this.height);
                ctx.lineTo(this.x, this.y + this.height);
                ctx.closePath();
                ctx.fill();
                
                // 绘制座舱
                ctx.fillStyle = '#2980b9';
                ctx.fillRect(this.x + this.width / 2 - 5, this.y + 10, 10, 15);
            }

            moveLeft() {
                this.x -= this.speed;
            }

            moveRight() {
                this.x += this.speed;
            }

            moveUp() {
                this.y -= this.speed;
            }

            moveDown() {
                this.y += this.speed;
            }

            fire() {
                if (this.canFire) {
                    this.canFire = false;
                    this.fireCounter = this.fireRate;
                    return new Bullet(
                        this.x + this.width / 2 - 2, 
                        this.y, 
                        -GAME_CONFIG.BULLET_SPEED,
                        '#3498db'
                    );
                }
                return null;
            }
        }

        // 敌机类
        class Enemy extends GameObject {
            constructor(x, y) {
                super(x, y, 35, 35, COLLISION_TYPES.RECTANGLE);
                this.speed = GAME_CONFIG.ENEMY_SPEED;
                this.fireRate = GAME_CONFIG.ENEMY_FIRE_RATE + Math.random() * 60;
                this.fireCounter = Math.random() * this.fireRate;
                this.movePattern = Math.floor(Math.random() * 3); // 0: 直线, 1: 波浪, 2: 之字形
                this.initialX = x;
                this.time = 0;
            }

            update(canvasWidth, canvasHeight) {
                this.time++;
                
                // 根据移动模式更新位置
                switch (this.movePattern) {
                    case 0: // 直线
                        this.y += this.speed;
                        break;
                    case 1: // 波浪
                        this.y += this.speed;
                        this.x = this.initialX + Math.sin(this.time * 0.05) * 50;
                        break;
                    case 2: // 之字形
                        this.y += this.speed;
                        if (this.time % 30 === 0) {
                            this.x += this.speed * 3 * (Math.random() > 0.5 ? 1 : -1);
                        }
                        break;
                }
                
                // 边界检查
                this.x = Math.max(0, Math.min(canvasWidth - this.width, this.x));
                
                // 更新射击冷却
                this.fireCounter--;
                if (this.fireCounter <= 0) {
                    this.fireCounter = this.fireRate + Math.random() * 60;
                    return this.fire();
                }
                
                return null;
            }

            render(ctx) {
                ctx.fillStyle = '#e74c3c';
                ctx.beginPath();
                ctx.moveTo(this.x + this.width / 2, this.y + this.height);
                ctx.lineTo(this.x + this.width, this.y);
                ctx.lineTo(this.x, this.y);
                ctx.closePath();
                ctx.fill();
                
                // 绘制细节
                ctx.fillStyle = '#c0392b';
                ctx.fillRect(this.x + this.width / 2 - 5, this.y + 5, 10, 15);
            }

            fire() {
                return new Bullet(
                    this.x + this.width / 2 - 2, 
                    this.y + this.height, 
                    GAME_CONFIG.ENEMY_BULLET_SPEED,
                    '#e74c3c'
                );
            }
        }

        // 子弹类
        class Bullet extends GameObject {
            constructor(x, y, speed, color) {
                super(x, y, 4, 12, COLLISION_TYPES.RECTANGLE);
                this.speed = speed;
                this.color = color;
            }

            update(canvasWidth, canvasHeight) {
                this.y += this.speed;
                
                // 检查是否超出画布
                if (!this.isWithinCanvas(canvasWidth, canvasHeight)) {
                    this.active = false;
                }
            }

            render(ctx) {
                ctx.fillStyle = this.color;
                ctx.fillRect(this.x, this.y, this.width, this.height);
                
                // 添加发光效果
                ctx.shadowColor = this.color;
                ctx.shadowBlur = 5;
                ctx.fillRect(this.x, this.y, this.width, this.height);
                ctx.shadowBlur = 0;
            }
        }

        // 背景类
        class Background {
            constructor(canvasWidth, canvasHeight) {
                this.canvasWidth = canvasWidth;
                this.canvasHeight = canvasHeight;
                this.height = canvasHeight * 2;
                this.backgroundY = 0;
                this.stars = [];
                
                // 生成星星
                for (let i = 0; i < 100; i++) {
                    this.stars.push({
                        x: Math.random() * canvasWidth,
                        y: Math.random() * this.height,
                        size: Math.random() * 2 + 1,
                        speed: Math.random() * 2 + 1
                    });
                }
            }

            update() {
                // 更新星星位置
                for (let star of this.stars) {
                    star.y += star.speed;
                    if (star.y > this.height) {
                        star.y = 0;
                        star.x = Math.random() * this.canvasWidth;
                    }
                }
                
                // 更新背景滚动位置
                this.backgroundY += GAME_CONFIG.BACKGROUND_SCROLL_SPEED;
                if (this.backgroundY >= this.canvasHeight) {
                    this.backgroundY = 0;
                }
            }

            render(ctx) {
                // 绘制星空背景
                ctx.fillStyle = '#000';
                ctx.fillRect(0, 0, this.canvasWidth, this.canvasHeight);
                
                // 绘制星星
                ctx.fillStyle = '#fff';
                for (let star of this.stars) {
                    ctx.beginPath();
                    ctx.arc(star.x, star.y, star.size, 0, Math.PI * 2);
                    ctx.fill();
                }
                
                // 绘制云层（简化版）
                ctx.fillStyle = 'rgba(255, 255, 255, 0.1)';
                for (let i = 0; i < 5; i++) {
                    const cloudY = (i * 150 + this.backgroundY) % this.canvasHeight;
                    ctx.beginPath();
                    ctx.ellipse(100, cloudY, 60, 20, 0, 0, Math.PI * 2);
                    ctx.ellipse(160, cloudY - 10, 50, 20, 0, 0, Math.PI * 2);
                    ctx.ellipse(220, cloudY, 70, 20, 0, 0, Math.PI * 2);
                    ctx.fill();
                }
            }
        }

            resizeCanvas() {
                const container = document.getElementById('gameContainer');
                const containerWidth = container.clientWidth;
                const containerHeight = container.clientHeight;
                
                // 保持宽高比
