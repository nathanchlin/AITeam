        this.currentLevel = levelNumber;

        this.enemies = [];
        this.bullets = [];
        this.powerUps = [];
        
        // 设置背景

        for (let i = 0; i < waveConfig.count; i++) {
            setTimeout(() => {
                this.spawnEnemy(waveConfig.type, waveConfig.speed);
            }, i * waveConfig.delay);
        }
    }
    
    spawnEnemy(type, speed) {
        let enemy;
        const x = Math.random() * (this.width - 50) + 25;
        const y = -50;
        
        switch (type) {
            case 'basic':
                enemy = new BasicEnemy(x, y, speed);
                break;
            case 'fast':
                enemy = new FastEnemy(x, y, speed);
                break;
            case 'zigzag':
                enemy = new ZigzagEnemy(x, y, speed);
                break;
            case 'formation':
                enemy = new FormationEnemy(x, y, speed);
                break;
            case 'dive':
                enemy = new DiveEnemy(x, y, speed);
                break;
            case 'mixed':
                const enemyTypes = ['basic', 'fast', 'zigzag'];
                const randomType = enemyTypes[Math.floor(Math.random() * enemyTypes.length)];
                this.spawnEnemy(randomType, speed);
                return; // 不添加到enemies数组，因为已经递归调用了
            default:
                enemy = new BasicEnemy(x, y, speed);
        }
        
        this.enemies.push(enemy);
    }

    update() {
        if (this.state !== GameState.PLAYING) return;
        
        // 更新FPS
        this.updateFPS();
        
        // 更新背景
        this.background.update();
        
        // 更新玩家
        if (this.player) {
            this.player.update(this.keys, this.width, this.height);
        }
        
        // 更新敌人
        this.enemies = this.enemies.filter(enemy => {
            enemy.update(this.width, this.height);
            
            // 检查是否超出屏幕
            if (enemy.y > this.height + 50) {
                return false;
            }
            
            // 检查是否与玩家碰撞
            if (this.player && this.checkCollision(enemy, this.player)) {
                this.playerHit();
                return false;
            }
            
            // 敌人射击
            if (Math.random() < enemy.shootProbability) {
                this.bullets.push(new EnemyBullet(enemy.x, enemy.y));
            }
            
            return true;
        });
        
        // 更新子弹
        this.bullets = this.bullets.filter(bullet => {
            bullet.update();
            
            // 检查是否超出屏幕
            if (bullet.y < -10 || bullet.y > this.height + 10 || 
                bullet.x < -10 || bullet.x > this.width + 10) {
                return false;
            }
            
            // 检查子弹碰撞
            if (bullet.isPlayerBullet) {
                // 玩家子弹与敌人碰撞
                for (let i = 0; i < this.enemies.length; i++) {
                    if (this.checkCollision(bullet, this.enemies[i])) {
                        this.enemies[i].hit(bullet.damage);
                        if (this.enemies[i].health <= 0) {
                            this.score += this.enemies[i].points;
                            this.createExplosion(this.enemies[i].x, this.enemies[i].y);
                            
                            // 掉落道具
                            if (Math.random() < 0.1) {
                                this.spawnPowerUp(this.enemies[i].x, this.enemies[i].y);
                            }
                            
                            this.enemies.splice(i, 1);
                        }
                        return false;
                    }
                }
            } else {
                // 敌人子弹与玩家碰撞
                if (this.player && this.checkCollision(bullet, this.player)) {
                    this.playerHit();
                    return false;
                }
            }
            
            return true;
        });
        
        // 更新道具
        this.powerUps = this.powerUps.filter(powerUp => {
            powerUp.update();
            
            // 检查是否超出屏幕
            if (powerUp.y > this.height + 20) {
                return false;
            }
            
            // 检查与玩家碰撞
            if (this.player && this.checkCollision(powerUp, this.player)) {
                this.player.applyPowerUp(powerUp.type);
                return false;
            }
            
            return true;
        });
        
        // 更新粒子效果
        this.particles = this.particles.filter(particle => {
            particle.update();
            return particle.life > 0;
        });
        
        // 检查Boss生成
        if (!this.bossSpawned && this.enemies.length === 0 && this.currentWave >= this.levels[this.currentLevel - 1].enemyWaves.length) {
            setTimeout(() => {
                this.spawnBoss();
            }, 2000);
        }
        
        // 检查关卡完成
        if (this.bossSpawned && this.enemies.length === 0 && this.boss.health <= 0) {
            this.levelComplete();
        }
        
        // 检查游戏结束
        if (this
