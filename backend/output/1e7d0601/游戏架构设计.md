# 1942游戏架构设计

## 1. 总体架构概述

为1942风格游戏设计一个模块化、可扩展的架构，采用组件化设计模式，便于后续功能扩展和维护。

### 1.1 架构原则
- **单一职责原则**：每个模块负责特定功能
- **依赖注入**：模块间通过接口通信，降低耦合
- **事件驱动**：使用事件系统处理模块间通信
- **数据驱动**：游戏配置和数据外部化

## 3. 游戏主循环设计

```csharp
public class GameLoop {
    private enum GameLoopState {
        INIT,
        UPDATE,
        FIXED_UPDATE,
        RENDER,
        SHUTDOWN
    }
    
    private GameLoopState currentState = GameLoopState.INIT;
    private float accumulatedTime = 0f;
    
    public void Run() {
        while (currentState != GameLoopState.SHUTDOWN) {
            switch (currentState) {
                case GameLoopState.INIT:
                    Initialize();
                    currentState = GameLoopState.UPDATE;
                    break;
                    
                case GameLoopState.UPDATE:
                    Update();
                    break;
                    
                case GameLoopState.FIXED_UPDATE:
                    FixedUpdate();
                    break;
                    
                case GameLoopState.RENDER:
                    Render();
                    break;
            }
        }
    }
    
    private void Initialize() {
        // 初始化所有系统
        EventSystem.Instance.Initialize();
        SceneManager.Instance.Initialize();
        InputManager.Instance.Initialize();
        // ...其他系统初始化
    }
    
    private void Update() {
        float deltaTime = TimeManager.Instance.GetDeltaTime();
        accumulatedTime += deltaTime;
        
        // 处理输入
        InputManager.Instance.Update();
        
        // 更新游戏逻辑
        GameStateManager.Instance.Update(deltaTime);
        PlayerController.Instance.Update(deltaTime);
        EnemyAI.Instance.Update(deltaTime);
        BulletSystem.Instance.Update(deltaTime);
        
        // 处理事件
        EventSystem.Instance.ProcessEvents();
        
        // 如果达到固定时间间隔，执行物理更新
        if (accumulatedTime >= TimeManager.Instance.fixedDeltaTime) {
            currentState = GameLoopState.FIXED_UPDATE;
        }
    }
    
    private void FixedUpdate() {
        // 固定时间步长的物理更新
        float fixedDeltaTime = TimeManager.Instance.fixedDeltaTime;
        
        CollisionSystem.Instance.Update(fixedDeltaTime);
        PowerUpSystem.Instance.Update(fixedDeltaTime);
        
        accumulatedTime -= TimeManager.Instance.fixedDeltaTime;
        if (accumulatedTime < TimeManager.Instance.fixedDeltaTime) {
            currentState = GameLoopState.RENDER;
        }
    }
    
    private void Render() {
        // 渲染更新
        RenderSystem.Instance.Update();
        AnimationSystem.Instance.Update();
        EffectSystem.Instance.Update();
        UIManager.Instance.Update();
        
        currentState = GameLoopState.UPDATE;
    }
}
```

```csharp
public class GameStateManager {
    private GameState currentState = GameState.MENU;
    private Dictionary<GameState, IGameStateHandler> stateHandlers;
    
    public GameStateManager() {
        stateHandlers = new Dictionary<GameState, IGameStateHandler>();
        RegisterStateHandlers();
    }
    
    private void RegisterStateHandlers() {
        stateHandlers[GameState.MENU] = new MenuStateHandler();
        stateHandlers[GameState.PLAYING] = new PlayingStateHandler();
        stateHandlers[GameState.PAUSED] = new PausedStateHandler();
        stateHandlers[GameState.GAME_OVER] = new GameOverStateHandler();
        stateHandlers[GameState.SETTINGS] = new SettingsStateHandler();
        stateHandlers[GameState.CREDITS] = new CreditsStateHandler();
    }
    
    public void ChangeState(GameState newState) {
        if (currentState == newState) return;
        
        stateHandlers[currentState].Exit();
        currentState = newState;
        stateHandlers[currentState].Enter();
    }
    
    public void Update(float deltaTime) {
        stateHandlers[currentState].Update(deltaTime);
    }
    
    // 其他状态管理方法...
}

public interface IGameStateHandler {
    void Enter();
    void Exit();
    void Update(float deltaTime);
}
```

```csharp
public abstract class GameEntity {
    public string Id { get; protected set; }
    public Vector2 Position { get; set; }
    public Vector2 Velocity { get; set; }
    public float Rotation { get; set; }
    public bool IsActive { get; set; }
    
    private List<IComponent> components = new List<IComponent>();
    
    public void AddComponent(IComponent component) {
        components.Add(component);
        component.Initialize(this);
    }
    
    public T GetComponent<T>() where T : IComponent {
        return components.OfType<T>().FirstOrDefault();
    }
    
    public virtual void Update(float deltaTime) {
        foreach (var component in components) {
            component.Update(deltaTime);
        }
    }
}
```

```csharp
public interface IGamePlugin {
    string Name { get; }
    string Version { get; }
    void Initialize(IGameContext context);
    void Update(float deltaTime);
    void Shutdown();
}

public class PluginManager {
    private List<IGamePlugin> plugins = new List<IGamePlugin>();
    
    public void LoadPlugin(string pluginPath) {
        var assembly = Assembly.LoadFrom(pluginPath);
        var pluginType = assembly.GetTypes().FirstOrDefault(t => 
            typeof(IGamePlugin).IsAssignableFrom(t) && !t.IsAbstract);
            
        if (pluginType != null) {
            var plugin = (IGamePlugin)Activator.CreateInstance(pluginType);
            plugins.Add(plugin);
            plugin.Initialize(GameContext.Instance);
        }
    }
}
```

### 8.2 脚本系统
```csharp
public class ScriptingSystem {
    private Dictionary<string, GameObject> gameObjects = new Dictionary<string, GameObject>();
    
    public void AttachScript(string objectId, Type scriptType) {
        if (gameObjects.TryGetValue(objectId, out var gameObject)) {
            var script = (MonoBehaviour)Activator.CreateInstance(scriptType);
            script.gameObject = gameObject;
            gameObject.AddComponent(script);
        }
    }
    
    public void ExecuteScriptMethod(string objectId, string methodName, params object[] parameters) {
        if (gameObjects.TryGetValue(objectId, out var gameObject)) {
            var scripts = gameObject.GetComponents<MonoBehaviour>();
            foreach (var script in scripts) {
                var method = script.GetType().GetMethod(methodName);
                if (method != null) {
                    method.Invoke(script, parameters);
                }
            }
        }
    }
}
```
