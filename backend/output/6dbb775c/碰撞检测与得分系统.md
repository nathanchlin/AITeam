```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>黑白棋游戏</title>
<style>
  body {
    margin: 0;
    display: flex;
    justify-content: center;
    align-items: center;
    height: 100vh;
    background-color: #f0f0f0;
  }
  canvas {
    border: 2px solid #000;
  }
</style>
</head>
<body>
<canvas id="gameCanvas" width="400" height="400"></canvas>
<script>
  const canvas = document.getElementById('gameCanvas');
  const ctx = canvas.getContext('2d');
  const boardSize = 8;
  const cellSize = canvas.width / boardSize;
  let board = [];
  let currentPlayer = 'black';
  let scoreBlack = 0;
  let scoreWhite = 0;

  function initializeBoard() {
    for (let i = 0; i < boardSize; i++) {
      board[i] = [];
      for (let j = 0; j < boardSize; j++) {
        board[i][j] = null;
      }
    }
  }

  function drawBoard() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    ctx.fillStyle = 'black';
    ctx.fillRect(0, 0, canvas.width, canvas.height);
    for (let i = 0; i < boardSize; i++) {
      for (let j = 0; j < boardSize; j++) {
        ctx.beginPath();
        ctx.moveTo(j * cellSize, i * cellSize);
        ctx.lineTo((j + 1) * cellSize, i * cellSize);
        ctx.lineTo((j + 1) * cellSize, (i + 1) * cellSize);
        ctx.lineTo(j * cellSize, (i + 1) * cellSize);
        ctx.lineTo(j * cellSize, i * cellSize);
        ctx.stroke();
        if (board[i][j]) {
          drawPiece(i, j, board[i][j]);
        }
      }
    }
    drawScore();
  }

  function drawPiece(x, y, color) {
    const offset = 10;
    const pieceSize = cellSize - 2 * offset;
    const startX = x * cellSize + offset;
    const startY = y * cellSize + offset;
    ctx.fillStyle = color;
    ctx.beginPath();
    ctx.arc(startX + pieceSize / 2, startY + pieceSize / 2, pieceSize / 2 - 2, 0, 2 * Math.PI);
    ctx.fill();
  }

  function drawScore() {
    ctx.fillStyle = 'black';
    ctx.font = '20px Arial';
    ctx.fillText(`Score - Black: ${scoreBlack}`, 10, 30);
    ctx.fillText(`Score - White: ${scoreWhite}`, 10, 60);
  }

  function onCanvasClick(event) {
    const x = Math.floor((event.offsetX - 10) / cellSize);
    const y = Math.floor((event.offsetY - 10) / cellSize);
    if (board[x][y] === null) {
      board[x][y] = currentPlayer;
      currentPlayer = currentPlayer === 'black' ? 'white' : 'black';
      updateScore(x, y);
      drawBoard();
    }
  }

  function updateScore(x, y) {
    const currentPlayerScore = currentPlayer === 'black' ? scoreBlack : scoreWhite;
    const opponentPlayerScore = currentPlayer === 'black' ? scoreWhite : scoreBlack;
    let surroundedCount = 0;

    // Horizontal
    for (let i = -1; i <= 1; i++) {
      if (x + i >= 0 && x + i < boardSize && y >= 0 && y < boardSize) {
        if (board[x + i][y] !== currentPlayer) {
          surroundedCount++;
        }
      }
    }

    // Vertical
    for (let j = -1; j <= 1; j++) {
      if (x >= 0 && x < boardSize && y + j >= 0 && y + j < boardSize) {
        if (board[x][y + j] !== currentPlayer) {
          surroundedCount++;
        }
      }
    }

    if (surroundedCount === 4) {
      currentPlayerScore += 10; // Example score for a complete surrounded piece
      if (currentPlayer === 'black') {
        scoreBlack = currentPlayerScore;
      } else {
        scoreWhite = currentPlayerScore;
      }
    }
  }

  function startGameLoop() {
    requestAnimationFrame(startGameLoop);
    drawBoard();
  }

  window.onload = function() {
    canvas.addEventListener('click', onCanvasClick);
    initializeBoard();
    drawBoard();
    startGameLoop();
  };
</script>
</body>
</html>
```

在这个代码中，我添加了以下功能：
1. `drawScore` 函数，用于在屏幕上绘制当前得分。
2. `updateScore` 函数，用于更新得分。每当有棋子被放置时，它会检查这个棋子是否被对方完全包围，如果是，则根据规则增加分数。
3. 修改了 `onCanvasClick` 函数，使其在点击后调用 `updateScore` 函数。

这个得分系统是一个简单的示例，它根据一个棋子是否被四个方向上的对方棋子完全包围来增加分数。在真实的游戏中，你可能需要一个更复杂的算法来处理各种不同的棋局情况。
