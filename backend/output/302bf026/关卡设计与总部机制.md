        // 坦克类
        class Tank {
            constructor(x, y, color, isPlayer = false) {
                this.x = x;
                this.y = y;
                this.width = 30;
                this.height = 30;
                this.color = color;
                this.speed = isPlayer ? PLAYER_SPEED : ENEMY_SPEED;
                this.direction = 'up';
                this.isPlayer = isPlayer;
                this.health = 1;
                this.canFire = true;
                this.fireRate = isPlayer ? 500 : 2000;
                this.lastFire = 0;
                this.hasShield = false;
                this.shieldTime = 0;
            }
            
            draw() {
                ctx.save();
                ctx.translate(this.x + this.width/2, this.y + this.height/2);
                
                // 绘制坦克主体
                ctx.fillStyle = this.color;
                ctx.fillRect(-this.width/2, -this.height/2, this.width, this.height);
                
                // 绘制坦克炮管
                ctx.fillStyle = this.color;
                switch(this.direction) {
                    case 'up':
                        ctx.fillRect(-3, -this.height/2 - 10, 6, 15);
                        break;
                    case 'down':
                        ctx.fillRect(-3, this.height/2 - 5, 6, 15);
                        break;
                    case 'left':
                        ctx.fillRect(-this.width/2 - 10, -3, 15, 6);
                        break;
                    case 'right':
                        ctx.fillRect(this.width/2 - 5, -3, 15, 6);
                        break;
                }
                
                // 绘制护盾
                if (this.hasShield) {
                    ctx.strokeStyle = 'rgba(0, 255, 255, 0.7)';
                    ctx.lineWidth = 2;
                    ctx.beginPath();
                    ctx.arc(0, 0, this.width, 0, Math.PI * 2);
                    ctx.stroke();
                }
                
                ctx.restore();
            }
            
            move(direction) {
                this.direction = direction;
                let newX = this.x;
                let newY = this.y;
                
                switch(direction) {
                    case 'up':
                        newY -= this.speed;
                        break;
                    case 'down':
                        newY += this.speed;
                        break;
                    case 'left':
                        newX -= this.speed;
                        break;
                    case 'right':
                        newX += this.speed;
                        break;
                }
                
                // 边界检查
                if (newX >= 0 && newX <= canvas.width - this.width &&
                    newY >= 0 && newY <= canvas.height - this.height) {
                    
                    // 障碍物碰撞检测
                    let canMove = true;
                    for (let obstacle of obstacles) {
                        if (this.checkCollision({x: newX, y: newY, width: this.width, height: this.height}, obstacle)) {
                            canMove = false;
                            break;
                        }
                    }
                    
                    if (canMove) {
                        this.x = newX;
                        this.y = newY;
                    }
                }
            }
            
            fire() {
                if (!this.canFire) return;
                
                let now = Date.now();
                if (now - this.lastFire < this.fireRate) return;
                
                this.lastFire = now;
                this.canFire = false;
                
                let bulletX = this.x + this.width/2;
                let bulletY = this.y + this.height/2;
                
                let bullet = new Bullet(
                    bulletX - 2, 
                    bulletY - 2, 
                    this.direction, 
                    this.isPlayer,
                    this.isPlayer ? BULLET_SPEED : ENEMY_BULLET_SPEED
                );
                
                if (this.isPlayer) {
                    bullets.push(bullet);
                } else {
                    enemyBullets.push(bullet);
                }
                
                setTimeout(() => {
                    this.canFire = true;
                }, this.fireRate / 2);
            }
            
            checkCollision(rect1, rect2) {
                return rect1.x < rect2.x + rect2.width &&
                       rect1.x + rect1.width > rect2.x &&
                       rect1.y < rect2.y + rect2.height &&
                       rect1.y + rect1.height > rect2.y;

            checkOccupation(tank) {
                let tankRect = {
                    x: tank.x,
                    y: tank.y,
                    width: tank.width,
                    height: tank.height
                };
                
                if (this.checkCollision(tankRect, this)) {
                    if (!this.occupied) {
                        this.occupied = true;
                        this.occupier = tank;
                        return true;
                    }
                } else {
                    if (this.occupied && this.occupier === tank) {
                        this.occupied = false;
                        this.occupier = null;
                    }
                }
                return false;
            }
