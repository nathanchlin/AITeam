        // 游戏配置
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        const gameInfo = {
            playerLives: document.getElementById('playerLives'),
            score: document.getElementById('score'),
            gameOver: document.getElementById('gameOver'),
            gameOverText: document.getElementById('gameOverText'),
            finalScore: document.getElementById('finalScore'),
            restartBtn: document.getElementById('restartBtn')

        // 游戏常量
        const TILE_SIZE = 40;
        const TANK_SPEED = 2;
        const BULLET_SPEED = 5;
        const ENEMY_BULLET_SPEED = 3;
        const ENEMY_SPAWN_DELAY = 2000; // 敌人生成间隔(毫秒)
        const ENEMY_SHOOT_PROBABILITY = 0.01; // 敌人射击概率

        // 游戏状态
        let gameRunning = true;
        let score = 0;
        let lastEnemySpawn = 0;
        let keys = {};

        // 游戏对象数组
        let player = null;
        let enemies = [];
        let playerBullets = [];
        let enemyBullets = [];
        let obstacles = [];
        let headquarters = null;

        // 坦克类
        class Tank {
            constructor(x, y, color, isPlayer = false) {
                this.x = x;
                this.y = y;
                this.width = TILE_SIZE - 10;
                this.height = TILE_SIZE - 10;
                this.color = color;
                this.direction = 'up'; // up, down, left, right
                this.isPlayer = isPlayer;
                this.speed = isPlayer ? TANK_SPEED : TANK_SPEED * 0.7;
                this.lastShot = 0;
                this.shootCooldown = isPlayer ? 500 : 1000; // 射击冷却时间(毫秒)
                this.lives = isPlayer ? 3 : 1;
            }

            draw() {
                ctx.save();
                ctx.fillStyle = this.color;
                
                // 根据方向旋转坦克
                ctx.translate(this.x + this.width/2, this.y + this.height/2);
                
                switch(this.direction) {
                    case 'up':
                        ctx.rotate(0);
                        break;
                    case 'right':
                        ctx.rotate(Math.PI/2);
                        break;
                    case 'down':
                        ctx.rotate(Math.PI);
                        break;
                    case 'left':
                        ctx.rotate(-Math.PI/2);
                        break;
                }
                
                // 绘制坦克主体
                ctx.fillRect(-this.width/2, -this.height/2, this.width, this.height);
                
                // 绘制坦克炮管
                ctx.fillStyle = this.isPlayer ? '#2ecc71' : '#e74c3c';
                ctx.fillRect(-5, -this.height/2 - 10, 10, 15);
                
                ctx.restore();
            }

            move(direction) {
                this.direction = direction;
                let newX = this.x;
                let newY = this.y;
                
                switch(direction) {
                    case 'up':
                        newY -= this.speed;
                        break;
                    case 'down':
                        newY += this.speed;
                        break;
                    case 'left':
                        newX -= this.speed;
                        break;
                    case 'right':
                        newX += this.speed;
                        break;
                }
                
                // 边界检查
                if (newX >= 0 && newX <= canvas.width - this.width &&
                    newY >= 0 && newY <= canvas.height - this.height) {
                    
                    // 碰撞检测 - 检查与障碍物的碰撞
                    let collision = false;
                    for (let obstacle of obstacles) {
                        if (this.checkCollision(newX, newY, this.width, this.height, obstacle)) {
                            collision = true;
                            break;
                        }
                    }
                    
                    // 检查与总部的碰撞
                    if (headquarters && this.checkCollision(newX, newY, this.width, this.height, headquarters)) {
                        collision = true;
                    }
                    
                    // 检查与其他坦克的碰撞
                    for (let enemy of enemies) {
                        if (enemy !== this && this.checkCollision(newX, newY, this.width, this.height, enemy)) {
                            collision = true;
                            break;
                        }
                    }
                    
                    // 检查与玩家坦克的碰撞
                    if (!this.isPlayer && this.checkCollision(newX, newY, this.width, this.height, player)) {
                        collision = true;
                    }
                    
                    if (!collision) {
                        this.x = newX;
                        this.y = newY;
                    }
                }
            }

            shoot() {
                const now = Date.now();
                if (now - this.lastShot > this.shootCooldown) {
                    let bulletX = this.x + this.width/2;
                    let bulletY = this.y + this.height/2;
                    
                    // 根据坦克方向调整子弹初始位置
                    switch(this.direction) {
                        case 'up':
                            bulletY -= this.height/2;
                            break;
                        case 'down':
                            bulletY += this.height/2;
                            break;
                        case 'left':
                            bulletX -= this.width/2;
                            break;
                        case 'right':
                            bulletX += this.width/2;
                            break;
                    }
                    
                    const bullet = new Bullet(
                        bulletX, 
                        bulletY, 
                        this.direction, 
                        this.isPlayer ? BULLET_SPEED : ENEMY_BULLET_SPEED,
                        this.isPlayer
                    );
                    
                    if (this.isPlayer) {
                        playerBullets.push(bullet);
                    } else {
                        enemyBullets.push(bullet);
                    }
                    
                    this.lastShot = now;
                }
            }

            checkCollision(x, y, width, height, obj) {
                return x < obj.x + obj.width &&
                       x + width > obj.x &&
                       y < obj.y + obj.height &&
                       y + height > obj.y;
            }

            takeDamage() {
                this.lives--;
                if (this.lives <= 0) {
                    return true; // 坦克被摧毁
                }
                return false; // 坦克受损但未摧毁
            }
        }

        // 子弹类
        class Bullet {
            constructor(x, y, direction, speed, isPlayerBullet) {
                this.x = x;
                this.y = y;
                this.width = 6;
                this.height = 10;
                this.direction = direction;
                this.speed = speed;
                this.isPlayerBullet = isPlayerBullet;
            }

            draw() {
                ctx.fillStyle = this.isPlayerBullet ? '#f1c40f' : '#e74c3c';
                ctx.fillRect(this.x - this.width/2, this.y - this.height/2, this.width, this.height);
            }

            update() {
                switch(this.direction) {
                    case 'up':
                        this.y -= this.speed;
                        break;
                    case 'down':
                        this.y += this.speed;
                        break;
                    case 'left':
                        this.x -= this.speed;
                        break;
                    case 'right':
                        this.x += this.speed;
                        break;
                }
                
                // 检查子弹是否超出边界
                return this.x < 0 || this.x > canvas.width || 
                       this.y < 0 || this.y > canvas.height;
            }

            checkCollision(obj) {
                return this.x < obj.x + obj.width &&
                       this.x + this.width > obj.x &&
                       this.y < obj.y + obj.height &&
                       this.y + this.height > obj.y;
            }
        }

        // 障碍物类
        class Obstacle {
            constructor(x, y, type = 'wall') {
                this.x = x;
                this.y = y;
                this.width = TILE_SIZE;
                this.height = TILE_SIZE;
                this.type = type;
            }

            draw() {
                ctx.fillStyle = '#7f8c8d';
                ctx.fillRect(this.x, this.y, this.width, this.height);
                
                // 添加纹理
                ctx.strokeStyle = '#95a5a6';
                ctx.lineWidth = 2;
                ctx.strokeRect(this.x, this.y, this.width, this.height);
                
                if (this.type === 'brick') {
                    // 绘制砖块纹理
                    ctx.beginPath();
                    ctx.moveTo(this.x + this.width/2, this.y);
                    ctx.lineTo(this.x + this.width/2, this.y + this.height);
                    ctx.stroke();
                    
                    ctx.beginPath();
                    ctx.moveTo(this.x, this.y + this.height/2);
                    ctx.lineTo(this.x + this.width, this.y + this.height/2);
                    ctx.stroke();
                }
            }
        }

        // 总部类
        class Headquarters {
            constructor(x, y) {
                this.x = x;
                this.y = y;
                this.width = TILE_SIZE * 2;
                this.height = TILE_SIZE * 2;
                this.health = 3;
            }

            draw() {
                // 绘制总部建筑
                ctx.fillStyle = '#3498db';
                ctx.fillRect(this.x, this.y, this.width, this.height);
                
                // 绘制旗帜
                ctx.fillStyle = '#e74c3c';
                ctx.fillRect(this.x + this.width/2 - 5, this.y - 20, 10, 20);
                
                // 绘制生命值指示器
                ctx.fillStyle = '#2ecc71';
                for (let i = 0; i < this.health; i++) {
                    ctx.fillRect(this.x + 5 + i * 15, this.y + 5, 10, 10);
                }
            }

            takeDamage() {
                this.health--;
                if (this.health <= 0) {
                    return true; // 总部被摧毁
                }
                return false; // 总部受损但未摧毁
            }
        }

        // 初始化游戏
        function initGame() {
            // 重置游戏状态
            gameRunning = true;
            score = 0;
            lastEnemySpawn = 0;
            keys = {};
            
            // 清空游戏对象数组
            enemies = [];
            playerBullets = [];
            enemyBullets = [];
            obstacles = [];
            
            // 创建玩家坦克
            player = new Tank(100, canvas.height / 2, '#2ecc71', true);
            
            // 创建障碍物
            createObstacles();
            
            // 创建总部
            headquarters = new Headquarters(canvas.width - 100, canvas.height / 2 - 40);
            
            // 更新UI
            gameInfo.playerLives.textContent = player.lives;
            gameInfo.score.textContent = score;
            gameInfo.gameOver.style.display = 'none';
        }

        // 创建障碍物
        function createObstacles() {
            // 创建一些随机障碍物
            for (let i = 0; i < 15; i++) {
                let x, y;
                let validPosition = false;
                
                // 确保障碍物不会生成在玩家或总部附近
                while (!validPosition) {
                    x = Math.floor(Math.random() * (canvas.width / TILE_SIZE)) * TILE_SIZE;
                    y = Math.floor(Math.random() * (canvas.height / TILE_SIZE)) * TILE_SIZE;
                    
                    // 检查位置是否有效
                    validPosition = true;
                    
                    // 避免在玩家初始位置附近生成障碍物
                    if (x < 200 && y > canvas.height/2 - 100 && y < canvas.height/2 + 100) {
                        validPosition = false;
                    }
                    
                    // 避免在总部附近生成障碍物
                    if (x > canvas.width - 200 && y > canvas.height/2 - 100 && y < canvas.height/2 + 100) {
                        validPosition = false;
                    }
                }
                
                const type = Math.random() > 0.7 ? 'brick' : 'wall';
                obstacles.push(new Obstacle(x, y, type));
            }
        }

        // 生成敌人
        function spawnEnemy() {
            const now = Date.now();
            if (now - lastEnemySpawn > ENEMY_SPAWN_DELAY && enemies.length < 5) {
                // 在左侧随机位置生成敌人
                const y = Math.floor(Math.random() * (canvas.height / TILE_SIZE)) * TILE_SIZE;
                const enemy = new Tank(0, y, '#e74c3c', false);
                
                // 随机初始方向
                const directions = ['up', 'down', 'left', 'right'];
                enemy.direction = directions[Math.floor(Math.random() * directions.length)];
                
                enemies.push(enemy);
                lastEnemySpawn = now;
            }
        }

        // 更新敌人AI
        function updateEnemyAI() {
            for (let enemy of enemies) {
                // 随机移动
                if (Math.random() < 0.02) {
                    const directions = ['up', 'down', 'left', 'right'];
                    enemy.direction = directions[Math.floor(Math.random() * directions.length)];
                }
                
                // 移动敌人
                enemy.move(enemy.direction);
                
                // 随机射击
                if (Math.random() < ENEMY_SHOOT_PROBABILITY) {
                    enemy.shoot();
                }
            }
        }

        // 碰撞检测 - 子弹与坦克
        function checkBulletTankCollisions() {
            // 玩家子弹与敌人碰撞
            for (let i = playerBullets.length - 1; i >= 0; i--) {
                const bullet = playerBullets[i];
                
                for (let j = enemies.length - 1; j >= 0; j--) {
                    const enemy = enemies[j];
                    
                    if (bullet.checkCollision(enemy)) {
                        // 移除子弹
                        playerBullets.splice(i, 1);
                        
                        // 敌人受到伤害
                        if (enemy.takeDamage()) {
                            // 敌人被摧毁
                            enemies.splice(j, 1);
                            score += 100;
                            gameInfo.score.textContent = score;
                        }
                        
                        break;
                    }
                }
            }
            
            // 敌人子弹与玩家碰撞
            for (let i = enemyBullets.length - 1; i >= 0; i--) {
                const bullet = enemyBullets[i];
                
                if (bullet.checkCollision(player)) {
                    // 移除子弹
                    enemyBullets.splice(i, 1);
                    
                    // 玩家受到伤害
                    if (player.takeDamage()) {
                        // 玩家被摧毁
                        gameOver(false);
                    } else {
                        gameInfo.playerLives.textContent = player.lives;
                    }
                }
            }
        }

        // 碰撞检测 - 子弹与障碍物
        function checkBulletObstacleCollisions() {
            // 玩家子弹与障碍物碰撞
            for (let i = playerBullets.length - 1; i >= 0; i--) {
                const bullet = playerBullets[i];
                
                for (let j = obstacles.length - 1; j >= 0; j--) {
                    const obstacle = obstacles[j];
                    
                    if (bullet.checkCollision(obstacle)) {
                        // 移除子弹
                        playerBullets.splice(i, 1);
                        
                        // 移除障碍物
                        obstacles.splice(j, 1);
                        score += 10;
                        gameInfo.score.textContent = score;
                        
                        break;
                    }
                }
            }
            
            // 敌人子弹与障碍物碰撞
            for (let i = enemyBullets.length - 1; i >= 0; i--) {
                const bullet = enemyBullets[i];
                
                for (let j = obstacles.length - 1; j >= 0; j--) {
                    const obstacle = obstacles[j];
                    
                    if (bullet.checkCollision(obstacle)) {
                        // 移除子弹
                        enemyBullets.splice(i, 1);
                        
                        // 移除障碍物
                        obstacles.splice(j, 1);
                        
                        break;
                    }
                }
            }
        }

        // 碰撞检测 - 子弹与总部
        function check
