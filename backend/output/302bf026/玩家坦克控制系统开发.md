    <script>
        // 游戏常量
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        const GAME_WIDTH = canvas.width;
        const GAME_HEIGHT = canvas.height;
        
        // 游戏状态
        const game = {
            keys: {},
            lastTime: 0,
            player: null,
            bullets: [],
            enemies: [],
            headquarters: [],
            particles: [],
            score: 0,
            gameOver: false
        };
        
        // 坦克类
        class Tank {
            constructor(x, y, color, isPlayer = false) {
                this.x = x;
                this.y = y;
                this.width = 30;
                this.height = 30;
                this.color = color;
                this.angle = 0;
                this.speed = isPlayer ? 2 : 1.5;
                this.rotationSpeed = 0.05;
                this.isPlayer = isPlayer;
                this.health = 100;
                this.maxHealth = 100;
                this.lastShot = 0;
                this.shootCooldown = isPlayer ? 500 : 1000;
                this.turretAngle = 0;
                this.moving = false;
                this.animationFrame = 0;
            }
            
            update(deltaTime) {
                // 更新动画帧
                if (this.moving) {
                    this.animationFrame = (this.animationFrame + 0.2) % 4;
                } else {
                    this.animationFrame = 0;
                }
                this.moving = false;
                
                // 更新炮塔角度
                if (this.isPlayer) {
                    // 玩家坦克炮塔始终指向鼠标
                    const dx = game.mouseX - this.x;
                    const dy = game.mouseY - this.y;
                    this.turretAngle = Math.atan2(dy, dx);
                } else {
                    // 敌人坦克炮塔指向玩家
                    const dx = game.player.x - this.x;
                    const dy = game.player.y - this.y;
                    this.turretAngle = Math.atan2(dy, dx);
                }
            }
            
            move(direction) {
                this.moving = true;
                switch(direction) {
                    case 'up':
                        this.x += Math.cos(this.angle) * this.speed;
                        this.y += Math.sin(this.angle) * this.speed;
                        break;
                    case 'down':
                        this.x -= Math.cos(this.angle) * this.speed;
                        this.y -= Math.sin(this.angle) * this.speed;
                        break;
                    case 'left':
                        this.angle -= this.rotationSpeed;
                        break;
                    case 'right':
                        this.angle += this.rotationSpeed;
                        break;
                }
                
                // 边界检查
                this.x = Math.max(this.width/2, Math.min(GAME_WIDTH - this.width/2, this.x));
                this.y = Math.max(this.height/2, Math.min(GAME_HEIGHT - this.height/2, this.y));
            }
            
            shoot() {
                const now = Date.now();
                if (now - this.lastShot > this.shootCooldown) {
                    this.lastShot = now;
                    
                    // 计算子弹发射位置（炮塔顶端）
                    const turretLength = this.width * 0.8;
                    const bulletX = this.x + Math.cos(this.turretAngle) * turretLength;
                    const bulletY = this.y + Math.sin(this.turretAngle) * turretLength;
                    
                    // 创建子弹
                    const bullet = new Bullet(
                        bulletX, 
                        bulletY, 
                        this.turretAngle, 
                        this.isPlayer ? '#00ff00' : '#ff0000',
                        this.isPlayer
                    );
                    game.bullets.push(bullet);
                    
                    // 添加射击特效
                    this.createMuzzleFlash();
                }
            }
            
            createMuzzleFlash() {
                for (let i = 0; i < 5; i++) {
                    const particle = new Particle(
                        this.x + Math.cos(this.turretAngle) * this.width * 0.8,
                        this.y + Math.sin(this.turretAngle) * this.width * 0.8,
                        this.turretAngle + (Math.random() - 0.5) * 0.5,
                        this.isPlayer ? '#00ff00' : '#ff0000'
                    );
                    game.particles.push(particle);
                }
            }
            
            draw() {
                ctx.save();
                ctx.translate(this.x, this.y);
                
                // 绘制坦克主体
                ctx.rotate(this.angle);
                ctx.fillStyle = this.color;
                ctx.fillRect(-this.width/2, -this.height/2, this.width, this.height);
                
                // 绘履带
                ctx.fillStyle = 'rgba(0, 0, 0, 0.3)';
                ctx.fillRect(-this.width/2, -this.height/2 - 3, this.width, 6);
                ctx.fillRect(-this.width/2, this.height/2 - 3, this.width, 6);
                
                // 绘制履带动画
                if (this.moving) {
                    const offset = (this.animationFrame * 5) % 20;
                    ctx.fillStyle = 'rgba(0, 0, 0, 0.5)';
                    for (let i = -this.width/2; i < this.width/2; i += 10) {
                        ctx.fillRect(i + offset, -this.height/2 - 3, 8, 6);
                        ctx.fillRect(i + offset, this.height/2 - 3, 8, 6);
                    }
                }
                
                // 绘制炮塔
                ctx.rotate(this.angle - this.angle + this.turretAngle);
                ctx.fillStyle = this.color;
                ctx.fillRect(0, -this.width/6, this.width * 0.8, this.width/3);
                
                // 绘制炮管
                ctx.fillStyle = this.color;
                ctx.fillRect(this.width * 0.3, -this.width/12, this.width * 0.6, this.width/6);
                
                ctx.restore();
                
                // 绘制血条
                if (this.health < this.maxHealth) {
                    ctx.fillStyle = 'rgba(255, 0, 0, 0.5)';
                    ctx.fillRect(this.x - this.width/2, this.y - this.height/2 - 10, this.width, 4);
                    ctx.fillStyle = 'rgba(0, 255, 0, 0.5)';
                    ctx.fillRect(this.x - this.width/2, this.y - this.height/2 - 10, this.width * (this.health / this.maxHealth), 4);
                }
            }
            
            takeDamage(amount) {
                this.health -= amount;
                if (this.health <= 0) {
                    this.destroy();
                    return true;
                }
                return false;
            }
            
            destroy() {
                // 创建爆炸特效
                for (let i = 0; i < 20; i++) {
                    const particle = new Particle(
                        this.x,
                        this.y,
                        Math.random() * Math.PI * 2,
                        this.isPlayer ? '#00ff00' : '#ff0000'
                    );
                    game.particles.push(particle);
                }
                
                if (!this.isPlayer) {
                    game.score += 100;
                }
            }
        }
        
        // 子弹类
        class Bullet {
            constructor(x, y, angle, color, isPlayerBullet = false) {
                this.x = x;
                this.y = y;
                this.angle = angle;
                this.speed = 5;
                this.radius = 3;
                this.color = color;
                this.isPlayerBullet = isPlayerBullet;
                this.lifetime = 100;
            }
            
            update() {
                this.x += Math.cos(this.angle) * this.speed;
                this.y += Math.sin(this.angle) * this.speed;
                this.lifetime--;
                
                // 边界检查
                if (this.x < 0 || this.x > GAME_WIDTH || 
                    this.y < 0 || this.y > GAME_HEIGHT || 
                    this.lifetime <= 0) {
                    return false;
                }
                
                // 碰撞检测
                if (this.isPlayerBullet) {
                    for (let i = 0; i < game.enemies.length; i++) {
                        const enemy = game.enemies[i];
                        const dx = this.x - enemy.x;
                        const dy = this.y - enemy.y;
                        const distance = Math.sqrt(dx * dx + dy * dy);
                        
                        if (distance < enemy.width/2) {
                            if (enemy.takeDamage(25)) {
                                game.enemies.splice(i, 1);
                            }
                            return false;
                        }
                    }
                    
                    // 检查是否击中总部
                    for (let i = 0; i < game.headquarters.length; i++) {
                        const hq = game.headquarters[i];
                        const dx = this.x - hq.x;
                        const dy = this.y - hq.y;
                        const distance = Math.sqrt(dx * dx + dy * dy);
                        
                        if (distance < hq.radius) {
                            hq.health -= 10;
                            if (hq.health <= 0) {
                                game.gameOver = true;
                            }
                            return false;
                        }
                    }
                } else {
                    // 敌人子弹
                    const dx = this.x - game.player.x;
                    const dy = this.y - game.player.y;
                    const distance = Math.sqrt(dx * dx + dy * dy);
                    
                    if (distance < game.player.width/2) {
                        if (game.player.takeDamage(20)) {
                            game.gameOver = true;
                        }
                        return false;
                    }
                }
                
                return true;
            }
            
            draw() {
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
                ctx.fillStyle = this.color;
                ctx.fill();
                
                // 绘制拖尾效果
                ctx.beginPath();
                ctx.moveTo(this.x, this.y);
                ctx.lineTo(
                    this.x - Math.cos(this.angle) * 10,
                    this.y - Math.sin(this.angle) * 10
                );
                ctx.strokeStyle = this.color;
                ctx.lineWidth = 2;
                ctx.stroke();
            }
        }
        
        // 粒子效果类
        class Particle {
            constructor(x, y, angle, color) {
                this.x = x;
                this.y = y;
                this.angle = angle;
                this.speed = Math.random() * 3 + 1;
                this.radius = Math.random() * 3 + 1;
                this.color = color;
                this.lifetime = 20;
                this.alpha = 1;
            }
            
            update() {
                this.x += Math.cos(this.angle) * this.speed;
                this.y += Math.sin(this.angle) * this.speed;
                this.lifetime--;
                this.alpha = this.lifetime / 20;
                
                return this.lifetime > 0;
            }
            
            draw() {
                ctx.save();
                ctx.globalAlpha = this.alpha;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
                ctx.fillStyle = this.color;
                ctx.fill();
                ctx.restore();
            }
        }
        
        // 总部类
        class Headquarters {
            constructor(x, y, isPlayer = true) {
                this.x = x;
                this.y = y;
                this.radius = 40;
                this.color = isPlayer ? '#3498db' : '#e74c3c';
                this.health = 100;
                this.maxHealth = 100;
                this.isPlayer = isPlayer;
            }
            
            draw() {
                // 绘制总部建筑
                ctx.fillStyle = this.color;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
                ctx.fill();
                
                // 绘制建筑细节
                ctx.fillStyle = 'rgba(255, 255, 255, 0.3)';
                ctx.beginPath();
                ctx.arc(this.x - 10, this.y - 10, 10, 0, Math.PI * 2);
                ctx.fill();
                
                // 绘制血条
                if (this.health < this.maxHealth) {
                    ctx.fillStyle = 'rgba(255, 0, 0, 0.5)';
                    ctx.fillRect(this.x - this.radius, this.y - this.radius - 15, this.radius * 2, 6);
                    ctx.fillStyle = 'rgba(0, 255, 0, 0.5)';
                    ctx.fillRect(this.x - this.radius, this.y - this.radius - 15, this.radius * 2 * (this.health / this.maxHealth), 6);
                }
            }
        }
        
        // 初始化游戏
        function init() {
            // 创建玩家坦克
            game.player = new Tank(GAME_WIDTH / 2, GAME_HEIGHT - 100, '#3498db', true);
            
            // 创建敌方坦克
            for (let i = 0; i < 3; i++) {
                const enemy = new Tank(
                    Math.random() * (GAME_WIDTH - 100) + 50,
                    Math.random() * 200 + 50,
                    '#e74c3c',
                    false
                );
                game.enemies.push(enemy);
            }
            
            // 创建总部
            game.headquarters.push(new Headquarters(GAME_WIDTH / 2, 50, false)); // 敌方总部
            
            // 鼠标位置
            game.mouseX = game.player.x;
            game.mouseY = game.player.y;
            
            // 绑定事件
            bindEvents();
        }
        
        // 绑定事件
        function bindEvents() {
            // 键盘事件
            window.addEventListener('keydown', (e) => {
                game.keys[e.key.toLowerCase()] = true;
            });
            
            window.addEventListener('keyup', (e) => {
                game.keys[e.key.toLowerCase()] = false;
            });
            
            // 鼠标事件
            canvas.addEventListener('mousemove', (e) => {
                const rect = canvas.getBoundingClientRect();
                game.mouseX = e.clientX - rect.left;
                game.mouseY = e.clientY - rect.top;
            });
            
            canvas.addEventListener('click', () => {
                if (!game.gameOver) {
                    game.player.shoot();
                }
            });
            
            // 触摸控制事件
            const touchButtons = {
                upBtn: 'up',
                downBtn: 'down',
                leftBtn: 'left',
                rightBtn: 'right',
                shootBtn: 'shoot'
            };
            
            for (const [btnId, action] of Object.entries(touchButtons)) {
                const btn = document.getElementById(btnId);
                
                btn.addEventListener('touchstart', (e) => {
                    e.preventDefault();
                    if (action === 'shoot') {
                        game.player.shoot();
                    } else {
                        game.keys[action] = true;
                    }
                });
                
                btn.addEventListener('touchend', (e) => {
                    e.preventDefault();
                    if (action !== 'shoot') {
                        game.keys[action] = false;
                    }
                });
                
                btn.addEventListener('mousedown', (e) => {
                    if (action === 'shoot') {
                        game.player.shoot();
                    } else
