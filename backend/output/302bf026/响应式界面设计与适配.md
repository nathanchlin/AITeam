        // 坦克类
        class Tank {
            constructor(x, y, color, isPlayer = false) {
                this.x = x;
                this.y = y;
                this.width = GAME_CONFIG.tankSize;
                this.height = GAME_CONFIG.tankSize;
                this.color = color;
                this.speed = isPlayer ? GAME_CONFIG.tankSpeed : GAME_CONFIG.enemyTankSpeed;
                this.direction = 'up';
                this.isPlayer = isPlayer;
                this.bullets = [];
                this.lastFireTime = 0;
                this.fireInterval = isPlayer ? 500 : 1000;
            }

            draw() {
                ctx.save();
                ctx.translate(this.x + this.width / 2, this.y + this.height / 2);
                
                // 根据方向旋转坦克
                let rotation = 0;
                switch(this.direction) {
                    case 'up': rotation = 0; break;
                    case 'right': rotation = Math.PI / 2; break;
                    case 'down': rotation = Math.PI; break;
                    case 'left': rotation = -Math.PI / 2; break;
                }
                ctx.rotate(rotation);
                
                // 绘制坦克主体
                ctx.fillStyle = this.color;
                ctx.fillRect(-this.width / 2, -this.height / 2, this.width, this.height);
                
                // 绘制坦克炮管
                ctx.fillStyle = this.isPlayer ? '#2ecc71' : '#e74c3c';
                ctx.fillRect(0, -this.width / 8, this.width * 0.8, this.width / 4);
                
                // 绘制坦克中心
                ctx.fillStyle = '#34495e';
                ctx.beginPath();
                ctx.arc(0, 0, this.width / 4, 0, Math.PI * 2);
                ctx.fill();
                
                ctx.restore();
            }

            move(direction) {
                this.direction = direction;
                let newX = this.x;
                let newY = this.y;
                
                switch(direction) {
                    case 'up':
                        newY -= this.speed;
                        break;
                    case 'down':
                        newY += this.speed;
                        break;
                    case 'left':
                        newX -= this.speed;
                        break;
                    case 'right':
                        newX += this.speed;
                        break;
                }
                
                // 边界检测
                if (newX >= 0 && newX + this.width <= gameState.canvasWidth &&
                    newY >= 0 && newY + this.height <= gameState.canvasHeight) {
                    
                    // 检测与其他坦克的碰撞
                    let collision = false;
                    for (let tank of enemies) {
                        if (tank !== this && this.checkCollision(newX, newY, tank)) {
                            collision = true;
                            break;
                        }
                    }
                    
                    if (!collision) {
                        this.x = newX;
                        this.y = newY;
                    }
                }
            }

            checkCollision(x, y, other) {
                return x < other.x + other.width &&
                       x + this.width > other.x &&
                       y < other.y + other.height &&
                       y + this.height > other.y;
            }

            fire() {
                const now = Date.now();
                if (now - this.lastFireTime > this.fireInterval) {
                    this.lastFireTime = now;
                    
                    let bulletX = this.x + this.width / 2;
                    let bulletY = this.y + this.height / 2;
                    let bulletDirection = this.direction;
                    
                    const bullet = new Bullet(bulletX, bulletY, bulletDirection, this.isPlayer);
                    if (this.isPlayer) {
                        bullets.push(bullet);
                    } else {
                        enemyBullets.push(bullet);
                    }
                }
            }
        }

        // 子弹类
        class Bullet {
            constructor(x, y, direction, isPlayerBullet = true) {
                this.x = x;
                this.y = y;
                this.width = GAME_CONFIG.bulletSize;
                this.height = GAME_CONFIG.bulletSize;
                this.direction = direction;
                this.speed = GAME_CONFIG.bulletSpeed;
                this.isPlayerBullet = isPlayerBullet;
                this.active = true;
            }

            draw() {
                ctx.fillStyle = this.isPlayerBullet ? '#2ecc71' : '#e74c3c';
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.width / 2, 0, Math.PI * 2);
                ctx.fill();
            }

            update() {
                switch(this.direction) {
                    case 'up':
                        this.y -= this.speed;
                        break;
                    case 'down':
                        this.y += this.speed;
                        break;
                    case 'left':
                        this.x -= this.speed;
                        break;
                    case 'right':
                        this.x += this.speed;
                        break;
                }
                
                // 检查是否超出边界
                if (this.x < 0 || this.x > gameState.canvasWidth ||
                    this.y < 0 || this.y > gameState.canvasHeight) {
                    this.active = false;
                }
            }
        }

        // 基地类
        class Base {
            constructor(x, y, isPlayerBase = true) {
                this.x = x;
                this.y = y;
                this.width = GAME_CONFIG.baseSize;
                this.height = GAME_CONFIG.baseSize;
                this.health = GAME_CONFIG.baseHealth;
                this.isPlayerBase = isPlayerBase;
                this.color = isPlayerBase ? '#3498db' : '#e74c3c';
            }

            draw() {
                ctx.fillStyle = this.color;
                ctx.fillRect(this.x, this.y, this.width, this.height);
                
                // 绘制基地标志
                ctx.fillStyle = 'white';
                ctx.font = '16px Arial';
                ctx.textAlign = 'center';
                ctx.textBaseline = 'middle';
                ctx.fillText(this.isPlayerBase ? 'P' : 'E', this.x + this.width / 2, this.y + this.height / 2);
            }

            takeDamage(damage) {
                this.health -= damage;
                if (this.health <= 0) {
                    this.health = 0;
                    return true; // 基地被摧毁
                }
                return false;
            }
        }

        // 调整画布大小
        function resizeCanvas() {
            const container = document.getElementById('gameContainer');
            const containerWidth = container.clientWidth;
            const containerHeight = container.clientHeight;
            
            // 计算缩放比例
            const scaleX = containerWidth / GAME_CONFIG.canvasWidth;
            const scaleY = containerHeight / GAME_CONFIG.canvasHeight;
            gameState.scale = Math.min(scaleX, scaleY);
            
            // 设置画布尺寸
            gameState.canvasWidth = containerWidth;
            gameState.canvasHeight = containerHeight;
            
            canvas.width = containerWidth;
            canvas.height = containerHeight;
        }

        // 生成敌人坦克
        function spawnEnemy() {
            if (enemies.length < GAME_CONFIG.maxEnemies) {
                const x = Math.random() * (gameState.canvasWidth - GAME_CONFIG.tankSize);
                const enemy = new Tank(x, 20, '#e74c3c', false);
                enemies.push(enemy);
            }
        }

        // 更新游戏状态
        function update() {
            if (gameState.isGameOver || gameState.isPaused) return;
            
            // 更新玩家坦克
            if (keys['ArrowUp'] || keys['w']) player.move('up');
            if (keys['ArrowDown'] || keys['s']) player.move('down');
            if (keys['ArrowLeft'] || keys['a']) player.move('left');
            if (keys['ArrowRight'] || keys['d']) player.move('right');
            if (keys[' ']) player.fire();
            
            // 更新敌人坦克
            const now = Date.now();
            
            // 生成新敌人
            if (now - lastEnemySpawnTime > GAME_CONFIG.enemySpawnInterval / gameState.level) {
                spawnEnemy();
                lastEnemySpawnTime = now;
            }
            
            // 更新敌人坦克行为
            for (let i = enemies.length - 1; i >= 0; i--) {
                const enemy = enemies[i];
                
                // 简单的AI：随机移动
                if (Math.random() < 0.02) {
                    const directions = ['up', 'down', 'left', 'right'];
                    enemy.direction = directions[Math.floor(Math.random() * directions.length)];
                }
                
                enemy.move(enemy.direction);
                
                // 敌人射击
                if (now - lastEnemyFireTime > GAME_CONFIG.enemyFireInterval / gameState.level) {
                    enemy.fire();
                    lastEnemyFireTime = now;
                }
                
                // 检测敌人是否到达基地
                for (let base of bases) {
                    if (base.isPlayerBase && enemy.checkCollision(enemy.x, enemy.y, base)) {
                        // 游戏结束
                        gameOver();
                        return;
                    }
                }
            }

                bullet.update();

                    continue;
