        // 游戏主类
        class Game {
            constructor() {
                this.canvas = document.getElementById('gameCanvas');
                this.ctx = this.canvas.getContext('2d');
                this.state = GameState.MENU;
                this.lastTime = 0;
                this.animationId = null;
                
                // 游戏对象
                this.player = null;
                this.enemies = [];
                this.bullets = [];

            setupEventListeners() {
                // 键盘事件

                    if (e.key === ' ' && this.state === GameState.PLAYING) {

                    this.keys[e.key] = false;

                // 重置游戏状态

                    this.enemies.push(new Tank(x, y, 'enemy'));
                }
            }

                    this.lastTime = timestamp;

            update(deltaTime) {
                if (this.state !== GameState.PLAYING) return;
                
                // 更新玩家

                // 更新子弹
                this.updateBullets();
                
                // 检测碰撞

                // 检查游戏状态

            updatePlayer() {
                // 处理玩家输入

                    this.player.moveUp();
                }

                    this.player.moveDown();
                }

                this.player.update();
                
                // 限制玩家在画布内
                this.player.x = Math.max(0, Math.min(GameConfig.canvasWidth - GameConfig.tankSize, this.player.x));
                this.player.y = Math.max(0, Math.min(GameConfig.canvasHeight - GameConfig.tankSize, this.player.y));
            }
            
            updateEnemies() {
                // 更新每个敌人
                for (let i = this.enemies.length - 1; i >= 0; i--) {
                    const enemy = this.enemies[i];
                    
                    // 简单的AI：随机移动
                    if (Math.random() < 0.02) {
                        const directions = ['up', 'down', 'left', 'right'];
                        const randomDirection = directions[Math.floor(Math.random() * directions.length)];
                        enemy.setDirection(randomDirection);
                    }
                    
                    // 随机开火
                    if (Math.random() < GameConfig.enemyFireRate) {
                        enemy.fire();
                    }
                    
                    // 更新敌人位置
                    enemy.update();
                    
                    // 限制敌人在画布内
                    enemy.x = Math.max(0, Math.min(GameConfig.canvasWidth - GameConfig.tankSize, enemy.x));
                    enemy.y = Math.max(0, Math.min(GameConfig.canvasHeight - GameConfig.tankSize, enemy.y));
                }
            }
            
            updateBullets() {
                // 更新所有子弹
                for (let i = this.bullets.length - 1; i >= 0; i--) {
                    const bullet = this.bullets[i];
                    bullet.update();
                    
                    // 移除超出画布的子弹
                    if (bullet.x < 0 || bullet.x > GameConfig.canvasWidth || 
                        bullet.y < 0 || bullet.y > GameConfig.canvasHeight) {
                        this.bullets.splice(i, 1);
                    }
                }
            }
            
            checkCollisions() {
                // 检查子弹与坦克的碰撞
                for (let i = this.bullets.length - 1; i >= 0; i--) {
                    const bullet = this.bullets[i];
                    
                    // 检查子弹与玩家的碰撞
                    if (bullet.owner === 'enemy' && this.checkCollision(bullet, this.player)) {
                        this.playerHit();
                        this.bullets.splice(i, 1);
                        continue;
                    }
                    
                    // 检查子弹与敌人的碰撞
                    for (let j = this.enemies.length - 1; j >= 0; j--) {
                        if (bullet.owner === 'player' && this.checkCollision(bullet, this.enemies[j])) {
                            this.enemies.splice(j, 1);
                            this.bullets.splice(i, 1);
                            GameConfig.score += 100;
                            break;
                        }
                    }
                    
                    // 检查子弹与总部的碰撞
                    for (let j = this.headquarters.length - 1; j >= 0; j--) {
                        if (this.checkCollision(bullet, this.headquarters[j])) {
                            if (bullet.owner === 'player' && this.headquarters[j].type === 'enemy') {
                                this.winLevel();
                            } else if (bullet.owner === 'enemy' && this.headquarters[j].type === 'player') {
                                this.playerHeadquartersHit();
                            }
                            this.bullets.splice(i, 1);
                            break;
                        }
                    }
                    
                    // 检查子弹与障碍物的碰撞
                    for (let j = this.obstacles.length - 1; j >= 0; j--) {
                        if (this.checkCollision(bullet, this.obstacles[j])) {
                            this.bullets.splice(i, 1);
                            break;
                        }
                    }
                }
                
                // 检查坦克与障碍物的碰撞
                for (const obstacle of this.obstacles) {
                    if (this.checkCollision(this.player, obstacle)) {
                        this.player.revertPosition();
                    }
                    
                    for (const enemy of this.enemies) {
                        if (this.checkCollision(enemy, obstacle)) {
                            enemy.revertPosition();
                        }
                    }
                }
            }
            
            checkCollision(obj1, obj2) {
                return obj1.x < obj2.x + obj2.width &&
                       obj1.x + obj1.width > obj2.x &&
                       obj1.y < obj2.y + obj2.height &&
                       obj1.y + obj1.height > obj2.y;
            }
            
            playerHit() {
                GameConfig.playerLives--;
                if (GameConfig.playerLives <= 0) {
                    this.gameOver(false);
                } else {
                    // 重置玩家位置
                    this.player.x = GameConfig.canvasWidth / 2;
                    this.player.y = GameConfig.canvasHeight - 50;
                }
            }
            
            playerHeadquartersHit() {
                this.gameOver(false);
            }
            
            winLevel() {
                GameConfig.level++;
                this.enemies = [];
                this.createEnemies();
                this.obstacles = [];
                this.createObstacles();
                
                // 重置玩家位置
                this.player.x = GameConfig.canvasWidth / 2;
                this.player.y = GameConfig.canvasHeight - 50;
            }
            
            checkGameState() {
                // 检查是否所有敌人都被消灭
                if (this.enemies.length === 0) {
                    this.winLevel();
                }
            }
            
            gameOver(isWin) {
                this.state = GameState.GAME_OVER;
                document.getElementById('gameOverTitle').textContent = isWin ? '胜利!' : '游戏结束';
                document.getElementById('finalScore').textContent = GameConfig.score;
                document.getElementById('gameOver').style.display = 'block';
            }
            
            render() {
                // 清空画布
                this.ctx.fillStyle = '#111';
                this.ctx.fillRect(0, 0, GameConfig.canvasWidth, GameConfig.canvasHeight);
                
                // 绘制障碍物
                for (const obstacle of this.obstacles) {
                    obstacle.render(this.ctx);
                }
                
                // 绘制总部
                for (const headquarters of this.headquarters) {
                    headquarters.render(this.ctx);
                }
                
                // 绘制玩家
                this.player.render(this.ctx);
                
                // 绘制敌人
                for (const enemy of this.enemies) {
                    enemy.render(this.ctx);
                }
                
                // 绘制子弹
                for (const bullet of this.bullets) {
                    bullet.render(this.ctx);
                }
            }
        }

        // 坦克类
        class Tank {
            constructor(x, y, owner) {
                this.x = x;
                this.y = y;
                this.owner = owner;
                this.width = GameConfig.tankSize;
                this.height = GameConfig.tankSize;
                this.speed = owner === 'player' ? GameConfig.tankSpeed : GameConfig.enemyTankSpeed;
                this.direction = 'up';
                this.lastPosition = { x: x, y: y };
                this.canFire = true;
                this.fireCooldown = 0;
            }
            
            moveLeft() {
                this.lastPosition = { x: this.x, y: this.y };
                this.x -= this.speed;
                this.direction = 'left';
            }
            
            moveRight() {
                this.lastPosition = { x: this.x, y: this.y };
                this.x += this.speed;
                this.direction = 'right';
            }
            
            moveUp() {
                this.lastPosition = { x: this.x, y: this.y };
                this.y -= this.speed;
                this.direction = 'up';
            }
            
            moveDown() {
                this.lastPosition = { x: this.x, y: this.y };
                this.y += this.speed;
                this.direction = 'down';
            }
            
            setDirection(direction) {
                this.direction = direction;
            }
            
            revertPosition() {
                this.x = this.lastPosition.x;
                this.y = this.lastPosition.y;
            }
            
            fire() {
                if (!this.canFire) return;
                
                let bulletX = this.x + this.width / 2 - GameConfig.bulletSize / 2;
                let bulletY = this.y + this.height / 2 - GameConfig.bulletSize / 2;

                    case 'up':
                        bulletY = this.y - GameConfig.bulletSize;
                        break;
                    case 'down':
                        bulletY = this.y + this.height;
                        break;
                    case 'left':
                        bulletX = this.x - GameConfig.bulletSize;
                        break;
                    case 'right':
                        bulletX = this.x + this.width;
                        break;

                            this.x + this.width,
                            this.y + this.height / 2 - barrelWidth / 2,
                            barrelSize,
                            barrelWidth
                        );

                }

            }
