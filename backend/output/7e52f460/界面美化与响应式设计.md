# 贪吃蛇游戏界面美化与响应式设计

我将为贪吃蛇游戏设计一个美观且响应式的界面，确保在不同屏幕尺寸下都能良好显示。以下是完整的实现方案：

## 设计思路

1. **现代化视觉风格**：采用深色主题配合霓虹色彩，营造游戏氛围
2. **响应式布局**：使用CSS Grid和Flexbox实现自适应布局
3. **动画效果**：添加平滑的过渡和动画增强用户体验
4. **触摸友好**：为移动设备添加触摸控制按钮
5. **视觉反馈**：通过颜色变化和动画提供游戏状态反馈

    <script>
        document.addEventListener('DOMContentLoaded', () => {
            // 游戏配置
            const BOARD_SIZE = 20;
            const INITIAL_SPEED = 150;
            const SPEED_INCREMENT = 10;
            const POINTS_PER_FOOD = 10;
            
            // 游戏状态
            let snake = [];
            let direction = 'right';
            let nextDirection = 'right';
            let food = {};
            let score = 0;
            let highScore = localStorage.getItem('snakeHighScore') || 0;
            let gameInterval;
            let isPaused = false;
            let isGameOver = false;
            let currentSpeed = INITIAL_SPEED;
            
            // DOM 元素
            const gameBoard = document.getElementById('game-board');
            const scoreElement = document.getElementById('score');
            const highScoreElement = document.getElementById('high-score');
            const speedElement = document.getElementById('speed');
            const startBtn = document.getElementById('start-btn');
            const pauseBtn = document.getElementById('pause-btn');
            const resetBtn = document.getElementById('reset-btn');
            const restartBtn = document.getElementById('restart-btn');
            const gameOverOverlay = document.getElementById('game-over');
            const finalScoreElement = document.getElementById('final-score');
            const arrowButtons = document.querySelectorAll('.arrow-btn');
            
            // 初始化高分显示
            highScoreElement.textContent = highScore;
            
            // 创建游戏板
            function createBoard() {
                gameBoard.innerHTML = '';
                for (let i = 0; i < BOARD_SIZE * BOARD_SIZE; i++) {
                    const cell = document.createElement('div');
                    cell.classList.add('cell');
                    gameBoard.appendChild(cell);
                }
            }
            
            // 获取随机位置
            function getRandomPosition() {
                return {
                    x: Math.floor(Math.random() * BOARD_SIZE),
                    y: Math.floor(Math.random() * BOARD_SIZE)
                };
            }
            
            // 检查位置是否在蛇身上
            function isOnSnake(position) {
                return snake.some(segment => segment.x === position.x && segment.y === position.y);
            }
            
            // 生成食物
            function generateFood() {
                let newFood;
                do {
                    newFood = getRandomPosition();
                } while (isOnSnake(newFood));
                
                food = newFood;
                updateBoard();
            }
            
            // 初始化游戏
            function initGame() {
                snake = [
                    { x: 10, y: 10 },
                    { x: 9, y: 10 },
                    { x: 8, y: 10 }
                ];
                direction = 'right';
                nextDirection = 'right';
                score = 0;
                currentSpeed = INITIAL_SPEED;
                isPaused = false;
                isGameOver = false;
                
                scoreElement.textContent = score;
                speedElement.textContent = Math.floor((INITIAL_SPEED - currentSpeed) / SPEED_INCREMENT) + 1;
                
                generateFood();
                updateBoard();
            }
            
            // 更新游戏板
            function updateBoard() {
                const cells = document.querySelectorAll('.cell');
                
                // 清除所有单元格
                cells.forEach(cell => {
                    cell.classList.remove('snake', 'snake-head', 'food');
                });
                
                // 绘制蛇
                snake.forEach((segment, index) => {
                    const cellIndex = segment.y * BOARD_SIZE + segment.x;
                    if (cells[cellIndex]) {
                        if (index === 0) {
                            cells[cellIndex].classList.add('snake-head');
                        } else {
                            cells[cellIndex].classList.add('snake');
                        }
                    }
                });
                
                // 绘制食物
                const foodCellIndex = food.y * BOARD_SIZE + food.x;
                if (cells[foodCellIndex]) {
                    cells[foodCellIndex].classList.add('food');
                }
            }
            
            // 移动蛇
            function moveSnake() {
                if (isPaused || isGameOver) return;
                
                direction = nextDirection;
                
                const head = { ...snake[0] };
                
                switch (direction) {
                    case 'up':
                        head.y -= 1;
                        break;
                    case 'down':
                        head.y += 1;
                        break;
                    case 'left':
                        head.x -= 1;
                        break;
                    case 'right':
                        head.x += 1;
                        break;
                }
                
                // 检查碰撞
                if (
                    head.x < 0 || head.x >= BOARD_SIZE ||
                    head.y < 0 || head.y >= BOARD_SIZE ||
                    snake.some(segment => segment.x === head.x && segment.y === head.y)
                ) {
                    gameOver();
                    return;
                }
                
                snake.unshift(head);
                
                // 检查是否吃到食物
                if (head.x === food.x && head.y === food.y) {
                    score += POINTS_PER_FOOD;
                    scoreElement.textContent = score;
                    
                    // 更新速度
                    if (score % 50 === 0 && currentSpeed > 50) {
                        currentSpeed -= SPEED_INCREMENT;
                        speedElement.textContent = Math.floor((INITIAL_SPEED - currentSpeed) / SPEED_INCREMENT) + 1;
                        clearInterval(gameInterval);
                        gameInterval = setInterval(gameLoop, currentSpeed);
                    }
                    
                    generateFood();
                } else {
                    snake.pop();
                }
                
                updateBoard();
            }
            
            // 游戏循环
            function gameLoop() {
                moveSnake();
            }
            
            // 游戏结束
            function gameOver() {
                isGameOver = true;
                clearInterval(gameInterval);
                
                if (score > highScore) {
                    highScore = score;
                    highScoreElement.textContent = highScore;
                    localStorage.setItem('snakeHighScore', highScore);
                }
                
                finalScoreElement.textContent = score;
                gameOverOverlay.style.display = 'flex';
            }
            
            // 开始游戏
            function startGame() {
                if (gameInterval) return;
                
                if (isGameOver) {
                    initGame();
                }
                
                gameInterval = setInterval(gameLoop, currentSpeed);
                startBtn.textContent = '游戏中';
                startBtn.disabled = true;
            }
            
            // 暂停游戏
            function pauseGame() {
                if (!gameInterval || isGameOver) return;
                
                isPaused = !isPaused;
