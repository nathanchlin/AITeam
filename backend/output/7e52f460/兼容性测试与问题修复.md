## 3. 发现的兼容性问题及修复

### 问题1：IE 11兼容性问题
**问题描述**：在IE 11中，游戏无法正常加载，部分CSS样式不显示，JavaScript控制功能失效。

**原因分析**：
1. IE 11不支持某些CSS3特性（如flexbox）
2. IE 11对ES6语法支持不完全

**修复方案**：
```javascript
// 替换ES6语法为ES5兼容代码
// 原代码：const canvas = document.getElementById('gameCanvas');
// 修复后：
var canvas = document.getElementById('gameCanvas');

// 替换箭头函数
// 原代码：const handleKeyDown = (e) => {
// 修复后：
function handleKeyDown(e) {
    // 函数内容保持不变
}

// 添加IE 11 flexbox兼容样式
.game-container {
    display: -ms-flexbox;
    display: -webkit-flex;
    display: flex;
    -ms-flex-direction: column;
    -webkit-flex-direction: column;
    flex-direction: column;
    -ms-flex-pack: center;
    -webkit-justify-content: center;
    justify-content: center;
    -ms-flex-align: center;
    -webkit-align-items: center;
    align-items: center;
    height: 100vh;
    background-color: #f0f0f0;
}
```

### 问题2：移动设备触摸控制问题
**问题描述**：在部分移动设备上，触摸控制不够灵敏，蛇的移动方向响应延迟。

**原因分析**：
1. 移动设备触摸事件处理不够完善
2. 缺乏触摸方向判断逻辑

**修复方案**：
```javascript
// 添加触摸事件处理
var touchStartX = 0;
var touchStartY = 0;

canvas.addEventListener('touchstart', function(e) {
    e.preventDefault();
    touchStartX = e.touches[0].clientX;
    touchStartY = e.touches[0].clientY;
}, false);

canvas.addEventListener('touchmove', function(e) {
    e.preventDefault();
}, false);

canvas.addEventListener('touchend', function(e) {
    e.preventDefault();
    var touchEndX = e.changedTouches[0].clientX;
    var touchEndY = e.changedTouches[0].clientY;
    
    var dx = touchEndX - touchStartX;
    var dy = touchEndY - touchStartY;
    
    if (Math.abs(dx) > Math.abs(dy)) {
        // 水平滑动
        if (dx > 0 && direction !== 'left') {
            direction = 'right';
        } else if (dx < 0 && direction !== 'right') {
            direction = 'left';
        }
    } else {
        // 垂直滑动
        if (dy > 0 && direction !== 'up') {
            direction = 'down';
        } else if (dy < 0 && direction !== 'down') {
            direction = 'up';
        }
    }
}, false);
```

### 问题3：响应式布局问题
**问题描述**：在小屏幕设备上，游戏区域显示过小，难以操作。

**原因分析**：
1. 游戏画布大小固定，未根据屏幕尺寸调整
2. 控制按钮在小屏幕上过小

**修复方案**：
```css
/* 响应式布局调整 */
.game-container {
    width: 100%;
    max-width: 600px;
    padding: 10px;
    box-sizing: border-box;
}

#gameCanvas {
    width: 100%;
    max-width: 500px;
    height: auto;
    border: 2px solid #333;
}

.controls {
    width: 100%;
    max-width: 500px;
    display: flex;
    justify-content: space-between;
    margin-top: 20px;
}

button {
    padding: 10px 20px;
    font-size: 16px;
    min-width: 100px;
}

/* 小屏幕调整 */
@media (max-width: 600px) {
    button {
        padding: 8px 15px;
        font-size: 14px;
        min-width: 80px;
    }
}

/* 超小屏幕调整 */
@media (max-width: 400px) {
    button {
        padding: 6px 12px;
        font-size: 12px;
        min-width: 60px;
    }
}
```

### 问题4：游戏性能问题
**问题描述**：在低性能设备上，游戏运行卡顿，动画不流畅。

**原因分析**：
1. 游戏循环频率过高
2. 未使用requestAnimationFrame优化

**修复方案**：
```javascript
// 使用requestAnimationFrame优化游戏循环
function gameLoop() {
    if (!gameRunning) return;
    
    // 更新游戏状态
    update();
    
    // 渲染游戏
    render();
    
    // 控制游戏速度
    setTimeout(function() {
        requestAnimationFrame(gameLoop);
    }, 100); // 控制速度，可根据设备性能调整
}

// 替换原来的setInterval
// 原代码：setInterval(gameLoop, 100);
// 修复后：
requestAnimationFrame(gameLoop);
```

    <script>
        // 获取画布和上下文
        var canvas = document.getElementById('gameCanvas');
        var ctx = canvas.getContext('2d');
        var scoreElement = document.getElementById('score');
        var finalScoreElement = document.getElementById('finalScore');
        var startBtn = document.getElementById('startBtn');
        var pauseBtn = document.getElementById('pauseBtn');
        var restartBtn = document.getElementById('restartBtn');
        var gameOverElement = document.getElementById('gameOver');
        
        // 游戏变量
        var gridSize = 20;
        var tileCount = canvas.width / gridSize;
        var snake = [
            {x: 10, y: 10}
        ];
        var food = {};
        var dx = 0;
        var dy = 0;
        var score = 0;
        var gameRunning = false;
        var gamePaused = false;
        
        // 初始化食物位置
        function generateFood() {
            food = {
                x: Math.floor(Math.random() * tileCount),
                y: Math.floor(Math.random() * tileCount)
            };
            
            // 确保食物不会生成在蛇身上
            for (var i = 0; i < snake.length; i++) {
                if (snake[i].x === food.x && snake[i].y === food.y) {
                    generateFood();
                    break;
                }
            }
        }
        
        // 绘制游戏
        function drawGame() {
            // 清除画布
            ctx.fillStyle = '#fff';
            ctx.fillRect(0, 0, canvas.width, canvas.height);
            
            // 绘制蛇
            ctx.fillStyle = '#4CAF50';
            for (var i = 0; i < snake.length; i++) {
                ctx.fillRect(snake[i].x * gridSize, snake[i].y * gridSize, gridSize - 2, gridSize - 2);
            }
            
            // 绘制食物
            ctx.fillStyle = '#f44336';
            ctx.fillRect(food.x * gridSize, food.y * gridSize, gridSize - 2, gridSize - 2);
        }
        
        // 更新游戏状态
        function update() {
            if (!gameRunning || gamePaused) return;
            
            // 移动蛇头
            var head = {x: snake[0].x + dx, y: snake[0].y + dy};
            
            // 检查碰撞墙壁
            if (head.x < 0 || head.x >= tileCount || head.y < 0 || head.y >= tileCount) {
                gameOver();
                return;
            }
            
            // 检查碰撞自身
            for (var i = 0; i < snake.length; i++) {
                if (head.x === snake[i].x && head.y === snake[i].y) {
                    gameOver();
                    return;
                }
            }
            
            // 添加新头部
            snake.unshift(head);
            
            // 检查是否吃到食物
            if (head.x === food.x && head.y === food.y) {
                score += 10;
                scoreElement.textContent = score;
                generateFood();
            } else {
                // 移除尾部
                snake.pop();
            }
        }
        
        // 游戏循环
        function gameLoop() {
            if (!gameRunning) return;
            
            update();
            drawGame();
            
            // 控制游戏速度
            setTimeout(function() {
                requestAnimationFrame(gameLoop);
            }, 100);
        }
        
        // 开始游戏
        function startGame() {
            if (gameRunning) return;
            
            // 重置游戏状态
            snake = [{x: 10, y: 10}];
            dx = 0;
            dy = 0;
            score = 0;
            scoreElement.textContent = score;
            gameRunning = true;
            gamePaused = false;
            gameOverElement.style.display = 'none';
            
            generateFood();
            gameLoop();
        }
        
        // 暂停/继续游戏
        function togglePause() {
            if (!gameRunning) return;
            
            gamePaused = !gamePaused;
            pauseBtn.textContent = gamePaused ? '继续' : '暂停';
            
            if (!gamePaused) {
                gameLoop();
            }
        }
        
        // 游戏结束
        function gameOver() {
            gameRunning = false;
            finalScoreElement.textContent = score;
            gameOverElement.style.display = 'block';
        }
        
        // 键盘控制
        function handleKeyDown(e) {
            if (!gameRunning || gamePaused) return;
            
            // 防止蛇直接反向移动
            if (e.key === 'ArrowUp' && dy !== 1) {
                dx = 0;
                dy = -1;
            } else if (e.key === 'ArrowDown' && dy !== -1) {
                dx = 0;
                dy = 1;

                dx = -1;
                dy = 0;
            } else if (e.key === 'ArrowRight' && dx !== -1) {
                dx = 1;
                dy = 0;
            }
