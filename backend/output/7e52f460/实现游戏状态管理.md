    <script>
        // 游戏状态管理
        const GameState = {
            IDLE: 'idle',       // 游戏未开始
            PLAYING: 'playing', // 游戏中
            PAUSED: 'paused',   // 游戏暂停
            GAME_OVER: 'gameOver' // 游戏结束
        };

        // 游戏配置
        const GRID_SIZE = 20;
        const CELL_SIZE = 20;
        const INITIAL_SPEED = 150;
        const SPEED_INCREMENT = 2;
        
        // 获取DOM元素
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        const scoreElement = document.getElementById('score');
        const highScoreElement = document.getElementById('highScore');
        const finalScoreElement = document.getElementById('finalScore');
        const startBtn = document.getElementById('startBtn');
        const pauseBtn = document.getElementById('pauseBtn');
        const resetBtn = document.getElementById('resetBtn');
        const restartBtn = document.getElementById('restartBtn');
        const gameOverOverlay = document.getElementById('gameOverOverlay');
        const pauseIndicator = document.getElementById('pauseIndicator');
        
        // 游戏状态变量
        let gameState = GameState.IDLE;
        let snake = [];
        let direction = { x: 1, y: 0 };
        food = {};
        let score = 0;
        let highScore = localStorage.getItem('snakeHighScore') || 0;
        let gameSpeed = INITIAL_SPEED;
        let gameLoop = null;
        
        // 初始化高分显示
        highScoreElement.textContent = highScore;
        
        // 初始化游戏
        function initGame() {
            // 初始化蛇的位置
            snake = [
                { x: 10, y: 10 },
                { x: 9, y: 10 },
                { x: 8, y: 10 }
            ];
            
            // 初始方向
            direction = { x: 1, y: 0 };
            
            // 生成食物
            generateFood();
            
            // 重置分数
            score = 0;
            scoreElement.textContent = score;
            
            // 重置游戏速度
            gameSpeed = INITIAL_SPEED;
            
            // 更新按钮状态
            updateButtons();
            
            // 隐藏游戏结束界面
            gameOverOverlay.style.display = 'none';
            pauseIndicator.style.display = 'none';
        }
        
        // 生成食物
        function generateFood() {
            do {
                food = {
                    x: Math.floor(Math.random() * GRID_SIZE),
                    y: Math.floor(Math.random() * GRID_SIZE)
                };
            } while (snake.some(segment => segment.x === food.x && segment.y === food.y));
        }
        
        // 绘制游戏
        function draw() {
            // 清空画布
            ctx.fillStyle = '#000';
            ctx.fillRect(0, 0, canvas.width, canvas.height);
            
            // 绘制网格线
            ctx.strokeStyle = '#111';
            ctx.lineWidth = 0.5;
            
            for (let i = 0; i <= GRID_SIZE; i++) {
                ctx.beginPath();
                ctx.moveTo(i * CELL_SIZE, 0);
                ctx.lineTo(i * CELL_SIZE, canvas.height);
                ctx.stroke();
                
                ctx.beginPath();
                ctx.moveTo(0, i * CELL_SIZE);
                ctx.lineTo(canvas.width, i * CELL_SIZE);
                ctx.stroke();
            }
            
            // 绘制蛇
            snake.forEach((segment, index) => {
                if (index === 0) {
                    // 蛇头
                    ctx.fillStyle = '#0f0';
                    ctx.shadowColor = '#0f0';
                    ctx.shadowBlur = 10;
                } else {
                    // 蛇身
                    ctx.fillStyle = '#0a0';
                    ctx.shadowBlur = 0;
                }
                
                ctx.fillRect(
                    segment.x * CELL_SIZE + 1,
                    segment.y * CELL_SIZE + 1,
                    CELL_SIZE - 2,
                    CELL_SIZE - 2
                );
            });
            
            // 绘制食物
            ctx.fillStyle = '#f00';
            ctx.shadowColor = '#f00';
            ctx.shadowBlur = 10;
            ctx.beginPath();
            ctx.arc(
                food.x * CELL_SIZE + CELL_SIZE / 2,
                food.y * CELL_SIZE + CELL_SIZE / 2,
                CELL_SIZE / 2 - 2,
                0,
                Math.PI * 2
            );
            ctx.fill();
            ctx.shadowBlur = 0;
        }
        
        // 更新游戏状态
        function update() {
            // 计算蛇头的新位置
            const head = { ...snake[0] };
            head.x += direction.x;
            head.y += direction.y;
            
            // 检查是否撞墙
            if (head.x < 0 || head.x >= GRID_SIZE || head.y < 0 || head.y >= GRID_SIZE) {
                gameOver();
                return;
            }
            
            // 检查是否撞到自己
            if (snake.some(segment => segment.x === head.x && segment.y === head.y)) {
                gameOver();
                return;
            }
            
            // 添加新的蛇头
            snake.unshift(head);
            
            // 检查是否吃到食物
            if (head.x === food.x && head.y === food.y) {
                // 增加分数
                score += 10;
                scoreElement.textContent = score;
                
                // 更新最高分
                if (score > highScore) {
                    highScore = score;
                    highScoreElement.textContent = highScore;
                    localStorage.setItem('snakeHighScore', highScore);
                }
                
                // 生成新食物
                generateFood();
                
                // 增加游戏速度
                if (gameSpeed > 50) {
                    gameSpeed -= SPEED_INCREMENT;
                    clearInterval(gameLoop);
                    gameLoop = setInterval(gameStep, gameSpeed);
                }
            } else {
                // 如果没有吃到食物，移除蛇尾
                snake.pop();
            }
        }
        
        // 游戏步骤
        function gameStep() {
            if (gameState === GameState.PLAYING) {
                update();
                draw();
            }
        }
        
        // 开始游戏
        function startGame() {
            if (gameState === GameState.IDLE || gameState === GameState.GAME_OVER) {
                initGame();
                gameState = GameState.PLAYING;
                updateButtons();
                
                // 开始游戏循环
                gameLoop = setInterval(gameStep, gameSpeed);
                draw();
            }
        }
        
        // 暂停游戏
        function pauseGame() {
            if (gameState === GameState.PLAYING) {
                gameState = GameState.PAUSED;
                pauseIndicator.style.display = 'block';
                updateButtons();
            } else if (gameState === GameState.PAUSED) {
                gameState = GameState.PLAYING;
                pauseIndicator.style.display = 'none';
                updateButtons();
            }
        }
        
        // 重置游戏
        function resetGame() {
            // 清除游戏循环
            if (gameLoop) {
                clearInterval(gameLoop);
                gameLoop = null;
            }
            
            // 重置游戏状态
            gameState = GameState.IDLE;
            
            // 初始化游戏
            initGame();
            
            // 绘制初始状态
            draw();
        }
        
        // 游戏结束
        function gameOver() {
            // 清除游戏循环
            if (gameLoop) {
                clearInterval(gameLoop);
                gameLoop = null;
            }
            
            // 更新游戏状态
            gameState = GameState.GAME_OVER;
            
            // 显示游戏结束界面
            finalScoreElement.textContent = score;
            gameOverOverlay.style.display = 'flex';
            
            // 更新按钮状态
            updateButtons();
        }
        
        // 更新按钮状态
        function updateButtons() {
            switch (gameState) {
                case GameState.IDLE:
                    startBtn.textContent = '开始游戏';
                    startBtn.disabled = false;
                    pauseBtn.disabled = true;
                    pauseBtn.textContent = '暂停';
                    break;
                case GameState.PLAYING:
                    startBtn.disabled = true;
                    pauseBtn.disabled = false;
                    pauseBtn.textContent = '暂停';
                    break;
                case GameState.PAUSED:
                    startBtn.disabled = true;
                    pauseBtn.disabled = false;
                    pauseBtn.textContent = '继续';
                    break;
                case GameState.GAME_OVER:
                    startBtn.textContent = '重新开始';
                    startBtn.disabled = false;
                    pauseBtn.disabled = true;
                    pauseBtn.textContent = '暂停';
                    break;
            }
        }
        
        // 键盘控制
        document.addEventListener('keydown', (e) => {
            if (gameState !== GameState.PLAYING) return;
            
            switch (e.key) {
                case 'ArrowUp':
                    if (direction.y === 0) {
                        direction = { x: 0, y: -1 };
                    }
                    break;
                case 'ArrowDown':
                    if (direction.y === 0) {
                        direction = { x: 0, y: 1 };
                    }
                    break;
                case 'ArrowLeft':
                    if (direction.x === 0) {
                        direction = { x: -1, y: 0 };
                    }
                    break;
                case 'ArrowRight':
                    if (direction.x === 0) {
                        direction = { x: 1, y: 0 };
                    }
                    break;
            }
        });
        
        // 按钮事件监听
        startBtn.addEventListener('click', startGame);
        pauseBtn.addEventListener('click', pauseGame);
        resetBtn.addEventListener('click', resetGame);
        restartBtn.addEventListener('click', startGame);
        
        // 初始化游戏
        initGame();
        draw();
    </script>
</body>
</html>

## 游戏状态管理实现详解

### 1. 状态定义

我使用了枚举方式定义了四种游戏状态：
- `IDLE`: 游戏未开始状态
- `PLAYING`: 游戏进行中状态
- `PAUSED`: 游戏暂停状态
- `GAME_OVER`: 游戏结束状态

### 2. 状态管理核心功能

#### 状态转换逻辑
- **开始游戏**: 从 `IDLE` 或 `GAME_OVER` 状态转换为 `PLAYING` 状态
- **暂停/继续**: 在 `PLAYING` 和 `PAUSED` 状态之间切换
- **游戏结束**: 从 `PLAYING` 状态转换为 `GAME_OVER` 状态
- **重置游戏**: 任何时候都可以重置为 `IDLE` 状态

#### 状态控制函数
- `startGame()`: 处理游戏开始逻辑
- `pauseGame()`: 处理游戏暂停/继续逻辑
- `resetGame()`: 处理游戏重置逻辑
- `gameOver()`: 处理游戏结束逻辑

### 3. 状态可视化反馈

为了增强用户体验，我添加了多种状态反馈：
- 按钮状态根据游戏状态动态变化
- 暂停时显示"暂停"提示
- 游戏结束时弹出游戏结束界面
- 分数和最高分实时更新

### 4. 游戏流程控制

游戏流程通过状态机模式管理：
1. 初始状态为 `IDLE`，显示"开始游戏"按钮
2. 点击"开始游戏"后进入 `PLAYING` 状态，游戏开始运行
3. 游戏中可以点击"暂停"按钮进入 `PAUSED` 状态
4. 在 `PAUSED` 状态可以点击"继续"按钮回到 `PLAYING` 状态
5. 游戏碰撞检测失败后进入 `GAME_OVER` 状态
6. 任何时候都可以点击"重置"按钮回到 `IDLE` 状态

### 5. 附加功能

- **本地存储最高分**: 使用 localStorage 保存最高分
- **动态游戏速度**: 随着得分增加，游戏速度逐渐加快
- **视觉反馈**: 添加了发光效果和阴影增强游戏体验

这个实现确保了游戏状态的正确转换和管理，提供了流畅的游戏体验和直观的用户界面。
