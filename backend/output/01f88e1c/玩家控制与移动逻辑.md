# 玩家控制与移动逻辑

```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>Snake Game</title>
<style>
  body {
    margin: 0;
    overflow: hidden;
    display: flex;
    justify-content: center;
    align-items: center;
    height: 100vh;
    background-color: #f0f0f0;
  }
  canvas {
    border: 1px solid black;
  }
</style>
</head>
<body>
<canvas id="gameCanvas"></canvas>
<script>
  const canvas = document.getElementById('gameCanvas');
  const ctx = canvas.getContext('2d');

  canvas.width = 400;
  canvas.height = 400;

  let snake = [{ x: 9 * 20, y: 10 * 20 }];
  let food = {
    x: Math.floor(Math.random() * 20) * 20,
    y: Math.floor(Math.random() * 20) * 20
  };
  let score = 0;
  let d;
  let gameInterval;
  let speed = 100;

  window.onload = () => {
    document.addEventListener('keydown', direction);
    gameInterval = setInterval(gameLoop, speed);
  };

  function direction(event) {
    const key = event.keyCode;
    const left = 37, up = 38, right = 39, down = 40;

    if ((key === left || key === right) && (d === undefined || (d !== left && d !== right))) {
      d = key === left ? 'LEFT' : 'RIGHT';
    } else if ((key === up || key === down) && (d === undefined || (d !== up && d !== down))) {
      d = key === up ? 'UP' : 'DOWN';
    }
  }

  function gameLoop() {
    if (checkCollision()) {
      clearInterval(gameInterval);
      alert('Game Over');
      return;
    }

    ctx.fillStyle = 'white';
    ctx.fillRect(0, 0, canvas.width, canvas.height);

    ctx.fillStyle = 'red';
    ctx.fillRect(food.x, food.y, 20, 20);

    let nx = snake[0].x;
    let ny = snake[0].y;

    if (d === 'LEFT') nx -= 20;
    if (d === 'UP') ny -= 20;
    if (d === 'RIGHT') nx += 20;
    if (d === 'DOWN') ny += 20;

    let tail = [...snake];
    tail.pop(); // remove the tail
    tail.unshift({ x: nx, y: ny }); // add the new head
    snake = tail;

    if (nx === food.x && ny === food.y) {
      score++;
      food = {
        x: Math.floor(Math.random() * 20) * 20,
        y: Math.floor(Math.random() * 20) * 20
      };
      speed = Math.min(10, speed - 5); // Increase speed, up to a maximum of 10
      clearInterval(gameInterval);
      gameInterval = setInterval(gameLoop, speed);
    }

    ctx.fillStyle = 'green';
    for (let i = 0; i < snake.length; i++) {
      ctx.fillRect(snake[i].x, snake[i].y, 20, 20);
    }

    ctx.fillStyle = 'black';
    ctx.fillText('Score: ' + score, 5, canvas.height - 5);
  }

  function checkCollision() {
    for (let i = 1; i < snake.length; i++) {
      if (snake[0].x === snake[i].x && snake[0].y === snake[i].y) {
        return true;
      }
    }
    return false;
  }
</script>
</body>
</html>
```

This modified HTML code now includes the following changes:
- Added a `gameInterval` variable to control the game loop timing.
- Added a `speed` variable to control the speed of the snake, which increases when the snake eats food up to a maximum of 10.
- The game loop now resets the interval using `setInterval` with the updated speed after each food consumption.
- The direction change logic has been refined to prevent immediate opposite direction changes, which was causing an immediate turn around in the original code.
