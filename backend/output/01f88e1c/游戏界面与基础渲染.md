# 游戏界面与基础渲染

**Time**: 2026-02-26T20:35:38.156120

---

```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>Snake Game</title>
<style>
  body {
    margin: 0;
    overflow: hidden;
    display: flex;
    justify-content: center;
    align-items: center;
    height: 100vh;
    background-color: #f0f0f0;
  }
  canvas {
    border: 1px solid black;
  }
</style>
</head>
<body>
<canvas id="gameCanvas"></canvas>
<script>
  const canvas = document.getElementById('gameCanvas');
  const ctx = canvas.getContext('2d');

  canvas.width = 400;
  canvas.height = 400;

  let snake = [{ x: 9 * 20, y: 10 * 20 }];
  let food = {
    x: Math.floor(Math.random() * 20) * 20,
    y: Math.floor(Math.random() * 20) * 20
  };
  let score = 0;
  let d;

  window.onload = () => {
    document.addEventListener('keydown', direction);
    requestAnimationFrame(gameLoop);
  };

  function direction(event) {
    if (event.keyCode === 37 && d !== 'RIGHT') {
      d = 'LEFT';
    } else if (event.keyCode === 38 && d !== 'DOWN') {
      d = 'UP';
    } else if (event.keyCode === 39 && d !== 'LEFT') {
      d = 'RIGHT';
    } else if (event.keyCode === 40 && d !== 'UP') {
      d = 'DOWN';
    }
  }

  function gameLoop() {
    if (checkCollision()) {
      alert('Game Over');
      return;
    }

    ctx.fillStyle = 'white';
    ctx.fillRect(0, 0, canvas.width, canvas.height);

    ctx.fillStyle = 'red';
    ctx.fillRect(food.x, food.y, 20, 20);

    let nx = snake[0].x;
    let ny = snake[0].y;

    if (d === 'LEFT') nx -= 20;
    if (d === 'UP') ny -= 20;
    if (d === 'RIGHT') nx += 20;
    if (d === 'DOWN') ny += 20;

    let tail = [...snake];
    tail.pop(); // remove the tail
    tail.unshift({ x: nx, y: ny }); // add the new head
    snake = tail;

    if (nx === food.x && ny === food.y) {
      score++;
      food = {
        x: Math.floor(Math.random() * 20) * 20,
        y: Math.floor(Math.random() * 20) * 20
      };
    }

    ctx.fillStyle = 'green';
    for (let i = 0; i < snake.length; i++) {
      ctx.fillRect(snake[i].x, snake[i].y, 20, 20);
    }

    ctx.fillStyle = 'black';
    ctx.fillText('Score: ' + score, 5, canvas.height - 5);
    requestAnimationFrame(gameLoop);
  }

  function checkCollision() {
    for (let i = 1; i < snake.length; i++) {
      if (snake[0].x === snake[i].x && snake[0].y === snake[i].y) {
        return true;
      }
    }
    return false;
  }
</script>
</body>
</html>
```

这段代码创建了一个简单的贪吃蛇游戏，使用了Canvas API进行渲染。游戏界面包括蛇、食物和分数计数。游戏循环由`requestAnimationFrame`控制，键盘事件用于控制蛇的方向。游戏会在蛇撞到自己时结束。
