// 游戏配置
const config = {
    type: Phaser.AUTO,
    width: 800,
    height: 600,

        default: 'arcade',

            gravity: { y: 300 },
            debug: false

    scene: [BootScene, LoadingScene, MainMenuScene, GameScene, LevelTransitionScene, GameOverScene],
    backgroundColor: '#5C94FC'
};

// 创建游戏实例
const game = new Phaser.Game(config);

// 性能监控变量
let fps = 0;
let frameCount = 0;
let lastTime = performance.now();
let memoryUsage = 0;

const GameState = {

// Boot场景 - 初始化游戏
class BootScene extends Phaser.Scene {
    constructor() {
        super({ key: 'BootScene' });
    }

    preload() {
        // 加载加载界面资源
        this.load.image('loading-bg', 'assets/loading-bg.png');
        this.load.image('loading-bar', 'assets/loading-bar.png');
    }

    create() {
        // 初始化输入管理
        this.cursors = this.input.keyboard.createCursorKeys();
        this.keys = {
            jump: this.input.keyboard.addKey(Phaser.Input.Keyboard.KeyCodes.SPACE),
            run: this.input.keyboard.addKey(Phaser.Input.Keyboard.KeyCodes.SHIFT)
        };
        
        // 初始化游戏状态
        gameState = GameState.MENU;
        
        // 转到加载场景
        this.scene.start('LoadingScene');
    }
}

// 加载场景
class LoadingScene extends Phaser.Scene {
    constructor() {
        super({ key: 'LoadingScene' });
    }

    preload() {
        // 显示加载背景
        this.add.image(400, 300, 'loading-bg');
        
        // 创建加载进度条
        this.loadingBar = this.add.graphics();
        this.loadingBar.fillStyle(0xffffff, 1);
        this.loadingBar.fillRect(200, 500, 400, 20);
        
        // 加载游戏资源
        this.load.on('progress', this.loadProgress, this);
        this.load.on('complete', this.loadComplete, this);
        this.load.on('loaderror', this.loadError, this);
        
        // 加载精灵图集
        this.load.atlas('mario', 'assets/mario.png', 'assets/mario.json');
        this.load.atlas('enemies', 'assets/enemies.png', 'assets/enemies.json');
        this.load.atlas('items', 'assets/items.png', 'assets/items.json');
        
        // 加载地图
        this.tilemapData = 'assets/level' + currentLevel + '.json';
        this.load.tilemapTiledJSON('level' + currentLevel, this.tilemapData);
        this.load.image('tileset', 'assets/tileset.png');
        
        // 加载音效
        this.load.audio('jump', 'assets/sounds/jump.mp3');
        this.load.audio('coin', 'assets/sounds/coin.mp3');
        this.load.audio('die', 'assets/sounds/die.mp3');
        this.load.audio('powerup', 'assets/sounds/powerup.mp3');
        this.load.audio('bgm', 'assets/sounds/bgm.mp3');
    }

    loadProgress(progress) {
        // 更新进度条
        this.loadingBar.clear();
        this.loadingBar.fillStyle(0xffffff, 1);
        this.loadingBar.fillRect(200, 500, 400 * progress, 20);
        
        // 显示加载百分比
        if (!this.loadingText) {
            this.loadingText = this.add.text(400, 450, '加载中...', {
                fontSize: '24px',
                fill: '#ffffff'
            }).setOrigin(0.5);
        }
        this.loadingText.setText(`加载中... ${Math.round(progress * 100)}%`);
    }

    loadComplete() {
        console.log('所有资源加载完成');
        this.scene.start('MainMenuScene');
    }

    loadError(file) {
        console.error(`资源加载失败: ${file.key}`);
        // 处理加载错误，显示错误信息
        if (!this.errorText) {
            this.errorText = this.add.text(400, 450, '资源加载失败，请刷新页面重试', {
                fontSize: '24px',
                fill: '#ff0000'
            }).setOrigin(0.5);
        }
    }
}

// 游戏场景
class GameScene extends Phaser.Scene {
    constructor() {
        super({ key: 'GameScene' });
    }

    init() {
        // 初始化游戏状态
        gameState = GameState.PLAYING;
        playerScore = 0;
        playerLives = 3;
        
        // 初始化对象池
        this.enemyPool = new EnemyPool(this);
        this.coinPool = new CoinPool(this);
        this.powerUpPool = new PowerUpPool(this);
        
        // 初始化关卡数据
        this.levelData = {
            coins: 0,
            totalCoins: 0,
            enemies: 0,
            time: 300, // 5分钟
            star: false
        };
    }

    preload() {
        // 资源已在加载场景中加载，这里不需要重复加载
    }

    create() {
        // 创建地图
        this.map = this.make.tilemap({ key: 'level' + currentLevel });
        const tileset = this.map.addTilesetImage('tileset', 'tileset');
        
        // 创建图层
        this.groundLayer = this.map.createLayer('ground', tileset, 0, 0);
        this.decorationLayer = this.map.createLayer('decoration', tileset, 0, 0);
        
        // 设置碰撞
        this.groundLayer.setCollisionByExclusion([-1]);
        
        // 创建玩家
        this.player = new Player(this, 100, 300, 'mario');
        this.physics.add.collider(this.player, this.groundLayer);
        
        // 创建相机
        this.cameras.main.setBounds(0, 0, this.map.widthInPixels, this.map.heightInPixels);
        this.cameras.main.startFollow(this.player);
        this.cameras.main.setDeadzone(200, 0);
        
        // 创建UI
        this.createUI();
        
        // 创建收集品
        this.createCollectibles();
        
        // 创建敌人
        this.createEnemies();
        
        // 创建终点旗帜
        this.createFlag();
        
        // 添加音效
        this.sounds = {
            jump: this.sound.add('jump'),
            coin: this.sound.add('coin'),
            die: this.sound.add('die'),
            powerup: this.sound.add('powerup')
        };
        
        // 添加背景音乐
        this.bgm = this.sound.add('bgm', { loop: true, volume: 0.5 });
        this.bgm.play();
        
        // 添加计时器
        this.timeText = this.add.text(700, 20, `时间: ${this.formatTime(this.levelData.time)}`, {
            fontSize: '20px',
            fill: '#ffffff'
        });
        
        // 添加事件监听
        this.events.on('playerDeath', this.onPlayerDeath, this);
        this.events.on('coinCollected', this.onCoinCollected, this);
        this.events.on('powerUpCollected', this.onPowerUpCollected, this);
        this.events.on('levelComplete', this.onLevelComplete, this);
        
        // 开始计时器
        this.timerEvent = this.time.addEvent({
            delay: 1000,
            callback: this.updateTimer,
            callbackScope: this,
            loop: true
        });
        
        // 初始化性能监控
        this.initPerformanceMonitor();
    }
    
    initPerformanceMonitor() {
        // 更新对象计数
        this.time.addEvent({
            delay: 1000,
            callback: function() {
                const objectCount = this.children.list.length;
                this.objectCountText.setText(`对象数: ${objectCount}`);
                
                // 更新全局对象计数
                window.objectCount = objectCount;
            },
            loop: true
        });
    }
    
    formatTime(seconds) {
        const minutes = Math.floor(seconds / 60);
        const remainingSeconds = seconds % 60;
        return `${minutes}:${remainingSeconds < 10 ? '0' : ''}${remainingSeconds}`;
    }
    
    updateTimer() {
        this.levelData.time--;
        this.timeText.setText(`时间: ${this.formatTime(this.levelData.time)}`);
        
        if (this.levelData.time <= 0) {
            this.events.emit('playerDeath');
        }
    }
    
    createUI() {
        // 创建分数显示
        this.scoreText = this.add.text(20, 20, `分数: ${playerScore}`, {
            fontSize: '24px',
            fill: '#ffffff'
        });
        
        // 创建生命值显示
        this.livesText = this.add.text(20, 60, `生命: ${playerLives}`, {
            fontSize: '24px',
            fill: '#ffffff'
        });
        
        // 创建关卡显示
        this.levelText = this.add.text(20, 100, `关卡: ${currentLevel}`, {
            fontSize: '24px',
            fill: '#ffffff'
        });
    }
    
    createCollectibles() {
        // 创建金币
        const coinObjects = this.map.getObjectLayer('coins').objects;
        this.levelData.totalCoins = coinObjects.length;
        
        coinObjects.forEach(coin => {
            this.coinPool.get(coin.x, coin.y);
        });
        
        // 创建蘑菇
        const powerUpObjects = this.map.getObjectLayer('powerups').objects;
        powerUpObjects.forEach(powerUp => {
            this.powerUpPool.get(powerUp.x, powerUp.y, powerUp.type);
        });
    }
    
    createEnemies() {
        // 创建敌人
        const enemyObjects = this.map.getObjectLayer('enemies').objects;
        this.levelData.enemies = enemyObjects.length;
        
        enemyObjects.forEach(enemy => {
            this.enemyPool.get(enemy.x, enemy.y, enemy.type);
        });
    }
    
    createFlag() {
        // 创建终点旗帜
        const flagObject = this.map.getObjectLayer('flag').objects[0];
        this.flag = new Flag(this, flagObject.x, flagObject.y, 'mario', 'flag');
        this.physics.add.collider(this.flag, this.groundLayer);
    }
    
    update() {
        // 更新玩家
        this.player.update();
        
        // 更新敌人
        this.enemyPool.getAll().forEach(enemy => {
            enemy.update();
        });
        
        // 更新收集品
        this.coinPool.getAll().forEach(coin => {
            coin.update();
        });
        
        this.powerUpPool.getAll().forEach(powerUp => {
            powerUp.update();
        });
        
        // 检查碰撞
        this.checkCollisions();
        
        // 更新性能监控
        this.updatePerformanceMonitor();
    }
    
    checkCollisions() {
        // 玩家与金币碰撞
        this.physics.add.overlap(this.player, this.coinPool.getAll(), (player, coin) => {
            if (coin.active) {
                coin.collect();
                this.events.emit('coinCollected', coin.value);
            }
        });
        
        // 玩家与蘑菇碰撞
        this.physics.add.overlap(this.player, this.powerUpPool.getAll(), (player, powerUp) => {
            if (powerUp.active && !this.player.isPoweredUp) {
                powerUp.collect();
                this.events.emit('powerUpCollected', powerUp.type);
            }
        });
        
        // 玩家与敌人碰撞
        this.physics.add.overlap(this.player, this.enemyPool.getAll(), (player, enemy) => {
            if (enemy.active) {
                // 如果玩家从上方跳到敌人头上
                if (player.body.velocity.y > 0 && player.y < enemy.y) {
                    enemy.stomp();
                    this.player.jump();
                } else {
                    // 玩家受伤或死亡
                    this.player.hit();
                }
            }
        });
        
        // 玩家与旗帜碰撞
        this.physics.add.overlap(this.player, this.flag, () => {
            if (this.flag.active) {
                this.flag.trigger();
                this.events.emit('levelComplete');
            }
        });
    }
    
    updatePerformanceMonitor() {
        // 计算FPS
        frameCount++;
        const currentTime = performance.now();
        if (currentTime >= lastTime + 1000) {
            fps = Math.round((frameCount * 1000) / (currentTime - lastTime));
            frameCount = 0;
            lastTime = currentTime;
            
            // 更新FPS显示
            document.getElementById('fps').textContent = fps;
            
            // 获取内存使用情况（如果浏览器支持）
            if (performance.memory) {
                memoryUsage = Math.round(performance.memory.usedJSHeapSize / 1048576);
                document
