class BootScene extends Phaser.Scene {
    constructor() {
        super({ key: 'BootScene' });
    }

    preload() {
        // 加载启动画面资源
        this.load.image('boot-screen', 'assets/images/boot-screen.png');
    }

    create() {
        // 显示启动画面
        this.add.image(400, 300, 'boot-screen');
        
        // 添加错误处理
        this.load.on('loaderror', (file) => {
            console.error(`资源加载失败: ${file.key}`);
            // 使用默认资源或显示错误信息
        });
        
        // 初始化完成后切换到预加载场景
        this.time.delayedCall(1000, () => {
            this.scene.start('PreloadScene');
        });
    }
}

class PreloadScene extends Phaser.Scene {
    constructor() {
        super({ key: 'PreloadScene' });
    }

    preload() {
        // 显示加载进度条
        let loadingText = this.add.text(400, 300, '加载中...', { 
            fontSize: '24px', 
            fill: '#ffffff' 
        });
        this.load.on('progress', function (progress) {
            loadingText.setText(`加载中... ${Math.round(progress * 100)}%`);
        });

        // 加载精灵图集
        this.load.spritesheet('mario', 'assets/images/mario.png', {
            frameWidth: 32,
            frameHeight: 32
        });
        
        this.load.spritesheet('enemies', 'assets/images/enemies.png', {
            frameWidth: 32,
            frameHeight: 32
        });
        
        // 加载地图瓦片
        this.load.tilemapTiledJSON('map', 'assets/maps/level1.json');
        this.load.image('tiles', 'assets/images/tiles.png');
        
        // 加载音效
        this.load.audio('jump', 'assets/sounds/jump.wav');
        this.load.audio('coin', 'assets/sounds/coin.wav');
        this.load.audio('die', 'assets/sounds/die.wav');
    }

    create() {
        // 资源加载完成后的处理
        this.load.on('filecomplete', function (key, type, data) {
            console.log(`资源加载完成: ${key}`);
        });
        
        // 所有资源加载完成后切换到游戏场景
        this.scene.start('GameScene');
    }
}

## 游戏主场景 (GameScene.js)

```javascript
class GameScene extends Phaser.Scene {
    constructor() {
        super({ key: 'GameScene' });
        this.player = null;
        this.cursors = null;
        this.platforms = null;
        this.enemies = null;
        this.coins = null;
        this.score = 0;
        this.scoreText = null;
        this.fpsText = null;
        this.gameOver = false;
        this.enemyPool = null;
        this.platformPool = null;
    }

    preload() {
        // 预加载资源已在PreloadScene中完成
    }

    create() {
        // 初始化性能监控
        this.initPerformanceMonitoring();
        
        // 创建地图
        this.createMap();
        
        // 创建玩家
        this.createPlayer();
        
        // 创建敌人和平台
        this.createEnemies();
        this.createPlatforms();
        
        // 创建收集品
        this.createCoins();
        
        // 创建UI
        this.createUI();
        
        // 初始化碰撞检测
        this.initCollisions();
        
        // 添加输入处理
        this.initInputHandling();
        
        // 添加游戏状态处理
        this.initGameStateHandling();
    }

    update(time, delta) {
        // 更新性能监控
        this.updatePerformanceMetrics();
        
        // 游戏主循环
        if (!this.gameOver) {
            // 处理玩家输入
            this.handlePlayerInput();
            
            // 更新玩家状态
            this.updatePlayer();
            
            // 更新敌人
            this.updateEnemies();
            
            // 检查碰撞
            this.checkCollisions();
            
            // 检查游戏结束条件
            this.checkGameOverConditions();
        }
    }

    initPerformanceMonitoring() {
        // 创建FPS显示
        this.fpsText = this.add.text(10, 10, 'FPS: 0', { 
            fontSize: '16px', 
            fill: '#ffffff' 
        });
        
        // 创建性能指标显示
        this.metricsText = this.add.text(10, 30, '', { 
            fontSize: '14px', 
            fill: '#ffffff' 
        });
        
        // 性能监控计时器
        this.time.addEvent({
            delay: 1000,
            callback: () => {
                this.fpsText.setText(`FPS: ${Math.round(game.loop.actualFps)}`);
                this.metricsText.setText(
                    `内存: ${performance.memory ? Math.round(performance.memory.usedJSHeapSize / 1048576) + 'MB' : 'N/A'} | ` +
                    `敌人: ${this.enemies ? this.enemies.getChildren().length : 0} | ` +
                    `平台: ${this.platforms ? this.platforms.getChildren().length : 0}`
                );
            },
            loop: true
        });
    }

    updatePerformanceMetrics() {
        // 更新性能指标
        performanceMetrics.fps = Math.round(game.loop.actualFps);
        performanceMetrics.enemies = this.enemies ? this.enemies.getChildren().length : 0;
        performanceMetrics.platforms = this.platforms ? this.platforms.getChildren().length : 0;
        
        if (performance.memory) {
            performanceMetrics.memory = Math.round(performance.memory.usedJSHeapSize / 1048576);
        }
    }

    createMap() {
        // 创建地图
        const map = this.make.tilemap({ key: 'map' });
        const tileset = map.addTilesetImage('tileset', 'tiles');
        
        // 创建图层
        this.groundLayer = map.createLayer('Ground', tileset, 0, 0);
        this.decorationLayer = map.createLayer('Decoration', tileset, 0, 0);
        
        // 设置碰撞区域
        this.groundLayer.setCollisionByProperty({ collides: true });
        
        // 添加物理边界
        this.physics.world.setBounds(0, 0, map.widthInPixels, map.heightInPixels);
    }

    createPlayer() {
        // 创建玩家精灵
        this.player = this.physics.add.sprite(100, 450, 'mario');
        
        // 设置玩家物理属性
        this.player.setBounce(0.2);
        this.player.setCollideWorldBounds(true);
        this.player.body.setGravityY(1000);
        
        // 创建玩家动画
        this.anims.create({
            key: 'left',
            frames: this.anims.generateFrameNumbers('mario', { start: 0, end: 3 }),
            frameRate: 10,
            repeat: -1
        });
        
        this.anims.create({
            key: 'turn',
            frames: [{ key: 'mario', frame: 4 }],
            frameRate: 20
        });
        
        this.anims.create({
            key: 'right',
            frames: this.anims.generateFrameNumbers('mario', { start: 5, end: 8 }),
            frameRate: 10,
            repeat: -1
        });
        
        this.anims.create({
            key: 'idle',
            frames: [{ key: 'mario', frame: 0 }],
            frameRate: 10
        });
        
        this.anims.create({
            key: 'jump',
            frames: [{ key: 'mario', frame: 9 }],
            frameRate: 10
        });
        
        // 设置初始动画
        this.player.anims.play('idle');
    }

    createEnemies() {
        // 创建敌人群组
        this.enemies = this.physics.add.group();
        
        // 创建对象池
        this.enemyPool = this.physics.add.group({
            classType: Enemy,
            maxSize: 50,
            runChildUpdate: true
        });
        
        // 初始化一些敌人
        for (let i = 0; i < 5; i++) {
            const enemy = this.enemyPool.get(200 + i * 150, 400, 'goomba');
            if (enemy) {
                this.enemies.add(enemy);
            }
        }
    }

    createPlatforms() {
        // 创建平台群组
        this.platforms = this.physics.add.staticGroup();
        
        // 创建对象池
        this.platformPool = this.physics.add.staticGroup({
            classType: Platform,
            maxSize: 20
        });
        
        // 初始化一些平台
        for (let i = 0; i < 3; i++) {
            const platform = this.platformPool.get(300 + i * 200, 300, 'moving');
            if (platform) {
                this.platforms.add(platform);
            }
        }
    }

    createCoins() {
        // 创建金币群组
        this.coins = this.physics.add.group({
            classType: Coin,
            maxSize: 30,
            runChildUpdate: true
        });
        
        // 初始化一些金币
        for (let i = 0; i < 10; i++) {
            const coin = this.coins.get(150 + i * 100, 200, 'coin');
            if (coin) {
                coin.setActive(true);
                coin.setVisible(true);
            }
        }
    }

    createUI() {
        // 创建分数显示
        this.scoreText = this.add.text(16, 16, '分数: 0', { 
            fontSize: '24px', 
            fill: '#fff' 
        });
        
        // 创建生命值显示
        this.livesText = this.add.text(16, 48, '生命: 3', { 
            fontSize: '24px', 
            fill: '#fff' 
        });
    }

    initCollisions() {
        // 玩家与地面碰撞
        this.physics.add.collider(this.player, this.groundLayer);
        
        // 玩家与平台碰撞
        this.physics.add.collider(this.player, this.platforms);
        
        // 玩家与敌人碰撞
        this.physics.add.collider(this.player, this.enemies, this.playerEnemyCollision, null, this);
        
        // 玩家与金币碰撞
        this.physics.add.overlap(this.player, this.coins, this.playerCoinCollision, null, this);
        
        // 敌人与地面碰撞
        this.physics.add.collider(this.enemies, this.groundLayer);
        
        // 敌人与平台碰撞
        this.physics.add.collider(this.enemies, this.platforms);
    }

    initInputHandling() {
        // 创建键盘输入
        this.cursors = this.input.keyboard.createCursorKeys();
        this.spaceBar = this.input.keyboard.addKey(Phaser.Input.Keyboard.KeyCodes.SPACE);
        
        // 添加输入错误处理
        this.input.keyboard.on('keydown', (event) => {
            if (event.keyCode === Phaser.Input.Keyboard.KeyCodes.UNKNOWN) {
                console.warn('未知按键被按下');
            }
        });
    }

    initGameStateHandling() {
        // 监听游戏状态变化
        this.events.on('gameover', () => {
            this.gameOver = true;
            this.showGameOverScreen();
        });
        
        this.events.on('restart', () => {
            this.restartGame();
        });
    }

    handlePlayerInput() {
        // 处理左右移动
        if (this.cursors.left.isDown) {
            this.player.setVelocityX(-200);
            this.player.anims.play('left', true);
        } else if (this.cursors.right.isDown) {
            this.player.setVelocityX(200);
            this.player.anims.play('right', true);
        } else {
            // 静止状态
            this.player.setVelocityX(0);
            this.player.anims.play('idle');
        }
        
        // 处理跳跃
        if (this.cursors.up.isDown && this.player.body.touching.down) {
            this.player.setVelocityY(-500);
            this.player.anims.play('jump');
            this.sound.play('jump');
        }
    }

    updatePlayer() {
        // 更新玩家动画状态
        if (this.player.body.velocity.y < 0) {
            this.player.anims.play('jump');
        } else if (this.player.body.velocity.y > 0) {
            this.player.anims.play('idle');
        }
        
        // 检查玩家是否掉出地图
        if (this.player.y > this.sys.canvas.height) {
            this.playerDie();
        }
    }

    updateEnemies() {
        // 更新所有敌人
        this.enemies.children.entries.forEach(enemy => {
            enemy.update();
        });
    }

    checkCollisions() {
        // 检查玩家与敌人的碰撞
        this.physics.overlap(this.player, this.enemies, (player, enemy) => {
            if (player.body.velocity.y > 0 && enemy.body.touching.up) {
                // 踩踏敌人
                player.setVelocityY(-300);
                enemy.destroy();
                this.score += 100;
                this.scoreText.setText(`分数: ${this.score}`);
                this.sound.play('coin');
            } else {
                // 玩家受伤
                this.playerHit();
            }
        });
        
        // 检查玩家与金币的碰撞
        this.physics.overlap(this.player, this.coins, (player, coin) => {
            coin.collect();
            this.score += 50;
            this.scoreText.setText(`分数: ${this.score}`);
            this.sound.play('coin');
        });
    }

    playerEnemyCollision(player, enemy) {
        // 玩家与敌人碰撞处理
        if (player.body.velocity.y > 0 && enemy.body.touching.up) {
            // 踩踏敌人
            player.setVelocityY(-300);
            enemy.destroy();
            this.score += 100;
            this.scoreText.setText(`分数: ${this.score}`);
            this.sound.play('coin');
        } else {
            // 玩家受伤
            this.playerHit();
        }
    }

    playerCoinCollision(player, coin) {
        // 玩家与金币碰撞处理
        coin.collect();
        this.score += 50;
        this.scoreText.setText(`分数: ${this.score}`);
        this.sound.play('coin');
    }

    playerHit() {
        // 玩家受伤处理
        this.sound.play('die');
        this.playerDie();
    }

    playerDie() {
        // 玩家死亡处理
        const lives = parseInt(this.livesText.getText().split(':')[1].trim());
        const newLives = lives - 1;
        
        if (newLives <= 0) {
            this.events.emit('gameover');
        } else {
            this.livesText.setText(`生命: ${newLives}`);
            this.player.setPosition(100, 450);
            this.player.setVelocity(0, 0);
            this.player.anims.play('idle');
        }
    }

    checkGameOverConditions() {
        // 检查游戏结束条件
        if (this.player.y > this.sys.canvas.height) {
            this.playerDie();
        }
    }

    showGameOverScreen() {
        // 显示游戏结束画面
        this.gameOverText = this.add.text(
            this.sys.canvas.width / 2, 
            this.sys.canvas.height / 2 - 50, 
            '游戏结束', 
            { 
                fontSize: '48px', 
                fill: '#fff' 
            }
        );
        
        this.restartText = this.add.text(
            this.sys.canvas.width / 2, 
            this.sys.canvas.height / 2 + 20, 
            '按空格键重新开始', 
            { 
                fontSize: '24px', 
                fill: '#fff' 
            }
        );
        
        this.spaceBar.on('down', () => {
            this.events.emit('restart');
        });
    }

    restartGame() {
        // 重启游戏
        this.gameOver = false;
        this.score = 0;
        this.scoreText.setText('分数: 0');
        this.livesText.setText('生命: 3');
        
        // 重置玩家位置
        this.player.setPosition(100, 450);
        this.player.setVelocity(0, 0);
        this.player.anims.play('idle');
        
        // 清除游戏结束文本
        if (this.gameOverText) this.gameOverText.destroy();
        if (this.restartText) this.restartText.destroy();
        
        // 重新生成敌人和金币
        this.enemies.clear(true);
        this.coins.clear(true);
        
        for (let i = 0; i < 5; i++) {
            const enemy = this.enemyPool.get(200 + i * 150, 400, 'goomba');
            if (enemy) {
                this.enemies.add(enemy);
            }
