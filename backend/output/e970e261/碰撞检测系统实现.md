// 游戏配置
const config = {
    type: Phaser.AUTO,
    width: 800,
    height: 600,
    backgroundColor: '#5C94FC',
    parent: 'game-container',
    physics: {
        default: 'arcade',
        arcade: {
            gravity: { y: 300 },
            debug: false
        }
    },
    scene: [PreloadScene, GameScene]
};

// 游戏全局变量
const game = new Phaser.Game(config);

// 性能监控类
class PerformanceMonitor {
    constructor(scene) {
        this.scene = scene;
        this.fps = 0;
        this.frameCount = 0;
        this.lastTime = performance.now();
        this.objectsCount = 0;
        
        // 创建性能显示元素
        this.fpsElement = document.getElementById('fps');
        this.objectsElement = document.getElementById('objects');
        this.memoryElement = document.getElementById('memory');
    }
    
    update() {
        this.frameCount++;
        const currentTime = performance.now();
        
        // 每秒更新一次FPS
        if (currentTime >= this.lastTime + 1000) {
            this.fps = Math.round((this.frameCount * 1000) / (currentTime - this.lastTime));
            this.frameCount = 0;
            this.lastTime = currentTime;
            
            // 更新显示
            this.fpsElement.textContent = this.fps;
            this.objectsElement.textContent = this.scene.children.list.length;
            
            // 更新内存使用情况（如果浏览器支持）
            if (performance.memory) {
                const memoryMB = Math.round(performance.memory.usedJSHeapSize / 1048576);
                this.memoryElement.textContent = memoryMB;
            }
        }
    }
}

// 预加载场景
class PreloadScene extends Phaser.Scene {
    constructor() {
        super({ key: 'PreloadScene' });
    }
    
    preload() {
        // 显示加载进度
        let loadingText = this.add.text(this.cameras.main.width / 2, this.cameras.main.height / 2 - 50, 'Loading...', {
            fontSize: '32px',
            fill: '#fff'
        });
        
        let progressBar = this.add.graphics();
        let progressBox = this.add.graphics();
        let width = 320;
        let height = 50;
        
        progressBox.fillStyle(0x222222, 0.8);
        progressBox.fillRect(this.cameras.main.width / 2 - width / 2, this.cameras.main.height / 2 - height / 2, width, height);
        
        // 监听加载进度
        this.load.on('progress', function (value) {
            progressBar.clear();
            progressBar.fillStyle(0x00ff00, 1);
            progressBar.fillRect(this.cameras.main.width / 2 - width / 2, this.cameras.main.height / 2 - height / 2, width * value, height);
            
            loadingText.setText(`Loading... ${Math.round(value * 100)}%`);
        }, this);
        
        // 监听加载完成
        this.load.on('complete', function () {
            loadingText.setText('Loading Complete!');
        }, this);
        
        // 加载资源
        // 玩家精灵
        this.load.spritesheet('player', 'assets/mario.png', {
            frameWidth: 32,
            frameHeight: 32
        });
        
        // 地图块
        this.load.image('tiles', 'assets/tileset.png');
        
        // 敌人精灵
        this.load.spritesheet('enemies', 'assets/enemies.png', {
            frameWidth: 32,
            frameHeight: 32
        });
        
        // 收集物
        this.load.image('coin', 'assets/coin.png');
        
        // 音效
        this.load.audio('jump', 'assets/jump.mp3');
        this.load.audio('coin', 'assets/coin.mp3');
        this.load.audio('hurt', 'assets/hurt.mp3');
        
        // 加载错误处理
        this.load.on('loaderror', function (file) {
            console.error(`Error loading resource: ${file.key}`);
            // 使用默认资源
            if (file.key === 'player') {
                this.load.image('player', 'assets/default_player.png');
            }
        }, this);
    }
    
    create() {
        // 创建地图
        this.createMap();
        
        // 创建玩家
        this.player = new Player(this, 100, 400);
        
        // 创建敌人
        this.enemies = this.physics.add.group();
        
        // 创建收集物
        this.coins = this.physics.add.group();
        
        // 设置碰撞检测
        this.setupCollisions();
        
        // 启动游戏场景
        this.scene.start('GameScene', {
            player: this.player,
            enemies: this.enemies,
            coins: this.coins
        });
    }
    
    createMap() {
        // 创建地图层
        this.map = this.make.tilemap({ key: 'level1' });
        const tileset = this.map.addTilesetImage('tileset', 'tiles');
        
        // 创建平台层
        this.platforms = this.map.createLayer('platforms', tileset, 0, 0);
        this.platforms.setCollisionByExclusion([-1], true);
        
        // 创建装饰层
        this.decorations = this.map.createLayer('decorations', tileset, 0, 0);
    }
    
    setupCollisions() {
        // 玩家与平台碰撞
        this.physics.add.collider(this.player, this.platforms);
        
        // 玩家与敌人碰撞
        this.physics.add.collider(this.player, this.enemies, this.playerHitEnemy, null, this);
        
        // 玩家与收集物碰撞
        this.physics.add.overlap(this.player, this.coins, this.playerCollectCoin, null, this);
        
        // 敌人与平台碰撞
        this.physics.add.collider(this.enemies, this.platforms);
        
        // 敌人与敌人碰撞
        this.physics.add.collider(this.enemies, this.enemies);
    }
    
    playerHitEnemy(player, enemy) {
        // 处理玩家与敌人碰撞
        if (player.body.velocity.y > 0 && enemy.body.velocity.y >= 0) {
            // 玩家从上方踩踏敌人
            player.setVelocityY(-200);
            enemy.destroy();
        } else {
            // 玩家受到伤害
            player.hurt();
        }
    }
    
    playerCollectCoin(player, coin) {
        // 处理玩家收集金币
        coin.destroy();
        // 添加得分逻辑
    }
}

// 游戏场景
class GameScene extends Phaser.Scene {
    constructor() {
        super({ key: 'GameScene' });
    }
    
    init(data) {
        // 从预加载场景获取游戏对象
        this.player = data.player;
        this.enemies = data.enemies;
        this.coins = data.coins;
        
        // 创建性能监控
        this.performanceMonitor = new PerformanceMonitor(this);
        
        // 创建UI
        this.createUI();
        
        // 创建敌人和收集物
        this.createLevelObjects();
        
        // 设置输入
        this.setupInput();
        
        // 设置游戏状态
        this.gameState = 'playing';
    }
    
    createUI() {
        // 创建得分显示
        this.scoreText = this.add.text(16, 16, 'Score: 0', { fontSize: '24px', fill: '#fff' });
        this.score = 0;
        
        // 创建生命值显示
        this.livesText = this.add.text(16, 48, 'Lives: 3', { fontSize: '24px', fill: '#fff' });
        this.lives = 3;
        
        // 创建时间显示
        this.timeText = this.add.text(16, 80, 'Time: 0', { fontSize: '24px', fill: '#fff' });
        this.time = 0;
        
        // 创建游戏状态文本
        this.statusText = this.add.text(this.cameras.main.width / 2, this.cameras.main.height / 2, '', {
            fontSize: '48px',
            fill: '#fff',
            align: 'center'
        }).setOrigin(0.5);
        this.statusText.visible = false;
    }
    
    createLevelObjects() {
        // 创建敌人
        for (let i = 0; i < 5; i++) {
            let enemy = new Enemy(this, 200 + i * 150, 400, 'goomba');
            this.enemies.add(enemy);
        }
        
        // 创建收集物
        for (let i = 0; i < 10; i++) {
            let coin = this.physics.add.sprite(100 + i * 70, 300 - i * 30, 'coin');
            this.coins.add(coin);
        }
    }
    
    setupInput() {
        // 键盘输入
        this.cursors = this.input.keyboard.createCursorKeys();
        this.spaceKey = this.input.keyboard.addKey(Phaser.Input.Keyboard.KeyCodes.SPACE);
        
        // 输入错误处理
        this.input.on('pointerdown', function (pointer) {
            if (pointer.x < 0 || pointer.x > this.cameras.main.width || 
                pointer.y < 0 || pointer.y > this.cameras.main.height) {
                console.warn('Invalid pointer position');
            }
        }, this);
    }
    
    update(time, delta) {
        // 更新性能监控
        this.performanceMonitor.update();
        
        // 更新游戏状态
        if (this.gameState === 'playing') {
            // 更新玩家
            this.player.update(this.cursors, this.spaceKey.isDown);
            
            // 更新敌人
            this.enemies.children.entries.forEach(enemy => {
                enemy.update();
            });
            
            // 检查碰撞
            this.checkCollisions();
            
            // 更新游戏时间
            this.time += delta / 1000;
            this.timeText.setText(`Time: ${Math.floor(this.time)}`);
            
            // 检查游戏结束条件
            if (this.lives <= 0) {
                this.gameOver();
            }
        }
        
        // 更新相机跟随玩家
        this.cameras.main.startFollow(this.player, true, 0.1, 0.1);
    }
    
    checkCollisions() {
        // 检查玩家与敌人碰撞
        this.physics.overlap(this.player, this.enemies, (player, enemy) => {
            if (player.body.velocity.y > 0 && enemy.body.velocity.y >= 0) {
                // 玩家从上方踩踏敌人
                player.setVelocityY(-200);
                enemy.destroy();
                this.score += 100;
                this.scoreText.setText(`Score: ${this.score}`);
            } else {
                // 玩家受到伤害
                player.hurt();
            }
        });
        
        // 检查玩家与收集物碰撞
        this.physics.overlap(this.player, this.coins, (player, coin) => {
            coin.destroy();
            this.score += 50;
            this.scoreText.setText(`Score: ${this.score}`);
        });
    }
    
    gameOver() {
        this.gameState = 'gameover';
        this.statusText.setText('GAME OVER');
        this.statusText.visible = true;
        
        // 添加重新开始功能
        this.input.keyboard.on('keydown-ENTER', function () {
            this.restartGame();
        }, this);
    }
    
    restartGame() {
        // 重置游戏状态
        this.gameState = 'playing';
        this.score = 0;
        this.lives = 3;
        this.time = 0;
        
        // 更新UI
        this.scoreText.setText(`Score: ${this.score}`);
        this.livesText.setText(`Lives: ${this.lives}`);
        this.timeText.setText(`Time: ${this.time}`);
        this.statusText.visible = false;
        
        // 重置玩家位置
        this.player.setPosition(100, 400);
        this.player.setVelocity(0, 0);
        this.player.respawn();
        
        // 重新创建敌人和收集物
        this.enemies.clear(true, true);
        this.coins.clear(true, true);
        this.createLevelObjects();
    }
}

// 玩家类
class Player extends Phaser.Physics.Arcade.Sprite {
    constructor(scene, x, y) {
        super(scene, x, y, 'player');
        
        // 添加到物理系统
        scene.physics.add.existing(this);
        scene.add.existing(this);
        
        // 设置物理属性
        this.setBounce(0.2);
        this.setCollideWorldBounds(true);
        this.body.setSize(24, 28, 4, 4);
        
        // 设置动画
        this.setupAnimations();
        
        // 游戏状态
        this.isHurt = false;
        this.invulnerableTime = 0;
        this.isJumping = false;
        this.jumpCount = 0;
        this.maxJumps = 2; // 允许二段跳
    }
    
    setupAnimations() {
        // 创建所有需要的动画
        this.scene.anims.create({
            key: 'idle',
            frames: this.scene.anims.generateFrameNumbers('player', { start: 0, end: 0 }),
            frameRate: 10,
            repeat: -1
        });
        
        this.scene.anims.create({
            key: 'run',
            frames: this.scene.anims.generateFrameNumbers('player', { start: 1, end: 3 }),
            frameRate: 10,
            repeat: -1
        });
        
        this.scene.anims.create({
            key: 'jump',
            frames: this.scene.anims.generateFrameNumbers('player', { start: 4, end: 4 }),
            frameRate: 10,
            repeat: -1
        });
        
        this.scene.anims.create({
            key: 'fall',
            frames: this.scene.anims.generateFrameNumbers('player', { start: 5, end: 5 }),
            frameRate: 10,
            repeat: -1
        });
        
        this.scene.anims.create({
            key: 'hurt',
            frames: this.scene.anims.generateFrameNumbers('player', { start: 6, end: 6 }),
            frameRate: 10,
            repeat: -1
        });
        
        // 设置默认动画
        this.anims.play('idle');
    }
    
    update(cursors, isJumping) {
        // 无敌时间处理
        if (this.isHurt) {
            this.invulnerableTime -= this.scene.game.loop.delta;
            if (this.invulnerableTime <= 0) {
                this.isHurt = false;
                this.alpha = 1;
            } else {
                // 闪烁效果
                this.alpha = this.invulnerableTime % 100 < 50 ? 1 : 0.5;
            }
        }
        
        // 水平移动
        if (cursors.left.isDown) {
            this.setVelocityX(-160);
            this.scaleX = -1;
            if (this.body.blocked.down && !this.isJumping) {
                this.anims.play('run', true);
            }
        } else if (cursors.right.isDown) {
            this.setVelocityX(160);
            this.scaleX = 1;
            if (this.body.blocked.down && !this.isJumping) {
                this.anims.play('run', true);
            }
        } else {
            this.setVelocityX(0);
            if (this.body.blocked.down && !this.isJumping) {
                this.anims.play('idle', true);
            }
        }
        
        // 跳跃
        if (isJumping && this.jumpCount < this.maxJumps) {
            this.jump();
        }
        
        // 更新跳跃状态
        if (this.body.blocked.down) {
            this.isJumping = false;
            this.jumpCount = 0;
            if (Math.abs(this.body.velocity.x) < 1) {
                this.anims.play('idle', true);
            }
        } else {
            this.isJumping = true;
            if (this.body.velocity.y < 0) {
                this.anims.play('jump', true);
            } else {
                this.anims.play('fall', true);
            }
        }
    }
    
    jump() {
        // 跳跃逻辑
        this.setVelocityY(-330);
        this.jumpCount++;
        this.isJumping = true;
        
        // 播放跳跃音效
        if (this.scene.sound.get('jump')) {
            this.scene.sound.play('jump');
        }
    }
    
    hurt() {
        // 处理受伤逻辑
        if (!this.isHurt) {
            this.isHurt = true;
            this.invulnerableTime = 1000; // 1秒无敌时间
            this.lives--;
            this.scene.livesText.setText(`Lives: ${this.scene.lives}`);
            
            // 播放受伤音效
