const Utils = {
    lerp: (a, b, t) => a + (b - a) * t,
    clamp: (val, min, max) => Math.max(min, Math.min(max, val)),
    randomRange: (min, max) => Math.random() * (max - min) + min,
    distance: (x1, y1, x2, y2) => Math.sqrt((x2 - x1) ** 2 + (y2 - y1) ** 2),
    easeOutCubic: (t) => 1 - Math.pow(1 - t, 3)
};

class Vector2 {
    constructor(x = 0, y = 0) {
        this.x = x;
        this.y = y;
    }
    add(v) { return new Vector2(this.x + v.x, this.y + v.y); }
    sub(v) { return new Vector2(this.x - v.x, this.y - v.y); }
    mul(s) { return new Vector2(this.x * s, this.y * s); }
    dot(v) { return this.x * v.x + this.y * v.y; }
    length() { return Math.sqrt(this.x * this.x + this.y * this.y); }
    normalize() {
        const len = this.length();
        return len === 0 ? new Vector2() : new Vector2(this.x / len, this.y / len);
    }
    clone() { return new Vector2(this.x, this.y); }
}

class Particle {
    constructor() { this.reset(); }
    
    reset() {
        this.x = 0; this.y = 0;
        this.vx = 0; this.vy = 0;
        this.size = 5; this.color = '#fff';
        this.alpha = 1; this.life = 60;
        this.maxLife = 60; this.active = false;
    }
    
    init(x, y, vx, vy, size, color, life) {
        this.x = x; this.y = y;
        this.vx = vx; this.vy = vy;
        this.size = size; this.color = color;
        this.life = life; this.maxLife = life;
        this.alpha = 1; this.active = true;
    }
    
    update() {
        if (!this.active) return;
        this.x += this.vx;
        this.y += this.vy;
        this.vy += 0.15;
        this.vx *= 0.98;
        this.life--;
        this.alpha = Math.max(0, this.life / this.maxLife);
        if (this.life <= 0) this.active = false;
    }
    
    render(ctx) {
        if (!this.active || this.alpha <= 0) return;
        ctx.save();
        ctx.globalAlpha = this.alpha;
        ctx.fillStyle = this.color;
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
        ctx.fill();
        ctx.restore();
    }
}

// 粒子系统

class ParticleSystem {
    constructor(maxParticles = 300) {
        this.particles = [];
        for (let i = 0; i < maxParticles; i++) {
            this.particles.push(new Particle());
        }
    }
    
    emit(x, y, count, colors, sizeRange, speedRange) {
        let emitted = 0;
        for (const p of this.particles) {
            if (!p.active && emitted < count) {
                const angle = Utils.randomRange(0, Math.PI * 2);
                const speed = Utils.randomRange(speedRange[0], speedRange[1]);
                const color = colors[Math.floor(Math.random() * colors.length)];
                p.init(
                    x, y,
                    Math.cos(angle) * speed,
                    Math.sin(angle) * speed - 2,
                    Utils.randomRange(sizeRange[0], sizeRange[1]),
                    color,
                    Utils.randomRange(30, 60)
                );
                emitted++;
            }
        }
    }
    
    update() { this.particles.forEach(p => p.update()); }
    render(ctx) { this.particles.forEach(p => p.render(ctx)); }
}

class Rigidbody {
    constructor(x, y, width, height, options = {}) {
        this.pos = new Vector2(x, y);
        this.vel = new Vector2(0, 0);
        this.width = width;
        this.height = height;
        this.angle = 0;
        this.angularVel = 0;
        this.mass = (width * height) * (options.density || 1) * 0.01;
        this.restitution = options.restitution || 0.3;
        this.friction = options.friction || 0.5;
        this.static = options.static || false;
        this.health = options.health || 100;
        this.maxHealth = this.health;
        this.destroyed = false;
        this.sleeping = false;
        this.sleepTimer = 0;
        this.color = options.color || '#708090';
        this.darkColor = options.darkColor || '#4A5568';
    }
    
    get centerX() { return this.pos.x + this.width / 2; }
    get centerY() { return this.pos.y + this.height / 2; }
    
    applyImpulse(ix, iy) {
        if (this.static) return;
        this.vel.x += ix / this.mass;
        this.vel.y += iy / this.mass;
        this.sleeping = false;
        this.sleepTimer = 0;
    }
    
    update() {
        if (this.static || this.destroyed) return;
        
        // 检查睡眠
        if (Math.abs(this.vel.x) < 0.1 && Math.abs(this.vel.y) < 0.1) {
            this.sleepTimer++;
            if (this.sleepTimer > 60) {
                this.sleeping = true;
                return;
            }
        } else {
            this.sleepTimer = 0;
            this.sleeping = false;
        }
        
        // 应用重力
        this.vel.y += CONFIG.GRAVITY;
        
        // 空气阻力
        this.vel.x *= CONFIG.AIR_RESISTANCE;
        this.vel.y *= CONFIG.AIR_RESISTANCE;
        
        // 更新位置
        this.pos.x += this.vel.x;
        this.pos.y += this.vel.y;
        
        // 角速度
        this.angle += this.angularVel;
        this.angularVel *= 0.95;
        
        // 地面碰撞
        const groundY = CONFIG.CANVAS_HEIGHT - CONFIG.GROUND_HEIGHT;
        if (this.pos.y + this.height > groundY) {
            this.pos.y = groundY - this.height;
            this.vel.y *= -this.restitution;
            this.vel.x *= CONFIG.GROUND_FRICTION;
            this.angularVel *= 0.8;
        }
        
        // 边界
        if (this.pos.x < 0) { this.pos.x = 0; this.vel.x *= -0.5; }
        if (this.pos.x + this.width > CONFIG.CANVAS_WIDTH) { 
            this.pos.x = CONFIG.CANVAS_WIDTH - this.width; 
            this.vel.x *= -0.5; 
        }
    }
    
    takeDamage(amount) {
        this.health -= amount;
        if (this.health <= 0) {
            this.destroyed = true;
            return true;
        }
        return false;
    }
    
    intersects(other) {
        return this.pos.x < other.pos.x + other.width &&
               this.pos.x + this.width > other.pos.x &&
               this.pos.y < other.pos.y + other.height &&
               this.pos.y + this.height > other.pos.y;
    }
    
    render(ctx) {
        if (this.destroyed) return;
        
        ctx.save();
        ctx.translate(this.centerX, this.centerY);
        ctx.rotate(this.angle);
        
        // 主体
        const damageRatio = this.health / this.maxHealth;
        ctx.fillStyle = damageRatio < 0.5 ? this.darkColor : this.color;
        ctx.fillRect(-this.width/2, -this.height/2, this.width, this.height);
        
        // 边框
        ctx.strokeStyle = this.darkColor;
        ctx.lineWidth = 2;
        ctx.strokeRect(-this.width/2, -this.height/2, this.width, this.height);
        
        // 裂纹效果
        if (damageRatio < 0.7) {
            ctx.strokeStyle = 'rgba(0,0,0,0.3)';
            ctx.lineWidth = 2;
            ctx.beginPath();
            ctx.moveTo(-this.width/4, -this.height/2);
            ctx.lineTo(0, 0);
            ctx.lineTo(this.width/4, this.height/2);
            ctx.stroke();
        }
        
        ctx.restore();
    }
}

class Bird {
    constructor(x, y, type = 'red') {
        this.pos = new Vector2(x, y);
        this.vel = new Vector2(0, 0);
        this.radius = CONFIG.BIRD.RADIUS;
        this.type = type;
        this.config = CONFIG.BIRD.TYPES[type];
        this.launched = false;
        this.active = true;
        this.rotation = 0;
        this.trail = [];
        this.startPos = new Vector2(x, y);
    }
    
    reset() {
        this.pos = this.startPos.clone();
        this.vel = new Vector2(0, 0);
        this.launched = false;
        this.active = true;
        this.rotation = 0;
        this.trail = [];
    }
    
    update() {
        if (!this.active) return;
        
        if (this.launched) {
            // 物理更新
            this.vel.y += CONFIG.GRAVITY;
            this.vel.x *= CONFIG.AIR_RESISTANCE;
            this.vel.y *= CONFIG.AIR_RESISTANCE;
            
            this.pos.x += this.vel.x;
            this.pos.y += this.vel.y;
            
            this.rotation += this.vel.x * 0.02;
            
            // 轨迹
            if (this.trail.length === 0 || 
                Utils.distance(this.pos.x, this.pos.y, 
                this.trail[this.trail.length-1].x, 
                this.trail[this.trail.length-1].y) > 10) {
                this.trail.push({ x: this.pos.x, y: this.pos.y, alpha: 1 });
                if (this.trail.length > 30) this.trail.shift();
            }
            
            // 更新轨迹透明度
            this.trail.forEach((t, i) => {
                t.alpha = (i + 1) / this.trail.length * 0.5;
            });
            
            // 地面碰撞
            const groundY = CONFIG.CANVAS_HEIGHT - CONFIG.GROUND_HEIGHT - this.radius;
            if (this.pos.y > groundY) {
                this.pos.y = groundY;
                this.vel.y *= -0.4;
                this.vel.x *= 0.7;
            }
            
            // 边界检查
            if (this.pos.x < this.radius) { this.pos.x = this.radius; this.vel.x *= -0.5; }
            if (this.pos.x > CONFIG.CANVAS_WIDTH - this.radius) { 
                this.pos.x = CONFIG.CANVAS_WIDTH - this.radius; 
                this.vel.x *= -0.5; 
            }
            
            // 停止检测
            if (Math.abs(this.vel.x) < 0.2 && Math.abs(this.vel.y) < 0.2 && 
                this.pos.y >= groundY - 5) {
                this.active = false;
            }
            
            // 超出画面
            if (this.pos.y > CONFIG.CANVAS_HEIGHT + 100) {
                this.active = false;
            }
        }
    }
    
    launch(velocity) {
        this.vel = velocity.mul(this.config.power);
        this.launched = true;
    }
    
    checkCollision(body) {
        if (body.destroyed) return false;
        
        const closestX = Utils.clamp(this.pos.x, body.pos.x, body.pos.x + body.width);
        const closestY = Utils.clamp(this.pos.y, body.pos.y, body.pos.y + body.height);
        
        const dist = Utils.distance(this.pos.x, this.pos.y, closestX, closestY);
        return dist < this.radius;
    }
    
    checkCircleCollision(other, radius) {
        const dist = Utils.distance(this.pos.x, this.pos.y, other.x, other.y);
        return dist < this.radius + radius;
    }
    
    render(ctx) {
        // 渲染轨迹
        if (this.launched && this.trail.length > 0) {
            this.trail.forEach((t, i) => {
                ctx.beginPath();
                ctx.arc(t.x, t.y, this.radius * 0.3 * (i / this.trail.length), 0, Math.PI * 2);
                ctx.fillStyle = `rgba(255,255,255,${t.alpha * 0.3})`;
                ctx.fill();
            });
        }
        
        ctx.save();
        ctx.translate(this.pos.x, this.pos.y);
        ctx.rotate(this.rotation);
        
        // 发光效果
        if (this.launched && Math.abs(this.vel.x) + Math.abs(this.vel.y) > 5) {
            ctx.shadowColor = this.config.color;
            ctx.shadowBlur = 20;
        }
        
        // 身体
        ctx.fillStyle = this.config.color;
        ctx.beginPath();
        ctx.arc(0, 0, this.radius, 0, Math.PI * 2);
        ctx.fill();
        
        // 深色部分
        ctx.fillStyle = this.config.dark;
        ctx.beginPath();
        ctx.arc(0, 5, this.radius * 0.8, 0, Math.PI);
        ctx.fill();
        
        // 眼睛
        ctx.fillStyle = '#fff';
        ctx.beginPath();
        ctx.ellipse(-6, -4, 6, 8, 0, 0, Math.PI * 2);
        ctx.ellipse(6, -4, 6, 8, 0, 0, Math.PI * 2);
        ctx.fill();
        
        // 瞳孔
        ctx.fillStyle = '#000';
        ctx.beginPath();
        ctx.arc(-4, -4, 3, 0, Math.PI * 2);
        ctx.arc(4, -4, 3, 0, Math.PI * 2);
        ctx.fill();
        
        // 眉毛
        ctx.strokeStyle = this.config.dark;
        ctx.lineWidth = 3;
        ctx.beginPath();
        ctx.moveTo(-12, -12);
        ctx.lineTo(-2, -8);
        ctx.moveTo(12, -12);
        ctx.lineTo(2, -8);
        ctx.stroke();
        
        // 嘴巴
        ctx.fillStyle = '#FF9800';
        ctx.beginPath();
        ctx.moveTo(0, 4);
        ctx.lineTo(-8, 10);
        ctx.lineTo(0, 8);
        ctx.lineTo(8, 10);
        ctx.closePath();
        ctx.fill();
        
        ctx.restore();
    }
}

class Pig {
    constructor(x, y, size = 'normal') {
        this.pos = new Vector2(x, y);
        this.size = size;
        this.radius = size === 'king' ? 35 : (size === 'small' ? 18 : CONFIG.PIG.RADIUS);
        this.health = size === 'king' ? 100 : (size === 'small' ? 30 : CONFIG.PIG.HEALTH);
        this.maxHealth = this.health;
        this.destroyed = false;
        this.vel = new Vector2(0, 0);
        this.rotation = 0;
    }
    
    update() {
        if (this.destroyed) return;
        
        // 简单物理
        this.vel.y += CONFIG.GRAVITY * 0.5;
        this.pos.x += this.vel.x;
        this.pos.y += this.vel.y;
        this.vel.x *= 0.95;
        this.vel.y *= 0.95;
        
        // 地面
        const groundY = CONFIG.CANVAS_HEIGHT - CONFIG.GROUND_HEIGHT - this.radius;
        if (this.pos.y > groundY) {
            this.pos.y = groundY;
            this.vel.y *= -0.3;
            this.vel.x *= 0.8;
        }
    }
    
    takeDamage(amount) {
        this.health -= amount;
        if (this.health <= 0) {
            this.destroyed = true;
            return true;
        }
        return false;
    }
    
    applyImpulse(ix, iy) {
        this.vel.x += ix * 0.5;
        this.vel.y += iy * 0.5;
    }
    
    render(ctx) {
        if (this.destroyed) return;
        
        const healthRatio = this.health / this.maxHealth;
        
        ctx.save();
        ctx.translate(this.pos.x, this.pos.y);
        ctx.rotate(this.rotation);
        
        // 身体
        ctx.fillStyle = healthRatio < 0.5 ? CONFIG.PIG.COLORS.dark : CONFIG.PIG.COLORS.normal;
        ctx.beginPath();
        ctx.arc(0, 0, this.radius, 0, Math.PI * 2);
        ctx.fill();
        
        // 耳朵
        ctx.fillStyle = CONFIG.PIG.COLORS.dark;
        ctx.beginPath();
        ctx.ellipse(-this.radius * 0.7, -this.radius * 0.7, this.radius * 0.3, this.radius * 0.4, -0.5, 0, Math.PI * 2);
        ctx.ellipse(this.radius * 0.7, -this.radius * 0.7, this.radius * 0.3, this.radius * 0.4, 0.5, 0, Math.PI * 2);
        ctx.fill();
        
        // 眼睛
        ctx.fillStyle = '#fff';
        ctx.beginPath();
        ctx.arc(-this.radius * 0.3, -this.radius * 0.2, this.radius * 0.25, 0, Math.PI * 2);
        ctx.arc(this.radius * 0.3, -this.radius * 0.2, this.radius * 0.25, 0, Math.PI * 2);
        ctx.fill();
        
        // 瞳孔
        ctx.fillStyle = '#000';
        ctx.beginPath();
        ctx.arc(-this.radius * 0.3, -this.radius * 0.15, this.radius * 0.12, 0, Math.PI * 2);
        ctx.arc(this.radius * 0.3, -this.radius * 0.15, this.radius * 0.12, 0, Math.PI * 2);
        ctx.fill();
        
        // 鼻子
        ctx.fillStyle = CONFIG.PIG.COLORS.dark;
        ctx.beginPath();
        ctx.ellipse(0, this.radius * 0.2, this.radius * 0.35, this.radius * 0.25, 0, 0, Math.PI * 2);
        ctx.fill();
        
        // 鼻孔
        ctx.fillStyle = '#33691E';
        ctx.beginPath();
        ctx.ellipse(-this.radius * 0.12, this.radius * 0.2, this.radius * 0.08, this.radius * 0.1, 0, 0, Math.PI * 2);
        ctx.ellipse(this.radius * 0.12, this.radius * 0.2, this.radius * 0.08, this.radius * 0.1, 0, 0, Math.PI * 2);
        ctx.fill();
        
        // 国王皇冠
        if (this.size === 'king') {
            ctx.fillStyle = '#FFD700';
            ctx.beginPath();
            ctx.moveTo(-this.radius * 0.6, -this.radius * 0.8);
            ctx.lineTo(-this.radius * 0.4, -this.radius * 1.3);
            ctx.lineTo(-this.radius * 0.2, -this.radius * 0.9);
            ctx.lineTo(0, -this.radius * 1.4);
            ctx.lineTo(this.radius * 0.2, -this.radius * 0.9);
            ctx.lineTo(this.radius * 0.4, -this.radius * 1.3);
            ctx.lineTo(this.radius * 0.6, -this.radius * 0.8);
            ctx.closePath();
            ctx.fill();
        }
        
        // 受伤表情
        if (healthRatio < 0.5) {
            ctx.strokeStyle = '#000';
            ctx.lineWidth = 2;
            ctx.beginPath();
            ctx.moveTo(-this.radius * 0.45, -this.radius * 0.3);
            ctx.lineTo(-this.radius * 0.15, -this.radius * 0.1);
            ctx.moveTo(-this.radius * 0.15, -this.radius * 0.3);
            ctx.lineTo(-this.radius * 0.45, -this.radius * 0.1);
            ctx.moveTo(this.radius * 0.45, -this.radius * 0.3);
            ctx.lineTo(this.radius * 0.15, -this.radius * 0.1);
            ctx.moveTo(this.radius * 0.15, -this.radius * 0.3);
            ctx.lineTo(this.radius * 0.45, -this.radius * 0.1);
            ctx.stroke();
        }
        
        ctx.restore();
    }
}

// 弹弓类

class Slingshot {
    constructor(x, y) {
        this.x = x;
        this.y = y;
        this.width = 30;
        this.height = 120;
        this.pullX = x;
        this.pullY = y - 50;
        this.maxStretch = CONFIG.SLINGSHOT.MAX_STRETCH;
        this.isPulling = false;
    }
    
    pull(mouseX, mouseY) {
        const dx = mouseX - this.x;
        const dy = mouseY - (this.y - 50);
        const dist = Math.sqrt(dx * dx + dy * dy);
        
        if (dist <= this.maxStretch) {
            this.pullX = mouseX;
            this.pullY = mouseY;
        } else {
            const angle = Math.atan2(dy, dx);
            this.pullX = this.x + Math.cos(angle) * this.maxStretch;
            this.pullY = (this.y - 50) + Math.sin(angle) * this.maxStretch;
        }
    }
    
    release() {
        const dx = this.x - this.pullX;
        const dy = (this.y - 50) - this.pullY;
        this.pullX = this.x;
        this.pullY = this.y - 50;
        return new Vector2(dx * CONFIG.SLINGSHOT.LAUNCH_POWER, dy * CONFIG.SLINGSHOT.LAUNCH_POWER);
    }
    
    getLaunchVelocity() {
        const dx = this.x - this.pullX;
        const dy = (this.y - 50) - this.pullY;
        return new Vector2(dx * CONFIG.SLINGSHOT.LAUNCH_POWER, dy * CONFIG.SLINGSHOT.LAUNCH_POWER);
    }
    
    isInRange(mouseX, mouseY) {
        const dist = Utils.distance(mouseX, mouseY, this.x, this.y - 50);
        return dist < 80;
    }
    
    render(ctx, bird = null) {
        // 后支架
        ctx.fillStyle = '#5D4037';
        ctx.fillRect(this.x - 20, this.y, 12, -this.height + 20);
        
        // 后皮筋
        ctx.strokeStyle = '#3E2723';
        ctx.lineWidth = 6;
        ctx.beginPath();
        ctx.moveTo(this.x - 14, this.y - this.height + 30);
        ctx.lineTo(this.pullX, this.pullY);
        ctx.stroke();
        
        // 渲染小鸟（如果在弹弓上）
        if (bird && !bird.launched) {
            bird.pos.x = this.pullX;
            bird.pos.y = this.pullY;
            bird.render(ctx);
        }
        
        // 前支架
        ctx.fillStyle = '#8D6E63';
        ctx.fillRect(this.x + 8, this.y, 12, -this.height + 20);
        
        // 前皮筋
        ctx.strokeStyle = '#5D4037';
        ctx.lineWidth = 6;
        ctx.beginPath();
        ctx.moveTo(this.x + 14, this.y - this.height + 30);
        ctx.lineTo(this.pullX, this.pullY);
        ctx.stroke();
        
        // 预测轨迹
        if (this.isPulling && bird) {
            this.renderTrajectory(ctx, bird);
        }
    }
    
    renderTrajectory(ctx, bird) {
        const velocity = this.getLaunchVelocity().mul(bird.config.power);
        let x = this.pullX;
        let y = this.pullY;
        let vx = velocity.x;
        let vy = velocity.y;
        
        ctx.fillStyle = 'rgba(255, 255, 255, 0.4)';
        for (let i = 0; i < 50; i++) {
            vy += CONFIG.GRAVITY;
            x += vx;
            y += vy;
            
            if (y > CONFIG.CANVAS_HEIGHT - CONFIG.GROUND_HEIGHT) break;
            
            if (i % 3 === 0) {
                ctx.beginPath();
                ctx.arc(x, y, 3, 0, Math.PI * 2);
                ctx.fill();
            }
        }
    }
}

class Game {
    constructor() {
        this.canvas = document.getElementById('gameCanvas');
        this.ctx = this.canvas.getContext('2d');
        
        this.state = 'menu'; // menu, playing, gameOver
        this.score = 0;
        this.currentLevel = 0;
        
        this.particles = new ParticleSystem(500);
        this.slingshot = new Slingshot(CONFIG.SLINGSHOT.X, CONFIG.SLINGSHOT.Y);
        
        this.birds = [];
        this.currentBirdIndex = 0;
        this.blocks = [];
        this.pigs = [];
        
        this.isDragging = false;
        this.waitingForNextBird = false;
        this.waitTimer = 0;
        
        this.groundOffset = 0;
        
        this.setupEventListeners();
        this.setupUI();
        this.loop();
    }
    
    setupEventListeners() {
        // 鼠标事件
        this.canvas.addEventListener('mousedown', (e) => this.onMouseDown(e));
        this.canvas.addEventListener('mousemove', (e) => this.onMouseMove(e));
        this.canvas.addEventListener('mouseup', (e) => this.onMouseUp(e));
        
        // 触摸事件
        this.canvas.addEventListener('touchstart', (e) => {
            e.preventDefault();
            const touch = e.touches[0];
            const rect = this.canvas.getBoundingClientRect();
            this.onMouseDown({ clientX: touch.clientX, clientY: touch.clientY, target: this.canvas });
        });
        this.canvas.addEventListener('touchmove', (e) => {
            e.preventDefault();
            const touch = e.touches[0];
            this.onMouseMove({ clientX: touch.clientX, clientY: touch.clientY });
        });
        this.canvas.addEventListener('touchend', (e) => {
            e.preventDefault();
            this.onMouseUp({});
        });
        
        // 键盘事件
        document.addEventListener('keydown', (e) => {
            if (e.key === 'r' || e.key === 'R') {
                this.resetLevel();
            }
        });
    }
    
    setupUI() {
        document.getElementById('playBtn').addEventListener('click', () => this.startGame());
        document.getElementById('resetBtn').addEventListener('click', () => this.resetLevel());
        document.getElementById('restartBtn').addEventListener('click', () => {
            document.getElementById('gameOverScreen').classList.remove('visible');
            this.resetLevel();
            this.startGame();
        });
    }
    
    getMousePos(e) {
        const rect = this.canvas.getBoundingClientRect();
        return {
            x: (e.clientX - rect.left) * (this.canvas.width / rect.width),
            y: (e.clientY - rect.top) * (this.canvas.height / rect.height)
        };
    }
    
    onMouseDown(e) {
        if (this.state !== 'playing') return;
        
        const pos = this.getMousePos(e);
        
        if (this.slingshot.isInRange(pos.x, pos.y) && !this.waitingForNextBird) {
            const currentBird = this.getCurrentBird();
            if (currentBird && !currentBird.launched) {
                this.isDragging = true;
                this.slingshot.isPulling = true;
            }
        }
    }
    
    onMouseMove(e) {
        if (this.state !== 'playing' || !this.isDragging) return;
        
        const pos = this.getMousePos(e);
        this.slingshot.pull(pos.x, pos.y);
    }
    
    onMouseUp(e) {
        if (this.state !== 'playing' || !this.isDragging) return;
        
        this.isDragging = false;
        this.slingshot.isPulling = false;
        
        const velocity = this.slingshot.release();
        const currentBird = this.getCurrentBird();
        
        if (currentBird && velocity.length() > 1) {
            currentBird.launch(velocity);
            this.waitingForNextBird = true;
            this.waitTimer = 0;
        }
    }
    
    getCurrentBird() {
        return this.birds[this.currentBirdIndex];
    }
    
    loadLevel(levelIndex) {
        const level = LEVELS[levelIndex];
        if (!level) return;
        
        // 清空
        this.birds = [];
        this.blocks = [];
        this.pigs = [];
        this.currentBirdIndex = 0;
        
        // 加载小鸟
        level.birds.forEach((type, i) => {
            this.birds.push(new Bird(
                CONFIG.SLINGSHOT.X - 60 - i * 40,
                CONFIG.CANVAS_HEIGHT - CONFIG.GROUND_HEIGHT - 25,
                type
            ));
        });
        
        // 加载结构
        level.structures.forEach(item => {
            if (item.type === 'block') {
                const material = CONFIG.MATERIALS[item.material];
                this.blocks.push(new Rigidbody(
                    item.x - item.w / 2,
                    item.y - item.h / 2,
                    item.w,
                    item.h,
                    {
                        density: material.density,
                        health: material.health,
                        color: material.color,
                        darkColor: material.darkColor
                    }
                ));
            } else if (item.type === 'pig') {
                this.pigs.push(new Pig(item.x, item.y, item.size));
            }
        });
        
        this.updateUI();
    }
    
    startGame() {
        document.getElementById('mainMenu').classList.add('hidden');
        document.getElementById('gameUI').style.display = 'flex';
        document.getElementById('controlsHint').style.display = 'flex';
        this.state = 'playing';
        this.score = 0;
        this.loadLevel(this.currentLevel);
    }
    
    resetLevel() {
        this.score = 0;
        this.waitingForNextBird = false;
        this.waitTimer = 0;
        this.loadLevel(this.currentLevel);
        document.getElementById('gameOverScreen').classList.remove('visible');
    }
    
    nextBird() {
        this.currentBirdIndex++;
        this.waitingForNextBird = false;
        this.waitTimer = 0;
        this.updateUI();
        
        // 检查游戏结束
        if (this.currentBirdIndex >= this.birds.length) {
            this.checkGameOver();
        }
    }
    
    checkGameOver() {
        const pigsAlive = this.pigs.filter(p => !p.destroyed).length;
        
        if (pigsAlive === 0) {
            // 胜利
            this.state = 'gameOver';
            document.getElementById('gameOverTitle').textContent = 'LEVEL COMPLETE!';
            document.getElementById('finalScore').textContent = `Score: ${this.score}`;
            document.getElementById('gameOverScreen').classList.add('visible');
        } else if (this.currentBirdIndex >= this.birds.length) {
            // 失败
            this.state = 'gameOver';
            document.getElementById('gameOverTitle').textContent = 'LEVEL FAILED';
            document.getElementById('finalScore').textContent = `Score: ${this.score}`;
            document.getElementById('gameOverScreen').classList.add('visible');
        }
    }
    
    updateUI() {
        document.getElementById('scoreValue').textContent = this.score;
        document.getElementById('pigsLeft').textContent = this.pigs.filter(p => !p.destroyed).length;
        
        // 更新小鸟图标
        const birdsLeftEl = document.getElementById('birdsLeft');
        birdsLeftEl.innerHTML = '';
        this.birds.forEach((bird, i) => {
            const div = document.createElement('div');
            div.className = 'bird-icon' + (i < this.currentBirdIndex ? ' used' : '');
            div.style.background = bird.config.color;
            birdsLeftEl.appendChild(div);
        });
    }
    
    update() {
        if (this.state !== 'playing') return;
        
        this.particles.update();
        
        // 更新小鸟
        this.birds.forEach(bird => bird.update());
        
        // 更新方块
        this.blocks.forEach(block => block.update());
        
        // 更新猪
        this.pigs.forEach(pig => pig.update());
        
        // 碰撞检测
        this.handleCollisions();
        
        // 等待下一只小鸟
        if (this.waitingForNextBird) {
            const currentBird = this.getCurrentBird();
            if (currentBird && !currentBird.active) {
                this.waitTimer++;
                if (this.waitTimer > 60) {
                    this.nextBird();
                }
            }
        }
        
        // 检查是否所有东西都静止了
        this.checkGameOver();
    }
    
    handleCollisions() {
        const currentBird = this.getCurrentBird();
        
        if (currentBird && currentBird.launched && currentBird.active) {
            // 小鸟与方块碰撞
            this.blocks.forEach(block => {
                if (currentBird.checkCollision(block)) {
                    const impact = currentBird.vel.length() * 10;
                    
                    if (block.takeDamage(impact)) {
                        // 方块被摧毁
                        this.score += 50;
                        this.particles.emit(
                            block.centerX, block.centerY, 20,
                            [block.color, block.darkColor, '#FFE4B5'],
                            [3, 8], [2, 8]
                        );
                    } else {
                        // 方块受损
                        this.particles.emit(
                            currentBird.pos.x, currentBird.pos.y, 5,
                            ['#fff', block.color],
                            [2, 4], [1, 3]
                        );
                    }
                    
                    // 应用冲量
                    block.applyImpulse(
                        currentBird.vel.x * 2,
                        currentBird.vel.y * 2
                    );
                    
                    // 小鸟反弹
                    currentBird.vel = currentBird.vel.mul(-0.3);
                    this.updateUI();
                }
            });
            
            // 小鸟与猪碰撞
            this.pigs.forEach(pig => {
                if (!pig.destroyed && currentBird.checkCircleCollision(pig.pos, pig.radius)) {
                    const impact = currentBird.vel.length() * 15;
                    
                    if (pig.takeDamage(impact)) {
                        this.score += 100;
                        this.particles.emit(
                            pig.pos.x, pig.pos.y, 25,
                            ['#7CB342', '#558B2F', '#C5E1A5'],
                            [4, 10], [3, 10]
                        );
                    }
                    
                    pig.applyImpulse(currentBird.vel.x, currentBird.vel.y);
                    currentBird.vel = currentBird.vel.mul(-0.2);
                    this.updateUI();
                }
            });
        }
        
        // 方块之间的碰撞
        for (let i = 0; i < this.blocks.length; i++) {
            for (let j = i + 1; j < this.blocks.length; j++) {
                const a = this.blocks[i];
                const b = this.blocks[j];
                
                if (!a.destroyed && !b.destroyed && a.intersects(b)) {
                    this.resolveBlockCollision(a, b);
                }
            }
        }
        
        // 方块与猪碰撞
        this.blocks.forEach(block => {
            if (block.destroyed) return;
            
            this.pigs.forEach(pig => {
                if (pig.destroyed) return;
                
                const closestX = Utils.clamp(pig.pos.x, block.pos.x, block.pos.x + block.width);
                const closestY = Utils.clamp(pig.pos.y, block.pos.y, block.pos.y + block.height);
                const dist = Utils.distance(pig.pos.x, pig.pos.y, closestX, closestY);
                
                if (dist < pig.radius) {
                    const impact = (Math.abs(block.vel.x) + Math.abs(block.vel.y)) * 5;
                    if (impact > 5 && pig.takeDamage(impact)) {
                        this.score += 100;
                        this.particles.emit(
                            pig.pos.x, pig.pos.y, 25,
                            ['#7CB342', '#558B2F', '#C5E1A5'],
                            [4, 10], [3, 10]
                        );
                        this.updateUI();
                    }
                    pig.applyImpulse(block.vel.x, block.vel.y);
                }
            });
        });
    }
    
    resolveBlockCollision(a, b) {
        // 简单的分离算法
        const overlapX = Math.min(a.pos.x + a.width - b.pos.x, b.pos.x + b.width - a.pos.x);
        const overlapY = Math.min(a.pos.y + a.height - b.pos.y, b.pos.y + b.height - a.pos.y);
        
        if (overlapX < overlapY) {
            if (a.centerX < b.centerX) {
                a.pos.x -= overlapX / 2;
                b.pos.x += overlapX / 2;
            } else {
                a.pos.x += overlapX / 2;
                b.pos.x -= overlapX / 2;
            }
            const avgVelX = (a.vel.x + b.vel.x) / 2;
            a.vel.x = avgVelX * 0.5;
            b.vel.x = avgVelX * 0.5;
        } else {
            if (a.centerY < b.centerY) {
                a.pos.y -= overlapY / 2;
                b.pos.y += overlapY / 2;
            } else {
                a.pos.y += overlapY / 2;
                b.pos.y -= overlapY / 2;
            }
            const avgVelY = (a.vel.y + b.vel.y) / 2;
            a.vel.y = avgVelY * 0.3;
            b.vel.y = avgVelY * 0.3;
        }
    }
    
    render() {
        // 清空画布
        this.ctx.clearRect(0, 0, CONFIG.CANVAS_WIDTH, CONFIG.CANVAS_HEIGHT);
        
        // 渲染背景
        this.renderBackground();
        
        // 渲染地面
        this.renderGround();
        
        if (this.state === 'playing' || this.state === 'gameOver') {
            // 渲染方块
            this.blocks.forEach(block => block.render(this.ctx));
            
            // 渲染猪
            this.pigs.forEach(pig => pig.render(this.ctx));
            
            // 渲染弹弓
            const currentBird = this.getCurrentBird();
            this.slingshot.render(this.ctx, currentBird);
            
            // 渲染已发射的小鸟
            this.birds.forEach((bird, i) => {
                if (i < this.currentBirdIndex || (bird.launched && bird !== currentBird)) {
                    bird.render(this.ctx);
                }
            });
            
            // 渲染粒子
            this.particles.render(this.ctx);
        }
    }
    
    renderBackground() {
        // 天空渐变
        const gradient = this.ctx.createLinearGradient(0, 0, 0, CONFIG.CANVAS_HEIGHT - CONFIG.GROUND_HEIGHT);
        gradient.addColorStop(0, '#87CEEB');
        gradient.addColorStop(0.5, '#B0E0E6');
        gradient.addColorStop(1, '#E0F7FA');
        this.ctx.fillStyle = gradient;
        this.ctx.fillRect(0, 0, CONFIG.CANVAS_WIDTH, CONFIG.CANVAS_HEIGHT - CONFIG.GROUND_HEIGHT);
        
        // 云朵
        this.ctx.fillStyle = 'rgba(255, 255, 255, 0.8)';
        this.renderCloud(100, 80, 60);
        this.renderCloud(300, 120, 45);
        this.renderCloud(550, 60, 70);
        this.renderCloud(800, 100, 55);
        this.renderCloud(1050, 70, 50);
        
        // 远山
        this.ctx.fillStyle = '#A5D6A7';
        this.ctx.beginPath();
        this.ctx.moveTo(0, CONFIG.CANVAS_HEIGHT - CONFIG.GROUND_HEIGHT);
        this.ctx.lineTo(200, CONFIG.CANVAS_HEIGHT - CONFIG.GROUND_HEIGHT - 100);
        this.ctx.lineTo(400, CONFIG.CANVAS_HEIGHT - CONFIG.GROUND_HEIGHT - 60);
