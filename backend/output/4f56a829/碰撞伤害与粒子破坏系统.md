const Utils = {
    lerp: (a, b, t) => a + (b - a) * t,
    clamp: (val, min, max) => Math.max(min, Math.min(max, val)),
    randomRange: (min, max) => Math.random() * (max - min) + min,
    randomInt: (min, max) => Math.floor(Utils.randomRange(min, max + 1)),
    distance: (x1, y1, x2, y2) => Math.sqrt((x2 - x1) ** 2 + (y2 - y1) ** 2),
    
    safeColor: (r, g, b, a = 1) => {
        r = Utils.clamp(Math.round(r), 0, 255);
        g = Utils.clamp(Math.round(g), 0, 255);
        b = Utils.clamp(Math.round(b), 0, 255);
        a = Utils.clamp(a, 0, 1);
        return `rgba(${r},${g},${b},${a})`;
    },
    
    pickRandom: (arr) => arr[Math.floor(Math.random() * arr.length)],
    
    noise: (x) => {
        const n = Math.sin(x * 12.9898) * 43758.5453;
        return n - Math.floor(n);
    },
    
    easeOutCubic: (t) => 1 - Math.pow(1 - t, 3),
    easeOutElastic: (t) => {
        const c4 = (2 * Math.PI) / 3;
        return t === 0 ? 0 : t === 1 ? 1 : Math.pow(2, -10 * t) * Math.sin((t * 10 - 0.75) * c4) + 1;
    },
    easeOutBack: (t) => {
        const c1 = 1.70158;
        const c3 = c1 + 1;
        return 1 + c3 * Math.pow(t - 1, 3) + c1 * Math.pow(t - 1, 2);
    },
    easeInQuad: (t) => t * t,
    easeOutQuad: (t) => 1 - (1 - t) * (1 - t)
};

class Vector2 {
    constructor(x = 0, y = 0) {
        this.x = x;
        this.y = y;
    }

    add(v) { return new Vector2(this.x + v.x, this.y + v.y); }
    sub(v) { return new Vector2(this.x - v.x, this.y - v.y); }
    mul(s) { return new Vector2(this.x * s, this.y * s); }
    dot(v) { return this.x * v.x + this.y * v.y; }
    cross(v) { return this.x * v.y - this.y * v.x; }
    length() { return Math.sqrt(this.x * this.x + this.y * this.y); }
    lengthSq() { return this.x * this.x + this.y * this.y; }

    normalize() {
        const len = this.length();
        if (len === 0) return new Vector2(0, 0);
        return new Vector2(this.x / len, this.y / len);
    }

    rotate(angle) {
        const cos = Math.cos(angle);
        const sin = Math.sin(angle);
        return new Vector2(this.x * cos - this.y * sin, this.x * sin + this.y * cos);
    }

    clone() { return new Vector2(this.x, this.y); }
    set(x, y) { this.x = x; this.y = y; return this; }

    static fromAngle(angle, length = 1) {
        return new Vector2(Math.cos(angle) * length, Math.sin(angle) * length);
    }
}

class Particle {
    constructor() {
        this.reset();
    }

    reset() {
        this.x = 0;
        this.y = 0;
        this.vx = 0;
        this.vy = 0;
        this.size = 5;
        this.color = '#fff';
        this.alpha = 1;
        this.life = 60;
        this.maxLife = 60;
        this.gravity = 0.1;
        this.rotation = 0;
        this.rotationSpeed = 0;
        this.type = 'circle';
        this.shimmer = false;
        this.glow = false;
        this.fadeOut = true;
        this.active = false;
    }

    init(x, y, vx, vy, size, color, life, options = {}) {
        this.x = x;
        this.y = y;
        this.vx = vx;
        this.vy = vy;
        this.size = size;
        this.color = color;
        this.life = life;
        this.maxLife = life;
        this.alpha = 1;
        this.rotation = options.rotation || 0;
        this.rotationSpeed = options.rotationSpeed || 0;
        this.gravity = options.gravity ?? 0.1;
        this.type = options.type || 'circle';
        this.shimmer = options.shimmer || false;
        this.glow = options.glow || false;
        this.fadeOut = options.fadeOut !== false;
        this.active = true;
    }

    update() {
        if (!this.active) return;

        this.x += this.vx;
        this.y += this.vy;
        this.vy += this.gravity;
        this.vx *= 0.98;
        this.rotation += this.rotationSpeed;
        this.life--;

        if (this.fadeOut) {
            this.alpha = Math.max(0, this.life / this.maxLife);
        }

        if (this.life <= 0) {
            this.active = false;
        }
    }

    render(ctx) {
        if (!this.active || this.alpha <= 0) return;

        ctx.save();
        ctx.globalAlpha = this.alpha;
        ctx.translate(this.x, this.y);
        ctx.rotate(this.rotation);

        if (this.glow) {
            ctx.shadowBlur = 15;
            ctx.shadowColor = this.color;
        }

        if (this.shimmer) {
            const shimmerAlpha = 0.5 + Math.sin(Date.now() * 0.02 + this.x) * 0.5;
            ctx.globalAlpha = this.alpha * shimmerAlpha;
        }

        ctx.fillStyle = this.color;

        switch (this.type) {
            case 'circle':
                ctx.beginPath();
                ctx.arc(0, 0, this.size, 0, Math.PI * 2);
                ctx.fill();
                break;

            case 'shard':
                ctx.beginPath();
                ctx.moveTo(0, -this.size);
                ctx.lineTo(this.size * 0.5, this.size * 0.5);
                ctx.lineTo(-this.size * 0.5, this.size * 0.5);
                ctx.closePath();
                ctx.fill();
                break;

            case 'debris':
                ctx.fillRect(-this.size / 2, -this.size / 2, this.size, this.size * 0.6);
                break;

            case 'dust':
                ctx.beginPath();
                ctx.arc(0, 0, this.size, 0, Math.PI * 2);
                ctx.fill();
                break;

            case 'spark':
                ctx.beginPath();
                ctx.moveTo(-this.size, 0);
                ctx.lineTo(0, -this.size * 0.3);
                ctx.lineTo(this.size, 0);
                ctx.lineTo(0, this.size * 0.3);
                ctx.closePath();
                ctx.fill();
                break;

            case 'flame':
                const gradient = ctx.createRadialGradient(0, 0, 0, 0, 0, this.size);
                gradient.addColorStop(0, this.color);
                gradient.addColorStop(0.5, this.color + '80');
                gradient.addColorStop(1, 'transparent');
                ctx.fillStyle = gradient;
                ctx.beginPath();
                ctx.arc(0, 0, this.size, 0, Math.PI * 2);
                ctx.fill();
                break;
        }

        ctx.restore();
    }
}

class ParticleSystem {
    constructor(maxParticles = 500) {
        this.particles = [];
        this.maxParticles = maxParticles;
        
        // 预分配粒子池
        for (let i = 0; i < maxParticles; i++) {
            this.particles.push(new Particle());
        }
        
        this.activeCount = 0;
    }

    emit(x, y, config, velocity = null) {
        const count = Utils.randomInt(config.count.min, config.count.max);
        let emitted = 0;

        for (const particle of this.particles) {
            if (!particle.active) {
                const angle = velocity ? 
                    Math.atan2(velocity.y, velocity.x) + Utils.randomRange(-0.5, 0.5) :
                    Utils.randomRange(0, Math.PI * 2);
                const speed = Utils.randomRange(config.speed.min, config.speed.max);
                
                const vx = Math.cos(angle) * speed;
                const vy = Math.sin(angle) * speed - Utils.randomRange(1, 3);
                
                const size = Utils.randomRange(config.size.min, config.size.max);
                const color = Utils.pickRandom(config.colors);
                const life = Utils.randomInt(config.life.min, config.life.max);
                
                particle.init(x, y, vx, vy, size, color, life, {
                    type: config.type,
                    gravity: config.gravity,
                    rotation: Utils.randomRange(0, Math.PI * 2),
                    rotationSpeed: Utils.randomRange(-0.2, 0.2),
                    shimmer: config.shimmer || false,
                    glow: config.glow || false,
                    fadeOut: config.fadeOut !== false
                });
                
                emitted++;
                this.activeCount++;
                
                if (emitted >= count) break;
            }
        }
    }

    emitExplosion(x, y, radius, config) {
        // 发射爆炸火焰粒子
        this.emit(x, y, config);
        
        // 发射额外的火花
        for (let i = 0; i < 10; i++) {
            const angle = (Math.PI * 2 / 10) * i;
            const dist = Utils.randomRange(10, radius * 0.3);
            const px = x + Math.cos(angle) * dist;
            const py = y + Math.sin(angle) * dist;
            
            this.emit(px, py, {
                type: 'spark',
                colors: ['#FFD700', '#FFA500', '#FF4500'],
                count: { min: 3, max: 6 },
                size: { min: 3, max: 6 },
                speed: { min: 4, max: 8 },
                life: { min: 20, max: 40 },
                gravity: 0.05
            });
        }
    }

    emitCollision(x, y, impulse, material) {
        // 根据冲量大小调整粒子数量
        const intensity = Math.min(impulse / 20, 2);
        const config = { ...material.particles };
        config.count = {
            min: Math.floor(config.count.min * intensity),
            max: Math.floor(config.count.max * intensity)
        };
        
        this.emit(x, y, config);
    }

    update() {
        this.activeCount = 0;
        for (const particle of this.particles) {
            if (particle.active) {
                particle.update();
                if (particle.active) this.activeCount++;
            }
        }
    }

    render(ctx, camera) {
        ctx.save();
        
        for (const particle of this.particles) {
            if (particle.active) {
                // 转换为屏幕坐标
                const screenX = (particle.x - camera.x) * camera.zoom;
                const screenY = (particle.y - camera.y) * camera.zoom;
                
                ctx.save();
                ctx.translate(screenX, screenY);
                ctx.scale(camera.zoom, camera.zoom);
                ctx.translate(-particle.x, -particle.y);
                
                particle.render(ctx);
                
                ctx.restore();
            }
        }
        
        ctx.restore();
    }
}

    addExplosion(x, y, intensity = 1) {
        this.addFlash(x, y, intensity);
        this.addShockwave(x, y, intensity);
        
        // 添加多个闪光点
        for (let i = 0; i < 5; i++) {
            const angle = Utils.randomRange(0, Math.PI * 2);
            const dist = Utils.randomRange(10, 30) * intensity;
            this.effects.push({
                type: 'flash',
                x: x + Math.cos(angle) * dist,
                y: y + Math.sin(angle) * dist,
                intensity: intensity * 0.5,
                radius: 5,
                maxRadius: 30,
                alpha: 1,
                life: 10
            });
        }
    }

    addDamageNumber(x, y, damage, isKill = false) {
        this.effects.push({
            type: 'damage',
            x, y,
            damage: Math.round(damage),
            isKill,
            offsetY: 0,
            alpha: 1,
            scale: 1,
            life: 60,
            maxLife: 60
        });
    }

    update() {
        for (let i = this.effects.length - 1; i >= 0; i--) {
            const effect = this.effects[i];
            effect.life--;

            if (effect.type === 'flash') {
                effect.radius += (effect.maxRadius - effect.radius) * 0.3;
                effect.alpha = effect.life / 15;
            } else if (effect.type === 'shockwave') {
                effect.radius += (effect.maxRadius - effect.radius) * 0.1;
                effect.alpha = (effect.life / effect.maxLife) * 0.6;
            } else if (effect.type === 'damage') {
                effect.offsetY -= 1.5;
                effect.scale = 1 + Math.sin((1 - effect.life / effect.maxLife) * Math.PI) * 0.3;
                effect.alpha = effect.life / effect.maxLife;
            }

            if (effect.life <= 0) {
                this.effects.splice(i, 1);
            }
        }
    }

    render(ctx, camera) {
        for (const effect of this.effects) {
            const screenX = (effect.x - camera.x) * camera.zoom;
            const screenY = (effect.y - camera.y) * camera.zoom;

            if (effect.type === 'flash') {
                ctx.save();
                ctx.globalAlpha = effect.alpha;
                
                const radius = effect.radius * camera.zoom;
                const gradient = ctx.createRadialGradient(
                    screenX, screenY, 0,
                    screenX, screenY, radius
                );
                gradient.addColorStop(0, '#FFFFFF');
                gradient.addColorStop(0.3, '#FFFFAA');
                gradient.addColorStop(0.6, '#FFAA00');
                gradient.addColorStop(1, 'transparent');
                
                ctx.fillStyle = gradient;
                ctx.beginPath();
                ctx.arc(screenX, screenY, radius, 0, Math.PI * 2);
                ctx.fill();
                ctx.restore();
            } else if (effect.type === 'shockwave') {
                ctx.save();
                ctx.globalAlpha = effect.alpha;
                ctx.strokeStyle = '#FFFFFF';
                ctx.lineWidth = 3 * camera.zoom;
                
                const radius = effect.radius * camera.zoom;
                ctx.beginPath();
                ctx.arc(screenX, screenY, radius, 0, Math.PI * 2);
                ctx.stroke();
                
                ctx.lineWidth = 1.5 * camera.zoom;
                ctx.globalAlpha = effect.alpha * 0.5;
                ctx.beginPath();
                ctx.arc(screenX, screenY, radius * 1.3, 0, Math.PI * 2);
                ctx.stroke();
                ctx.restore();
            } else if (effect.type === 'damage') {
                ctx.save();
                ctx.globalAlpha = effect.alpha;
                ctx.font = `bold ${16 * effect.scale * camera.zoom}px Arial`;
                ctx.textAlign = 'center';
                ctx.textBaseline = 'middle';
                
                const color = effect.isKill ? '#FF4444' : '#FFFF00';
                ctx.fillStyle = color;
                ctx.strokeStyle = '#000';
                ctx.lineWidth = 3 * camera.zoom;
                
                const text = effect.isKill ? `-${effect.damage}!` : `-${effect.damage}`;
                const y = screenY + effect.offsetY * camera.zoom;
                
                ctx.strokeText(text, screenX, y);
                ctx.fillText(text, screenX, y);
                ctx.restore();
            }
        }
    }
}

// 动画状态机
// ============================================
class AnimationStateMachine {
    constructor() {
        this.currentState = 'idle';
        this.previousState = null;
        this.stateTime = 0;
        this.transitionTime = 0;
        this.isTransitioning = false;
        
        this.states = {
            idle: {
                duration: Infinity,
                loop: true,
                squashX: 1, squashY: 1,
                bobSpeed: 0.05, bobAmount: 2
            },
            aiming: {
                duration: Infinity,
                loop: true,
                squashX: 1.15, squashY: 0.85,
                stretchDirection: true
            },
            flying: {
                duration: Infinity,
                loop: true,
                squashX: 0.85, squashY: 1.2,
                rotateToVelocity: true
            },
            ability: {
                duration: 500,
                loop: false,
                squashX: 0.7, squashY: 1.4,
                flash: true
            },
            hit: {
                duration: 300,
                loop: false,
                squashX: 1.3, squashY: 0.7,
                shake: true
            },
            falling: {
                duration: Infinity,
                loop: true,
                squashX: 0.9, squashY: 1.1,
                rotateToVelocity: true
            }
        };
    }

    setState(newState, force = false) {
        if (newState === this.currentState && !force) return;
        this.previousState = this.currentState;
        this.currentState = newState;
        this.stateTime = 0;
        this.isTransitioning = true;
        this.transitionTime = 0;
    }

    update(deltaTime) {
        this.stateTime += deltaTime;
        this.transitionTime = Math.min(this.transitionTime + deltaTime * 5, 1);
        if (this.transitionTime >= 1) this.isTransitioning = false;
        
        const state = this.states[this.currentState];
        if (!state.loop && this.stateTime >= state.duration) {
            this.setState('falling');
        }
    }

    getCurrentState() {
        return this.states[this.currentState];
    }

    getAnimationData(velocity = { x: 0, y: 0 }) {
        const state = this.states[this.currentState];
        const time = this.stateTime / 1000;
        
        let scaleX = state.squashX || 1;
        let scaleY = state.squashY || 1;
        let rotation = 0;
        let bobOffset = 0;
        
        if (this.currentState === 'idle') {
            bobOffset = Math.sin(time * state.bobSpeed * 60) * state.bobAmount;
        }
        
        if (this.currentState === 'aiming') {
            const breathe = Math.sin(time * 4) * 0.05;
            scaleX += breathe;
            scaleY -= breathe * 0.5;
        }
        
        if (state.rotateToVelocity) {
            const speed = Math.sqrt(velocity.x ** 2 + velocity.y ** 2);
            if (speed > 0.5) {
                rotation = Math.atan2(velocity.y, velocity.x);
                const stretchFactor = Math.min(speed / 15, 0.3);
                scaleX = 0.85 - stretchFactor * 0.2;
                scaleY = 1.2 + stretchFactor * 0.3;
            }
        }
        
        if (this.currentState === 'ability') {
            const pulse = Math.sin(time * 30) * 0.2;
            scaleX += pulse;
            scaleY += pulse;
        }
        
        if (state.shake) {
            const shake = Math.sin(time * 80) * 0.15;
            scaleX += shake;
            rotation += shake * 0.5;
        }
        
        if (this.isTransitioning) {
            const t = Utils.easeOutCubic(this.transitionTime);
            scaleX = Utils.lerp(1, scaleX, t);
            scaleY = Utils.lerp(1, scaleY, t);
        }
        
        return {
            scaleX, scaleY, rotation, bobOffset,
            flash: state.flash && Math.sin(time * 20) > 0
        };
    }
}

class SoundManager {
    constructor() {
        this.audioContext = null;
        this.enabled = true;
        this.initialized = false;
    }

    init() {
        if (this.initialized) return;
        try {
            this.audioContext = new (window.AudioContext || window.webkitAudioContext)();
            this.initialized = true;
        } catch (e) {
            console.warn('Web Audio API not supported');
            this.enabled = false;
        }
    }

    playLaunch(power) {
        if (!this.enabled || !this.audioContext) return;
        
        const oscillator = this.audioContext.createOscillator();
        const gainNode = this.audioContext.createGain();
        
        oscillator.connect(gainNode);
        gainNode.connect(this.audioContext.destination);
        
        const baseFreq = 200 + power * 8;
        oscillator.frequency.setValueAtTime(baseFreq, this.audioContext.currentTime);
        oscillator.frequency.exponentialRampToValueAtTime(80, this.audioContext.currentTime + 0.3);
        oscillator.type = 'sine';
        
        gainNode.gain.setValueAtTime(0.3, this.audioContext.currentTime);
        gainNode.gain.exponentialRampToValueAtTime(0.01, this.audioContext.currentTime + 0.3);
        
        oscillator.start(this.audioContext.currentTime);
        oscillator.stop(this.audioContext.currentTime + 0.3);
    }

    playCollision(impulse, material) {
        if (!this.enabled || !this.audioContext) return;
        
        const oscillator = this.audioContext.createOscillator();
        const gainNode = this.audioContext.createGain();
        
        oscillator.connect(gainNode);
        gainNode.connect(this.audioContext.destination);
        
        let baseFreq = 200;
        let type = 'sine';
        
        if (material === 'wood') {
            baseFreq = 150 + impulse * 3;
            type = 'triangle';
        } else if (material === 'glass') {
            baseFreq = 400 + impulse * 5;
            type = 'sine';
        } else if (material === 'stone') {
            baseFreq = 100 + impulse * 2;
            type = 'square';
        }
        
        oscillator.frequency.setValueAtTime(baseFreq, this.audioContext.currentTime);
        oscillator.frequency.exponentialRampToValueAtTime(baseFreq * 0.5, this.audioContext.currentTime + 0.15);
        oscillator.type = type;
        
        const volume = Utils.clamp(0.1 + impulse * 0.02, 0.1, 0.4);
        gainNode.gain.setValueAtTime(volume, this.audioContext.currentTime);
        gainNode.gain.exponentialRampToValueAtTime(0.01, this.audioContext.currentTime + 0.15);
        
        oscillator.start(this.audioContext.currentTime);
        oscillator.stop(this.audioContext.currentTime + 0.15);
    }

    playExplosion() {
        if (!this.enabled || !this.audioContext) return;
        
        const bufferSize = this.audioContext.sampleRate * 0.4;
        const buffer = this.audioContext.createBuffer(1, bufferSize, this.audioContext.sampleRate);
        const data = buffer.getChannelData(0);
        
        for (let i = 0; i < bufferSize; i++) {
            data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (bufferSize * 0.15));
        }
        
        const source = this.audioContext.createBufferSource();
        const gainNode = this.audioContext.createGain();
        source.buffer = buffer;
        source.connect(gainNode);
        gainNode.connect(this.audioContext.destination);
        gainNode.gain.setValueAtTime(0.5, this.audioContext.currentTime);
        
        source.start();
    }

    playDestroy(material) {
        if (!this.enabled || !this.audioContext) return;
        
        const bufferSize = this.audioContext.sampleRate * 0.2;
        const buffer = this.audioContext.createBuffer(1, bufferSize, this.audioContext.sampleRate);
        const data = buffer.getChannelData(0);
        
        let decay = 0.1;
        if (material === 'glass') decay = 0.3;
        else if (material === 'stone') decay = 0.08;
        
        for (let i = 0; i < bufferSize; i++) {
            data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (bufferSize * decay));
        }
        
        const source = this.audioContext.createBufferSource();
        const gainNode = this.audioContext.createGain();
        source.buffer = buffer;
        source.connect(gainNode);
        gainNode.connect(this.audioContext.destination);
        gainNode.gain.setValueAtTime(0.3, this.audioContext.currentTime);
        
        source.start();
    }
}

class PhysicsBody {
    constructor(x, y, width, height, material = 'wood') {
        this.position = new Vector2(x, y);
        this.velocity = new Vector2(0, 0);
        this.width = width;
        this.height = height;
        this.rotation = 0;
        this.angularVelocity = 0;
        this.mass = width * height * CONFIG.MATERIALS[material].density;
        this.material = material;
        this.materialConfig = CONFIG.MATERIALS[material];
        
        // 伤害系统
        this.health = this.materialConfig.health;
        this.maxHealth = this.materialConfig.health;
        this.damageLevel = 0; // 0-1 伤害程度
        this.isDestroyed = false;
        
        // 睡眠系统
        this.isSleeping = false;
        this.sleepTimer = 0;
        
        // 碰撞冷却
        this.collisionCooldown = 0;
    }

    takeDamage(damage, impactPoint = null) {
        if (this.isDestroyed) return false;
        
        this.health -= damage;
        this.damageLevel = 1 - (this.health / this.maxHealth);
        
        if (this.health <= 0) {
            this.isDestroyed = true;
            return true; // 被摧毁
        }
        
        return false;
    }

    getImpactPoint(other) {
        return new Vector2(
            (this.position.x + other.position.x) / 2,
            (this.position.y + other.position.y) / 2
        );
    }

    calculateImpulse(other) {
        const relativeVelocity = this.velocity.sub(other.velocity);
        const speed = relativeVelocity.length();
        const combinedMass = this.mass + other.mass;
        return speed * combinedMass / 100;
    }

    update(deltaTime, groundY) {
        if (this.isDestroyed) return;
        
        // 碰撞冷却
        if (this.collisionCooldown > 0) {
            this.collisionCooldown -= deltaTime;
        }
        
        // 重力
        this.velocity.y += CONFIG.PHYSICS.GRAVITY;
        
        // 空气阻力
        this.velocity.x *= CONFIG.PHYSICS.AIR_RESISTANCE;
        this.velocity.y *= CONFIG.PHYSICS.AIR_RESISTANCE;
        
        // 角速度阻尼
        this.angularVelocity *= CONFIG.PHYSICS.ANGULAR_DAMPING;
        
        // 更新位置
        this.position.x += this.velocity.x;
        this.position.y += this.velocity.y;
        this.rotation += this.angularVelocity;
        
        // 地面碰撞
        const bottom = this.position.y + this.height / 2;
        if (bottom > groundY) {
            this.position.y = groundY - this.height / 2;
            
            if (Math.abs(this.velocity.y) > 2) {
                this.velocity.y *= -CONFIG.MATERIALS[this.material].restitution;
            } else {
                this.velocity.y = 0;
            }
            
            this.velocity.x *= CONFIG.PHYSICS.GROUND_FRICTION;
            this.angularVelocity *= 0.9;
        }
        
        // 睡眠检测
        if (CONFIG.PHYSICS.ALLOW_SLEEP) {
            const speed = this.velocity.length();
            if (speed < CONFIG.PHYSICS.SLEEP_THRESHOLD) {
                this.sleepTimer++;
                if (this.sleepTimer > CONFIG.PHYSICS.SLEEP_TIME) {
                    this.isSleeping = true;
                }
            } else {
                this.sleepTimer = 0;
                this.isSleeping = false;
            }
        }
    }

    render(ctx, camera) {
        if (this.isDestroyed) return;
        
        const screenX = (this.position.x - camera.x) * camera.zoom;
        const screenY = (this.position.y - camera.y) * camera.zoom;
        const screenW = this.width * camera.zoom;
        const screenH = this.height * camera.zoom;
        
        ctx.save();
        ctx.translate(screenX, screenY);
        ctx.rotate(this.rotation);
        
        // 根据伤害程度选择颜色
        const colors = this.materialConfig.colors;
        let mainColor = colors.main;
        
        if (this.damageLevel > 0.3) {
            mainColor = this.damageLevel > 0.7 ? colors.damaged : 
                Utils.lerp !== undefined ? colors.dark : mainColor;
        }
        
        // 绘制阴影
        ctx.fillStyle = 'rgba(0,0,0,0.3)';
        ctx.fillRect(-screenW / 2 + 3, -screenH / 2 + 3, screenW, screenH);
        
        // 绘制主体
        ctx.fillStyle = mainColor;
        ctx.fillRect(-screenW / 2, -screenH / 2, screenW, screenH);
        
        // 绘制高光
        ctx.fillStyle = colors.light;
        ctx.fillRect(-screenW / 2, -screenH / 2, screenW, screenH * 0.15);
        
        // 绘制边框
        ctx.strokeStyle = colors.dark;
        ctx.lineWidth = 2;
        ctx.strokeRect(-screenW / 2, -screenH / 2, screenW, screenH);
        
        // 绘制裂纹（根据伤害程度）
        if (this.damageLevel > 0.2) {
            this.renderDamage(ctx, screenW, screenH);
        }
        
        // 绘制血条
        if (this.damageLevel > 0) {
            this.renderHealthBar(ctx, screenW, screenH);
        }
        
        ctx.restore();
    }

    renderDamage(ctx, w, h) {
        ctx.strokeStyle = 'rgba(0,0,0,0.5)';
        ctx.lineWidth = 1;
        
        const crackCount = Math.floor(this.damageLevel * 5);
        for (let i = 0; i < crackCount; i++) {
            const startX = Utils.noise(this.position.x + i) * w - w / 2;
            const startY = Utils.noise(this.position.y + i) * h - h / 2;
            
            ctx.beginPath();
            ctx.moveTo(startX, startY);
            
            let x = startX;
            let y = startY;
            for (let j = 0; j < 3; j++) {
                x += Utils.randomRange(-w * 0.2, w * 0.2);
                y += Utils.randomRange(-h * 0.2, h * 0.2);
                ctx.lineTo(x, y);
            }
            ctx.stroke();
        }
    }

    renderHealthBar(ctx, w, h) {
        const barWidth = w * 0.8;
        const barHeight = 4;
        const barY = -h / 2 - 10;
        
        // 背景
        ctx.fillStyle = 'rgba(0,0,0,0.5)';
        ctx.fillRect(-barWidth / 2, barY, barWidth, barHeight);
        
        // 血量
        const healthPercent = this.health / this.maxHealth;
        const hue = healthPercent * 120; // 0=红, 120=绿
        ctx.fillStyle = `hsl(${hue}, 70%, 50%)`;
        ctx.fillRect(-barWidth / 2, barY, barWidth * healthPercent, barHeight);
    }
}

class Bird {
    constructor(x, y, type = 'red') {
        this.position = new Vector2(x, y);
        this.velocity = new Vector2(0, 0);
        this.radius = CONFIG.BIRD.RADIUS;
        this.type = type;
        this.colors = CONFIG.BIRD.COLORS[type];
        this.animation = new AnimationStateMachine();
        
        this.isLaunched = false;
        this.abilityUsed = false;
        this.isDead = false;
        
        this.mass = type === 'black' ? 2 : type === 'yellow' ? 0.8 : 1;
        this.rotation = 0;
        
        // 特殊能力冷却
        this.abilityCooldown = 0;
    }

    launch(velocity) {
        this.velocity = velocity;
        this.isLaunched = true;
        this.animation.setState('flying');
    }

    useAbility() {
        if (this.abilityUsed || !this.isLaunched) return;
        
        this.abilityUsed = true;
        this.animation.setState('ability');
        
        switch (this.type) {
            case 'yellow':
                // 加速
                this.velocity.x *= 2;
                this.velocity.y *= 0.5;
                break;
            case 'black':
                // 爆炸（在游戏中处理）
                break;
        }
    }

    update(deltaTime, groundY) {
        if (this.isDead) return;
        
        this.animation.update(deltaTime);
        
        if (this.isLaunched) {
            this.velocity.y += CONFIG.PHYSICS.GRAVITY;
            this.velocity.x *= CONFIG.PHYSICS.AIR_RESISTANCE;
            this.velocity.y *= CONFIG.PHYSICS.AIR_RESISTANCE;
            
            this.position.x += this.velocity.x;
            this.position.y += this.velocity.y;
            
            // 更新旋转
            const speed = this.velocity.length();
            if (speed > 1) {
                this.rotation = Math.atan2(this.velocity.y, this.velocity.x);
            }
            
            // 地面碰撞
            if (this.position.y + this.radius > groundY) {
                this.position.y = groundY - this.radius;
                this.velocity.y *= -0.5;
                this.velocity.x *= 0.8;
                
                if (Math.abs(this.velocity.y) < 1) {
                    this.velocity.y = 0;
                }
            }
            
            // 边界检测
            if (this.position.x > CONFIG.WORLD_WIDTH || this.position.x < -100 ||
                this.position.y > CONFIG.CANVAS_HEIGHT + 100) {
                this.isDead = true;
            }
        }
    }

    render(ctx, camera) {
        if (this.isDead) return;
        
        const screenX = (this.position.x - camera.x) * camera.zoom;
        const screenY = (this.position.y - camera.y) * camera.zoom;
        const screenRadius = this.radius * camera.zoom;
        
        const animData = this.animation.getAnimationData(this.velocity);
        
        ctx.save();
        ctx.translate(screenX, screenY + animData.bobOffset * camera.zoom);
        ctx.rotate(this.rotation + animData.rotation);
        ctx.scale(animData.scaleX, animData.scaleY);
        
        // 发光效果
        if (animData.flash) {
            ctx.shadowBlur = 20;
            ctx.shadowColor = this.colors.light;
        }
        
        // 身体
        const gradient = ctx.createRadialGradient(
            -screenRadius * 0.3, -screenRadius * 0.3, 0,
            0, 0, screenRadius
        );
        gradient.addColorStop(0, this.colors.light);
        gradient.addColorStop(0.7, this.colors.main);
        gradient.addColorStop(1, this.colors.dark);
        
        ctx.fillStyle = gradient;
        ctx.beginPath();
        ctx.arc(0, 0, screenRadius, 0, Math.PI * 2);
        ctx.fill();
        
        // 眼睛
        const eyeOffsetX = screenRadius * 0.3;
        const eyeOffsetY = -screenRadius * 0.1;
        const eyeSize = screenRadius * 0.25;
        
        ctx.fillStyle = '#FFFFFF';
        ctx.beginPath();
        ctx.ellipse(-eyeOffsetX, eyeOffsetY, eyeSize, eyeSize * 1.2, 0, 0, Math.PI * 2);
        ctx.ellipse(eyeOffsetX, eyeOffsetY, eyeSize, eyeSize * 1.2, 0, 0, Math.PI * 2);
        ctx.fill();
        
        // 瞳孔
        ctx.fillStyle = '#000000';
        ctx.beginPath();
        ctx.arc(-eyeOffsetX + eyeSize * 0.2, eyeOffsetY, eyeSize * 0.5, 0, Math.PI * 2);
        ctx.arc(eyeOffsetX + eyeSize * 0.2, eyeOffsetY, eyeSize * 0.5, 0, Math.PI * 2);
        ctx.fill();
        
        // 眉毛
        ctx.strokeStyle = this.colors.dark;
        ctx.lineWidth = screenRadius * 0.15;
        ctx.lineCap = 'round';
        
        ctx.beginPath();
        ctx.moveTo(-eyeOffsetX - eyeSize, eyeOffsetY - eyeSize * 1.5);
        ctx.lineTo(-eyeOffsetX + eyeSize * 0.5, eyeOffsetY - eyeSize * 2);
        ctx.stroke();
        
        ctx.beginPath();
        ctx.moveTo(eyeOffsetX - eyeSize * 0.5, eyeOffsetY - eyeSize * 2);
        ctx.lineTo(eyeOffsetX + eyeSize, eyeOffsetY - eyeSize * 1.5);
        ctx.stroke();
        
        // 嘴巴
        ctx.fillStyle = '#FF8C00';
        ctx.beginPath();
        ctx.moveTo(screenRadius * 0.6, screenRadius * 0.3);
        ctx.lineTo(screenRadius * 1.1, screenRadius * 0.4);
        ctx.lineTo(screenRadius * 0.6, screenRadius * 0.6);
        ctx.closePath();
        ctx.fill();
        
        ctx.restore();
    }
}

class Pig {

        this.position = new Vector2(x, y);
        this.velocity = new Vector2(0, 0);
        this.size = size;
        
        const sizeMultipliers = { small: 0.7, medium: 1, large: 1.4 };
        this.radius = 20 * (sizeMultipliers[size] || 1);

        this.maxHealth = this.health;
        
        this.isDead = false;
        this.rotation = 0;
        this.angularVelocity = 0;
        
        this.blinkTimer = 0;
        this.isBlinking = false;
    }

    takeDamage(damage) {
        this.health -= damage;
        if (this.health <= 0) {
            this.isDead = true;
            return true;
        }
        return false;
    }

    update(deltaTime, groundY) {
        if (this.isDead) return;
        
        this.velocity.y += CONFIG.PHYSICS.GRAVITY;
        this.velocity.x *= 0.99;
        this.velocity.y *= 0.99;
        this.angularVelocity *= 0.95;
        
        this.position.x += this.velocity.x;
        this.position.y += this.velocity.y;
        this.rotation += this.angularVelocity;

        if (this.position.y + this.radius > groundY) {
            this.position.y = groundY - this.radius;
            this.velocity.y *= -0.3;
            this.velocity.x *= 0.9;
        }
        
        // 眨眼
        this.blinkTimer += deltaTime;
        if (this.blinkTimer > 3000) {
            this.isBlinking = true;
            if (this.blinkTimer > 3150) {
                this.isBlinking = false;
                this.blinkTimer = 0;
            }
        }
    }

    render(ctx, camera) {
        if (this.isDead) return;
        
        const screenX = (this.position.x - camera.x) * camera.zoom;
        const screenY = (this.position.y - camera.y) * camera.zoom;
        const screenRadius = this.radius * camera.zoom;
        
        ctx.save();
        ctx.translate(screenX, screenY);
        ctx.rotate(this.rotation);
        
        // 身体
        const gradient = ctx.createRadialGradient(
            -screenRadius * 0.3, -screenRadius * 0.3, 0,
            0, 0, screenRadius
        );
        gradient.addColorStop(0, '#90EE90');
        gradient.addColorStop(0.7, '#228B22');
        gradient.addColorStop(1, '#006400');
        
        ctx.fillStyle = gradient;
        ctx.beginPath();
        ctx.arc(0, 0, screenRadius, 0, Math.PI * 2);
        ctx.fill();
        
        // 耳朵
        ctx.fillStyle = '#228B22';
        ctx.beginPath();
        ctx.ellipse(-screenRadius * 0.7, -screenRadius * 0.7, screenRadius * 0.25, screenRadius * 0.3, -0.5, 0, Math.PI * 2);
        ctx.ellipse(screenRadius * 0.7, -screenRadius * 0.7, screenRadius * 0.25, screenRadius * 0.3, 0.5, 0, Math.PI * 2);
        ctx.fill();
        
        // 眼睛
        const eyeOffsetX = screenRadius * 0.35;
        const eyeOffsetY = -screenRadius * 0.1;
        const eyeSize = screenRadius * 0.3;
        
        ctx.fillStyle = '#FFFFFF';
        ctx.beginPath();
        ctx.arc(-eyeOffsetX, eyeOffsetY, eyeSize, 0, Math.PI * 2);
        ctx.arc(eyeOffsetX, eyeOffsetY, eyeSize, 0, Math.PI * 2);
        ctx.fill();
        
        // 瞳孔（除非在眨眼）
        if (!this.isBlinking) {
            ctx.fillStyle = '#000000';
            ctx.beginPath();
            ctx.arc(-eyeOffsetX, eyeOffsetY, eyeSize * 0.5, 0, Math.PI * 2);
            ctx.arc(eyeOffsetX, eyeOffsetY, eyeSize * 0.5, 0, Math.PI * 2);
            ctx.fill();
        } else {
            ctx.strokeStyle = '#000000';
            ctx.lineWidth = 2;
            ctx.beginPath();
            ctx.moveTo(-eyeOffsetX - eyeSize, eyeOffsetY);
            ctx.lineTo(-eyeOffsetX + eyeSize, eyeOffsetY);
            ctx.moveTo(eyeOffsetX - eyeSize, eyeOffsetY);
            ctx.lineTo(eyeOffsetX + eyeSize, eyeOffsetY);
            ctx.stroke();
        }
        
        // 鼻子
        ctx.fillStyle = '#006400';
        ctx.beginPath();
        ctx.ellipse(0, screenRadius * 0.3, screenRadius * 0.35, screenRadius * 0.25, 0, 0, Math.PI * 2);
        ctx.fill();
        
        // 鼻孔
        ctx.fillStyle = '#004400';
        ctx.beginPath();
        ctx.ellipse(-screenRadius * 0.15, screenRadius * 0.3, screenRadius * 0.08, screenRadius * 0.12, 0, 0, Math.PI * 2);
        ctx.ellipse(screenRadius * 0.15, screenRadius * 0.3, screenRadius * 0.08, screenRadius * 0.12, 0, 0, Math.PI * 2);
        ctx.fill();
        
        // 血条
        if (this.health < this.maxHealth) {
            const barWidth = screenRadius * 2;
            const barHeight = 4;
            const barY = -screenRadius - 10;
            
            ctx.fillStyle = 'rgba(0,0,0,0.5)';
            ctx.fillRect(-barWidth / 2, barY, barWidth, barHeight);
            
            const healthPercent = this.health / this.maxHealth;
            ctx.fillStyle = `hsl(${healthPercent * 120}, 70%, 50%)`;
            ctx.fillRect(-barWidth / 2, barY, barWidth * healthPercent, barHeight);
        }
        
        ctx.restore();
    }
}

    setupEventListeners() {
        // 鼠标事件
        this.canvas.addEventListener('mousedown', (e) => this.onMouseDown(e));
        this.canvas.addEventListener('mousemove', (e) => this.onMouseMove(e));
        this.canvas.addEventListener('mouseup', (e) => this.onMouseUp(e));

        this.canvas.addEventListener('touchstart', (e) => this.onTouchStart(e));
        this.canvas.addEventListener('touchmove', (e) => this.onTouchMove(e));
        this.canvas.addEventListener('touchend', (e) => this.onTouchEnd(e));

                    this.currentBird.colors = CONFIG.BIRD.COLORS[this.currentBirdType];
                }
            });
        });
    }

    getMousePos(e) {
        const rect = this.canvas.getBoundingClientRect();
        return {
            x: (e.clientX - rect.left) / this.camera.zoom + this.camera.x,
            y: (e.clientY - rect.top) / this.camera.zoom + this.camera.y
        };
    }

    onMouseDown(e) {
        const pos = this.getMousePos(e);
        
        // 检查是否点击了弹弓上的小鸟
        if (this.currentBird && !this.currentBird.isLaunched) {
            const dist = Utils.distance(
                pos.x, pos.y,
                this.currentBird.position.x, this.currentBird.position.y
            );
            
            if (dist < this.currentBird.radius * 2) {
                this.isDragging = true;
                this.dragStart = { ...this.currentBird.position };
                this.dragCurrent = { ...pos };
                this.currentBird.animation.setState('aiming');
                
                document.getElementById('powerIndicator').style.display = 'block';
            }
        }
        
        // 检查是否点击了方块（用于测试伤害）
        for (const block of this.blocks) {
            if (!block.isDestroyed) {
                const dx = pos.x - block.position.x;
                const dy = pos.y - block.position.y;
                if (Math.abs(dx) < block.width / 2 && Math.abs(dy) < block.height / 2) {
                    // 手动造成伤害（测试用）
                    this.applyDamage(block, 30, pos);
                    break;
                }
            }
        }
    }

    onMouseMove(e) {
        if (this.isDragging && this.currentBird) {
            const pos = this.getMousePos(e);
            
            // 限制拉伸距离
            const dx = pos.x - CONFIG.SLINGSHOT.X;
            const dy = pos.y - (this.groundY - 80);
            const dist = Math.sqrt(dx * dx + dy * dy);
            
            if (dist > CONFIG.SLINGSHOT.MAX_STRETCH) {
                const angle = Math.atan2(dy, dx);
                this.dragCurrent.x = CONFIG.SLINGSHOT.X + Math.cos(angle) * CONFIG.SLINGSHOT.MAX_STRETCH;
                this.dragCurrent.y = this.groundY - 80 + Math.sin(angle) * CONFIG.SLINGSHOT.MAX_STRETCH;
            } else {
                this.dragCurrent = { ...pos };
            }
            
            this.currentBird.position.x = this.dragCurrent.x;
            this.currentBird.position.y = this.dragCurrent.y;
            
            // 更新力量指示器
            this.launchPower = dist / CONFIG.SLINGSHOT.MAX_STRETCH;
            const powerFill = document.getElementById('powerFill');
            powerFill.style.width = `${this.launchPower * 100}%`;
            
            const hue = 120 - this.launchPower * 120;
            powerFill.style.background = `hsl(${hue}, 80%, 50%)`;
            
            // 播放拉伸音效
            if (Math.random() < 0.1) {
                this.soundManager.playStretch(dist);
            }
        }
    }

    onMouseUp(e) {
        if (this.isDragging && this.currentBird) {
            const dx = CONFIG.SLINGSHOT.X - this.dragCurrent.x;
            const dy = (this.groundY - 80) - this.dragCurrent.y;
            const power = Math.sqrt(dx * dx + dy * dy) * CONFIG.BIRD.LAUNCH_POWER;
            
            if (power > CONFIG.BIRD.MIN_POWER) {
                const angle = Math.atan2(dy, dx);
                const velocity = new Vector2(
                    Math.cos(angle) * Math.min(power, CONFIG.BIRD.MAX_POWER),
                    Math.sin(angle) * Math.min(power, CONFIG.BIRD.MAX_POWER)
                );
                
                this.currentBird.launch(velocity);
                this.soundManager.playLaunch(power);
            } else {
                // 力量太小，重置位置
                this.currentBird.position.x = CONFIG.SLINGSHOT.X;
                this.currentBird.position.y = this.groundY - 80;
                this.currentBird.animation.setState('idle');
            }
            
            this.isDragging = false;
            document.getElementById('powerIndicator').style.display = 'none';
        }
    }

    onTouchStart(e) {
        e.preventDefault();
        const touch = e.touches[0];
        this.onMouseDown({ clientX: touch.clientX, clientY: touch.clientY });
    }

    onTouchMove(e) {
        e.preventDefault();
        const touch = e.touches[0];
        this.onMouseMove({ clientX: touch.clientX, clientY: touch.clientY });
    }

    onTouchEnd(e) {
        e.preventDefault();
        this.onMouseUp(e);
    }

    onKeyDown(e) {
        if (e.code === 'Space' && this.currentBird && this.currentBird.isLaunched) {
            this.currentBird.useAbility();
            
            // 黑鸟爆炸
            if (this.currentBird.type === 'black') {
                this.triggerExplosion(
                    this.currentBird.position.x,
                    this.currentBird.position.y,
                    150
                );
                this.currentBird.isDead = true;
            }
        }
        
        if (e.code === 'KeyR') {
            this.resetGame();
        }
    }

    triggerExplosion(x, y, radius) {
        const tntConfig = CONFIG.MATERIALS.tnt;
        
        // 发射爆炸粒子
        this.particleSystem.emitExplosion(x, y, radius, tntConfig.particles);
        
        // 添加特效
        this.effectsSystem.addExplosion(x, y, 1.5);
        
        // 播放音效
        this.soundManager.playExplosion();
        
        // 对周围物体造成伤害
        for (const block of this.blocks) {
            if (block.isDestroyed) continue;
            
            const dist = Utils.distance(x, y, block.position.x, block.position.y);
            if (dist < radius) {
                const damage = tntConfig.explosionDamage * (1 - dist / radius);
                this.applyDamage(block, damage, { x, y });
                
                // 施加力
                const angle = Math.atan2(
                    block.position.y - y,
                    block.position.x - x
                );
                const force = (1 - dist / radius) * 20;
                block.velocity.x += Math.cos(angle) * force;
                block.velocity.y += Math.sin(angle) * force - 5;
                block.isSleeping = false;
            }
        }
        
        // 对绿猪造成伤害
        for (const pig of this.pigs) {
            if (pig.isDead) continue;
            
            const dist = Utils.distance(x, y, pig.position.x, pig.position.y);
            if (dist < radius) {
                const damage = 80 * (1 - dist / radius);
                const killed = pig.takeDamage(damage);
                
                if (killed) {
                    this.score += 5000;
                    this.effectsSystem.addDamageNumber(pig.position.x, pig.position.y - pig.radius, 5000, true);
                    this.particleSystem.emit(pig.position.x, pig.position.y, {
                        type: 'dust',
                        colors: ['#90EE90', '#228B22', '#006400'],
                        count: { min: 10, max: 20 },
                        size: { min: 3, max: 6 },
                        speed: { min: 2, max: 5 },
                        life: { min: 30, max: 60 },
                        gravity: 0.1
                    });
                }
                
                // 施加力
                const angle = Math.atan2(pig.position.y - y, pig.position.x - x);
                const force = (1 - dist / radius) * 15;
                pig.velocity.x += Math.cos(angle) * force;
                pig.velocity.y += Math.sin(angle) * force - 5;
            }
        }
    }

    applyDamage(block, damage, impactPoint) {
        const destroyed = block.takeDamage(damage);
        
        // 发射碰撞粒子
        this.particleSystem.emitCollision(
            impactPoint.x, impactPoint.y,
            damage,
            block.materialConfig
        );
        
        // 添加碰撞特效
        this.effectsSystem.addFlash(impactPoint.x, impactPoint.y, damage / 50);
        this.effectsSystem.addShockwave(impactPoint.x, impactPoint.y, damage / 80);
        
        // 显示伤害数字
        const points = Math.round(damage * 10);
        this.score += points;
        this.effectsSystem.addDamageNumber(
            block.position.x,
            block.position.y - block.height / 2,
            points,
            destroyed
        );
        
        // 播放碰撞音效
        this.soundManager.playCollision(damage, block.material);
        
        if (destroyed) {
            // 发射破坏粒子
            this.particleSystem.emit(
                block.position.x,
                block.position.y,
                {
                    ...block.materialConfig.particles,
                    count: {
                        min: block.materialConfig.particles.count.max,
                        max: block.materialConfig.particles.count.max * 1.5
                    }
                }
            );
            
            this.soundManager.playDestroy(block.material);
            
            // 如果是TNT，触发连锁爆炸
            if (block.material === 'tnt') {
                this.triggerExplosion(block.position.x, block.position.y, block.materialConfig.explosionRadius);
            }
            
            this.score += 1000;
            this.effectsSystem.addDamageNumber(
                block.position.x,
                block.position.y,
                1000,
                true
            );
        }
    }

    checkCollisions() {
        if (!this.currentBird || !this.currentBird.isLaunched || this.currentBird.isDead) return;
        
        const bird = this.currentBird;
        
        // 小鸟与方块碰撞
        for (const block of this.blocks) {
            if (block.isDestroyed) continue;
            
            // 简单的圆-矩形碰撞检测
            const closestX = Utils.clamp(bird.position.x, 
                block.position.x - block.width / 2, 
                block.position.x + block.width / 2);
            const closestY = Utils.clamp(bird.position.y,
                block.position.y - block.height / 2,
                block.position.y + block.height / 2);
            
            const dist = Utils.distance(bird.position.x, bird.position.y, closestX, closestY);
            
            if (dist < bird.radius) {
                // 碰撞发生
                const relativeVelocity = Math.sqrt(bird.velocity.x ** 2 + bird.velocity.y ** 2);
                const impulse = relativeVelocity * bird.mass * 2;
                
                // 计算伤害
                const damage = impulse * (bird.type === 'black' ? 2 : bird.type === 'yellow' ? 1.5 : 1);
                
                // 应用伤害
                this.applyDamage(block, damage, { x: closestX, y: closestY });
                
                // 反弹
                const nx = (bird.position.x - closestX) / dist || 0;
                const ny = (bird.position.y - closestY) / dist || 0;
                
                const restitution = CONFIG.MATERIALS[block.material].restitution;
                bird.velocity.x = -bird.velocity.x * restitution + block.velocity.x;
                bird.velocity.y = -bird.velocity.y * restitution + block.velocity.y;
                
                // 分离
                bird.position.x = closestX + nx * bird.radius;
                bird.position.y = closestY + ny * bird.radius;
                
                // 方块也受力
                block.velocity.x += bird.velocity.x * 0.3;
                block.velocity.y += bird.velocity.y * 0.3;
                block.isSleeping = false;
            }
        }
        
        // 小鸟与绿猪碰撞
        for (const pig of this.pigs) {
            if (pig.isDead) continue;
            
            const dist = Utils.distance(
                bird.position.x, bird.position.y,
                pig.position.x, pig.position.y
            );
            
            if (dist < bird.radius + pig.radius) {
                const relativeVelocity = Math.sqrt(bird.velocity.x ** 2 + bird.velocity.y ** 2);
                const damage = relativeVelocity * bird.mass * 5;
                
                const killed = pig.takeDamage(damage);
                
                if (killed) {
                    this.score += 5000;
                    this.effectsSystem.addDamageNumber(pig.position.x, pig.position.y - pig.radius, 5000, true);
                    
                    // 绿猪死亡粒子
                    this.particleSystem.emit(pig.position.x, pig.position.y, {
                        type: 'dust',
                        colors: ['#90EE90', '#228B22', '#006400'],
                        count: { min: 15, max: 25 },
                        size: { min: 4, max: 8 },
                        speed: { min: 3, max: 7 },
                        life: { min: 40, max: 80 },
                        gravity: 0.12
                    });
                } else {
                    this.effectsSystem.addDamageNumber(pig.position.x, pig.position.y - pig.radius, damage * 10, false);
                }
                
                // 碰撞特效
                this.effectsSystem.addFlash(pig.position.x, pig.position.y, 0.5);
                
                // 反弹
                const nx = (bird.position.x - pig.position.x) / dist;
                const ny = (bird.position.y - pig.position.y) / dist;
                
                pig.velocity.x += bird.velocity.x * 0.5;
                pig.velocity.y += bird.velocity.y * 0.5;
                
                bird.velocity.x *= -0.5;
                bird.velocity.y *= -0.5;
            }
        }
        
        // 方块之间的碰撞（简化版）
        for (let i = 0; i < this.blocks.length; i++) {
            for (let j = i + 1; j < this.blocks.length; j++) {
                const a = this.blocks[i];
                const b = this.blocks[j];
                
                if (a.isDestroyed || b.isDestroyed) continue;
                
                // AABB碰撞检测
                if (Math.abs(a.position.x - b.position.x) < (a.width + b.width) / 2 &&
                    Math.abs(a.position.y - b.position.y) < (a.height + b.height) / 2) {
                    
                    // 计算冲量
                    const impulse = a.calculateImpulse(b);
                    
                    // 如果冲量足够大，造成伤害
                    if (impulse > a.materialConfig.destructionThreshold && a.collisionCooldown <= 0) {
                        const damage = impulse * 0.5;
                        const impactPoint = a.getImpactPoint(b);
                        this.applyDamage(a, damage, impactPoint);
                        a.collisionCooldown = 200;
                    }
                    
                    if (impulse > b.materialConfig.destructionThreshold && b.collisionCooldown <= 0) {
                        const damage = impulse * 0.5;
                        const impactPoint = b.getImpactPoint(a);
                        this.applyDamage(b, damage, impactPoint);
                        b.collisionCooldown = 200;
                    }
                    
                    // 简单分离
                    const dx = b.position.x - a.position.x;
                    const dy = b.position.y - a.position.y;
                    const overlapX = (a.width + b.width) / 2 - Math.abs(dx);
                    const overlapY = (a.height + b.height) / 2 - Math.abs(dy);
                    
                    if (overlapX < overlapY) {
                        const sign = dx > 0 ? 1 : -1;
                        a.position.x -= overlapX * 0.5 * sign;
                        b.position.x += overlapX * 0.5 * sign;
                        
                        const avgVx = (a.velocity.x + b.velocity.x) * 0.5;
                        a.velocity.x = avgVx - overlapX * sign * 0.1;
                        b.velocity.x = avgVx + overlapX * sign * 0.1;
                    } else {
                        const sign = dy > 0 ? 1 : -1;
                        a.position.y -= overlapY * 0.5 * sign;
                        b.position.y += overlapY * 0.5 * sign;
                        
                        const avgVy = (a.velocity.y + b.velocity.y) * 0.5;
                        a.velocity.y = avgVy - overlapY * sign * 0.1;
                        b.velocity.y = avgVy + overlapY * sign * 0.1;
                    }
                    
                    a.isSleeping = false;
                    b.isSleeping = false;
                }
            }
        }
        
        // 方块与绿猪碰撞
        for (const block of this.blocks) {
            if (block.isDestroyed) continue;
            
            for (const pig of this.pigs) {
                if (pig.isDead) continue;
                
                const closestX = Utils.clamp(pig.position.x,
                    block.position.x - block.width / 2,
                    block.position.x + block.width / 2);
                const closestY = Utils.clamp(pig.position.y,
                    block.position.y - block.height / 2,
                    block.position.y + block.height / 2);
                
                const dist = Utils.distance(pig.position.x, pig.position.y, closestX, closestY);
                
                if (dist < pig.radius) {
                    const speed = Math.sqrt(block.velocity.x ** 2 + block.velocity.y ** 2);
                    if (speed > 2) {
                        const damage = speed * 10;
                        const killed = pig.takeDamage(damage);
                        
                        if (killed) {
                            this.score += 5000;
                            this.effectsSystem.addDamageNumber(pig.position.x, pig.position.y - pig.radius, 5000, true);
                        } else {
                            this.effectsSystem.addDamageNumber(pig.position.x, pig.position.y - pig.radius, damage * 10, false);
                        }
                    }
                    
                    // 推开绿猪
                    const nx = (pig.position.x - closestX) / (dist || 1);
                    const ny = (pig.position.y - closestY) / (dist || 1);
                    
                    pig.position.x = closestX + nx * pig.radius;
                    pig.position.y = closestY + ny * pig.radius;
                    pig.velocity.x += block.velocity.x * 0.5;
                    pig.velocity.y += block.velocity.y * 0.5;
                }
            }
        }
    }

    updateCamera() {
        if (this.currentBird && this.currentBird.isLaunched && !this.currentBird.isDead) {
            // 跟随小鸟
            this.camera.targetX = this.currentBird.position.x - CONFIG.CANVAS_WIDTH / 3;
        } else {
            // 返回弹弓位置
            this.camera.targetX = 0;
        }
        
        // 平滑移动
        this.camera.x += (this.camera.targetX - this.camera.x) * CONFIG.CAMERA_LERP;
        
        // 限制相机范围
        this.camera.x = Utils.clamp(this.camera.x, 0, CONFIG.WORLD_WIDTH - CONFIG.CANVAS_WIDTH);
    }

    checkBirdReset() {
        if (this.currentBird && this.currentBird.isLaunched) {
            const speed = Math.sqrt(
                this.currentBird.velocity.x ** 2 + 
                this.currentBird.velocity.y ** 2
            );
            
            // 小鸟停止或出界
            if (this.currentBird.isDead || 
                (speed < 0.5 && this.currentBird.position.y > this.groundY - 50)) {
                
                setTimeout(() => {
                    this.birdsLeft--;
                    document.getElementById('birdsValue').textContent = this.birdsLeft;
                    
                    if (this.birdsLeft > 0) {
                        this.createBird();
                    } else {
                        // 游戏结束
                        this.currentBird = null;
                    }
                }, 1000);
                
                this.currentBird.isDead = true; // 防止重复触发
            }
        }
    }

    resetGame() {
        this.score = 0;
        this.birdsLeft = 3;
        document.getElementById('scoreValue').textContent = '0';
        document.getElementById('birdsValue').textContent = '3';
        
        this.blocks = [];
        this.pigs = [];
        this.particleSystem = new ParticleSystem(CONFIG.PARTICLES.MAX_PARTICLES);
        this.effectsSystem = new EffectsSystem();
        
        this.createLevel();
        this.createBird();
        
        this.camera.x = 0;
        this.camera.targetX = 0;
    }

    update(deltaTime) {
        // 更新当前小鸟
        if (this.currentBird) {
            this.currentBird.update(deltaTime, this.groundY);
        }
        
        // 更新方块
        for (const block of this.blocks) {
            block.update(deltaTime, this.groundY);
        }
        
        // 更新绿猪
        for (const pig of this.pigs) {
            pig.update(deltaTime, this.groundY);
        }
        
        // 检测碰撞
        this.checkCollisions();
        
        // 更新粒子
        this.particleSystem.update();
        
        // 更新特效
        this.effectsSystem.update();
        
        // 更新相机
        this.updateCamera();
        
        // 检查小鸟是否需要重置
        this.checkBirdReset();
        
        // 更新分数显示
        document.getElementById('scoreValue').textContent = this.score.toLocaleString();
        
        // 更新调试面板
        this.frameCount++;
        this.fpsTimer += deltaTime;
        if (this.fpsTimer >= 1000) {
            this.fps = this.frameCount;
            this.frameCount = 0;
            this.fpsTimer = 0;
        }
        
        const activeBlocks = this.blocks.filter(b => !b.isDestroyed).length;
        const activePigs = this.pigs.filter(p => !p.isDead).length;
        document.getElementById('debugPanel').textContent = 
            `FPS: ${this.fps} | Particles: ${this.particleSystem.activeCount} | Blocks: ${activeBlocks} | Pigs: ${activePigs}`;
    }

    render() {
        const ctx = this.ctx;
        
        // 清空画布
        ctx.fillStyle = '#1a1a2e';
        ctx.fillRect(0, 0, CONFIG.CANVAS_WIDTH, CONFIG.CANVAS_HEIGHT);
        
        // 绘制背景
        this.renderBackground();
        
        // 绘制地面
        this.renderGround();
        
        // 绘制弹弓后部
        this.renderSlingshotBack();
        
        // 绘制轨迹预测
        if (this.isDragging && this.currentBird) {
            this.renderTrajectory();
        }
        
        // 绘制方块
        for (const block of this.blocks) {
            block.render(ctx, this.camera);
        }
        
        // 绘制绿猪
        for (const pig of this.pigs) {
            pig.render(ctx, this.camera);
        }
        
        // 绘制当前小鸟
        if (this.currentBird) {
            this.currentBird.render(ctx, this.camera);
        }
        
        // 绘制弹弓前部和皮筋
        this.renderSlingshotFront();
        
        // 绘制粒子
        this.particleSystem.render(ctx, this.camera);
        
        // 绘制特效
        this.effectsSystem.render(ctx, this.camera);
    }

    renderBackground() {
        const ctx = this.ctx;
        
        // 天空渐变
        const skyGradient = ctx.createLinearGradient(0, 0, 0, CONFIG.CANVAS_HEIGHT);
        skyGradient.addColorStop(0, '#0f0c29');
        skyGradient.addColorStop(0.5, '#302b63');
        skyGradient.addColorStop(1, '#24243e');
        ctx.fillStyle = skyGradient;
        ctx.fillRect(0, 0, CONFIG.CANVAS_WIDTH, CONFIG.CANVAS_HEIGHT);
        
        // 星星
        ctx.fillStyle = 'rgba(255,255,255,0.5)';
        for (let i = 0; i < 50; i++) {
            const x = (Utils.noise(i * 100) * CONFIG.CANVAS_WIDTH + this.camera.x * 0.1) % CONFIG.CANVAS_WIDTH;
            const y = Utils.noise(i * 200) * (this.groundY - 100);
            const size = Utils.noise(i * 300) * 2 + 0.5;
            
            ctx.beginPath();
            ctx.arc(x, y, size, 0, Math.PI * 2);
            ctx.fill();
        }
        
        // 远山
        ctx.fillStyle = 'rgba(100, 100, 150, 0.3)';
        ctx.beginPath();
        ctx.moveTo(0, this.groundY);
        for (let x = 0; x <= CONFIG.CANVAS_WIDTH; x += 50) {
            const worldX = x + this.camera.x * 0.2;
            const y = this.groundY - 80 - Math.sin(worldX * 0.01) * 40 - Utils.noise(worldX * 0.005) * 30;
            ctx.lineTo(x, y);
        }
        ctx.lineTo(CONFIG.CANVAS_WIDTH, this.groundY);
        ctx.closePath();
        ctx.fill();
        
        // 近山
        ctx.fillStyle = 'rgba(80, 80, 120, 0.4)';
        ctx.beginPath();
        ctx.moveTo(0, this.groundY);
        for (let x = 0; x <= CONFIG.CANVAS_WIDTH; x += 30) {
            const worldX = x + this.camera.x * 0.4;
            const y = this.groundY - 50 - Math.sin(worldX * 0.015) * 25 - Utils.noise(worldX * 0.008) * 20;
            ctx.lineTo(x, y);
        }
        ctx.lineTo(CONFIG.CANVAS_WIDTH, this.groundY);
        ctx.closePath();
        ctx.fill();
    }

    renderGround() {
        const ctx = this.ctx;
        const groundScreenY = this.groundY;
        
        // 草地
        const grassGradient = ctx.createLinearGradient(0, groundScreenY, 0, CONFIG.CANVAS_HEIGHT);
        grassGradient.addColorStop(0, '#228B22');
        grassGradient.addColorStop(0.3, '#1e7b1e');
        grassGradient.addColorStop(1, '#145214');
        
        ctx.fillStyle = grassGradient;
        ctx.fillRect(0, groundScreenY, CONFIG.CANVAS_WIDTH, CONFIG.GROUND_HEIGHT);
        
        // 草地纹理
        ctx.strokeStyle = '#2d9e2d';
        ctx.lineWidth = 2;
        for (let x = 0; x < CONFIG.CANVAS_WIDTH; x += 15) {
            const offset = Utils.noise(x * 0.1 + this.camera.x * 0.1) * 5;
            ctx.beginPath();
            ctx.moveTo(x, groundScreenY);
            ctx.lineTo(x + offset, groundScreenY - 8 - Utils.noise(x) * 5);
            ctx.stroke();
        }
        
        // 泥土层
        ctx.fillStyle = '#8B4513';
        ctx.fillRect(0, groundScreenY + 30, CONFIG.CANVAS_WIDTH, CONFIG.GROUND_HEIGHT - 30);
    }

    renderSlingshotBack() {
        const ctx = this.ctx;
        const baseX = CONFIG.SLINGSHOT.X - this.camera.x;
        const baseY = this.groundY;
        
        // 后支架
        ctx.fillStyle = '#5D4037';
        ctx.beginPath();
        ctx.moveTo(baseX - 20, baseY);
        ctx.lineTo(baseX - 35, baseY - 120);
        ctx.lineTo(baseX - 25, baseY - 120);
        ctx.lineTo(baseX - 10, baseY);
        ctx.closePath();
        ctx.fill();
        
        ctx.strokeStyle = '#3E2723';
        ctx.lineWidth = 2;
        ctx.stroke();
    }

    renderSlingshotFront() {
        const ctx = this.ctx;
        const baseX = CONFIG.SLINGSHOT.X - this.camera.x;
        const baseY = this.groundY;
        
        // 皮筋
        if (this.isDragging && this.currentBird) {
            const birdScreenX = this.currentBird.position.x - this.camera.x;
            const birdScreenY = this.currentBird.position.y - this.camera.y;
            
            // 后皮筋
            ctx.strokeStyle = CONFIG.SLINGSHOT.BAND_STRETCH_COLOR;
            ctx.lineWidth = CONFIG.SLINGSHOT.BAND_WIDTH;
            ctx.lineCap = 'round';
            
            ctx.beginPath();
            ctx.moveTo(baseX - 30, baseY - 110);
            ctx.lineTo(birdScreenX, birdScreenY);
            ctx.stroke();
            
            // 前皮筋
            ctx.beginPath();
            ctx.moveTo(birdScreenX, birdScreenY);
            ctx.lineTo(baseX + 10, baseY - 110);
            ctx.stroke();
        } else {
            // 静止状态的皮筋
            ctx.strokeStyle = CONFIG.SLINGSHOT.BAND_COLOR;
            ctx.lineWidth = CONFIG.SLINGSHOT.BAND_WIDTH;
            ctx.lineCap = 'round';
            
            ctx.beginPath();
            ctx.moveTo(baseX - 30, baseY - 110);
            ctx.quadraticCurveTo(baseX - 10, baseY - 95, baseX + 10, baseY - 110);
            ctx.stroke();
        }
        
        // 前支架
        ctx.fillStyle = '#6D4C41';
        ctx.beginPath();
        ctx.moveTo(baseX + 5, baseY);
        ctx.lineTo(baseX + 25, baseY - 120);
        ctx.lineTo(baseX + 15, baseY - 120);
        ctx.lineTo(baseX - 5, baseY);
        ctx.closePath();
        ctx.fill();
        
        ctx.strokeStyle = '#3E2723';
        ctx.lineWidth = 2;
        ctx.stroke();
        
        // Y形顶部
        ctx.fillStyle = '#5D4037';
        ctx.beginPath();
        ctx.arc(baseX - 30, baseY - 115, 8, 0, Math.PI * 2);
        ctx.fill();
        
        ctx.beginPath();
        ctx.arc(baseX + 15, baseY - 115, 8, 0, Math.PI * 2);
        ctx.fill();
    }

    renderTrajectory() {
        const ctx = this.ctx;
        
        const dx = CONFIG.SLINGSHOT.X - this.dragCurrent.x;
        const dy = (this.groundY - 80) - this.dragCurrent.y;
        const power = Math.sqrt(dx * dx + dy * dy) * CONFIG.BIRD.LAUNCH_POWER;
        
        if (power < CONFIG.BIRD.MIN_POWER) return;
        
        const angle = Math.atan2(dy, dx);
        let vx = Math.cos(angle) * Math.min(power, CONFIG.BIRD.MAX_POWER);
        let vy = Math.sin(angle) * Math.min(power, CONFIG.BIRD.MAX_POWER);
        
        let x = this.dragCurrent.x;
        let y = this.dragCurrent.y;
        
        const points = CONFIG.TRAJECTORY.POINT_COUNT;
        
        for (let i = 0; i < points; i++) {
            vy += CONFIG.PHYSICS.GRAVITY;
            x += vx;
            y += vy;
            
            if (y > this.groundY) break;
            
            const screenX = (x - this.camera.x) * this.camera.zoom;
            const screenY = (y - this.camera.y) * this.camera.zoom;
            
            const progress = i / points;
            const size = Utils.lerp(CONFIG.TRAJECTORY.POINT_SIZE_START, CONFIG.TRAJECTORY.POINT_SIZE_END, progress);
            const alpha = 1 - progress;
            
            ctx.fillStyle = `rgba(255, 255, 255, ${alpha * 0.6})`;
            ctx.beginPath();
            ctx.arc(screenX, screenY, size, 0, Math.PI * 2);
            ctx.fill();
        }
    }

    gameLoop(timestamp) {
        const deltaTime = timestamp - this.lastTime;
        this.lastTime = timestamp;
        
        this.update(Math.min(deltaTime, 32)); // 限制最大帧时间
        this.render();
        
        requestAnimationFrame((t) => this.gameLoop(t));
    }
}

// 启动游戏
window.addEventListener('load', () => {
    new Game();
});
</script>
</body>
</html>

## 主要实现内容

### 1. 材质差异化系统
- **四种材质**：木头、玻璃、石头、TNT
- 每种材质有不同的：
  - 血量
  - 破坏阈值
  - 密度、弹性、摩擦力
  - 专属颜色方案

### 2. 粒子系统
- **木屑**（棕色方块碎片）- 长方形粒子，带旋转
- **玻璃碎片**（蓝色闪光）- 三角形碎片，带闪烁效果
- **石尘**（灰色烟雾）- 圆形粒子，带淡出效果
- **爆炸火焰**（红黄渐变）- 辐射渐变，带发光

### 3. 碰撞特效系统
- **白色闪光** - 碰撞瞬间的径向渐变闪光
- **冲击波** - 扩散的圆环效果
- **伤害数字** - 浮动显示的伤害值
- **TNT爆炸** - 连锁爆炸效果

### 4. 伤害计算
- 基于冲量（速度 × 质量）计算伤害
- 不同材质有不同的伤害系数
- 击杀时有特殊视觉反馈

### 操作说明
- **拖拽**小鸟瞄准发射
- **空格**使用小鸟技能
- **R**重置游戏
- **点击方块**测试伤害效果
