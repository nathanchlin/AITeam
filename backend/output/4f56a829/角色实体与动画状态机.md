const Utils = {
    lerp: (a, b, t) => a + (b - a) * t,
    clamp: (val, min, max) => Math.max(min, Math.min(max, val)),
    randomRange: (min, max) => Math.random() * (max - min) + min,
    randomInt: (min, max) => Math.floor(Utils.randomRange(min, max + 1)),
    distance: (x1, y1, x2, y2) => Math.sqrt((x2 - x1) ** 2 + (y2 - y1) ** 2),
    
    safeColor: (r, g, b, a = 1) => {
        r = Utils.clamp(Math.round(r), 0, 255);
        g = Utils.clamp(Math.round(g), 0, 255);
        b = Utils.clamp(Math.round(b), 0, 255);
        a = Utils.clamp(a, 0, 1);
        return `rgba(${r},${g},${b},${a})`;
    },
    
    noise: (x) => {
        const n = Math.sin(x * 12.9898) * 43758.5453;
        return n - Math.floor(n);
    },
    
    easeOutCubic: (t) => 1 - Math.pow(1 - t, 3),
    easeOutElastic: (t) => {
        const c4 = (2 * Math.PI) / 3;
        return t === 0 ? 0 : t === 1 ? 1 : Math.pow(2, -10 * t) * Math.sin((t * 10 - 0.75) * c4) + 1;
    },
    
    easeOutBack: (t) => {
        const c1 = 1.70158;
        const c3 = c1 + 1;
        return 1 + c3 * Math.pow(t - 1, 3) + c1 * Math.pow(t - 1, 2);
    }
};

class Vector2 {
    constructor(x = 0, y = 0) {
        this.x = x;
        this.y = y;
    }

    add(v) { return new Vector2(this.x + v.x, this.y + v.y); }
    sub(v) { return new Vector2(this.x - v.x, this.y - v.y); }
    mul(s) { return new Vector2(this.x * s, this.y * s); }
    dot(v) { return this.x * v.x + this.y * v.y; }
    cross(v) { return this.x * v.y - this.y * v.x; }
    length() { return Math.sqrt(this.x * this.x + this.y * this.y); }
    lengthSq() { return this.x * this.x + this.y * this.y; }

    normalize() {
        const len = this.length();
        if (len === 0) return new Vector2(0, 0);
        return new Vector2(this.x / len, this.y / len);
    }

    rotate(angle) {
        const cos = Math.cos(angle);
        const sin = Math.sin(angle);
        return new Vector2(this.x * cos - this.y * sin, this.x * sin + this.y * cos);
    }

    clone() { return new Vector2(this.x, this.y); }
    set(x, y) { this.x = x; this.y = y; return this; }

    static fromAngle(angle, length = 1) {
        return new Vector2(Math.cos(angle) * length, Math.sin(angle) * length);
    }
}

class AnimationStateMachine {
    constructor() {
        this.currentState = 'idle';
        this.previousState = null;
        this.stateTime = 0;
        this.transitionTime = 0;
        this.isTransitioning = false;
        
        // 状态配置
        this.states = {
            idle: {
                duration: Infinity,
                loop: true,
                squashX: 1, squashY: 1,
                bobSpeed: 0.05, bobAmount: 2
            },
            aiming: {
                duration: Infinity,
                loop: true,
                squashX: 1.15, squashY: 0.85,
                stretchDirection: true
            },
            flying: {
                duration: Infinity,
                loop: true,
                squashX: 0.85, squashY: 1.2,
                rotateToVelocity: true
            },
            ability: {
                duration: 500,
                loop: false,
                squashX: 0.7, squashY: 1.4,
                flash: true
            },
            hit: {
                duration: 300,
                loop: false,
                squashX: 1.3, squashY: 0.7,
                shake: true
            },
            falling: {
                duration: Infinity,
                loop: true,
                squashX: 0.9, squashY: 1.1,
                rotateToVelocity: true
            }
        };
    }

    setState(newState, force = false) {
        if (newState === this.currentState && !force) return;
        
        this.previousState = this.currentState;
        this.currentState = newState;
        this.stateTime = 0;
        this.isTransitioning = true;
        this.transitionTime = 0;
    }

    update(deltaTime) {
        this.stateTime += deltaTime;
        this.transitionTime = Math.min(this.transitionTime + deltaTime * 5, 1);
        
        if (this.transitionTime >= 1) {
            this.isTransitioning = false;
        }
        
        const state = this.states[this.currentState];
        if (!state.loop && this.stateTime >= state.duration) {
            this.setState('falling');
        }
    }

    getCurrentState() {
        return this.states[this.currentState];
    }

    getAnimationData(velocity = { x: 0, y: 0 }) {
        const state = this.states[this.currentState];
        const time = this.stateTime / 1000;
        
        let scaleX = state.squashX || 1;
        let scaleY = state.squashY || 1;
        let rotation = 0;
        let bobOffset = 0;
        
        // 闲置状态 - 上下浮动
        if (this.currentState === 'idle') {
            bobOffset = Math.sin(time * state.bobSpeed * 60) * state.bobAmount;
        }
        
        // 瞄准状态 - 呼吸效果
        if (this.currentState === 'aiming') {
            const breathe = Math.sin(time * 4) * 0.05;
            scaleX += breathe;
            scaleY -= breathe * 0.5;
        }
        
        // 飞行状态 - 根据速度旋转
        if (state.rotateToVelocity) {
            const speed = Math.sqrt(velocity.x ** 2 + velocity.y ** 2);
            if (speed > 0.5) {
                rotation = Math.atan2(velocity.y, velocity.x);
                
                // 速度越快，拉伸越明显
                const stretchFactor = Math.min(speed / 15, 0.3);
                scaleX = 0.85 - stretchFactor * 0.2;
                scaleY = 1.2 + stretchFactor * 0.3;
            }
        }
        
        // 技能状态 - 脉冲效果
        if (this.currentState === 'ability') {
            const pulse = Math.sin(time * 30) * 0.2;
            scaleX += pulse;
            scaleY += pulse;
        }
        
        // 击中状态 - 震动
        if (state.shake) {
            const shake = Math.sin(time * 80) * 0.15;
            scaleX += shake;
            rotation += shake * 0.5;
        }
        
        // 过渡插值
        if (this.isTransitioning) {
            const t = Utils.easeOutCubic(this.transitionTime);
            scaleX = Utils.lerp(1, scaleX, t);
            scaleY = Utils.lerp(1, scaleY, t);
        }
        
        return {
            scaleX,
            scaleY,
            rotation,
            bobOffset,
            flash: state.flash && Math.sin(time * 20) > 0
        };
    }
}

class SoundManager {
    constructor() {
        this.audioContext = null;
        this.enabled = true;
        this.initialized = false;
    }

    init() {
        if (this.initialized) return;
        try {
            this.audioContext = new (window.AudioContext || window.webkitAudioContext)();
            this.initialized = true;
        } catch (e) {
            console.warn('Web Audio API not supported');
            this.enabled = false;
        }
    }

    playLaunch(power) {
        if (!this.enabled || !this.audioContext) return;
        
        const oscillator = this.audioContext.createOscillator();
        const gainNode = this.audioContext.createGain();
        
        oscillator.connect(gainNode);
        gainNode.connect(this.audioContext.destination);
        
        const baseFreq = 200 + power * 8;
        oscillator.frequency.setValueAtTime(baseFreq, this.audioContext.currentTime);
        oscillator.frequency.exponentialRampToValueAtTime(80, this.audioContext.currentTime + 0.3);
        oscillator.type = 'sine';
        
        gainNode.gain.setValueAtTime(0.3, this.audioContext.currentTime);
        gainNode.gain.exponentialRampToValueAtTime(0.01, this.audioContext.currentTime + 0.3);
        
        oscillator.start(this.audioContext.currentTime);
        oscillator.stop(this.audioContext.currentTime + 0.3);
    }

    playStretch(power) {
        if (!this.enabled || !this.audioContext) return;
        
        const oscillator = this.audioContext.createOscillator();
        const gainNode = this.audioContext.createGain();
        
        oscillator.connect(gainNode);
        gainNode.connect(this.audioContext.destination);
        
        oscillator.frequency.setValueAtTime(150 + power * 2, this.audioContext.currentTime);
        oscillator.type = 'triangle';
        
        gainNode.gain.setValueAtTime(0.05, this.audioContext.currentTime);
        gainNode.gain.exponentialRampToValueAtTime(0.01, this.audioContext.currentTime + 0.1);
        
        oscillator.start(this.audioContext.currentTime);
        oscillator.stop(this.audioContext.currentTime + 0.1);
    }

    playBounce() {
        if (!this.enabled || !this.audioContext) return;
        
        const oscillator = this.audioContext.createOscillator();
        const gainNode = this.audioContext.createGain();
        
        oscillator.connect(gainNode);
        gainNode.connect(this.audioContext.destination);
        
        oscillator.frequency.setValueAtTime(300, this.audioContext.currentTime);
        oscillator.frequency.exponentialRampToValueAtTime(150, this.audioContext.currentTime + 0.1);
        oscillator.type = 'sine';
        
        gainNode.gain.setValueAtTime(0.2, this.audioContext.currentTime);
        gainNode.gain.exponentialRampToValueAtTime(0.01, this.audioContext.currentTime + 0.1);
        
        oscillator.start(this.audioContext.currentTime);
        oscillator.stop(this.audioContext.currentTime + 0.1);
    }

    playExplosion() {
        if (!this.enabled || !this.audioContext) return;
        
        const bufferSize = this.audioContext.sampleRate * 0.3;
        const buffer = this.audioContext.createBuffer(1, bufferSize, this.audioContext.sampleRate);
        const data = buffer.getChannelData(0);
        
        for (let i = 0; i < bufferSize; i++) {
            data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (bufferSize * 0.1));
        }
        
        const source = this.audioContext.createBufferSource();
        const gainNode = this.audioContext.createGain();
        source.buffer = buffer;
        source.connect(gainNode);
        gainNode.connect(this.audioContext.destination);
        gainNode.gain.setValueAtTime(0.4, this.audioContext.currentTime);
        
        source.start();
    }

    playAbility() {
        if (!this.enabled || !this.audioContext) return;
        
        const oscillator = this.audioContext.createOscillator();
        const gainNode = this.audioContext.createGain();
        
        oscillator.connect(gainNode);
        gainNode.connect(this.audioContext.destination);
        
        oscillator.frequency.setValueAtTime(400, this.audioContext.currentTime);
        oscillator.frequency.exponentialRampToValueAtTime(800, this.audioContext.currentTime + 0.15);
        oscillator.type = 'square';
        
        gainNode.gain.setValueAtTime(0.2, this.audioContext.currentTime);
        gainNode.gain.exponentialRampToValueAtTime(0.01, this.audioContext.currentTime + 0.15);
        
        oscillator.start(this.audioContext.currentTime);
        oscillator.stop(this.audioContext.currentTime + 0.15);
    }
}

class InputManager {
    constructor(canvas) {
        this.canvas = canvas;
        this.keys = {};
        this.mouse = {
            x: 0, y: 0,
            worldX: 0, worldY: 0,
            isDown: false, isDragging: false
        };
        this.dragStart = null;
        this.bindEvents();
    }

    bindEvents() {
        window.addEventListener('keydown', (e) => {
            this.keys[e.code] = true;
            if (e.code === 'Space') e.preventDefault();
        });

        window.addEventListener('keyup', (e) => {
            this.keys[e.code] = false;
        });

        this.canvas.addEventListener('mousedown', (e) => {
            const rect = this.canvas.getBoundingClientRect();
            this.mouse.x = e.clientX - rect.left;
            this.mouse.y = e.clientY - rect.top;
            this.mouse.isDown = true;
            this.dragStart = { x: this.mouse.x, y: this.mouse.y };
        });

        this.canvas.addEventListener('mouseup', () => {
            this.mouse.isDown = false;
            this.mouse.isDragging = false;
            this.dragStart = null;
        });

        this.canvas.addEventListener('mousemove', (e) => {
            const rect = this.canvas.getBoundingClientRect();
            this.mouse.x = e.clientX - rect.left;
            this.mouse.y = e.clientY - rect.top;

            if (this.mouse.isDown && this.dragStart) {
                const dx = this.mouse.x - this.dragStart.x;
                const dy = this.mouse.y - this.dragStart.y;
                if (Math.abs(dx) > 5 || Math.abs(dy) > 5) {
                    this.mouse.isDragging = true;
                }
            }
        });

        // 触摸支持
        this.canvas.addEventListener('touchstart', (e) => {
            e.preventDefault();
            const touch = e.touches[0];
            const rect = this.canvas.getBoundingClientRect();
            this.mouse.x = touch.clientX - rect.left;
            this.mouse.y = touch.clientY - rect.top;
            this.mouse.isDown = true;
            this.dragStart = { x: this.mouse.x, y: this.mouse.y };
        });

        this.canvas.addEventListener('touchend', (e) => {
            e.preventDefault();
            this.mouse.isDown = false;
            this.mouse.isDragging = false;
            this.dragStart = null;
        });

        this.canvas.addEventListener('touchmove', (e) => {
            e.preventDefault();
            const touch = e.touches[0];
            const rect = this.canvas.getBoundingClientRect();
            this.mouse.x = touch.clientX - rect.left;
            this.mouse.y = touch.clientY - rect.top;

            if (this.mouse.isDown && this.dragStart) {
                const dx = this.mouse.x - this.dragStart.x;
                const dy = this.mouse.y - this.dragStart.y;
                if (Math.abs(dx) > 5 || Math.abs(dy) > 5) {
                    this.mouse.isDragging = true;
                }
            }
        });
    }

    isKeyPressed(code) {
        return this.keys[code] === true;
    }

    updateWorldPosition(camera) {
        this.mouse.worldX = (this.mouse.x - CONFIG.CANVAS_WIDTH / 2) / camera.zoom + camera.x;
        this.mouse.worldY = (this.mouse.y - CONFIG.CANVAS_HEIGHT / 2) / camera.zoom + camera.y;
    }
}

class Camera {
    constructor() {
        this.position = new Vector2(CONFIG.CANVAS_WIDTH / 2, 0);
        this.target = new Vector2(CONFIG.CANVAS_WIDTH / 2, 0);
        this.zoom = 1.0;
        this.targetZoom = 1.0;
        this.x = CONFIG.CANVAS_WIDTH / 2;
        this.y = 0;
    }

    follow(targetX, targetY, immediate = false) {
        this.target.x = Utils.clamp(targetX, CONFIG.CANVAS_WIDTH / 2 / this.zoom, CONFIG.WORLD_WIDTH - CONFIG.CANVAS_WIDTH / 2 / this.zoom);
        this.target.y = Utils.clamp(targetY, -200, 200);
        
        if (immediate) {
            this.position.x = this.target.x;
            this.position.y = this.target.y;
        }
    }

    update() {
        this.position.x = Utils.lerp(this.position.x, this.target.x, CONFIG.CAMERA_LERP);
        this.position.y = Utils.lerp(this.position.y, this.target.y, CONFIG.CAMERA_LERP);
        this.zoom = Utils.lerp(this.zoom, this.targetZoom, CONFIG.CAMERA_LERP);
        
        this.x = this.position.x;
        this.y = this.position.y;
    }

    worldToScreen(worldX, worldY) {
        return {
            x: (worldX - this.position.x) * this.zoom + CONFIG.CANVAS_WIDTH / 2,
            y: (worldY - this.position.y) * this.zoom + CONFIG.CANVAS_HEIGHT / 2
        };
    }

    screenToWorld(screenX, screenY) {
        return {
            x: (screenX - CONFIG.CANVAS_WIDTH / 2) / this.zoom + this.position.x,
            y: (screenY - CONFIG.CANVAS_HEIGHT / 2) / this.zoom + this.position.y
        };
    }
}

// 实体基类
// ============================================
class Entity {
    constructor(x, y) {
        this.position = new Vector2(x, y);
        this.velocity = new Vector2(0, 0);
        this.acceleration = new Vector2(0, 0);
        this.rotation = 0;
        this.angularVelocity = 0;
        this.mass = 1;
        this.restitution = 0.6;
        this.friction = 0.3;
        this.isStatic = false;
        this.isSleeping = false;
        this.sleepTimer = 0;
        this.isActive = true;
        this.health = 100;
        this.maxHealth = 100;
    }

    update(deltaTime) {
        if (this.isStatic || this.isSleeping) return;
        
        // 应用重力
        this.velocity.y += CONFIG.PHYSICS.GRAVITY;
        
        // 应用空气阻力
        this.velocity.x *= CONFIG.PHYSICS.AIR_RESISTANCE;
        this.velocity.y *= CONFIG.PHYSICS.AIR_RESISTANCE;
        
        // 更新位置
        this.position.x += this.velocity.x;
        this.position.y += this.velocity.y;
        
        // 更新旋转
        this.rotation += this.angularVelocity;
        this.angularVelocity *= CONFIG.PHYSICS.ANGULAR_DAMPING;
        
        // 地面碰撞
        const groundY = -CONFIG.GROUND_HEIGHT;
        if (this.position.y > groundY) {
            this.position.y = groundY;
            this.velocity.y *= -this.restitution;
            this.velocity.x *= CONFIG.PHYSICS.GROUND_FRICTION;
            
            // 睡眠检测
            if (Math.abs(this.velocity.x) < CONFIG.PHYSICS.SLEEP_THRESHOLD && 
                Math.abs(this.velocity.y) < CONFIG.PHYSICS.SLEEP_THRESHOLD) {
                this.sleepTimer++;
                if (this.sleepTimer > CONFIG.PHYSICS.SLEEP_TIME) {
                    this.isSleeping = true;
                }
            } else {
                this.sleepTimer = 0;
            }
        }
        
        // 边界检测
        if (this.position.x < 0) {
            this.position.x = 0;
            this.velocity.x *= -this.restitution;
        }
        if (this.position.x > CONFIG.WORLD_WIDTH) {
            this.position.x = CONFIG.WORLD_WIDTH;
            this.velocity.x *= -this.restitution;
        }
    }

    wake() {
        this.isSleeping = false;
        this.sleepTimer = 0;
    }

    takeDamage(amount) {
        this.health -= amount;
        if (this.health <= 0) {
            this.isActive = false;
            return true;
        }
        return false;
    }

    render(ctx, camera) {
        // 子类实现
    }
}

class Bird extends Entity {
    static TYPES = {
        red: { 
            name: 'Red Bird',
            ability: 'none',
            power: 1.0,
            radius: 18
        },
        yellow: { 
            name: 'Chuck',
            ability: 'speed',
            power: 1.3,
            radius: 16,
            speedBoost: 2.5
        },
        black: { 
            name: 'Bomb',
            ability: 'explosion',
            power: 1.5,
            radius: 22,
            explosionRadius: 120,
            explosionDamage: 80
        }
    };

    constructor(x, y, type = 'red') {
        super(x, y);
        this.type = type;
        this.config = Bird.TYPES[type];
        this.radius = this.config.radius;
        this.mass = type === 'black' ? 2 : 1;
        
        // 状态机
        this.animState = new AnimationStateMachine();
        this.isLaunched = false;
        this.abilityUsed = false;
        this.abilityActive = false;
        
        // 视觉效果
        this.colors = CONFIG.BIRD.COLORS[type];
        this.trail = [];
        this.maxTrailLength = 20;
        
        // 眼睛动画
        this.eyeOffset = { x: 0, y: 0 };
        this.blinkTimer = 0;
        this.isBlinking = false;
        
        // 羽毛粒子
        this.feathers = [];
    }

    update(deltaTime) {
        super.update(deltaTime);
        
        // 更新动画状态机
        this.animState.update(deltaTime);
        
        // 更新轨迹
        if (this.isLaunched && !this.isSleeping) {
            this.trail.push({ x: this.position.x, y: this.position.y, age: 0 });
            if (this.trail.length > this.maxTrailLength) {
                this.trail.shift();
            }
        }
        
        // 更新轨迹年龄
        for (let t of this.trail) {
            t.age++;
        }
        
        // 眨眼计时器
        this.blinkTimer += deltaTime;
        if (this.blinkTimer > 3000 + Math.random() * 2000) {
            this.isBlinking = true;
            setTimeout(() => this.isBlinking = false, 150);
            this.blinkTimer = 0;
        }
        
        // 更新羽毛
        this.feathers = this.feathers.filter(f => {
            f.x += f.vx;
            f.y += f.vy;
            f.vy += 0.1;
            f.rotation += f.rotationSpeed;
            f.life -= deltaTime;
            return f.life > 0;
        });
        
        // 更新眼睛方向
        if (this.isLaunched && !this.isSleeping) {
            const speed = Math.sqrt(this.velocity.x ** 2 + this.velocity.y ** 2);
            if (speed > 1) {
                this.eyeOffset.x = Utils.lerp(this.eyeOffset.x, (this.velocity.x / speed) * 3, 0.1);
                this.eyeOffset.y = Utils.lerp(this.eyeOffset.y, (this.velocity.y / speed) * 3, 0.1);
            }
        }
    }

    launch(velocity) {
        this.velocity = velocity;
        this.isLaunched = true;
        this.animState.setState('flying');
        this.wake();
    }

    useAbility() {
        if (this.abilityUsed || !this.isLaunched) return false;
        
        this.abilityUsed = true;
        this.abilityActive = true;
        this.animState.setState('ability');
        
        switch (this.config.ability) {
            case 'speed':
                // 加速技能
                const speedBoost = this.config.speedBoost;
                this.velocity.x *= speedBoost;
                this.velocity.y *= speedBoost;
                this.spawnFeathers(8);
                break;
                
            case 'explosion':
                // 爆炸技能 - 延迟触发
                setTimeout(() => this.explode(), 500);
                break;
        }
        
        setTimeout(() => {
            this.abilityActive = false;
            if (this.config.ability !== 'explosion') {
                this.animState.setState('falling');
            }
        }, this.config.ability === 'speed' ? 200 : 500);
        
        return true;
    }

    explode() {
        if (!this.isActive) return;
        
        this.spawnFeathers(20);
        this.isActive = false;
        
        // 返回爆炸数据供游戏处理
        return {
            x: this.position.x,
            y: this.position.y,
            radius: this.config.explosionRadius,
            damage: this.config.explosionDamage
        };
    }

    spawnFeathers(count) {
        for (let i = 0; i < count; i++) {
            this.feathers.push({
                x: this.position.x,
                y: this.position.y,
                vx: (Math.random() - 0.5) * 8,
                vy: (Math.random() - 0.5) * 8 - 2,
                rotation: Math.random() * Math.PI * 2,
                rotationSpeed: (Math.random() - 0.5) * 0.3,
                size: 3 + Math.random() * 4,
                life: 1000 + Math.random() * 500
            });
        }
    }

    onHit() {
        this.animState.setState('hit');
        setTimeout(() => {
            if (this.isLaunched) {
                this.animState.setState('falling');
            }
        }, 300);
    }

    setAiming(isAiming) {
        if (!this.isLaunched) {
            this.animState.setState(isAiming ? 'aiming' : 'idle');
        }
    }

    render(ctx, camera) {
        if (!this.isActive) return;
        
        const screen = camera.worldToScreen(this.position.x, this.position.y);
        const animData = this.animState.getAnimationData(this.velocity);
        
        ctx.save();
        ctx.translate(screen.x, screen.y + animData.bobOffset);
        ctx.rotate(animData.rotation);
        ctx.scale(animData.scaleX * camera.zoom, animData.scaleY * camera.zoom);
        
        // 闪光效果
        if (animData.flash) {
            ctx.globalAlpha = 0.8;
        }
        
        // 绘制轨迹
        this.renderTrail(ctx, camera);
        
        // 绘制阴影
        this.renderShadow(ctx, camera, animData);
        
        // 根据类型绘制小鸟
        switch (this.type) {
            case 'red':
                this.renderRedBird(ctx);
                break;
            case 'yellow':
                this.renderYellowBird(ctx);
                break;
            case 'black':
                this.renderBlackBird(ctx);
                break;
        }
        
        ctx.restore();
        
        // 绘制羽毛粒子
        this.renderFeathers(ctx, camera);
    }

    renderTrail(ctx, camera) {
        if (this.trail.length < 2) return;
        
        ctx.save();
        ctx.rotate(-this.animState.getAnimationData(this.velocity).rotation);
        
        for (let i = 0; i < this.trail.length; i++) {
            const t = this.trail[i];
            const alpha = (i / this.trail.length) * 0.5;
            const size = (i / this.trail.length) * this.radius * 0.5;
            
            const localX = (t.x - this.position.x) * camera.zoom;
            const localY = (t.y - this.position.y) * camera.zoom;
            
            ctx.fillStyle = Utils.safeColor(
                parseInt(this.colors.main.slice(1, 3), 16),
                parseInt(this.colors.main.slice(3, 5), 16),
                parseInt(this.colors.main.slice(5, 7), 16),
                alpha
            );
            ctx.beginPath();
            ctx.arc(localX, localY, size, 0, Math.PI * 2);
            ctx.fill();
        }
        
        ctx.restore();
    }

    renderShadow(ctx, camera, animData) {
        ctx.save();
        ctx.rotate(-animData.rotation);
        ctx.scale(1 / animData.scaleX, 1 / animData.scaleY);
        
        const shadowY = CONFIG.CANVAS_HEIGHT - CONFIG.GROUND_HEIGHT - this.position.y;
        const shadowScale = Utils.clamp(1 - shadowY / 500, 0.3, 1);
        
        ctx.fillStyle = 'rgba(0, 0, 0, 0.2)';
        ctx.beginPath();
        ctx.ellipse(0, shadowY * camera.zoom / animData.scaleY, 
            this.radius * shadowScale * 1.2, this.radius * shadowScale * 0.3, 0, 0, Math.PI * 2);
        ctx.fill();
        
        ctx.restore();
    }

    renderRedBird(ctx) {
        const r = this.radius;

        const gradient = ctx.createRadialGradient(-r * 0.3, -r * 0.3, 0, 0, 0, r);
        gradient.addColorStop(0, this.colors.light);
        gradient.addColorStop(0.7, this.colors.main);
        gradient.addColorStop(1, this.colors.dark);
        
        ctx.fillStyle = gradient;
        ctx.beginPath();
        ctx.arc(0, 0, r, 0, Math.PI * 2);
        ctx.fill();

        ctx.rotate(0.2);

        ctx.lineTo(0, -r * 1.4);
        ctx.lineTo(r * 0.15, -r * 0.9);

            ctx.fill();

        // 技能激活时的发光效果
        if (this.abilityActive) {
            ctx.strokeStyle = `rgba(255, 100, 0, ${0.5 + Math.sin(Date.now() / 50) * 0.3})`;
            ctx.lineWidth = 4;
            ctx.beginPath();
            ctx.arc(0, 0, r * 1.3, 0, Math.PI * 2);
            ctx.stroke();
        }

    renderEyes(ctx, r) {
        const eyeSpacing = r * 0.35;
        const eyeRadius = r * 0.22;
        
        [-1, 1].forEach(side => {
            const ex = side * eyeSpacing + this.eyeOffset.x * 0.5;
            const ey = -r * 0.1 + this.eyeOffset.y * 0.3;
            
            // 眼白
            ctx.fillStyle = '#ffffff';
            ctx.beginPath();
            ctx.ellipse(ex, ey, eyeRadius, eyeRadius * 1.2, 0, 0, Math.PI * 2);
            ctx.fill();
            
            // 眼眶
            ctx.strokeStyle = '#333';
            ctx.lineWidth = 1;
            ctx.stroke();
            
            if (!this.isBlinking) {
                // 瞳孔
                ctx.fillStyle = '#1a1a1a';
                ctx.beginPath();
                ctx.arc(ex + this.eyeOffset.x * 0.3, ey + this.eyeOffset.y * 0.3, eyeRadius * 0.5, 0, Math.PI * 2);
                ctx.fill();
                
                // 高光
                ctx.fillStyle = '#ffffff';
                ctx.beginPath();
                ctx.arc(ex + this.eyeOffset.x * 0.2 - eyeRadius * 0.15, ey + this.eyeOffset.y * 0.2 - eyeRadius * 0.15, eyeRadius * 0.2, 0, Math.PI * 2);
                ctx.fill();
            } else {
                // 闭眼
                ctx.strokeStyle = '#1a1a1a';
                ctx.lineWidth = 2;
                ctx.beginPath();
                ctx.moveTo(ex - eyeRadius, ey);
                ctx.lineTo(ex + eyeRadius, ey);
                ctx.stroke();
            }
        });
    }

    renderBeak(ctx, r, color1, color2, sharp = false) {
        ctx.save();
        ctx.translate(0, r * 0.2);
        
        // 上嘴
        ctx.fillStyle = color1;
        ctx.beginPath();
        if (sharp) {
            ctx.moveTo(-r * 0.2, -r * 0.1);
            ctx.lineTo(r * 0.8, r * 0.1);
            ctx.lineTo(-r * 0.1, r * 0.2);
        } else {
            ctx.moveTo(-r * 0.25, -r * 0.15);
            ctx.lineTo(r * 0.6, 0);
            ctx.lineTo(-r * 0.15, r * 0.15);
        }
        ctx.closePath();
        ctx.fill();
        
        // 下嘴
        ctx.fillStyle = color2;
        ctx.beginPath();
        if (sharp) {
            ctx.moveTo(-r * 0.1, r * 0.2);
            ctx.lineTo(r * 0.5, r * 0.25);
            ctx.lineTo(-r * 0.05, r * 0.35);
        } else {
            ctx.moveTo(-r * 0.15, r * 0.15);
            ctx.lineTo(r * 0.4, r * 0.25);
            ctx.lineTo(-r * 0.05, r * 0.35);
        }
        ctx.closePath();
        ctx.fill();
        
        ctx.restore();
    }

    renderFeathers(ctx, camera) {
        for (const f of this.feathers) {
            const screen = camera.worldToScreen(f.x, f.y);
            ctx.save();
            ctx.translate(screen.x, screen.y);
            ctx.rotate(f.rotation);
            ctx.globalAlpha = f.life / 1500;
            ctx.fillStyle = this.colors.main;
            ctx.beginPath();
            ctx.ellipse(0, 0, f.size * camera.zoom, f.size * 0.4 * camera.zoom, 0, 0, Math.PI * 2);
            ctx.fill();
            ctx.restore();
        }
    }
}

class Pig extends Entity {
    static SIZES = {
        small: { radius: 18, health: 30, points: 500 },
        medium: { radius: 26, health: 50, points: 1000 },
        large: { radius: 36, health: 80, points: 2000 },
        king: { radius: 48, health: 150, points: 5000 }
    };

    constructor(x, y, size = 'medium') {
        super(x, y);
        this.size = size;
        this.config = Pig.SIZES[size];
        this.radius = this.config.radius;
        this.health = this.config.health;
        this.maxHealth = this.config.health;
        this.mass = this.radius / 18;
        
        // 表情状态
        this.expression = 'normal';
        this.damageLevel = 0;
        
        // 动画
        this.wobble = 0;
        this.wobbleSpeed = 0.02 + Math.random() * 0.01;
        
        // 皇冠（国王猪）
        this.hasCrown = size === 'king';
    }

    update(deltaTime) {
        super.update(deltaTime);
        
        // 摇摆动画
        this.wobble += this.wobbleSpeed * deltaTime;
        
        // 根据生命值更新表情
        const healthPercent = this.health / this.maxHealth;
        if (healthPercent > 0.6) {
            this.expression = 'normal';
        } else if (healthPercent > 0.3) {
            this.expression = 'worried';
        } else {
            this.expression = 'hurt';
        }
        this.damageLevel = 1 - healthPercent;
    }

    takeDamage(amount) {
        const destroyed = super.takeDamage(amount);
        if (destroyed) {
            this.spawnParticles();
        }
        return destroyed;
    }

    spawnParticles() {
        // 返回粒子数据供游戏渲染
        return {
            x: this.position.x,
            y: this.position.y,
            count: 10,
            color: '#7cb342'
        };
    }

    render(ctx, camera) {
        if (!this.isActive) return;
        
        const screen = camera.worldToScreen(this.position.x, this.position.y);
        const r = this.radius * camera.zoom;
        
        ctx.save();
        ctx.translate(screen.x, screen.y);
        
        // 摇摆效果
        const wobbleX = Math.sin(this.wobble) * 2 * camera.zoom;
        const wobbleY = Math.cos(this.wobble * 1.5) * 1.5 * camera.zoom;
        ctx.translate(wobbleX, wobbleY);
        
        // 受伤时的抖动
        if (this.damageLevel > 0.5) {
            ctx.translate(
                (Math.random() - 0.5) * this.damageLevel * 4,
                (Math.random() - 0.5) * this.damageLevel * 4
            );
        }
        
        // 阴影
        ctx.fillStyle = 'rgba(0, 0, 0, 0.2)';
        ctx.beginPath();
        ctx.ellipse(wobbleX * 0.5, r * 0.9, r * 0.8, r * 0.25, 0, 0, Math.PI * 2);
        ctx.fill();
        
        // 身体
        const gradient = ctx.createRadialGradient(-r * 0.3, -r * 0.3, 0, 0, 0, r);
        gradient.addColorStop(0, '#9ccc65');
        gradient.addColorStop(0.6, '#7cb342');
        gradient.addColorStop(1, '#558b2f');
        
        ctx.fillStyle = gradient;
        ctx.beginPath();
        ctx.arc(0, 0, r, 0, Math.PI * 2);
        ctx.fill();
        
        // 耳朵
        this.renderEars(ctx, r);
        
        // 眼睛
        this.renderEyes(ctx, r);
        
        // 鼻子
        this.renderSnout(ctx, r);
        
        // 皇冠
        if (this.hasCrown) {
            this.renderCrown(ctx, r);
        }
        
        // 受伤效果
        if (this.damageLevel > 0) {
            ctx.strokeStyle = `rgba(255, 0, 0, ${this.damageLevel * 0.5})`;
            ctx.lineWidth = 2;
            ctx.beginPath();
            ctx.arc(0, 0, r * (1 + this.damageLevel * 0.1), 0, Math.PI * 2);
            ctx.stroke();
        }
        
        ctx.restore();
    }

    renderEars(ctx, r) {
        const earPositions = [
            { x: -r * 0.6, y: -r * 0.6, rot: -0.5 },
            { x: r * 0.6, y: -r * 0.6, rot: 0.5 }
        ];
        
        earPositions.forEach(pos => {
            ctx.save();
            ctx.translate(pos.x, pos.y);
            ctx.rotate(pos.rot);
            
            ctx.fillStyle = '#7cb342';
            ctx.beginPath();
            ctx.ellipse(0, 0, r * 0.25, r * 0.35, 0, 0, Math.PI * 2);
            ctx.fill();
            
            ctx.fillStyle = '#9ccc65';
            ctx.beginPath();
            ctx.ellipse(0, 0, r * 0.15, r * 0.2, 0, 0, Math.PI * 2);
            ctx.fill();
            
            ctx.restore();
        });
    }

    renderEyes(ctx, r) {
        const eyeSpacing = r * 0.35;
        const eyeY = -r * 0.15;
        
        [-1, 1].forEach(side => {
            const ex = side * eyeSpacing;
            
            // 根据表情调整眼睛
            let eyeScaleY = 1;
            let pupilOffset = { x: 0, y: 0 };
            
            switch (this.expression) {
                case 'worried':
                    eyeScaleY = 1.3;
                    pupilOffset = { x: 0, y: r * 0.05 };
                    break;
                case 'hurt':
                    eyeScaleY = 0.7;
                    break;
            }
            
            // 眼白
            ctx.fillStyle = '#ffffff';
            ctx.beginPath();
            ctx.ellipse(ex, eyeY, r * 0.2, r * 0.22 * eyeScaleY, 0, 0, Math.PI * 2);
            ctx.fill();
            
            // 瞳孔
            if (this.expression !== 'hurt' || Math.random() > 0.1) {
                ctx.fillStyle = '#1a1a1a';
                ctx.beginPath();
                ctx.arc(ex + pupilOffset.x, eyeY + pupilOffset.y, r * 0.1, 0, Math.PI * 2);
                ctx.fill();
                
                // 高光
                ctx.fillStyle = '#ffffff';
                ctx.beginPath();
                ctx.arc(ex - r * 0.05, eyeY - r * 0.05, r * 0.05, 0, Math.PI * 2);
                ctx.fill();
            } else {
                // 受伤时的 X 眼
                ctx.strokeStyle = '#1a1a1a';
                ctx.lineWidth = 2;
                ctx.beginPath();
                ctx.moveTo(ex - r * 0.1, eyeY - r * 0.1);
                ctx.lineTo(ex + r * 0.1, eyeY + r * 0.1);
                ctx.moveTo(ex + r * 0.1, eyeY - r * 0.1);
                ctx.lineTo(ex - r * 0.1, eyeY + r * 0.1);
                ctx.stroke();
            }
            
            // 眉毛（担心时）
            if (this.expression === 'worried' || this.expression === 'hurt') {
                ctx.strokeStyle = '#558b2f';
                ctx.lineWidth = 2;
                ctx.beginPath();
                ctx.moveTo(ex - r * 0.15, eyeY - r * 0.3);
                ctx.lineTo(ex + r * 0.15, eyeY - r * 0.25);
                ctx.stroke();
            }
        });
    }

    renderSnout(ctx, r) {
        const snoutY = r * 0.2;
        
        // 鼻子主体
        ctx.fillStyle = '#8bc34a';
        ctx.beginPath();
        ctx.ellipse(0, snoutY, r * 0.4, r * 0.3, 0, 0, Math.PI * 2);
        ctx.fill();
        
        // 鼻孔
        ctx.fillStyle = '#558b2f';
        [-1, 1].forEach(side => {
            ctx.beginPath();
            ctx.ellipse(side * r * 0.12, snoutY, r * 0.08, r * 0.1, 0, 0, Math.PI * 2);
            ctx.fill();
        });
    }

    renderCrown(ctx, r) {
        ctx.save();
        ctx.translate(0, -r * 0.85);
        
        // 皇冠主体
        ctx.fillStyle = '#ffd700';
        ctx.beginPath();
        ctx.moveTo(-r * 0.4, r * 0.15);
        ctx.lineTo(-r * 0.35, -r * 0.3);
        ctx.lineTo(-r * 0.15, -r * 0.1);
        ctx.lineTo(0, -r * 0.4);
        ctx.lineTo(r * 0.15, -r * 0.1);
        ctx.lineTo(r * 0.35, -r * 0.3);
        ctx.lineTo(r * 0.4, r * 0.15);
        ctx.closePath();
        ctx.fill();
        
        // 宝石
        ctx.fillStyle = '#e94560';
        ctx.beginPath();
        ctx.arc(0, -r * 0.15, r * 0.08, 0, Math.PI * 2);
        ctx.fill();
        
        ctx.restore();
    }
}

class Block extends Entity {
    static MATERIALS = {
        wood: {
            color: '#deb887',
            darkColor: '#cd853f',
            lightColor: '#f5deb3',
            health: 40,
            density: 0.6,
            sound: 'wood'
        },
        stone: {
            color: '#808080',
            darkColor: '#606060',
            lightColor: '#a0a0a0',
            health: 100,
            density: 1.2,
            sound: 'stone'
        },
        glass: {
            color: '#b0e0e6',
            darkColor: '#87ceeb',
            lightColor: '#e0ffff',
            health: 20,
            density: 0.4,
            sound: 'glass'
        },
        tnt: {
            color: '#cc0000',
            darkColor: '#990000',
            lightColor: '#ff3333',
            health: 30,
            density: 0.5,
            sound: 'tnt',
            explosive: true,
            explosionRadius: 100
        }
    };

    static SHAPES = {
        square: { width: 40, height: 40 },
        rect: { width: 80, height: 40 },
        plank: { width: 120, height: 20 },
        tall: { width: 30, height: 80 }
    };

    constructor(x, y, material = 'wood', shape = 'square') {
        super(x, y);
        this.material = material;
        this.shape = shape;
        
        this.materialConfig = Block.MATERIALS[material];
        this.shapeConfig = Block.SHAPES[shape];
        
        this.width = this.shapeConfig.width;
        this.height = this.shapeConfig.height;
        this.health = this.materialConfig.health;
        this.maxHealth = this.materialConfig.health;
        this.mass = this.width * this.height * this.materialConfig.density / 1000;
        
        this.damageLevel = 0;
        this.crackLines = [];
        
        // TNT 特有属性
        this.fuseTimer = 0;
        this.isExploding = false;
    }

    update(deltaTime) {
        super.update(deltaTime);
        
        // 更新伤害等级
        this.damageLevel = 1 - (this.health / this.maxHealth);
        
        // 生成裂缝
        if (this.damageLevel > 0.3 && this.crackLines.length < 3) {
            if (Math.random() < 0.01) {
                this.crackLines.push({
                    x: (Math.random() - 0.5) * this.width * 0.8,
                    y: (Math.random() - 0.5) * this.height * 0.8,
                    angle: Math.random() * Math.PI,
                    length: 10 + Math.random() * 20
                });
            }
        }
        
        // TNT 爆炸计时
        if (this.isExploding) {
            this.fuseTimer += deltaTime;
            if (this.fuseTimer > 500) {
                return this.explode();
            }
        }
    }

    takeDamage(amount) {
        const destroyed = super.takeDamage(amount);
        
        if (destroyed && this.materialConfig.explosive && !this.isExploding) {
            this.isExploding = true;
            return false; // 还没完全销毁，等待爆炸
        }
        
        return destroyed;
    }

    explode() {
        this.isActive = false;
        return {
            x: this.position.x,
            y: this.position.y,
            radius: this.materialConfig.explosionRadius,
            damage: 60
        };
    }

    render(ctx, camera) {
        if (!this.isActive) return;
        
        const screen = camera.worldToScreen(this.position.x, this.position.y);
        const w = this.width * camera.zoom;
        const h = this.height * camera.zoom;
        
        ctx.save();
        ctx.translate(screen.x, screen.y);
        ctx.rotate(this.rotation);
        
        // 阴影
        ctx.fillStyle = 'rgba(0, 0, 0, 0.2)';
        ctx.fillRect(-w / 2 + 4, -h / 2 + 4, w, h);
        
        // 主体
        const colors = this.materialConfig;
        const gradient = ctx.createLinearGradient(-w / 2, -h / 2, w / 2, h / 2);
        gradient.addColorStop(0, colors.lightColor);
        gradient.addColorStop(0.5, colors.color);
        gradient.addColorStop(1, colors.darkColor);
        
        ctx.fillStyle = gradient;
        ctx.fillRect(-w / 2, -h / 2, w, h);
        
        // 材质特效
        this.renderMaterialEffect(ctx, w, h);
        
        // 裂缝
        this.renderCracks(ctx, w, h);
        
        // 边框
        ctx.strokeStyle = colors.darkColor;
        ctx.lineWidth = 2 * camera.zoom;
        ctx.strokeRect(-w / 2, -h / 2, w, h);
        
        // TNT 标签
        if (this.material === 'tnt') {
            this.renderTNTLabel(ctx, w, h);
            
            // 爆炸闪烁
            if (this.isExploding) {
                ctx.fillStyle = `rgba(255, 255, 0, ${0.5 + Math.sin(Date.now() / 30) * 0.5})`;
                ctx.fillRect(-w / 2, -h / 2, w, h);
            }
        }
        
        ctx.restore();
    }

    renderMaterialEffect(ctx, w, h) {
        switch (this.material) {
            case 'wood':
                // 木纹
                ctx.strokeStyle = 'rgba(139, 69, 19, 0.3)';
                ctx.lineWidth = 1;
                for (let i = 0; i < 5; i++) {
                    const y = -h / 2 + (i + 1) * h / 6;
                    ctx.beginPath();
                    ctx.moveTo(-w / 2, y);
                    ctx.bezierCurveTo(-w / 4, y + 2, w / 4, y - 2, w / 2, y);
                    ctx.stroke();
                }
                break;
                
            case 'stone':
                // 石头纹理
                ctx.fillStyle = 'rgba(0, 0, 0, 0.1)';
                for (let i = 0; i < 8; i++) {
                    const x = (Math.random() - 0.5) * w * 0.8;
                    const y = (Math.random() - 0.5) * h * 0.8;
                    const size = 2 + Math.random() * 4;
                    ctx.beginPath();
                    ctx.arc(x, y, size, 0, Math.PI * 2);
                    ctx.fill();
                }
                break;
                
            case 'glass':
                // 玻璃反光
                ctx.fillStyle = 'rgba(255, 255, 255, 0.3)';
                ctx.beginPath();
                ctx.moveTo(-w / 2 + 5, -h / 2 + 5);
                ctx.lineTo(-w / 2 + 15, -h / 2 + 5);
                ctx.lineTo(-w / 2 + 5, -h / 2 + 15);
                ctx.closePath();
                ctx.fill();
                break;
        }
    }

    renderCracks(ctx, w, h) {
        if (this.crackLines.length === 0) return;
        
        ctx.strokeStyle = 'rgba(0, 0, 0, 0.5)';
        ctx.lineWidth = 1;
        
        for (const crack of this.crackLines) {
            ctx.beginPath();
            ctx.moveTo(crack.x, crack.y);
            
            // 生成锯齿状裂缝
            let x = crack.x;
            let y = crack.y;
            const segments = 3;
            for (let i = 0; i < segments; i++) {
                x += Math.cos(crack.angle + (Math.random() - 0.5)) * crack.length / segments;
                y += Math.sin(crack.angle + (Math.random() - 0.5)) * crack.length / segments;
                ctx.lineTo(x, y);
            }
            ctx.stroke();
        }
    }

    renderTNTLabel(ctx, w, h) {
        ctx.fillStyle = '#ffffff';
        ctx.font = `bold ${h * 0.4}px Arial`;
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';
        ctx.fillText('TNT', 0, 0);
    }
}

class ParticleSystem {
    constructor() {
        this.particles = [];
    }

    emit(x, y, count, color, options = {}) {
        for (let i = 0; i < count; i++) {
            this.particles.push({
                x, y,
                vx: (Math.random() - 0.5) * (options.spread || 10),
                vy: (Math.random() - 0.5) * (options.spread || 10) - (options.upward || 3),
                size: options.size || (2 + Math.random() * 4),
                color,
                life: options.life || 1000,
                maxLife: options.life || 1000,
                gravity: options.gravity !== undefined ? options.gravity : 0.2,
                rotation: Math.random() * Math.PI * 2,
                rotationSpeed: (Math.random() - 0.5) * 0.2
            });
        }
    }

    emitExplosion(x, y, radius) {
        // 火焰粒子
        this.emit(x, y, 30, '#ff6600', { spread: 15, upward: 5, size: 6, life: 600 });
        this.emit(x, y, 20, '#ffcc00', { spread: 12, upward: 4, size: 4, life: 400 });
        
        // 烟雾粒子
        this.emit(x, y, 15, '#444444', { spread: 8, upward: 2, size: 10, life: 1200, gravity: -0.05 });
        
        // 碎片粒子
        this.emit(x, y, 20, '#8b4513', { spread: 20, upward: 8, size: 3, life: 800 });
    }

    update(deltaTime) {
        this.particles = this.particles.filter(p => {
            p.x += p.vx;
            p.y += p.vy;
            p.vy += p.gravity;
            p.vx *= 0.98;
            p.rotation += p.rotationSpeed;
            p.life -= deltaTime;
            return p.life > 0;
        });
    }

    render(ctx, camera) {
        for (const p of this.particles) {
            const screen = camera.worldToScreen(p.x, p.y);
            const alpha = p.life / p.maxLife;
            
            ctx.save();
            ctx.translate(screen.x, screen.y);
            ctx.rotate(p.rotation);
            ctx.globalAlpha = alpha;
            ctx.fillStyle = p.color;
            
            if (p.size > 5) {
                // 大粒子用方块
                ctx.fillRect(-p.size / 2 * camera.zoom, -p.size / 2 * camera.zoom, 
                    p.size * camera.zoom, p.size * camera.zoom);
            } else {
                // 小粒子用圆
                ctx.beginPath();
                ctx.arc(0, 0, p.size * camera.zoom, 0, Math.PI * 2);
                ctx.fill();
            }
            
            ctx.restore();
        }
    }
}

class SlingshotRenderer {
    constructor(x, groundY) {
        this.x = x;
        this.y = groundY;
        this.launchPoint = { x: x, y: groundY - 120 };
    }

    renderBack(ctx, camera) {
        const screen = camera.worldToScreen(this.x, this.y);
        const launchScreen = camera.worldToScreen(this.launchPoint.x, this.launchPoint.y);
        const zoom = camera.zoom;
        
        // 后支架
        ctx.fillStyle = '#5d4037';
        ctx.fillRect(screen.x - 12 * zoom, screen.y - 140 * zoom, 10 * zoom, 140 * zoom);
        
        // 后皮筋连接点
        ctx.fillStyle = '#8B4513';
        ctx.beginPath();
        ctx.arc(launchScreen.x - 15 * zoom, launchScreen.y, 8 * zoom, 0, Math.PI * 2);
        ctx.fill();
    }

    renderFront(ctx, camera, birdPosition = null, isStretching = false) {
        const screen = camera.worldToScreen(this.x, this.y);
        const launchScreen = camera.worldToScreen(this.launchPoint.x, this.launchPoint.y);
        const zoom = camera.zoom;
        
        // 前支架
        ctx.fillStyle = '#6d4c41';
        ctx.fillRect(screen.x + 2 * zoom, screen.y - 150 * zoom, 10 * zoom, 150 * zoom);
        
        // 前皮筋连接点
        ctx.fillStyle = '#8B4513';
        ctx.beginPath();
        ctx.arc(launchScreen.x + 15 * zoom, launchScreen.y, 8 * zoom, 0, Math.PI * 2);
        ctx.fill();
        
        // 皮筋
        if (birdPosition && isStretching) {
            const birdScreen = camera.worldToScreen(birdPosition.x, birdPosition.y);
            
            // 后皮筋
            ctx.strokeStyle = CONFIG.SLINGSHOT.BAND_COLOR;
            ctx.lineWidth = CONFIG.SLINGSHOT.BAND_WIDTH * zoom;
            ctx.beginPath();
            ctx.moveTo(launchScreen.x - 15 * zoom, launchScreen.y);
            ctx.lineTo(birdScreen.x, birdScreen.y);
            ctx.stroke();
            
            // 前皮筋
            ctx.beginPath();
            ctx.moveTo(birdScreen.x, birdScreen.y);
            ctx.lineTo(launchScreen.x + 15 * zoom, launchScreen.y);
            ctx.stroke();
        } else {
            // 静止状态的皮筋
            ctx.strokeStyle = CONFIG.SLINGSHOT.BAND_COLOR;
            ctx.lineWidth = CONFIG.SLINGSHOT.BAND_WIDTH * zoom;
            ctx.beginPath();
            ctx.moveTo(launchScreen.x - 15 * zoom, launchScreen.y);
            ctx.quadraticCurveTo(launchScreen.x, launchScreen.y + 20 * zoom, launchScreen.x + 15 * zoom, launchScreen.y);
            ctx.stroke();
        }
    }

    renderTrajectory(ctx, camera, velocity, startX, startY) {
        if (!velocity) return;
        
        ctx.save();
        
        let x = startX;
        let y = startY;
        let vx = velocity.x;
        let vy = velocity.y;
        
        const points = [];
        
        for (let i = 0; i < CONFIG.TRAJECTORY.POINT_COUNT; i++) {
            const screen = camera.worldToScreen(x, y);
            points.push({ x: screen.x, y: screen.y, progress: i / CONFIG.TRAJECTORY.POINT_COUNT });
            
            vx *= CONFIG.PHYSICS.AIR_RESISTANCE;
            vy *= CONFIG.PHYSICS.AIR_RESISTANCE;
            vy += CONFIG.PHYSICS.GRAVITY;
            x += vx;
            y += vy;
            
            if (y > -CONFIG.GROUND_HEIGHT) break;
        }
        
        points.forEach((p, i) => {
            const size = Utils.lerp(CONFIG.TRAJECTORY.POINT_SIZE_START, CONFIG.TRAJECTORY.POINT_SIZE_END, p.progress);
            const alpha = 1 - p.progress;
            
            ctx.fillStyle = `rgba(255, 255, 255, ${alpha * 0.7})`;
            ctx.beginPath();
            ctx.arc(p.x, p.y, size * camera.zoom, 0, Math.PI * 2);
            ctx.fill();
        });
        
        ctx.restore();
    }
}

class SceneRenderer {
    constructor(canvas) {
        this.canvas = canvas;
        this.ctx = canvas.getContext('2d');
        
        // 背景层
        this.clouds = this.generateClouds();
        this.hills = this.generateHills();
    }

    generateClouds() {
        const clouds = [];
        for (let i = 0; i < 15; i++) {
            clouds.push({
                x: Math.random() * CONFIG.WORLD_WIDTH,
                y: -200 - Math.random() * 300,
                width: 80 + Math.random() * 120,
                speed: 0.02 + Math.random() * 0.03
            });
        }
        return clouds;
    }

    generateHills() {
        return [
            { offset: 0, height: 80, color: '#2d5a3d' },
            { offset: 200, height: 120, color: '#1e4d2b' },
            { offset: 400, height: 60, color: '#3a6b4a' }
        ];
    }

    render(camera) {
        const ctx = this.ctx;
        
        // 天空渐变
        const skyGradient = ctx.createLinearGradient(0, 0, 0, CONFIG.CANVAS_HEIGHT);
        skyGradient.addColorStop(0, '#1a1a2e');
        skyGradient.addColorStop(0.5, '#16213e');
        skyGradient.addColorStop(1, '#0f3460');
        ctx.fillStyle = skyGradient;
        ctx.fillRect(0, 0, CONFIG.CANVAS_WIDTH, CONFIG.CANVAS_HEIGHT);
        
        // 星星
        this.renderStars(ctx);
        
        // 云朵
        this.renderClouds(ctx, camera);
        
        // 远山
        this.renderHills(ctx, camera);
        
        // 地面
        this.renderGround(ctx, camera);
    }

    renderStars(ctx) {
        ctx.fillStyle = 'rgba(255, 255, 255, 0.5)';
        for (let i = 0; i < 100; i++) {
            const x = Utils.noise(i * 0.1) * CONFIG.CANVAS_WIDTH;
            const y = Utils.noise(i * 0.2 + 100) * CONFIG.CANVAS_HEIGHT * 0.5;
            const size = Utils.noise(i * 0.3 + 200) * 2 + 0.5;
            const twinkle = Math.sin(Date.now() / 1000 + i) * 0.3 + 0.7;
            ctx.globalAlpha = twinkle * 0.8;
            ctx.beginPath();
            ctx.arc(x, y, size, 0, Math.PI * 2);
            ctx.fill();
        }
        ctx.globalAlpha = 1;
    }

    renderClouds(ctx, camera) {
        ctx.fillStyle = 'rgba(255, 255, 255, 0.1)';
        
        for (const cloud of this.clouds) {
            const parallax = 0.3;
            const screenX = (cloud.x - camera.x * parallax) % (CONFIG.WORLD_WIDTH + 200) - 100;
            
            if (screenX > -200 && screenX < CONFIG.CANVAS_WIDTH + 200) {
                ctx.beginPath();
                ctx.ellipse(screenX, cloud.y * camera.zoom + 100, 
                    cloud.width * camera.zoom * 0.5, 30 * camera.zoom, 0, 0, Math.PI * 2);
                ctx.fill();
                ctx.beginPath();
                ctx.ellipse(screenX - 30 * camera.zoom, cloud.y * camera.zoom + 110, 
                    cloud.width * camera.zoom * 0.3, 25 * camera.zoom, 0, 0, Math.PI * 2);
                ctx.fill();
            }
            
            cloud.x += cloud.speed;
            if (cloud.x > CONFIG.WORLD_WIDTH + 200) {
                cloud.x = -200;
            }
        }
    }

    renderHills(ctx, camera) {
        this.hills.forEach((hill, index) => {
            ctx.fillStyle = hill.color;
            ctx.beginPath();
            
            const parallax = 0.2 + index * 0.1;
            const baseY = CONFIG.CANVAS_HEIGHT - CONFIG.GROUND_HEIGHT;
            
            ctx.moveTo(0, baseY);
            
            for (let x = 0; x <= CONFIG.CANVAS_WIDTH; x += 20) {
                const worldX = x / camera.zoom + camera.x;
                const hillY = Math.sin((worldX + hill.offset) * 0.005) * hill.height;
                ctx.lineTo(x, baseY - hillY * camera.zoom * parallax);
            }
            
            ctx.lineTo(CONFIG.CANVAS_WIDTH, baseY);
            ctx.closePath();
            ctx.fill();
        });
    }

    renderGround(ctx, camera) {
        const groundY = CONFIG.CANVAS_HEIGHT - CONFIG.GROUND_HEIGHT;
        
        // 主地面
        const groundGradient = ctx.createLinearGradient(0, groundY, 0, CONFIG.CANVAS_HEIGHT);
        groundGradient.addColorStop(0, '#4a7c59');
        groundGradient.addColorStop(0.3, '#3d6b4f');
        groundGradient.addColorStop(1, '#2d5a3d');
        
        ctx.fillStyle = groundGradient;
        ctx.fillRect(0, groundY, CONFIG.CANVAS_WIDTH, CONFIG.GROUND_HEIGHT);
        
        // 草地纹理
        ctx.strokeStyle = '#5a8c69';
        ctx.lineWidth = 2;
        for (let x = 0; x < CONFIG.CANVAS_WIDTH; x += 15) {
            const grassHeight = 5 + Utils.noise(x * 0.1) * 10;
            const sway = Math.sin(Date.now() / 500 + x * 0.1) * 2;
            ctx.beginPath();
            ctx.moveTo(x, groundY);
            ctx.quadraticCurveTo(x + sway, groundY - grassHeight, x + sway * 0.5, groundY - grassHeight * 1.5);
            ctx.stroke();
        }
        
        // 地下土层
        ctx.fillStyle = '#8b6914';
        ctx.fillRect(0, groundY + 30, CONFIG.CANVAS_WIDTH, CONFIG.GROUND_HEIGHT - 30);
    }
}

    handleCameraInput() {
        const moveSpeed = 8;
        
        if (this.input.isKeyPressed('KeyA')) {
            this.camera.target.x -= moveSpeed;
        }
        if (this.input.isKeyPressed('KeyD')) {
            this.camera.target.x += moveSpeed;
        }
        if (this.input.isKeyPressed('KeyW')) {
            this.camera.targetZoom = Math.min(CONFIG.MAX_ZOOM, this.camera.targetZoom + 0.02);
        }
        if (this.input.isKeyPressed('KeyS')) {
            this.camera.targetZoom = Math.max(CONFIG.MIN_ZOOM, this.camera.targetZoom - 0.02);
        }
        
        // 跟随飞行的小鸟
        if (this.currentBird && this.currentBird.isLaunched && !this.currentBird.isSleeping) {
            this.camera.follow(this.currentBird.position.x, this.currentBird.position.y);
        }
    }

    handleAiming() {
        if (!this.currentBird || this.currentBird.isLaunched) return;
        
        const dist = Utils.distance(
            this.input.mouse.worldX, this.input.mouse.worldY,
            this.launchPoint.x, this.launchPoint.y
        );
        
        // 检测是否在弹弓附近
        if (dist < CONFIG.SLINGSHOT.MAX_STRETCH * 1.5) {
            if (this.input.mouse.isDown) {
                this.isAiming = true;
                this.currentBird.setAiming(true);
                
                // 计算拉伸
                let dx = this.input.mouse.worldX - this.launchPoint.x;
                let dy = this.input.mouse.worldY - this.launchPoint.y;
                const stretchDist = Math.sqrt(dx * dx + dy * dy);
                
                if (stretchDist > CONFIG.SLINGSHOT.MAX_STRETCH) {
                    dx = dx / stretchDist * CONFIG.SLINGSHOT.MAX_STRETCH;
                    dy = dy / stretchDist * CONFIG.SLINGSHOT.MAX_STRETCH;
                }
                
                this.currentBird.position.x = this.launchPoint.x + dx;
                this.currentBird.position.y = this.launchPoint.y + dy;
                
                // 计算发射速度
                const power = stretchDist * CONFIG.BIRD.LAUNCH_POWER;
                this.launchVelocity = new Vector2(-dx * CONFIG.BIRD.LAUNCH_POWER, -dy * CONFIG.BIRD.LAUNCH_POWER);
                
                // 更新力量条
                this.updatePowerIndicator(stretchDist / CONFIG.SLINGSHOT.MAX_STRETCH);
                
                // 拉伸音效
                if (stretchDist > 20) {
                    this.sound.playStretch(stretchDist);
                }
            } else if (this.isAiming) {
                // 释放发射
                this.launchBird();
            }
        }
        
        if (!this.input.mouse.isDown && this.isAiming) {
            this.isAiming = false;
            this.currentBird.setAiming(false);
            this.hidePowerIndicator();
        }
    }

    launchBird() {
        if (!this.launchVelocity) return;
        
        const power = this.launchVelocity.length();
        if (power < CONFIG.BIRD.MIN_POWER) {
            // 力量太小，重置位置
            this.currentBird.position.x = this.launchPoint.x;
            this.currentBird.position.y = this.launchPoint.y;
            this.isAiming = false;
            this.hidePowerIndicator();
            return;
        }
        
        // 应用力度修正
        const finalPower = Utils.clamp(power, CONFIG.BIRD.MIN_POWER, CONFIG.BIRD.MAX_POWER);
        const direction = this.launchVelocity.normalize();
        const finalVelocity = direction.mul(finalPower * this.currentBird.config.power);
        
        this.currentBird.launch(finalVelocity);
        this.birdsRemaining--;
        this.gameState = 'flying';
        
        this.sound.playLaunch(finalPower);
        this.isAiming = false;
        this.hidePowerIndicator();
        this.updateUI();
    }

    updatePowerIndicator(ratio) {
        const indicator = document.getElementById('powerIndicator');
        const fill = document.getElementById('powerFill');
        
        indicator.style.display = 'block';
        fill.style.width = `${ratio * 100}%`;
        
        // 颜色渐变
        const r = Math.floor(255 * ratio);
        const g = Math.floor(255 * (1 - ratio));
        fill.style.background = `rgb(${r}, ${g}, 0)`;
    }

    hidePowerIndicator() {
        document.getElementById('powerIndicator').style.display = 'none';
    }

    updateEntities(deltaTime) {
        // 更新小鸟
        for (const bird of this.birds) {
            bird.update(deltaTime);
            
            // 处理爆炸鸟的爆炸
            if (bird.type === 'black' && bird.abilityUsed && !bird.isActive) {
                const explosion = bird.explode();
                if (explosion) {
                    this.handleExplosion(explosion);
                }
            }
        }
        
        // 更新猪
        for (const pig of this.pigs) {
            pig.update(deltaTime);
        }
        
        // 更新方块
        for (const block of this.blocks) {
            const result = block.update(deltaTime);
            
            // 处理TNT爆炸
            if (result && block.material === 'tnt') {
                this.handleExplosion(result);
            }
        }
        
        // 清理不活跃的实体
        this.pigs = this.pigs.filter(p => {
            if (!p.isActive) {
                this.score += p.config.points;
                this.particles.emit(p.position.x, p.position.y, 15, '#7cb342', { spread: 8, upward: 5 });
                this.updateUI();
                return false;
            }
            return true;
        });
        
        this.blocks = this.blocks.filter(b => b.isActive);
    }

    handleExplosion(explosion) {
        this.sound.playExplosion();
        this.particles.emitExplosion(explosion.x, explosion.y, explosion.radius);
        
        // 对范围内的实体造成伤害
        for (const pig of this.pigs) {
            const dist = Utils.distance(explosion.x, explosion.y, pig.position.x, pig.position.y);
            if (dist < explosion.radius) {
                const damage = explosion.damage * (1 - dist / explosion.radius);
                pig.takeDamage(damage);
                
                // 击退效果
                const angle = Math.atan2(pig.position.y - explosion.y, pig.position.x - explosion.x);
                const force = (1 - dist / explosion.radius) * 10;
                pig.velocity.x += Math.cos(angle) * force;
                pig.velocity.y += Math.sin(angle) * force;
                pig.wake();
            }
        }
        
        for (const block of this.blocks) {
            const dist = Utils.distance(explosion.x, explosion.y, block.position.x, block.position.y);
            if (dist < explosion.radius) {
                const damage = explosion.damage * (1 - dist / explosion.radius);
                block.takeDamage(damage);
            }
        }
    }

    checkCollisions() {
        // 简单的碰撞检测
        for (const bird of this.birds) {
            if (!bird.isLaunched || bird.isSleeping) continue;
            
            // 小鸟与猪的碰撞
            for (const pig of this.pigs) {
                if (!pig.isActive) continue;
                
                const dist = Utils.distance(bird.position.x, bird.position.y, pig.position.x, pig.position.y);
                if (dist < bird.radius + pig.radius) {
                    // 计算伤害基于相对速度
                    const relVel = Math.sqrt(
                        (bird.velocity.x - pig.velocity.x) ** 2 + 
                        (bird.velocity.y - pig.velocity.y) ** 2
                    );
                    const damage = relVel * 5;
                    
                    if (pig.takeDamage(damage)) {
                        this.score += pig.config.points;
                        this.particles.emit(pig.position.x, pig.position.y, 15, '#7cb342');
                        this.updateUI();
                    }
                    
                    // 碰撞响应
                    this.resolveCollision(bird, pig);
                    bird.onHit();
                    this.sound.playBounce();
                }
            }
            
            // 小鸟与方块的碰撞
            for (const block of this.blocks) {
                if (!block.isActive) continue;
                
                // 简化的AABB与圆碰撞
                const closestX = Utils.clamp(bird.position.x, 
                    block.position.x - block.width / 2, 
                    block.position.x + block.width / 2);
                const closestY = Utils.clamp(bird.position.y, 
                    block.position.y - block.height / 2, 
                    block.position.y + block.height / 2);
                
                const dist = Utils.distance(bird.position.x, bird.position.y, closestX, closestY);
                
                if (dist < bird.radius) {
                    const relVel = Math.sqrt(bird.velocity.x ** 2 + bird.velocity.y ** 2);
                    const damage = relVel * 3;
                    
                    if (block.takeDamage(damage)) {
                        this.score += 50;
                        this.particles.emit(block.position.x, block.position.y, 10, block.materialConfig.color);
                        this.updateUI();
                    }
                    
                    // 碰撞响应
                    this.resolveBlockCollision(bird, block, closestX, closestY);
                    bird.onHit();
                    this.sound.playBounce();
                }
            }
        }
    }

    resolveCollision(a, b) {
        const dx = b.position.x - a.position.x;
        const dy = b.position.y - a.position.y;
        const dist = Math.sqrt(dx * dx + dy * dy);
        
        if (dist === 0) return;
        
        const nx = dx / dist;
        const ny = dy / dist;
        
        const relVelX = a.velocity.x - b.velocity.x;
        const relVelY = a.velocity.y - b.velocity.y;
        const relVelN = relVelX * nx + relVelY * ny;
        
        if (relVelN > 0) return;
        
        const restitution = Math.min(a.restitution, b.restitution);
        const impulse = -(1 + restitution) * relVelN / (1 / a.mass + 1 / b.mass);
        
        a.velocity.x += impulse * nx / a.mass;
        a.velocity.y += impulse * ny / a.mass;
        b.velocity.x -= impulse * nx / b.mass;
        b.velocity.y -= impulse * ny / b.mass;
        
        // 分离
        const overlap = (a.radius + b.radius) - dist;
        if (overlap > 0) {
            const sep = overlap / 2;
            a.position.x -= sep * nx;
            a.position.y -= sep * ny;
            b.position.x += sep * nx;
            b.position.y += sep * ny;
        }
        
        b.wake();
    }

    resolveBlockCollision(bird, block, closestX, closestY) {
        const dx = bird.position.x - closestX;
        const dy = bird.position.y - closestY;
        const dist = Math.sqrt(dx * dx + dy * dy);
        
        if (dist === 0) return;
        
        const nx = dx / dist;
        const ny = dy / dist;
        
        // 反弹
        const dot = bird.velocity.x * nx + bird.velocity.y * ny;
        bird.velocity.x -= 2 * dot * nx * bird.restitution;
        bird.velocity.y -= 2 * dot * ny * bird.restitution;
        
        // 分离
        const overlap = bird.radius - dist;
        if (overlap > 0) {
            bird.position.x += overlap * nx;
            bird.position.y += overlap * ny;
        }
        
        // 给方块施加力
        block.velocity.x += nx * 2;
        block.velocity.y += ny * 2;
        block.wake();
    }

    checkGameState() {
        // 检查是否所有猪都被消灭
        const activePigs = this.pigs.filter(p => p.isActive);
        if (activePigs.length === 0) {
            this.gameState = 'win';
            return;
        }
        
        // 检查当前小鸟是否停止
        if (this.currentBird && this.currentBird.isLaunched) {
            if (this.currentBird.isSleeping || !this.currentBird.isActive) {
                // 等待一会后生成下一只小鸟
                if (this.gameState !== 'spawning') {
                    this.gameState = 'spawning';
                    setTimeout(() => {
                        this.spawnNextBird();
                        this.camera.target.x = CONFIG.CANVAS_WIDTH / 2;
                        this.camera.targetZoom = 1;
                    }, 1500);
                }
            }
        }
    }

    reset() {
        this.birds = [];
        this.pigs = [];
        this.blocks = [];
        this.particles = new ParticleSystem();
        this.score = 0;
        this.birdsRemaining = 3;
        this.gameState = 'ready';
        
        this.setupLevel();
        this.spawnNextBird();
        this.camera.target.x = CONFIG.CANVAS_WIDTH / 2;
        this.camera.targetZoom = 1;
        this.updateUI();
    }

    updateDebugPanel() {
        if (!this.debugMode) return;
        
        const panel = document.getElementById('debugPanel');
        const bird = this.currentBird;
        
        let debugText = `FPS: ${this.fps}\n`;
        debugText += `State: ${this.gameState}\n`;
        if (bird) {
            debugText += `Bird: ${bird.type}\n`;
            debugText += `AnimState: ${bird.animState.currentState}\n`;
            debugText += `Pos: (${bird.position.x.toFixed(0)}, ${bird.position.y.toFixed(0)})\n`;
            debugText += `Vel: (${bird.velocity.x.toFixed(2)}, ${bird.velocity.y.toFixed(2)})\n`;
        }
        debugText += `Pigs: ${this.pigs.length}\n`;
        debugText += `Blocks: ${this.blocks.length}`;
        
        panel.textContent = debugText;
    }

    render() {
        const ctx = this.ctx;
        
        // 清空画布
        ctx.clearRect(0, 0, CONFIG.CANVAS_WIDTH, CONFIG.CANVAS_HEIGHT);
        
        // 渲染场景背景
        this.sceneRenderer.render(this.camera);
        
        // 渲染弹弓后部
        this.slingshot.renderBack(ctx, this.camera);
        
        // 渲染轨迹预测
        if (this.isAiming && this.launchVelocity && this.currentBird) {
            this.slingshot.renderTrajectory(ctx, this.camera, this.launchVelocity, 
                this.currentBird.position.x, this.currentBird.position.y);
        }
        
        // 渲染方块
        for (const block of this.blocks) {
            block.render(ctx, this.camera);
        }
        
        // 渲染猪
        for (const pig of this.pigs) {
            pig.render(ctx, this.camera);
        }
        
        // 渲染小鸟
        for (const bird of this.birds) {
            bird.render(ctx, this.camera);
        }
        
        // 渲染弹弓前部和皮筋
        const birdPos = this.isAiming && this.currentBird ? 
            { x: this.currentBird.position.x, y: this.currentBird.position.y } : null;
        this.slingshot.renderFront(ctx, this.camera, birdPos, this.isAiming);
        
        // 渲染粒子
        this.particles.render(ctx, this.camera);
        
        // 渲染游戏状态信息
        this.renderGameState();
    }

    renderGameState() {
        const ctx = this.ctx;
        
        if (this.gameState === 'win') {
            ctx.fillStyle = 'rgba(0, 0, 0, 0.5)';
            ctx.fillRect(0, 0, CONFIG.CANVAS_WIDTH, CONFIG.CANVAS_HEIGHT);
            
            ctx.fillStyle = '#ffd700';
            ctx.font = 'bold 48px Arial';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            ctx.fillText('LEVEL COMPLETE!', CONFIG.CANVAS_WIDTH / 2, CONFIG.CANVAS_HEIGHT / 2 - 30);
            
            ctx.fillStyle = '#ffffff';
            ctx.font = '24px Arial';
            ctx.fillText(`Score: ${this.score}`, CONFIG.CANVAS_WIDTH / 2, CONFIG.CANVAS_HEIGHT / 2 + 20);
            ctx.fillText('Press R to play again', CONFIG.CANVAS_WIDTH / 2, CONFIG.CANVAS_HEIGHT / 2 + 60);
        }
        
        if (this.gameState === 'gameover') {
            ctx.fillStyle = 'rgba(0, 0, 0, 0.5)';
            ctx.fillRect(0, 0, CONFIG.CANVAS_WIDTH, CONFIG.CANVAS_HEIGHT);
            
            ctx.fillStyle = '#e94560';
            ctx.font = 'bold 48px Arial';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            ctx.fillText('GAME OVER', CONFIG.CANVAS_WIDTH / 2, CONFIG.CANVAS_HEIGHT / 2 - 30);
            
            ctx.fillStyle = '#ffffff';
            ctx.font = '24px Arial';
            ctx.fillText(`Final Score: ${this.score}`, CONFIG.CANVAS_WIDTH / 2, CONFIG.CANVAS_HEIGHT / 2 + 20);
            ctx.fillText('Press R to try again', CONFIG.CANVAS_WIDTH / 2, CONFIG.CANVAS_HEIGHT / 2 + 60);
        }
    }

    gameLoop(timestamp = 0) {
        const deltaTime = timestamp - this.lastTime;
        this.lastTime = timestamp;
        
        // FPS计算
        this.frameCount++;
        if (this.frameCount % 30 === 0) {
            this.fps = Math.round(1000 / deltaTime);
        }
        
        this.update(deltaTime);
        this.render();
        
        requestAnimationFrame((t) => this.gameLoop(t));
    }
}

// 启动游戏
window.addEventListener('DOMContentLoaded', () => {
    const game = new Game();

## 实现说明

### 实体系统

1. **Entity 基类**：包含位置、速度、质量、碰撞等基础物理属性

2. **小鸟类**：
   - 三种类型：红色普通鸟、黄色加速鸟、黑色爆炸鸟
   - 每种类型有独特的能力和视觉特征
   - 完整的状态机系统

3. **绿猪类**：
   - 四种尺寸：small、medium、large、king
   - 不同尺寸有不同的血量和分值
   - 根据受伤程度显示不同表情
   - 国王猪有皇冠装饰

4. **方块类**：
   - 四种材质：木头、石头、玻璃、TNT
   - 不同材质有不同的耐久度和物理属性
   - TNT 方块会在被摧毁时爆炸

### 动画状态机

实现了 `AnimationStateMachine` 类，支持以下状态：
- **idle**：闲置状态，带有呼吸浮动动画
- **aiming**：瞄准状态，被拉伸时的变形
- **flying**：飞行状态，根据速度方向旋转和拉伸
- **ability**：技能激活状态，带有脉冲闪光效果
- **hit**：击中状态，震动效果
- **falling**：下落状态

### 挤压与拉伸

- 根据速度方向动态调整 scaleX 和 scaleY
- 飞行时沿运动方向拉伸
- 碰撞时产生挤压变形
- 使用平滑插值实现自然过渡
