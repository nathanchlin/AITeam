```html
    const Utils = {
        lerp: (a, b, t) => a + (b - a) * t,
        clamp: (val, min, max) => Math.max(min, Math.min(max, val)),
        randomRange: (min, max) => Math.random() * (max - min) + min,
        randomInt: (min, max) => Math.floor(Utils.randomRange(min, max + 1)),
        distance: (x1, y1, x2, y2) => Math.sqrt((x2 - x1) ** 2 + (y2 - y1) ** 2),
        
        safeColor: (r, g, b, a = 1) => {
            r = Utils.clamp(Math.round(r), 0, 255);
            g = Utils.clamp(Math.round(g), 0, 255);
            b = Utils.clamp(Math.round(b), 0, 255);
            a = Utils.clamp(a, 0, 1);
            return `rgba(${r},${g},${b},${a})`;

    class Vector2 {
        constructor(x = 0, y = 0) {
            this.x = x;
            this.y = y;
        }

        add(v) {
            return new Vector2(this.x + v.x, this.y + v.y);
        }

        sub(v) {
            return new Vector2(this.x - v.x, this.y - v.y);
        }

        mul(s) {
            return new Vector2(this.x * s, this.y * s);
        }

        dot(v) {
            return this.x * v.x + this.y * v.y;
        }

        cross(v) {
            return this.x * v.y - this.y * v.x;
        }

        length() {
            return Math.sqrt(this.x * this.x + this.y * this.y);
        }

        lengthSq() {
            return this.x * this.x + this.y * this.y;
        }

        normalize() {
            const len = this.length();
            if (len === 0) return new Vector2(0, 0);
            return new Vector2(this.x / len, this.y / len);
        }

        rotate(angle) {
            const cos = Math.cos(angle);
            const sin = Math.sin(angle);
            return new Vector2(
                this.x * cos - this.y * sin,
                this.x * sin + this.y * cos
            );
        }

        clone() {
            return new Vector2(this.x, this.y);
        }

        set(x, y) {
            this.x = x;
            this.y = y;
            return this;
        }

        static fromAngle(angle, length = 1) {
            return new Vector2(Math.cos(angle) * length, Math.sin(angle) * length);
        }
    }

    class SoundManager {
        constructor() {
            this.audioContext = null;
            this.enabled = true;
            this.initialized = false;
        }

        init() {
            if (this.initialized) return;
            try {
                this.audioContext = new (window.AudioContext || window.webkitAudioContext)();
                this.initialized = true;
            } catch (e) {
                console.warn('Web Audio API not supported');
                this.enabled = false;
            }
        }

        playLaunch(power) {
            if (!this.enabled || !this.audioContext) return;
            
            const oscillator = this.audioContext.createOscillator();
            const gainNode = this.audioContext.createGain();
            
            oscillator.connect(gainNode);
            gainNode.connect(this.audioContext.destination);
            
            // 根据力度调整音调
            const baseFreq = 200 + power * 8;
            oscillator.frequency.setValueAtTime(baseFreq, this.audioContext.currentTime);
            oscillator.frequency.exponentialRampToValueAtTime(80, this.audioContext.currentTime + 0.3);
            
            oscillator.type = 'sine';
            
            gainNode.gain.setValueAtTime(0.3, this.audioContext.currentTime);
            gainNode.gain.exponentialRampToValueAtTime(0.01, this.audioContext.currentTime + 0.3);
            
            oscillator.start(this.audioContext.currentTime);
            oscillator.stop(this.audioContext.currentTime + 0.3);
        }

        playStretch(power) {
            if (!this.enabled || !this.audioContext) return;
            
            const oscillator = this.audioContext.createOscillator();
            const gainNode = this.audioContext.createGain();
            
            oscillator.connect(gainNode);
            gainNode.connect(this.audioContext.destination);
            
            oscillator.frequency.setValueAtTime(150 + power * 2, this.audioContext.currentTime);
            oscillator.type = 'triangle';
            
            gainNode.gain.setValueAtTime(0.05, this.audioContext.currentTime);
            gainNode.gain.exponentialRampToValueAtTime(0.01, this.audioContext.currentTime + 0.1);
            
            oscillator.start(this.audioContext.currentTime);
            oscillator.stop(this.audioContext.currentTime + 0.1);
        }

        playBounce() {
            if (!this.enabled || !this.audioContext) return;
            
            const oscillator = this.audioContext.createOscillator();
            const gainNode = this.audioContext.createGain();
            
            oscillator.connect(gainNode);
            gainNode.connect(this.audioContext.destination);
            
            oscillator.frequency.setValueAtTime(300, this.audioContext.currentTime);
            oscillator.frequency.exponentialRampToValueAtTime(150, this.audioContext.currentTime + 0.1);
            oscillator.type = 'sine';
            
            gainNode.gain.setValueAtTime(0.2, this.audioContext.currentTime);
            gainNode.gain.exponentialRampToValueAtTime(0.01, this.audioContext.currentTime + 0.1);
            
            oscillator.start(this.audioContext.currentTime);
            oscillator.stop(this.audioContext.currentTime + 0.1);
        }

    class InputManager {
        constructor(canvas) {
            this.canvas = canvas;
            this.keys = {};
            this.mouse = {
                x: 0,
                y: 0,
                worldX: 0,
                worldY: 0,
                isDown: false,
                isDragging: false
            };
            this.dragStart = null;

            this.bindEvents();
        }

        bindEvents() {
            window.addEventListener('keydown', (e) => {
                this.keys[e.code] = true;
                if (e.code === 'Space') e.preventDefault();
            });

            window.addEventListener('keyup', (e) => {
                this.keys[e.code] = false;
            });

            this.canvas.addEventListener('mousedown', (e) => {
                const rect = this.canvas.getBoundingClientRect();
                this.mouse.x = e.clientX - rect.left;
                this.mouse.y = e.clientY - rect.top;
                this.mouse.isDown = true;
                this.dragStart = { x: this.mouse.x, y: this.mouse.y };
            });

            this.canvas.addEventListener('mouseup', () => {
                this.mouse.isDown = false;
                this.mouse.isDragging = false;
                this.dragStart = null;
            });

            this.canvas.addEventListener('mousemove', (e) => {
                const rect = this.canvas.getBoundingClientRect();
                this.mouse.x = e.clientX - rect.left;
                this.mouse.y = e.clientY - rect.top;

                if (this.mouse.isDown && this.dragStart) {
                    const dx = this.mouse.x - this.dragStart.x;
                    const dy = this.mouse.y - this.dragStart.y;
                    if (Math.abs(dx) > 5 || Math.abs(dy) > 5) {
                        this.mouse.isDragging = true;
                    }
                }
            });

            // 触摸支持
            this.canvas.addEventListener('touchstart', (e) => {
                e.preventDefault();
                const touch = e.touches[0];
                const rect = this.canvas.getBoundingClientRect();
                this.mouse.x = touch.clientX - rect.left;
                this.mouse.y = touch.clientY - rect.top;
                this.mouse.isDown = true;
                this.dragStart = { x: this.mouse.x, y: this.mouse.y };
            });

            this.canvas.addEventListener('touchend', (e) => {
                e.preventDefault();
                this.mouse.isDown = false;
                this.mouse.isDragging = false;
                this.dragStart = null;
            });

            this.canvas.addEventListener('touchmove', (e) => {
                e.preventDefault();
                const touch = e.touches[0];
                const rect = this.canvas.getBoundingClientRect();
                this.mouse.x = touch.clientX - rect.left;
                this.mouse.y = touch.clientY - rect.top;

                if (this.mouse.isDown && this.dragStart) {
                    const dx = this.mouse.x - this.dragStart.x;
                    const dy = this.mouse.y - this.dragStart.y;
                    if (Math.abs(dx) > 5 || Math.abs(dy) > 5) {
                        this.mouse.isDragging = true;
                    }
                }
            });
        }

        isKeyPressed(code) {
            return this.keys[code] === true;
        }

        updateWorldPosition(camera) {
            this.mouse.worldX = (this.mouse.x - CONFIG.CANVAS_WIDTH / 2) / camera.zoom + camera.x;
            this.mouse.worldY = (this.mouse.y - CONFIG.CANVAS_HEIGHT / 2) / camera.zoom + camera.y;
        }
    }

    class Camera {
        constructor() {
            this.position = new Vector2(CONFIG.CANVAS_WIDTH / 2, 0);
            this.target = new Vector2(CONFIG.CANVAS_WIDTH / 2, 0);
            this.zoom = 1.0;
            this.targetZoom = 1.0;
            this.x = CONFIG.CANVAS_WIDTH / 2;
            this.y = 0;
        }

        follow(targetX, targetY, immediate = false) {
            this.target.x = Utils.clamp(targetX, CONFIG.CANVAS_WIDTH / 2 / this.zoom, 
                                        CONFIG.WORLD_WIDTH - CONFIG.CANVAS_WIDTH / 2 / this.zoom);
            this.target.y = Utils.clamp(targetY, -200, 200);

            if (immediate) {
                this.position.x = this.target.x;
                this.position.y = this.target.y;
                this.x = this.target.x;
                this.y = this.target.y;
            }
        }

        setZoom(zoom) {
            this.targetZoom = Utils.clamp(zoom, CONFIG.MIN_ZOOM, CONFIG.MAX_ZOOM);
        }

        update(deltaTime) {
            this.position.x = Utils.lerp(this.position.x, this.target.x, CONFIG.CAMERA_LERP);
            this.position.y = Utils.lerp(this.position.y, this.target.y, CONFIG.CAMERA_LERP);
            this.zoom = Utils.lerp(this.zoom, this.targetZoom, 0.05);
            this.x = this.position.x;
            this.y = this.position.y;
        }

        worldToScreen(worldX, worldY) {
            return {
                x: (worldX - this.position.x) * this.zoom + CONFIG.CANVAS_WIDTH / 2,
                y: (worldY - this.position.y) * this.zoom + CONFIG.CANVAS_HEIGHT / 2
            };
        }

        screenToWorld(screenX, screenY) {
            return {
                x: (screenX - CONFIG.CANVAS_WIDTH / 2) / this.zoom + this.position.x,
                y: (screenY - CONFIG.CANVAS_HEIGHT / 2) / this.zoom + this.position.y
            };
        }
    }

    class Cloud {
        constructor(layer, worldWidth) {
            this.layer = layer;
            this.worldWidth = worldWidth;
            this.reset(true);
        }

        reset(initial = false) {
            this.width = Utils.randomRange(80, 200);
            this.height = Utils.randomRange(30, 60);
            this.x = initial ? Utils.randomRange(0, this.worldWidth) : -this.width;
            this.y = Utils.randomRange(30, 200);
            this.speed = Utils.randomRange(0.1, 0.3) * (1 + this.layer * 0.5);
            this.opacity = Utils.randomRange(0.3, 0.7) * (1 - this.layer * 0.2);
            
            this.puffs = [];
            const puffCount = Utils.randomInt(3, 5);
            for (let i = 0; i < puffCount; i++) {
                this.puffs.push({
                    offsetX: Utils.randomRange(-this.width * 0.3, this.width * 0.3),
                    offsetY: Utils.randomRange(-this.height * 0.2, this.height * 0.2),
                    radius: Utils.randomRange(this.height * 0.4, this.height * 0.7)
                });
            }
        }

        update(deltaTime) {
            this.x += this.speed;
            if (this.x > this.worldWidth + this.width) {
                this.reset();
            }
        }

        draw(ctx, camera) {
            const parallaxFactor = 0.1 + this.layer * 0.15;
            const screenX = (this.x - camera.position.x * parallaxFactor);
            const screenY = this.y;

            ctx.save();
            ctx.globalAlpha = this.opacity;

            this.puffs.forEach(puff => {
                const gradient = ctx.createRadialGradient(
                    screenX + puff.offsetX, screenY + puff.offsetY, 0,
                    screenX + puff.offsetX, screenY + puff.offsetY, Math.max(1, puff.radius)
                );
                gradient.addColorStop(0, 'rgba(255,255,255,0.9)');
                gradient.addColorStop(0.5, 'rgba(255,255,255,0.6)');
                gradient.addColorStop(1, 'rgba(255,255,255,0)');

                ctx.fillStyle = gradient;
                ctx.beginPath();
                ctx.arc(screenX + puff.offsetX, screenY + puff.offsetY, Math.max(1, puff.radius), 0, Math.PI * 2);
                ctx.fill();
            });

            ctx.restore();
        }
    }

    class BackgroundRenderer {
        constructor(worldWidth) {
            this.worldWidth = worldWidth;
            this.clouds = [];
            this.stars = [];
            
            for (let layer = 0; layer < 3; layer++) {
                const cloudCount = 4 - layer;
                for (let i = 0; i < cloudCount; i++) {
                    this.clouds.push(new Cloud(layer, worldWidth));
                }
            }

            for (let i = 0; i < 50; i++) {
                this.stars.push({
                    x: Utils.randomRange(0, CONFIG.CANVAS_WIDTH),
                    y: Utils.randomRange(10, 150),
                    size: Utils.randomRange(1, 3),
                    twinkle: Utils.randomRange(0, Math.PI * 2),
                    speed: Utils.randomRange(1, 3)
                });
            }
        }

        update(deltaTime) {
            this.clouds.forEach(cloud => cloud.update(deltaTime));
            
            this.stars.forEach(star => {
                star.twinkle += star.speed * deltaTime * 0.1;
            });
        }

        draw(ctx, camera) {
            const skyGradient = ctx.createLinearGradient(0, 0, 0, CONFIG.CANVAS_HEIGHT);
            skyGradient.addColorStop(0, '#0f0c29');
            skyGradient.addColorStop(0.3, '#302b63');
            skyGradient.addColorStop(0.7, '#24243e');
            skyGradient.addColorStop(1, '#1a1a2e');
            
            ctx.fillStyle = skyGradient;
            ctx.fillRect(0, 0, CONFIG.CANVAS_WIDTH, CONFIG.CANVAS_HEIGHT);

            this.stars.forEach(star => {
                const alpha = 0.3 + Math.sin(star.twinkle) * 0.3;
                ctx.fillStyle = `rgba(255, 255, 255, ${alpha})`;
                ctx.beginPath();
                ctx.arc(star.x, star.y, star.size, 0, Math.PI * 2);
                ctx.fill();
            });

            this.drawMountains(ctx, camera);
            this.clouds.forEach(cloud => cloud.draw(ctx, camera));
        }

        drawMountains(ctx, camera) {
            const groundY = CONFIG.CANVAS_HEIGHT - CONFIG.GROUND_HEIGHT;
            
            // 远山（视差小）
            ctx.fillStyle = '#1a1a3a';
            ctx.beginPath();
            ctx.moveTo(0, groundY);
            for (let x = 0; x <= CONFIG.CANVAS_WIDTH; x += 60) {
                const parallax = (camera.position.x - CONFIG.CANVAS_WIDTH / 2) * 0.1;
                const height = Utils.noise((x + parallax) * 0.008) * 80 + 20;
                ctx.lineTo(x, groundY - height);
            }
            ctx.lineTo(CONFIG.CANVAS_WIDTH, groundY);
            ctx.closePath();
            ctx.fill();

            // 近山（视差大）
            ctx.fillStyle = '#16213e';
            ctx.beginPath();
            ctx.moveTo(0, groundY);
            for (let x = 0; x <= CONFIG.CANVAS_WIDTH; x += 40) {
                const parallax = (camera.position.x - CONFIG.CANVAS_WIDTH / 2) * 0.2;
                const height = Utils.noise((x + parallax) * 0.012 + 100) * 50 + 10;
                ctx.lineTo(x, groundY - height);
            }
            ctx.lineTo(CONFIG.CANVAS_WIDTH, groundY);
            ctx.closePath();
            ctx.fill();
        }
    }

    class Bird {
        constructor(x, y, colorIndex = 0) {
            this.position = new Vector2(x, y);
            this.velocity = new Vector2(0, 0);
            this.radius = CONFIG.BIRD.RADIUS;
            this.colorIndex = colorIndex;
            this.color = CONFIG.BIRD.COLORS[colorIndex % CONFIG.BIRD.COLORS.length];
            this.rotation = 0;
            this.angularVelocity = 0;
            this.state = 'ready'; // ready, aiming, flying, stopped
            this.groundY = CONFIG.CANVAS_HEIGHT - CONFIG.GROUND_HEIGHT - this.radius;
            this.trail = [];
            this.maxTrailLength = 30;
            this.bounceCount = 0;
        }

        launch(velocity) {
            this.velocity = velocity;
            this.state = 'flying';
            this.trail = [];
        }

        update(deltaTime, physics) {
            if (this.state === 'flying') {
                // 应用重力
                this.velocity.y += physics.GRAVITY;
                
                // 应用空气阻力
                this.velocity.x *= physics.AIR_RESISTANCE;
                this.velocity.y *= physics.AIR_RESISTANCE;
                
                // 更新位置
                this.position.x += this.velocity.x;
                this.position.y += this.velocity.y;
                
                // 旋转随速度变化
                this.angularVelocity = this.velocity.x * 0.02;
                this.rotation += this.angularVelocity;
                this.angularVelocity *= physics.ANGGULAR_DAMPING || 0.98;
                
                // 记录轨迹
                if (this.trail.length === 0 || 
                    Utils.distance(this.position.x, this.position.y, 
                                   this.trail[this.trail.length - 1].x, 
                                   this.trail[this.trail.length - 1].y) > 5) {
                    this.trail.push({ x: this.position.x, y: this.position.y });
                    if (this.trail.length > this.maxTrailLength) {
                        this.trail.shift();
                    }
                }
                
                // 地面碰撞
                if (this.position.y > this.groundY) {
                    this.position.y = this.groundY;
                    this.velocity.y = -this.velocity.y * physics.GROUND_FRICTION;
                    this.velocity.x *= physics.GROUND_FRICTION;
                    this.bounceCount++;
                    
                    // 检查是否停止
                    if (Math.abs(this.velocity.x) < 0.5 && Math.abs(this.velocity.y) < 0.5) {
                        this.state = 'stopped';
                        this.velocity.set(0, 0);
                    }
                }
                
                // 边界检查
                if (this.position.x < this.radius) {
                    this.position.x = this.radius;
                    this.velocity.x = -this.velocity.x * 0.6;
                }
                if (this.position.x > CONFIG.WORLD_WIDTH - this.radius) {
                    this.position.x = CONFIG.WORLD_WIDTH - this.radius;
                    this.velocity.x = -this.velocity.x * 0.6;
                }
            }
        }

        draw(ctx, camera) {
            const screen = camera.worldToScreen(this.position.x, this.position.y);
            const screenRadius = this.radius * camera.zoom;
            
            // 绘制轨迹
            if (this.state === 'flying' && this.trail.length > 1) {
                ctx.save();
                ctx.globalAlpha = 0.4;
                ctx.strokeStyle = this.color;
                ctx.lineWidth = 3 * camera.zoom;
                ctx.lineCap = 'round';
                ctx.beginPath();
                
                for (let i = 0; i < this.trail.length; i++) {
                    const tp = camera.worldToScreen(this.trail[i].x, this.trail[i].y);
                    if (i === 0) {
                        ctx.moveTo(tp.x, tp.y);
                    } else {
                        ctx.lineTo(tp.x, tp.y);
                    }
                }
                ctx.stroke();
                ctx.restore();
            }
            
            ctx.save();
            ctx.translate(screen.x, screen.y);
            ctx.rotate(this.rotation);
            
            // 发光效果
            const glowGradient = ctx.createRadialGradient(0, 0, 0, 0, 0, screenRadius * 1.5);
            glowGradient.addColorStop(0, this.color + '40');
            glowGradient.addColorStop(1, this.color + '00');
            ctx.fillStyle = glowGradient;
            ctx.beginPath();
            ctx.arc(0, 0, screenRadius * 1.5, 0, Math.PI * 2);
            ctx.fill();
            
            // 主体渐变
            const bodyGradient = ctx.createRadialGradient(
                -screenRadius * 0.3, -screenRadius * 0.3, 0,
                0, 0, screenRadius
            );
            bodyGradient.addColorStop(0, '#ffffff');
            bodyGradient.addColorStop(0.3, this.color);
            bodyGradient.addColorStop(1, this.darkenColor(this.color, 40));
            
            ctx.fillStyle = bodyGradient;
            ctx.beginPath();
            ctx.arc(0, 0, screenRadius, 0, Math.PI * 2);
            ctx.fill();
            
            // 眼睛
            const eyeOffset = screenRadius * 0.35;
            const eyeRadius = screenRadius * 0.25;
            
            // 眼白
            ctx.fillStyle = '#ffffff';
            ctx.beginPath();
            ctx.ellipse(-eyeOffset, -screenRadius * 0.1, eyeRadius * 1.2, eyeRadius, 0, 0, Math.PI * 2);
            ctx.fill();
            ctx.beginPath();
            ctx.ellipse(eyeOffset, -screenRadius * 0.1, eyeRadius * 1.2, eyeRadius, 0, 0, Math.PI * 2);
            ctx.fill();
            
            // 瞳孔 - 看向飞行方向
            const pupilOffset = this.state === 'flying' ? 
                Math.min(this.velocity.x * 0.02, eyeRadius * 0.4) : 0;
            ctx.fillStyle = '#1a1a2e';
            ctx.beginPath();
            ctx.arc(-eyeOffset + pupilOffset, -screenRadius * 0.1, eyeRadius * 0.5, 0, Math.PI * 2);
            ctx.fill();
            ctx.beginPath();
            ctx.arc(eyeOffset + pupilOffset, -screenRadius * 0.1, eyeRadius * 0.5, 0, Math.PI * 2);
            ctx.fill();
            
            // 眉毛 - 愤怒表情
            ctx.strokeStyle = '#1a1a2e';
            ctx.lineWidth = screenRadius * 0.12;
            ctx.lineCap = 'round';
            
            ctx.beginPath();
            ctx.moveTo(-screenRadius * 0.6, -screenRadius * 0.45);
            ctx.lineTo(-screenRadius * 0.15, -screenRadius * 0.55);
            ctx.stroke();
            
            ctx.beginPath();
            ctx.moveTo(screenRadius * 0.6, -screenRadius * 0.45);
            ctx.lineTo(screenRadius * 0.15, -screenRadius * 0.55);
            ctx.stroke();
            
            // 喙
            ctx.fillStyle = '#f39c12';
            ctx.beginPath();
            ctx.moveTo(0, screenRadius * 0.1);
            ctx.lineTo(-screenRadius * 0.25, screenRadius * 0.35);
            ctx.lineTo(screenRadius * 0.25, screenRadius * 0.35);
            ctx.closePath();
            ctx.fill();
            
            ctx.restore();
        }

        darkenColor(color, amount) {
            const hex = color.replace('#', '');
            const r = Math.max(0, parseInt(hex.substr(0, 2), 16) - amount);
            const g = Math.max(0, parseInt(hex.substr(2, 2), 16) - amount);
            const b = Math.max(0, parseInt(hex.substr(4, 2), 16) - amount);
            return `#${r.toString(16).padStart(2, '0')}${g.toString(16).padStart(2, '0')}${b.toString(16).padStart(2, '0')}`;
        }

        isStopped() {
            return this.state === 'stopped' || this.bounceCount > 5;
        }
    }

    class Slingshot {
        constructor(x) {
            this.x = x;
            this.baseY = CONFIG.CANVAS_HEIGHT - CONFIG.GROUND_HEIGHT;
            this.y = this.baseY - 100;
            this.width = 30;
            this.height = 120;
            this.forkHeight = 80;
            this.forkWidth = 50;
            
            // 橡皮筋锚点
            this.leftAnchor = new Vector2(x - this.forkWidth / 2, this.baseY - this.forkHeight);
            this.rightAnchor = new Vector2(x + this.forkWidth / 2, this.baseY - this.forkHeight);
            
            // 拉伸状态
            this.pulledPosition = new Vector2(x, this.y);
            this.isPulling = false;
            this.pullDistance = 0;
            this.pullAngle = 0;
            this.maxStretch = CONFIG.SLINGSHOT.MAX_STRETCH;
            
            // 发射的小鸟
            this.currentBird = null;
            
            // 橡皮筋弹性动画
            this.bandTension = 0;
            this.snapBackAnim = 0;
        }

        attachBird(bird) {
            this.currentBird = bird;
            bird.position.set(this.x, this.y);
            bird.state = 'ready';
        }

        startPull(worldX, worldY) {
            if (!this.currentBird || this.currentBird.state !== 'ready') return false;
            
            // 检查是否点击在弹弓区域
            const dist = Utils.distance(worldX, worldY, this.x, this.y);
            if (dist < 80) {
                this.isPulling = true;
                this.currentBird.state = 'aiming';
                return true;
            }
            return false;
        }

        updatePull(worldX, worldY) {
            if (!this.isPulling) return;
            
            // 计算拉伸向量
            const dx = this.x - worldX;
            const dy = this.y - worldY;
            
            // 限制最大拉伸距离
            const dist = Math.min(Math.sqrt(dx * dx + dy * dy), this.maxStretch);
            const angle = Math.atan2(dy, dx);
            
            this.pullDistance = dist;
            this.pullAngle = angle;
            this.bandTension = dist / this.maxStretch;
            
            // 计算小鸟位置（被拉向反方向）
            this.pulledPosition.x = this.x - Math.cos(angle) * dist;
            this.pulledPosition.y = this.y - Math.sin(angle) * dist;
            
            if (this.currentBird) {
                this.currentBird.position.set(this.pulledPosition.x, this.pulledPosition.y);
            }
        }

        release() {
            if (!this.isPulling || !this.currentBird) {
                this.isPulling = false;
                return null;
            }
            
            // 计算发射速度
            const power = this.pullDistance * CONFIG.BIRD.LAUNCH_POWER;
            const clampedPower = Utils.clamp(power, CONFIG.BIRD.MIN_POWER, CONFIG.BIRD.MAX_POWER);
            
            const velocity = new Vector2(
                Math.cos(this.pullAngle) * clampedPower,
                Math.sin(this.pullAngle) * clampedPower
            );
            
            // 发射小鸟
            const bird = this.currentBird;
            bird.launch(velocity);
            
            // 重置状态
            this.currentBird = null;
            this.isPulling = false;
            this.pullDistance = 0;
            this.snapBackAnim = 1;
            
            return { bird, power: clampedPower };
        }

        update(deltaTime) {
            // 橡皮筋回弹动画
            if (this.snapBackAnim > 0) {
                this.snapBackAnim -= deltaTime * 0.008;
                if (this.snapBackAnim < 0) this.snapBackAnim = 0;
            }
        }

        getLaunchVelocity() {
            if (this.pullDistance === 0) return new Vector2(0, 0);
            
            const power = this.pullDistance * CONFIG.BIRD.LAUNCH_POWER;
            const clampedPower = Utils.clamp(power, CONFIG.BIRD.MIN_POWER, CONFIG.BIRD.MAX_POWER);
            
            return new Vector2(
                Math.cos(this.pullAngle) * clampedPower,
                Math.sin(this.pullAngle) * clampedPower
            );
        }

        draw(ctx, camera) {
            const baseScreen = camera.worldToScreen(this.x, this.baseY);
            const leftScreen = camera.worldToScreen(this.leftAnchor.x, this.leftAnchor.y);
            const rightScreen = camera.worldToScreen(this.rightAnchor.x, this.rightAnchor.y);
            const pulledScreen = camera.worldToScreen(this.pulledPosition.x, this.pulledPosition.y);
            const zoom = camera.zoom;
            
            // 绘制后橡皮筋（在小鸟后面）
            if (this.isPulling || this.snapBackAnim > 0) {
                this.drawBand(ctx, leftScreen, pulledScreen, zoom, 'back');
                this.drawBand(ctx, rightScreen, pulledScreen, zoom, 'back');
            }
            
            // 绘制弹弓支架
            this.drawFrame(ctx, baseScreen, leftScreen, rightScreen, zoom);
            
            // 绘制轨迹预测
            if (this.isPulling && this.pullDistance > 10) {
                this.drawTrajectory(ctx, camera);
            }
            
            // 绘制前橡皮筋（在小鸟前面）
            if (this.isPulling || this.snapBackAnim > 0) {
                this.drawBand(ctx, leftScreen, pulledScreen, zoom, 'front');
                this.drawBand(ctx, rightScreen, pulledScreen, zoom, 'front');
            } else {
                // 静止状态的橡皮筋
                this.drawRestingBand(ctx, leftScreen, rightScreen, zoom);
            }
        }

        drawFrame(ctx, base, left, right, zoom) {
            // 主支架
            const woodGradient = ctx.createLinearGradient(base.x - 15 * zoom, base.y, base.x + 15 * zoom, base.y);
            woodGradient.addColorStop(0, '#5D4037');
            woodGradient.addColorStop(0.5, '#8D6E63');
            woodGradient.addColorStop(1, '#5D4037');
            
            ctx.fillStyle = woodGradient;
            
            // 左支柱
            ctx.beginPath();
            ctx.moveTo(base.x - 8 * zoom, base.y);
            ctx.lineTo(left.x - 5 * zoom, left.y + 10 * zoom);
            ctx.lineTo(left.x + 3 * zoom, left.y + 15 * zoom);
            ctx.lineTo(base.x - 2 * zoom, base.y);
            ctx.closePath();
            ctx.fill();
            
            // 右支柱
            ctx.beginPath();
            ctx.moveTo(base.x + 8 * zoom, base.y);
            ctx.lineTo(right.x + 5 * zoom, right.y + 10 * zoom);
            ctx.lineTo(right.x - 3 * zoom, right.y + 15 * zoom);
            ctx.lineTo(base.x + 2 * zoom, base.y);
            ctx.closePath();
            ctx.fill();
            
            // 叉子顶部装饰
            ctx.fillStyle = '#A0522D';
            ctx.beginPath();
            ctx.arc(left.x, left.y, 6 * zoom, 0, Math.PI * 2);
            ctx.fill();
            ctx.beginPath();
            ctx.arc(right.x, right.y, 6 * zoom, 0, Math.PI * 2);
            ctx.fill();
        }

        drawBand(ctx, anchor, target, zoom, layer) {
            const tension = this.bandTension;
            const bandWidth = CONFIG.SLINGSHOT.BAND_WIDTH * zoom * (1 - tension * 0.3);
            
            // 根据张力调整颜色
            const r = Math.round(139 + tension * 50);
            const g = Math.round(69 - tension * 20);
            const b = Math.round(19);
            const bandColor = `rgb(${r},${g},${b})`;
            
            ctx.strokeStyle = bandColor;
            ctx.lineWidth = bandWidth;
            ctx.lineCap = 'round';
            
            // 计算控制点使橡皮筋有弹性曲线
            const midX = (anchor.x + target.x) / 2;
            const midY = (anchor.y + target.y) / 2;
            const curve = (1 - tension) * 20 * zoom;
            
            ctx.beginPath();
            ctx.moveTo(anchor.x, anchor.y);
            
            if (layer === 'back') {
                ctx.quadraticCurveTo(midX + curve, midY, target.x, target.y);
            } else {
                ctx.quadraticCurveTo(midX - curve, midY, target.x, target.y);
            }
            
            ctx.stroke();
        }

        drawRestingBand(ctx, left, right, zoom) {
            ctx.strokeStyle = CONFIG.SLINGSHOT.BAND_COLOR;
            ctx.lineWidth = CONFIG.SLINGSHOT.BAND_WIDTH * zoom;
            ctx.lineCap = 'round';
            
            // 弯曲的静止橡皮筋
            const midX = (left.x + right.x) / 2;
            const midY = (left.y + right.y) / 2 + 15 * zoom;
            
            ctx.beginPath();
            ctx.moveTo(left.x, left.y);
            ctx.quadraticCurveTo(midX, midY, right.x, right.y);
            ctx.stroke();
        }

        drawTrajectory(ctx, camera) {
            const velocity = this.getLaunchVelocity();
            const startX = this.pulledPosition.x;
            const startY = this.pulledPosition.y;
            
            ctx.save();
            
            const points = CONFIG.TRAJECTORY.POINT_COUNT;
            let x = startX;
            let y = startY;
            let vx = velocity.x;
            let vy = velocity.y;
            
            for (let i = 0; i < points; i++) {
                // 模拟物理
                vy += CONFIG.PHYSICS.GRAVITY;
                vx *= CONFIG.PHYSICS.AIR_RESISTANCE;
                vy *= CONFIG.PHYSICS.AIR_RESISTANCE;
                x += vx;
                y += vy;
                
                // 检查是否碰到地面
                if (y > CONFIG.CANVAS_HEIGHT - CONFIG.GROUND_HEIGHT) break;
                
                const screen = camera.worldToScreen(x, y);
                
                // 计算点的透明度和大小
                const progress = i / points;
                const alpha = 1 - progress * 0.8;
                const size = Utils.lerp(CONFIG.TRAJECTORY.POINT_SIZE_START, 
                                        CONFIG.TRAJECTORY.POINT_SIZE_END, progress) * camera.zoom;
                
                // 绘制点
                ctx.fillStyle = `rgba(255, 255, 255, ${alpha})`;
                ctx.beginPath();
                ctx.arc(screen.x, screen.y, Math.max(1, size), 0, Math.PI * 2);
                ctx.fill();
            }
            
            ctx.restore();
        }
    }

    class GroundRenderer {
        constructor() {
            this.grassBlades = [];
            this.generateGrass();
        }

        generateGrass() {
            for (let i = 0; i < 200; i++) {
                this.grassBlades.push({
                    x: Utils.randomRange(0, CONFIG.WORLD_WIDTH),
                    height: Utils.randomRange(8, 25),
                    sway: Utils.randomRange(0, Math.PI * 2),
                    swaySpeed: Utils.randomRange(0.02, 0.05)
                });
            }
        }

        update(deltaTime) {
            this.grassBlades.forEach(blade => {
                blade.sway += blade.swaySpeed;
            });
        }

        draw(ctx, camera) {
            const groundY = CONFIG.CANVAS_HEIGHT - CONFIG.GROUND_HEIGHT;
            
            // 地面基础
            const groundGradient = ctx.createLinearGradient(0, groundY, 0, CONFIG.CANVAS_HEIGHT);
            groundGradient.addColorStop(0, '#3d5a3d');
            groundGradient.addColorStop(0.1, '#2d4a2d');
            groundGradient.addColorStop(0.5, '#1d3a1d');
            groundGradient.addColorStop(1, '#0d2a0d');
            
            ctx.fillStyle = groundGradient;
            ctx.fillRect(0, groundY, CONFIG.CANVAS_WIDTH, CONFIG.GROUND_HEIGHT);
            
            // 草地层
            ctx.fillStyle = '#4a7a4a';
            ctx.beginPath();
            ctx.moveTo(0, groundY);
            
            for (let x = 0; x <= CONFIG.CANVAS_WIDTH; x += 20) {
                const worldX = x / camera.zoom + camera.position.x - CONFIG.CANVAS_WIDTH / (2 * camera.zoom);
                const noise = Utils.noise(worldX * 0.05) * 8;
                ctx.lineTo(x, groundY + noise);
            }
            
            ctx.lineTo(CONFIG.CANVAS_WIDTH, groundY + 10);
            ctx.lineTo(0, groundY + 10);
            ctx.closePath();
            ctx.fill();
            
            // 草叶
            ctx.strokeStyle = '#5a9a5a';
            ctx.lineWidth = 2;
            
            this.grassBlades.forEach(blade => {
                const screenX = (blade.x - camera.position.x) * camera.zoom + CONFIG.CANVAS_WIDTH / 2;
                if (screenX < -20 || screenX > CONFIG.CANVAS_WIDTH + 20) return;
                
                const sway = Math.sin(blade.sway) * 5;
                
                ctx.beginPath();
                ctx.moveTo(screenX, groundY);
                ctx.quadraticCurveTo(
                    screenX + sway,
                    groundY - blade.height / 2,
                    screenX + sway * 1.5,
                    groundY - blade.height
                );
                ctx.stroke();
            });
        }
    }

    class Game {
        constructor() {
            this.canvas = document.getElementById('gameCanvas');
            this.ctx = this.canvas.getContext('2d');
            this.canvas.width = CONFIG.CANVAS_WIDTH;
            this.canvas.height = CONFIG.CANVAS_HEIGHT;
            
            this.score = 0;
            this.birdsLeft = 3;
            this.currentBirdIndex = 0;
            
            // 初始化系统
            this.camera = new Camera();
            this.input = new InputManager(this.canvas);
            this.sound = new SoundManager();
            this.background = new BackgroundRenderer(CONFIG.WORLD_WIDTH);
            this.ground = new GroundRenderer();
            
            // 弹弓
            this.slingshot = new Slingshot(CONFIG.SLINGSHOT.X);
            
            // 小鸟队列
            this.birds = [];
            this.flyingBirds = [];
            
            // 初始化小鸟
            this.spawnNextBird();
            
            // 游戏状态
            this.isAiming = false;
            this.lastTime = 0;
            this.running = true;
            
            // UI元素
            this.scoreDisplay = document.getElementById('scoreDisplay');
            this.birdsDisplay = document.getElementById('birdsDisplay');
            this.debugPanel = document.getElementById('debugPanel');
            this.powerIndicator = document.getElementById('powerIndicator');
            this.powerFill = document.getElementById('powerFill');
            
            this.bindEvents();
            this.gameLoop(0);
        }

        spawnNextBird() {
            if (this.birdsLeft <= 0) {
                // 游戏结束逻辑
                return;
            }
            
            const bird = new Bird(
                CONFIG.SLINGSHOT.X,
                CONFIG.SLINGSHOT.Y - 100,
                this.currentBirdIndex
            );
            
            this.slingshot.attachBird(bird);
            this.currentBirdIndex++;
            this.birdsLeft--;
            this.updateUI();
        }

        bindEvents() {
            // 鼠标事件
            this.canvas.addEventListener('mousedown', (e) => {
                this.sound.init(); // 需要用户交互来初始化音频
                this.handlePointerDown(e);
            });
            
            this.canvas.addEventListener('mousemove', (e) => {
                this.handlePointerMove(e);
            });
            
            this.canvas.addEventListener('mouseup', (e) => {
                this.handlePointerUp(e);
            });
            
            // 触摸事件
            this.canvas.addEventListener('touchstart', (e) => {
                this.sound.init();
                this.handlePointerDown(e.touches[0]);
            });
            
            this.canvas.addEventListener('touchmove', (e) => {
                this.handlePointerMove(e.touches[0]);
            });
            
            this.canvas.addEventListener('touchend', (e) => {
                this.handlePointerUp(e);
            });
            
            // 键盘事件
            window.addEventListener('keydown', (e) => {
                if (e.code === 'KeyR') {
                    this.resetGame();
                }
            });
        }

        handlePointerDown(e) {
            const rect = this.canvas.getBoundingClientRect();
            const screenX = e.clientX - rect.left;
            const screenY = e.clientY - rect.top;
            
            const world = this.camera.screenToWorld(screenX, screenY);
            
            if (this.slingshot.startPull(world.x, world.y)) {
                this.isAiming = true;
                this.powerIndicator.style.display = 'block';
            }
        }

        handlePointerMove(e) {
            if (!this.isAiming) return;
            
            const rect = this.canvas.getBoundingClientRect();
            const screenX = e.clientX - rect.left;
            const screenY = e.clientY - rect.top;
            
            const world = this.camera.screenToWorld(screenX, screenY);
            this.slingshot.updatePull(world.x, world.y);
            
            // 更新力度指示器
            const powerPercent = (this.slingshot.pullDistance / CONFIG.SLINGSHOT.MAX_STRETCH) * 100;
            this.powerFill.style.width = powerPercent + '%';
            
            // 根据力度改变颜色
            let color;
            if (powerPercent < 30) {
                color = '#4ade80'; // 绿色
            } else if (powerPercent < 70) {
                color = '#fbbf24'; // 黄色
            } else {
                color = '#ef4444'; // 红色
            }
            this.powerFill.style.backgroundColor = color;
            
            // 播放拉伸音效
            if (Math.random() < 0.1) {
                this.sound.playStretch(this.slingshot.pullDistance);
            }
        }

        handlePointerUp(e) {
            if (!this.isAiming) return;
            
            const result = this.slingshot.release();
            
            if (result) {
                this.flyingBirds.push(result.bird);
                this.sound.playLaunch(result.power);
                
                // 延迟生成下一只小鸟
                setTimeout(() => {
                    this.spawnNextBird();
                }, 2000);
            }
            
            this.isAiming = false;
            this.powerIndicator.style.display = 'none';
        }

        resetGame() {
            this.score = 0;
            this.birdsLeft = 3;
            this.currentBirdIndex = 0;
            this.flyingBirds = [];
            
            this.camera = new Camera();
            this.slingshot = new Slingshot(CONFIG.SLINGSHOT.X);
            
            this.spawnNextBird();
            this.updateUI();
        }

        updateUI() {
            this.scoreDisplay.textContent = this.score;
            this.birdsDisplay.textContent = this.birdsLeft;
        }

        update(deltaTime) {
            // 相机控制
            if (this.input.isKeyPressed('KeyA')) {
                this.camera.follow(this.camera.target.x - 10, this.camera.target.y);
            }
            if (this.input.isKeyPressed('KeyD')) {
                this.camera.follow(this.camera.target.x + 10, this.camera.target.y);
            }
            if (this.input.isKeyPressed('KeyW')) {
                this.camera.setZoom(this.camera.targetZoom + 0.02);
            }
            if (this.input.isKeyPressed('KeyS')) {
                this.camera.setZoom(this.camera.targetZoom - 0.02);
            }
            
            // 跟随飞行的小鸟
            let followBird = null;
            this.flyingBirds.forEach(bird => {
                if (bird.state === 'flying') {
                    followBird = bird;
                }
            });
            
            if (followBird) {
                this.camera.follow(followBird.position.x, followBird.position.y);
            } else if (this.slingshot.currentBird) {
                // 回到弹弓位置
                this.camera.follow(CONFIG.SLINGSHOT.X, 0);
            }
            
            // 更新各系统
            this.camera.update(deltaTime);
            this.background.update(deltaTime);
            this.ground.update(deltaTime);
            this.slingshot.update(deltaTime);
            this.input.updateWorldPosition(this.camera);
            
            // 更新飞行中的小鸟
            this.flyingBirds.forEach(bird => {
                bird.update(deltaTime, CONFIG.PHYSICS);
            });
            
            // 移除停止的小鸟
            this.flyingBirds = this.flyingBirds.filter(bird => !bird.isStopped());
            
            // 更新调试面板
            this.debugPanel.innerHTML = `
                FPS: ${Math.round(1000 / deltaTime)}<br>
                Camera: (${Math.round(this.camera.x)}, ${Math.round(this.camera.y)})<br>
                Zoom: ${this.camera.zoom.toFixed(2)}<br>
                Birds Flying: ${this.flyingBirds.length}
            `;
        }

        render() {
            // 清空画布
            this.ctx.clearRect(0, 0, CONFIG.CANVAS_WIDTH, CONFIG.CANVAS_HEIGHT);
            
            // 绘制背景
            this.background.draw(this.ctx, this.camera);
            
            // 绘制地面
            this.ground.draw(this.ctx, this.camera);
            
            // 绘制弹弓（后层）
            this.slingshot.draw(this.ctx, this.camera);
            
            // 绘制瞄准中的小鸟
            if (this.slingshot.currentBird) {
                this.slingshot.currentBird.draw(this.ctx, this.camera);
            }
            
            // 绘制飞行中的小鸟
            this.flyingBirds.forEach(bird => {
                bird.draw(this.ctx, this.camera);
            });
        }

        gameLoop(timestamp) {
            const deltaTime = timestamp - this.lastTime;
            this.lastTime = timestamp;
            
            if (this.running) {
                this.update(deltaTime || 16);
                this.render();
            }
            
            requestAnimationFrame((t) => this.gameLoop(t));
        }
    }

    // 启动游戏
    window.addEventListener('load', () => {
        new Game();
    });

</script>
```

## 实现说明

### 核心功能

1. **弹弓交互系统** (`Slingshot` 类)
   - 检测弹弓区域的点击/触摸
   - 计算拉伸矢量（方向 + 力度）
   - 限制最大拉伸距离
   - 释放时计算初速度

2. **橡皮筋视觉效果**
   - 根据张力动态调整颜色（从棕色到红棕色）
   - 使用贝塞尔曲线实现弹性形变
   - 张力越大，橡皮筋越细
   - 静止时呈现自然弯曲状态

3. **轨迹预测**
   - 使用物理模拟预测飞行路径
   - 虚线点从大变小，透明度渐变
   - 实时更新预测轨迹

4. **音效反馈** (`SoundManager` 类)
   - 使用 Web Audio API 生成音效
   - 拉伸音效（随力度变化音调）
   - 发射音效（弹性释放感）

5. **力度指示器**
   - 实时显示拉力百分比
   - 颜色从绿色→黄色→红色变化

### 交互方式

- **鼠标拖拽**：在弹弓区域按住并拖动来瞄准
- **触摸支持**：完全支持移动端触摸操作
- **释放发射**：松开鼠标/手指自动发射小鸟

### 视觉效果

- 小鸟带有渐变、发光、愤怒表情
- 飞行轨迹拖尾效果
- 相机自动跟随飞行中的小鸟
