        // 游戏引擎核心模块
        const GameEngine = (function() {

                level: 1,
                keys: {},
                lastTime: 0

            // 游戏配置
            const config = {

            // 渲染系统
            const renderer = {
                clear() {
                    ctx.fillStyle = '#111';
                    ctx.fillRect(0, 0, config.canvasWidth, config.canvasHeight);
                },
                
                drawEntity(entity) {
                    ctx.save();
                    
                    if (entity.type === 'player' || entity.type === 'enemy') {
                        // 绘制坦克
                        ctx.fillStyle = entity.color;
                        ctx.fillRect(entity.x - entity.width/2, entity.y - entity.height/2, entity.width, entity.height);
                        
                        // 绘制炮管
                        ctx.fillStyle = '#333';
                        const barrelLength = entity.width * 0.8;
                        const barrelWidth = entity.width * 0.2;
                        
                        if (entity.direction === 'up') {
                            ctx.fillRect(entity.x - barrelWidth/2, entity.y - entity.height/2 - barrelLength/2, barrelWidth, barrelLength);
                        } else if (entity.direction === 'down') {
                            ctx.fillRect(entity.x - barrelWidth/2, entity.y + entity.height/2 - barrelLength/2, barrelWidth, barrelLength);
                        } else if (entity.direction === 'left') {
                            ctx.fillRect(entity.x - entity.width/2 - barrelLength/2, entity.y - barrelWidth/2, barrelLength, barrelWidth);
                        } else if (entity.direction === 'right') {
                            ctx.fillRect(entity.x + entity.width/2 - barrelLength/2, entity.y - barrelWidth/2, barrelLength, barrelWidth);
                        }
                    } else if (entity.type === 'bullet') {
                        // 绘制子弹
                        ctx.fillStyle = entity.color;
                        ctx.beginPath();
                        ctx.arc(entity.x, entity.y, entity.radius, 0, Math.PI * 2);
                        ctx.fill();
                    } else if (entity.type === 'wall') {
                        // 绘制墙壁
                        ctx.fillStyle = entity.color;
                        ctx.fillRect(entity.x, entity.y, entity.width, entity.height);
                    }
                    
                    ctx.restore();
                },
                
                render() {
                    this.clear();
                    
                    // 渲染所有实体
                    for (const entity of entities.walls) {
                        this.drawEntity(entity);
                    }
                    
                    if (entities.player) {
                        this.drawEntity(entities.player);
                    }
                    
                    for (const enemy of entities.enemies) {
                        this.drawEntity(enemy);
                    }
                    
                    for (const bullet of entities.bullets) {
                        this.drawEntity(bullet);
                    }
                }
            };

                        state.keys[e.key] = false;

                updatePlayer() {
                    if (!entities.player) return;
                    
                    const player = entities.player;
                    let moved = false;

                        player.y -= config.playerSpeed;
                        player.direction = 'up';
                        moved = true;

                    if (state.keys['ArrowDown'] || state.keys['s']) {
                        player.y += config.playerSpeed;
                        player.direction = 'down';
                        moved = true;
                    }
                    if (state.keys['ArrowLeft'] || state.keys['a']) {
                        player.x -= config.playerSpeed;
                        player.direction = 'left';
                        moved = true;
                    }
                    if (state.keys['ArrowRight'] || state.keys['d']) {
                        player.x += config.playerSpeed;
                        player.direction = 'right';
                        moved = true;
                    }
                    
                    // 边界检查
                    player.x = Math.max(player.width/2, Math.min(config.canvasWidth - player.width/2, player.x));
                    player.y = Math.max(player.height/2, Math.min(config.canvasHeight - player.height/2, player.y));
                }

            // 实体创建函数
            function createPlayer() {
                return {

            function createEnemy() {
                const side = Math.floor(Math.random() * 4);
                let x, y;
                
                switch(side) {
                    case 0: // 上
                        x = Math.random() * config.canvasWidth;
                        y = -20;
                        break;
                    case 1: // 右
                        x = config.canvasWidth + 20;
                        y = Math.random() * config.canvasHeight;
                        break;
                    case 2: // 下
                        x = Math.random() * config.canvasWidth;
                        y = config.canvasHeight + 20;
                        break;
                    case 3: // 左

                        y = Math.random() * config.canvasHeight;
                        break;
                }
                
                return {
                    type: 'enemy',
                    x: x,
                    y: y,
                    width: 25,
                    height: 25,
                    color: '#ff4444',

                    speed: config.enemySpeed,

            function createBullet(x, y, direction, isPlayerBullet = true) {
                return {
                    type: 'bullet',
                    x: x,
                    y: y,
                    radius: 4,
                    color: isPlayerBullet ? '#FFEB3B' : '#FF5722',
                    direction: direction,
                    speed: config.bulletSpeed,
                    isPlayerBullet: isPlayerBullet
                };
            }
            
            function createWall(x, y, width, height) {
                return {
                    type: 'wall',
                    x: x,
                    y: y,
                    width: width,
                    height: height,
                    color: '#795548'
                };
            }
            
            // 子弹发射函数
            function fireBullet(shooter) {
                const bulletSpeed = config.bulletSpeed;
                let bulletX = shooter.x;
                let bulletY = shooter.y;
                
                // 根据坦克方向调整子弹起始位置
                if (shooter.direction === 'up') {
                    bulletY -= shooter.height / 2;
                } else if (shooter.direction === 'down') {
                    bulletY += shooter.height / 2;
                } else if (shooter.direction === 'left') {
                    bulletX -= shooter.width / 2;
                } else if (shooter.direction === 'right') {
                    bulletX += shooter.width / 2;
                }
                
                entities.bullets.push(createBullet(bulletX, bulletY, shooter.direction, shooter.type === 'player'));
            }
            
            // 碰撞检测
            function checkCollision(rect1, rect2) {
                return rect1.x < rect2.x + rect2.width &&
                       rect1.x + rect1.width > rect2.x &&
                       rect1.y < rect2.y + rect2.height &&
                       rect1.y + rect1.height > rect2.y;
            }
            
            function checkCircleRectCollision(circle, rect) {
                let distX = Math.abs(circle.x - rect.x - rect.width/2);
                let distY = Math.abs(circle.y - rect.y - rect.height/2);
                
                if (distX > (rect.width/2 + circle.radius)) return false;
                if (distY > (rect.height/2 + circle.radius)) return false;
                
                if (distX <= (rect.width/2)) return true;
                if (distY <= (rect.height/2)) return true;
                
                let dx = distX - rect.width/2;
                let dy = distY - rect.height/2;
                return (dx*dx + dy*dy <= (circle.radius*circle.radius));
            }
            
            // 更新游戏逻辑
            function update(currentTime) {
                if (!state.isRunning) return;
                
                const deltaTime = currentTime - state.lastTime;
                state.lastTime = currentTime;
                
                // 更新玩家
                input.updatePlayer();
                
                // 更新敌人
                for (let i = entities.enemies.length - 1; i >= 0; i--) {
                    const enemy = entities.enemies[i];
                    
                    // 简单的AI移动
                    if (Math.random() < 0.02) {
                        const directions = ['up', 'down', 'left', 'right'];
                        enemy.direction = directions[Math.floor(Math.random() * directions.length)];
                    }
                    
                    // 移动敌人
                    if (enemy.direction === 'up') {
                        enemy.y -= enemy.speed;
                    } else if (enemy.direction === 'down') {
                        enemy.y += enemy.speed;
                    } else if (enemy.direction === 'left') {
                        enemy.x -= enemy.speed;
                    } else if (enemy.direction === 'right') {
                        enemy.x += enemy.speed;
                    }
                    
                    // 敌人射击
                    if (currentTime - enemy.lastShot > config.enemyBulletInterval) {
                        fireBullet(enemy);
                        enemy.lastShot = currentTime;
                    }
                    
                    // 移除超出屏幕的敌人
                    if (enemy.x < -50 || enemy.x > config.canvasWidth + 50 ||
                        enemy.y < -50 || enemy.y > config.canvasHeight + 50) {
                        entities.enemies.splice(i, 1);
                    }
                }
                
                // 更新子弹
                for (let i = entities.bullets.length - 1; i >= 0; i--) {
                    const bullet = entities.bullets[i];
                    
                    // 移动子弹
                    if (bullet.direction === 'up') {
                        bullet.y -= bullet.speed;
                    } else if (bullet.direction === 'down') {
                        bullet.y += bullet.speed;
                    } else if (bullet.direction === 'left') {
                        bullet.x -= bullet.speed;
                    } else if (bullet.direction === 'right') {
                        bullet.x += bullet.speed;
                    }
                    
                    // 移除超出屏幕的子弹
                    if (bullet.x < 0 || bullet.x > config.canvasWidth ||
                        bullet.y < 0 || bullet.y > config.canvasHeight) {
                        entities.bullets.splice(i, 1);
                        continue;
                    }
                    
                    // 子弹与墙壁碰撞检测
                    let hitWall = false;
                    for (const wall of entities.walls) {
                        if (checkCircleRectCollision(bullet, wall)) {
                            entities.bullets.splice(i, 1);
                            hitWall = true;
                            break;
                        }
                    }
                    if (hitWall) continue;
                    
                    // 玩家子弹与敌人碰撞检测
                    if (bullet.isPlayerBullet && entities.player) {
                        for (let j = entities.enemies.length - 1; j >= 0; j--) {
                            const enemy = entities.enemies[j];
                            if (checkCircleRectCollision(bullet, enemy)) {
                                entities.enemies.splice(j, 1);
                                entities.bullets.splice(i, 1);
                                state.score += 100;
                                updateUI();
                                break;
                            }
                        }
                    }
                    
                    // 敌人子弹与玩家碰撞检测
                    if (!bullet.isPlayerBullet && entities.player) {
                        if (checkCircleRectCollision(bullet, entities.player)) {
                            entities.bullets.splice(i, 1);
                            state.lives--;
                            updateUI();
                            
                            if (state.lives <= 0) {
                                gameOver();
                            }
                        }
                    }
                }
                
                // 生成新敌人
                if (Math.random() < 0.01) {
                    entities.enemies.push(createEnemy());
                }
            }

            // 游戏循环
            function gameLoop(currentTime) {
                update(currentTime);
                renderer.render();
