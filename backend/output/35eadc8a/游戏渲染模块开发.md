我将为您创建一个完整的贪吃蛇游戏，使用Canvas进行渲染。这是一个单文件HTML应用，包含了所有必要的CSS和JavaScript代码。

```html
<!DOCTYPE html>
<html lang="zh-CN">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>贪吃蛇游戏</title>
    <style>
        body {
            margin: 0;
            padding: 0;
            display: flex;
            flex-direction: column;
            justify-content: center;
            align-items: center;
            background-color: #f0f0f0;
            font-family: Arial, sans-serif;
        }
        
        h1 {
            color: #333;
            margin-bottom: 20px;
        }
        
        #gameCanvas {
            border: 2px solid #333;
            background-color: #fff;
            box-shadow: 0 0 10px rgba(0, 0, 0, 0.1);
        }
        
        #score {
            margin-top: 20px;
            font-size: 24px;
            color: #333;
        }
        
        #gameOver {
            position: absolute;
            top: 50%;
            left: 50%;
            transform: translate(-50%, -50%);
            background-color: rgba(0, 0, 0, 0.8);
            color: white;
            padding: 20px;
            border-radius: 10px;
            text-align: center;
            display: none;
        }
        
        #restartButton {
            margin-top: 15px;
            padding: 10px 20px;
            background-color: #4CAF50;
            color: white;
            border: none;
            border-radius: 5px;
            cursor: pointer;
            font-size: 16px;
        }
        
        #restartButton:hover {
            background-color: #45a049;
        }
    </style>
</head>
<body>
    <h1>贪吃蛇游戏</h1>
    <canvas id="gameCanvas" width="400" height="400"></canvas>
    <div id="score">得分: 0</div>
    <div id="gameOver">
        <h2>游戏结束!</h2>
        <p id="finalScore"></p>
        <button id="restartButton">重新开始</button>
    </div>

    <script>
        // 游戏配置
        const GRID_SIZE = 20;
        const CELL_SIZE = 20;
        const INITIAL_SPEED = 150;
        
        // 获取DOM元素
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        const scoreElement = document.getElementById('score');
        const gameOverElement = document.getElementById('gameOver');
        const finalScoreElement = document.getElementById('finalScore');
        const restartButton = document.getElementById('restartButton');
        
        // 游戏状态
        let snake = [];
        let direction = 'right';
        let food = {};
        let score = 0;
        let gameRunning = false;
        let gameSpeed = INITIAL_SPEED;
        let lastRenderTime = 0;
        let gameLoopId = null;
        
        // 初始化游戏
        function initGame() {
            // 初始化蛇
            snake = [
                {x: 6, y: 10},
                {x: 5, y: 10},
                {x: 4, y: 10}
            ];
            
            // 设置初始方向
            direction = 'right';
            
            // 生成第一个食物
            generateFood();
            
            // 重置分数和游戏状态
            score = 0;
            gameSpeed = INITIAL_SPEED;
            gameRunning = true;
            updateScore();
            
            // 隐藏游戏结束界面
            gameOverElement.style.display = 'none';
        }
        
        // 生成食物
        function generateFood() {
            // 确保食物不会生成在蛇身上
            do {
                food = {
                    x: Math.floor(Math.random() * GRID_SIZE),
                    y: Math.floor(Math.random() * GRID_SIZE)
                };
            } while (snake.some(segment => segment.x === food.x && segment.y === food.y));
        }
        
        // 绘制游戏元素
        function draw() {
            // 清空画布
            ctx.fillStyle = '#fff';
            ctx.fillRect(0, 0, canvas.width, canvas.height);
            
            // 绘制网格线（可选）
            ctx.strokeStyle = '#eee';
            ctx.lineWidth = 0.5;
            for (let i = 0; i <= GRID_SIZE; i++) {
                // 垂直线
                ctx.beginPath();
                ctx.moveTo(i * CELL_SIZE, 0);
                ctx.lineTo(i * CELL_SIZE, canvas.height);
                ctx.stroke();
                
                // 水平线
                ctx.beginPath();
                ctx.moveTo(0, i * CELL_SIZE);
                ctx.lineTo(canvas.width, i * CELL_SIZE);
                ctx.stroke();
            }
            
            // 绘制蛇
            snake.forEach((segment, index) => {
                // 蛇头使用不同颜色
                if (index === 0) {
                    ctx.fillStyle = '#2ecc71';
                } else {
                    ctx.fillStyle = '#27ae60';
                }
                
                ctx.fillRect(
                    segment.x * CELL_SIZE + 1,
                    segment.y * CELL_SIZE + 1,
                    CELL_SIZE - 2,
                    CELL_SIZE - 2
                );
                
                // 添加圆角效果
                ctx.beginPath();
                ctx.roundRect(
                    segment.x * CELL_SIZE + 1,
                    segment.y * CELL_SIZE + 1,
                    CELL_SIZE - 2,
                    CELL_SIZE - 2,
                    3
                );
                ctx.fill();
            });
            
            // 绘制食物
            ctx.fillStyle = '#e74c3c';
            ctx.beginPath();
            ctx.arc(
                food.x * CELL_SIZE + CELL_SIZE / 2,
                food.y * CELL_SIZE + CELL_SIZE / 2,
                CELL_SIZE / 2 - 2,
                0,
                Math.PI * 2
            );
            ctx.fill();
        }
        
        // 更新游戏状态
        function update() {
            if (!gameRunning) return;
            
            // 计算新的蛇头位置
            const head = {...snake[0]};
            
            switch (direction) {
                case 'up':
                    head.y -= 1;
                    break;
                case 'down':
                    head.y += 1;
                    break;
                case 'left':
                    head.x -= 1;
                    break;
                case 'right':
                    head.x += 1;
                    break;
            }
            
            // 检查是否碰到墙壁
            if (head.x < 0 || head.x >= GRID_SIZE || head.y < 0 || head.y >= GRID_SIZE) {
                gameOver();
                return;
            }
            
            // 检查是否碰到自己
            if (snake.some(segment => segment.x === head.x && segment.y === head.y)) {
                gameOver();
                return;
            }
            
            // 将新头部添加到蛇身
            snake.unshift(head);
            
            // 检查是否吃到食物
            if (head.x === food.x && head.y === food.y) {
                // 增加分数
                score += 10;
                updateScore();
                
                // 生成新食物
                generateFood();
                
                // 每得50分增加游戏速度
                if (score % 50 === 0 && gameSpeed > 50) {
                    gameSpeed -= 10;
                }
            } else {
                // 如果没有吃到食物，移除尾部
                snake.pop();
            }
        }
        
        // 更新分数显示
        function updateScore() {
            scoreElement.textContent = `得分: ${score}`;
        }
        
        // 游戏结束
        function gameOver() {
            gameRunning = false;
            finalScoreElement.textContent = `最终得分: ${score}`;
            gameOverElement.style.display = 'block';
            
            // 取消动画帧
            if (gameLoopId) {
                cancelAnimationFrame(gameLoopId);
            }
        }
        
        // 游戏主循环
        function gameLoop(currentTime) {
            if (!gameRunning) return;
            
            // 计算自上次渲染以来的时间
            const secondsSinceLastRender = (currentTime - lastRenderTime) / 1000;
            
            // 如果时间间隔大于游戏速度，则更新游戏
            if (secondsSinceLastRender > gameSpeed / 1000) {
                update();
                draw();
                lastRenderTime = currentTime;
            }
            
            // 继续游戏循环
            gameLoopId = requestAnimationFrame(gameLoop);
        }
        
        // 处理键盘输入
        function handleKeyPress(event) {
            if (!gameRunning) return;
            
            // 防止方向键滚动页面
            if ([37, 38, 39, 40].includes(event.keyCode)) {
                event.preventDefault();
            }
            
            // 根据按键改变方向
            switch (event.keyCode) {
                case 37: // 左箭头
                    if (direction !== 'right') direction = 'left';
                    break;
                case 38: // 上箭头
                    if (direction !== 'down') direction = 'up';
                    break;
                case 39: // 右箭头
                    if (direction !== 'left') direction = 'right';
                    break;
                case 40: // 下箭头
                    if (direction !== 'up') direction = 'down';
                    break;
            }
        }
        
        // 重新开始游戏
        function restartGame() {
            if (gameLoopId) {
                cancelAnimationFrame(gameLoopId);
            }
            initGame();
            draw();
            gameLoopId = requestAnimationFrame(gameLoop);
        }
        
        // 初始化游戏事件监听
        document.addEventListener('DOMContentLoaded', () => {
            // 键盘事件
            document.addEventListener('keydown', handleKeyPress);
            
            // 重新开始按钮
            restartButton.addEventListener('click', restartGame);
            
            // 初始化并开始游戏
            initGame();
            draw();
            gameLoopId = requestAnimationFrame(gameLoop);
        });
    </script>
</body>
</html>
```

这个贪吃蛇游戏实现了以下功能：

1. **Canvas渲染**：
   - 使用Canvas API绘制蛇和食物
   - 添加了网格线使游戏区域更清晰
   - 蛇头使用不同颜色，食物为圆形

2. **游戏逻辑**：
   - 蛇可以上下左右移动
   - 吃到食物后蛇身变长，分数增加
   - 碰到墙壁或自己身体时游戏结束
   - 每得50分游戏速度会增加

3. **用户界面**：
   - 显示当前得分
   - 游戏结束时显示最终得分和重新开始按钮
   - 简洁美观的界面设计

4. **技术实现**：
   - 使用requestAnimationFrame实现游戏循环
   - 键盘事件监听控制蛇的移动方向
   - 防止蛇直接掉头（例如向右移动时不能直接向左）
   - 游戏状态管理（运行中、结束等）

所有代码都在一个HTML文件中，无需外部依赖，可以直接在浏览器中打开运行。
