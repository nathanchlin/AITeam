我将为您实现一个完整的贪吃蛇游戏，包含游戏结束界面和重启功能。以下是完整的单文件HTML代码：

```html
<!DOCTYPE html>
<html lang="zh-CN">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>贪吃蛇游戏</title>
    <style>
        body {
            margin: 0;
            padding: 0;
            display: flex;
            justify-content: center;
            align-items: center;
            height: 100vh;
            background-color: #f0f0f0;
            font-family: Arial, sans-serif;
        }
        .game-container {
            text-align: center;
        }
        canvas {
            border: 2px solid #333;
            background-color: #fff;
            box-shadow: 0 0 10px rgba(0, 0, 0, 0.2);
        }
        .score {
            font-size: 24px;
            margin-bottom: 10px;
            color: #333;
        }
        .game-over {
            position: absolute;
            top: 50%;
            left: 50%;
            transform: translate(-50%, -50%);
            background-color: rgba(0, 0, 0, 0.8);
            color: white;
            padding: 30px;
            border-radius: 10px;
            text-align: center;
            display: none;
        }
        .game-over h2 {
            margin-top: 0;
            font-size: 32px;
        }
        .game-over p {
            font-size: 24px;
            margin: 10px 0;
        }
        .restart-btn {
            background-color: #4CAF50;
            color: white;
            border: none;
            padding: 12px 24px;
            font-size: 18px;
            border-radius: 5px;
            cursor: pointer;
            transition: background-color 0.3s;
        }
        .restart-btn:hover {
            background-color: #45a049;
        }
        .controls {
            margin-top: 20px;
            color: #666;
        }
    </style>
</head>
<body>
    <div class="game-container">
        <div class="score">分数: <span id="score">0</span></div>
        <canvas id="gameCanvas" width="400" height="400"></canvas>
        <div class="controls">使用方向键控制蛇的移动</div>
        
        <div id="gameOver" class="game-over">
            <h2>游戏结束!</h2>
            <p>最终得分: <span id="finalScore">0</span></p>
            <button id="restartBtn" class="restart-btn">重新开始</button>
        </div>
    </div>

    <script>
        document.addEventListener('DOMContentLoaded', function() {
            const canvas = document.getElementById('gameCanvas');
            const ctx = canvas.getContext('2d');
            const scoreElement = document.getElementById('score');
            const finalScoreElement = document.getElementById('finalScore');
            const gameOverElement = document.getElementById('gameOver');
            const restartBtn = document.getElementById('restartBtn');
            
            // 游戏配置
            const gridSize = 20;
            const tileCount = canvas.width / gridSize;
            
            let snake = [];
            let direction = { x: 0, y: 0 };
            let food = {};
            let score = 0;
            let gameRunning = false;
            
            // 初始化游戏
            function initGame() {
                // 初始化蛇的位置
                snake = [
                    { x: 10, y: 10 }
                ];
                
                // 初始方向
                direction = { x: 0, y: 0 };
                
                // 生成食物
                generateFood();
                
                // 重置分数
                score = 0;
                scoreElement.textContent = score;
                
                // 隐藏游戏结束界面
                gameOverElement.style.display = 'none';
                
                // 开始游戏
                gameRunning = true;
                gameLoop();
            }
            
            // 生成食物
            function generateFood() {
                food = {
                    x: Math.floor(Math.random() * tileCount),
                    y: Math.floor(Math.random() * tileCount)
                };
                
                // 确保食物不会出现在蛇身上
                for (let segment of snake) {
                    if (segment.x === food.x && segment.y === food.y) {
                        generateFood();
                        break;
                    }
                }
            }
            
            // 绘制游戏
            function draw() {
                // 清除画布
                ctx.fillStyle = '#fff';
                ctx.fillRect(0, 0, canvas.width, canvas.height);
                
                // 绘制蛇
                ctx.fillStyle = '#2ecc71';
                for (let segment of snake) {
                    ctx.fillRect(segment.x * gridSize, segment.y * gridSize, gridSize - 2, gridSize - 2);
                }
                
                // 绘制蛇头（不同颜色）
                ctx.fillStyle = '#27ae60';
                ctx.fillRect(snake[0].x * gridSize, snake[0].y * gridSize, gridSize - 2, gridSize - 2);
                
                // 绘制食物
                ctx.fillStyle = '#e74c3c';
                ctx.beginPath();
                ctx.arc(
                    food.x * gridSize + gridSize / 2,
                    food.y * gridSize + gridSize / 2,
                    gridSize / 2 - 1,
                    0,
                    Math.PI * 2
                );
                ctx.fill();
            }
            
            // 更新游戏状态
            function update() {
                if (!gameRunning) return;
                
                // 计算新的蛇头位置
                const head = { x: snake[0].x + direction.x, y: snake[0].y + direction.y };
                
                // 检查是否撞墙
                if (head.x < 0 || head.x >= tileCount || head.y < 0 || head.y >= tileCount) {
                    gameOver();
                    return;
                }
                
                // 检查是否撞到自己
                for (let segment of snake) {
                    if (head.x === segment.x && head.y === segment.y) {
                        gameOver();
                        return;
                    }
                }
                
                // 将新头部添加到蛇的前面
                snake.unshift(head);
                
                // 检查是否吃到食物
                if (head.x === food.x && head.y === food.y) {
                    // 增加分数
                    score += 10;
                    scoreElement.textContent = score;
                    
                    // 生成新食物
                    generateFood();
                } else {
                    // 如果没有吃到食物，移除尾部
                    snake.pop();
                }
            }
            
            // 游戏结束
            function gameOver() {
                gameRunning = false;
                finalScoreElement.textContent = score;
                gameOverElement.style.display = 'block';
            }
            
            // 游戏循环
            function gameLoop() {
                update();
                draw();
                
                if (gameRunning) {
                    setTimeout(gameLoop, 100);
                }
            }
            
            // 键盘控制
            document.addEventListener('keydown', function(e) {
                if (!gameRunning && e.key !== ' ') return;
                
                // 防止蛇直接掉头
                switch (e.key) {
                    case 'ArrowUp':
                        if (direction.y === 0) {
                            direction = { x: 0, y: -1 };
                        }
                        break;
                    case 'ArrowDown':
                        if (direction.y === 0) {
                            direction = { x: 0, y: 1 };
                        }
                        break;
                    case 'ArrowLeft':
                        if (direction.x === 0) {
                            direction = { x: -1, y: 0 };
                        }
                        break;
                    case 'ArrowRight':
                        if (direction.x === 0) {
                            direction = { x: 1, y: 0 };
                        }
                        break;
                    case ' ':
                        // 空格键重新开始游戏
                        if (!gameRunning) {
                            initGame();
                        }
                        break;
                }
            });
            
            // 重新开始按钮
            restartBtn.addEventListener('click', initGame);
            
            // 初始化游戏
            initGame();
        });
    </script>
</body>
</html>
```

这个贪吃蛇游戏实现了以下功能：

1. **游戏核心机制**：
   - 使用Canvas API渲染游戏
   - 蛇可以上下左右移动
   - 吃到食物后蛇身变长，分数增加
   - 撞墙或撞到自己时游戏结束

2. **游戏结束界面**：
   - 显示半透明的黑色遮罩层
   - 显示"游戏结束"标题
   - 显示最终得分
   - 提供"重新开始"按钮

3. **重启功能**：
   - 点击"重新开始"按钮可以重新开始游戏
   - 按空格键也可以重新开始游戏

4. **用户界面**：
   - 显示当前分数
   - 游戏区域有边框和阴影效果
   - 提供操作说明

5. **游戏控制**：
   - 使用方向键控制蛇的移动
   - 防止蛇直接掉头（例如向右移动时不能立即向左）

游戏使用了原生Canvas API，没有依赖任何外部游戏框架，所有代码都在一个HTML文件中，可以直接在浏览器中打开运行。
