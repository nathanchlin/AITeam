```html
<!DOCTYPE html>
<html lang="zh-CN">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>贪吃蛇游戏</title>
    <style>
        body {
            margin: 0;

            display: flex;
            flex-direction: column;
            align-items: center;
            justify-content: center;
            height: 100vh;
            background-color: #f0f0f0;
            font-family: Arial, sans-serif;
        }
        
        #game-container {
            position: relative;
            margin-bottom: 20px;
        }
        
        #game-canvas {
            border: 2px solid #333;
            background-color: #fff;
            box-shadow: 0 4px 8px rgba(0, 0, 0, 0.1);
        }
        
        #score-board {
            display: flex;
            justify-content: space-between;
            width: 400px;
            margin-bottom: 10px;
            font-size: 18px;
            font-weight: bold;
        }
        
        #game-over {
            position: absolute;
            top: 50%;
            left: 50%;
            transform: translate(-50%, -50%);
            background-color: rgba(0, 0, 0, 0.7);
            color: white;

        }
        
        #instructions {
            text-align: center;
            margin-top: 20px;
            color: #666;
        }
    </style>
</head>
<body>
    <div id="score-board">
        <div>当前分数: <span id="current-score">0</span></div>
        <div>最高分数: <span id="high-score">0</span></div>
    </div>
    
    <div id="game-container">
        <canvas id="game-canvas" width="400" height="400"></canvas>
        <div id="game-over">
            <h2>游戏结束!</h2>
            <p>您的分数: <span id="final-score">0</span></p>
            <button id="restart-btn">重新开始</button>
        </div>
    </div>
    
    <div id="instructions">
        <p>使用方向键控制蛇的移动</p>
        <p>吃到食物增加分数和长度</p>
    </div>

    <script>
        // 游戏配置
        const GRID_SIZE = 20;
        const CELL_SIZE = 20;
        const INITIAL_SPEED = 150;
        const SPEED_INCREMENT = 2;
        
        // 游戏状态
        let canvas, ctx;
        let snake = [];
        let direction = { x: 1, y: 0 };
        let food = {};
        let score = 0;
        let highScore = localStorage.getItem('snakeHighScore') || 0;
        let gameRunning = false;
        let gameSpeed = INITIAL_SPEED;
        let lastRenderTime = 0;
        
        // 初始化游戏
        function initGame() {
            canvas = document.getElementById('game-canvas');
            ctx = canvas.getContext('2d');
            
            // 更新最高分数显示
            document.getElementById('high-score').textContent = highScore;
            
            // 初始化蛇
            snake = [
                { x: 10, y: 10 },
                { x: 9, y: 10 },
                { x: 8, y: 10 }
            ];
            
            // 初始方向
            direction = { x: 1, y: 0 };
            
            // 生成食物
            generateFood();
            
            // 重置分数和速度
            score = 0;
            gameSpeed = INITIAL_SPEED;
            updateScore();
            
            // 设置游戏状态
            gameRunning = true;
            document.getElementById('game-over').style.display = 'none';
            
            // 开始游戏循环
            requestAnimationFrame(gameLoop);
        }
        
        // 游戏主循环
        function gameLoop(currentTime) {
            if (!gameRunning) return;
            
            // 控制游戏速度
            const secondsSinceLastRender = (currentTime - lastRenderTime) / 1000;
            if (secondsSinceLastRender < 1 / (gameSpeed / 100)) {
                requestAnimationFrame(gameLoop);
                return;
            }
            
            lastRenderTime = currentTime;
            
            // 更新游戏状态
            update();
            
            // 渲染游戏
            render();
            
            // 继续游戏循环
            requestAnimationFrame(gameLoop);
        }
        
        // 更新游戏状态
        function update() {
            // 计算蛇的新头部位置
            const head = { ...snake[0] };
            head.x += direction.x;
            head.y += direction.y;
            
            // 检查是否撞墙
            if (head.x < 0 || head.x >= GRID_SIZE || head.y < 0 || head.y >= GRID_SIZE) {
                gameOver();
                return;
            }
            
            // 检查是否撞到自己
            for (let segment of snake) {
                if (head.x === segment.x && head.y === segment.y) {
                    gameOver();
                    return;
                }
            }
            
            // 将新头部添加到蛇的前面
            snake.unshift(head);
            
            // 检查是否吃到食物
            if (head.x === food.x && head.y === food.y) {
                // 增加分数
                score += 10;
                updateScore();
                
                // 增加游戏速度
                gameSpeed += SPEED_INCREMENT;
                
                // 生成新食物
                generateFood();
            } else {
                // 如果没有吃到食物，移除蛇的尾部
                snake.pop();
            }
        }
        
        // 渲染游戏
        function render() {
            // 清空画布
            ctx.fillStyle = '#fff';
            ctx.fillRect(0, 0, canvas.width, canvas.height);
            
            // 绘制蛇
            ctx.fillStyle = '#4CAF50';
            for (let i = 0; i < snake.length; i++) {
                const segment = snake[i];
                ctx.fillRect(
                    segment.x * CELL_SIZE,
                    segment.y * CELL_SIZE,
                    CELL_SIZE,
                    CELL_SIZE
                );
                
                // 绘制蛇头的不同颜色
                if (i === 0) {
                    ctx.fillStyle = '#2E7D32';
                    ctx.fillRect(
                        segment.x * CELL_SIZE + 2,
                        segment.y * CELL_SIZE + 2,
                        CELL_SIZE - 4,
                        CELL_SIZE - 4
                    );
                    ctx.fillStyle = '#4CAF50';
                }
            }
            
            // 绘制食物
            ctx.fillStyle = '#F44336';
            ctx.beginPath();
            ctx.arc(
                food.x * CELL_SIZE + CELL_SIZE / 2,
                food.y * CELL_SIZE + CELL_SIZE / 2,
                CELL_SIZE / 2 - 1,
                0,
                Math.PI * 2
            );
            ctx.fill();
        }
        
        // 生成食物
        function generateFood() {
            let newFood;
            do {
                newFood = {
                    x: Math.floor(Math.random() * GRID_SIZE),
                    y: Math.floor(Math.random() * GRID_SIZE)
                };
            } while (snake.some(segment => segment.x === newFood.x && segment.y === newFood.y));
            
            food = newFood;
        }
        
        // 更新分数显示
        function updateScore() {
            document.getElementById('current-score').textContent = score;
            
            // 更新最高分数
            if (score > highScore) {
                highScore = score;
                document.getElementById('high-score').textContent = highScore;
                localStorage.setItem('snakeHighScore', highScore);
            }
        }
        
        // 游戏结束
        function gameOver() {
            gameRunning = false;
            document.getElementById('final-score').textContent = score;
            document.getElementById('game-over').style.display = 'block';
        }
        
        // 键盘事件处理
        function handleKeyDown(e) {
            if (!gameRunning) return;
            
            switch (e.key) {
                case 'ArrowUp':
                    if (direction.y === 0) {
                        direction = { x: 0, y: -1 };
                    }
                    break;
                case 'ArrowDown':
                    if (direction.y === 0) {
                        direction = { x: 0, y: 1 };
                    }
                    break;
                case 'ArrowLeft':
                    if (direction.x === 0) {
                        direction = { x: -1, y: 0 };
                    }
                    break;
                case 'ArrowRight':
                    if (direction.x === 0) {
                        direction = { x: 1, y: 0 };
                    }
                    break;
            }
        }
        
        // 重新开始按钮事件
        function handleRestart() {
            initGame();
        }
        
        // 初始化事件监听
        document.addEventListener('DOMContentLoaded', function() {
            document.addEventListener('keydown', handleKeyDown);
            document.getElementById('restart-btn').addEventListener('click', handleRestart);
            
            // 初始化游戏
            initGame();
        });
    </script>
</body>
</html>
```

这个贪吃蛇游戏实现了以下功能：

1. **分数系统**：
   - 显示当前分数和最高分数
   - 吃到食物增加10分
   - 使用localStorage保存最高分数

2. **游戏功能**：
   - 蛇可以上下左右移动
   - 吃到食物后蛇身变长
   - 撞墙或撞到自己游戏结束
   - 游戏速度随分数增加而加快

3. **UI设计**：
   - 分数显示面板
   - 游戏结束提示
   - 重新开始按钮
   - 游戏说明

4. **技术实现**：
   - 使用Canvas API进行渲染
   - requestAnimationFrame实现游戏循环
   - 键盘事件控制蛇的移动
   - 碰撞检测逻辑

游戏操作简单直观，使用方向键控制蛇的移动，吃到红色食物增加分数和长度。游戏会随着分数增加而逐渐加快速度，增加挑战性。
