    <script>
        // 获取画布和上下文
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        
        // 游戏状态
        let gameRunning = false;
        let gamePaused = false;
        let score = 0;
        let lives = 3;
        
        // 球的属性
        const ball = {
            x: canvas.width / 2,
            y: canvas.height - 30,
            radius: 10,
            dx: 4,
            dy: -4,
            speed: 4,
            color: '#0095DD'
        };
        
        // 挡板属性
        const paddle = {
            width: 100,
            height: 15,
            x: (canvas.width - 100) / 2,
            y: canvas.height - 20,
            speed: 8,
            color: '#0095DD',
            dx: 0
        };
        
        // 砖块属性
        const brickRowCount = 5;
        const brickColumnCount = 9;
        const brickWidth = 75;
        const brickHeight = 20;
        const brickPadding = 10;
        const brickOffsetTop = 30;
        const brickOffsetLeft = 30;
        const bricks = [];
        
        // 键盘状态
        const rightPressed = false;
        const leftPressed = false;
        
        // 初始化砖块
        function initBricks() {
            bricks.length = 0;
            for (let c = 0; c < brickColumnCount; c++) {
                bricks[c] = [];
                for (let r = 0; r < brickRowCount; r++) {
                    const brickX = c * (brickWidth + brickPadding) + brickOffsetLeft;
                    const brickY = r * (brickHeight + brickPadding) + brickOffsetTop;
                    
                    // 不同行的砖块不同颜色和分数
                    const colors = ['#FF5252', '#FF4081', '#E040FB', '#7C4DFF', '#536DFE'];
                    const points = (brickRowCount - r) * 10;
                    
                    bricks[c][r] = {
                        x: brickX,
                        y: brickY,
                        status: 1,
                        color: colors[r],
                        points: points
                    };
                }
            }
        }
        
        // 绘制球
        function drawBall() {
            ctx.beginPath();
            ctx.arc(ball.x, ball.y, ball.radius, 0, Math.PI * 2);
            ctx.fillStyle = ball.color;
            ctx.fill();
            ctx.closePath();
        }
        
        // 绘制挡板
        function drawPaddle() {
            ctx.beginPath();
            ctx.rect(paddle.x, paddle.y, paddle.width, paddle.height);
            ctx.fillStyle = paddle.color;
            ctx.fill();
            ctx.closePath();
        }
        
        // 绘制砖块
        function drawBricks() {
            for (let c = 0; c < brickColumnCount; c++) {
                for (let r = 0; r < brickRowCount; r++) {
                    if (bricks[c][r].status === 1) {
                        const brick = bricks[c][r];
                        ctx.beginPath();
                        ctx.rect(brick.x, brick.y, brickWidth, brickHeight);
                        ctx.fillStyle = brick.color;
                        ctx.fill();
                        ctx.strokeStyle = '#333';
                        ctx.stroke();
                        ctx.closePath();
                    }
                }
            }
        }
        
        // 碰撞检测
        function collisionDetection() {
            // 球与砖块的碰撞检测
            for (let c = 0; c < brickColumnCount; c++) {
                for (let r = 0; r < brickRowCount; r++) {
                    const brick = bricks[c][r];
                    if (brick.status === 1) {
                        if (ball.x > brick.x && 
                            ball.x < brick.x + brickWidth && 
                            ball.y > brick.y && 
                            ball.y < brick.y + brickHeight) {
                            
                            ball.dy = -ball.dy;
                            brick.status = 0;
                            score += brick.points;
                            updateScore();
                            
                            // 检查是否所有砖块都被消除
                            if (score === getMaxScore()) {
                                gameWin();
                            }
                        }
                    }
                }
            }
            
            // 球与挡板的碰撞检测
            if (ball.x > paddle.x && 
                ball.x < paddle.x + paddle.width && 
                ball.y + ball.radius > paddle.y) {
                
                // 根据球击中挡板的位置改变反弹角度
                const hitPosition = (ball.x - (paddle.x + paddle.width / 2)) / (paddle.width / 2);
                ball.dx = hitPosition * ball.speed;
                ball.dy = -Math.abs(ball.dy);
            }
        }
        
        // 球边界检测
        function ballBoundaryCheck() {
            // 左右边界
            if (ball.x + ball.dx > canvas.width - ball.radius || ball.x + ball.dx < ball.radius) {
                ball.dx = -ball.dx;
            }
            
            // 上边界
            if (ball.y + ball.dy < ball.radius) {
                ball.dy = -ball.dy;
            } 
            // 下边界（球掉落）
            else if (ball.y + ball.dy > canvas.height - ball.radius) {
                lives--;
                updateLives();
                
                if (lives === 0) {
                    gameOver();
                } else {
                    resetBall();
                }
            }
        }
        
        // 挡板边界检测
        function paddleBoundaryCheck() {
            if (paddle.x < 0) {
                paddle.x = 0;
            } else if (paddle.x + paddle.width > canvas.width) {
                paddle.x = canvas.width - paddle.width;
            }
        }
        
        // 更新球的位置
        function updateBall() {
            ball.x += ball.dx;
            ball.y += ball.dy;
        }
        
        // 更新挡板位置
        function updatePaddle() {
            paddle.x += paddle.dx;
            paddleBoundaryCheck();
        }
        
        // 重置球的位置
        function resetBall() {
            ball.x = canvas.width / 2;
            ball.y = canvas.height - 30;
            ball.dx = 4 * (Math.random() > 0.5 ? 1 : -1);
            ball.dy = -4;
        }
        
        // 游戏主循环
        function gameLoop() {
            if (!gameRunning || gamePaused) return;
            
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            drawBricks();
            drawBall();
            drawPaddle();
            
            collisionDetection();
            ballBoundaryCheck();
            updateBall();
            updatePaddle();
            
            requestAnimationFrame(gameLoop);
        }
        
        // 开始游戏
        function startGame() {
            gameRunning = true;
            gamePaused = false;
            score = 0;
            lives = 3;
            updateScore();
            updateLives();
            initBricks();
            resetBall();
            paddle.x = (canvas.width - paddle.width) / 2;
            gameLoop();
        }
        
        // 暂停游戏
        function pauseGame() {
            gamePaused = !gamePaused;
            if (!gamePaused) {
                gameLoop();
            }
        }
        
        // 游戏结束
        function gameOver() {
            gameRunning = false;
            document.getElementById('gameOverText').textContent = '游戏结束';
            document.getElementById('finalScore').textContent = score;
            document.getElementById('gameOverModal').style.display = 'block';
        }
        
        // 游戏胜利
        function gameWin() {
            gameRunning = false;
            document.getElementById('gameOverText').textContent = '恭喜你赢了！';
            document.getElementById('finalScore').textContent = score;
            document.getElementById('gameOverModal').style.display = 'block';
        }
        
        // 重新开始游戏
        function restartGame() {
            document.getElementById('gameOverModal').style.display = 'none';
            startGame();
        }
        
        // 更新分数显示
        function updateScore() {
            document.getElementById('score').textContent = score;
        }
        
        // 更新生命显示
        function updateLives() {
            document.getElementById('lives').textContent = lives;
        }
        
        // 计算最大可能分数
        function getMaxScore() {
            let maxScore = 0;
            for (let c = 0; c < brickColumnCount; c++) {
                for (let r = 0; r < brickRowCount; r++) {
                    maxScore += (brickRowCount - r) * 10;
                }
            }
            return maxScore;
        }
        
        // 键盘事件处理
        document.addEventListener("keydown", keyDownHandler);
        document.addEventListener("keyup", keyUpHandler);
        
        function keyDownHandler(e) {
            if (e.key === "Right" || e.key === "ArrowRight" || e.key === "d" || e.key === "D") {
                paddle.dx = paddle.speed;
            } else if (e.key === "Left" || e.key === "ArrowLeft" || e.key === "a" || e.key === "A") {
                paddle.dx = -paddle.speed;
            }
        }
        
        function keyUpHandler(e) {
            if (e.key === "Right" || e.key === "ArrowRight" || e.key === "d" || e.key === "D") {
                paddle.dx = 0;
            } else if (e.key === "Left" || e.key === "ArrowLeft" || e.key === "a" || e.key === "A") {
                paddle.dx = 0;
            }
        }
        
        // 触摸事件处理（移动设备支持）
        let touchX = null;
        
        canvas.addEventListener("touchstart", function(e) {
            touchX = e.touches[0].clientX;
        });
        
        canvas.addEventListener("touchmove", function(e) {
            e.preventDefault();
            const touch = e.touches[0];
            const relativeX = touch.clientX - canvas.offsetLeft;
            
            if (relativeX > 0 && relativeX < canvas.width) {
                paddle.x = relativeX - paddle.width / 2;
                paddleBoundaryCheck();
            }
        });
        
        // 按钮事件处理
        document.getElementById("startBtn").addEventListener("click", startGame);
        document.getElementById("pauseBtn").addEventListener("click", pauseGame);
        document.getElementById("restartBtn").addEventListener("click", restartGame);
        document.getElementById("playAgainBtn").addEventListener("click", restartGame);
        
        // 初始化游戏
        initBricks();
        drawBricks();
        drawBall();
        drawPaddle();
    </script>
</body>

## 实现说明

### 1. 球体物理引擎

球体物理引擎实现了以下核心功能：

- **位置与速度**：球有位置(x, y)和速度(dx, dy)属性
- **边界反弹**：当球碰到画布边界时，相应的速度分量会反转
- **碰撞检测**：实现了球与挡板、球与砖块的碰撞检测
- **反弹角度**：根据球击中挡板的位置调整反弹角度，使游戏更有趣

### 2. 关键功能实现

- **砖块生成与渲染**：`initBricks()` 和 `drawBricks()` 函数
- **碰撞检测**：`collisionDetection()` 函数处理球与砖块、挡板的碰撞
- **游戏循环**：`gameLoop()` 函数使用 requestAnimationFrame 实现流畅的游戏循环
- **游戏状态管理**：实现了开始、暂停、重新开始等功能

### 3. 边界情况处理

- **球边界检测**：`ballBoundaryCheck()` 处理球超出画布边界的情况
- **挡板边界检测**：`paddleBoundaryCheck()` 防止挡板移出画布
- **游戏结束判定**：球掉落失去生命，生命为0时游戏结束

### 4. 额外功能

- **分数系统**：不同行的砖块有不同分值
- **生命系统**：玩家有3条生命
- **触摸支持**：添加了触摸事件处理，支持移动设备
- **游戏结束界面**：显示最终得分和重新开始选项
