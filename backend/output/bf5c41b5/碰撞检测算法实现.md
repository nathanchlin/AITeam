```html

    <script>
        // 获取画布和上下文
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        const scoreElement = document.getElementById('score');
        const livesElement = document.getElementById('lives');
        const gameMessageElement = document.getElementById('gameMessage');
        const restartBtn = document.getElementById('restartBtn');
        const pauseBtn = document.getElementById('pauseBtn');

        // 游戏变量
        let score = 0;
        let lives = 3;
        let isPaused = false;
        let isGameOver = false;
        let animationId;

        // 球的属性
        const ball = {
            x: canvas.width / 2,
            y: canvas.height - 30,
            radius: 10,
            dx: 5,
            dy: -5,
            speed: 5
        };

        // 挡板属性
        const paddle = {
            width: 100,
            height: 15,
            x: (canvas.width - 100) / 2,
            dx: 0,
            speed: 10
        };

        // 砖块属性
        const brickRowCount = 5;
        const brickColumnCount = 9;
        const brickWidth = 75;
        const brickHeight = 20;
        const brickPadding = 10;
        const brickOffsetTop = 60;
        const brickOffsetLeft = 30;
        const bricks = [];

        // 键盘控制变量
        let rightPressed = false;
        let leftPressed = false;

        // 触摸控制变量
        let touchStartX = 0;

        // 初始化砖块
        function initBricks() {
            for (let c = 0; c < brickColumnCount; c++) {
                bricks[c] = [];
                for (let r = 0; r < brickRowCount; r++) {
                    // 不同行的砖块不同颜色和分数
                    const colors = ['#FF5252', '#FF4081', '#E040FB', '#7C4DFF', '#536DFE'];
                    const points = (brickRowCount - r) * 10;
                    bricks[c][r] = { 
                        x: 0, 
                        y: 0, 
                        status: 1, 
                        color: colors[r],
                        points: points
                    };
                }
            }
        }

        // 绘制球
        function drawBall() {
            ctx.beginPath();
            ctx.arc(ball.x, ball.y, ball.radius, 0, Math.PI * 2);
            ctx.fillStyle = '#FFF';
            ctx.fill();
            ctx.closePath();
        }

        // 绘制挡板
        function drawPaddle() {
            ctx.beginPath();
            ctx.rect(paddle.x, canvas.height - paddle.height, paddle.width, paddle.height);
            ctx.fillStyle = '#0095DD';
            ctx.fill();
            ctx.closePath();
        }

        // 绘制砖块
        function drawBricks() {
            for (let c = 0; c < brickColumnCount; c++) {
                for (let r = 0; r < brickRowCount; r++) {
                    if (bricks[c][r].status === 1) {
                        const brickX = c * (brickWidth + brickPadding) + brickOffsetLeft;
                        const brickY = r * (brickHeight + brickPadding) + brickOffsetTop;
                        bricks[c][r].x = brickX;
                        bricks[c][r].y = brickY;
                        
                        ctx.beginPath();
                        ctx.rect(brickX, brickY, brickWidth, brickHeight);
                        ctx.fillStyle = bricks[c][r].color;
                        ctx.fill();
                        ctx.strokeStyle = '#FFF';
                        ctx.lineWidth = 1;
                        ctx.stroke();
                        ctx.closePath();
                    }
                }
            }
        }

        // 碰撞检测 - 球与砖块
        function collisionDetection() {
            for (let c = 0; c < brickColumnCount; c++) {
                for (let r = 0; r < brickRowCount; r++) {
                    const b = bricks[c][r];
                    if (b.status === 1) {
                        // 检测球是否与砖块碰撞
                        if (ball.x + ball.radius > b.x && 
                            ball.x - ball.radius < b.x + brickWidth && 
                            ball.y + ball.radius > b.y && 
                            ball.y - ball.radius < b.y + brickHeight) {
                            
                            // 计算球中心到砖块各边的距离
                            const ballLeft = ball.x - ball.radius;
                            const ballRight = ball.x + ball.radius;
                            const ballTop = ball.y - ball.radius;
                            const ballBottom = ball.y + ball.radius;
                            
                            const brickLeft = b.x;
                            const brickRight = b.x + brickWidth;
                            const brickTop = b.y;
                            const brickBottom = b.y + brickHeight;
                            
                            // 计算球中心到砖块各边的距离
                            const leftDist = ballRight - brickLeft;
                            const rightDist = brickRight - ballLeft;
                            const topDist = ballBottom - brickTop;
                            const bottomDist = brickBottom - ballTop;
                            
                            // 找出最小距离，确定碰撞方向
                            const minDist = Math.min(leftDist, rightDist, topDist, bottomDist);
                            
                            if (minDist === leftDist || minDist === rightDist) {
                                ball.dx = -ball.dx;
                            } else {
                                ball.dy = -ball.dy;
                            }
                            
                            b.status = 0;
                            score += b.points;
                            scoreElement.textContent = score;
                            
                            // 检查是否所有砖块都被消除
                            if (score === getTotalPoints()) {
                                gameWin();
                            }
                        }
                    }
                }
            }
        }

        // 计算总分数
        function getTotalPoints() {
            let total = 0;
            for (let c = 0; c < brickColumnCount; c++) {
                for (let r = 0; r < brickRowCount; r++) {
                    total += (brickRowCount - r) * 10;
                }
            }
            return total;
        }

        // 球边界检测
        function ballBoundaryCheck() {
            // 左右边界
            if (ball.x + ball.dx > canvas.width - ball.radius || ball.x + ball.dx < ball.radius) {
                ball.dx = -ball.dx;
            }
            
            // 上边界
            if (ball.y + ball.dy < ball.radius) {
                ball.dy = -ball.dy;
            }
            // 下边界 - 球掉落
            else if (ball.y + ball.dy > canvas.height - ball.radius) {
                // 检查是否碰到挡板
                if (ball.x > paddle.x && ball.x < paddle.x + paddle.width) {
                    // 根据球击中挡板的位置调整反弹角度
                    const hitPos = (ball.x - paddle.x) / paddle.width;
                    ball.dx = 8 * (hitPos - 0.5); // -4 到 4 之间的值
                    ball.dy = -ball.dy;
                } else {
                    // 球掉落
                    lives--;
                    livesElement.textContent = lives;
                    
                    if (lives === 0) {
                        gameOver();
                    } else {
                        resetBall();
                        showMessage("失去一条生命! 剩余 " + lives + " 条");
                    }
                }
            }
        }

        // 挡板边界检测
        function paddleBoundaryCheck() {
            if (paddle.x < 0) {
                paddle.x = 0;
            } else if (paddle.x + paddle.width > canvas.width) {
                paddle.x = canvas.width - paddle.width;
            }
        }

        // 重置球的位置
        function resetBall() {
            ball.x = canvas.width / 2;
            ball.y = canvas.height - 30;
            ball.dx = 5 * (Math.random() > 0.5 ? 1 : -1);
            ball.dy = -5;
        }

        // 游戏主循环
        function gameLoop() {
            if (isPaused || isGameOver) {
                return;
            }
            
            // 清除画布
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            // 绘制游戏元素
            drawBricks();
            drawBall();
            drawPaddle();
            
            // 碰撞检测
            collisionDetection();
            ballBoundaryCheck();
            paddleBoundaryCheck();
            
            // 移动球
            ball.x += ball.dx;
            ball.y += ball.dy;
            
            // 移动挡板
            paddle.x += paddle.dx;
            
            // 继续游戏循环
            animationId = requestAnimationFrame(gameLoop);
        }

        // 键盘事件处理
        function keyDownHandler(e) {
            if (e.key === "Right" || e.key === "ArrowRight") {
                rightPressed = true;
            } else if (e.key === "Left" || e.key === "ArrowLeft") {
                leftPressed = true;
            }
        }

        function keyUpHandler(e) {
            if (e.key === "Right" || e.key === "ArrowRight") {
                rightPressed = false;
            } else if (e.key === "Left" || e.key === "ArrowLeft") {
                leftPressed = false;
            }
        }

        // 触摸事件处理
        function touchStartHandler(e) {
            touchStartX = e.touches[0].clientX;
        }

        function touchMoveHandler(e) {
            e.preventDefault();
            const touchX = e.touches[0].clientX;
            const diffX = touchX - touchStartX;
            
            paddle.x += diffX;
            paddleBoundaryCheck();
            
            touchStartX = touchX;
        }

        // 更新挡板移动
        function updatePaddleMovement() {
            if (rightPressed) {
                paddle.dx = paddle.speed;
            } else if (leftPressed) {
                paddle.dx = -paddle.speed;
            } else {
                paddle.dx = 0;
            }
        }

        // 游戏结束
        function gameOver() {
            isGameOver = true;
            cancelAnimationFrame(animationId);
            showMessage("游戏结束! 最终得分: " + score);
        }

        // 游戏胜利
        function gameWin() {
            isGameOver = true;
            cancelAnimationFrame(animationId);
            showMessage("恭喜你赢了! 最终得分: " + score);
        }

        // 显示消息
        function showMessage(message) {
            gameMessageElement.textContent = message;
            setTimeout(() => {
                gameMessageElement.textContent = "";
            }, 3000);
        }

        // 重新开始游戏
        function restartGame() {
            // 重置游戏状态
            score = 0;
            lives = 3;
            isPaused = false;
            isGameOver = false;
            
            // 更新UI
            scoreElement.textContent = score;
            livesElement.textContent = lives;
            gameMessageElement.textContent = "";
            pauseBtn.textContent = "暂停";
            
            // 重置球和挡板
            resetBall();
            paddle.x = (canvas.width - paddle.width) / 2;
            
            // 重新初始化砖块
            initBricks();
            
            // 重新开始游戏循环
            if (animationId) {
                cancelAnimationFrame(animationId);
            }
            gameLoop();
        }

        // 暂停/继续游戏
        function togglePause() {
            if (isGameOver) return;
            
            isPaused = !isPaused;
            pauseBtn.textContent = isPaused ? "继续" : "暂停";
            
            if (!isPaused) {
                gameLoop();
            }
        }

        // 事件监听器
        document.addEventListener("keydown", keyDownHandler);
        document.addEventListener("keyup", keyUpHandler);
        canvas.addEventListener("touchstart", touchStartHandler);
        canvas.addEventListener("touchmove", touchMoveHandler);
        restartBtn.addEventListener("click", restartGame);
        pauseBtn.addEventListener("click", togglePause);

        // 持续更新挡板移动
        setInterval(updatePaddleMovement, 16);

        // 初始化游戏
        initBricks();
        gameLoop();
    </script>
</body>
```

## 碰撞检测算法详解

### 1. 球与砖块碰撞检测

我实现了一种精确的碰撞检测算法，可以确定球是从哪个方向撞击砖块的：

```javascript
function collisionDetection() {
    for (let c = 0; c < brickColumnCount; c++) {
        for (let r = 0; r < brickRowCount; r++) {
            const b = bricks[c][r];
            if (b.status === 1) {
                // 检测球是否与砖块碰撞
                if (ball.x + ball.radius > b.x && 
                    ball.x - ball.radius < b.x + brickWidth && 
                    ball.y + ball.radius > b.y && 
                    ball.y - ball.radius < b.y + brickHeight) {
                    
                    // 计算球中心到砖块各边的距离
                    const ballLeft = ball.x - ball.radius;
                    const ballRight = ball.x + ball.radius;
                    const ballTop = ball.y - ball.radius;
                    const ballBottom = ball.y + ball.radius;
                    
                    const brickLeft = b.x;
                    const brickRight = b.x + brickWidth;
                    const brickTop = b.y;
                    const brickBottom = b.y + brickHeight;
                    
                    // 计算球中心到砖块各边的距离
                    const leftDist = ballRight - brickLeft;
                    const rightDist = brickRight - ballLeft;
                    const topDist = ballBottom - brickTop;
                    const bottomDist = brickBottom - ballTop;
                    
                    // 找出最小距离，确定碰撞方向
                    const minDist = Math.min(leftDist, rightDist, topDist, bottomDist);
                    
                    if (minDist === leftDist || minDist === rightDist) {
                        ball.dx = -ball.dx;
                    } else {
                        ball.dy = -ball.dy;
                    }
                    
                    b.status = 0;
                    score += b.points;
                    scoreElement.textContent = score;
                    
                    // 检查是否所有砖块都被消除
                    if (score === getTotalPoints()) {
                        gameWin();
                    }
                }
            }
        }
    }
}
```

**算法特点**：
- 使用AABB（轴对齐边界框）检测球与砖块的碰撞
- 计算球到砖块各边的距离，确定碰撞方向
- 根据碰撞方向正确反弹球
- 不同行的砖块有不同颜色和分数
