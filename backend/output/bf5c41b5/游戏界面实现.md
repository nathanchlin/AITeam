```html
    <script>
        // 获取画布和上下文
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        const scoreElement = document.getElementById('score');
        const livesElement = document.getElementById('lives');
        const gameMessageElement = document.getElementById('gameMessage');
        const startBtn = document.getElementById('startBtn');
        const pauseBtn = document.getElementById('pauseBtn');
        const restartBtn = document.getElementById('restartBtn');
        
        // 游戏变量
        let score = 0;
        let lives = 3;
        let gameRunning = false;
        let gamePaused = false;
        let animationId = null;
        
        // 球的属性
        const ball = {
            x: canvas.width / 2,
            y: canvas.height - 30,
            radius: 8,
            dx: 4,
            dy: -4,
            speed: 4
        };
        
        // 挡板属性
        const paddle = {
            width: 75,
            height: 10,
            x: (canvas.width - 75) / 2,
            dx: 0,
            speed: 7
        };
        
        // 砖块属性
        const brickRowCount = 5;
        const brickColumnCount = 9;
        const brickWidth = 50;
        const brickHeight = 20;
        const brickPadding = 10;
        const brickOffsetTop = 30;
        const brickOffsetLeft = 30;
        const bricks = [];
        
        // 键盘状态
        const rightPressed = false;
        const leftPressed = false;
        
        // 初始化砖块
        function initBricks() {
            for (let c = 0; c < brickColumnCount; c++) {
                bricks[c] = [];
                for (let r = 0; r < brickRowCount; r++) {
                    const brickX = c * (brickWidth + brickPadding) + brickOffsetLeft;
                    const brickY = r * (brickHeight + brickPadding) + brickOffsetTop;
                    
                    // 不同行的砖块不同颜色
                    const colors = ['#e74c3c', '#e67e22', '#f1c40f', '#2ecc71', '#3498db'];
                    
                    bricks[c][r] = { 
                        x: brickX, 
                        y: brickY, 
                        status: 1,
                        color: colors[r]
                    };
                }
            }
        }
        
        // 绘制球
        function drawBall() {
            ctx.beginPath();
            ctx.arc(ball.x, ball.y, ball.radius, 0, Math.PI * 2);
            ctx.fillStyle = '#fff';
            ctx.fill();
            ctx.closePath();
        }
        
        // 绘制挡板
        function drawPaddle() {
            ctx.beginPath();
            ctx.rect(paddle.x, canvas.height - paddle.height, paddle.width, paddle.height);
            ctx.fillStyle = '#0095DD';
            ctx.fill();
            ctx.closePath();
        }
        
        // 绘制砖块
        function drawBricks() {
            for (let c = 0; c < brickColumnCount; c++) {
                for (let r = 0; r < brickRowCount; r++) {
                    if (bricks[c][r].status === 1) {
                        const brickX = bricks[c][r].x;
                        const brickY = bricks[c][r].y;
                        
                        ctx.beginPath();
                        ctx.rect(brickX, brickY, brickWidth, brickHeight);
                        ctx.fillStyle = bricks[c][r].color;
                        ctx.fill();
                        ctx.strokeStyle = '#000';
                        ctx.stroke();
                        ctx.closePath();
                    }
                }
            }
        }
        
        // 碰撞检测
        function collisionDetection() {
            for (let c = 0; c < brickColumnCount; c++) {
                for (let r = 0; r < brickRowCount; r++) {
                    const b = bricks[c][r];
                    if (b.status === 1) {
                        if (ball.x > b.x && ball.x < b.x + brickWidth && 
                            ball.y > b.y && ball.y < b.y + brickHeight) {
                            ball.dy = -ball.dy;
                            b.status = 0;
                            score += 10;
                            scoreElement.textContent = score;
                            
                            // 检查是否所有砖块都被消除
                            if (score === brickRowCount * brickColumnCount * 10) {
                                gameWin();
                            }
                        }
                    }
                }
            }
        }
        
        // 球边界检测
        function ballBoundaryCheck() {
            // 左右边界
            if (ball.x + ball.dx > canvas.width - ball.radius || ball.x + ball.dx < ball.radius) {
                ball.dx = -ball.dx;
            }
            
            // 上边界
            if (ball.y + ball.dy < ball.radius) {
                ball.dy = -ball.dy;
            } 
            // 下边界（球掉落）
            else if (ball.y + ball.dy > canvas.height - ball.radius) {
                // 检查是否碰到挡板
                if (ball.x > paddle.x && ball.x < paddle.x + paddle.width) {
                    // 根据球击中挡板的位置改变反弹角度
                    const hitPos = (ball.x - paddle.x) / paddle.width;
                    ball.dx = 8 * (hitPos - 0.5); // -4 到 4 之间的值
                    ball.dy = -ball.dy;
                } else {
                    // 球掉落
                    lives--;
                    livesElement.textContent = lives;
                    
                    if (lives === 0) {
                        gameOver();
                    } else {
                        resetBall();
                        gameMessageElement.textContent = `生命值减少！剩余 ${lives} 条生命`;
                        gameMessageElement.className = 'lose-message';
                    }
                }
            }
        }
        
        // 挡板边界检测
        function paddleBoundaryCheck() {
            if (paddle.x < 0) {
                paddle.x = 0;
            } else if (paddle.x + paddle.width > canvas.width) {
                paddle.x = canvas.width - paddle.width;
            }
        }
        
        // 重置球的位置
        function resetBall() {
            ball.x = canvas.width / 2;
            ball.y = canvas.height - 30;
            ball.dx = 4;
            ball.dy = -4;
            paddle.x = (canvas.width - paddle.width) / 2;
            gamePaused = true;
            
            setTimeout(() => {
                if (gameRunning) {
                    gamePaused = false;
                }
            }, 1000);
        }
        
        // 游戏主循环
        function gameLoop() {
            if (!gameRunning || gamePaused) return;
            
            // 清除画布
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            // 绘制游戏元素
            drawBricks();
            drawBall();
            drawPaddle();
            
            // 碰撞检测
            collisionDetection();
            
            // 边界检测
            ballBoundaryCheck();
            paddleBoundaryCheck();
            
            // 更新球的位置
            ball.x += ball.dx;
            ball.y += ball.dy;
            
            // 更新挡板位置
            paddle.x += paddle.dx;
            
            // 继续游戏循环
            animationId = requestAnimationFrame(gameLoop);
        }
        
        // 键盘按下事件
        function keyDownHandler(e) {
            if (e.key === "Right" || e.key === "ArrowRight") {
                paddle.dx = paddle.speed;
            } else if (e.key === "Left" || e.key === "ArrowLeft") {
                paddle.dx = -paddle.speed;
            }
        }
        
        // 键盘释放事件
        function keyUpHandler(e) {
            if (e.key === "Right" || e.key === "ArrowRight" || 
                e.key === "Left" || e.key === "ArrowLeft") {
                paddle.dx = 0;
            }
        }
        
        // 触摸事件处理
        let touchX = null;
        
        function touchStartHandler(e) {
            touchX = e.touches[0].clientX;
        }
        
        function touchMoveHandler(e) {
            if (touchX === null) return;
            
            const touch = e.touches[0];
            const deltaX = touch.clientX - touchX;
            
            paddle.x += deltaX;
            touchX = touch.clientX;
            
            // 防止挡板移出画布
            paddleBoundaryCheck();
        }
        
        function touchEndHandler() {
            touchX = null;
        }
        
        // 开始游戏
        function startGame() {
            if (!gameRunning) {
                gameRunning = true;
                gamePaused = false;
                gameMessageElement.textContent = '';
                gameMessageElement.className = '';
                gameLoop();
            }
        }
        
        // 暂停游戏
        function pauseGame() {
            if (gameRunning) {
                gamePaused = !gamePaused;
                if (gamePaused) {
                    gameMessageElement.textContent = '游戏已暂停';
                    gameMessageElement.className = 'pause-message';
                    cancelAnimationFrame(animationId);
                } else {
                    gameMessageElement.textContent = '';
                    gameMessageElement.className = '';
                    gameLoop();
                }
            }
        }
        
        // 重新开始游戏
        function restartGame() {
            // 取消之前的动画
            cancelAnimationFrame(animationId);
            
            // 重置游戏状态
            score = 0;
            lives = 3;
            gameRunning = false;
            gamePaused = false;
            
            // 更新UI
            scoreElement.textContent = score;
            livesElement.textContent = lives;
            gameMessageElement.textContent = '';
            gameMessageElement.className = '';
            
            // 重置球和挡板位置
            resetBall();
            
            // 重新初始化砖块
            initBricks();
            
            // 清除画布并重新绘制
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            drawBricks();
            drawBall();
            drawPaddle();
        }
        
        // 游戏胜利
        function gameWin() {
            gameRunning = false;
            gameMessageElement.textContent = '恭喜你赢了！';
            gameMessageElement.className = 'win-message';
            cancelAnimationFrame(animationId);
        }
        
        // 游戏结束
        function gameOver() {
            gameRunning = false;
            gameMessageElement.textContent = '游戏结束！';
            gameMessageElement.className = 'lose-message';
            cancelAnimationFrame(animationId);
        }
        
        // 事件监听器
        document.addEventListener("keydown", keyDownHandler);
        document.addEventListener("keyup", keyUpHandler);
        
        // 触摸事件
        canvas.addEventListener('touchstart', touchStartHandler);
        canvas.addEventListener('touchmove', touchMoveHandler);
        canvas.addEventListener('touchend', touchEndHandler);
        
        // 按钮事件
        startBtn.addEventListener("click", startGame);
        pauseBtn.addEventListener("click", pauseGame);
        restartBtn.addEventListener("click", restartGame);
        
        // 初始化游戏
        initBricks();
        drawBricks();
        drawBall();
        drawPaddle();
    </script>
</body>
</html>
```

## 实现说明

### 1. 游戏界面设计
- 使用了现代、简洁的UI设计，包含游戏画布、得分显示、生命值显示和控制按钮
- 响应式设计，适配不同屏幕尺寸
- 清晰的游戏状态提示（胜利、失败、暂停等）

### 2. 游戏功能实现
- **砖块生成与渲染**：`initBricks()` 和 `drawBricks()` 函数实现了彩色砖块的生成和渲染
- **碰撞检测**：`collisionDetection()` 实现了球与砖块的碰撞检测，消除砖块并增加得分
- **球与挡板碰撞**：球击中挡板时会根据击中位置改变反弹角度，增加游戏可玩性
- **游戏循环**：`gameLoop()` 实现了游戏的主循环，包括绘制、碰撞检测和边界检查
- **游戏状态管理**：实现了开始、暂停、重新开始等功能

### 3. 边界情况处理
- **球边界检测**：`ballBoundaryCheck()` 处理球超出画布边界的情况，包括左右上反弹和下边界掉落
- **挡板边界检测**：`paddleBoundaryCheck()` 防止挡板移出画布边界
- **游戏结束判定**：球掉落减少生命值，生命值为0时游戏结束；所有砖块消除时游戏胜利

### 4. 控制方式
- **键盘控制**：支持左右方向键控制挡板
- **触摸控制**：支持移动设备的触摸滑动控制挡板
- **按钮控制**：提供开始、暂停和重新开始按钮

### 5. 用户体验优化
- 游戏暂停时显示提示信息
- 球掉落后显示剩余生命值提示
- 不同行的砖块使用不同颜色，增加视觉区分度
- 球击中挡板不同位置会改变反弹角度，增加策略性

这个实现满足了所有原始需求，并修复了之前测试中发现的问题，提供了完整的游戏体验。
