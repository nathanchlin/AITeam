### JavaScript游戏逻辑

```javascript
document.addEventListener('DOMContentLoaded', () => {
    // 游戏配置
    const GAME_CONFIG = {
        initialSize: 4,
        minSize: 4,
        maxSize: 6,
        winValue: 2046,
        maxHistory: 10,
        animationDuration: 150,
        colors: {
            2: '#eee4da',
            4: '#ede0c8',
            8: '#f2b179',
            16: '#f59563',
            32: '#f67c5f',
            64: '#f65e3b',
            128: '#edcf72',
            256: '#edcc61',
            512: '#edc850',
            1024: '#edc53f',
            2048: '#edc22e',
            4096: '#3c3a32',
            8192: '#3c3a32'
        }
    };

    // 游戏状态
    class GameState {
        constructor(size) {
            this.size = size;
            this.grid = [];
            this.score = 0;
            this.bestScore = parseInt(localStorage.getItem('bestScore') || '0');
            this.gameOver = false;
            this.won = false;
            this.moveCount = 0;
            this.startTime = Date.now();
            this.history = [];
            this.gridContainer = document.getElementById('grid-container');
            this.tileContainer = document.getElementById('tile-container');
            this.scoreElement = document.getElementById('score');
            this.bestScoreElement = document.getElementById('best-score');
            this.gameMessageElement = document.getElementById('game-message');
            this.tryAgainButton = document.getElementById('try-again-btn');
            this.undoButton = document.getElementById('undo-btn');
            this.newGameButton = document.getElementById('new-game-btn');
            this.sizeSelector = document.getElementById('size-selector');
            this.gameTimeElement = document.getElementById('game-time');
            this.moveCountElement = document.getElementById('move-count');
            
            this.initializeGrid();
            this.setupEventListeners();
            this.updateDisplay();
        }

        initializeGrid() {
            // 设置CSS变量
            document.documentElement.style.setProperty('--grid-size', this.size);
            
            // 清空容器
            this.gridContainer.innerHTML = '';
            this.tileContainer.innerHTML = '';
            
            // 创建网格
            this.grid = Array(this.size).fill().map(() => Array(this.size).fill(0));
            
            // 创建网格单元格
            this.gridContainer.style.gridTemplateColumns = `repeat(${this.size}, 1fr)`;
            this.gridContainer.style.gridTemplateRows = `repeat(${this.size}, 1fr)`;
            
            for (let i = 0; i < this.size * this.size; i++) {
                const cell = document.createElement('div');
                cell.className = 'grid-cell';
                this.gridContainer.appendChild(cell);
            }
            
            // 添加初始方块
            this.addRandomTile();
            this.addRandomTile();
        }

        setupEventListeners() {
            // 键盘事件
            document.addEventListener('keydown', (e) => this.handleKeyDown(e));
            
            // 触摸事件
            let touchStartX = 0;
            let touchStartY = 0;
            
            this.tileContainer.addEventListener('touchstart', (e) => {
                touchStartX = e.touches[0].clientX;
                touchStartY = e.touches[0].clientY;
            }, { passive: true });
            
            this.tileContainer.addEventListener('touchend', (e) => {
                if (!touchStartX || !touchStartY) return;
                
                const touchEndX = e.changedTouches[0].clientX;
                const touchEndY = e.changedTouches[0].clientY;
                
                const dx = touchEndX - touchStartX;
                const dy = touchEndY - touchStartY;
                
                if (Math.abs(dx) > Math.abs(dy)) {
                    if (dx > 0) {
                        this.move('right');
                    } else {
                        this.move('left');
                    }
                } else {
                    if (dy > 0) {
                        this.move('down');
                    } else {
                        this.move('up');
                    }
                }
                
                touchStartX = 0;
                touchStartY = 0;
            }, { passive: true });
            
            // 按钮事件
            this.tryAgainButton.addEventListener('click', () => this.restart());
            this.undoButton.addEventListener('click', () => this.undo());
            this.newGameButton.addEventListener('click', () => this.restart());
            this.sizeSelector.addEventListener('change', (e) => {
                const newSize = parseInt(e.target.value);
                if (newSize !== this.size) {
                    this.size = newSize;
                    this.restart();
                }
            });
        }

        handleKeyDown(e) {
            if (this.gameOver) return;
            
            switch(e.key) {
                case 'ArrowUp':
                    e.preventDefault();
                    this.move('up');
                    break;
                case 'ArrowDown':
                    e.preventDefault();
                    this.move('down');
                    break;
                case 'ArrowLeft':
                    e.preventDefault();
                    this.move('left');
                    break;
                case 'ArrowRight':
                    e.preventDefault();
                    this.move('right');
                    break;
            }
        }

        move(direction) {
            // 保存当前状态到历史记录
            this.saveToHistory();
            
            const previousGrid = this.grid.map(row => [...row]);
            const previousScore = this.score;
            let moved = false;
            
            // 根据方向旋转网格，简化逻辑
            let rotatedGrid = this.rotateGrid(this.grid, direction);
            
            // 移动和合并方块
            for (let row = 0; row < this.size; row++) {
                const filteredRow = rotatedGrid[row].filter(val => val !== 0);
                const mergedRow = [];
                
                for (let i = 0; i < filteredRow.length; i++) {
                    if (i < filteredRow.length - 1 && filteredRow[i] === filteredRow[i + 1]) {
                        mergedRow.push(filteredRow[i] * 2);
                        this.score += filteredRow[i] * 2;
                        i++;
                    } else {
                        mergedRow.push(filteredRow[i]);
                    }
                }
                
                // 填充剩余空间
                while (mergedRow.length < this.size) {
                    mergedRow.push(0);
                }
                
                rotatedGrid[row] = mergedRow;
                
                // 检查是否有移动
                for (let col = 0; col < this.size; col++) {
                    if (previousGrid[row][col] !== rotatedGrid[row][col]) {
                        moved = true;
                        break;
                    }
                }
            }
            
            // 旋转回原始方向
            this.grid = this.rotateGrid(rotatedGrid, direction === 'left' ? 'right' : direction === 'right' ? 'left' : direction === 'up' ? 'down' : 'up');
            
            if (moved) {
                this.moveCount++;
                this.addRandomTile();
                this.updateDisplay();
                
                // 检查游戏是否结束
                if (this.isGameOver()) {
                    this.gameOver = true;
                    this.showGameOver();
                } else if (this.hasWon() && !this.won) {
                    this.won = true;
                    this.showWin();
                }
            } else {
                // 如果没有移动，恢复历史记录
                this.history.pop();
            }
        }

        rotateGrid(grid, direction) {
            let rotated = grid;
            
            // 根据方向旋转网格
            if (direction === 'right')
