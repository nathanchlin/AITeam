## JavaScript 模块

### app.js (主应用逻辑)

```javascript
document.addEventListener('DOMContentLoaded', () => {
    // 初始化游戏控制器
    const gameController = new GameController();
    
    // 绑定新游戏按钮
    document.getElementById('new-game-button').addEventListener('click', () => {
        gameController.restart();
    });
    
    document.querySelector('.retry-button').addEventListener('click', () => {
        gameController.restart();
    });
    
    // 键盘控制
    document.addEventListener('keydown', (event) => {
        switch (event.key) {
            case 'ArrowUp':
                event.preventDefault();
                gameController.move('up');
                break;
            case 'ArrowDown':
                event.preventDefault();
                gameController.move('down');
                break;
            case 'ArrowLeft':
                event.preventDefault();
                gameController.move('left');
                break;
            case 'ArrowRight':
                event.preventDefault();
                gameController.move('right');
                break;
        }
    });
    
    // 触摸控制
    let touchStartX = 0;
    let touchStartY = 0;
    
    document.addEventListener('touchstart', (event) => {
        if (event.touches.length > 0) {
            touchStartX = event.touches[0].clientX;
            touchStartY = event.touches[0].clientY;
        }
    });
    
    document.addEventListener('touchend', (event) => {
        if (!touchStartX || !touchStartY) {
            return;
        }
        
        let touchEndX = 0;
        let touchEndY = 0;
        
        if (event.changedTouches.length > 0) {
            touchEndX = event.changedTouches[0].clientX;
            touchEndY = event.changedTouches[0].clientY;
        }
        
        const dx = touchEndX - touchStartX;
        const dy = touchEndY - touchStartY;
        
        // 确定滑动方向
        if (Math.abs(dx) > Math.abs(dy)) {
            // 水平滑动
            if (dx > 0) {
                gameController.move('right');
            } else {
                gameController.move('left');
            }
        } else {
            // 垂直滑动
            if (dy > 0) {
                gameController.move('down');
            } else {
                gameController.move('up');
            }
        }
        
        touchStartX = 0;
        touchStartY = 0;
    });
});
```

### grid.js (网格管理)

```javascript
class Grid {
    constructor(size = 4) {
        this.size = size;
        this.cells = this.empty();
    }
    
    // 创建空网格
    empty() {
        const cells = [];
        for (let i = 0; i < this.size; i++) {
            cells[i] = [];
            for (let j = 0; j < this.size; j++) {
                cells[i][j] = null;
            }
        }
        return cells;
    }
    
    // 随机获取一个空单元格
    randomAvailableCell() {
        const cells = this.availableCells();
        if (cells.length > 0) {
            return cells[Math.floor(Math.random() * cells.length)];
        }
        return null;
    }
    
    // 获取所有空单元格
    availableCells() {
        const cells = [];
        for (let i = 0; i < this.size; i++) {
            for (let j = 0; j < this.size; j++) {
                if (!this.cells[i][j]) {
                    cells.push({ x: i, y: j });
                }
            }
        }
        return cells;
    }
    
    // 检查是否有空单元格
    hasAvailableCells() {
        return !!this.availableCells().length;
    }
    
    // 检查是否可以移动
    canMove(direction) {
        // 检查是否有空单元格
        if (this.hasAvailableCells()) {
            return true;
        }
        
        // 检查相邻单元格是否可以合并
        switch (direction) {
            case 'left':
            case 'right':
                return this.canMoveHorizontal();
            case 'up':
            case 'down':
                return this.canMoveVertical();
        }
        return false;
    }
    
    // 检查水平方向是否可以移动
    canMoveHorizontal() {
        for (let i = 0; i < this.size; i++) {
            for (let j = 0; j < this.size - 1; j++) {
                if (this.cells[i][j] && this.cells[i][j] === this.cells[i][j + 1]) {
                    return true;
                }
            }
        }
        return false;
    }
    
    // 检查垂直方向是否可以移动
    canMoveVertical() {
        for (let i = 0; i < this.size - 1; i++) {
            for (let j = 0; j < this.size; j++) {
                if (this.cells[i][j] && this.cells[i][j] === this.cells[i + 1][j]) {
                    return true;
                }
            }
        }
        return false;
    }
    
    // 检查整个网格是否可以移动
    canMoveAny() {
        return this.canMove('left') || this.canMove('right') || 
               this.canMove('up') || this.canMove('down');
    }
    
    // 在指定位置插入方块
    insertTile(tile) {
        this.cells[tile.x][tile.y] = tile;
    }
    
    // 移除指定位置的方块
    removeTile(tile) {
        this.cells[tile.x][tile.y] = null;
    }
    
    // 获取指定行
    getRow(row) {
        return this.cells[row];
    }
    
    // 获取指定
