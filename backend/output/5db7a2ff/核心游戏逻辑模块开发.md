### JavaScript游戏逻辑 (script.js)

```javascript
document.addEventListener('DOMContentLoaded', () => {
    // 游戏状态
    let gameState = {
        board: [],
        score: 0,
        bestScore: parseInt(localStorage.getItem('bestScore') || 0),
        moves: 0,
        gridSize: 4,
        gameOver: false,
        gameWon: false,
        previousState: null,
        startTime: null,
        elapsedTime: 0,
        timerInterval: null
    };

    // DOM元素
    const gameBoard = document.getElementById('game-board');
    const scoreElement = document.getElementById('score');
    const bestScoreElement = document.getElementById('best-score');
    const movesElement = document.getElementById('moves');
    const timerElement = document.getElementById('timer');
    const newGameBtn = document.getElementById('new-game-btn');
    const undoBtn = document.getElementById('undo-btn');
    const difficultySelect = document.getElementById('difficulty');
    const gameMessage = document.getElementById('game-message');

    // 初始化游戏
    function initGame() {
        // 重置游戏状态
        gameState.board = Array(gameState.gridSize).fill().map(() => Array(gameState.gridSize).fill(0));
        gameState.score = 0;
        gameState.moves = 0;
        gameState.gameOver = false;
        gameState.gameWon = false;
        gameState.previousState = null;
        gameState.startTime = Date.now();
        gameState.elapsedTime = 0;
        
        // 清除计时器
        if (gameState.timerInterval) {
            clearInterval(gameState.timerInterval);
        }
        
        // 启动计时器
        gameState.timerInterval = setInterval(updateTimer, 1000);
        
        // 更新UI
        updateScore();
        updateMoves();
        updateTimer();
        
        // 创建游戏板
        createGameBoard();
        
        // 添加初始方块
        addNewTile();
        addNewTile();
        
        // 隐藏游戏消息
        gameMessage.classList.remove('show');
        
        // 更新最高分显示
        bestScoreElement.textContent = gameState.bestScore;
    }

    // 创建游戏板
    function createGameBoard() {
        gameBoard.innerHTML = '';
        const gridContainer = document.createElement('div');
        gridContainer.className = 'grid-container';
        gridContainer.style.gridTemplateColumns = `repeat(${gameState.gridSize}, 1fr)`;
        gridContainer.style.gridTemplateRows = `repeat(${gameState.gridSize}, 1fr)`;
        
        for (let i = 0; i < gameState.gridSize * gameState.gridSize; i++) {
            const tile = document.createElement('div');
            tile.className = 'tile';
            tile.id = `tile-${i}`;
            gridContainer.appendChild(tile);
        }
        
        gameBoard.appendChild(gridContainer);
    }

    // 添加新方块
    function addNewTile() {
        const emptyCells = [];
        for (let row = 0; row < gameState.gridSize; row++) {
            for (let col = 0; col < gameState.gridSize; col++) {
                if (gameState.board[row][col] === 0) {
                    emptyCells.push({ row, col });
                }
            }
        }
        
        if (emptyCells.length > 0) {
            const randomCell = emptyCells[Math.floor(Math.random() * emptyCells.length)];
            const value = Math.random() < 0.9 ? 2 : 4;
            gameState.board[randomCell.row][randomCell.col] = value;
            
            // 添加动画效果
            setTimeout(() => {
                const tileElement = document.getElementById(`tile-${randomCell.row * gameState.gridSize + randomCell.col}`);
                tileElement.classList.add('tile-new', `tile-${value}`);
                tileElement.textContent = value;
            }, 10);
        }
    }

    // 更新游戏板显示
    function updateBoard() {
        for (let row = 0; row < gameState.gridSize; row++) {
            for (let col = 0; col < gameState.gridSize; col++) {
                const tileElement = document.getElementById(`tile-${row * gameState.gridSize + col}`);
                const value = gameState.board[row][col];
                
                // 清除所有tile类
                tileElement.className = 'tile';
                
                if (value > 0) {
                    tileElement.classList.add(`tile-${value}`);
                    tileElement.textContent = value;
                    
                    // 检查是否达到2048
                    if (value === 2048 && !gameState.gameWon) {
                        gameState.gameWon = true;
                        showMessage('恭喜！你达到了2048！');
                    }
                } else {
                    tileElement.textContent = '';
                }
            }
        }
    }

    // 保存当前状态
    function saveState() {
        gameState.previousState = {
            board: gameState.board.map(row => [...row]),
            score: gameState.score,
            moves: gameState.moves
        };
    }

    // 撤销操作
    function undo() {
        if (gameState.previousState && !gameState.gameOver) {
            gameState.board = gameState.previousState.board.map(row => [...row]);
            gameState.score = gameState.previousState.score;
            gameState.moves = gameState.previousState.moves;
            
            updateBoard();
            updateScore();
            updateMoves();
            
            gameState.previousState = null;
            undoBtn.disabled = true;
        }
    }

    // 移动方块
    function move(direction) {
        if (gameState.gameOver) return;
        
        saveState();
        let moved = false;
        const newBoard = gameState.board.map(row => [...row]);
        
        // 根据方向处理移动
        if (direction === 'left') {
            for (let row = 0; row < gameState.gridSize; row++) {
                const newRow = processLine(newBoard[row]);
                if (newRow.changed) moved = true;
                newBoard[row] = newRow.line;
            }
        } else if (direction === 'right') {
            for (let row = 0; row < gameState.gridSize; row++) {
                const reversedRow = [...newBoard[row]].reverse();
                const newRow = processLine(reversedRow);
                if (newRow.changed) moved = true;
                newBoard[row] = newRow.line.reverse();
            }
        } else if (direction === 'up') {
            for (let col = 0; col < gameState.gridSize; col++) {
                const column = [];
                for (let row = 0; row < gameState.gridSize; row++) {
                    column.push(newBoard[row][col]);
                }
                const newColumn = processLine(column);
                if (newColumn.changed) moved = true;
                for (let row = 0; row < gameState.gridSize; row++) {
                    newBoard[row][col] = newColumn.line[row];
                }
            }
        } else if (direction === 'down') {
            for (let col = 0; col < gameState.gridSize; col++) {
                const column = [];
                for (let row = 0; row < gameState.gridSize; row++) {
                    column.push(newBoard[row][col]);
                }
                const reversedColumn = [...column].reverse();
                const newColumn = processLine(reversedColumn);
                if (newColumn.changed) moved = true;
                const finalColumn = newColumn.line.reverse();
                for (let row = 0; row < gameState.gridSize; row++) {
                    newBoard[row][col] = finalColumn[row];
                }
            }
        }
        
        if (moved) {
            gameState.board = newBoard;
            gameState.moves++;
            updateMoves();
            
            // 添加新方块
            setTimeout(() => {
                addNewTile();
                updateBoard();
                
                // 检查游戏是否结束
                if (isGameOver()) {
                    gameState.gameOver = true;
                    clearInterval(gameState.timerInterval);
                    showMessage('游戏结束！');
                    
                    // 更新最高分
                    if (gameState.score > gameState.bestScore) {
                        gameState.bestScore = gameState.score;
                        localStorage.setItem('bestScore', gameState.bestScore);
                        bestScoreElement.textContent = gameState.bestScore;
                    }
                }
            }, 150);
