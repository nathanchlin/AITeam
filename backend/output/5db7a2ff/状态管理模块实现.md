## 3. JavaScript游戏逻辑

```javascript
class Game2046 {
    constructor() {
        this.size = 4; // 默认4x4游戏板
        this.board = [];
        this.score = 0;
        this.bestScore = parseInt(localStorage.getItem('bestScore') || '0');
        this.gameOver = false;
        this.won = false;
        this.previousState = null;
        this.startTime = null;
        this.gameTime = 0;
        this.timerInterval = null;
        
        // DOM元素
        this.boardElement = document.getElementById('game-board');
        this.scoreElement = document.getElementById('current-score');
        this.bestScoreElement = document.getElementById('best-score');
        this.gameTimeElement = document.getElementById('game-time');
        this.messageElement = document.getElementById('game-message');
        this.newGameBtn = document.getElementById('new-game-btn');
        this.undoBtn = document.getElementById('undo-btn');
        this.difficultyBtn = document.getElementById('difficulty-btn');
        
        // 初始化
        this.init();
    }
    
    init() {
        // 设置游戏板样式
        this.boardElement.style.gridTemplateColumns = `repeat(${this.size}, 1fr)`;
        this.boardElement.style.gridTemplateRows = `repeat(${this.size}, 1fr)`;
        
        // 更新最高分显示
        this.bestScoreElement.textContent = this.bestScore;
        
        // 添加事件监听
        this.newGameBtn.addEventListener('click', () => this.newGame());
        this.undoBtn.addEventListener('click', () => this.undo());
        this.difficultyBtn.addEventListener('click', () => this.toggleDifficulty());
        
        // 键盘事件
        document.addEventListener('keydown', (e) => this.handleKeyPress(e));
        
        // 触摸事件
        let touchStartX = 0;
        let touchStartY = 0;
        
        this.boardElement.addEventListener('touchstart', (e) => {
            touchStartX = e.touches[0].clientX;
            touchStartY = e.touches[0].clientY;
        });
        
        this.boardElement.addEventListener('touchend', (e) => {
            if (!touchStartX || !touchStartY) return;
            
            let touchEndX = e.changedTouches[0].clientX;
            let touchEndY = e.changedTouches[0].clientY;
            
            let dx = touchEndX - touchStartX;
            let dy = touchEndY - touchStartY;
            
            if (Math.abs(dx) > Math.abs(dy)) {
                if (dx > 0) {
                    this.move('right');
                } else {
                    this.move('left');
                }
            } else {
                if (dy > 0) {
                    this.move('down');
                } else {
                    this.move('up');
                }
            }
            
            touchStartX = 0;
            touchStartY = 0;
        });
        
        // 开始新游戏
        this.newGame();
    }
    
    newGame() {
        // 重置游戏状态
        this.board = Array(this.size).fill().map(() => Array(this.size).fill(0));
        this.score = 0;
        this.gameOver = false;
        this.won = false;
        this.previousState = null;
        this.startTime = Date.now();
        this.gameTime = 0;
        
        // 清除计时器
        if (this.timerInterval) {
            clearInterval(this.timerInterval);
        }
        
        // 启动计时器
        this.timerInterval = setInterval(() => {
            this.gameTime = Math.floor((Date.now() - this.startTime) / 1000);
            this.updateTimeDisplay();
        }, 1000);
        
        // 更新UI
        this.updateScore();
        this.updateTimeDisplay();
        this.messageElement.textContent = '';
        this.undoBtn.disabled = true;
        
        // 添加两个初始方块
        this.addNewTile();
        this.addNewTile();
        
        // 渲染游戏板
        this.render();
    }
    
    addNewTile() {
        // 找出所有空位置
        const emptyCells = [];
        for (let r = 0; r < this.size; r++) {
            for (let c = 0; c < this.size; c++) {
                if (this.board[r][c] === 0) {
                    emptyCells.push({r, c});
                }
            }
        }
        
        // 如果有空位置，随机添加一个新方块
        if (emptyCells.length > 0) {
            const randomCell = emptyCells[Math.floor(Math.random() * emptyCells.length)];
            this.board[randomCell.r][randomCell.c] = Math.random() < 0.9 ? 2 : 4;
            return true;
        }
        return false;
    }
    
    move(direction) {
        if (this.gameOver) return;
        
        // 保存当前状态，用于撤销功能
        this.saveState();
        
        let moved = false;
        const newBoard = this.board.map(row => [...row]);
        
        // 根据方向处理移动
        if (direction === 'left') {
            for (let r = 0; r < this.size; r++) {
                const row = this.slideAndMergeRow(newBoard[r]);
                if (row.moved) moved = true;
                newBoard[r] = row.row;
            }
        } else if (direction === 'right') {
            for (let r = 0; r < this.size; r++) {
                const row = this.slideAndMergeRow(newBoard[r].slice().reverse());
                if (row.moved) moved = true;
                newBoard[r] = row.row.reverse();
            }
        } else if (direction === 'up') {
            for (let c = 0; c < this.size; c++) {
                const column = [];
                for (let r = 0; r < this.size; r++) {
                    column.push(newBoard[r][c]);
                }
                const result = this.slideAndMergeRow(column);
                if (result.moved) moved = true;
                for (let r = 0; r < this.size; r++) {
                    newBoard[r][c] = result.row[r];
                }
            }
        } else if (direction === 'down') {
            for (let c = 0; c < this.size; c++) {
                const column = [];
                for (let r = 0; r < this.size; r++) {
                    column.push(newBoard[r][c]);
                }
                const result = this.slideAndMergeRow(column.slice().reverse());
                if (result.moved) moved = true;
                const reversedRow = result.row.reverse();
                for (let r = 0; r < this.size; r++) {
                    newBoard[r][c] = reversedRow[r];
                }
            }
        }
        
        // 如果有移动，添加新方块并检查游戏状态
        if (moved) {
            this.board = newBoard;
            this.addNewTile();
            this.render();
            this.updateScore();
            
            // 检查是否获胜
            if (!this.won && this.checkWin()) {
                this.won = true;
                this.messageElement.textContent = '恭喜你赢了！';
            }
            
            // 检查游戏是否结束
            if (this.checkGameOver()) {
                this.gameOver = true;
                clearInterval(this.timerInterval);
                this.messageElement.textContent = '游戏结束！';
            }
            
            // 启用撤销按钮
            this.undoBtn.disabled = false;
        }
    }
    
    slideAndMergeRow(row) {
        // 移除零
        let newRow = row.filter(val => val !== 0);
        let moved = false;
        
        // 检查是否有移动
        for (let i = 0; i < row.length; i++) {
            if (row[i] !== (newRow[i] || 0)) {
                moved = true;
                break;
            }
        }
        
        // 合并相同的数字
        for (let i = 0; i < newRow.length - 1; i++) {
            if (newRow[i] === newRow[i + 1]) {
                newRow[i] *= 2;
                this.score += newRow[i];
                newRow.splice(i + 1, 1);
            }
        }
        
        // 补充零
        while (newRow.length < this.size) {
            newRow.push(0);
        }
        
        return { row: newRow, moved };
    }
    
    checkWin() {
        for (let r = 0; r < this.size; r++) {
            for (let c = 0; c < this.size; c++) {
                if (this.board[r][c] === 2046) {
                    return true;
                }
            }
        }
        return false;
    }
    
    checkGameOver() {
        // 检查是否有空格
        for (let r = 0; r < this.size; r++) {
            for (let c = 0; c < this.size; c++) {
                if (this.board[r][c] === 0) {
                    return false;
                }
            }
        }
        
        // 检查是否有相邻的相同数字
        for (let r = 0; r < this.size; r++) {
            for (let c = 0; c < this.size; c++) {
                if (c < this.size - 1 && this.board[r][c] === this.board[r][c + 1]) {
                    return false;
                }
                if (r < this.size - 1 && this.board[r][c] === this.board[r + 1][c]) {
                    return false;
                }
            }
        }
        
        return true;
    }
    
    saveState() {
        // 保存当前状态用于撤销
        this.previousState = {
            board: this.board.map(row => [...row]),
            score: this.score,
            gameOver: this.gameOver,
            won: this.won
        };
    }
    
    undo() {
        if (this.previousState) {
            // 恢复之前的状态
            this.board = this.previousState.board.map(row => [...row]);
            this.score = this.previousState.score;
            this.gameOver = this.previousState.gameOver;
            this.won = this.previous
