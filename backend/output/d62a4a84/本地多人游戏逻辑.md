  const canvas = document.getElementById('gameCanvas');
  const ctx = canvas.getContext('2d');

  canvas.width = window.innerWidth;
  canvas.height = window.innerHeight;

  let ball1 = {
    x: canvas.width / 4,
    y: canvas.height / 2,
    radius: 30,
    color: 'red',
    velocityX: 5,
    velocityY: 5
  };

  let ball2 = {
    x: canvas.width * 3 / 4,
    y: canvas.height / 2,
    radius: 30,
    color: 'blue',
    velocityX: -5,
    velocityY: 5
  };

  let score1 = 0;
  let score2 = 0;

  function drawBall(ball) {
    ctx.beginPath();
    ctx.arc(ball.x, ball.y, ball.radius, 0, Math.PI * 2);
    ctx.fillStyle = ball.color;
    ctx.fill();
    ctx.closePath();
  }

  function drawScore() {
    ctx.font = '24px Arial';
    ctx.fillStyle = 'black';
    ctx.fillText('Player 1: ' + score1, 10, 30);
    ctx.fillText('Player 2: ' + score2, canvas.width - 150, 30);
  }

  function draw() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    drawBall(ball1);
    drawBall(ball2);
    drawScore();

    // Update ball1 position
    ball1.x += ball1.velocityX;
    ball1.y += ball1.velocityY;

    // Update ball2 position
    ball2.x += ball2.velocityX;
    ball2.y += ball2.velocityY;

    // Reverse direction when ball hits the canvas wall
    if (ball1.x + ball1.radius > canvas.width || ball1.x - ball1.radius < 0) {
      ball1.velocityX = -ball1.velocityX;
    }
    if (ball1.y + ball1.radius > canvas.height || ball1.y - ball1.radius < 0) {
      ball1.velocityY = -ball1.velocityY;
    }

    if (ball2.x + ball2.radius > canvas.width || ball2.x - ball2.radius < 0) {
      ball2.velocityX = -ball2.velocityX;
    }
    if (ball2.y + ball2.radius > canvas.height || ball2.y - ball2.radius < 0) {
      ball2.velocityY = -ball2.velocityY;
    }

    // Collision detection and scoring
    if (detectCollision(ball1, ball2)) {
      swapVelocity(ball1, ball2);
      score1 += 10;
      ball1.radius += 2;
      ball2.radius += 2;
    }

    requestAnimationFrame(draw);
  }

  function detectCollision(ball1, ball2) {
    let dx = ball2.x - ball1.x;
    let dy = ball2.y - ball1.y;
    let distance = Math.sqrt(dx * dx + dy * dy);
    return distance < (ball1.radius + ball2.radius);
  }

  function swapVelocity(ball1, ball2) {
    let tempX = ball1.velocityX;
    let tempY = ball1.velocityY;
    ball1.velocityX = ball2.velocityX;
    ball1.velocityY = ball2.velocityY;
    ball2.velocityX = tempX;
    ball2.velocityY = tempY;
  }
