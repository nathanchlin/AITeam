# 游戏启动与退出功能

根据您的要求，以下是一个完整的单文件HTML，它实现了游戏启动和退出的功能。在这个例子中，我将使用一个按钮来控制游戏的开始和结束。

```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>球球大作战 - 游戏启动与退出</title>
<style>
  body {
    margin: 0;
    display: flex;
    justify-content: center;
    align-items: center;
    height: 100vh;
    background-color: #f0f0f0;
  }
  canvas {
    border: 2px solid #000;
  }
  #controls {
    position: absolute;
    bottom: 10px;
    left: 10px;
  }
  button {
    padding: 5px 10px;
    margin-right: 5px;
    cursor: pointer;
  }
</style>
</head>
<body>
<canvas id="gameCanvas"></canvas>
<div id="controls">
  <button id="startButton">开始游戏</button>
  <button id="exitButton">退出游戏</button>
</div>
<script>
  const canvas = document.getElementById('gameCanvas');
  const ctx = canvas.getContext('2d');
  const startButton = document.getElementById('startButton');
  const exitButton = document.getElementById('exitButton');
  let gameRunning = false;

  canvas.width = window.innerWidth * 0.8;
  canvas.height = window.innerHeight * 0.8;

  let ball = {
    x: canvas.width / 2,
    y: canvas.height / 2,
    radius: 30,
    color: 'red',
    velocityX: 5,
    velocityY: 5
  };

  let score = 0;

  function drawBall() {
    ctx.beginPath();
    ctx.arc(ball.x, ball.y, ball.radius, 0, Math.PI * 2);
    ctx.fillStyle = ball.color;
    ctx.fill();
    ctx.closePath();
  }

  function drawScore() {
    ctx.font = '24px Arial';
    ctx.fillStyle = 'black';
    ctx.fillText('Score: ' + score, 10, 30);
  }

  function draw() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    drawBall();
    drawScore();
    ball.x += ball.velocityX;
    ball.y += ball.velocityY;

    // Reverse direction when ball hits the canvas wall
    if (ball.x + ball.radius > canvas.width || ball.x - ball.radius < 0) {
      ball.velocityX = -ball.velocityX;
    }

    if (ball.y + ball.radius > canvas.height || ball.y - ball.radius < 0) {
      ball.velocityY = -ball.velocityY;
    }

    requestAnimationFrame(draw);
  }

  function handleKeyDown(event) {
    switch (event.key) {
      case 'ArrowLeft':
        ball.velocityX = -5;
        break;
      case 'ArrowRight':
        ball.velocityX = 5;
        break;
      case 'ArrowUp':
        ball.velocityY = -5;
        break;
      case 'ArrowDown':
        ball.velocityY = 5;
        break;
    }
  }

  function handleKeyUp(event) {
    switch (event.key) {
      case 'ArrowLeft':
      case 'ArrowRight':
        ball.velocityX = 0;
        break;
      case 'ArrowUp':
      case 'ArrowDown':
        ball.velocityY = 0;
        break;
    }
  }

  function startGame() {
    if (!gameRunning) {
      gameRunning = true;
      document.addEventListener('keydown', handleKeyDown);
      document.addEventListener('keyup', handleKeyUp);
      draw();
    }
  }

  function exitGame() {
    gameRunning = false;
    document.removeEventListener('keydown', handleKeyDown);
    document.removeEventListener('keyup', handleKeyUp);
  }

  startButton.addEventListener('click', startGame);
  exitButton.addEventListener('click', exitGame);

  window.onload = function() {
    document.addEventListener('keydown', handleKeyDown);
    document.addEventListener('keyup', handleKeyUp);
  };
</script>
</body>
</html>
```

在这个代码中，我添加了两个按钮：“开始游戏”和“退出游戏”。点击“开始游戏”按钮将启动游戏，球体开始移动，并且键盘方向键可以控制球体的移动。点击“退出游戏”按钮将停止游戏，球体的移动将停止，并且不再响应键盘事件。

游戏循环使用`requestAnimationFrame`来保持流畅的动画效果，并且所有的逻辑都封装在同一个HTML文件中，满足您的要求。
