        // 资源管理器 - 负责加载和管理游戏资源
        class ResourceManager {
            constructor() {
                this.resources = {};
                this.loadingQueue = [];
                this.loadedCount = 0;
                this.totalResources = 0;
                this.loadingCallbacks = [];
                this.cache = new Map(); // 用于缓存已加载的资源
            }
            
            // 添加资源到加载队列
            addResource(name, url, type = 'image') {
                this.loadingQueue.push({ name, url, type });
                this.totalResources++;
                return this;
            }
            
            // 设置加载完成回调
            onLoaded(callback) {
                this.loadingCallbacks.push(callback);
                return this;
            }
            
            // 开始加载所有资源
            loadAll() {
                return new Promise((resolve) => {
                    const loadNext = () => {
                        if (this.loadingQueue.length === 0) {
                            this.loadedCount = this.totalResources;
                            this.updateProgress(100);
                            
                            // 执行所有回调
                            this.loadingCallbacks.forEach(callback => callback());
                            resolve();
                            return;
                        }
                        
                        const resource = this.loadingQueue.shift();
                        this.loadResource(resource)
                            .then(() => {
                                this.loadedCount++;
                                const progress = Math.round((this.loadedCount / this.totalResources) * 100);
                                this.updateProgress(progress);
                                loadNext();
                            })
                            .catch(error => {
                                console.error(`加载资源失败: ${resource.name}`, error);
                                loadNext();
                            });
                    };
                    
                    loadNext();
                });
            }
            
            // 加载单个资源
            loadResource({ name, url, type }) {
                return new Promise((resolve, reject) => {
                    // 检查缓存
                    if (this.cache.has(name)) {
                        this.resources[name] = this.cache.get(name);
                        resolve();
                        return;
                    }
                    
                    switch (type) {
                        case 'image':
                            const img = new Image();
                            img.src = url;
                            img.onload = () => {
                                this.resources[name] = img;
                                this.cache.set(name, img);
                                resolve();
                            };
                            img.onerror = () => reject(new Error(`图片加载失败: ${url}`));
                            break;
                            
                        case 'audio':
                            const audio = new Audio();
                            audio.src = url;
                            audio.load();
                            audio.oncanplaythrough = () => {
                                this.resources[name] = audio;
                                this.cache.set(name, audio);
                                resolve();
                            };
                            audio.onerror = () => reject(new Error(`音频加载失败: ${url}`));
                            break;
                            
                        default:
                            reject(new Error(`不支持的资源类型: ${type}`));
                    }
                });
            }
            
            // 更新加载进度
            updateProgress(percent) {
                const progressElement = document.getElementById('loadingProgress');
                if (progressElement) {
                    progressElement.style.width = `${percent}%`;
                }
            }
            
            // 获取资源
            get(name) {
                return this.resources[name];
            }
            
            // 检查资源是否已加载
            isLoaded(name) {
                return this.resources.hasOwnProperty(name);
            }

                        // 预加载逻辑
                    }, 1000);
                }
            }
            
            // 释放资源（内存管理）
            release(name) {
                if (this.resources[name]) {
                    delete this.resources[name];
                    this.cache.delete(name);
                }
            }
            
            // 释放所有资源
            releaseAll() {
                this.resources = {};
                this.cache.clear();
            }
        }

            // 创建音效（使用Web Audio API）
            createSounds() {
                const audioContext = new (window.AudioContext || window.webkitAudioContext)();
                
                // 匹配音效
                const createBeep = (frequency, duration) => {
                    const oscillator = audioContext.createOscillator();
                    const gainNode = audioContext.createGain();
                    
                    oscillator.connect(gainNode);
                    gainNode.connect(audioContext.destination);
                    
                    oscillator.frequency.value = frequency;
                    gainNode.gain.value = 0.3;
                    
                    oscillator.start();
                    oscillator.stop(audioContext.currentTime + duration);
                    
                    return oscillator;
                };
                
                // 匹配音效
                this.resourceManager.resources['matchSound'] = {
                    play: () => {
                        if (!this.soundEnabled) return;
                        createBeep(523.25, 0.1); // C5
                        setTimeout(() => createBeep(659.25, 0.1), 50); // E5
                        setTimeout(() => createBeep(783.99, 0.2), 100); // G5
                    }
                };
                
                // 交换音效
                this.resourceManager.resources['swapSound'] = {
                    play: () => {
                        if (!this.soundEnabled) return;
                        createBeep(440, 0.1); // A4
                    }
                };
                
                // 无效移动音效
                this.resourceManager.resources['invalidSound'] = {
                    play: () => {

                        createBeep(220, 0.2); // A3
                    }
                };
            }

                    }
                }

            // 生成随机宝石
            generateRandomGem(row, col) {
                let gemType;
                do {
                    gemType = Math.floor(Math.random() * GAME_CONFIG.gemShapes.length);
                } while (this.wouldCreateMatch(row, col, gemType));
                
                return {
                    type: gemType,
                    row: row,
                    col: col,
                    x: col * GAME_CONFIG.tileSize,
                    y: row * GAME_CONFIG.tileSize,
                    scale: 1,
                    rotation: 0,
                    isFalling: false,
                    isMatched: false
                };
            }
            
            // 检查是否会产生匹配
            wouldCreateMatch(row, col, gemType) {
                // 检查水平方向
                let horizontalCount = 1;
                
                // 向左检查
                for (let c = col - 1; c >= 0 && this.board[row] && this.board[row][c] && this.board[row][c].type === gemType; c--) {
                    horizontalCount++;
                }
                
                // 向右检查
                for (let c = col + 1; c < GAME_CONFIG.cols && this.board[row] && this.board[row][c] && this.board[row][c].type === gemType; c++) {
                    horizontalCount++;
                }
                
                if (horizontalCount >= GAME_CONFIG.matchSize) {
                    return true;
                }
                
                // 检查垂直方向
                let verticalCount = 1;
                
                // 向上检查
                for (let r = row - 1; r >= 0 && this.board[r] && this.board[r][col] && this.board[r][col].type === gemType; r--) {
                    verticalCount++;
                }
                
                // 向下检查
                for (let r = row + 1; r < GAME_CONFIG.rows && this.board[r] && this.board[r][col] && this.board[r][col].type === gemType; r++) {
                    verticalCount++;
                }
                
                if (verticalCount >= GAME_CONFIG.matchSize) {
                    return true;
                }
                
                return false;
            }

            // 处理点击事件
            handleClick(e) {
                if (this.isPaused || this.isAnimating) return;
                
                const col = Math.floor(e.offsetX / GAME_CONFIG.tileSize);
                const row = Math.floor(e.offsetY / GAME_CONFIG.tileSize);
                
                if (row < 0 || row >= GAME_CONFIG.rows || col < 0 || col >= GAME_CONFIG.cols) return;
                
                const clickedTile = this.board[row][col];
                if (!clickedTile) return;
                
                if (!this.selectedTile) {
                    // 选择第一个宝石
                    this.selectedTile = clickedTile;
