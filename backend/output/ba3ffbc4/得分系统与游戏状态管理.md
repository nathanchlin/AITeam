    <script>
        // 游戏常量
        const GRID_SIZE = 8; // 8x8的网格
        const CELL_SIZE = 60; // 每个格子的大小
        const CELL_MARGIN = 2; // 格子之间的间隙
        const COLORS = ['#FF5252', '#FF4081', '#E040FB', '#7C4DFF', '#536DFE', '#448AFF', '#40C4FF', '#18FFFF', '#64FFDA', '#69F0AE'];
        const MIN_MATCH = 3; // 最少匹配数量
        const INITIAL_MOVES = 20; // 初始步数
        const SCORE_PER_MATCH = 100; // 每次匹配的基础分数
        const BONUS_MULTIPLIER = 1.5; // 连击倍数
        
        // 游戏状态
        class GameState {
            constructor() {
                this.score = 0;
                this.level = 1;
                this.moves = INITIAL_MOVES;
                this.targetScore = 1000;
                this.grid = [];
                this.selectedCell = null;
                this.isAnimating = false;
                this.combo = 0;
                this.gameOver = false;
                this.levelComplete = false;
            }
            
            reset() {
                this.score = 0;
                this.moves = INITIAL_MOVES;
                this.targetScore = 1000 * this.level;
                this.selectedCell = null;
                this.combo = 0;
                this.gameOver = false;
                this.levelComplete = false;
            }
            
            nextLevel() {
                this.level++;
                this.reset();
                this.targetScore = 1000 * this.level;
                this.moves = INITIAL_MOVES + (this.level - 1) * 2; // 每关增加2步
            }
            
            updateScore(points) {
                this.score += points;
                if (this.score >= this.targetScore) {
                    this.levelComplete = true;
                }
            }
            
            decrementMoves() {
                this.moves--;
                if (this.moves <= 0) {
                    this.gameOver = true;
                }
            }
        }
        
        // 游戏主类
        class Match3Game {
            constructor(canvasId) {
                this.canvas = document.getElementById(canvasId);
                this.ctx = this.canvas.getContext('2d');
                this.state = new GameState();
                this.animationQueue = [];
                this.messageTimeout = null;
                
                // 初始化游戏
                this.init();
                
                // 绑定事件
                this.bindEvents();
                
                // 开始游戏循环
                this.gameLoop();
            }
            
            init() {
                // 初始化网格
                this.initGrid();
                
                // 更新UI
                this.updateUI();
                
                // 显示欢迎消息
                this.showMessage("欢迎来到消消乐游戏!", "info");
            }
            
            initGrid() {
                this.state.grid = [];
                for (let row = 0; row < GRID_SIZE; row++) {
                    this.state.grid[row] = [];
                    for (let col = 0; col < GRID_SIZE; col++) {
                        // 确保初始网格没有匹配
                        let color;
                        do {
                            color = this.getRandomColor();
                        } while (this.wouldCreateMatch(row, col, color));
                        
                        this.state.grid[row][col] = {
                            color: color,
                            row: row,
                            col: col,
                            scale: 1,
                            alpha: 1,
                            isMatched: false
                        };
                    }
                }
            }
            
            getRandomColor() {
                return COLORS[Math.floor(Math.random() * COLORS.length)];
            }
            
            wouldCreateMatch(row, col, color) {
                // 检查水平方向
                let horizontalCount = 1;
                
                // 向左检查
                for (let c = col - 1; c >= 0 && this.state.grid[row][c] && this.state.grid[row][c].color === color; c--) {
                    horizontalCount++;
                }
                
                // 向右检查
                for (let c = col + 1; c < GRID_SIZE && this.state.grid[row][c] && this.state.grid[row][c].color === color; c++) {
                    horizontalCount++;
                }
                
                // 检查垂直方向
                let verticalCount = 1;
                
                // 向上检查
                for (let r = row - 1; r >= 0 && this.state.grid[r][col] && this.state.grid[r][col].color === color; r--) {
                    verticalCount++;
                }
                
                // 向下检查
                for (let r = row + 1; r < GRID_SIZE && this.state.grid[r][col] && this.state.grid[r][col].color === color; r++) {
                    verticalCount++;
                }
                
                return horizontalCount >= MIN_MATCH || verticalCount >= MIN_MATCH;
            }
            
            bindEvents() {
                // 鼠标点击事件
                this.canvas.addEventListener('click', (e) => {
                    if (this.state.isAnimating || this.state.gameOver || this.state.levelComplete) return;
                    
                    const rect = this.canvas.getBoundingClientRect();
                    const x = e.clientX - rect.left;
                    const y = e.clientY - rect.top;
                    
                    const col = Math.floor(x / (CELL_SIZE + CELL_MARGIN));
                    const row = Math.floor(y / (CELL_SIZE + CELL_MARGIN));
                    
                    if (row >= 0 && row < GRID_SIZE && col >= 0 && col < GRID_SIZE) {
                        this.handleCellClick(row, col);
                    }
                });
                
                // 按钮事件
                document.getElementById('newGameBtn').addEventListener('click', () => {
                    this.state = new GameState();
                    this.init();
                });
                
                document.getElementById('resetBtn').addEventListener('click', () => {
                    this.state.reset();
                    this.initGrid();
                    this.updateUI();
                    this.showMessage("关卡已重置", "info");
                });
            }
            
            handleCellClick(row, col) {
                const cell = this.state.grid[row][col];
                
                if (!cell) return;
                
                if (!this.state.selectedCell) {
                    // 选择第一个格子
                    this.state.selectedCell = { row, col };
                    cell.selected = true;
                } else {
                    const selected = this.state.selectedCell;
                    
                    // 检查是否是相邻格子
                    const rowDiff = Math.abs(row - selected.row);
                    const colDiff = Math.abs(col - selected.col);
                    
                    if ((rowDiff === 1 && colDiff === 0) || (rowDiff === 0 && colDiff === 1)) {
                        // 交换格子
                        this.swapCells(selected.row, selected.col, row, col);
                    } else {
                        // 选择新的格子
                        this.state.grid[selected.row][selected.col].selected = false;
                        this.state.selectedCell = { row, col };
                        cell.selected = true;
                    }
                }
            }
            
            swapCells(row1, col1, row2, col2) {
                const cell1 = this.state.grid[row1][col1];
                const cell2 = this.state.grid[row2][col2];
                
                // 交换颜色
                const tempColor = cell1.color;
                cell1.color = cell2.color;
                cell2.color = tempColor;
                
                // 取消选择
                cell1.selected = false;
                this.state.selectedCell = null;
                
                // 检查是否有匹配
                const matches = this.findMatches();
                
                if (matches.length > 0) {
                    // 有效移动
                    this.state.decrementMoves();
                    this.state.combo = 0;
                    this.processMatches(matches);
                } else {
                    // 无效移动，交换回来
                    const tempColor = cell1.color;
                    cell1.color = cell2.color;
                    cell2.color = tempColor;
                    
                    this.showMessage("无效移动!", "warning");
                }
            }
            
            findMatches() {
                const matches = [];
                
                // 查找水平匹配
                for (let row = 0; row < GRID_SIZE; row++) {
                    let count = 1;
                    let currentColor = this.state.grid[row][0].color;
                    
                    for (let col = 1; col < GRID_SIZE; col++) {
                        if (this.state.grid[row][col].color === currentColor) {
                            count++;
                        } else {
                            if (count >= MIN_MATCH) {
                                for (let i = col - count; i < col; i++) {
                                    matches.push({ row, col: i });
                                }
                            }
                            count = 1;
                            currentColor = this.state.grid[row][col].color;
                        }
                    }
                    
                    if (count >= MIN_MATCH) {
                        for (let i = GRID_SIZE - count; i < GRID_SIZE; i++) {
                            matches.push({ row, col: i });
                        }
                    }
                }
                
                // 查找垂直匹配
                for (let col = 0; col < GRID_SIZE; col++) {
                    let count = 1;
                    let currentColor = this.state.grid[0][col].color;
                    
                    for (let row = 1; row < GRID_SIZE; row++) {
                        if (this.state.grid[row][col].color === currentColor) {
                            count++;
                        } else {
                            if (count >= MIN_MATCH) {
                                for (let i = row - count; i < row; i++) {
                                    matches.push({ row: i, col });
                                }
                            }
                            count = 1;
                            currentColor = this.state.grid[row][col].color;
                        }
                    }
                    
                    if (count >= MIN_MATCH) {
                        for (let i = GRID_SIZE - count; i < GRID_SIZE; i++) {
                            matches.push({ row: i, col });
                        }
                    }
                }
                
                // 去重
                const uniqueMatches = [];
                const matchSet = new Set();
                
                for (const match of matches) {
                    const key = `${match.row},${match.col}`;
                    if (!matchSet.has(key)) {
                        matchSet.add(key);
                        uniqueMatches.push(match);
                    }
                }
                
                return uniqueMatches;
            }
            
            processMatches(matches) {
                this.state.isAnimating = true;
                
                // 标记匹配的格子
                for (const match of matches) {
                    this.state.grid[match.row][match.col].isMatched = true;
                }
                
                // 计算分数
                const matchCount = matches.length;
                const baseScore = matchCount * SCORE_PER_MATCH;
                const comboBonus = this.state.combo > 0 ? Math.floor(baseScore * (this.state.combo * 0.5)) : 0;
                const totalScore = baseScore + comboBonus;
                
                this.state.updateScore(totalScore);
                this.state.combo++;
                
                // 显示得分消息
                if (this.state.combo > 1) {
                    this.showMessage(`连击 x${this.state.combo}! +${totalScore}`, "success");
                } else {
                    this.showMessage(`+${totalScore}`, "success");
                }
                
                // 动画效果
                this.animateMatches(matches, () => {
                    // 移除匹配的格子
                    for (const match of matches) {
                        this.state.grid[match.row][match.col] = null;
                    }
                    
                    // 下落和填充
                    this.dropCells();
                    this.fillEmptyCells();
                    
                    // 检查新的匹配
                    setTimeout(() => {
                        const newMatches = this.findMatches();
                        if (newMatches.length > 0) {
                            // 连锁反应
                            this.processMatches(newMatches);
                        } else {
                            // 没有新的匹配，结束动画
                            this.state.isAnimating = false;
                            this.checkGameState();
                        }
                    }, 300);
                });
            }
            
            animateMatches(matches, callback) {
                const duration = 300; // 动画持续时间（毫秒）
                const startTime = Date.now();
                
                const animate = () => {
                    const elapsed = Date.now() - startTime;
                    const progress = Math.min(elapsed / duration, 1);
                    
                    // 更新匹配格子的缩放和透明度
                    for (const match of matches) {
                        const cell = this.state.grid[match.row][match.col];
                        if (cell) {
                            cell.scale = 1 - progress;
                            cell.alpha = 1 - progress;
                        }
                    }
                    
                    if (progress < 1) {
                        requestAnimationFrame(animate);
                    } else {
                        callback();
                    }
                };
                
                animate();
            }
            
            dropCells() {
                // 让格子下落填补空缺
                for (let col = 0; col < GRID_SIZE; col++) {
                    let emptyRow = GRID_SIZE - 1;
                    
                    for (let row = GRID_SIZE - 1; row >= 0; row--) {
                        if (this.state.grid[row][col] !== null) {
                            if (row !== emptyRow) {
                                this.state.grid[emptyRow][col] = this.state.grid[row][col];
                                this.state.grid[emptyRow][col].row = emptyRow;
                                this.state.grid[row][col] = null;
                            }
                            emptyRow--;
                        }
                    }
                }
            }
            
            fillEmptyCells() {
                // 填充空格子
                for (let col = 0; col < GRID_SIZE; col++) {
                    for (let row = 0; row < GRID_SIZE; row++) {
                        if (this.state.grid[row][col] === null) {
                            this.state.grid[row][col] = {
                                color: this.getRandomColor(),
                                row: row,
                                col: col,
                                scale: 1,
                                alpha: 1,
                                isMatched: false
                            };
                        }
                    }
                }
            }
            
            checkGameState() {
                this.updateUI();
                
                if (this.state.levelComplete) {
                    this.showMessage(`恭喜通过第 ${this.state.level} 关!`, "success");
                    setTimeout(() => {
                        this.state.nextLevel();
                        this.initGrid();
                        this.updateUI();
                        this.showMessage(`欢迎来到第 ${this.state.level} 关!`, "info");
                    }, 2000);
                } else if (this.state.gameOver) {
                    this.showMessage("游戏结束! 点击新游戏重新开始", "warning");
                } else {
                    // 检查是否还有可能的移动
                    if (!this.hasPossibleMoves()) {
                        this.showMessage("没有可能的移动，重新生成网格", "info");
                        this.initGrid();
                    }
                }
            }
            
            hasPossibleMoves() {
                // 检查是否有可能的移动
                for (let row = 0; row < GRID_SIZE; row++) {
                    for (let col = 0; col < GRID_SIZE; col++) {
                        // 尝试向右交换
                        if (col < GRID_SIZE - 1) {
                            this.swapCells(row, col, row, col + 1);
                            const matches = this.findMatches();
                            this.swapCells(row,
