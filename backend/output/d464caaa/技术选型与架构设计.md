# 俄罗斯方块对战游戏技术方案

## 技术选型

基于需求，我将使用原生 JavaScript + Canvas API 来实现这个俄罗斯方块对战游戏。不使用外部游戏框架，而是自己构建完整的游戏架构。

## 架构设计

### 核心模块划分

1. **游戏引擎模块 (GameEngine)**
   - 负责游戏循环管理
   - 帧率控制
   - 游戏状态管理

2. **渲染模块 (Renderer)**
   - 负责Canvas绘制
   - 游戏元素渲染
   - 动画效果处理

3. **输入处理模块 (InputHandler)**
   - 键盘事件处理
   - 玩家操作响应

4. **游戏逻辑模块 (GameLogic)**
   - 俄罗斯方块核心规则
   - 碰撞检测
   - 行消除逻辑

5. **AI模块 (AIPlayer)**
   - AI对手逻辑
   - 难度调整

6. **方块模块 (Tetromino)**
   - 方块形状定义
   - 旋转逻辑

7. **游戏状态模块 (GameState)**
   - 分数统计
   - 游戏进度管理

    <script>
        // 游戏常量
        const COLS = 10;
        const ROWS = 20;
        const BLOCK_SIZE = 30;
        const COLORS = [
            '#FF0D72', '#0DC2FF', '#0DFF72', '#F538FF',
            '#FF8E0D', '#FFE138', '#3877FF'
        ];

        // 方块形状定义
        const SHAPES = [
            [[1, 1, 1, 1]], // I
            [[1, 1], [1, 1]], // O
            [[0, 1, 0], [1, 1, 1]], // T
            [[1, 1, 0], [0, 1, 1]], // S
            [[0, 1, 1], [1, 1, 0]], // Z
            [[1, 0, 0], [1, 1, 1]], // J
            [[0, 0, 1], [1, 1, 1]]  // L
        ];

        // 游戏引擎类
        class GameEngine {
            constructor(canvasId, isAI = false) {
                this.canvas = document.getElementById(canvasId);
                this.ctx = this.canvas.getContext('2d');
                this.isAI = isAI;
                this.board = this.createBoard();
                this.currentPiece = null;
                this.score = 0;
                this.gameOver = false;
                this.dropCounter = 0;
                this.lastTime = 0;
                this.dropInterval = 1000; // 初始下落速度
                this.gameLoop = this.gameLoop.bind(this);
            }

            createBoard() {
                return Array.from({ length: ROWS }, () => Array(COLS).fill(0));
            }

            start() {
                this.spawnPiece();
                requestAnimationFrame(this.gameLoop);
            }

            gameLoop(time = 0) {
                if (this.gameOver) return;

                const deltaTime = time - this.lastTime;
                this.lastTime = time;

                this.dropCounter += deltaTime;
                if (this.dropCounter > this.dropInterval) {
                    this.drop();
                    this.dropCounter = 0;
                }

                this.draw();
                requestAnimationFrame(this.gameLoop);
            }

            spawnPiece() {
                const typeId = Math.floor(Math.random() * SHAPES.length);
                this.currentPiece = {
                    x: Math.floor(COLS / 2) - Math.floor(SHAPES[typeId][0].length / 2),
                    y: 0,
                    typeId: typeId,
                    rotation: 0,
                    color: COLORS[typeId]
                };

                // 检查游戏是否结束
                if (this.collision()) {
                    this.gameOver = true;
                    this.onGameOver();
                }
            }

            collision() {
                const shape = this.getCurrentShape();
                for (let y = 0; y < shape.length; y++) {
                    for (let x = 0; x < shape[y].length; x++) {
                        if (shape[y][x]) {
                            const boardX = this.currentPiece.x + x;
                            const boardY = this.currentPiece.y + y;
                            
                            if (boardX < 0 || boardX >= COLS || boardY >= ROWS) {
                                return true;
                            }
                            
                            if (boardY >= 0 && this.board[boardY][boardX]) {
                                return true;
                            }
                        }
                    }
                }
                return false;
            }

            getCurrentShape() {
                const shape = SHAPES[this.currentPiece.typeId];
                const rotation = this.currentPiece.rotation % 4;
                
                if (rotation === 0) return shape;
                if (rotation === 1) return this.rotateMatrix(shape);
                if (rotation === 2) return this.rotateMatrix(this.rotateMatrix(shape));
                return this.rotateMatrix(this.rotateMatrix(this.rotateMatrix(shape)));
            }

            rotateMatrix(matrix) {
                const rows = matrix.length;
                const cols = matrix[0].length;
                const rotated = Array.from({ length: cols }, () => Array(rows).fill(0));
                
                for (let y = 0; y < rows; y++) {
                    for (let x = 0; x < cols; x++) {
                        rotated[x][rows - 1 - y] = matrix[y][x];
                    }
                }
                
                return rotated;
            }

            move(dirX) {
                this.currentPiece.x += dirX;
                if (this.collision()) {
                    this.currentPiece.x -= dirX;
                    return false;
                }
                return true;
            }

            rotate() {
                const prevRotation = this.currentPiece.rotation;
                this.currentPiece.rotation = (this.currentPiece.rotation + 1) % 4;
                
                if (this.collision()) {
                    this.currentPiece.rotation = prevRotation;
                    return false;
                }
                return true;
            }

            drop() {
                this.currentPiece.y++;
                if (this.collision()) {
                    this.currentPiece.y--;
                    this.lockPiece();
                    this.clearLines();
                    this.spawnPiece();
                }
            }

            hardDrop() {
                while (!this.collision()) {
                    this.currentPiece.y++;
                }
                this.currentPiece.y--;
                this.lockPiece();
                this.clearLines();
                this.spawnPiece();
            }

            lockPiece() {
                const shape = this.getCurrentShape();
                for (let y = 0; y < shape.length; y++) {
                    for (let x = 0; x < shape[y].length; x++) {
                        if (shape[y][x]) {
                            const boardX = this.currentPiece.x + x;
                            const boardY = this.currentPiece.y + y;
                            if (boardY >= 0) {
                                this.board[boardY][boardX] = this.currentPiece.color;
                            }
                        }
                    }
                }
            }

            clearLines() {
                let linesCleared = 0;
                
                for (let y = ROWS - 1; y >= 0; y--) {
                    if (this.board[y].every(cell => cell !== 0)) {
                        this.board.splice(y, 1);
                        this.board.unshift(Array(COLS).fill(0));
                        linesCleared++;
                        y++; // 重新检查当前行
                    }
                }
                
                if (linesCleared > 0) {
                    this.score += linesCleared * 100 * linesCleared; // 连击加分
                    this.updateScore();
                    this.increaseSpeed();
                }
            }

            increaseSpeed() {
                // 每得1000分增加速度
                if (this.score > 0 && this.score % 1000 === 0) {
                    this.dropInterval = Math.max(100, this.dropInterval - 100);
                }
            }

            updateScore() {
                if (this.isAI) {
                    document.getElementById('aiScore').textContent = this.score;
                } else {
                    document.getElementById('playerScore').textContent = this.score;
                }
            }

            draw() {
                // 清空画布
                this.ctx.fillStyle = '#000';
                this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);
                
                // 绘制网格
                this.drawGrid();
                
                // 绘制已锁定的方块
                for (let y = 0; y < ROWS; y++) {
                    for (let x = 0; x < COLS; x++) {
                        if (this.board[y][x]) {
                            this.drawBlock(x, y, this.board[y][x]);
                        }
                    }
                }
                
                // 绘制当前方块
                if (this.currentPiece) {
                    const shape = this.getCurrentShape();
                    for (let y = 0; y < shape.length; y++) {
                        for (let x = 0; x < shape[y].length; x++) {
                            if (shape[y][x]) {
                                this.drawBlock(
                                    this.currentPiece.x + x,
                                    this.currentPiece.y + y,
                                    this.currentPiece.color
                                );
                            }
                        }
                    }
                    
                    // 绘制方块阴影
                    this.drawShadow();
                }
            }

            drawGrid() {
                this.ctx.strokeStyle = '#222';
                this.ctx.lineWidth = 0.5;
                
                for (let x = 0; x <= COLS; x++) {
                    this.ctx.beginPath();
                    this.ctx.moveTo(x * BLOCK_SIZE, 0);
                    this.ctx.lineTo(x * BLOCK_SIZE, ROWS * BLOCK_SIZE);
                    this.ctx.stroke();
                }
                
                for (let y = 0; y <= ROWS; y++) {
                    this.ctx.beginPath();
                    this.ctx.moveTo(0, y * BLOCK_SIZE);
                    this.ctx.lineTo(COLS * BLOCK_SIZE, y * BLOCK_SIZE);
                    this.ctx.stroke();
                }
            }

            drawBlock(x, y, color) {
                this.ctx.fillStyle = color;
                this.ctx.fillRect(x * BLOCK_SIZE, y * BLOCK_SIZE, BLOCK_SIZE, BLOCK_SIZE);
                
                // 添加高光效果
                this.ctx.fillStyle = 'rgba(255, 255, 255, 0.3)';
                this.ctx.fillRect(x * BLOCK_SIZE, y * BLOCK_SIZE, BLOCK_SIZE, 4);
                this.ctx.fillRect(x * BLOCK_SIZE, y * BLOCK_SIZE, 4, BLOCK_SIZE);
                
                // 添加阴影效果
                this.ctx.fillStyle = 'rgba(0, 0, 0, 0.3)';
                this.ctx.fillRect(x * BLOCK_SIZE + BLOCK_SIZE - 4, y * BLOCK_SIZE, 4, BLOCK_SIZE);
                this.ctx.fillRect(x * BLOCK_SIZE, y * BLOCK_SIZE + BLOCK_SIZE - 4, BLOCK_SIZE, 4);
            }

            drawShadow() {
                let shadowY = this.currentPiece.y;
                
                while (true) {
                    shadowY++;
                    const shape = this.getCurrentShape();
                    let collision = false;
                    
                    for (let y = 0; y < shape.length; y++) {
                        for (let x = 0; x < shape[y].length; x++) {
                            if (shape[y][x]) {
                                const boardX = this.currentPiece.x + x;
                                const boardY = shadowY + y;
                                
                                if (boardY >= ROWS || (boardY >= 0 && this.board[boardY][boardX])) {
                                    collision = true;
                                    break;
                                }
                            }
                        }
                        if (collision) break;
                    }
                    
                    if (collision) break;
                }
                
                shadowY--;
                
                const shape = this.getCurrentShape();
                this.ctx.fillStyle = 'rgba(255, 255, 255, 0.2)';
                for (let y = 0; y < shape.length; y++) {
                    for (let x = 0; x < shape[y].length; x++) {
                        if (shape[y][x]) {
                            this.ctx.fillRect(
                                (this.currentPiece.x + x) * BLOCK_SIZE,
                                (shadowY + y) * BLOCK_SIZE,
                                BLOCK_SIZE,
                                BLOCK_SIZE
                            );
                        }
                    }
                }
            }

            onGameOver() {
                // 游戏结束逻辑将在外部实现
            }
        }

        // AI玩家类
        class AIPlayer extends GameEngine {
            constructor(canvasId) {
                super(canvasId, true);
                this.thinkingTime = 0;
                this.thinkInterval = 500; // AI思考间隔
                this.targetX = 0;
                this.targetRotation = 0;
                this.isThinking = false;
            }

            gameLoop(time = 0) {
                if (this.gameOver) return;

                const deltaTime = time - this.lastTime;
                this.lastTime = time;

                this.dropCounter += deltaTime;
                this.thinkingTime += deltaTime;
                
                // AI决策
                if (this.thinkingTime > this.thinkInterval && !this.isThinking) {
                    this.makeDecision();
                    this.thinkingTime = 0;
                }
                
                // 执行决策
                if (this.isThinking) {
                    this.executeDecision();
                }
                
                if (this.dropCounter > this.dropInterval) {
                    this.drop();
                    this.dropCounter = 0;
                }

                this.draw();
                requestAnimationFrame(this.gameLoop);
            }

            makeDecision() {
                // 简单的AI策略：找到最佳位置
                const bestMove = this.findBestMove();
                this.targetX = bestMove.x;
                this.targetRotation = bestMove.rotation;
                this.isThinking = true;
            }

            findBestMove() {
                let bestScore = -Infinity;
                let bestMove = { x: 0, rotation: 0 };
                
                // 尝试所有可能的旋转和位置
                for (let rotation = 0; rotation < 4; rotation++) {
                    const shape = this.getRotatedShape(rotation);
                    const shapeWidth = shape[0].length;
                    
                    for (let x = -shapeWidth + 1; x < COLS; x++) {
                        // 模拟放置
                        const boardCopy = this.board.map(row => [...row]);
                        const currentPieceCopy = {
                            ...this.currentPiece,
                            x: x,
                            rotation: rotation
                        };
                        
                        // 计算最终位置
                        let finalY = 0;
                        while (true) {
                            finalY++;
                            const shape = this.getRotatedShape(rotation);
                            let collision = false;
                            
                            for (let y = 0; y < shape.length; y++) {
                                for (let x = 0; x < shape[y].length; x++) {
                                    if (shape[y][x]) {
                                        const boardX = currentPieceCopy.x + x;
                                        const boardY = finalY + y;
                                        
                                        if (boardY >= ROWS || (boardY >= 0 && boardCopy[boardY][boardX])) {
                                            collision = true;
                                            break;
                                        }
                                    }
                                }
                                if (collision) break;
                            }
                            
                            if (collision) break;
                        }
                        finalY--;
                        
                        // 计算分数
                        const score = this.evaluateMove(shape, x, finalY, boardCopy);
                        
                        if (score > bestScore) {
                            bestScore = score;
                            bestMove = { x: x, rotation: rotation };
                        }
                    }
                }
                
                return bestMove;
            }

            getRotatedShape(rotation) {
