        class Piece {
            constructor(shape, color) {
                this.shape = shape;
                this.color = color;
                this.x = Math.floor((COLS - shape[0].length) / 2);
                this.y = 0;
            }
            
            rotate() {
                // 顺时针旋转90度
                const rows = this.shape.length;
                const cols = this.shape[0].length;
                const rotated = Array(cols).fill().map(() => Array(rows).fill(0));
                
                for (let r = 0; r < rows; r++) {
                    for (let c = 0; c < cols; c++) {
                        rotated[c][rows - 1 - r] = this.shape[r][c];
                    }
                }
                
                return rotated;
            }

                let ghostY = this.y;
                
                while (this.canMoveTo(this.x, ghostY + 1, this.shape, board)) {
                    ghostY++;

                return ghostY;
            }
            
            canMoveTo(newX, newY, shape = this.shape, board = this.board) {
                for (let r = 0; r < shape.length; r++) {
                    for (let c = 0; c < shape[r].length; c++) {
                        if (shape[r][c]) {
                            const boardX = newX + c;
                            const boardY = newY + r;
                            
                            if (boardX < 0 || boardX >= COLS || boardY >= ROWS) {
                                return false;
                            }
                            
                            if (boardY >= 0 && board[boardY][boardX]) {
                                return false;
                            }
                        }
                    }
                }
                return true;
            }

                            const boardX = this.x + c;
                            
                            if (boardY >= 0) {
                                board[boardY][boardX] = this.color;
                            }
                        }
                    }
                }
            }

                );
            }

                this.currentPiece = this.nextPiece;
                this.generateNextPiece();

                    this.gameState.gameOver = true;
                    return false;
                }
                
                return true;

                if (!this.currentPiece || this.gameState.gameOver) return;

                    this.gameState.lines += clearedLines;

                    if (newLevel > this.gameState.level) {
                        this.gameState.level = newLevel;
                        this.gameState.dropInterval = Math.max(100, MOVE_INTERVAL - (newLevel - 1) * 100);
                    }
                    
                    // 如果消除足够多的行，发送攻击
                    if (clearedLines >= LINE_ATTACK_THRESHOLD) {
                        this.gameState.pendingAttacks += Math.floor(clearedLines / LINE_ATTACK_THRESHOLD);
                    }
                }
                
                // 生成新方块

            }

                        this.gameState.board.unshift(Array(COLS).fill(0));
                        linesCleared++;
                        r++; // 重新检查当前行
                    }
                }
