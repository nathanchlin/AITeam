    <script>
        // 游戏常量
        const COLS = 10;
        const ROWS = 20;
        const BLOCK_SIZE = 30;
        
        // 方块形状定义
        const SHAPES = [
            // I
            [[1, 1, 1, 1]],
            // O
            [[1, 1],
             [1, 1]],
            // T
            [[0, 1, 0],
             [1, 1, 1]],
            // S
            [[0, 1, 1],
             [1, 1, 0]],
            // Z
            [[1, 1, 0],
             [0, 1, 1]],
            // J
            [[1, 0, 0],
             [1, 1, 1]],
            // L
            [[0, 0, 1],
             [1, 1, 1]]
        ];
        
        // 方块颜色
        const COLORS = [
            '#00f0f0', // I - 青色
            '#f0f000', // O - 黄色
            '#a000f0', // T - 紫色
            '#00f000', // S - 绿色
            '#f00000', // Z - 红色
            '#0000f0', // J - 蓝色
            '#f0a000'  // L - 橙色
        ];
        
        // 游戏状态
        class GameState {
            constructor() {
                this.board = Array(ROWS).fill().map(() => Array(COLS).fill(0));
                this.currentPiece = null;
                this.nextPiece = null;
                this.score = 0;
                this.lines = 0;
                this.level = 1;
                this.gameOver = false;
                this.paused = false;
                this.gameMode = 'ai';
                this.aiDifficulty = 'medium';
                this.gameSpeed = 5;
                this.dropCounter = 0;
                this.lastTime = 0;
                this.aiDropCounter = 0;
                this.aiLastTime = 0;
            }
            
            reset() {
                this.board = Array(ROWS).fill().map(() => Array(COLS).fill(0));
                this.currentPiece = null;
                this.nextPiece = null;
                this.score = 0;
                this.lines = 0;
                this.level = 1;
                this.gameOver = false;
                this.paused = false;
                this.dropCounter = 0;
                this.aiDropCounter = 0;
            }
        }
        
        // 方块类
        class Piece {
            constructor(shape, color) {
                this.shape = shape;
                this.color = color;
                this.x = Math.floor((COLS - shape[0].length) / 2);
                this.y = 0;
            }
            
            rotate() {
                const rows = this.shape.length;
                const cols = this.shape[0].length;
                const rotated = Array(cols).fill().map(() => Array(rows).fill(0));
                
                for (let r = 0; r < rows; r++) {
                    for (let c = 0; c < cols; c++) {
                        rotated[c][rows - 1 - r] = this.shape[r][c];
                    }
                }
                
                return rotated;
            }
        }
        
        // 游戏主类
        class TetrisGame {
            constructor(canvasId, nextPieceCanvasId) {
                this.canvas = document.getElementById(canvasId);
                this.ctx = this.canvas.getContext('2d');
                this.nextPieceCanvas = document.getElementById(nextPieceCanvasId);
                this.nextPieceCtx = this.nextPieceCanvas.getContext('2d');
                
                this.state = new GameState();
                this.init();
            }
            
            init() {
                this.generateNewPiece();
                this.generateNewPiece(); // 生成下一个方块
                this.setupEventListeners();
                this.setupControls();
                this.gameLoop();
            }
            
            setupEventListeners() {
                document.addEventListener('keydown', (e) => this.handleKeyDown(e));
                document.getElementById('pauseButton').addEventListener('click', () => this.togglePause());
                document.getElementById('restartButton').addEventListener('click', () => this.restart());
                
                // 配置更改事件
                document.getElementById('gameMode').addEventListener('change', (e) => {
                    this.state.gameMode = e.target.value;
                });
                
                document.getElementById('aiDifficulty').addEventListener('change', (e) => {
                    this.state.aiDifficulty = e.target.value;
                    this.updateDifficultyDescription();
                });
                
                document.getElementById('gameSpeed').addEventListener('input', (e) => {
                    this.state.gameSpeed = parseInt(e.target.value);
                    document.getElementById('speedValue').textContent = e.target.value;
                });
            }
            
            setupControls() {
                this.updateDifficultyDescription();
            }
            
            updateDifficultyDescription() {
                const descriptions = {
                    'easy': 'AI反应较慢，适合新手',
                    'medium': 'AI反应速度适中，有一定挑战性',
                    'hard': 'AI反应迅速，难度较高',
                    'expert': 'AI几乎完美，极具挑战性'
                };
                
                document.getElementById('difficultyDescription').textContent = 
                    descriptions[this.state.aiDifficulty];
            }
            
            generateNewPiece() {
                if (this.state.nextPiece) {
                    this.state.currentPiece = this.state.nextPiece;
                } else {
                    const randomIndex = Math.floor(Math.random() * SHAPES.length);
                    this.state.currentPiece = new Piece(
                        SHAPES[randomIndex],
                        COLORS[randomIndex]
                    );
                }
                
                const randomIndex = Math.floor(Math.random() * SHAPES.length);
                this.state.nextPiece = new Piece(
                    SHAPES[randomIndex],
                    COLORS[randomIndex]
                );
                
                // 检查游戏是否结束
                if (this.checkCollision(this.state.currentPiece)) {
                    this.state.gameOver = true;
                    this.showGameOver();
                }
                
                this.drawNextPiece();
            }
            
            drawNextPiece() {
                this.nextPieceCtx.clearRect(0, 0, this.nextPieceCanvas.width, this.nextPieceCanvas.height);
                
                if (!this.state.nextPiece) return;
                
                const blockSize = 20;
                const offsetX = (this.nextPieceCanvas.width - this.state.nextPiece.shape[0].length * blockSize) / 2;
                const offsetY = (this.nextPieceCanvas.height - this.state.nextPiece.shape.length * blockSize) / 2;
                
                for (let r = 0; r < this.state.nextPiece.shape.length; r++) {
                    for (let c = 0; c < this.state.nextPiece.shape[r].length; c++) {
                        if (this.state.nextPiece.shape[r][c]) {
                            this.nextPieceCtx.fillStyle = this.state.nextPiece.color;
                            this.nextPieceCtx.fillRect(
                                offsetX + c * blockSize,
                                offsetY + r * blockSize,
                                blockSize - 1,
                                blockSize - 1
                            );
                        }
                    }
                }
            }
            
            handleKeyDown(e) {
                if (this.state.gameOver || this.state.paused) return;
                
                switch (e.key) {
                    case 'ArrowLeft':
                        this.movePiece(-1, 0);
                        break;
                    case 'ArrowRight':
                        this.movePiece(1, 0);
                        break;
                    case 'ArrowDown':
                        this.movePiece(0, 1);
                        break;
                    case 'ArrowUp':
                        this.rotatePiece();
                        break;
                    case ' ':
                        this.hardDrop();
                        break;
                    case 'p':
                    case 'P':
                        this.togglePause();
                        break;
                }
            }
            
            movePiece(dx, dy) {
                const newPiece = new Piece(
                    this.state.currentPiece.shape,
                    this.state.currentPiece.color
                );
                newPiece.x = this.state.currentPiece.x + dx;
                newPiece.y = this.state.currentPiece.y + dy;
                
                if (!this.checkCollision(newPiece)) {
                    this.state.currentPiece = newPiece;
                    return true;
                }
                
                if (dy > 0) {
                    this.lockPiece();
                }
                
                return false;
            }
            
            rotatePiece() {
                const rotated = this.state.currentPiece.rotate();
                const originalShape = this.state.currentPiece.shape;
                
                this.state.currentPiece.shape = rotated;
                
                if (this.checkCollision(this.state.currentPiece)) {
                    // 尝试墙踢
                    for (let kick of [-1, 1, -2, 2]) {
                        this.state.currentPiece.x += kick;
                        if (!this.checkCollision(this.state.currentPiece)) {
                            return;
                        }
                        this.state.currentPiece.x -= kick;
                    }
                    
                    // 如果墙踢失败，恢复原始形状
                    this.state.currentPiece.shape = originalShape;
                }
            }
            
            hardDrop() {
                while (this.movePiece(0, 1)) {
                    this.state.score += 2;
                }
                this.updateScore();
            }
            
            checkCollision(piece) {
                for (let r = 0; r < piece.shape.length; r++) {
                    for (let c = 0; c < piece.shape[r].length; c++) {
                        if (piece.shape[r][c]) {
                            const newX = piece.x + c;
                            const newY = piece.y + r;
                            
                            if (newX < 0 || newX >= COLS || newY >= ROWS) {
                                return true;
                            }
                            
                            if (newY >= 0 && this.state.board[newY][newX]) {
                                return true;
                            }
                        }
                    }
                }
                return false;
            }
            
            lockPiece() {
                for (let r = 0; r < this.state.currentPiece.shape.length; r++) {
                    for (let c = 0; c < this.state.currentPiece.shape[r].length; c++) {
                        if (this.state.currentPiece.shape[r][c]) {
                            const x = this.state.currentPiece.x + c;
                            const y = this.state.currentPiece.y + r;
                            
                            if (y < 0) {
                                this.state.gameOver = true;
                                this.showGameOver();
                                return;
                            }
                            
                            this.state.board[y][x] = this.state.currentPiece.color;
                        }
                    }
                }
                
                this.clearLines();
                this.generateNewPiece();
            }
            
            clearLines() {
                let linesCleared = 0;
                
                for (let r = ROWS - 1; r >= 0; r--) {
                    if (this.state.board[r].every(cell => cell !== 0)) {
                        this.state.board.splice(r, 1);
                        this.state.board.unshift(Array(COLS).fill(0));
                        linesCleared++;
                        r++; // 重新检查当前行
                    }
                }
                
                if (linesCleared > 0) {
                    this.state.lines += linesCleared;
                    this.state.score += [40, 100, 300, 1200][linesCleared - 1] * this.state.level;
                    this.state.level
