        // 方块类
        class Piece {
            constructor(shape, color) {
                this.shape = shape;
                this.color = color;
                this.x = Math.floor((COLS - shape[0].length) / 2);
                this.y = 0;
            }
            
            rotate() {
                // 转置矩阵
                const rotated = this.shape[0].map((_, i) =>
                    this.shape.map(row => row[i])
                );
                
                // 反转每一行
                for (let i = 0; i < rotated.length; i++) {
                    rotated[i] = rotated[i].reverse();

                return rotated;

                for (let y = 0; y < this.shape.length; y++) {
                    for (let x = 0; x < this.shape[y].length; x++) {
                        if (this.shape[y][x] !== 0) {
                            const newX = this.x + x + dx;
                            const newY = this.y + y + dy;
                            
                            if (newX < 0 || newX >= COLS || newY >= ROWS) {
                                return false;
                            }
                            
                            if (newY >= 0 && board[newY][newX] !== 0) {
                                return false;
                            }
                        }
                    }
                }
                return true;
            }
            
            lock(board) {
                for (let y = 0; y < this.shape.length; y++) {
                    for (let x = 0; x < this.shape[y].length; x++) {
                        if (this.shape[y][x] !== 0) {
                            const boardY = this.y + y;
                            const boardX = this.x + x;

                                // 方块锁定在顶部，游戏结束
                                return false;
                            }
                            
                            board[boardY][boardX] = this.color;
                        }
                    }
                }
                return true;
            }

        class TetrisAI {
            constructor() {
                this.weights = {
                    height: -0.3,
                    holes: -0.5,
                    bumpiness: -0.2,
                    lines: 0.8
                };
            }
            
            // 评估棋盘状态
            evaluateBoard(board) {
                let score = 0;
                let heights = Array(COLS).fill(0);
                let holes = 0;
                let bumpiness = 0;
                
                // 计算每列的高度
                for (let col = 0; col < COLS; col++) {
                    for (let row = ROWS - 1; row >= 0; row--) {
                        if (board[row][col] !== 0) {
                            heights[col] = ROWS - row;
                            break;
                        }
                    }
                }
                
                // 计算空洞和凹凸不平
                let lastHeight = heights[0];
                for (let col = 0; col < COLS; col++) {
                    // 计算空洞
                    for (let row = ROWS - heights[col]; row < ROWS; row++) {
                        if (board[row][col] === 0) {
                            holes++;
                        }
                    }
                    
                    // 计算凹凸不平
                    bumpiness += Math.abs(heights[col] - lastHeight);
                    lastHeight = heights[col];
                }
                
                // 计算完整行数
                let lines = 0;
                for (let row = 0; row < ROWS; row++) {
                    if (board[row].every(cell => cell !== 0)) {
                        lines++;
                    }
                }
                
                // 加权计算分数
                score += heights.reduce((sum, h) => sum + h, 0) * this.weights.height;
                score += holes * this.weights.holes;
                score += bumpiness * this.weights.bumpiness;
                score += lines * this.weights.lines;
                
                return score;
            }
            
            // 获取所有可能的移动
            getPossibleMoves(piece, board) {
                const moves = [];
                
                // 尝试所有可能的x位置

                    // 尝试所有可能的旋转

                            this.getRotatedShape(piece.shape, rotation),
                            piece.color
                        );
                        testPiece.x = x;
                        testPiece.y = 0;
                        
                        // 找到可以落下的最低位置
                        let y = 0;
                        while (testPiece.canMove(0, y + 1, board)) {
                            y++;
                        }
                        testPiece.y = y;
                        
                        // 检查是否可以锁定
                        if (testPiece.canMove(0, 0, board)) {
                            moves.push({
                                piece: testPiece,
                                rotation: rotation,
                                x: x,
                                y: y
                            });
                        }
                    }
                }
                
                return moves;
            }
            
            // 旋转形状
            getRotatedShape(shape, rotations) {
                let rotated = shape;
                for (let i = 0; i < rotations; i++) {
                    rotated = rotated[0].map((_, i) =>
                        rotated.map(row => row[i])
                    );
                    for (let j = 0; j < rotated.length; j++) {
                        rotated[j] = rotated[j].reverse();
                    }
                }
                return rotated;
            }

            selectBestMove(piece, board) {
                const moves = this.getPossibleMoves(piece, board);
                let bestMove = null;
                let bestScore = -Infinity;

                        bestMove = move;
                    }
                }
