                for (let y = 0; y < this.currentPiece.shape.length; y++) {

                                return true;
                            }
                        }
                    }

                return false;
            }

                        if (this.currentPiece.shape[y][x] !== 0) {

                    if (this.isAI && gameState.isRunning && !gameState.isPaused) {
                        this.addObstacle();
                    }
                }

            addObstacle() {
                const linesToAdd = Math.min(Math.floor(this.lines / 10), 5);
                for (let i = 0; i < linesToAdd; i++) {

                }

                        this.aiMove();

                    this.currentPiece.y = bestMove.y;

                        }
                    } else {
                        // 没有碰撞，直接固定
                        this.merge();
                        this.clearLines();
                        
                        if (!this.generatePiece()) {
                            this.handleGameOver();
                        }
                    }
                } else {
                    // 如果找不到最佳移动，硬降
                    if (!this.moveDown()) {
                        this.handleGameOver();
                    }
                }
            }
            
            calculateBestMove() {
                const possibleMoves = [];
                const originalPiece = {
                    shape: this.currentPiece.shape.map(row => [...row]),
                    x: this.currentPiece.x,
                    y: this.currentPiece.y
                };
                
                // 尝试所有可能的旋转和位置
                for (let rotation = 0; rotation < 4; rotation++) {
                    const currentShape = this.currentPiece.shape;
                    
                    // 尝试所有水平位置
                    for (let x = 0; x < COLS; x++) {
                        this.currentPiece.x = x;
                        this.currentPiece.y = 0;
                        
                        // 模拟下落
                        while (!this.collision()) {
                            this.currentPiece.y++;
                        }
                        this.currentPiece.y--;
                        
                        // 计算得分
                        const score = this.evaluatePosition();
                        
                        possibleMoves.push({
                            x: x,
                            y: this.currentPiece.y,
                            shape: currentShape.map(row => [...row]),
                            score: score
                        });
                        
                        // 恢复原始位置
                        this.currentPiece.x = originalPiece.x;
                        this.currentPiece.y = originalPiece.y;
                    }
                    
                    // 旋转方块
                    this.currentPiece.shape = this.currentPiece.rotate();
                }
                
                // 恢复原始方块
                this.currentPiece = {
                    shape: originalPiece.shape.map(row => [...row]),
                    x: originalPiece.x,
                    y: originalPiece.y
                };
                
                // 选择最佳移动
                if (possibleMoves.length > 0) {
                    return possibleMoves.reduce((best, move) => 
                        move.score > best.score ? move : best
                    );
                }
                
                return null;
            }
            
            evaluatePosition() {
                let score = 0;
                
                // 评估高度（越低越好）
                const height = this.currentPiece.y;
                score -= height * 2;
                
                // 评估空洞（越少越好）
                let holes = 0;
                for (let x = 0; x < COLS; x++) {
                    let blockFound = false;
                    for (let y = 0; y < ROWS; y++) {
                        if (this.grid[y][x] !== 0) {
                            blockFound = true;
                        } else if (blockFound) {
                            holes++;
                        }
                    }
                }
                score -= holes * 10;
                
                // 评估行完成度（越接近完成越好）
                for (let y = 0; y < ROWS; y++) {
                    const filledBlocks = this.grid[y].filter(cell => cell !== 0).length;
                    if (filledBlocks > 0 && filledBlocks < COLS) {
                        score += (filledBlocks / COLS) * 5;
                    }
                }
                
                // 评估平滑度（表面越平滑越好）
                let roughness = 0;
                let lastHeight = -1;
                for (let x = 0; x < COLS; x++) {
                    let height = 0;
                    for (let y = 0; y < ROWS; y++) {
                        if (this.grid[y][x] !== 0) {
                            height = ROWS - y;
                            break;
                        }
                    }
                    if (lastHeight !== -1) {
                        roughness += Math.abs(height - lastHeight);
                    }
                    lastHeight = height;
                }
                score -= roughness;
                
                return score;
            }
