        // 游戏常量
        const COLS = 10;
        const ROWS = 20;
        const BLOCK_SIZE = 30;
        const COLORS = [
            '#FF0D72', '#0DC2FF', '#0DFF72', '#F538FF',
            '#FF8E0D', '#FFE138', '#3877FF'
        ];
        
        // 方块形状定义
        const SHAPES = [
            [[1, 1, 1, 1]], // I
            [[1, 1], [1, 1]], // O
            [[1, 1, 1], [0, 1, 0]], // T
            [[1, 1, 1], [1, 0, 0]], // L
            [[1, 1, 1], [0, 0, 1]], // J
            [[0, 1, 1], [1, 1, 0]], // S
            [[1, 1, 0], [0, 1, 1]]  // Z
        ];
        
        // 游戏类
        class TetrisGame {
            constructor(canvasId, nextCanvasId, isAI = false) {
                this.canvas = document.getElementById(canvasId);
                this.ctx = this.canvas.getContext('2d');
                this.nextCanvas = document.getElementById(nextCanvasId);
                this.nextCtx = this.nextCanvas.getContext('2d');
                this.isAI = isAI;
                
                this.board = this.createBoard();
                this.currentPiece = null;
                this.nextPiece = null;
                this.score = 0;
                this.lines = 0;
                this.level = 1;
                this.dropCounter = 0;
                this.lastTime = 0;
                this.gameOver = false;
                this.dropInterval = 1000;
                
                // AI相关属性
                this.aiMoveTimer = 0;
                this.aiMoveInterval = 100; // AI移动间隔(毫秒)
                
                this.init();
            }
            
            createBoard() {
                return Array(ROWS).fill().map(() => Array(COLS).fill(0));
            }
            
            init() {
                this.nextPiece = this.createPiece();
                this.spawnPiece();
                this.draw();
            }
            
            createPiece() {
                const typeId = Math.floor(Math.random() * SHAPES.length);
                return {
                    x: Math.floor(COLS / 2) - Math.floor(SHAPES[typeId][0].length / 2),
                    y: 0,
                    type: typeId,
                    color: COLORS[typeId],
                    shape: SHAPES[typeId]
                };
            }
            
            spawnPiece() {
                this.currentPiece = this.nextPiece;
                this.nextPiece = this.createPiece();
                
                if (this.collision()) {
                    this.gameOver = true;
                    if (this.isAI) {
                        document.getElementById('aiGameOver').style.display = 'block';
                        document.getElementById('aiFinalScore').textContent = this.score;
                    } else {
                        document.getElementById('playerGameOver').style.display = 'block';
                        document.getElementById('playerFinalScore').textContent = this.score;
                    }
                }
                
                this.drawNextPiece();
            }
            
            collision() {
                for (let y = 0; y < this.currentPiece.shape.length; y++) {
                    for (let x = 0; x < this.currentPiece.shape[y].length; x++) {
                        if (this.currentPiece.shape[y][x] !== 0) {
                            const boardX = this.currentPiece.x + x;
                            const boardY = this.currentPiece.y + y;
                            
                            if (boardX < 0 || boardX >= COLS || boardY >= ROWS) {
                                return true;
                            }
                            
                            if (boardY >= 0 && this.board[boardY][boardX] !== 0) {
                                return true;
                            }
                        }
                    }
                }
                return false;
            }
            
            merge() {
                for (let y = 0; y < this.currentPiece.shape.length; y++) {
                    for (let x = 0; x < this.currentPiece.shape[y].length; x++) {
                        if (this.currentPiece.shape[y][x] !== 0) {
                            const boardX = this.currentPiece.x + x;
                            const boardY = this.currentPiece.y + y;
                            
                            if (boardY >= 0) {
                                this.board[boardY][boardX] = this.currentPiece.color;
                            }
                        }
                    }
                }
            }
            
            rotate() {
                const rotated = [];
                const rows = this.currentPiece.shape.length;
                const cols = this.currentPiece.shape[0].length;
                
                for (let i = 0; i < cols; i++) {
                    rotated.push([]);
                    for (let j = rows - 1; j >= 0; j--) {
                        rotated[i].push(this.currentPiece.shape[j][i]);
                    }
                }
                
                const prevShape = this.currentPiece.shape;
                this.currentPiece.shape = rotated;
                
                if (this.collision()) {
                    this.currentPiece.shape = prevShape;
                }
            }
            
            move(dir) {
                this.currentPiece.x += dir;
                
                if (this.collision()) {
                    this.currentPiece.x -= dir;
                    return false;
                }
                
                return true;
            }
            
            drop() {
                this.currentPiece.y++;
                
                if (this.collision()) {
                    this.currentPiece.y--;
                    this.merge();
                    this.clearLines();
                    this.spawnPiece();
                    return false;
                }
                
                this.dropCounter = 0;
                return true;
            }
            
            hardDrop() {
                while (this.drop()) {}
            }
            
            clearLines() {
                let linesCleared = 0;
                
                for (let y = ROWS - 1; y >= 0; y--) {
                    if (this.board[y].every(cell => cell !== 0)) {
                        this.board.splice(y, 1);
                        this.board.unshift(Array(COLS).fill(0));
                        linesCleared++;
                        y++; // 重新检查当前行
                    }
                }
                
                if (linesCleared > 0) {
                    this.lines += linesCleared;
                    this.score += linesCleared * 100 * this.level;
                    this.level = Math.floor(this.lines / 10) + 1;
                    this.dropInterval = Math.max(100, 1000 - (this.level - 1) * 100);
                    
                    // 更新UI
                    if (this.isAI) {
                        document.getElementById('aiScore').textContent = this.score;
                        document.getElementById('aiLines').textContent = this.lines;
                        document.getElementById('aiLevel').textContent = this.level;
                    } else {
                        document.getElementById('playerScore').textContent = this.score;
                        document.getElementById('playerLines').textContent = this.lines;
                        document.getElementById('playerLevel').textContent = this.level;
                    }
                }
            }
            
            update(time = 0) {
                if (this.gameOver) return;
                
                const deltaTime = time - this.lastTime;
                this.lastTime = time;
                
                this.dropCounter += deltaTime;
                
                if (this.dropCounter > this.dropInterval) {
                    this.drop();
                }
                
                // AI逻辑
                if (this.isAI) {
                    this.aiMoveTimer += deltaTime;
                    if (this.aiMoveTimer > this.aiMoveInterval) {
                        this.makeAIMove();
                        this.aiMoveTimer = 0;
                    }
                }
            }
            
            makeAIMove() {
                // 简单的AI策略：尝试将方块放在最可能消除行的位置
                const bestMove = this.findBestMove();
                
                if (bestMove) {
                    this.currentPiece.x = bestMove.x;
                    this.currentPiece.y = bestMove.y;
                    this.currentPiece.rotation = bestMove.rotation;
                    
                    // 应用旋转
                    const rotations = bestMove.rotation;
                    const originalShape = this.currentPiece.shape;
                    
                    for (let i = 0; i < rotations; i++) {
                        this.rotate();
                    }
                    
                    // 如果碰撞，尝试移动
                    if (this.collision()) {
                        this.currentPiece.x = originalShape.x;
                        this.currentPiece.shape = originalShape;
                    }
                }
                
                // 执行下落
                this.drop();
            }
            
            findBestMove() {
                const moves = [];
                
                // 尝试所有可能的x位置和旋转
                for (let x = 0; x < COLS; x++) {
                    for (let rotation = 0; rotation < 4; rotation++) {
                        const move = {
                            x: x,
                            y: 0,
                            rotation: rotation,
                            score: 0
                        };
                        
                        // 创建临时棋盘
                        const tempBoard = this.board.map(row => [...row]);
                        const tempPiece = {
                            ...this.currentPiece,
                            x: x,
                            shape: this.getRotatedShape(this.currentPiece.shape, rotation)
                        };
                        
                        // 模拟放置方块
                        let y = 0;
                        while (!this.checkCollisionAt(tempBoard, tempPiece, y + 1)) {
                            y++;
                        }
                        
                        // 评估这个位置
                        move.y = y;
                        move.score = this.evaluateMove(tempBoard, tempPiece, y);
                        
                        moves.push(move);
                    }
                }
                
                // 找到最佳移动
                moves.sort((a, b) => b.score - a.score);
                return moves[0];
            }
            
            getRotatedShape(shape, rotations) {
                let rotated = shape;
                for (let i = 0; i < rotations; i++) {
                    rotated = this.rotateShape(rotated);
                }
                return rotated;
            }
            
            rotateShape(shape) {
                const rotated = [];
                const rows = shape.length;
                const cols = shape[0].length;
                
                for (let i = 0; i < cols; i++) {
                    rotated.push([]);
                    for (let j = rows - 1; j >= 0; j--) {
                        rotated[i].push(shape[j][i]);
                    }
                }
                
                return rotated;
            }
            
            checkCollisionAt(board, piece, y) {
                for (let py = 0; py < piece.shape.length; py++) {
                    for (let px = 0; px < piece.shape[py].length; px++) {
                        if (piece.shape[py][px] !== 0) {
                            const boardX = piece.x + px;
                            const boardY = piece.y + py + y;
                            
                            if (boardX < 0 || boardX >= COLS || boardY >= ROWS) {
                                return true;
                            }
                            
                            if (boardY >= 0 && board[boardY][boardX] !== 0) {
                                return true;
                            }
                        }
                    }
                }
                return false;
            }
            
            evaluateMove(board, piece, y) {
                // 创建临时棋盘
                const tempBoard = board.map(row => [...row]);
                
                // 放置方块
                for (let py = 0; py < piece.shape.length; py++) {
                    for (let px = 0; px < piece.shape[py].length; px++) {
                        if (piece.shape[py][px] !== 0) {
                            const boardX = piece.x + px;
                            const boardY = piece.y + py + y;
                            
                            if (boardY >= 0) {
                                tempBoard[boardY][boardX] = piece.color;
                            }
                        }
                    }
                }
                
                // 评估分数
                let score = 0;
                
                // 1. 检查可消除的行
                let linesCleared = 0;
                for (let row = ROWS - 1; row >= 0; row--) {
                    if (tempBoard[row].every(cell => cell !== 0)) {
                        linesCleared++;
                        score += 100;
                    }
                }
                
                // 2. 奖励堆叠高度较低的位置
                const heights = this.getColumnHeights(tempBoard);
                const maxHeight = Math.max(...heights);
                score += (ROWS - maxHeight) * 10;
                
                // 3. 奖励更平整的表面
                const roughness = this.calculateRoughness(heights);
                score -= roughness * 5;
                
                // 4. 奖励创建"洞"较少的位置
                const holes = this.countHoles(tempBoard);
                score -= holes * 20;
                
                return score;
            }
            
            getColumnHeights(board) {
                const heights = Array(COLS).fill(0);
                
                for (let x = 0; x < COLS; x++) {
                    for (let y = ROWS - 1; y >= 0; y--) {
                        if (board[y][x] !== 0) {
                            heights[x] = ROWS - y;
                            break;
                        }
                    }
                }
                
                return heights;
            }
            
            calculateRoughness(heights) {
                let roughness
