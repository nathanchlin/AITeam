        // 游戏常量
        const COLS = 10;
        const ROWS = 20;
        const BLOCK_SIZE = 30;
        const INITIAL_SPEED = 1000; // 初始下落速度（毫秒）
        const SPEED_INCREMENT = 50; // 每级速度减少量
        const LINE_SCORE = 100; // 消除一行的基础分数
        const LEVEL_THRESHOLD = 10; // 每消除10行升一级
        
        // 方块形状定义
        const SHAPES = [
            [[1, 1, 1, 1]], // I
            [[1, 1], [1, 1]], // O
            [[0, 1, 0], [1, 1, 1]], // T
            [[1, 0, 0], [1, 1, 1]], // L
            [[0, 0, 1], [1, 1, 1]], // J
            [[0, 1, 1], [1, 1, 0]], // S
            [[1, 1, 0], [0, 1, 1]]  // Z
        ];

        const COLORS = [

            '#f0f000', // O - 黄色

            '#f0a000', // L - 橙色

                this.offscreenCanvas.width = BLOCK_SIZE;

            initBoard() {
                this.board = Array(ROWS).fill().map(() => Array(COLS).fill(0));

            renderBlockToOffscreen(shapeIndex, color) {
                this.offscreenCtx.clearRect(0, 0, BLOCK_SIZE, BLOCK_SIZE);
                this.offscreenCtx.fillStyle = color;
                
                const shape = SHAPES[shapeIndex];
                const shapeWidth = shape[0].length;
                const shapeHeight = shape.length;
                
                const blockSize = BLOCK_SIZE / Math.max(shapeWidth, shapeHeight);
                const offsetX = (BLOCK_SIZE - shapeWidth * blockSize) / 2;
                const offsetY = (BLOCK_SIZE - shapeHeight * blockSize) / 2;
                
                for (let y = 0; y < shapeHeight; y++) {
                    for (let x = 0; x < shapeWidth; x++) {
                        if (shape[y][x]) {
                            this.offscreenCtx.fillRect(
                                offsetX + x * blockSize,
                                offsetY + y * blockSize,
                                blockSize - 1,
                                blockSize - 1
                            );
                        }
                    }
                }
                
                return this.offscreenCanvas;
            }
            
            // 创建新方块
            createPiece() {
                const shapeIndex = Math.floor(Math.random() * SHAPES.length);
                this.currentShapeIndex = shapeIndex;
                this.currentPiece = SHAPES[shapeIndex];
                this.currentX = Math.floor((COLS - this.currentPiece[0].length) / 2);
                this.currentY = 0;
                
                // 检查游戏是否结束
                if (this.collision()) {
                    this.gameOver = true;
                    return false;

                return true;

            // 碰撞检测
            collision() {
                for (let y = 0; y < this.currentPiece.length; y++) {
                    for (let x = 0; x < this.currentPiece[y].length; x++) {
                        if (this.currentPiece[y][x]) {
                            const boardX = this.currentX + x;
                            const boardY = this.currentY + y;
                            
                            if (
                                boardX < 0 || 
                                boardX >= COLS || 
                                boardY >= ROWS ||
                                (boardY >= 0 && this.board[boardY][boardX])
                            ) {
                                return true;
                            }
                        }
                    }
                }
                return false;
            }
            
            // 旋转方块
            rotate() {
                const rotated = this.currentPiece[0].map((_, i) =>
                    this.currentPiece.map(row => row[i]).reverse()
                );
                
                const previousPiece = this.currentPiece;
                this.currentPiece = rotated;
                
                if (this.collision()) {
                    this.currentPiece = previousPiece;
                }
            }
            
            // 移动方块
            move(dir) {
                this.currentX += dir;
                if (this.collision()) {
                    this.currentX -= dir;
                    return false;
                }
                return true;
            }
            
            // 硬降（直接落到底部）
            hardDrop() {
                while (this.move(1)) {
                    this.score += 2;
                }
                this.drop();
            }
            
            // 方块下落
            drop() {
                this.currentY++;
                if (this.collision()) {
                    this.currentY--;
                    this.merge();
                    this.clearLines();
                    if (!this.createPiece()) {
                        return false;
                    }
                }
                return true;
            }
            
            // 合并方块到游戏板
            merge() {
                for (let y = 0; y < this.currentPiece.length; y++) {
                    for (let x = 0; x < this.currentPiece[y].length; x++) {
                        if (this.currentPiece[y][x]) {
                            const boardY = this.currentY + y;
                            if (boardY >= 0) {
                                this.board[boardY][this.currentX + x] = this.currentShapeIndex + 1;
                            }
                        }
                    }
                }
            }
            
            // 清除完整的行
            clearLines() {
                let linesCleared = 0;
                
                for (let y = ROWS - 1; y >= 0; y--) {
                    if (this.board[y].every(cell => cell !== 0)) {
                        this.board.splice(y, 1);
                        this.board.unshift(Array(COLS).fill(0));
                        linesCleared++;
                        y++; // 重新检查当前行，因为上面的行下移了
                    }
                }
                
                if (linesCleared > 0) {
                    // 更新分数
                    this.lines += linesCleared;
                    this.score += linesCleared * LINE_SCORE * this.level;
                    
                    // 更新等级
                    const newLevel = Math.floor(this.lines / LEVEL_THRESHOLD) + 1;
                    if (newLevel > this.level) {
                        this.level = newLevel;
                        this.dropInterval = Math.max(100, INITIAL_SPEED - (this.level - 1) * SPEED_INCREMENT);
                    }
                    
                    // 更新UI
                    this.updateUI();
                }
            }

            updateUI() {

            }

            render() {
                // 清空画布
                this.ctx.fillStyle = '#000';

                // 绘制已固定的方块
                for (let y = 0; y < ROWS; y++) {
                    for (let x = 0; x < COLS; x++) {
                        if (this.board[y][x]) {
                            const blockX = x * BLOCK_SIZE;
                            const blockY = y * BLOCK_SIZE;
                            const color = COLORS[this.board[y][x] - 1];
                            
                            // 使用预渲染的方块
                            const blockCanvas = this.renderBlockToOffscreen(this.board[y][x] - 1, color);
                            this.ctx.drawImage(blockCanvas, blockX, blockY);
                        }
                    }
                }
                
                // 绘制当前方块
                if (this.currentPiece) {
                    const color = COLORS[this.currentShapeIndex];
                    
                    for (let y = 0; y < this.currentPiece.length; y++) {
                        for (let x = 0; x < this.currentPiece[y].length; x++) {
                            if (this.currentPiece[y][x]) {
                                const blockX = (this.currentX + x) * BLOCK_SIZE;
                                const blockY = (this.currentY + y) * BLOCK_SIZE;
                                
                                // 使用预渲染的方块
                                const blockCanvas = this.renderBlockToOffscreen(this.currentShapeIndex, color);
                                this.ctx.drawImage(blockCanvas, blockX, blockY);
                            }
                        }
                    }
                }
            }
            
            // 绘制网格线
            drawGrid() {
                this.ctx.strokeStyle = '#222';
                this.ctx.lineWidth = 0.5;
                
                // 垂直线
                for (let x = 1; x < COLS; x++) {
                    this.ctx.beginPath();
                    this.ctx.moveTo(x * BLOCK_SIZE, 0);
                    this.ctx.lineTo(x * BLOCK_SIZE, this.canvas.height);
                    this.ctx.stroke();
                }
                
                // 水平线
                for (let y = 1; y < ROWS; y++) {
                    this.ctx.beginPath();
                    this.ctx.moveTo(0, y * BLOCK_SIZE);
                    this.ctx.lineTo(this.canvas.width, y * BLOCK_SIZE);
                    this.ctx.stroke();
                }
            }
            
            // 更新游戏状态
            update(deltaTime) {
                if (this.gameOver) return;
                
                this.dropCounter += deltaTime;
                if (this.dropCounter > this.dropInterval) {
                    this.drop();
                    this.dropCounter = 0;
                }
            }
            
            // 游戏结束
            endGame() {
                this.gameOver = true;
                const gameOverElement = this.isAI ? document.getElementById('aiGameOver') : document.getElementById('playerGameOver');
                const finalScoreElement = this.isAI ? document.getElementById('aiFinalScore') : document.getElementById('playerFinalScore');
                
                finalScoreElement.textContent = this.score;
                gameOverElement.style.display = 'block';
            }
