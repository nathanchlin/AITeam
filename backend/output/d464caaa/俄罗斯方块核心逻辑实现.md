    <script>
        // 游戏常量
        const COLS = 10;
        const ROWS = 20;
        const BLOCK_SIZE = 30;
        const COLORS = [
            '#FF0D72', '#0DC2FF', '#0DFF72', '#F538FF',
            '#FF8E0D', '#FFE138', '#3877FF'
        ];
        
        // 方块形状定义
        const SHAPES = [
            [[1, 1, 1, 1]], // I
            [[1, 1], [1, 1]], // O
            [[1, 1, 1], [0, 1, 0]], // T
            [[1, 1, 1], [1, 0, 0]], // L
            [[1, 1, 1], [0, 0, 1]], // J
            [[0, 1, 1], [1, 1, 0]], // S
            [[1, 1, 0], [0, 1, 1]]  // Z
        ];
        
        // 方块类
        class Piece {
            constructor(shape, color) {
                this.shape = shape;
                this.color = color;
                this.x = Math.floor(COLS / 2) - Math.floor(shape[0].length / 2);
                this.y = 0;
            }
            
            rotate() {
                const rotated = [];
                const rows = this.shape.length;
                const cols = this.shape[0].length;
                
                for (let i = 0; i < cols; i++) {
                    rotated[i] = [];
                    for (let j = rows - 1; j >= 0; j--) {
                        rotated[i][rows - 1 - j] = this.shape[j][i];
                    }
                }
                
                return rotated;
            }
        }
        
        // 游戏板类
        class Board {
            constructor(canvasId, isAI = false) {
                this.canvas = document.getElementById(canvasId);
                this.ctx = this.canvas.getContext('2d');
                this.grid = Array(ROWS).fill().map(() => Array(COLS).fill(0));
                this.currentPiece = null;
                this.score = 0;
                this.lines = 0;
                this.level = 1;
                this.isAI = isAI;
                this.gameOver = false;
                this.dropInterval = 1000;
                this.lastDropTime = 0;
                this.gameOverElement = document.getElementById(isAI ? 'aiGameOver' : 'playerGameOver');
                this.scoreElement = document.getElementById(isAI ? 'aiScore' : 'playerScore');
                this.linesElement = document.getElementById(isAI ? 'aiLines' : 'playerLines');
                this.levelElement = document.getElementById(isAI ? 'aiLevel' : 'playerLevel');
                this.finalScoreElement = document.getElementById(isAI ? 'aiFinalScore' : 'playerFinalScore');
                
                if (!isAI) {
                    this.setupControls();
                }
            }
            
            setupControls() {
                document.addEventListener('keydown', (e) => {
                    if (this.gameOver) return;
                    
                    switch(e.key) {
                        case 'ArrowLeft':
                            this.movePiece(-1, 0);
                            break;
                        case 'ArrowRight':
                            this.movePiece(1, 0);
                            break;
                        case 'ArrowDown':
                            this.movePiece(0, 1);
                            break;
                        case 'ArrowUp':
                            this.rotatePiece();
                            break;
                        case ' ':
                            this.hardDrop();
                            break;
                    }
                });
            }
            
            start() {
                this.spawnPiece();
                this.gameLoop();
            }
            
            gameLoop() {
                if (this.gameOver) return;
                
                const now = Date.now();
                if (now - this.lastDropTime > this.dropInterval) {
                    this.movePiece(0, 1);
                    this.lastDropTime = now;
                }
                
                this.draw();
                
                if (this.isAI) {
                    this.aiMove();
                }
                
                requestAnimationFrame(() => this.gameLoop());
            }
            
            spawnPiece() {
                const randomIndex = Math.floor(Math.random() * SHAPES.length);
                const shape = SHAPES[randomIndex];
                const color = COLORS[randomIndex];
                this.currentPiece = new Piece(shape, color);
                
                if (this.collision()) {
                    this.endGame();
                }
            }
            
            collision() {
                for (let y = 0; y < this.currentPiece.shape.length; y++) {
                    for (let x = 0; x < this.currentPiece.shape[y].length; x++) {
                        if (this.currentPiece.shape[y][x] !== 0) {
                            const newX = this.currentPiece.x + x;
                            const newY = this.currentPiece.y + y;
                            
                            if (newX < 0 || newX >= COLS || newY >= ROWS) {
                                return true;
                            }
                            
                            if (newY >= 0 && this.grid[newY][newX] !== 0) {
                                return true;
                            }
                        }
                    }
                }
                return false;
            }
            
            movePiece(dx, dy) {
                this.currentPiece.x += dx;
                this.currentPiece.y += dy;
                
                if (this.collision()) {
                    this.currentPiece.x -= dx;
                    this.currentPiece.y -= dy;
                    
                    if (dy > 0) {
                        this.lockPiece();
                    }
                }
            }
            
            rotatePiece() {
                const originalShape = this.currentPiece.shape;
                this.currentPiece.shape = this.currentPiece.rotate();
                
                if (this.collision()) {
                    this.currentPiece.shape = originalShape;
                }
            }
            
            hardDrop() {
                while (!this.collision()) {
                    this.currentPiece.y++;
                }
                this.currentPiece.y--;
                this.lockPiece();
            }
            
            lockPiece() {
                for (let y = 0; y < this.currentPiece.shape.length; y++) {
                    for (let x = 0; x < this.currentPiece.shape[y].length; x++) {
                        if (this.currentPiece.shape[y][x] !== 0) {
                            const gridX = this.currentPiece.x + x;
                            const gridY = this.currentPiece.y + y;
                            
                            if (gridY >= 0) {
                                this.grid[gridY][gridX] = this.currentPiece.color;
                            }
                        }
                    }
                }
                
                this.clearLines();
                this.spawnPiece();
            }
            
            clearLines() {
                let linesCleared = 0;
                
                for (let y = ROWS - 1; y >= 0; y--) {
                    if (this.grid[y].every(cell => cell !== 0)) {
                        this.grid.splice(y, 1);
                        this.grid.unshift(Array(COLS).fill(0));
                        linesCleared++;
                        y++; // 检查同一行，因为数组已改变
                    }
                }
                
                if (linesCleared > 0) {
                    this.lines += linesCleared;
                    this.score += linesCleared * 100 * this.level;
                    this.level = Math.floor(this.lines / 10) + 1;
                    this.dropInterval = Math.max(100, 1000 - (this.level - 1) * 100);
                    
                    this.updateScore();
                }
            }
            
            updateScore() {
                this.scoreElement.textContent = this.score;
                this.linesElement.textContent = this.lines;
                this.levelElement.textContent = this.level;
            }
            
            endGame() {
                this.gameOver = true;
                this.finalScoreElement.textContent = this.score;
                this.gameOverElement.style.display = 'block';
            }
            
            draw() {
                // 清空画布
                this.ctx.fillStyle = '#0f0f1e';
                this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);
                
                // 绘制网格
                this.ctx.strokeStyle = '#1a1a2e';
                this.ctx.lineWidth = 0.5;
                
                for (let i = 0; i <= COLS; i++) {
                    this.ctx.beginPath();
                    this.ctx.moveTo(i * BLOCK_SIZE, 0);
                    this.ctx.lineTo(i * BLOCK_SIZE, ROWS * BLOCK_SIZE);
                    this.ctx.stroke();
                }
                
                for (let i = 0; i <= ROWS; i++) {
                    this.ctx.beginPath();
                    this.ctx.moveTo(0, i * BLOCK_SIZE);
                    this.ctx.lineTo(COLS * BLOCK_SIZE, i * BLOCK_SIZE);
                    this.ctx.stroke();
                }
                
                // 绘制已锁定的方块
                for (let y = 0; y < ROWS; y++) {
                    for (let x = 0; x < COLS; x++) {
                        if (this.grid[y][x] !== 0) {
                            this.drawBlock(x, y, this.grid[y][x]);
                        }
                    }
                }
                
                // 绘制当前方块
                if (this.currentPiece) {
                    for (let y = 0; y < this.currentPiece.shape.length; y++) {
                        for (let x = 0; x < this.currentPiece.shape[y].length; x++) {
                            if (this.currentPiece.shape[y][x] !== 0) {
                                this.drawBlock(
                                    this.currentPiece.x + x,
                                    this.currentPiece.y + y,
                                    this.currentPiece.color
                                );
                            }
                        }
                    }
                }
            }
            
            drawBlock(x, y, color) {
                this.ctx.fillStyle = color;
                this.ctx.fillRect(x * BLOCK_SIZE, y * BLOCK_SIZE, BLOCK_SIZE, BLOCK_SIZE);
                
                // 添加高光效果
                this.ctx.fillStyle = 'rgba(255, 255, 255, 0.3)';
                this.ctx.fillRect(x * BLOCK_SIZE, y * BLOCK_SIZE, BLOCK_SIZE, 4);
                this.ctx.fillRect(x * BLOCK_SIZE, y * BLOCK_SIZE, 4, BLOCK_SIZE);
                
                // 添加阴影效果
                this.ctx.fillStyle = 'rgba(0, 0, 0, 0.3)';
                this.ctx.fillRect(x * BLOCK_SIZE + BLOCK_SIZE - 4, y * BLOCK_SIZE, 4, BLOCK_SIZE);
                this.ctx.fillRect(x * BLOCK_SIZE, y * BLOCK_SIZE + BLOCK_SIZE - 4, BLOCK_SIZE, 4);
            }
            
            // AI相关方法
            aiMove() {
                if (!this.currentPiece || this.gameOver) return;
                
                // 简单的AI策略：找到最佳位置
                const bestMove = this.findBestMove();
                
                // 执行移动
                this.currentPiece.x = bestMove.x;
                this.currentPiece.y = bestMove.y;
                this.currentPiece.shape = bestMove.shape;
                
                // 如果碰撞，锁定方块
                if (this.collision()) {
                    this.currentPiece.y--;
                    this.lockPiece();
                }
            }
            
            findBestMove() {
                const moves = [];
                const originalPiece = this.currentPiece;
                
                // 尝试所有可能的旋转
                for (let rotation = 0; rotation < 4; rotation++) {
                    // 尝试所有可能的x位置
                    for (let x = 0; x < COLS; x++) {
                        const testPiece = new Piece(
                            originalPiece.shape,
                            originalPiece.color
                        );
                        
                        // 应用旋转
                        for (let i = 0; i < rotation; i++) {
                            testPiece.shape = testPiece.rotate();
                        }
                        
                        testPiece.x = x;
                        testPiece.y = 0;
                        
                        // 找到最低位置
                        while (!this.collision()) {
                            testPiece.y++;
                        }
                        testPiece.y--;
                        
                        // 计算分数
                        const score = this.evaluatePosition(testPiece);
                        moves.push({
                            x: testPiece.x,
                            y: testPiece.y,
                            shape: testPiece.shape,
                            score: score
                        });
                    }
                }
                
                // 返回最佳移动
                moves.sort((a, b) => b.score - a.score);
                return moves[0];
            }
            
            evaluatePosition(piece) {
                // 简单的评估函数
                let score = 0;
                
                // 创建临时网格
                const tempGrid = this.grid.map(row => [...row]);
                
                // 将方块放置在临时网格中
                for (let y = 0; y < piece.shape.length; y++) {
                    for (let x = 0; x < piece.shape[y].length; x++) {
                        if (piece.shape[y][x] !== 0) {
                            const gridX = piece.x + x;
                            const gridY = piece.y + y;
                            
                            if (gridY >= 0) {
                                tempGrid[gridY][gridX] = piece.color;
                            }
                        }
                    }
                }
                
                // 评估行
                let linesCleared = 0;
                for (let y = ROWS - 1; y >= 0; y--) {
                    if (tempGrid[y].every(cell => cell !== 0)) {
                        linesCleared++;
                        score += 100;
                    }
                }
                
                // 评估高度（越低越好）
                let maxHeight = 0;
                for (let x = 0; x < COLS; x++) {
                    for (let y = 0; y < ROWS; y++) {
                        if (tempGrid[y][x] !== 0) {
                            maxHeight = Math.max(maxHeight, ROWS -
