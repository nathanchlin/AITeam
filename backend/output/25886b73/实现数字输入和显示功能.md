### JavaScript 功能实现

实现数字输入和显示的核心逻辑：

```javascript
document.addEventListener('DOMContentLoaded', () => {
    const display = document.getElementById('display');
    let currentValue = '0';
    let previousValue = '';
    let operation = null;
    let shouldResetDisplay = false;
    
    // 更新显示屏
    function updateDisplay() {
        // 限制显示长度，防止溢出
        if (currentValue.length > 12) {
            currentValue = parseFloat(currentValue).toExponential(6);
        }
        display.textContent = currentValue;
    }
    
    // 处理数字输入
    function inputNumber(value) {
        if (shouldResetDisplay) {
            currentValue = '';
            shouldResetDisplay = false;
        }
        
        if (currentValue === '0' && value !== '.') {
            currentValue = value;
        } else {
            currentValue += value;
        }
        updateDisplay();
    }
    
    // 处理小数点
    function inputDecimal() {
        if (shouldResetDisplay) {
            currentValue = '0';
            shouldResetDisplay = false;
        }
        
        if (!currentValue.includes('.')) {
            currentValue += '.';
            updateDisplay();
        }
    }
    
    // 处理运算符
    function inputOperator(op) {
        if (operation !== null && !shouldResetDisplay) {
            calculate();
        }
        
        previousValue = currentValue;
        operation = op;
        shouldResetDisplay = true;
    }
    
    // 计算结果
    function calculate() {
        if (operation === null || shouldResetDisplay) {
            return;
        }
        
        let prev = parseFloat(previousValue);
        let current = parseFloat(currentValue);
        let result;
        
        switch (operation) {
            case '+':
                result = prev + current;
                break;
            case '-':
                result = prev - current;
                break;
            case '*':
                result = prev * current;
                break;
            case '/':
                if (current === 0) {
                    alert('不能除以零！');
                    return;
                }
                result = prev / current;
                break;
            default:
                return;
        }
        
        currentValue = String(result);
        operation = null;
        previousValue = '';
        shouldResetDisplay = true;
        updateDisplay();
    }
    
    // 清除计算器
    function clear() {
        currentValue = '0';
        previousValue = '';
        operation = null;
        shouldResetDisplay = false;
        updateDisplay();
    }
    
    // 添加按钮事件监听
    document.querySelectorAll('.btn').forEach(button => {
        button.addEventListener('click', () => {
            const value = button.getAttribute('data-value');
            
            if (button.classList.contains('number')) {
                inputNumber(value);
            } else if (button.classList.contains('decimal')) {
                inputDecimal();
            } else if (button.classList.contains('operator')) {
                inputOperator(value);
            } else if (button.classList.contains('equals')) {
                calculate();
            } else if (button.classList.contains('clear')) {
                clear();
            }
        });
    });
    
    // 键盘支持
    document.addEventListener('keydown', (e) => {
        if (e.key >= '0' && e.key <= '9' || e.key === '.') {
            inputNumber(e.key);
        } else if (e.key === '+' || e.key === '-' || e.key === '*' || e.key === '/') {
            inputOperator(e.key);
        } else if (e.key === 'Enter' || e.key === '=') {
            calculate();
        } else if (e.key === 'Escape' || e.key === 'c' || e.key === 'C') {
            clear();
        }
    });
});
```

## 功能说明

1. **数字输入功能**：
   - 支持0-9的数字输入
   - 自动处理前导零
   - 支持小数点输入，防止重复输入小数点

2. **显示功能**：
   - 实时更新显示内容
   - 限制显示长度，防止溢出
   - 长数字自动转换为科学计数法

3. **额外功能**：
   - 键盘支持，可以使用键盘输入数字和运算符
   - 清除功能，重置计算器状态
   - 运算符处理，支持基本的加减乘除运算

4. **用户体验优化**：
   - 按钮悬停和点击效果
   - 不同类型按钮的视觉区分
   - 响应式设计，适应不同屏幕尺寸

## 实现细节

1. **状态管理**：
   - 使用`currentValue`存储当前输入的数字
   - 使用`previousValue`存储运算前的数字
   - 使用`operation`存储当前运算符
   - 使用`shouldResetDisplay`标志控制是否需要重置显示

2. **输入处理**：
   - 数字输入时自动处理前导零
   - 小数点输入时检查是否已存在小数点
   - 运算符输入时保存当前值并准备下一次运算

3. **计算逻辑**：
   - 支持基本的四则运算
   - 处理除以零的错误情况
   - 计算后自动重置状态
