### 2. JavaScript 输入验证和错误处理实现

以下是完整的 JavaScript 代码，包含全面的输入验证和错误处理：

```javascript
// 获取DOM元素
const display = document.getElementById('display');
const errorMessage = document.getElementById('errorMessage');

// 清除错误信息
function clearError() {
    errorMessage.textContent = '';
}

// 显示错误信息
function showError(message) {
    errorMessage.textContent = message;
    setTimeout(clearError, 5000); // 5秒后自动清除错误信息
}

// 验证输入是否为有效数字
function isValidNumber(input) {
    // 检查是否为空
    if (input === '') {
        return false;
    }
    
    // 检查是否为有效数字格式
    const numberRegex = /^-?\d*\.?\d+$/;
    return numberRegex.test(input);
}

// 验证表达式是否有效
function isValidExpression(expression) {
    // 检查表达式是否为空
    if (expression === '') {
        return false;
    }
    
    // 检查表达式是否以运算符开头或结尾
    if (/^[+\-*/.]|[+\-*/.]$/.test(expression)) {
        return false;
    }
    
    // 检查是否有连续的运算符
    if (/[+\-*/]{2,}/.test(expression)) {
        return false;
    }
    
    // 检查是否有多个小数点
    const parts = expression.split(/[+\-*/]/);
    for (const part of parts) {
        if (part.split('.').length > 2) {
            return false;
        }
    }
    
    return true;
}

// 添加字符到显示
function appendToDisplay(value) {
    clearError();
    
    // 如果当前显示为0或错误状态，则替换而不是追加
    if (display.value === '0' || display.value === 'Error') {
        display.value = value;
    } else {
        display.value += value;
    }
}

// 清除显示
function clearDisplay() {
    display.value = '';
    clearError();
}

// 删除最后一个字符
function deleteLast() {
    clearError();
    if (display.value.length > 0) {
        display.value = display.value.slice(0, -1);
    }
}

// 计算表达式
function calculate() {
    try {
        const expression = display.value;
        
        // 验证表达式
        if (!isValidExpression(expression)) {
            throw new Error('无效的表达式');
        }
        
        // 检查是否以运算符结尾
        if (/^[+\-*/]$/.test(expression)) {
            throw new Error('表达式不能以运算符结尾');
        }
        
        // 检查除以零的情况
        if (expression.includes('/0')) {
            throw new Error('不能除以零');
        }
        
        // 检查除法运算符后是否为空
        if (expression.match(/\/\s*$/)) {
            throw new Error('除法运算符后必须有数字');
        }
        
        // 检查是否有无效的运算符组合
        if (/[+\-*/]{2,}/.test(expression)) {
            throw new Error('无效的运算符组合');
        }
        
        // 尝试计算表达式
        const result = eval(expression);
        
        // 检查结果是否为有效数字
        if (!isValidNumber(result.toString())) {
            throw new Error('计算结果无效');
        }
        
        // 显示结果
        display.value = result;
    } catch (error) {
        // 处理各种错误情况
        if (error instanceof SyntaxError) {
            showError('语法错误: ' + error.message);
            display.value = 'Error';
        } else if (error.message === 'Cannot divide by zero') {
            showError('错误: 不能除以零');
            display.value = 'Error';
        } else {
            showError('错误: ' + error.message);
            display.value = 'Error';
        }
    }
}

// 键盘事件处理
document.addEventListener('keydown', function(event) {
    // 防止输入非数字和运算符
    if (event.key.match(/[0-9+\-*/.=]|Backspace|Delete|Enter/)) {
        if (event.key === 'Enter') {
            calculate();
        } else if (event.key === 'Backspace' || event.key === 'Delete') {
            deleteLast();
        } else {
            appendToDisplay(event.key);
        }
        event.preventDefault();
    }
});
```

## 验证和错误处理功能详解

### 1. 输入验证功能

- **数字验证**：`isValidNumber()` 函数确保输入的是有效数字，包括整数、小数和负数
- **表达式验证**：`isValidExpression()` 函数检查整个表达式是否有效，包括：
  - 防止空表达式
  - 防止以运算符开头或结尾
  - 防止连续运算符
  - 防止数字中有多个小数点

### 2. 错误处理机制

- **除以零检测**：在计算前检查表达式是否包含除以零的情况
- **语法错误处理**：使用 try-catch 捕获 eval() 可能抛出的语法错误
- **运算符有效性检查**：确保运算符使用正确，没有无效组合
- **结果验证**：确保计算结果是有效数字

### 3. 用户反馈

- **错误信息显示**：在错误信息区域显示具体的错误原因
- **自动清除错误**：错误信息会在5秒后自动消失
- **键盘事件处理**：限制键盘输入只能输入有效字符

### 4. 防御性编程

- **防止空输入**：确保用户不能提交空表达式
- **防止无效运算**：检查运算符使用是否正确
- **防止计算溢出**：通过 try-catch 捕获可能的计算错误

## 最佳实践总结

1. **防御性编程**：始终验证所有输入，不要假设用户会正确使用程序
2. **明确的错误信息**：提供清晰、具体的错误信息，帮助用户理解问题所在
3. **优雅的错误处理**：不要让程序崩溃，而是捕获异常并提供有意义的反馈
4. **输入限制**：限制用户只能输入有效字符，防止无效输入
5. **用户友好**：提供直观的界面和反馈，提升用户体验

这个实现方案确保了计算器能够处理各种异常情况，并提供良好的用户体验。通过全面的输入验证和错误处理，计算器将更加健壮和可靠。
