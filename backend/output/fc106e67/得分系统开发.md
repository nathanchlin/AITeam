## 系统设计

### 核心功能

2. 分数显示界面
3. 连击系统
4. 特殊奖励机制
5. 最高分记录

## 代码实现

```python
class ScoreSystem:
    def __init__(self):
        self.current_score = 0
        self.high_score = 0
        self.combo_multiplier = 1
        self.combo_timer = 0
        self.combo_threshold = 2.0  # 2秒内需要击中下一个目标保持连击
        self.distance_score = 0
        self.kill_score = 0
        self.bonus_score = 0
        self.font = None  # 将在游戏初始化时设置
        
    def update(self, dt):
        """更新分数系统，包括连击计时器"""
        # 更新连击计时器
        if self.combo_timer > 0:
            self.combo_timer -= dt
            if self.combo_timer <= 0:
                self.combo_multiplier = 1  # 重置连击倍数
                
    def add_distance_score(self, distance):
        """根据移动距离增加分数"""
        points = int(distance * 10)  # 每10单位距离得1分
        self.distance_score += points
        self._update_total_score(points)
        
    def add_kill_score(self, enemy_type):
        """根据击败的敌人类型增加分数"""
        # 不同敌人类型有不同分值
        enemy_scores = {
            "normal": 10,
            "fast": 20,
            "strong": 30,
            "boss": 100
        }
        
        points = enemy_scores.get(enemy_type, 5)  # 默认5分
        self.kill_score += points * self.combo_multiplier
        self._update_total_score(points * self.combo_multiplier)
        
        # 增加连击倍数和时间
        self.combo_multiplier = min(self.combo_multiplier + 0.5, 5)  # 最大5倍
        self.combo_timer = self.combo_threshold
        
    def add_bonus_score(self, bonus_type):
        """添加奖励分数"""
        bonus_scores = {
            "coin": 50,
            "time_bonus": 200,
            "perfect_dodge": 100,
            "level_complete": 1000
        }
        
        points = bonus_scores.get(bonus_type, 0)
        self.bonus_score += points
        self._update_total_score(points)
        
    def _update_total_score(self, points):
        """更新总分并检查最高分"""
        self.current_score += points
        
        # 检查是否创造新纪录
        if self.current_score > self.high_score:
            self.high_score = self.current_score
            
    def reset(self):
        """重置当前游戏分数"""
        self.current_score = 0
        self.combo_multiplier = 1
        self.combo_timer = 0
        self.distance_score = 0
        self.kill_score = 0
        self.bonus_score = 0
        
    def draw(self, screen):
        """在屏幕上绘制分数信息"""
        if self.font:
            # 绘制当前分数
            score_text = self.font.render(f"Score: {self.current_score}", True, (255, 255, 255))
            screen.blit(score_text, (10, 10))
            
            # 绘制最高分
            high_score_text = self.font.render(f"High Score: {self.high_score}", True, (255, 255, 255))
            screen.blit(high_score_text, (10, 50))
            
            # 绘制连击倍数
            if self.combo_multiplier > 1:
                combo_text = self.font.render(f"Combo: x{self.combo_multiplier:.1f}", True, (255, 255, 0))
                screen.blit(combo_text, (10, 90))
                
                # 绘制连击进度条
                bar_width = 100
                bar_height = 10
                bar_x = 10
                bar_y = 130
                
                # 背景条
                pygame.draw.rect(screen, (100, 100, 100), (bar_x, bar_y, bar_width, bar_height))
                
                # 进度条
                progress = self.combo_timer / self.combo_threshold
                pygame.draw.rect(screen, (255, 255, 0), (bar_x, bar_y, int(bar_width * progress), bar_height))
```

## 使用示例

```python
# 在游戏主循环中使用
score_system = ScoreSystem()
score_system.font = pygame.font.Font(None, 36)  # 设置字体

# 游戏主循环
clock = pygame.time.Clock()
running = True

while running:
    dt = clock.tick(60) / 1000.0  # 转换为秒
    
    # 处理事件...
    
    # 更新分数系统
    score_system.update(dt)
    
    # 根据游戏事件添加分数
    # 例如：忍者移动了50个单位
    score_system.add_distance_score(50)
    
    # 例如：击败了一个普通敌人
    score_system.add_kill_score("normal")
    
    # 绘制分数
    score_system.draw(screen)
    
    # 更新显示...
    pygame.display.flip()
```
