# 忍者角色控制系统实现

我将为您实现一个完整的忍者角色控制系统，包含移动、跳跃、滑行等基本跑酷动作。这个系统将采用面向对象的设计，确保代码的可维护性和扩展性。

## 角色控制系统设计

### 1. 基础架构

```python
import pygame
import math

class Ninja:
    def __init__(self, x, y, game_width, game_height):
        # 初始化位置和尺寸
        self.x = x
        self.y = y
        self.width = 40
        self.height = 60
        self.game_width = game_width
        self.game_height = game_height
        
        # 移动属性
        self.velocity_x = 0
        self.velocity_y = 0
        self.speed = 5
        self.jump_power = 15
        self.gravity = 0.8
        
        # 状态管理
        self.is_jumping = False
        self.is_sliding = False
        self.is_facing_right = True
        
        # 动画相关
        self.animation_frame = 0
        self.animation_timer = 0
        self.animation_speed = 5
        
        # 滑行相关
        self.slide_duration = 0
        self.slide_max_duration = 30  # 滑行持续帧数
        self.slide_cool_down = 0
        
        # 地面高度
        self.ground_y = game_height - 100
        
        # 设置初始位置在地面
        self.y = self.ground_y - self.height
```

### 2. 移动控制系统

```python
    def move_left(self):
        """向左移动"""
        self.velocity_x = -self.speed
        self.is_facing_right = False
        if self.is_sliding:
            self.velocity_x *= 1.5  # 滑行时移动速度增加
            
    def move_right(self):
        """向右移动"""
        self.velocity_x = self.speed
        self.is_facing_right = True
        if self.is_sliding:
            self.velocity_x *= 1.5  # 滑行时移动速度增加
            
    def stop_horizontal_movement(self):
        """停止水平移动"""
        self.velocity_x = 0
```

### 3. 跳跃系统

```python
    def jump(self):
        """执行跳跃动作"""
        if not self.is_jumping and not self.is_sliding:
            self.velocity_y = -self.jump_power
            self.is_jumping = True
            
    def update_jump(self):
        """更新跳跃状态"""
        if self.is_jumping:
            self.velocity_y += self.gravity
            self.y += self.velocity_y
            
            # 检查是否落地
            if self.y >= self.ground_y - self.height:
                self.y = self.ground_y - self.height
                self.is_jumping = False
                self.velocity_y = 0
```

### 4. 滑行系统

```python
    def slide(self):
        """执行滑行动作"""
        if not self.is_jumping and not self.is_sliding and self.slide_cool_down <= 0:
            self.is_sliding = True
            self.slide_duration = self.slide_max_duration
            self.height = 30  # 滑行时高度减半
            
    def update_slide(self):
        """更新滑行状态"""
        if self.is_sliding:
            self.slide_duration -= 1
            
            # 检查滑行是否结束
            if self.slide_duration <= 0:
                self.is_sliding = False
                self.height = 60  # 恢复原始高度
                self.slide_cool_down = 60  # 设置滑行冷却时间
                
        # 更新冷却时间
        if self.slide_cool_down > 0:
            self.slide_cool_down -= 1
```

### 5. 更新和渲染系统

```python
    def update(self):
        """更新忍者状态"""
        # 更新位置
        self.x += self.velocity_x
        
        # 边界检查
        if self.x < 0:
            self.x = 0
        elif self.x > self.game_width - self.width:
            self.x = self.game_width - self.width
            
        # 更新跳跃状态
        self.update_jump()
        
        # 更新滑行状态
        self.update_slide()
        
        # 更新动画
        self.update_animation()
        
    def update_animation(self):
        """更新动画帧"""
        self.animation_timer += 1
        if self.animation_timer >= self.animation_speed:
            self.animation_timer = 0
            self.animation_frame = (self.animation_frame + 1) % 4
            
    def draw(self, screen):
        """绘制忍者角色"""
        # 根据状态选择颜色
        if self.is_sliding:
            color = (0, 150, 0)  # 滑行时颜色稍深
        else:
            color = (0, 255, 0)  # 正常绿色
            
        # 绘制忍者身体
        pygame.draw.rect(screen, color, (self.x, self.y, self.width, self.height))
        
        # 绘制眼睛（指示朝向）
        eye_y = self.y + 15
        if self.is_facing_right:
            pygame.draw.circle(screen, (255, 255, 255), (self.x + 25, eye_y), 5)
            pygame.draw.circle(screen, (0, 0, 0), (self.x + 27, eye_y), 2)
        else:
            pygame.draw.circle(screen, (255, 255, 255), (self.x + 15, eye_y), 5)
            pygame.draw.circle(screen, (0, 0, 0), (self.x + 13, eye_y), 2)
        
        # 绘制动画效果（简单的移动线条）
        if not self.is_sliding:
            line_offset = math.sin(self.animation_frame * math.pi / 2) * 5
            pygame.draw.line(screen, (0, 200, 0), 
                            (self.x + self.width // 2, self.y + self.height),
                            (self.x + self.width // 2 + line_offset, self.y + self.height + 10), 2)
```

### 6. 输入处理系统

```python
class NinjaController:
    def __init__(self, ninja):
        self.ninja = ninja
        
    def handle_input(self, keys):
        """处理键盘输入"""
        # 左右移动
        if keys[pygame.K_LEFT] or keys[pygame.K_a]:
            self.ninja.move_left()
        elif keys[pygame.K_RIGHT] or keys[pygame.K_d]:
            self.ninja.move_right()
        else:
            self.ninja.stop_horizontal_movement()
            
        # 跳跃
        if keys[pygame.K_SPACE] or keys[pygame.K_UP] or keys[pygame.K_w]:
            self.ninja.jump()
            
        # 滑行
        if keys[pygame.K_DOWN] or keys[pygame.K_s]:
            self.ninja.slide()
```

### 7. 完整游戏示例

```python
def main():
    # 初始化Pygame
    pygame.init()
    
    # 设置窗口
    WIDTH, HEIGHT = 800, 600
    screen = pygame.display.set_mode((WIDTH, HEIGHT))
    pygame.display.set_caption("忍者必须死 - 跑酷游戏")
    clock = pygame.time.Clock()
    
    # 创建忍者
    ninja = Ninja(100, HEIGHT - 160, WIDTH, HEIGHT)
    controller = NinjaController(ninja)
    
    # 游戏主循环
    running = True
    while running:
        # 事件处理
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                running = False
                
        # 获取键盘状态
        keys = pygame.key.get_pressed()
        
        # 处理输入
        controller.handle_input(keys)
        
        # 更新忍者
        ninja.update()
        
        # 绘制
        screen.fill((135, 206, 235))  # 天蓝色背景
        
        # 绘制地面
        pygame.draw.rect(screen, (100, 100, 100), (0, HEIGHT - 100, WIDTH, 100))
        
        # 绘制忍者
        ninja.draw(screen)
        
        # 显示操作提示
        font = pygame.font.SysFont(None, 24)
        instructions = [
            "操作说明:",
            "←/A: 向左移动",
            "→/D: 向右移动",
            "空格/↑/W: 跳跃",
            "↓/S: 滑行"
        ]
        for i, text in enumerate(instructions):
            text_surface = font.render(text, True, (255, 255, 255))
            screen.blit(text_surface, (10, 10 + i * 25))
        
        # 更新显示
        pygame.display.flip()
        clock.tick(60)
    
    pygame.quit()

if __name__ == "__main__":
    main()
```
