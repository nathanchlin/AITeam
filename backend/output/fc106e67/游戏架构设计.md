```python
# 游戏主控制器
class Game:
    def __init__(self):
        self.screen = Screen()
        self.clock = Clock()
        self.running = True
        self.score = 0
        self.game_state = "MENU"  # MENU, PLAYING, GAME_OVER
        
        # 初始化各模块
        self.player = Player()
        self.obstacle_manager = ObstacleManager()
        self.collision_detector = CollisionDetector()
        self.renderer = Renderer()
        
    def run(self):
        while self.running:
            self.handle_events()
            self.update()
            self.render()
            self.clock.tick(60)  # 60 FPS
            
    def handle_events(self):
        # 处理用户输入和系统事件
        pass
        
    def update(self):
        # 更新游戏逻辑
        if self.game_state == "PLAYING":
            self.player.update()
            self.obstacle_manager.update()
            if self.collision_detector.check_collision(self.player, self.obstacle_manager):
                self.game_state = "GAME_OVER"
                
    def render(self):
        # 渲染游戏画面
        self.screen.clear()
        if self.game_state == "PLAYING":
            self.renderer.render_game(self.player, self.obstacle_manager, self.score)
        elif self.game_state == "MENU":
            self.renderer.render_menu()
        elif self.game_state == "GAME_OVER":
            self.renderer.render_game_over(self.score)
        self.screen.update()
```

```python
class Player:
    def __init__(self):
        self.x = 100  # 初始位置
        self.y = 300
        self.width = 40
        self.height = 60
        self.color = (0, 255, 0)  # 绿色忍者
        self.velocity_y = 0
        self.is_jumping = False
        self.is_sliding = False
        self.is_attacking = False
        self.animation_frame = 0
        self.animation_timer = 0
        
    def update(self):
        # 处理重力
        if self.y < 300:  # 地面高度
            self.velocity_y += 0.8  # 重力加速度
        else:
            self.y = 300
            self.velocity_y = 0
            self.is_jumping = False
            
        self.y += self.velocity_y
        
        # 更新动画
        self.animation_timer += 1
        if self.animation_timer > 10:  # 每10帧更新一次动画
            self.animation_frame = (self.animation_frame + 1) % 4
            self.animation_timer = 0
            
    def jump(self):
        if not self.is_jumping:
            self.velocity_y = -15  # 跳跃初速度
            self.is_jumping = True
            
    def slide(self):
        if not self.is_sliding and not self.is_jumping:
            self.is_sliding = True
            self.height = 30  # 滑铲时高度减半
            # 设置滑铲持续时间
            self.slide_timer = 30
            
    def attack(self):
        if not self.is_attacking:
            self.is_attacking = True
            self.attack_timer = 20  # 攻击持续20帧
            
    def get_rect(self):
        return pygame.Rect(self.x, self.y, self.width, self.height)
        
    def draw(self, screen):
        # 根据状态绘制不同的忍者动画
        if self.is_sliding:
            # 绘制滑铲状态
            pygame.draw.rect(screen, self.color, 
                           (self.x, self.y + 30, self.width, self.height))
        elif self.is_attacking:
            # 绘制攻击状态
            pygame.draw.rect(screen, self.color, 
                           (self.x - 20, self.y, self.width + 20, self.height))
        else:
            # 绘制正常状态
            pygame.draw.rect(screen, self.color, 
                           (self.x, self.y, self.width, self.height))
```

## 3. 障碍物生成模块

```python
import random

class Obstacle:
    def __init__(self, x, y, width, height, obstacle_type):
        self.x = x
        self.y = y
        self.width = width
        self.height = height
        self.type = obstacle_type  # "GROUND", "AIR", "SLIDE"
        self.color = (255, 0, 0)  # 红色障碍物
        self.passed = False
        
    def update(self, speed):
        self.x -= speed  # 向左移动
        
    def draw(self, screen):
        if self.type == "GROUND":
            pygame.draw.rect(screen, self.color, (self.x, self.y, self.width, self.height))
        elif self.type == "AIR":
            pygame.draw.rect(screen, self.color, (self.x, self.y - 100, self.width, self.height))
        elif self.type == "SLIDE":
            pygame.draw.rect(screen, self.color, (self.x, self.y + 30, self.width, self.height))
            
    def get_rect(self):
        return pygame.Rect(self.x, self.y, self.width, self.height)

class ObstacleManager:
    def __init__(self):
        self.obstacles = []
        self.spawn_timer = 0
        self.spawn_interval = 60  # 初始生成间隔
        self.speed = 5  # 初始速度
        
    def update(self):
        # 更新所有障碍物
        for obstacle in self.obstacles[:]:
            obstacle.update(self.speed)
            
            # 移除屏幕外的障碍物
            if obstacle.x + obstacle.width < 0:
                self.obstacles.remove(obstacle)
                if not obstacle.passed:
                    # 玩家成功通过障碍物，增加分数
                    self.score += 1
                    
        # 生成新障碍物
        self.spawn_timer += 1
        if self.spawn_timer >= self.spawn_interval:
            self.spawn_obstacle()
            self.spawn_timer = 0
            
            # 随着游戏进行，增加难度
            if self.spawn_interval > 30:
                self.spawn_interval -= 0.1
            if self.speed < 10:
                self.speed += 0.01
                
    def spawn_obstacle(self):
        # 随机生成不同类型的障碍物
        obstacle_type = random.choice(["GROUND", "AIR", "SLIDE"])
        
        if obstacle_type == "GROUND":
            x = 800  # 从屏幕右侧生成
            y = 300  # 地面高度
            width = random.randint(30, 60)
            height = random.randint(40, 80)
        elif obstacle_type == "AIR":
            x = 800
            y = 300
            width = random.randint(40, 70)
            height = random.randint(40, 60)
        elif obstacle_type == "SLIDE":
            x = 800
            y = 300
            width = random.randint(50, 80)
            height = random.randint(20, 40)
            
        self.obstacles.append(Obstacle(x, y, width, height, obstacle_type))
        
    def draw(self, screen):
        for obstacle in self.obstacles:
            obstacle.draw(screen)
```

## 4. 碰撞检测模块

```python
class CollisionDetector:
    def __init__(self):
        pass
        
    def check_collision(self, player, obstacle_manager):
        player_rect = player.get_rect()
        
        for obstacle in obstacle_manager.obstacles:
            obstacle_rect = obstacle.get_rect()
            
            # 根据玩家状态调整碰撞检测
            if player.is_sliding:
                # 滑铲时检测区域降低
                player_rect = pygame.Rect(player.x, player.y + 30, player.width, player.height)
            elif player.is_attacking:
                # 攻击时检测区域扩大
                player_rect = pygame.Rect(player.x - 20, player.y, player.width + 20, player.height)
                
            # 简单的矩形碰撞检测
            if player_rect.colliderect(obstacle_rect):
                # 根据障碍物类型进行特殊处理
                if obstacle.type == "SLIDE" and not player.is_sliding:
                    return True
                elif obstacle.type == "AIR" and player.is_sliding:
                    return True
                elif obstacle.type == "GROUND":
                    return True
                    
        return False
```

```python
class InputHandler:
    def __init__(self, player):
        self.player = player
        
    def handle_input(self, event):
        if event.type == pygame.KEYDOWN:
            if event.key == pygame.K_SPACE:
                self.player.jump()
            elif event.key == pygame.K_DOWN:
                self.player.slide()
            elif event.key == pygame.K_x:
                self.player.attack()
```
