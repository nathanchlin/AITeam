# 游戏框架搭建：忍者必须死的跑酷游戏

我将为您创建一个基础的HTML5 Canvas游戏框架，用于开发"忍者必须死的跑酷游戏"。这个框架将包含游戏循环、渲染系统和基本结构。

## 完整代码实现

```html
<!DOCTYPE html>
<html lang="zh-CN">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>忍者必须死 - 跑酷游戏</title>
    <style>
        body {
            margin: 0;
            padding: 0;
            background-color: #111;
            display: flex;
            justify-content: center;
            align-items: center;
            height: 100vh;
            font-family: 'Arial', sans-serif;
            overflow: hidden;
        }
        
        #gameContainer {
            position: relative;
            width: 800px;
            height: 400px;
            border: 2px solid #333;
            box-shadow: 0 0 20px rgba(0, 255, 0, 0.3);
        }
        
        #gameCanvas {
            background-color: #222;
            display: block;
        }
        
        #gameInfo {
            position: absolute;
            top: 10px;
            left: 10px;
            color: #0f0;
            font-size: 16px;
            text-shadow: 0 0 5px rgba(0, 255, 0, 0.5);
        }
        
        #startScreen {
            position: absolute;
            top: 0;
            left: 0;
            width: 100%;
            height: 100%;
            background-color: rgba(0, 0, 0, 0.8);
            display: flex;
            flex-direction: column;
            justify-content: center;
            align-items: center;
            color: #0f0;
            z-index: 10;
        }
        
        #startScreen h1 {
            font-size: 36px;
            margin-bottom: 20px;
            text-shadow: 0 0 10px rgba(0, 255, 0, 0.8);
        }
        
        #startButton {
            padding: 10px 20px;
            font-size: 18px;
            background-color: #0f0;
            color: #000;
            border: none;
            border-radius: 5px;
            cursor: pointer;
            transition: all 0.3s;
        }
        
        #startButton:hover {
            background-color: #0a0;
            box-shadow: 0 0 15px rgba(0, 255, 0, 0.7);
        }
    </style>
</head>
<body>
    <div id="gameContainer">
        <canvas id="gameCanvas" width="800" height="400"></canvas>
        <div id="gameInfo">
            <div>得分: <span id="score">0</span></div>
            <div>生命: <span id="lives">3</span></div>
        </div>
        <div id="startScreen">
            <h1>忍者必须死</h1>
            <button id="startButton">开始游戏</button>
        </div>
    </div>

    <script>
        // 游戏配置
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        const gameInfo = {
            score: document.getElementById('score'),
            lives: document.getElementById('lives')
        };
        const startScreen = document.getElementById('startScreen');
        const startButton = document.getElementById('startButton');
        
        // 游戏状态
        const game = {
            running: false,
            score: 0,
            lives: 3,
            speed: 5,
            gravity: 0.5,
            frameCount: 0
        };
        
        // 忍者对象
        const ninja = {
            x: 100,
            y: 200,
            width: 40,
            height: 60,
            velocityY: 0,
            jumping: false,
            color: '#0f0', // 绿色忍者
            draw: function() {
                ctx.fillStyle = this.color;
                // 绘制忍者身体
                ctx.fillRect(this.x, this.y, this.width, this.height);
                
                // 绘制忍者头部
                ctx.beginPath();
                ctx.arc(this.x + this.width/2, this.y - 10, 15, 0, Math.PI * 2);
                ctx.fill();
                
                // 绘制忍者眼睛
                ctx.fillStyle = '#000';
                ctx.beginPath();
                ctx.arc(this.x + this.width/2 - 5, this.y - 12, 3, 0, Math.PI * 2);
                ctx.arc(this.x + this.width/2 + 5, this.y - 12, 3, 0, Math.PI * 2);
                ctx.fill();
                
                // 绘制忍者武器
                ctx.strokeStyle = '#fff';
                ctx.lineWidth = 3;
                ctx.beginPath();
                ctx.moveTo(this.x + this.width, this.y + 20);
                ctx.lineTo(this.x + this.width + 20, this.y + 10);
                ctx.stroke();
            },
            update: function() {
                // 应用重力
                this.velocityY += game.gravity;
                this.y += this.velocityY;
                
                // 地面碰撞检测
                const groundY = canvas.height - 100 - this.height;
                if (this.y > groundY) {
                    this.y = groundY;
                    this.velocityY = 0;
                    this.jumping = false;
                }
            },
            jump: function() {
                if (!this.jumping) {
                    this.velocityY = -12;
                    this.jumping = true;
                }
            }
        };
        
        // 地面对象
        const ground = {
            y: canvas.height - 100,
            height: 100,
            draw: function() {
                ctx.fillStyle = '#333';
                ctx.fillRect(0, this.y, canvas.width, this.height);
                
                // 绘制地面纹理
                ctx.strokeStyle = '#555';
                ctx.lineWidth = 1;
                for (let i = 0; i < canvas.width; i += 20) {
                    ctx.beginPath();
                    ctx.moveTo(i, this.y);
                    ctx.lineTo(i + 10, this.y + 10);
                    ctx.stroke();
                }
            }
        };
        
        // 障碍物数组
        let obstacles = [];
        
        // 创建障碍物
        function createObstacle() {
            const height = Math.random() * 40 + 30;
            obstacles.push({
                x: canvas.width,
                y: ground.y - height,
                width: 30,
                height: height,
                color: '#f00',
                passed: false
            });
        }
        
        // 绘制障碍物
        function drawObstacles() {
            obstacles.forEach(obstacle => {
                ctx.fillStyle = obstacle.color;
                ctx.fillRect(obstacle.x, obstacle.y, obstacle.width, obstacle.height);
                
                // 添加危险标记
                ctx.fillStyle = '#ff0';
                ctx.font = '20px Arial';
                ctx.fillText('!', obstacle.x + 10, obstacle.y - 10);
            });
        }
        
        // 更新障碍物
        function updateObstacles() {
            // 移动障碍物
            obstacles = obstacles.filter(obstacle => {
                obstacle.x -= game.speed;
                
                // 检测是否通过障碍物
                if (!obstacle.passed && obstacle.x + obstacle.width < ninja.x) {
                    obstacle.passed = true;
                    game.score += 10;
                    gameInfo.score.textContent = game.score;
                }
                
                // 碰撞检测
                if (ninja.x < obstacle.x + obstacle.width &&
                    ninja.x + ninja.width > obstacle.x &&
                    ninja.y < obstacle.y + obstacle.height &&
                    ninja.y + ninja.height > obstacle.y) {
                    game.lives--;
                    gameInfo.lives.textContent = game.lives;
                    
                    // 重置障碍物位置
                    obstacle.x = canvas.width;
                    obstacle.passed = false;
                    
                    if (game.lives <= 0) {
                        endGame();
                    }
                }
                
                return obstacle.x > -obstacle.width;
            });
            
            // 随机生成新障碍物
            if (game.frameCount % 100 === 0) {
                createObstacle();
            }
        }
        
        // 绘制背景
        function drawBackground() {
            // 天空渐变
            const gradient = ctx.createLinearGradient(0, 0, 0, canvas.height);
            gradient.addColorStop(0, '#1a1a2e');
            gradient.addColorStop(1, '#16213e');
            ctx.fillStyle = gradient;
            ctx.fillRect(0, 0, canvas.width, canvas.height);
            
            // 绘制月亮
            ctx.fillStyle = '#fff';
            ctx.beginPath();
            ctx.arc(700, 50, 30, 0, Math.PI * 2);
            ctx.fill();
            
            // 绘制星星
            ctx.fillStyle = '#fff';
            for (let i = 0; i < 50; i++) {
                const x = Math.random() * canvas.width;
                const y = Math.random() * canvas.height * 0.7;
                const size = Math.random() * 2;
                ctx.fillRect(x, y, size, size);
            }
        }
        
        // 游戏循环
        function gameLoop() {
            if (!game.running) return;
            
            // 清除画布
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            // 绘制背景
            drawBackground();
            
            // 更新和绘制地面
            ground.draw();
            
            // 更新和绘制忍者
            ninja.update();
            ninja.draw();
            
            // 更新和绘制障碍物
            updateObstacles();
            drawObstacles();
            
            // 增加游戏难度
            if (game.frameCount % 500 === 0) {
                game.speed += 0.5;
            }
            
            game.frameCount++;
            requestAnimationFrame(gameLoop);
        }
        
        // 开始游戏
        function startGame() {
            game.running = true;
            game.score = 0;
            game.lives = 3;
            game.speed = 5;
            game.frameCount = 0;
            
            gameInfo.score.textContent = game.score;
            gameInfo.lives.textContent = game.lives;
            
            obstacles = [];
            ninja.y = 200;
            ninja.velocityY = 0;
            ninja.jumping = false;
            
            startScreen.style.display = 'none';
            gameLoop();
        }
        
        // 结束游戏
        function endGame() {
            game.running = false;
            startScreen.style.display = 'flex';
            startButton.textContent = '重新开始';
            
            // 显示游戏结束信息
            const gameOverText = document.createElement('div');
            gameOverText.style.fontSize = '24px';
            gameOverText.style.marginTop = '20px';
            gameOverText.textContent = `游戏结束! 最终得分: ${game.score}`;
            startScreen.insertBefore(gameOverText, startButton);
            
            // 移除游戏结束文本
            setTimeout(() => {
                startScreen.removeChild(gameOverText);
            }, 5000);
        }
        
        // 键盘控制
        document.addEventListener('keydown', (e) => {
            if (e.code === 'Space' && game.running) {
                e.preventDefault();
                ninja.jump();
            }
        });
        
        // 触摸控制（移动设备）
        canvas.addEventListener('touchstart', (e) => {
            e.preventDefault();
            if (game.running) {
                ninja.jump();
            }
        });
        
        // 开始按钮事件
        startButton.addEventListener('click', startGame);
    </script>
</body>
</html>
```

- 实现了跳跃物理系统，包括重力和地面碰撞检测

#### 障碍物系统
- 随机生成不同高度的障碍物

- 随着游戏进行增加难度
