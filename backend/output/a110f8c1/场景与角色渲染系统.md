        // ==================== 场景渲染器 ====================
        class SceneRenderer {
            constructor(game) {
                this.game = game;
                this.width = game.width;
                this.height = game.height;
                
                // 预生成云雾
                this.clouds = [];
                for (let i = 0; i < 8; i++) {
                    this.clouds.push({
                        x: Math.random() * this.width * 2,
                        y: this.height * 0.2 + Math.random() * this.height * 0.4,
                        size: 100 + Math.random() * 200,
                        speed: 0.02 + Math.random() * 0.03,
                        opacity: 0.03 + Math.random() * 0.05
                    });
                }
                
                // 预生成竹子数据
                this.bambooGroups = this.generateBambooGroups();
            }
            
            generateBambooGroups() {
                const groups = [];
                for (let layer = 0; layer < 3; layer++) {
                    const count = 8 + layer * 4;
                    for (let i = 0; i < count; i++) {
                        groups.push({
                            x: (i / count) * this.width * 2 + Math.random() * 100,
                            height: 200 + Math.random() * 300 + layer * 50,
                            segments: 5 + Math.floor(Math.random() * 4),
                            layer: layer,
                            lean: (Math.random() - 0.5) * 0.1
                        });
                    }
                }
                return groups;
            }
            
            render() {
                const ctx = this.game.ctx;
                const offset = this.game.parallaxOffset;
                
                // 绘制天空渐变
                this.drawSky(ctx);
                
                // 第1层 - 最远山
                this.drawMountainLayer(ctx, offset * 0.1, 0.15, CONFIG.COLORS.mountainFar, this.height * 0.35);
                
                // 第2层 - 远山
                this.drawMountainLayer(ctx, offset * 0.2, 0.25, CONFIG.COLORS.mountainMid, this.height * 0.45);
                
                // 第3层 - 中景山
                this.drawMountainLayer(ctx, offset * 0.4, 0.35, CONFIG.COLORS.mountainNear, this.height * 0.55);
                
                // 绘制云雾层
                this.drawMist(ctx, offset);
                
                // 第4层 - 远景竹林
                this.drawBambooForest(ctx, offset * 0.6, 0.6, 2);
                
                // 第5层 - 近景竹林
                this.drawBambooForest(ctx, offset * 0.8, 0.85, 0);
                
                // 绘制地面雾气
                this.drawGroundMist(ctx, offset);
            }
            
            drawSky(ctx) {
                const gradient = ctx.createLinearGradient(0, 0, 0, this.height);
                gradient.addColorStop(0, '#0a0a12');
                gradient.addColorStop(0.3, '#12121f');
                gradient.addColorStop(0.6, '#1a1a2a');
                gradient.addColorStop(1, '#101018');
                ctx.fillStyle = gradient;
                ctx.fillRect(0, 0, this.width, this.height);
                
                // 添加微妙的光晕
                const glowGradient = ctx.createRadialGradient(
                    this.width * 0.3, this.height * 0.2, 0,
                    this.width * 0.3, this.height * 0.2, this.width * 0.5
                );
                glowGradient.addColorStop(0, 'rgba(62, 205, 196, 0.03)');
                glowGradient.addColorStop(1, 'rgba(62, 205, 196, 0)');
                ctx.fillStyle = glowGradient;
                ctx.fillRect(0, 0, this.width, this.height);
            }
            
            drawMountainLayer(ctx, offset, heightRatio, color, baseY) {
                ctx.save();
                ctx.fillStyle = color;
                ctx.beginPath();
                
                const amplitude = this.height * heightRatio;
                let x = -offset % (this.width * 2) - this.width * 0.5;
                
                ctx.moveTo(x, this.height);
                
                // 绘制起伏的山脉
                const segments = 12;
                for (let i = 0; i <= segments * 3; i++) {
                    const px = x + (i / segments) * (this.width * 2 / 3);
                    const noise1 = Math.sin(i * 0.5 + offset * 0.01) * amplitude * 0.3;
                    const noise2 = Math.sin(i * 0.8 + offset * 0.02) * amplitude * 0.2;
                    const noise3 = Math.sin(i * 1.5) * amplitude * 0.1;
                    const py = baseY + noise1 + noise2 + noise3;
                    
                    if (i === 0) {
                        ctx.lineTo(px, py);
                    } else {
                        const cpx = x + ((i - 0.5) / segments) * (this.width * 2 / 3);
                        const cpy = baseY + Math.sin((i - 0.5) * 0.5 + offset * 0.01) * amplitude * 0.3;
                        ctx.quadraticCurveTo(cpx, cpy, px, py);
                    }
                }
                
                ctx.lineTo(x + this.width * 2, this.height);
                ctx.closePath();
                ctx.fill();
                ctx.restore();
            }
            
            drawMist(ctx, offset) {
                ctx.save();
                this.clouds.forEach(cloud => {
                    const x = (cloud.x - offset * cloud.speed) % (this.width * 2);
                    const adjustedX = x < 0 ? x + this.width * 2 : x;
                    
                    const gradient = ctx.createRadialGradient(
                        adjustedX, cloud.y, 0,
                        adjustedX, cloud.y, cloud.size
                    );
                    gradient.addColorStop(0, `rgba(220, 215, 200, ${cloud.opacity})`);
                    gradient.addColorStop(0.5, `rgba(220, 215, 200, ${cloud.opacity * 0.5})`);
                    gradient.addColorStop(1, 'rgba(220, 215, 200, 0)');
                    
                    ctx.fillStyle = gradient;
                    ctx.fillRect(adjustedX - cloud.size, cloud.y - cloud.size * 0.5, 
                                cloud.size * 2, cloud.size);
                });
                ctx.restore();
            }
            
            drawBambooForest(ctx, offset, groundY, layerIndex) {
                ctx.save();
                const bamboos = this.bambooGroups.filter(b => b.layer === layerIndex);
                const scale = 0.4 + layerIndex * 0.3;
                const alpha = 0.3 + layerIndex * 0.25;
                
                bamboos.forEach(bamboo => {
                    const x = bamboo.x - offset;
                    if (x < -100 || x > this.width + 100) return;
                    
                    const baseY = this.height * groundY;
                    const height = bamboo.height * scale;
                    
                    ctx.globalAlpha = alpha;
                    
                    // 绘制竹竿
                    this.drawBambooStalk(ctx, x, baseY, height, bamboo.segments, bamboo.lean, scale);
                });
                
                ctx.restore();
            }
            
            drawBambooStalk(ctx, x, baseY, height, segments, lean, scale) {
                const segmentHeight = height / segments;
                const baseWidth = 8 * scale;
                
                ctx.save();
                ctx.translate(x, baseY);
                
                // 绘制竹节
                for (let i = 0; i < segments; i++) {
                    const segY = -i * segmentHeight;
                    const segWidth = baseWidth * (1 - i * 0.05);
                    const leanOffset = lean * i * segmentHeight;
                    
                    // 竹节渐变
                    const gradient = ctx.createLinearGradient(
                        leanOffset - segWidth, segY,
                        leanOffset + segWidth, segY
                    );
                    gradient.addColorStop(0, '#1a3a2a');
                    gradient.addColorStop(0.3, '#2a5a4a');
                    gradient.addColorStop(0.7, '#2a5a4a');
                    gradient.addColorStop(1, '#1a3a2a');
                    
                    ctx.fillStyle = gradient;
                    
                    // 绘制竹节
                    ctx.beginPath();
                    ctx.ellipse(leanOffset, segY, segWidth, segmentHeight * 0.6, 0, 0, Math.PI * 2);
                    ctx.fill();
                    
                    // 绘制竹节环
                    if (i > 0) {
                        ctx.strokeStyle = '#3a6a5a';
                        ctx.lineWidth = 2 * scale;
                        ctx.beginPath();
                        ctx.ellipse(leanOffset, segY, segWidth * 1.1, 3 * scale, 0, 0, Math.PI * 2);
                        ctx.stroke();
                    }
                    
                    // 绘制竹叶
                    if (i > 0 && i < segments - 1 && Math.random() > 0.3) {
                        this.drawBambooLeaf(ctx, leanOffset + segWidth, segY - segmentHeight * 0.3, scale);
                        if (Math.random() > 0.5) {
                            this.drawBambooLeaf(ctx, leanOffset - segWidth, segY - segmentHeight * 0.5, -scale);
                        }
                    }
                }
                
                ctx.restore();
            }
            
            drawBambooLeaf(ctx, x, y, direction) {
                ctx.save();
                ctx.translate(x, y);
                ctx.scale(direction, 1);
                
                ctx.fillStyle = '#3a6a5a';
                ctx.beginPath();
                ctx.moveTo(0, 0);
                ctx.quadraticCurveTo(15, -5, 30, -2);
                ctx.quadraticCurveTo(15, 2, 0, 0);
                ctx.fill();
                
                ctx.restore();
            }
            
            drawGroundMist(ctx, offset) {
                const gradient = ctx.createLinearGradient(0, this.height - 100, 0, this.height);
                gradient.addColorStop(0, 'rgba(220, 215, 200, 0)');
                gradient.addColorStop(0.5, 'rgba(220, 215, 200, 0.05)');
                gradient.addColorStop(1, 'rgba(220, 215, 200, 0.1)');
                
                ctx.fillStyle = gradient;
                ctx.fillRect(0, this.height - 100, this.width, 100);
            }
        }
        
        // ==================== 角色类 ====================
        class Character {
            constructor(game, x, groundY) {
                this.game = game;
                this.x = x;
                this.groundY = groundY;
                this.y = groundY;
                
                this.state = 'idle';
                this.currentFrame = 0;
                this.frameTimer = 0;
                this.totalFrames = CONFIG.IDLE_FRAMES;
                this.frameDuration = CONFIG.FRAME_DURATION;
                
                this.breathPhase = 0;
                this.direction = 1; // 1: 右, -1: 左
                
                this.attackPhase = 0;
                this.attackRecovery = false;
                
                // 角色颜色
                this.colors = {
                    body: '#d4a54a',
                    bodyDark: '#a67c32',
                    eye: '#1a1a2a',
                    eyeHighlight: '#ffffff',
                    cheek: 'rgba(201, 74, 74, 0.4)',
                    aura: 'rgba(212, 165, 74, 0.3)',
                    attackGlow: 'rgba(62, 205, 196, 0.6)'
                };
            }
            
            setState(newState) {
                if (this.state === 'attack' && this.attackPhase < 0.9) {
                    return; // 攻击动画未完成不切换
                }
                
                this.state = newState;
                this.currentFrame = 0;
                this.frameTimer = 0;
                this.attackPhase = 0;
                this.attackRecovery = false;
                
                switch (newState) {
                    case 'idle':
                        this.totalFrames = CONFIG.IDLE_FRAMES;
                        this.frameDuration = CONFIG.FRAME_DURATION;
                        break;
                    case 'walk':
                        this.totalFrames = 4;
                        this.frameDuration = 100;
                        break;
                    case 'attack':
                        this.totalFrames = CONFIG.ATTACK_FRAMES;
                        this.frameDuration = 80;
                        break;
                }
            }
            
            update(deltaTime) {
                this.frameTimer += deltaTime;
                
                if (this.frameTimer >= this.frameDuration) {
                    this.frameTimer = 0;
                    this.currentFrame = (this.currentFrame + 1) % this.totalFrames;
                    
                    // 攻击动画结束
                    if (this.state === 'attack' && this.currentFrame === 0) {
                        this.setState('idle');
                    }
                }
                
                // 呼吸动画
                this.breathPhase += CONFIG.BREATH_SPEED * deltaTime;
                
                // 攻击阶段
                if (this.state === 'attack') {
                    this.attackPhase = (this.currentFrame + this.frameTimer / this.frameDuration) / this.totalFrames;
                }
                
                // 行走时角色朝向
                if (this.game.moveDirection !== 0) {
                    this.direction = this.game.moveDirection;
                }
            }
            
            render(ctx) {
                ctx.save();
                ctx.translate(this.x, this.y);
                ctx.scale(this.direction, 1);
                
                // 呼吸偏移
                const breathOffset = Math.sin(this.breathPhase) * CONFIG.BREATH_AMPLITUDE;
                const breathScale = 1 + Math.sin(this.breathPhase) * 0.02;
                
                ctx.translate(0, breathOffset);
                ctx.scale(1, breathScale);
                
                // 绘制光环
                this.drawAura(ctx);
                
                // 根据状态绘制
                switch (this.state) {
                    case 'idle':
                        this.drawIdle(ctx, this.currentFrame);
                        break;
                    case 'walk':
                        this.drawWalk(ctx, this.currentFrame);
                        break;
                    case 'attack':
                        this.drawAttack(ctx, this.currentFrame, this.attackPhase);
                        break;
                }
                
                ctx.restore();
            }
            
            drawAura(ctx) {
                // 底部光晕
                const gradient = ctx.createRadialGradient(0, 20, 0, 0, 20, 60);
                gradient.addColorStop(0, 'rgba(212, 165, 74, 0.2)');
                gradient.addColorStop(0.5, 'rgba(212, 165, 74, 0.1)');
                gradient.addColorStop(1, 'rgba(212, 165, 74, 0)');
                
                ctx.fillStyle = gradient;
                ctx.beginPath();
                ctx.ellipse(0, 20, 60, 20, 0, 0, Math.PI * 2);
                ctx.fill();
            }
            
            drawIdle(ctx, frame) {
                // 身体微微晃动
                const sway = Math.sin(frame * 0.5) * 1;
                
                ctx.save();
                ctx.translate(sway, 0);
                
                // 绘制身体
                this.drawBody(ctx, frame);
                
                // 绘制头部
                this.drawHead(ctx, frame);
                
                // 绘制耳朵
                this.drawEars(ctx, frame);
                
                ctx.restore();
            }
            
            drawWalk(ctx, frame) {
                // 行走动画 - 上下弹跳
                const bounce = Math.abs(Math.sin(frame * Math.PI / 2)) * 5;
                const legSwing = Math.sin(frame * Math.PI / 2) * 8;
                
                ctx.save();
                ctx.translate(0, -bounce);
                
                // 绘制身体
                this.drawBody(ctx, frame);
                
                // 绘制头部
                this.drawHead(ctx, frame);
                
                // 绘制耳朵（带摆动）
                this.drawEars(ctx, frame, legSwing * 0.3);
                
                ctx.restore();
            }
            
            drawAttack(ctx, frame, phase) {
                ctx.save();
                
                // 攻击前摇 - 蓄力
                if (phase < 0.3) {
                    const charge = phase / 0.3;
                    ctx.translate(-10 * charge, 5 * charge);
                    ctx.rotate(-0.2 * charge);
                    
                    // 蓄力特效
                    this.drawChargeEffect(ctx, charge);
                }
                // 攻击释放
                else if (phase < 0.6) {
                    const release = (phase - 0.3) / 0.3;
                    ctx.translate(20 * release, -10 * release);
                    ctx.rotate(0.3 * release);
                    
                    // 攻击特效
                    this.drawAttackEffect(ctx, release);
                }
                // 攻击后摇
                else {
                    const recovery = (phase - 0.6) / 0.4;
                    ctx.translate(20 * (1 - recovery), -10 * (1 - recovery));
                    ctx.rotate(0.3 * (1 - recovery));
                }
                
                // 绘制身体
                this.drawBody(ctx, frame);
                
                // 绘制头部
                this.drawHead(ctx, frame, phase >= 0.3 && phase < 0.6);
                
                // 绘制耳朵
                this.drawEars(ctx, frame);
                
                ctx.restore();
            }
            
            drawBody(ctx, frame) {
                const wobble = Math.sin(frame * 0.8) * 2;
                
                // 身体 - 椭圆形
                const bodyGradient = ctx.createRadialGradient(0, 0, 0, 0, 0, 35);
                bodyGradient.addColorStop(0, '#f0d890');
                bodyGradient.addColorStop(0.5, this.colors.body);
                bodyGradient.addColorStop(1, this.colors.bodyDark);
                
                ctx.fillStyle = bodyGradient;
                ctx.beginPath();
                ctx.ellipse(0, 0, 28 + wobble, 35, 0, 0, Math.PI * 2);
                ctx.fill();
                
                // 肚子高光
                ctx.fillStyle = 'rgba(255, 255, 255, 0.3)';
                ctx.beginPath();
                ctx.ellipse(-8, -10, 8, 12, -0.3, 0, Math.PI * 2);
                ctx.fill();
                
                // 小脚丫
                ctx.fillStyle = this.colors.bodyDark;
                ctx.beginPath();
                ctx.ellipse(-12, 32, 8, 5, 0, 0, Math.PI * 2);
                ctx.fill();
                ctx.beginPath();
                ctx.ellipse(12, 32, 8, 5, 0, 0, Math.PI * 2);
                ctx.fill();
                
                // 小手臂
                ctx.fillStyle = this.colors.body;
                ctx.beginPath();
                ctx.ellipse(-25, 5, 7, 10, 0.3, 0, Math.PI * 2);
                ctx.fill();
                ctx.beginPath();
                ctx.ellipse(25, 5, 7, 10, -0.3, 0, Math.PI * 2);
                ctx.fill();
            }
            
            drawHead(ctx, frame, isAttacking) {
                // 头部 - 稍大的圆形
                const headGradient = ctx.createRadialGradient(0, -45, 0, 0, -45, 30);
                headGradient.addColorStop(0, '#f0d890');
                headGradient.addColorStop(0.6, this.colors.body);
                headGradient.addColorStop(1, this.colors.bodyDark);
                
                ctx.fillStyle = headGradient;
                ctx.beginPath();
                ctx.arc(0, -45, 26, 0, Math.PI * 2);
                ctx.fill();
                
                // 腮红
                ctx.fillStyle = this.colors.cheek;
                ctx.beginPath();
                ctx.ellipse(-15, -40, 6, 4, 0, 0, Math.PI * 2);
                ctx.fill();
                ctx.beginPath();
                ctx.ellipse(15, -40, 6, 4, 0, 0, Math.PI * 2);
                ctx.fill();
                
                // 眼睛
                this.drawEyes(ctx, frame, isAttacking);
                
                // 嘴巴
                ctx.strokeStyle = this.colors.bodyDark;
                ctx.lineWidth = 2;
                ctx.lineCap = 'round';
                ctx.beginPath();
                
                if (isAttacking) {
                    // 攻击时张嘴
                    ctx.ellipse(0, -32, 4, 3, 0, 0, Math.PI * 2);
                    ctx.fillStyle = '#8a4a4a';
                    ctx.fill();
                } else {
                    // 微笑
                    ctx.arc(0, -35, 5, 0.2, Math.PI - 0.2);
                    ctx.stroke();
                }
            }
            
            drawEyes(ctx, frame, isAttacking) {
                const blinkPhase = (this.game.time % 3000) / 3000;
                const isBlinking = blinkPhase > 0.95;
                
                const eyeHeight = isBlinking ? 1 : 8;
                const eyeY = -48;
                
                // 左眼
                ctx.fillStyle = '#1a1a2a';
                if (isAttacking) {
                    // 攻击时眼睛变锐利
                    ctx.beginPath();
                    ctx.moveTo(-12, eyeY - 6);
                    ctx.lineTo(-6, eyeY);
                    ctx.lineTo(-12, eyeY + 6);
                    ctx.closePath();
                    ctx.fill();
                } else {
                    ctx.beginPath();
                    ctx.ellipse(-12, eyeY, 5, eyeHeight, 0, 0, Math.PI * 2);
                    ctx.fill();
                    
                    // 眼睛高光
                    if (!isBlinking) {
                        ctx.fillStyle = '#ffffff';
                        ctx.beginPath();
                        ctx.arc(-10, eyeY - 3, 2, 0, Math.PI * 2);
                        ctx.fill();
                    }
                }
                
                // 右眼
                ctx.fillStyle = '#1a1a2a';
                if (isAttacking) {
                    ctx.beginPath();
                    ctx.moveTo(12, eyeY - 6);
                    ctx.lineTo(6, eyeY);
                    ctx.lineTo(12, eyeY + 6);
                    ctx.closePath();
                    ctx.fill();
                } else {
                    ctx.beginPath();
                    ctx.ellipse(12, eyeY, 5, eyeHeight, 0, 0, Math.PI * 2);
                    ctx.fill();
                    
                    if (!isBlinking) {
                        ctx.fillStyle = '#ffffff';
                        ctx.beginPath();
                        ctx.arc(14, eyeY - 3, 2, 0, Math.PI * 2);
                        ctx.fill();
                    }
                }
            }
            
            drawEars(ctx, frame, swing) {
                swing = swing || 0;
                
                // 左耳 - 尖尖的精灵耳
                ctx.save();
                ctx.translate(-20, -60);
                ctx.rotate(-0.3 + swing * 0.02);
                
                const earGradient = ctx.createLinearGradient(0, 0, 0, -25);
                earGradient.addColorStop(0, this.colors.body);
                earGradient.addColorStop(1, this.colors.bodyDark);
                
                ctx.fillStyle = earGradient;
                ctx.beginPath();
                ctx.moveTo(0, 0);
                ctx.quadraticCurveTo(-5, -12, 0, -25);
                ctx.quadraticCurveTo(5, -12, 8, 0);
                ctx.closePath();
                ctx.fill();
                
                // 耳朵内部
                ctx.fillStyle = 'rgba(201, 74, 74, 0.3)';
                ctx.beginPath();
                ctx.moveTo(2, -2);
                ctx.quadraticCurveTo(-2, -10, 2, -18);
                ctx.quadraticCurveTo(5, -10, 5, -2);
                ctx.closePath();
                ctx.fill();
                
                ctx.restore();
                
                // 右耳
                ctx.save();
                ctx.translate(20, -60);
                ctx.scale(-1, 1);
                ctx.rotate(-0.3 - swing * 0.02);
                
                ctx.fillStyle = earGradient;
                ctx.beginPath();
                ctx.moveTo(0, 0);
                ctx.quadraticCurveTo(-5, -12, 0, -25);
                ctx.quadraticCurveTo(5, -12, 8, 0);
                ctx.closePath();
                ctx.fill();
                
                ctx.fillStyle = 'rgba(201, 74, 74, 0.3)';
                ctx.beginPath();
                ctx.moveTo(2, -2);
                ctx.quadraticCurveTo(-2, -10, 2, -18);
                ctx.quadraticCurveTo(5, -10, 5, -2);
                ctx.closePath();
                ctx.fill();
                
                ctx.restore();
            }
            
            drawChargeEffect(ctx, charge) {
                // 蓄力时的能量聚集效果
                const alpha = charge * 0.5;
                const gradient = ctx.createRadialGradient(0, 0, 0, 0, 0, 50 + charge * 30);
                gradient.addColorStop(0, `rgba(62, 205, 196, ${alpha})`);
                gradient.addColorStop(0.5, `rgba(62, 205, 196, ${alpha * 0.3})`);
                gradient.addColorStop(1, 'rgba(62, 205, 196, 0)');
                
                ctx.fillStyle = gradient;
                ctx.beginPath();
                ctx.arc(0, 0, 50 + charge * 30, 0, Math.PI * 2);
                ctx.fill();
                
                // 能量线条
                ctx.strokeStyle = `rgba(62, 205, 196, ${alpha})`;
                ctx.lineWidth = 2;
                for (let i = 0; i < 6; i++) {
                    const angle = (i / 6) * Math.PI * 2 + this.game.time * 0.005;
                    const r = 30 + charge * 20;
                    ctx.beginPath();
                    ctx.moveTo(Math.cos(angle) * 20, Math.sin(angle) * 20);
                    ctx.lineTo(Math.cos(angle) * r, Math.sin(angle) * r);
                    ctx.stroke();
                }
            }
            
            drawAttackEffect(ctx, release) {
                // 攻击释放特效
                const alpha = (1 - release) * 0.8;
                
                // 攻击弧线
                ctx.strokeStyle = `rgba(62, 205, 196, ${alpha})`;
                ctx.lineWidth = 4;
                ctx.lineCap = 'round';
                
                ctx.beginPath();
                ctx.arc(40, 0, 50, -Math.PI * 0.6, Math.PI * 0.2);
                ctx.stroke();
                
                // 攻击粒子
                ctx.fillStyle = `rgba(212, 165, 74, ${alpha})`;
                for (let i = 0; i < 5; i++) {
                    const angle = -Math.PI * 0.3 + (i / 4) * Math.PI * 0.5;
                    const dist = 40 + release * 60 + Math.random() * 20;
                    const size = 3 + Math.random() * 4;
                    ctx.beginPath();
                    ctx.arc(
                        Math.cos(angle) * dist,
                        Math.sin(angle) * dist,
                        size, 0, Math.PI * 2
                    );
                    ctx.fill();
                }
            }
        }
        
        // ==================== 灵气粒子类 ====================
        class SpiritParticle {
            constructor(game) {
                this.game = game;
                this.reset();
            }
            
            reset() {
                this.x = Math.random() * this.game.width;
                this.y = Math.random() * this.game.height * 0.7;
                this.size = 2 + Math.random() * 4;
                this.speedX = (Math.random() - 0.5) * 0.5;
                this.speedY = -0.2 - Math.random() * 0.3;
                this.opacity = 0.2 + Math.random() * 0.4;
                this.pulse = Math.random() * Math.PI * 2;
                this.pulseSpeed = 0.002 + Math.random() * 0.003;
                
                // 随机颜色
                const colors = [
                    [212, 165, 74],   // 金色
                    [62, 205, 196],   // 青色
                    [200, 180, 140]   // 米色
                ];
                this.color = colors[Math.floor(Math.random() * colors.length)];
            }
            
            update(deltaTime) {
                this.x += this.speedX - this.game.moveDirection * 0.5;
                this.y += this.speedY;
                this.pulse += this.pulseSpeed * deltaTime;
                
                // 边界检测
                if (this.y < -20 || this.x < -20 || this.x > this.game.width + 20) {
                    this.reset();
                    this.y = this.game.height * 0.7 + Math.random() * 100;
                }
            }
            
            render(ctx) {
                const pulseSize = this.size + Math.sin(this.pulse) * 2;
                const pulseOpacity = this.opacity * (0.5 + Math.sin(this.pulse) * 0.5);
                
                const gradient = ctx.createRadialGradient(
                    this.x, this.y, 0,
                    this.x, this.y, pulseSize * 2
                );
                gradient.addColorStop(0, `rgba(${this.color[0]}, ${this.color[1]}, ${this.color[2]}, ${pulseOpacity})`);
                gradient.addColorStop(0.5, `rgba(${this.color[0]}, ${this.color[1]}, ${this.color[2]}, ${pulseOpacity * 0.3})`);
                gradient.addColorStop(1, `rgba(${this.color[0]}, ${this.color[1]}, ${this.color[2]}, 0)`);
                
                ctx.fillStyle = gradient;
                ctx.beginPath();
                ctx.arc(this.x, this.y, pulseSize * 2, 0, Math.PI * 2);
                ctx.fill();
            }
        }
        
        // ==================== 初始化游戏 ====================
        window.onload = function() {
            window.game = new Game();
        };
        
    })();
    </script>

## 代码说明

### 核心架构

1. **Game 类** - 游戏主控制器
   - 管理游戏循环、帧率控制
   - 处理键盘/鼠标输入
   - 协调场景和角色的渲染

2. **SceneRenderer 类** - 场景渲染器
   - 5层视差滚动：远山 → 中景山 → 近景山 → 远景竹林 → 近景竹林
   - 水墨风格云雾系统（8个动态云团）
   - 程序化生成的竹林系统

3. **Character 类** - 角色系统
   - 三种状态：待机、行走、攻击
   - 帧动画系统
   - 呼吸动画（正弦波驱动的缩放和位移）
   - 攻击特效（蓄力 → 释放 → 回收）

4. **SpiritParticle 类** - 灵气粒子
   - 漂浮的发光粒子增添仙侠氛围

### 视觉特色

- **水墨配色**：深墨蓝底色 + 淡墨山峦 + 金色点缀
- **视差滚动**：每层不同的滚动速度（0.1x ~ 0.8x）
- **角色设计**：Q版小妖精，尖耳朵、圆身体、大眼睛

### 操作方式

- **A/D 或 方向键**：移动视角（触发视差滚动）
- **空格/J**：触发攻击动画
- **底部按钮**：点击控制
