```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>寻道大千挂机游戏</title>
<style>
  body {
    margin: 0;
    overflow: hidden;
  }
  canvas {
    display: block;
  }
  #ui {
    position: absolute;
    bottom: 10px;
    left: 10px;
    background-color: rgba(255, 255, 255, 0.8);
    padding: 5px;
    border-radius: 5px;
  }
</style>
</head>
<body>
<canvas id="gameCanvas"></canvas>
<div id="ui">Score: 0</div>
<script>
const canvas = document.getElementById('gameCanvas');
const ctx = canvas.getContext('2d');
const role = {
  x: canvas.width / 2,
  y: canvas.height - 30,
  width: 20,
  height: 30,
  color: 'green'
};
const background = {
  color: 'skyblue'
};
const score = 0;
let keys = {};

canvas.width = window.innerWidth;
canvas.height = window.innerHeight;

function drawBackground() {
  ctx.fillStyle = background.color;
  ctx.fillRect(0, 0, canvas.width, canvas.height);
}

function drawRole() {
  ctx.fillStyle = role.color;
  ctx.fillRect(role.x, role.y, role.width, role.height);
}

function drawUI() {
  ctx.fillStyle = 'black';
  ctx.fillText(`Score: ${score}`, 10, 20);
}

function moveRole() {
  if (keys.ArrowUp) {
    role.y -= 5;
  }
  if (keys.ArrowDown) {
    role.y += 5;
  }
  if (keys.ArrowLeft) {
    role.x -= 5;
  }
  if (keys.ArrowRight) {
    role.x += 5;
  }
  // Ensure the role stays within the canvas bounds
  role.x = Math.max(0, Math.min(role.x, canvas.width - role.width));
  role.y = Math.max(0, Math.min(role.y, canvas.height - role.height));
}

function loop() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  drawBackground();
  drawRole();
  drawUI();
  moveRole();
  requestAnimationFrame(loop);
}

window.addEventListener('DOMContentLoaded', () => {
  loop();
});

window.addEventListener('keydown', (e) => {
  keys[e.key] = true;
});

window.addEventListener('keyup', (e) => {
  keys[e.key] = false;
});
</script>
</body>
</html>
```

这段代码创建了一个简单的挂机游戏，玩家可以使用键盘的箭头键控制一个绿色的方块角色在画布上移动。游戏会在浏览器中直接运行，不需要任何额外的库或框架。代码中已经包含了所有必需的元素和功能，如`<canvas>`元素、`getContext('2d')`、`requestAnimationFrame`游戏循环、`DOMContentLoaded`初始化以及键盘事件绑定。
