        // ==================== 工具函数 ====================
        const Utils = {
            lerp: (a, b, t) => a + (b - a) * t,
            clamp: (val, min, max) => Math.max(min, Math.min(max, val)),
            random: (min, max) => Math.random() * (max - min) + min,
            randomInt: (min, max) => Math.floor(Math.random() * (max - min + 1)) + min,
            easeOutQuad: t => t * (2 - t),
            easeOutCubic: t => 1 - Math.pow(1 - t, 3),
            easeOutElastic: t => {
                const c4 = (2 * Math.PI) / 3;
                return t === 0 ? 0 : t === 1 ? 1 : Math.pow(2, -10 * t) * Math.sin((t * 10 - 0.75) * c4) + 1;
            },
            easeOutBounce: t => {
                const n1 = 7.5625;
                const d1 = 2.75;
                if (t < 1 / d1) return n1 * t * t;
                if (t < 2 / d1) return n1 * (t -= 1.5 / d1) * t + 0.75;
                if (t < 2.5 / d1) return n1 * (t -= 2.25 / d1) * t + 0.9375;
                return n1 * (t -= 2.625 / d1) * t + 0.984375;
            },
            formatNumber: num => {
                if (num >= 1000000) return (num / 1000000).toFixed(1) + 'M';
                if (num >= 1000) return (num / 1000).toFixed(1) + 'K';
                return Math.floor(num).toString();
            }
        };

        // ==================== 粒子基类 ====================
        class Particle {
            constructor(x, y, options = {}) {
                this.x = x;
                this.y = y;
                this.vx = options.vx || 0;
                this.vy = options.vy || 0;
                this.ax = options.ax || 0;
                this.ay = options.ay || 0;
                this.size = options.size || 4;
                this.sizeEnd = options.sizeEnd || 0;
                this.life = options.life || 1;
                this.maxLife = this.life;
                this.color = options.color || '#ffffff';
               .colorEnd = options.colorEnd || null;
                this.alpha = options.alpha || 1;
                this.alphaEnd = options.alphaEnd || 0;
                this.rotation = options.rotation || 0;
                this.rotationSpeed = options.rotationSpeed || 0;
                this.shape = options.shape || 'circle';
                this.gravity = options.gravity || 0;
                this.friction = options.friction || 1;
            }

            update(dt) {
                this.vx += this.ax * dt;
                this.vy += this.ay * dt;
                this.vy += this.gravity * dt;
                this.vx *= this.friction;
                this.vy *= this.friction;
                this.x += this.vx * dt;
                this.y += this.vy * dt;
                this.rotation += this.rotationSpeed * dt;
                this.life -= dt;
                return this.life > 0;
            }

            draw(ctx) {
                const progress = 1 - (this.life / this.maxLife);
                const size = Utils.lerp(this.size, this.sizeEnd, progress);
                if (size <= 0.1) return;
                
                const alpha = Utils.lerp(this.alpha, this.alphaEnd, progress);
                
                ctx.save();
                ctx.translate(this.x, this.y);
                ctx.rotate(this.rotation);
                ctx.globalAlpha = Utils.clamp(alpha, 0, 1);
                
                if (this.colorEnd) {
                    // 渐变颜色处理
                    ctx.fillStyle = this.color;
                } else {
                    ctx.fillStyle = this.color;
                }
                
                if (this.shape === 'circle') {
                    ctx.beginPath();
                    ctx.arc(0, 0, Math.max(0.5, size), 0, Math.PI * 2);
                    ctx.fill();
                } else if (this.shape === 'rect') {
                    ctx.fillRect(-size / 2, -size / 2, size, size);
                } else if (this.shape === 'star') {
                    this.drawStar(ctx, 0, 0, 5, size, size / 2);
                } else if (this.shape === 'spark') {
                    this.drawSpark(ctx, size);
                }
                
                ctx.restore();
            }

            drawStar(ctx, cx, cy, spikes, outerRadius, innerRadius) {
                let rot = Math.PI / 2 * 3;
                let step = Math.PI / spikes;
                ctx.beginPath();
                ctx.moveTo(cx, cy - outerRadius);
                for (let i = 0; i < spikes; i++) {
                    ctx.lineTo(cx + Math.cos(rot) * outerRadius, cy + Math.sin(rot) * outerRadius);
                    rot += step;
                    ctx.lineTo(cx + Math.cos(rot) * innerRadius, cy + Math.sin(rot) * innerRadius);
                    rot += step;
                }
                ctx.lineTo(cx, cy - outerRadius);
                ctx.closePath();
                ctx.fill();
            }

            drawSpark(ctx, size) {
                ctx.beginPath();
                ctx.moveTo(0, -size);
                ctx.lineTo(size * 0.3, 0);
                ctx.lineTo(0, size);
                ctx.lineTo(-size * 0.3, 0);
                ctx.closePath();
                ctx.fill();
            }
        }

        // ==================== 挥砍轨迹特效 ====================
        class SlashTrail {
            constructor(x, y, angle, length, options = {}) {
                this.x = x;
                this.y = y;
                this.angle = angle;
                this.length = length;
                this.width = options.width || 30;
                this.life = options.life || 0.4;
                this.maxLife = this.life;
                this.color = options.color || '#d4a54a';
                this.segments = [];
                this.generateSegments();
            }

            generateSegments() {
                const segCount = 12;
                for (let i = 0; i < segCount; i++) {
                    const t = i / (segCount - 1);
                    const offset = Math.sin(t * Math.PI) * this.width * 0.5;
                    this.segments.push({
                        offset: offset,
                        delay: t * 0.05
                    });
                }
            }

            update(dt) {
                this.life -= dt;
                return this.life > 0;
            }

            draw(ctx) {
                const progress = 1 - (this.life / this.maxLife);
                const alpha = 1 - progress;
                
                ctx.save();
                ctx.translate(this.x, this.y);
                ctx.rotate(this.angle);
                
                // 绘制主轨迹
                ctx.beginPath();
                const drawProgress = Utils.clamp(progress * 2, 0, 1);
                const visibleSegments = Math.floor(this.segments.length * drawProgress);
                
                for (let i = 0; i < visibleSegments; i++) {
                    const seg = this.segments[i];
                    const segProgress = i / Math.max(1, visibleSegments - 1);
                    const x = segProgress * this.length - this.length / 2;
                    const fadeAlpha = alpha * (1 - Math.abs(segProgress - 0.5) * 1.5);
                    
                    ctx.save();
                    ctx.globalAlpha = Utils.clamp(fadeAlpha, 0, 1);
                    
                    // 创建渐变
                    const gradient = ctx.createRadialGradient(x, 0, 0, x, 0, Math.max(1, seg.offset + 5));
                    gradient.addColorStop(0, this.color);
                    gradient.addColorStop(0.5, this.color + 'aa');
                    gradient.addColorStop(1, this.color + '00');
                    ctx.fillStyle = gradient;
                    
                    ctx.beginPath();
                    ctx.arc(x, 0, Math.max(1, seg.offset + 5), 0, Math.PI * 2);
                    ctx.fill();
                    
                    ctx.restore();
                }
                
                // 绘制发光边缘
                ctx.strokeStyle = this.color;
                ctx.lineWidth = 3;
                ctx.globalAlpha = alpha * 0.8;
                ctx.lineCap = 'round';
                
                ctx.beginPath();
                ctx.moveTo(-this.length / 2, 0);
                for (let i = 1; i < visibleSegments; i++) {
                    const seg = this.segments[i];
                    const segProgress = i / (this.segments.length - 1);
                    const x = segProgress * this.length - this.length / 2;
                    ctx.lineTo(x, seg.offset);
                }
                ctx.stroke();
                
                ctx.beginPath();
                ctx.moveTo(-this.length / 2, 0);
                for (let i = 1; i < visibleSegments; i++) {
                    const seg = this.segments[i];
                    const segProgress = i / (this.segments.length - 1);
                    const x = segProgress * this.length - this.length / 2;
                    ctx.lineTo(x, -seg.offset);
                }
                ctx.stroke();
                
                ctx.restore();
            }
        }

        // ==================== 命中火花特效 ====================
        class HitSpark {
            constructor(x, y, options = {}) {
                this.x = x;
                this.y = y;
                this.particles = [];
                this.life = options.life || 0.6;
                this.maxLife = this.life;
                this.color = options.color || '#ffcc44';
                this.generateParticles(options.count || 20);
            }

            generateParticles(count) {
                for (let i = 0; i < count; i++) {
                    const angle = Utils.random(0, Math.PI * 2);
                    const speed = Utils.random(100, 400);
                    this.particles.push(new Particle(this.x, this.y, {
                        vx: Math.cos(angle) * speed,
                        vy: Math.sin(angle) * speed,
                        size: Utils.random(2, 6),
                        sizeEnd: 0,
                        life: Utils.random(0.2, 0.5),
                        color: this.color,
                        alpha: 1,
                        alphaEnd: 0,
                        gravity: 200,
                        friction: 0.96,
                        shape: Utils.random() > 0.5 ? 'circle' : 'spark'
                    }));
                }
                
                // 添加核心闪光
                this.coreFlash = {
                    size: 30,
                    alpha: 1
                };
            }

            update(dt) {
                this.life -= dt;
                this.particles = this.particles.filter(p => p.update(dt));
                
                if (this.coreFlash) {
                    this.coreFlash.alpha -= dt * 4;
                    this.coreFlash.size += dt * 100;
                    if (this.coreFlash.alpha <= 0) {
                        this.coreFlash = null;
                    }
                }
                
                return this.life > 0 || this.particles.length > 0;
            }

            draw(ctx) {
                // 绘制核心闪光
                if (this.coreFlash) {
                    const gradient = ctx.createRadialGradient(
                        this.x, this.y, 0,
                        this.x, this.y, Math.max(1, this.coreFlash.size)
                    );
                    gradient.addColorStop(0, `rgba(255, 255, 255, ${this.coreFlash.alpha})`);
                    gradient.addColorStop(0.3, `rgba(255, 220, 100, ${this.coreFlash.alpha * 0.6})`);
                    gradient.addColorStop(1, 'rgba(255, 200, 50, 0)');
                    
                    ctx.fillStyle = gradient;
                    ctx.beginPath();
                    ctx.arc(this.x, this.y, Math.max(1, this.coreFlash.size), 0, Math.PI * 2);
                    ctx.fill();
                }
                
                // 绘制粒子
                this.particles.forEach(p => p.draw(ctx));
            }
        }

        // ==================== 伤害数字特效 ====================
        class DamageNumber {
            constructor(x, y, value, options = {}) {
                this.x = x;
                this.y = y;
                this.value = value;
                this.isCrit = options.isCrit || false;
                this.isDodge = options.isDodge || false;
                this.life = options.life || 1.5;
                this.maxLife = this.life;
                this.scale = 0;
                this.targetScale = this.isCrit ? 1.8 : 1.2;
                this.vy = -80;
                this.alpha = 1;
                this.shakeOffset = 0;
                this.shakeTime = 0;
                
                if (this.isCrit) {
                    this.color = '#ffd700';
                    this.prefix = '暴击 ';
                } else if (this.isDodge) {
                    this.color = '#888888';
                    this.prefix = '闪避';
                } else {
                    this.color = '#ffffff';
                    this.prefix = '';
                }
            }

            update(dt) {
                this.life -= dt;
                
                // 弹出动画
                const appearProgress = Utils.clamp((this.maxLife - this.life) / 0.2, 0, 1);
                this.scale = Utils.easeOutElastic(appearProgress) * this.targetScale;
                
                // 上浮
                this.y += this.vy * dt;
                this.vy *= 0.95;
                
                // 暴击震动
                if (this.isCrit && this.life > this.maxLife - 0.3) {
                    this.shakeTime += dt * 30;
                    this.shakeOffset = Math.sin(this.shakeTime) * 3;
                } else {
                    this.shakeOffset = 0;
                }
                
                // 淡出
                if (this.life < 0.5) {
                    this.alpha = this.life / 0.5;
                }
                
                return this.life > 0;
            }

            draw(ctx) {
                ctx.save();
                ctx.translate(this.x + this.shakeOffset, this.y);
                ctx.scale(this.scale, this.scale);
                ctx.globalAlpha = this.alpha;
                
                const text = this.prefix + Utils.formatNumber(this.value);
                ctx.font = this.isCrit ? 'bold 28px "Ma Shan Zheng"' : 'bold 22px "Ma Shan Zheng"';
                ctx.textAlign = 'center';
                ctx.textBaseline = 'middle';
                
                // 描边
                ctx.strokeStyle = 'rgba(0, 0, 0, 0.8)';
                ctx.lineWidth = 4;
                ctx.strokeText(text, 0, 0);
                
                // 发光效果
                if (this.isCrit) {
                    ctx.shadowColor = this.color;
                    ctx.shadowBlur = 15;
                }
                
                ctx.fillStyle = this.color;
                ctx.fillText(text, 0, 0);
                
                ctx.restore();
            }
        }

        // ==================== 敌人死亡特效（灵气消散） ====================
        class DeathEffect {
            constructor(x, y, options = {}) {
                this.x = x;
                this.y = y;
                this.life = 2;
                this.maxLife = this.life;
                this.soulParticles = [];
                this.rings = [];
                this.color = options.color || '#64b5f6';
                this.generateEffects();
            }

            generateEffects() {
                // 灵气粒子
                for (let i = 0; i < 30; i++) {
                    const angle = Utils.random(0, Math.PI * 2);
                    const distance = Utils.random(10, 40);
                    this.soulParticles.push({
                        x: this.x + Math.cos(angle) * distance,
                        y: this.y + Math.sin(angle) * distance,
                        size: Utils.random(4, 12),
                        alpha: 1,
                        vy: Utils.random(-60, -30),
                        vx: Utils.random(-20, 20),
                        angle: Utils.random(0, Math.PI * 2),
                        rotSpeed: Utils.random(-2, 2)
                    });
                }
                
                // 扩散光环
                for (let i = 0; i < 3; i++) {
                    this.rings.push({
                        radius: 10,
                        alpha: 0.8,
                        delay: i * 0.15
                    });
                }
            }

            update(dt) {
                this.life -= dt;
                
                // 更新灵气粒子
                this.soulParticles.forEach(p => {
                    p.y += p.vy * dt;
                    p.x += p.vx * dt;
                    p.vy -= 20 * dt;
                    p.angle += p.rotSpeed * dt;
                    p.alpha = Utils.clamp((this.life - 0.5) / 1.5, 0, 1);
                    p.size *= 0.995;
                });
                
                // 更新光环
                this.rings.forEach(r => {
                    if (this.life < this.maxLife - r.delay) {
                        r.radius += 150 * dt;
                        r.alpha -= dt * 0.6;
                    }
                });
                
                return this.life > 0;
            }

            draw(ctx) {
                ctx.save();
                
                // 绘制扩散光环
                this.rings.forEach(r => {
                    if (r.alpha > 0) {
                        ctx.beginPath();
                        ctx.arc(this.x, this.y, Math.max(1, r.radius), 0, Math.PI * 2);
                        ctx.strokeStyle = this.color;
                        ctx.globalAlpha = Utils.clamp(r.alpha, 0, 1);
                        ctx.lineWidth = 3;
                        ctx.stroke();
                    }
                });
                
                // 绘制灵气粒子
                this.soulParticles.forEach(p => {
                    if (p.alpha <= 0 || p.size < 1) return;
                    
                    ctx.save();
                    ctx.translate(p.x, p.y);
                    ctx.rotate(p.angle);
                    ctx.globalAlpha = p.alpha;
                    
                    // 灵气光晕
                    const gradient = ctx.createRadialGradient(0, 0, 0, 0, 0, Math.max(1, p.size));
                    gradient.addColorStop(0, 'rgba(255, 255, 255, 0.9)');
                    gradient.addColorStop(0.4, this.color);
                    gradient.addColorStop(1, 'rgba(100, 181, 246, 0)');
                    
                    ctx.fillStyle = gradient;
                    ctx.beginPath();
                    ctx.arc(0, 0, Math.max(1, p.size), 0, Math.PI * 2);
                    ctx.fill();
                    
                    // 内核
                    ctx.fillStyle = 'rgba(255, 255, 255, 0.8)';
                    ctx.beginPath();
                    ctx.arc(0, 0, Math.max(0.5, p.size * 0.3), 0, Math.PI * 2);
                    ctx.fill();
                    
                    ctx.restore();
                });
                
                ctx.restore();
            }
        }

        // ==================== 升级光效 ====================
        class LevelUpEffect {
            constructor(x, y) {
                this.x = x;
                this.y = y;
                this.life = 2.5;
                this.maxLife = this.life;
                this.rings = [];
                this.lightBeams = [];
                this.particles = [];
                this.generateEffects();
            }

            generateEffects() {
                // 光环
                for (let i = 0; i < 5; i++) {
                    this.rings.push({
                        radius: 20,
                        targetRadius: 200 + i * 40,
                        alpha: 1,
                        delay: i * 0.1,
                        started: false
                    });
                }
                
                // 光柱
                for (let i = 0; i < 8; i++) {
                    const angle = (i / 8) * Math.PI * 2;
                    this.lightBeams.push({
                        angle: angle,
                        length: 0,
                        targetLength: 300,
                        width: Utils.random(15, 30),
                        alpha: 0,
                        delay: Utils.random(0, 0.3)
                    });
                }
                
                // 金色粒子
                for (let i = 0; i < 50; i++) {
                    const angle = Utils.random(0, Math.PI * 2);
                    const speed = Utils.random(50, 200);
                    this.particles.push(new Particle(this.x, this.y, {
                        vx: Math.cos(angle) * speed,
                        vy: Math.sin(angle) * speed - 100,
                        size: Utils.random(3, 8),
                        sizeEnd: 0,
                        life: Utils.random(1, 2),
                        color: Utils.random() > 0.5 ? '#ffd700' : '#ffaa00',
                        alpha: 1,
                        alphaEnd: 0,
                        gravity: -50,
                        shape: 'star'
                    }));
                }
            }

            update(dt) {
                this.life -= dt;
                
                const elapsed = this.maxLife - this.life;
                
                // 更新光环
                this.rings.forEach((r, i) => {
                    if (elapsed > r.delay) {
                        r.started = true;
                        const ringElapsed = elapsed - r.delay;
                        const progress = Utils.easeOutCubic(Utils.clamp(ringElapsed / 0.8, 0, 1));
                        r.radius = Utils.lerp(20, r.targetRadius, progress);
                        r.alpha = 1 - progress;
                    }
                });
                
                // 更新光柱
                this.lightBeams.forEach(b => {
                    if (elapsed > b.delay) {
                        const beamElapsed = elapsed - b.delay;
                        const progress = Utils.easeOutQuad(Utils.clamp(beamElapsed / 0.5, 0, 1));
                        b.length = b.targetLength * progress;
                        b.alpha = Math.sin(progress * Math.PI) * 0.6;
                    }
                });
                
                // 更新粒子
                this.particles = this.particles.filter(p => p.update(dt));
                
                return this.life > 0;
            }

            draw(ctx) {
                ctx.save();
                
                // 绘制光柱
                this.lightBeams.forEach(b => {
                    if (b.alpha <= 0) return;
                    
                    ctx.save();
                    ctx.translate(this.x, this.y);
                    ctx.rotate(b.angle);
                    ctx.globalAlpha = b.alpha;
                    
                    const gradient = ctx.createLinearGradient(0, 0, b.length, 0);
                    gradient.addColorStop(0, '#ffd700');
                    gradient.addColorStop(0.5, 'rgba(255, 215, 0, 0.5)');
                    gradient.addColorStop(1, 'rgba(255, 215, 0, 0)');
                    
                    ctx.fillStyle = gradient;
                    ctx.beginPath();
                    ctx.moveTo(0, -b.width / 2);
                    ctx.lineTo(b.length, -b.width / 4);
                    ctx.lineTo(b.length, b.width / 4);
                    ctx.lineTo(0, b.width / 2);
                    ctx.closePath();
                    ctx.fill();
                    
                    ctx.restore();
                });
                
                // 绘制光环
                this.rings.forEach(r => {
                    if (!r.started || r.alpha <= 0) return;
                    
                    ctx.beginPath();
                    ctx.arc(this.x, this.y, Math.max(1, r.radius), 0, Math.PI * 2);
                    ctx.strokeStyle = '#ffd700';
                    ctx.globalAlpha = r.alpha;
                    ctx.lineWidth = 4;
                    ctx.stroke();
                    
                    // 光晕
                    const gradient = ctx.createRadialGradient(
                        this.x, this.y, Math.max(1, r.radius - 10),
                        this.x, this.y, Math.max(2, r.radius + 10)
                    );
                    gradient.addColorStop(0, 'rgba(255, 215, 0, 0)');
                    gradient.addColorStop(0.5, `rgba(255, 215, 0, ${r.alpha * 0.3})`);
                    gradient.addColorStop(1, 'rgba(255, 215, 0, 0)');
                    
                    ctx.fillStyle = gradient;
                    ctx.beginPath();
                    ctx.arc(this.x, this.y, Math.max(2, r.radius + 10), 0, Math.PI * 2);
                    ctx.fill();
                });
                
                // 绘制中心光球
                const coreAlpha = Utils.clamp((this.life - 0.5) / 2, 0, 1);
                if (coreAlpha > 0) {
                    const coreSize = 60 + Math.sin((this.maxLife - this.life) * 5) * 10;
                    const gradient = ctx.createRadialGradient(
                        this.x, this.y, 0,
                        this.x, this.y, Math.max(1, coreSize)
                    );
                    gradient.addColorStop(0, `rgba(255, 255, 255, ${coreAlpha})`);
                    gradient.addColorStop(0.3, `rgba(255, 230, 150, ${coreAlpha * 0.8})`);
                    gradient.addColorStop(1, 'rgba(255, 200, 100, 0)');
                    
                    ctx.fillStyle = gradient;
                    ctx.beginPath();
                    ctx.arc(this.x, this.y, Math.max(1, coreSize), 0, Math.PI * 2);
                    ctx.fill();
                }
                
                // 绘制粒子
                this.particles.forEach(p => p.draw(ctx));
                
                ctx.restore();
            }
        }

        class RippleEffect {
            constructor(x, y, container) {
                this.x = x;
                this.y = y;
                this.container = container;
                this.radius = 0;
                this.maxRadius = 100;
                this.alpha = 0.6;
                this.life = 0.6;
            }

            update(dt) {
                this.life -= dt;
                const progress = 1 - (this.life / 0.6);
                this.radius = progress * this.maxRadius;
                this.alpha = 0.6 * (1 - progress);
                return this.life > 0;
            }

            draw(ctx) {
                if (this.alpha <= 0) return;
                
                ctx.save();
                ctx.globalAlpha = this.alpha;
                ctx.strokeStyle = '#d4a54a';
                ctx.lineWidth = 2;
                ctx.beginPath();
                ctx.arc(this.x, this.y, Math.max(1, this.radius), 0, Math.PI * 2);
                ctx.stroke();
                ctx.restore();
            }
        }

        // ==================== 屏幕震动 ====================
        class ScreenShake {
            constructor() {
                this.intensity = 0;
                this.duration = 0;
                this.offsetX = 0;
                this.offsetY = 0;
            }

            shake(intensity, duration) {
                this.intensity = intensity;
                this.duration = duration;
            }

            update(dt) {
                if (this.duration > 0) {
                    this.duration -= dt;
                    const progress = this.duration / 0.4;
                    const currentIntensity = this.intensity * progress;
                    this.offsetX = Utils.random(-currentIntensity, currentIntensity);
                    this.offsetY = Utils.random(-currentIntensity, currentIntensity);
                } else {
                    this.offsetX = 0;
                    this.offsetY = 0;
                }
            }

            apply(ctx) {
                ctx.translate(this.offsetX, this.offsetY);
            }
        }

        class Enemy {
            constructor(x, y, type = 'normal') {
                this.x = x;
                this.y = y;
                this.type = type;
                this.radius = type === 'boss' ? 50 : 35;
                this.hp = type === 'boss' ? CONFIG.ENEMY.baseHp * 5 : CONFIG.ENEMY.baseHp;
                this.maxHp = this.hp;
                this.atk = type === 'boss' ? CONFIG.ENEMY.baseAtk * 2 : CONFIG.ENEMY.baseAtk;
                this.name = type === 'boss' ? '妖王' : '小妖';
                this.color = type === 'boss' ? '#9c64b5' : '#64b5f6';
                this.hitFlash = 0;
                this.scale = 1;
                this.wobbleTime = 0;
                this.alive = true;
            }

            takeDamage(damage, isCrit) {
                this.hp -= damage;
                this.hitFlash = 0.15;
                this.scale = isCrit ? 1.15 : 1.08;
                
                if (this.hp <= 0) {
                    this.alive = false;
                    return true; // 死亡
                }
                return false;
            }

            update(dt) {
                if (this.hitFlash > 0) {
                    this.hitFlash -= dt;
                }
                this.scale = Utils.lerp(this.scale, 1, dt * 8);
                this.wobbleTime += dt * 2;
            }

            draw(ctx) {
                if (!this.alive) return;
                
                ctx.save();
                ctx.translate(this.x, this.y);
                ctx.scale(this.scale, this.scale);
                
                // 呼吸动画
                const breathe = Math.sin(this.wobbleTime) * 0.03;
                ctx.scale(1 + breathe, 1 - breathe);
                
                // 受击闪烁
                if (this.hitFlash > 0) {
                    ctx.filter = 'brightness(2)';
                }
                
                // 敌人身体
                const gradient = ctx.createRadialGradient(0, -5, 0, 0, 0, Math.max(1, this.radius));
                gradient.addColorStop(0, this.color);
                gradient.addColorStop(0.7, this.adjustColor(this.color, -30));
                gradient.addColorStop(1, this.adjustColor(this.color, -50));
                
                ctx.fillStyle = gradient;
                ctx.beginPath();
                ctx.arc(0, 0, Math.max(1, this.radius), 0, Math.PI * 2);
                ctx.fill();
                
                // 眼睛
                ctx.fillStyle = '#ffffff';
                ctx.beginPath();
                ctx.arc(-this.radius * 0.3, -this.radius * 0.2, 6, 0, Math.PI * 2);
                ctx.arc(this.radius * 0.3, -this.radius * 0.2, 6, 0, Math.PI * 2);
                ctx.fill();
                
                ctx.fillStyle = '#1a1a2e';
                ctx.beginPath();
                ctx.arc(-this.radius * 0.3, -this.radius * 0.2, 3, 0, Math.PI * 2);
                ctx.arc(this.radius * 0.3, -this.radius * 0.2, 3, 0, Math.PI * 2);
                ctx.fill();
                
                // 嘴巴
                ctx.strokeStyle = '#1a1a2e';
                ctx.lineWidth = 2;
                ctx.beginPath();
                ctx.arc(0, this.radius * 0.2, 8, 0, Math.PI);
                ctx.stroke();
                
                ctx.restore();
                
                // 血条
                this.drawHealthBar(ctx);
            }

            drawHealthBar(ctx) {
                const barWidth = this.radius * 2.5;
                const barHeight = 8;
                const barX = this.x - barWidth / 2;
                const barY = this.y - this.radius - 20;
                
                // 背景
                ctx.fillStyle = 'rgba(0, 0, 0, 0.6)';
                ctx.fillRect(barX, barY, barWidth, barHeight);
                
                // 血量
                const hpPercent = Math.max(0, this.hp / this.maxHp);
                const gradient = ctx.createLinearGradient(barX, barY, barX + barWidth * hpPercent, barY);
                gradient.addColorStop(0, '#d64545');
                gradient.addColorStop(1, '#ff6b6b');
                ctx.fillStyle = gradient;
                ctx.fillRect(barX, barY, barWidth * hpPercent, barHeight);
                
                // 边框
                ctx.strokeStyle = 'rgba(255, 255, 255, 0.3)';
                ctx.lineWidth = 1;
                ctx.strokeRect(barX, barY, barWidth, barHeight);
            }

            adjustColor(color, amount) {
                const hex = color.replace('#', '');
                const r = Math.max(0, Math.min(255, parseInt(hex.substr(0, 2), 16) + amount));
                const g = Math.max(0, Math.min(255, parseInt(hex.substr(2, 2), 16) + amount));
                const b = Math.max(0, Math.min(255, parseInt(hex.substr(4, 2), 16) + amount));
                return `rgb(${r}, ${g}, ${b})`;
            }
        }

        // ==================== 背景渲染器 ====================
        class Background {
            constructor(width, height) {
                this.width = width;
                this.height = height;
                this.layers = [];
                this.time = 0;
                this.stars = [];
                this.mistParticles = [];
                this.generateBackground();
            }

            generateBackground() {
                // 星星
                for (let i = 0; i < 80; i++) {
                    this.stars.push({
                        x: Utils.random(0, this.width),
                        y: Utils.random(0, this.height * 0.6),
                        size: Utils.random(1, 3),
                        alpha: Utils.random(0.3, 1),
                        twinkleSpeed: Utils.random(1, 3)
                    });
                }
                
                // 雾气粒子
                for (let i = 0; i < 20; i++) {
                    this.mistParticles.push({
                        x: Utils.random(0, this.width),
                        y: Utils.random(this.height * 0.5, this.height),
                        size: Utils.random(100, 300),
                        alpha: Utils.random(0.05, 0.15),
                        speed: Utils.random(5, 20)
                    });
                }
            }

            update(dt) {
                this.time += dt;
                
                this.mistParticles.forEach(m => {
                    m.x += m.speed * dt;
                    if (m.x > this.width + m.size) {
                        m.x = -m.size;
                    }
                });
            }

            draw(ctx) {
                // 渐变背景
                const gradient = ctx.createLinearGradient(0, 0, 0, this.height);
                gradient.addColorStop(0, '#0a0a18');
                gradient.addColorStop(0.4, '#101025');
                gradient.addColorStop(0.7, '#151530');
                gradient.addColorStop(1, '#0d0d1a');
                ctx.fillStyle = gradient;
                ctx.fillRect(0, 0, this.width, this.height);
                
                // 绘制星星
                this.stars.forEach(s => {
                    const twinkle = Math.sin(this.time * s.twinkleSpeed) * 0.3 + 0.7;
                    ctx.fillStyle = `rgba(255, 255, 255, ${s.alpha * twinkle})`;
                    ctx.beginPath();
                    ctx.arc(s.x, s.y, s.size, 0, Math.PI * 2);
                    ctx.fill();
                });
                
                // 绘制月亮
                this.drawMoon(ctx);
                
                // 绘制远山
                this.drawMountains(ctx);
                
                // 绘制雾气
                this.drawMist(ctx);
            }

            drawMoon(ctx) {
                const moonX = this.width * 0.8;
                const moonY = 80;
                const moonRadius = 40;
                
                // 月晕
                const haloGradient = ctx.createRadialGradient(moonX, moonY, moonRadius, moonX, moonY, moonRadius * 3);
                haloGradient.addColorStop(0, 'rgba(255, 250, 220, 0.2)');
                haloGradient.addColorStop(0.5, 'rgba(255, 250, 220, 0.05)');
                haloGradient.addColorStop(1, 'rgba(255, 250, 220, 0)');
                ctx.fillStyle = haloGradient;
                ctx.beginPath();
                ctx.arc(moonX, moonY, moonRadius * 3, 0, Math.PI * 2);
                ctx.fill();
                
                // 月亮
                const moonGradient = ctx.createRadialGradient(moonX - 10, moonY - 10, 0, moonX, moonY, moonRadius);
                moonGradient.addColorStop(0, '#fffaec');
                moonGradient.addColorStop(0.7, '#f0e6c8');
                moonGradient.addColorStop(1, '#d4c4a8');
                ctx.fillStyle = moonGradient;
                ctx.beginPath();
                ctx.arc(moonX, moonY, moonRadius, 0, Math.PI * 2);
                ctx.fill();
            }

            drawMountains(ctx) {
                // 远山
                ctx.fillStyle = 'rgba(20, 20, 40, 0.8)';
                ctx.beginPath();
                ctx.moveTo(0, this.height);
                ctx.lineTo(0, this.height * 0.7);
                ctx.bezierCurveTo(
                    this.width * 0.2, this.height * 0.5,
                    this.width * 0.4, this.height * 0.65,
                    this.width * 0.5, this.height * 0.55
                );
                ctx.bezierCurveTo(
                    this.width * 0.6, this.height * 0.45,
                    this.width * 0.8, this.height * 0.6,
                    this.width, this.height * 0.5
                );
                ctx.lineTo(this.width, this.height);
                ctx.closePath();
                ctx.fill();
                
                // 近山
                ctx.fillStyle = 'rgba(15, 15, 30, 0.9)';
                ctx.beginPath();
                ctx.moveTo(0, this.height);
                ctx.lineTo(0, this.height * 0.8);
                ctx.bezierCurveTo(
                    this.width * 0.3, this.height * 0.65,
                    this.width * 0.5, this.height * 0.75,
                    this.width * 0.7, this.height * 0.7
                );
                ctx.bezierCurveTo(
                    this.width * 0.85, this.height * 0.65,
                    this.width * 0.95, this.height * 0.72,
                    this.width, this.height * 0.68
                );
                ctx.lineTo(this.width, this.height);
                ctx.closePath();
                ctx.fill();
            }

            drawMist(ctx) {
                this.mistParticles.forEach(m => {
                    const gradient = ctx.createRadialGradient(m.x, m.y, 0, m.x, m.y, m.size);
                    gradient.addColorStop(0, `rgba(100, 120, 150, ${m.alpha})`);
                    gradient.addColorStop(0.5, `rgba(80, 100, 130, ${m.alpha * 0.5})`);
                    gradient.addColorStop(1, 'rgba(60, 80, 110, 0)');
                    ctx.fillStyle = gradient;
                    ctx.beginPath();
                    ctx.arc(m.x, m.y, m.size, 0, Math.PI * 2);
                    ctx.fill();
                });
            }
        }

        // ==================== 战斗系统 ====================
        class BattleSystem {
            constructor(player, vfxManager) {
                this.player = player;
                this.vfxManager = vfxManager;
                this.enemies = [];
                this.comboCount = 0;
                this.totalDamage = 0;
                this.waveNumber = 1;
                this.maxWaves = 5;
                this.spawnTimer = 0;
                this.spawnInterval = 3;
            }

            spawnEnemy() {
                const isBoss = this.waveNumber === this.maxWaves && this.enemies.length === 0;
                const x = Utils.random(canvas.width * 0.5, canvas.width * 0.85);
                const y = Utils.random(canvas.height * 0.3, canvas.height * 0.7);
                this.enemies.push(new Enemy(x, y, isBoss ? 'boss' : 'normal'));
            }

            performAttack() {
                if (!this.player.canAttack() || this.enemies.length === 0) return;
                
                this.player.attack();
                
                const target = this.enemies[0];
                const isCrit = Math.random() < this.player.critRate;
                let damage = this.player.atk;
                
                if (isCrit) {
                    damage *= this.player.critDmg;
                }
                
                // 计算攻击方向
                const dx = target.x - this.player.x;
                const dy = target.y - this.player.y;
                const angle = Math.atan2(dy, dx);
                
                // 挥砍轨迹
                const slashX = this.player.x + dx * 0.5;
                const slashY = this.player.y + dy * 0.5;
                this.vfxManager.addSlashTrail(slashX, slashY, angle, 150, {
                    color: isCrit ? '#ffd700' : '#d4a54a',
                    width: isCrit ? 50 : 30
                });
                
                // 命中火花
                this.vfxManager.addHitSpark(target.x, target.y, {
                    color: isCrit ? '#ffd700' : '#ff8844',
                    count: isCrit ? 35 : 20
                });
                
                // 伤害数字
                const dmgX = target.x + Utils.random(-20, 20);
                const dmgY = target.y - target.radius - 10;
                this.vfxManager.addDamageNumber(dmgX, dmgY, damage, {
                    isCrit: isCrit,
                    isDodge: false
                });
                
                // 暴击震动
                if (isCrit) {
                    this.vfxManager.triggerScreenShake(12, 0.3);
                }
                
                // 应用伤害
                const killed = target.takeDamage(damage, isCrit);
                
                // 更新统计
                this.comboCount++;
                this.totalDamage += damage;
                
                if (killed) {
                    this.onEnemyKilled(target);
                }
            }

            onEnemyKilled(enemy) {
                // 死亡特效
                this.vfxManager.addDeathEffect(enemy.x, enemy.y, {
                    color: enemy.color
                });
                
                // 移除敌人
                const index = this.enemies.indexOf(enemy);
                if (index > -1) {
                    this.enemies.splice(index, 1);
                }
                
                // 检查波次
                if (this.enemies.length === 0) {
                    if (this.waveNumber < this.maxWaves) {
                        this.waveNumber++;
                        this.spawnTimer = 1;
                    } else {
                        // 通关，触发升级效果
                        this.onLevelUp();
                        this.waveNumber = 1;
                    }
                }
            }

            onLevelUp() {
                const centerX = canvas.width / 2;
                const centerY = canvas.height / 2;
                this.vfxManager.addLevelUpEffect(centerX, centerY);
                this.vfxManager.triggerScreenShake(8, 0.5);
                
                // 显示升级提示
                const toast = document.getElementById('levelUpToast');
                toast.classList.add('show');
                setTimeout(() => toast.classList.remove('show'), 2000);
                
                // 提升属性
                this.player.atk *= 1.15;
                this.player.critRate = Math.min(0.8, this.player.critRate + 0.02);
            }

            update(dt) {
                this.player.update(dt);
                this.enemies.forEach(e => e.update(dt));
                
                // 自动攻击
                if (this.enemies.length > 0 && this.player.canAttack()) {
                    this.performAttack();
                }
                
                // 自动生成敌人
                if (this.enemies.length < 3) {
                    this.spawnTimer -= dt;
                    if (this.spawnTimer <= 0) {
                        this.spawnEnemy();
                        this.spawnTimer = this.spawnInterval;
                    }
                }
                
                // 更新UI
                document.getElementById('comboCount').textContent = this.comboCount;
                document.getElementById('totalDmg').textContent = Utils.formatNumber(this.totalDamage);
                document.getElementById('atkValue').textContent = Utils.formatNumber(this.player.atk);
            }

            draw(ctx) {
                this.player.draw(ctx);
                this.enemies.forEach(e => e.draw(ctx));
            }
        }

        // ==================== 游戏主类 ====================
        class Game {
            constructor() {
                this.canvas = document.getElementById('gameCanvas');
                this.ctx = this.canvas.getContext('2d');
                this.resize();

                            color: color,

                    color: color,
                    count: 40

                    isCrit: true
                });

                this.battleSystem.totalDamage += damage;
                this.battleSystem.comboCount += 3;

                            color: '#ffd700',
                            width: 60,
                            life: 0.6

                        this.vfxManager.addHitSpark(enemy.x, enemy.y, {
                            color: '#ffd700',
                            count: 50
                        });
                        
                        const damage = this.player.atk * 5;
                        this.vfxManager.addDamageNumber(enemy.x, enemy.y - 30, damage, {
                            isCrit: true
                        });
                        
                        enemy.takeDamage(damage, true);
                        this.battleSystem.totalDamage += damage;
                    }, i * 150);
                });
                
                this.vfxManager.triggerScreenShake(20, 0.6);
                this.battleSystem.comboCount += 10;
            }

            initUI() {
                // 初始生成敌人
                for (let i = 0; i < 2; i++) {
                    this.battleSystem.spawnEnemy();
                }
            }

            update(dt) {
                this.background.update(dt);
                this.battleSystem.update(dt);

            }

            draw() {

                this.background.draw(this.ctx);
                this.battleSystem.draw(this.ctx);
                
                this.ctx.save();

            loop(currentTime) {
                if (!this.running) return;
                
                this.deltaTime = Math.min((currentTime - this.lastTime) / 1000, 0.1);
                this.lastTime = currentTime;
                
                this.update(this.deltaTime);
                this.draw();
                
                requestAnimationFrame((t) => this.loop(t));
            }

            start() {
                this.lastTime = performance.now();
                this.loop(this.lastTime);
            }
        }

## 功能说明

1. **挥砍轨迹**

   - 暴击时显示金色加粗轨迹

   - 重力影响的粒子飘散

   - 灰色：闪避提示
   - 弹性缩放动画 + 上浮淡出

4. **暴击屏幕震动**
   - 强度可调节的震动效果
   - 自动衰减恢复

   - 多层扩散光环

   - 光柱爆发
   - 环形波纹扩散
   - 金色星星粒子
