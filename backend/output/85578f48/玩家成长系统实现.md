        class AgarGame {

                this.canvas = document.getElementById('gameCanvas');

            generateRandomColor() {
                const hue = Math.floor(Math.random() * 360);
                return `hsl(${hue}, 70%, 50%)`;

            generateFood() {
                this.foods = [];
                for (let i = 0; i < CONFIG.FOOD_COUNT; i++) {
                    this.foods.push({

                        }],

                        ejectCooldown: 0
                    });
                }
            }
            
            handleMouseMove(e) {
                if (this.state !== GameState.PLAYING) return;
                
                const rect = this.canvas.getBoundingClientRect();
                this.mouse.x = e.clientX - rect.left;
                this.mouse.y = e.clientY - rect.top;
            }
            
            handleKeyDown(e) {
                this.keys[e.key] = true;
                
                if (e.key === ' ' && this.state === GameState.PLAYING) {
                    e.preventDefault();
                    this.splitPlayer();
                } else if (e.key.toLowerCase() === 'w' && this.state === GameState.PLAYING) {
                    this.ejectFood();
                } else if (e.key.toLowerCase() === 'p' && this.state === GameState.PLAYING) {
                    this.togglePause();
                }
            }
            
            handleKeyUp(e) {
                this.keys[e.key] = false;
            }
            
            splitPlayer() {
                if (this.state !== GameState.PLAYING) return;
                
                const now = Date.now();
                if (now - this.player.splitCooldown < CONFIG.SPLIT_COOLDOWN) return;
                
                // 只能在有足够大的细胞时分裂
                const mainCell = this.player.cells.find(cell => cell.id === 'main');
                if (!mainCell || mainCell.size < CONFIG.PLAYER_MIN_SIZE * 1.5) return;
                
                // 分裂成更小的细胞
                const newCells = [];
                const cellSize = mainCell.size * CONFIG.PLAYER_SPLIT_SIZE_FACTOR;
                
                // 保留主细胞，但缩小它
                mainCell.size = cellSize;
                
                // 创建新的细胞
                for (let i = 0; i < CONFIG.SPLIT_COUNT; i++) {
                    const angle = (Math.PI * 2 / CONFIG.SPLIT_COUNT) * i;
                    const distance = mainCell.size + cellSize + 5;
                    
                    newCells.push({
                        id: `split-${i}`,
                        x: mainCell.x + Math.cos(angle) * distance,
                        y: mainCell.y + Math.sin(angle) * distance,
                        size: cellSize,
                        dx: Math.cos(angle) * CONFIG.EJECT_FOOD_SPEED,
                        dy: Math.sin(angle) * CONFIG.EJECT_FOOD_SPEED
                    });
                }
                
                this.player.cells = this.player.cells.concat(newCells);
                this.player.splitCooldown = now;
                
                // 更新玩家大小
                this.updatePlayerSize();
            }
            
            ejectFood() {
                if (this.state !== GameState.PLAYING) return;
                
                const now = Date.now();
                if (now - this.player.ejectCooldown < CONFIG.EJECT_COOLDOWN) return;
                
                // 只能在有足够大的细胞时吐出食物
                const mainCell = this.player.cells.find(cell => cell.id === 'main');
                if (!mainCell || mainCell.size < CONFIG.PLAYER_MIN_SIZE + CONFIG.EJECT_FOOD_SIZE * 2) return;
                
                // 计算吐出方向
                const dx = this.mouse.x - this.canvas.width / 2;
                const dy = this.mouse.y - this.canvas.height / 2;
                const distance = Math.sqrt(dx * dx + dy * dy);
                
                if (distance === 0) return;
                
                const angle = Math.atan2(dy, dx);
                
                // 创建食物
                const ejectedFood = {
                    id: `ejected-${Date.now()}`,
                    x: mainCell.x,
                    y: mainCell.y,
                    size: CONFIG.EJECT_FOOD_SIZE,
                    color: this.generateRandomColor(),
                    dx: Math.cos(angle) * CONFIG.EJECT_FOOD_SPEED,
                    dy: Math.sin(angle) * CONFIG.EJECT_FOOD_SPEED,
                    isEjected: true
                };
                
                // 缩小主细胞
                mainCell.size -= CONFIG.EJECT_FOOD_SIZE;
                
                // 添加到食物数组
                this.foods.push(ejectedFood);
                
                this.player.ejectCooldown = now;
                
                // 更新玩家大小
                this.updatePlayerSize();
            }
            
            updatePlayerSize() {
                // 计算玩家总大小
                const totalSize = this.player.cells.reduce((sum, cell) => sum + cell.size, 0);
                this.player.size = totalSize;
                
                // 更新UI
                document.getElementById('size').textContent = Math.floor(totalSize);
            }
            
            togglePause() {
                if (this.state === GameState.PLAYING) {
                    this.state = GameState.PAUSED;
                } else if (this.state === GameState.PAUSED) {
                    this.state = GameState.PLAYING;
                }
            }
            
            updateCamera() {
                if (!this.player) return;
                
                // 计算目标相机位置
                const mainCell = this.player.cells.find(cell => cell.id === 'main');
                if (!mainCell) return;
                
                this.targetCamera.x = mainCell.x - this.canvas.width / 2;
                this.targetCamera.y = mainCell.y - this.canvas.height / 2;
                
                // 平滑移动相机
                this.camera.x += (this.targetCamera.x - this.camera.x) * CONFIG.CAMERA_SMOOTHNESS;
                this.camera.y += (this.targetCamera.y - this.camera.y) * CONFIG.CAMERA_SMOOTHNESS;
                
                // 限制相机范围
                this.camera.x = Math.max(0, Math.min(CONFIG.WORLD_WIDTH - this.canvas.width, this.camera.x));
                this.camera.y = Math.max(0, Math.min(CONFIG.WORLD_HEIGHT - this.canvas.height, this.camera.y));
            }

                if (!this.player || this.state !== GameState.PLAYING) return;
                
                const now = Date.now();
                
                // 更新冷却时间
