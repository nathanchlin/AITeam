        // 游戏配置

        // 游戏状态

        // 玩家类
        class Player {
            constructor(id, username, x, y, size, color) {
                this.id = id;
                this.username = username;
                this.x = x;
                this.y = y;
                this.size = size;
                this.color = color;
                this.targetX = x;
                this.targetY = y;
                this.speed = this.calculateSpeed();
                this.isAlive = true;
                this.score = 0;
                this.splitCooldown = 0;
                this.ejectCooldown = 0;
                this.ejectedMass = [];
            }

            calculateSpeed() {
                // 越大越慢
                return CONFIG.SPEED_FACTOR / Math.sqrt(this.size);
            }

            update(deltaTime) {
                if (!this.isAlive) return;

                // 更新冷却时间
                if (this.splitCooldown > 0) {
                    this.splitCooldown -= deltaTime;
                }
                if (this.ejectCooldown > 0) {
                    this.ejectCooldown -= deltaTime;
                }

                // 更新喷射物
                this.ejectedMass = this.ejectedMass.filter(mass => {
                    mass.x += mass.vx;
                    mass.y += mass.vy;
                    mass.size *= 0.98; // 逐渐缩小
                    
                    // 检查是否被其他玩家吃掉
                    for (const [id, player] of gameState.players) {
                        if (id !== this.id && player.isAlive) {
                            const dx = mass.x - player.x;
                            const dy = mass.y - player.y;
                            const distance = Math.sqrt(dx * dx + dy * dy);
                            
                            if (distance < player.size) {
                                // 玩家吃掉喷射物
                                player.size += mass.size * 0.5;
                                player.score += Math.floor(mass.size * CONFIG.SCORE_PER_FOOD);
                                updatePlayerStats();
                                return false; // 移除这个喷射物
                            }
                        }
                    }
                    
                    return mass.size > 1; // 保留足够大的喷射物
                });

                // 移动到目标位置
                const dx = this.targetX - this.x;
                const dy = this.targetY - this.y;
                const distance = Math.sqrt(dx * dx + dy * dy);

                if (distance > 1) {
                    const moveX = (dx / distance) * this.speed * deltaTime / 16;
                    const moveY = (dy / distance) * this.speed * deltaTime / 16;
                    
                    this.x += moveX;
                    this.y += moveY;
                    
                    // 边界检查
                    this.x = Math.max(this.size, Math.min(CONFIG.MAP_WIDTH - this.size, this.x));
                    this.y = Math.max(this.size, Math.min(CONFIG.MAP_HEIGHT - this.size, this.y));
                }
            }

            draw(ctx, cameraX, cameraY) {
                if (!this.isAlive) return;

                const screenX = this.x - cameraX;
                const screenY = this.y - cameraY;

                // 绘制球体
                ctx.beginPath();
                ctx.arc(screenX, screenY, this.size, 0, Math.PI * 2);
                ctx.fillStyle = this.color;
                ctx.fill();
                ctx.strokeStyle = 'rgba(0, 0, 0, 0.2)';
                ctx.lineWidth = 2;
                ctx.stroke();

                // 绘制用户名
                ctx.fillStyle = '#333';
                ctx.font = '12px Arial';
                ctx.textAlign = 'center';
                ctx.textBaseline = 'middle';
                ctx.fillText(this.username, screenX, screenY);

                // 绘制分裂冷却指示器
                if (this.splitCooldown > 0) {
                    ctx.beginPath();
                    ctx.arc(screenX, screenY, this.size + 5, 0, Math.PI * 2 * (1 - this.splitCooldown / 2000));
                    ctx.strokeStyle = 'rgba(255, 255, 255, 0.7)';
                    ctx.lineWidth = 2;
                    ctx.stroke();
                }

                // 绘制喷射物
                this.ejectedMass.forEach(mass => {
                    const massScreenX = mass.x - cameraX;
                    const massScreenY = mass.y - cameraY;
                    
                    ctx.beginPath();
                    ctx.arc(massScreenX, massScreenY, mass.size, 0, Math.PI * 2);
                    ctx.fillStyle = this.color;
                    ctx.fill();
                });
            }

            split() {
                if (this.splitCooldown > 0 || this.size < CONFIG.MIN_BALL_SIZE * 2) return;
                
                this.splitCooldown = 2000; // 2秒冷却时间
                
                // 创建新球体
                const newSize = this.size * CONFIG.SPLIT_FACTOR;
                const newPlayer = new Player(
                    Date.now().toString(),
                    this.username,
                    this.x + this.size,
                    this.y,
                    newSize,
                    this.color
                );
                
                // 更新当前玩家
                this.size = newSize;
                this.speed = this.calculateSpeed();
                
                // 添加到游戏状态
                gameState.players.set(newPlayer.id, newPlayer);
                
                // 如果是本地玩家，更新视图
                if (this.id === gameState.player.id) {
                    gameState.player = this;
                    updatePlayerStats();
                }
                
                // 发送分裂事件到服务器
                if (gameState.socket) {
                    gameState.socket.emit('playerSplit', {
                        playerId: this.id,
                        newPlayer: {
                            id: newPlayer.id,
                            x: newPlayer.x,
                            y: newPlayer.y,
                            size: newPlayer.size,
                            color: newPlayer.color
                        }
                    });
                }
                
                showNotification('分裂成功！');
            }

            eject() {
                if (this.ejectCooldown > 0 || this.size < CONFIG.MIN_BALL_SIZE + CONFIG.EJECT_SIZE) return;
                
                this.ejectCooldown = 1000; // 1秒冷却时间
                
                // 创建喷射物
                const ejectAngle = Math.atan2(
                    this.targetY - this.y,
                    this.targetX - this.x
                );
                
                const ejectedMass = {
                    id: Date.now().toString(),
                    x: this.x + Math.cos(ejectAngle) * this.size,
                    y: this.y + Math.sin(ejectAngle) * this.size,
                    size: CONFIG.EJECT_SIZE,
                    vx: Math.cos(ejectAngle) * CONFIG.EJECT_SPEED,
                    vy: Math.sin(ejectAngle) * CONFIG.EJECT_SPEED
                };
                
                this.ejectedMass.push(ejectedMass);
                this.size -= CONFIG.EJECT_SIZE;
                this.speed = this.calculateSpeed();
                
                // 更新视图
                if (this.id === gameState.player.id) {
                    gameState.player = this;
                    updatePlayerStats();
                }
                
                // 发送喷射事件到服务器
                if (gameState.socket) {
                    gameState.socket.emit('playerEject', {
                        playerId: this.id,
                        ejectedMass: ejectedMass
                    });
                }
            }

            eatFood(food) {
                // 增加大小和分数
                this.size += food.size * 0.1;
                this.score += Math.floor(food.size * CONFIG.SCORE_PER_FOOD);
                
                // 限制最大尺寸
                if (this.size > CONFIG.MAX_BALL_SIZE) {
                    this.size = CONFIG.MAX_BALL_SIZE;
                }
                
                // 更新速度
                this.speed = this.calculateSpeed();
                
                // 显示得分弹出
                showScorePopup(food.x, food.y, `+${Math.floor(food.size * CONFIG.SCORE_PER_FOOD)}`);
                
                // 更新视图
                if (this.id === gameState.player.id) {
                    gameState.player = this;
                    updatePlayerStats();
                }
            }
