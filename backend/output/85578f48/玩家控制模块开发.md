        // 球体类
        class Ball {
            constructor(x, y, radius, color, isPlayer = false, name = '') {
                this.x = x;
                this.y = y;
                this.radius = radius;
                this.color = color;
                this.isPlayer = isPlayer;
                this.name = name;
                this.vx = 0;
                this.vy = 0;
                this.targetX = x;
                this.targetY = y;
                this.speed = 0;
                this.mass = Math.PI * radius * radius;
            }

            update(deltaTime) {
                // 计算速度
                const dx = this.targetX - this.x;
                const dy = this.targetY - this.y;
                const distance = Math.sqrt(dx * dx + dy * dy);
                
                if (distance > 1) {
                    // 根据大小计算速度
                    const baseSpeed = CONFIG.PLAYER_SPEED_FACTOR * 1000;
                    this.speed = baseSpeed / Math.sqrt(this.radius);
                    
                    // 计算速度分量
                    this.vx = (dx / distance) * this.speed * deltaTime / 1000;
                    this.vy = (dy / distance) * this.speed * deltaTime / 1000;
                    
                    // 更新位置
                    this.x += this.vx;
                    this.y += this.vy;
                } else {
                    this.vx = 0;
                    this.vy = 0;
                }
                
                // 边界检查
                this.x = Math.max(this.radius, Math.min(CONFIG.WORLD_WIDTH - this.radius, this.x));
                this.y = Math.max(this.radius, Math.min(CONFIG.WORLD_HEIGHT - this.radius, this.y));
            }

            draw(context, cameraX, cameraY) {
                const screenX = this.x - cameraX;
                const screenY = this.y - cameraY;
                
                // 只绘制在屏幕内的球体
                if (screenX + this.radius > 0 && screenX - this.radius < canvas.width &&
                    screenY + this.radius > 0 && screenY - this.radius < canvas.height) {
                    
                    // 绘制球体
                    context.beginPath();
                    context.arc(screenX, screenY, this.radius, 0, Math.PI * 2);
                    context.fillStyle = this.color;
                    context.fill();
                    
                    // 绘制边框
                    context.strokeStyle = 'rgba(0, 0, 0, 0.2)';
                    context.lineWidth = 2;
                    context.stroke();
                    
                    // 如果是玩家球体，绘制名字
                    if (this.isPlayer) {
                        context.fillStyle = 'white';
                        context.font = `bold ${Math.max(12, this.radius / 4)}px Arial`;
                        context.textAlign = 'center';
                        context.textBaseline = 'middle';
                        context.fillText(this.name, screenX, screenY);
                    }
                }
            }

            canEat(other) {
                if (this === other) return false;
                if (!this.isPlayer && !other.isPlayer) return false;
                
                const distance = Math.sqrt(
                    Math.pow(this.x - other.x, 2) + 
                    Math.pow(this.y - other.y, 2)
                );
                
                return distance < this.radius && this.radius > other.radius * 1.1;
            }

            eat(other) {
                // 计算新的大小
                const newMass = this.mass + other.mass * 0.8;
                this.radius = Math.sqrt(newMass / Math.PI);
                this.mass = newMass;
                
                // 限制最大大小
                if (this.radius > CONFIG.MAX_PLAYER_SIZE) {
                    this.radius = CONFIG.MAX_PLAYER_SIZE;
                    this.mass = Math.PI * this.radius * this.radius;
                }
            }

            split() {
                if (this.radius < CONFIG.SPLIT_MIN_SIZE) return null;
                
                // 创建一个新的球体
                const newRadius = this.radius / 2;
                const angle = Math.random() * Math.PI * 2;
                const distance = this.radius + newRadius;
                
                const newBall = new Ball(
                    this.x + Math.cos(angle) * distance,
                    this.y + Math.sin(angle) * distance,
                    newRadius,
                    this.color,
                    this.isPlayer,
                    this.name
                );
                
                // 设置分裂后的速度
                const speed = CONFIG.SPLIT_SPEED_FACTOR * 1000;
                newBall.vx = Math.cos(angle) * speed * gameState.deltaTime / 1000;
                newBall.vy = Math.sin(angle) * speed * gameState.deltaTime / 1000;
                
                // 更新原球体
                this.radius = newRadius;
                this.mass = Math.PI * this.radius * this.radius;
                
                return newBall;
            }

            eject() {
                if (this.radius < CONFIG.MIN_PLAYER_SIZE + CONFIG.EJECT_SIZE) return null;
                
                // 减少玩家大小
                this.radius -= CONFIG.EJECT_SIZE;
                this.mass = Math.PI * this.radius * this.radius;
                
                // 创建吐出的球体
                const angle = Math.atan2(
                    gameState.mouse.worldY - this.y,
                    gameState.mouse.worldX - this.x
                );
                
                const ejectedBall = new Ball(
                    this.x + Math.cos(angle) * (this.radius + CONFIG.EJECT_SIZE),
                    this.y + Math.sin(angle) * (this.radius + CONFIG.EJECT_SIZE),
                    CONFIG.EJECT_SIZE,
                    this.color,
                    false
                );
                
                // 设置吐出球的速度
                ejectedBall.vx = Math.cos(angle) * CONFIG.EJECT_SPEED;
                ejectedBall.vy = Math.sin(angle) * CONFIG.EJECT_SPEED;
                
                return ejectedBall;
            }

            // 计算时间差
