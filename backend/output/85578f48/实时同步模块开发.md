        // 实时同步模块
        class RealtimeSync {
            constructor() {
                this.stateBuffer = [];
                this.lastProcessedTime = 0;
                this.maxBufferSize = 100;
                this.reconciliationThreshold = 100; // ms
                this.predictionEnabled = true;
                this.smoothingFactor = 0.2;
            }

            // 收集本地状态
            collectLocalState(player) {
                return {
                    id: player.id,
                    x: player.x,
                    y: player.y,
                    size: player.size,
                    score: player.score,
                    timestamp: Date.now()
                };
            }

            // 应用远程状态
            applyRemoteState(remoteState) {
                const localPlayer = gameState.player;
                if (!localPlayer || remoteState.id === localPlayer.id) return;

                // 检查是否需要状态更新
                const timeDiff = remoteState.timestamp - this.lastProcessedTime;
                if (timeDiff < this.reconciliationThreshold) return;

                // 更新玩家状态
                const player = gameState.players.get(remoteState.id);
                if (player) {
                    // 平滑插值
                    if (this.predictionEnabled) {
                        player.x += (remoteState.x - player.x) * this.smoothingFactor;
                        player.y += (remoteState.y - player.y) * this.smoothingFactor;
                    } else {
                        player.x = remoteState.x;
                        player.y = remoteState.y;
                    }
                    
                    player.size = remoteState.size;
                    player.score = remoteState.score;
                }

                this.lastProcessedTime = remoteState.timestamp;
            }

            // 状态冲突解决
            resolveConflicts(localState, remoteState) {
                // 基于时间戳和玩家大小决定优先级
                if (remoteState.timestamp > localState.timestamp) {
                    // 远程更新更新，应用远程状态
                    this.applyRemoteState(remoteState);
                    
                    // 如果本地状态更新时间更近，保留本地位置
                    if (localState.timestamp > remoteState.timestamp - 100) {
                        const player = gameState.players.get(remoteState.id);
                        if (player && player.id === gameState.player.id) {
                            player.x = localState.x;
                            player.y = localState.y;
                        }
                    }
                }
            }

            // 状态插值
            interpolateStates(player) {
                // 简单的位置插值
                if (player.targetX !== undefined && player.targetY !== undefined) {
                    player.x += (player.targetX - player.x) * 0.1;
                    player.y += (player.targetY - player.y) * 0.1;
                }
            }

            // 添加状态到缓冲区
            addToBuffer(state) {
                this.stateBuffer.push(state);
                
                // 限制缓冲区大小
                if (this.stateBuffer.length > this.maxBufferSize) {
                    this.stateBuffer.shift();
                }
            }

            // 处理状态缓冲区
            processBuffer() {
                const now = Date.now();
                
                // 处理缓冲区中的所有状态
                while (this.stateBuffer.length > 0) {
                    const state = this.stateBuffer.shift();
                    
                    // 检查是否过期
                    if (now - state.timestamp > 1000) continue;
                    
                    // 应用状态
                    this.applyRemoteState(state);
                }
            }
        }

            // 发送消息到服务器
            sendToServer(type, data) {
                if (!this.isConnected) {
                    // 如果未连接，将消息加入队列
                    this.messageQueue.push({ type, data });
                    return;
                }

                // 发送消息
                this.socket.emit(type, data);
            }

            // 处理消息队列
            processMessageQueue() {
                while (this.messageQueue.length > 0 && this.isConnected) {
                    const message = this.messageQueue.shift();
                    this.sendToServer(message.type, message.data);
                }
            }

            // 调度重连
            scheduleReconnect() {
                if (this.reconnectTimer) return;
                
                this.reconnectAttempts++;
                
                if (this.reconnectAttempts > this.maxReconnectAttempts) {
                    console.error('达到最大重连次数');
                    showError('无法连接到服务器，请检查网络连接');
                    return;
                }

                const delay = Math.min(1000 * Math.pow(2, this.reconnectAttempts), 30000);
                console.log(`将在 ${delay}ms 后尝试重新连接...`);

                this.reconnectTimer = setTimeout(() => {
                    console.log('尝试重新连接...');
                    this.socket.connect();
                    this.reconnectTimer = null;
                }, delay);
            }
