    <script>
        // 游戏配置
        const CONFIG = {
            WORLD_WIDTH: 3000,
            WORLD_HEIGHT: 3000,
            FOOD_COUNT: 100,
            FOOD_MIN_SIZE: 5,
            FOOD_MAX_SIZE: 15,
            PLAYER_MIN_SIZE: 20,
            PLAYER_MAX_SIZE: 200,
            PLAYER_SPEED_FACTOR: 0.2,
            PLAYER_SPLIT_SIZE_FACTOR: 0.6,
            FOOD_COLOR: ['#FF5252', '#FFD740', '#69F0AE', '#448AFF', '#E040FB'],
            PLAYER_COLORS: ['#FF5252', '#FF4081', '#E040FB', '#7C4DFF', '#536DFE', '#448AFF', '#40C4FF', '#18FFFF', '#64FFDA', '#69F0AE', '#B2FF59', '#EEFF41', '#FFFF00', '#FFD740', '#FFAB40', '#FF6E40']
        };

        // 游戏状态
        const gameState = {
            world: {
                width: CONFIG.WORLD_WIDTH,
                height: CONFIG.WORLD_HEIGHT
            },
            camera: {
                x: 0,
                y: 0,
                width: 0,
                height: 0
            },
            foods: [],
            players: [],
            currentPlayer: null,
            leaderboard: [],
            isRunning: false,
            isPaused: false,
            keys: {},
            mouse: {
                x: 0,
                y: 0
            }
        };

        // 食物类
        class Food {
            constructor(x, y, size) {
                this.x = x;
                this.y = y;
                this.size = size || Math.random() * (CONFIG.FOOD_MAX_SIZE - CONFIG.FOOD_MIN_SIZE) + CONFIG.FOOD_MIN_SIZE;
                this.color = CONFIG.FOOD_COLOR[Math.floor(Math.random() * CONFIG.FOOD_COLOR.length)];
                this.id = Math.random().toString(36).substr(2, 9);
            }

            draw(ctx, camera) {
                const screenX = this.x - camera.x + camera.width / 2;
                const screenY = this.y - camera.y + camera.height / 2;
                
                // 只绘制在屏幕范围内的食物
                if (screenX + this.size > 0 && screenX - this.size < camera.width &&
                    screenY + this.size > 0 && screenY - this.size < camera.height) {
                    
                    ctx.beginPath();
                    ctx.arc(screenX, screenY, this.size, 0, Math.PI * 2);
                    ctx.fillStyle = this.color;
                    ctx.fill();
                    
                    // 添加光泽效果
                    ctx.beginPath();
                    ctx.arc(screenX - this.size * 0.3, screenY - this.size * 0.3, this.size * 0.2, 0, Math.PI * 2);
                    ctx.fillStyle = 'rgba(255, 255, 255, 0.5)';
                    ctx.fill();
                }
            }
        }

        // 玩家类
        class Player {
            constructor(name, x, y, size, color) {
                this.name = name;
                this.x = x;
                this.y = y;
                this.size = size || CONFIG.PLAYER_MIN_SIZE;
                this.color = color || CONFIG.PLAYER_COLORS[Math.floor(Math.random() * CONFIG.PLAYER_COLORS.length)];
                this.speed = this.calculateSpeed();
                this.targetX = x;
                this.targetY = y;
                this.score = 0;
                this.alive = true;
                this.splitCooldown = 0;
                this.ejectCooldown = 0;
                this.ejects = [];
                this.id = Math.random().toString(36).substr(2, 9);
            }

            calculateSpeed() {
                // 速度与大小成反比
                return Math.max(1, (CONFIG.PLAYER_MAX_SIZE - this.size) * CONFIG.PLAYER_SPEED_FACTOR);
            }

            update(deltaTime) {
                if (!this.alive) return;

                // 更新冷却时间
                if (this.splitCooldown > 0) {
                    this.splitCooldown -= deltaTime;
                }
                if (this.ejectCooldown > 0) {
                    this.ejectCooldown -= deltaTime;
                }

                // 更新喷射物
                this.ejects = this.ejects.filter(eject => {
                    eject.update(deltaTime);
                    return eject.life > 0;
                });

                // 平滑移动到目标位置
                const dx = this.targetX - this.x;
                const dy = this.targetY - this.y;
                const distance = Math.sqrt(dx * dx + dy * dy);
                
                if (distance > 0.1) {
                    const moveX = (dx / distance) * this.speed;
                    const moveY = (dy / distance) * this.speed;
                    
                    this.x += moveX;
                    this.y += moveY;
                    
                    // 边界检查
                    this.x = Math.max(this.size, Math.min(gameState.world.width - this.size, this.x));
                    this.y = Math.max(this.size, Math.min(gameState.world.height - this.size, this.y));
                }
            }

            draw(ctx, camera) {
                if (!this.alive) return;

                const screenX = this.x - camera.x + camera.width / 2;
                const screenY = this.y - camera.y + camera.height / 2;

                // 只绘制在屏幕范围内的玩家
                if (screenX + this.size > -camera.width && screenX - this.size < camera.width * 2 &&
                    screenY + this.size > -camera.height && screenY - this.size < camera.height * 2) {
                    
                    // 绘制玩家球体
                    ctx.beginPath();
                    ctx.arc(screenX, screenY, this.size, 0, Math.PI * 2);
                    ctx.fillStyle = this.color;
                    ctx.fill();
                    
                    // 添加边框
                    ctx.strokeStyle = 'rgba(0, 0, 0, 0.2)';
                    ctx.lineWidth = 2;
                    ctx.stroke();
                    
                    // 添加光泽效果
                    ctx.beginPath();
                    ctx.arc(screenX - this.size * 0.3, screenY - this.size * 0.3, this.size * 0.2, 0, Math.PI * 2);
                    ctx.fillStyle = 'rgba(255, 255, 255, 0.5)';
                    ctx.fill();
                    
                    // 绘制玩家名称
                    ctx.fillStyle = '#333';
                    ctx.font = `${Math.max(12, this.size / 3)}px Arial`;
                    ctx.textAlign = 'center';
                    ctx.textBaseline = 'middle';
                    ctx.fillText(this.name, screenX, screenY);
                }

                // 绘制喷射物
                this.ejects.forEach(eject => eject.draw(ctx, camera));
            }

            split() {
                if (this.splitCooldown > 0 || this.size < CONFIG.PLAYER_MIN_SIZE * 2) return;
                
                const newSize = this.size * CONFIG.PLAYER_SPLIT_SIZE_FACTOR;
                if (newSize < CONFIG.PLAYER_MIN_SIZE) return;
                
                this.size = newSize;
                this.speed = this.calculateSpeed();
                this.splitCooldown = 1000; // 1秒冷却
                
                // 创建分裂后的新球
                const angle = Math.atan2(this.targetY - this.y, this.targetX - this.x);
                const newX = this.x + Math.cos(angle) * this.size * 2;
                const newY = this.y + Math.sin(angle) * this.size * 2;
                
                const newPlayer = new Player(this.name, newX, newY, newSize, this.color);
                newPlayer.score = this.score / 2;
                this.score = this.score / 2;
                
                gameState.players.push(newPlayer);
                
                return newPlayer;
            }

            eject() {
                if (this.ejectCooldown > 0) return;
                
                this.ejectCooldown = 500; // 0.5秒冷却
                
                // 创建喷射物
                const angle = Math.atan2(this.targetY - this.y, this.targetX - this.x);
                const ejectX = this.x + Math.cos(angle) * this.size;
                const ejectY = this.y + Math.sin(angle) * this.size;
                
                const eject = new Eject(ejectX, ejectY, angle, this.color);
                this.ejects.push(eject);
                
                // 反向推力
                this.x -= Math.cos(angle) * 5;
                this.y -= Math.sin(angle) * 5;
            }
        }

        // 喷射物类
        class Eject {
            constructor(x, y, angle, color) {
                this.x = x;
                this.y = y;
                this.angle = angle;
                this.size = 5;
                this.color = color;
                this.speed = 15;
                this.life = 1000; // 1秒生命周期
            }

            update(deltaTime) {
                this.x += Math.cos(this.angle) * this.speed;
                this.y += Math.sin(this.angle) * this.speed;
                this.life -= deltaTime;
                
                // 边界检查
                if (this.x < 0 || this.x > gameState.world.width || 
                    this.y < 0 || this.y > gameState.world.height) {
                    this.life = 0;
                }
            }

            draw(ctx, camera) {
                if (this.life <= 0) return;
                
                const screenX = this.x - camera.x + camera.width / 2;
                const screenY = this.y - camera.y + camera.height / 2;
                
                ctx.beginPath();
                ctx.arc(screenX, screenY, this.size, 0, Math.PI * 2);
                ctx.fillStyle = this.color;
                ctx.fill();
                
                // 添加光泽效果
                ctx.beginPath();
                ctx.arc(screenX - this.size * 0.3, screenY - this.size * 0.3, this.size * 0.3, 0, Math.PI * 2);
                ctx.fillStyle = 'rgba(255, 255, 255, 0.7)';
                ctx.fill();
            }
        }

        // AI玩家类
        class AIPlayer extends Player {
            constructor(name, x, y, size, color) {
                super(name, x, y, size, color);
                this.targetFood = null;
                this.direction = Math.random() * Math.PI * 2;
                this.directionChangeTimer = 0;
            }

            updateAI(deltaTime) {
                if (!this.alive) return;
                
                // 寻找最近的食物
                if (!this.targetFood || this.targetFood.eaten) {
                    let closestFood = null;
                    let closestDistance = Infinity;
                    
                    gameState.foods.forEach(food => {
                        const dx = food.x - this.x;
                        const dy = food.y - this.y;
                        const distance = Math.sqrt(dx * dx + dy * dy);
                        
                        if (distance < closestDistance) {
                            closestDistance = distance;
                            closestFood = food;
                        }
                    });
                    
                    this.targetFood = closestFood;
                }
                
                // 更新移动方向
                this.directionChangeTimer -= deltaTime;
                
                if (this.directionChangeTimer <= 0 || this.targetFood) {
                    if (this.targetFood) {
                        // 向食物移动
                        this.targetX = this.targetFood.x;
                        this.targetY = this.targetFood.y;
                    } else {
                        // 随机移动
                        this.direction = Math.random() * Math.PI * 2;
                        this.targetX = this.x + Math.cos(this.direction) * 200;
                        this.targetY = this.y + Math.sin(this.direction) * 200;
                        
                        // 确保目标在世界范围内
                        this.targetX = Math.max(this.size, Math.min(gameState.world.width - this.size, this.targetX));
                        this.targetY = Math.max(this.size, Math.min(gameState.world.height - this.size, this.targetY));
                    }
                    
                    this.directionChangeTimer = 1000 + Math.random() * 2000; // 1-3秒改变一次方向
                }
                
                // 随机分裂
                if (Math.random() < 0.001 && this.size > CONFIG.PLAYER_MIN_SIZE * 2) {
                    this.split();
                }
                
                // 随机喷射
                if (Math.random() < 0.01) {
                    this.eject();
                }
            }
        }
