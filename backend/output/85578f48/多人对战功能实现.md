                document.getElementById('serverStatus').style.color = '#F44336';

        // 更新游戏状态
        function updateGameState(data) {
            if (data.player) {
                gameState.player = data.player;
                updateCamera();
            }
            
            // 更新其他玩家
            gameState.players = new Map();
            data.players.forEach(p => {
                if (p.id !== gameState.player?.id) {
                    gameState.players.set(p.id, p);
                }
            });
            
            // 更新食物
            gameState.foods = data.foods || [];
            
            // 更新喷射物
            gameState.pellets = data.pellets || [];
            
            // 更新排行榜
            gameState.leaderboard = data.leaderboard || [];

            // 更新玩家分数和排名
            if (gameState.player) {
                document.getElementById('score').textContent = Math.floor(gameState.player.score);
                document.getElementById('rank').textContent = gameState.player.rank || '-';
            }
        }
        
        // 更新相机位置
        function updateCamera() {
            if (!gameState.player) return;
            
            if (gameState.isSpectating && gameState.spectatedPlayerId) {
                const spectatedPlayer = gameState.players.get(gameState.spectatedPlayerId) || gameState.player;
                camera.x = spectatedPlayer.x - canvas.width / 2;
                camera.y = spectatedPlayer.y - canvas.height / 2;
            } else {
                camera.x = gameState.player.x - canvas.width / 2;
                camera.y = gameState.player.y - canvas.height / 2;
            }
            
            // 限制相机在地图范围内
            camera.x = Math.max(0, Math.min(MAP_WIDTH - canvas.width, camera.x));
            camera.y = Math.max(0, Math.min(MAP_HEIGHT - canvas.height, camera.y));
        }

            // 处理输入

            // 更新游戏状态
            updateGameStateFromInput();
            
            // 渲染游戏
            render();
            
            // 继续循环
            requestAnimationFrame(gameLoop);
        }
        
        // 处理输入
        function handleInput() {
            if (!gameState.player || gameState.isSpectating) return;
            
            // 计算鼠标方向
            const dx = mouse.x - canvas.width / 2;
            const dy = mouse.y - canvas.height / 2;
            const distance = Math.sqrt(dx * dx + dy * dy);
            
            if (distance > 0) {
                // 发送移动方向到服务器
                const angle = Math.atan2(dy, dx);
                socket.emit('playerMove', { angle });
            }
        }
