        // 球体类
        class Ball {
            constructor(x, y, radius, color, isPlayer = false) {
                this.x = x;
                this.y = y;
                this.radius = radius;
                this.color = color;
                this.isPlayer = isPlayer;
                this.vx = 0;
                this.vy = 0;
                this.targetX = x;
                this.targetY = y;
                this.name = isPlayer ? gameState.player.name : 'AI';
                this.score = 0;
            }

            update() {
                if (this.isPlayer) {
                    // 玩家控制逻辑
                    const dx = gameState.mouseX - CONFIG.VIEWPORT_WIDTH / 2;
                    const dy = gameState.mouseY - CONFIG.VIEWPORT_HEIGHT / 2;
                    const distance = Math.sqrt(dx * dx + dy * dy);
                    
                    if (distance > 5) {
                        const speed = CONFIG.PLAYER_SPEED * (gameState.isSpeedBoosting ? CONFIG.SPEED_BOOST_MULTIPLIER : 1);
                        this.vx = (dx / distance) * speed;
                        this.vy = (dy / distance) * speed;
                    } else {
                        this.vx = 0;
                        this.vy = 0;
                    }
                } else {
                    // AI控制逻辑
                    const speed = CONFIG.AI_SPEED;
                    const angle = Math.random() * Math.PI * 2;
                    this.vx = Math.cos(angle) * speed;
                    this.vy = Math.sin(angle) * speed;
                    
                    // AI寻找食物
                    const nearestFood = this.findNearestFood();
                    if (nearestFood) {
                        const dx = nearestFood.x - this.x;
                        const dy = nearestFood.y - this.y;
                        const distance = Math.sqrt(dx * dx + dy * dy);
                        
                        if (distance < 200) {
                            this.vx = (dx / distance) * speed;
                            this.vy = (dy / distance) * speed;
                        }
                    }
                    
                    // 避免边界
                    if (this.x < 100 || this.x > CONFIG.WORLD_WIDTH - 100) {
                        this.vx *= -1;
                    }
                    if (this.y < 100 || this.y > CONFIG.WORLD_HEIGHT - 100) {
                        this.vy *= -1;
                    }
                }
                
                // 更新位置
                this.x += this.vx;
                this.y += this.vy;
                
                // 边界检查
                this.x = Math.max(this.radius, Math.min(CONFIG.WORLD_WIDTH - this.radius, this.x));
                this.y = Math.max(this.radius, Math.min(CONFIG.WORLD_HEIGHT - this.radius, this.y));
                
                // 更新分数
                this.score = Math.floor(this.radius - CONFIG.PLAYER_MIN_SIZE);
            }

            findNearestFood() {
                let nearestFood = null;
                let minDistance = Infinity;
                
                for (const food of gameState.foods) {
                    const dx = food.x - this.x;
                    const dy = food.y - this.y;
                    const distance = Math.sqrt(dx * dx + dy * dy);
                    
                    if (distance < minDistance) {
                        minDistance = distance;
                        nearestFood = food;
                    }
                }
                
                return nearestFood;
            }

            draw(context, offsetX = 0, offsetY = 0) {
                context.beginPath();
                context.arc(this.x - offsetX, this.y - offsetY, this.radius, 0, Math.PI * 2);
                context.fillStyle = this.color;
                context.fill();
                
                // 绘制边框
                context.strokeStyle = this.isPlayer ? '#000' : '#fff';
                context.lineWidth = 2;
                context.stroke();
                
                // 绘制名字
                if (this.radius > 15) {
                    context.fillStyle = '#fff';
                    context.font = `${Math.max(12, this.radius / 4)}px Arial`;
                    context.textAlign = 'center';
                    context.textBaseline = 'middle';
                    context.fillText(this.name, this.x - offsetX, this.y - offsetY);
                }
            }

            canEat(other) {
                const dx = this.x - other.x;
                const dy = this.y - other.y;
                const distance = Math.sqrt(dx * dx + dy * dy);
                return distance < this.radius && this.radius > other.radius * 1.1;
            }

            eat(other) {
                // 计算新的大小
                const area = Math.PI * this.radius * this.radius + Math.PI * other.radius * other.radius;
                this.radius = Math.sqrt(area / Math.PI);
                this.radius = Math.min(this.radius, CONFIG.PLAYER_MAX_SIZE);
                
                // 更新分数
                this.score = Math.floor(this.radius - CONFIG.PLAYER_MIN_SIZE);
                
                // 如果是玩家，更新UI
                if (this.isPlayer) {
                    document.getElementById('score').textContent = this.score;
                    document.getElementById('size').textContent = Math.floor(this.radius);
                }
            }

            split() {
                if (this.radius > CONFIG.PLAYER_MIN_SIZE * 2) {
                    const newRadius = this.radius * CONFIG.PLAYER_SPLIT_SIZE_RATIO;
                    this.radius = newRadius;
                    
                    // 创建新球
                    const angle = Math.random() * Math.PI * 2;
                    const distance = this.radius * 2;
                    const newX = this.x + Math.cos(angle) * distance;
                    const newY = this.y + Math.sin(angle) * distance;
                    
                    const newBall = new Ball(
                        newX,
                        newY,
                        newRadius,
                        this.color,
                        this.isPlayer
                    );
                    
                    // 设置新球的速度
                    newBall.vx = Math.cos(angle) * CONFIG.PLAYER_SPEED * 2;
                    newBall.vy = Math.sin(angle) * CONFIG.PLAYER_SPEED * 2;
                    
                    // 更新分数
                    this.score = Math.floor(this.radius - CONFIG.PLAYER_MIN_SIZE);
                    if (this.isPlayer) {
                        document.getElementById('score').textContent = this.score;
                        document.getElementById('size').textContent = Math.floor(this.radius);
                    }
                    
                    return newBall;
                }
                return null;
            }
        }

        // 食物类
        class Food {
            constructor(x, y) {
                this.x = x;
                this.y = y;
                this.radius = Math.random() * (CONFIG.FOOD_MAX_SIZE - CONFIG.FOOD_MIN_SIZE) + CONFIG.FOOD_MIN_SIZE;
                this.color = this.getRandomColor();
            }

            getRandomColor() {
                const colors = ['#FF5252', '#FF4081', '#E040FB', '#7C4DFF', '#536DFE', '#448AFF', '#40C4FF', '#18FFFF', '#64FFDA', '#69F0AE', '#B2FF59', '#EEFF41', '#FFFF00', '#FFD740', '#FFAB40'];
                return colors[Math.floor(Math.random() * colors.length)];
            }

            draw(context, offsetX = 0, offsetY = 0) {
                context.beginPath();
                context.arc(this.x - offsetX, this.y - offsetY, this.radius, 0, Math.PI * 2);
                context.fillStyle = this.color;
                context.fill();
            }
        }

            // 生成AI玩家

                gameState.aiPlayers.push(new Ball
