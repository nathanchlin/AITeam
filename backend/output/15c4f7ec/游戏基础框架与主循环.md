                this.state = GameState.PLAYING;

            }
            
            handleKeyDown(e) {
                this.keys[e.key] = true;
                
                // 防止方向键滚动页面
                if (['ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight', ' '].includes(e.key)) {
                    e.preventDefault();
                }

            handleKeyUp(e) {
                this.keys[e.key] = false;
            }
            
            update(deltaTime) {
                if (this.state !== GameState.PLAYING) return;
                
                // 更新玩家
                this.player.update(this.keys, deltaTime);
                
                // 玩家射击
                if (this.keys[' ']) {
                    const bullet = this.player.shoot();
                    if (bullet) {
                        this.bullets.push(bullet);
                    }
                }
                
                // 更新子弹
                this.bullets = this.bullets.filter(bullet => {
                    bullet.update(deltaTime);
                    return bullet.y > -10;
                });
                
                // 更新敌人
                this.enemies = this.enemies.filter(enemy => {
                    enemy.update(deltaTime);
                    return enemy.y < this.height + 50;
                });
                
                // 更新敌人子弹
                this.enemyBullets = this.enemyBullets.filter(bullet => {
                    bullet.update(deltaTime);
                    return bullet.y < this.height + 10;
                });
                
                // 更新粒子效果
                this.particles = this.particles.filter(particle => {
                    particle.update(deltaTime);
                    return particle.life > 0;
                });
                
                // 生成敌人
                this.enemySpawnTimer += deltaTime;
                if (this.enemySpawnTimer >= this.enemySpawnInterval) {
                    this.spawnEnemy();
                    this.enemySpawnTimer = 0;
                    // 随着游戏进行，敌人生成速度加快
                    this.enemySpawnInterval = Math.max(300, 1000 - this.score / 100);
                }
                
                // 碰撞检测
                this.checkCollisions();
                
                // 更新UI
                document.getElementById('score').textContent = this.score;
                document.getElementById('lives').textContent = this.lives;
                
                // 检查游戏结束
                if (this.lives <= 0) {
                    this.gameOver();
                }
            }
            
            spawnEnemy() {
                const x = Math.random() * (this.width - 40) + 20;
                const enemy = new Enemy(x, -50);
                this.enemies.push(enemy);
            }
            
            checkCollisions() {
                // 玩家子弹与敌人碰撞
                for (let i = this.bullets.length - 1; i >= 0; i--) {
                    for (let j = this.enemies.length - 1; j >= 0; j--) {
                        if (this.checkCollision(this.bullets[i], this.enemies[j])) {
                            // 创建爆炸效果
                            this.createExplosion(this.enemies[j].x, this.enemies[j].y);
                            
                            // 增加分数
                            this.score += 100;
                            
                            // 移除子弹和敌人
                            this.bullets.splice(i, 1);
                            this.enemies.splice(j, 1);
                            
                            break;
                        }
                    }
                }
                
                // 敌人与玩家碰撞
                for (let i = this.enemies.length - 1; i >= 0; i--) {
                    if (this.checkCollision(this.player, this.enemies[i])) {
                        // 创建爆炸效果
                        this.createExplosion(this.player.x, this.player.y);
                        this.createExplosion(this.enemies[i].x, this.enemies[i].y);
                        
                        // 减少生命
                        this.lives--;
                        
                        // 移除敌人
                        this.enemies.splice(i, 1);
                        
                        // 检查游戏是否结束
                        if (this.lives <= 0) {
                            this.gameOver();
                        }
                    }
                }
                
                // 敌人子弹与玩家碰撞
                for (let i = this.enemyBullets.length - 1; i >= 0; i--) {
                    if (this.checkCollision(this.player, this.enemyBullets[i])) {
                        // 创建爆炸效果
                        this.createExplosion(this.player.x, this.player.y);
                        
                        // 减少生命
                        this.lives--;
                        
                        // 移除子弹
                        this.enemyBullets.splice(i, 1);
                        
                        // 检查游戏是否结束
                        if (this.lives <= 0) {
                            this.gameOver();
                        }
                    }
                }
            }
            
            checkCollision(obj1, obj2) {
                const dx = obj1.x - obj2.x;
                const dy = obj1.y - obj2.y;
                const distance = Math.sqrt(dx * dx + dy * dy);
                return distance < (obj1.radius + obj2.radius);
            }
            
            createExplosion(x, y) {
                for (let i = 0; i < 15; i++) {
                    const particle = new Particle(x, y);
                    this.particles.push(particle);
                }
            }
            
            gameOver() {
                this.state = GameState.GAME_OVER;
                document.getElementById('finalScore').textContent = this.score;
                document.getElementById('gameOver').style.display = 'block';
            }
            
            render() {
                // 清空画布
                this.ctx.fillStyle = '#000';
                this.ctx.fillRect(0, 0, this.width, this.height);
                
                // 绘制星空背景
                this.drawStars();
                
                if (this.state === GameState.PLAYING || this.state === GameState.PAUSED) {
                    // 绘制玩家
                    this.player.draw(this.ctx);
                    
                    // 绘制子弹
                    this.bullets.forEach(bullet => bullet.draw(this.ctx));
                    
                    // 绘制敌人
                    this.enemies.forEach(enemy => enemy.draw(this.ctx));
                    
                    // 绘制敌人子弹
                    this.enemyBullets.forEach(bullet => bullet.draw(this.ctx));
                    
                    // 绘制粒子效果
                    this.particles.forEach(particle => particle.draw(this.ctx));
                }
            }
            
            drawStars() {
                this.ctx.fillStyle = '#fff';
                for (let i = 0; i < 50; i++) {
                    const x = (i * 73) % this.width;
                    const y = (i * 37 + Date.now() * 0.05) % this.height;
                    this.ctx.fillRect(x, y, 1, 1);
                }
            }
            
            gameLoop(currentTime = 0) {
                const deltaTime = currentTime - this.lastTime;
                this.lastTime = currentTime;
                
                this.update(deltaTime);
                this.render();
                
                if (this.state === GameState.PLAYING) {
                    requestAnimationFrame((time) => this.gameLoop(time));
                }
            }
        }

        class Player {
            constructor(x, y) {
                this.x = x;
                this.y = y;
                this.width = 40;
                this.height = 50;
                this.speed = 300; // 像素/秒
                this.radius = 20;
                this.shootCooldown = 0;

            update(keys, deltaTime) {

                    this.x -= this.speed * deltaTime / 1000;

                }
                if (keys['ArrowDown'] && this.y < game.height - this.radius) {
                    this.y += this.speed * deltaTime / 1000;

                if (this.shootCooldown > 0) {
                    this.shootCooldown -= deltaTime;
                }
            }
            
            shoot() {
                if (this.shootCooldown <= 0) {
