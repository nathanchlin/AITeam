    <div id="gameContainer">

        <div id="gameInfo">
            <div>分数: <span id="score">0</span></div>
            <div>生命: <span id="lives">3</span></div>
        </div>
        <div id="gameOver">
            <div>游戏结束</div>

        <div id="startScreen">
            <h1>1942 打飞机游戏</h1>
            <p>使用方向键移动飞机，空格键射击</p>
            <button onclick="game.start()">开始游戏</button>
        </div>

    <script>
        // 游戏状态枚举
        const GameState = {
            MENU: 'menu',
            PLAYING: 'playing',
            PAUSED: 'paused',
            GAME_OVER: 'game_over'
        };

        // 游戏主类
        class Game {
            constructor() {
                this.canvas = document.getElementById('gameCanvas');
                this.ctx = this.canvas.getContext('2d');
                this.width = this.canvas.width;
                this.height = this.canvas.height;
                this.state = GameState.MENU;
                this.score = 0;
                this.lives = 3;
                this.player = null;
                this.bullets = [];
                this.enemies = [];
                this.particles = [];
                this.keys = {};
                this.lastTime = 0;
                this.enemySpawnTimer = 0;
                this.enemySpawnInterval = 1000; // 1秒生成一个敌人
                
                // 初始化事件监听
                this.initEventListeners();
            }
            
            initEventListeners() {
                // 键盘事件
                window.addEventListener('keydown', (e) => {
                    this.keys[e.key] = true;
                    // 防止空格键滚动页面
                    if (e.key === ' ') {
                        e.preventDefault();
                    }
                });
                
                window.addEventListener('keyup', (e) => {
                    this.keys[e.key] = false;
                });
                
                // 鼠标事件
                this.canvas.addEventListener('mousemove', (e) => {
                    if (this.state === GameState.PLAYING && this.player) {
                        const rect = this.canvas.getBoundingClientRect();
                        const mouseX = e.clientX - rect.left;
                        const mouseY = e.clientY - rect.top;
                        this.player.targetX = mouseX;
                        this.player.targetY = mouseY;
                    }
                });
                
                this.canvas.addEventListener('click', (e) => {
                    if (this.state === GameState.PLAYING && this.player) {
                        this.player.shoot();
                    }
                });
            }
            
            start() {
                document.getElementById('startScreen').style.display = 'none';
                this.state = GameState.PLAYING;
                this.initPlayer();
                this.gameLoop();
            }
            
            initPlayer() {
                this.player = new Player(this.width / 2, this.height - 100);
            }
            
            gameLoop(currentTime = 0) {
                const deltaTime = currentTime - this.lastTime;
                this.lastTime = currentTime;
                
                // 清除画布
                this.ctx.clearRect(0, 0, this.width, this.height);
                
                if (this.state === GameState.PLAYING) {
                    // 更新游戏逻辑
                    this.update(deltaTime);
                    
                    // 渲染游戏
                    this.render();
                    
                    // 继续游戏循环
                    requestAnimationFrame((time) => this.gameLoop(time));
                }
            }
            
            update(deltaTime) {
                // 更新玩家
                if (this.player) {
                    this.player.update(deltaTime, this.keys);
                    
                    // 边界检测
                    this.player.x = Math.max(0, Math.min(this.width - this.player.width, this.player.x));
                    this.player.y = Math.max(0, Math.min(this.height - this.player.height, this.player.y));
                }
                
                // 更新子弹
                for (let i = this.bullets.length - 1; i >= 0; i--) {
                    const bullet = this.bullets[i];
                    bullet.update(deltaTime);
                    
                    // 移除超出屏幕的子弹
                    if (bullet.y < 0 || bullet.y > this.height) {
                        this.bullets.splice(i, 1);
                    }
                }
                
                // 生成敌人
                this.enemySpawnTimer += deltaTime;
                if (this.enemySpawnTimer >= this.enemySpawnInterval) {
                    this.spawnEnemy();
                    this.enemySpawnTimer = 0;
                }
                
                // 更新敌人
                for (let i = this.enemies.length - 1; i >= 0; i--) {
                    const enemy = this.enemies[i];
                    enemy.update(deltaTime);
                    
                    // 移除超出屏幕的敌人
                    if (enemy.y > this.height) {
                        this.enemies.splice(i, 1);
                    }
                }
                
                // 更新粒子效果
                for (let i = this.particles.length - 1; i >= 0; i--) {
                    const particle = this.particles[i];
                    particle.update(deltaTime);
                    
                    // 移除生命周期结束的粒子
                    if (particle.life <= 0) {
                        this.particles.splice(i, 1);
                    }
                }
                
                // 检测碰撞
                this.checkCollisions();
            }
            
            render() {
                // 绘制玩家
                if (this.player) {
                    this.player.render(this.ctx);
                }
                
                // 绘制子弹
                this.bullets.forEach(bullet => bullet.render(this.ctx));
                
                // 绘制敌人
                this.enemies.forEach(enemy => enemy.render(this.ctx));
                
                // 绘制粒子效果
                this.particles.forEach(particle => particle.render(this.ctx));
                
                // 更新UI
                document.getElementById('score').textContent = this.score;
                document.getElementById('lives').textContent = this.lives;
            }
            
            spawnEnemy() {
                const x = Math.random() * (this.width - 40);
                this.enemies.push(new Enemy(x, -40));
            }
            
            checkCollisions() {
                // 子弹与敌人的碰撞
                for (let i = this.bullets.length - 1; i >= 0; i--) {
                    const bullet = this.bullets[i];
                    
                    for (let j = this.enemies.length - 1; j >= 0; j--) {
                        const enemy = this.enemies[j];
                        
                        if (this.isColliding(bullet, enemy)) {
                            // 创建爆炸效果
                            this.createExplosion(enemy.x + enemy.width / 2, enemy.y + enemy.height / 2);
                            
                            // 移除子弹和敌人
                            this.bullets.splice(i, 1);
                            this.enemies.splice(j, 1);
                            
                            // 增加分数
                            this.score += 100;
                            break;
                        }
                    }
                }
                
                // 玩家与敌人的碰撞
                if (this.player) {
                    for (let i = this.enemies.length - 1; i >= 0; i--) {
                        const enemy = this.enemies[i];
                        
                        if (this.isColliding(this.player, enemy)) {
                            // 创建爆炸效果
                            this.createExplosion(enemy.x + enemy.width / 2, enemy.y + enemy.height / 2);
                            
                            // 移除敌人
                            this.enemies.splice(i, 1);
                            
                            // 减少生命值
                            this.lives--;
                            
                            // 检查游戏是否结束
                            if (this.lives <= 0) {
                                this.gameOver();
                            }
                            break;
                        }
                    }
                }
            }
            
            isColliding(obj1, obj2) {
                return obj1.x < obj2.x + obj2.width &&
                       obj1.x + obj1.width > obj2.x &&
                       obj1.y < obj2.y + obj2.height &&
                       obj1.y + obj1.height > obj2.y;
            }
            
            createExplosion(x, y) {
                for (let i = 0; i < 15; i++) {
                    this.particles.push(new Particle(x, y));
                }
            }
            
            gameOver() {
                this.state = GameState.GAME_OVER;
                document.getElementById('finalScore').textContent = this.score;
                document.getElementById('gameOver').style.display = 'block';
            }
            
            restart() {
                this.score = 0;
                this.lives = 3;
                this.bullets = [];
                this.enemies = [];
                this.particles = [];
                this.enemySpawnTimer = 0;
                
                document.getElementById('gameOver').style.display = 'none';
                
                if (this.player) {
                    this.player.x = this.width / 2;
                    this.player.y = this.height - 100;
                }
                
                this.state = GameState.PLAYING;
                this.gameLoop();
            }
        }

        // 玩家类
        class Player {
            constructor(x, y) {
                this.x = x;
                this.y = y;
                this.width = 40;
                this.height = 40;
                this.speed = 0.3;
                this.bulletSpeed = 0.5;
                this.shootCooldown = 0;
                this.shootInterval = 200; // 毫秒
                this.targetX = x;
                this.targetY = y;
                this.isMouseControl = false;
            }
            
            update(deltaTime, keys) {
                // 键盘控制
                if (keys['ArrowLeft'] || keys['a']) {
                    this.x -= this.speed * deltaTime;
                }
                if (keys['ArrowRight'] || keys['d']) {
                    this.x += this.speed * deltaTime;
                }
                if (keys['ArrowUp'] || keys['w']) {
                    this.y -= this.speed * deltaTime;
                }
                if (keys['ArrowDown'] || keys['s']) {
                    this.y += this.speed * deltaTime;
                }
                
                // 鼠标控制（平滑移动）
                if (this.isMouseControl) {
                    const dx = this.targetX - this.x;
                    const dy = this.targetY - this.y;
                    const distance = Math.sqrt(dx * dx + dy * dy);
                    
                    if (distance > 5) {
                        this.x += (dx / distance) * this.speed * deltaTime * 2;
                        this.y += (dy / distance) * this.speed * deltaTime * 2;
                    }
                }
                
                // 射击冷却
                if (this.shootCooldown > 0) {
                    this.shootCooldown -= deltaTime;
                }
                
                // 空格键射击
                if (keys[' '] && this.shootCooldown <= 0) {
                    this.shoot();
                }
            }
            
            shoot() {
                if (this.shootCooldown <= 0) {
                    game.bullets.push(new Bullet(this.x + this.width / 2 - 2, this.y, this.bulletSpeed));
                    this.shootCooldown = this.shootInterval;
                }
            }
            
            render(ctx) {
                // 绘制玩家飞机（简单的三角形）
                ctx.fillStyle = '#0ff';
                ctx.beginPath();
                ctx.moveTo(this.x + this.width / 2, this.y);
                ctx.lineTo(this.x, this.y + this.height);
                ctx.lineTo(this.x + this.width, this.y + this.height);
                ctx.closePath();
                ctx.fill();
                
                // 绘制飞机细节
                ctx.fillStyle = '#fff';
                ctx.fillRect(this.x + this.width / 2 - 2, this.y + 10, 4, 15);
            }
        }

        // 子弹类
        class Bullet {
            constructor(x, y, speed) {
                this.x = x;
                this.y = y;
                this.width = 4;
                this.height = 10;
                this.speed = speed;
            }
            
            update(deltaTime) {
                this.y -= this.speed * deltaTime;
            }
            
            render(ctx) {
                ctx.fillStyle = '#ff0';
                ctx.fillRect(this.x, this.y, this.width, this.height);
                
                // 添加发光效果
                ctx.shadowBlur = 10;
                ctx.shadowColor = '#ff0';
                ctx.fillRect(this.x, this.y, this.width, this.height);
                ctx.shadowBlur = 0;
            }
        }

        // 敌人类
        class Enemy {
            constructor(x, y) {
                this.x = x;
                this.y = y;
                this.width = 30;
                this.height = 30;
                this.speed = 0.15;
            }
            
            update(deltaTime) {
                this.y += this.speed * deltaTime;
            }
            
            render(ctx) {
                // 绘制敌人飞机（倒三角形）
                ctx.fillStyle = '#f00';
                ctx.beginPath();
                ctx.moveTo(this.x + this.width / 2, this.y + this.height);
                ctx.lineTo(this.x, this.y);
                ctx.lineTo(this.x + this.width, this.y);
                ctx.closePath();
                ctx.fill();
                
                // 绘制飞机细节
                ctx.fillStyle = '#800';
                ctx.fillRect(this.x + this.width / 2 - 1, this.y + 5, 2, 10);
            }
        }

        // 粒子效果类
        class Particle {
            constructor(x, y) {
                this.x = x;
                this.y = y;
                this.size = Math.random() * 5 + 2;
                this.speedX = (Math.random() - 0.5) * 0.3;
                this.speedY = (Math.random() - 0.5) * 0.3;
                this.life = 1.0;
                this.decay = Math.random() * 0.02 + 0.01;
                this.color = `hsl(${Math.random() * 60}, 100%, 50%)`;
            }
            
            update(deltaTime) {
                this.x += this.speedX * deltaTime;
                this.y += this.speedY * deltaTime;
                this.life -= this.decay;
                this.size *= 0.98;
            }
            
            render(ctx) {
                ctx.globalAlpha = this.life;
                ctx.fillStyle = this.color;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                ctx.fill();
                ctx.globalAlpha = 1;
            }
        }

        // 创建游戏实例
        const game = new Game();
    </script>

这个实现包含了以下功能：

1. **玩家飞机控制**：
   - 键盘控制：使用方向键或WASD键移动飞机
   - 鼠标控制：鼠标移动控制飞机位置，点击鼠标射击
   - 平滑移动和边界检测

2. **射击系统**：
   - 空格键或鼠标点击发射子弹
   - 射击冷却时间限制射击频率
   - 子弹自动向上移动

3. **边界检测**：
   - 玩家飞机不能移出游戏区域
   - 子弹和敌人超出屏幕边界时自动移除

4. **视觉效果**：
   - 玩家飞机和敌人的简单图形表示
   - 爆炸粒子效果
   - 发光的子弹效果

5. **游戏状态管理**：
   - 开始菜单
   - 游戏进行中
   - 游戏结束

这个实现使用了原生Canvas API，没有依赖任何外部游戏框架，可以直接在浏览器中运行。游戏逻辑完整，包含了玩家控制、射击、碰撞检测和边界检测等所有要求的功能。
