                            }

                            this.y = newY;

                                this.lastMoveTime = now;
                            }
                        }
                    }
                }

                            break;

            checkWallCollision(x, y, wall) {

            checkTankCollision(x, y, tank) {

                       y < tank.y + tank.height &&

                       this.y < wall.y + wall.height &&
                       this.y + this.height > wall.y;

                return this.x < tank.x + tank.width &&
