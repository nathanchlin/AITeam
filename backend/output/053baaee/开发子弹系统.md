## 子弹系统设计

### 子弹类设计

```javascript
class Bullet {
  constructor(x, y, direction, isPlayerBullet = false) {
    this.x = x;
    this.y = y;
    this.direction = direction; // 'up', 'down', 'left', 'right'
    this.speed = 5;
    this.isPlayerBullet = isPlayerBullet;
    this.active = true;
    this.size = 4; // 子弹大小
    this.damage = 1;
  }

  // 更新子弹位置
  update() {
    if (!this.active) return;

    switch (this.direction) {
      case 'up':
        this.y -= this.speed;
        break;
      case 'down':
        this.y += this.speed;
        break;
      case 'left':
        this.x -= this.speed;
        break;
      case 'right':
        this.x += this.speed;
        break;
    }

    // 检查子弹是否超出边界
    if (this.x < 0 || this.x > canvas.width || 
        this.y < 0 || this.y > canvas.height) {
      this.active = false;
    }
  }

  // 绘制子弹
  draw(ctx) {
    if (!this.active) return;

    ctx.fillStyle = this.isPlayerBullet ? '#00FF00' : '#FF0000';
    ctx.fillRect(this.x - this.size/2, this.y - this.size/2, this.size, this.size);
  }

  // 获取子弹的边界框
  getBounds() {
    return {
      left: this.x - this.size/2,
      right: this.x + this.size/2,
      top: this.y - this.size/2,
      bottom: this.y + this.size/2
    };
  }
}
```

### 子弹管理器

```javascript
class BulletManager {
  constructor() {
    this.bullets = [];
    this.maxBullets = 3; // 玩家同时最多拥有的子弹数
    this.shootCooldown = 300; // 射击冷却时间（毫秒）
    this.lastShotTime = 0;
  }

  // 发射子弹
  shoot(x, y, direction, isPlayerBullet = false) {
    const now = Date.now();
    
    // 如果是玩家子弹，检查射击冷却和最大子弹数
    if (isPlayerBullet) {
      if (this.bullets.filter(b => b.isPlayerBullet && b.active).length >= this.maxBullets) {
        return; // 已达到最大子弹数
      }
      
      if (now - this.lastShotTime < this.shootCooldown) {
        return; // 射击冷却中
      }
      
      this.lastShotTime = now;
    }

    // 创建新子弹
    const bullet = new Bullet(x, y, direction, isPlayerBullet);
    this.bullets.push(bullet);
  }

  // 更新所有子弹
  update() {
    this.bullets.forEach(bullet => bullet.update());
    
    // 移除非活跃的子弹
    this.bullets = this.bullets.filter(bullet => bullet.active);
  }

  // 绘制所有子弹
  draw(ctx) {
    this.bullets.forEach(bullet => bullet.draw(ctx));
  }

  // 检测子弹与目标的碰撞
  checkCollisions(targets) {
    this.bullets.forEach(bullet => {
      if (!bullet.active) return;

      const bulletBounds = bullet.getBounds();
      
      targets.forEach(target => {
        if (!target.active) return;
        
        const targetBounds = target.getBounds();
        
        // 检测碰撞
        if (bulletBounds.left < targetBounds.right &&
            bulletBounds.right > targetBounds.left &&
            bulletBounds.top < targetBounds.bottom &&
            bulletBounds.bottom > targetBounds.top) {
          
          // 处理碰撞
          bullet.active = false;
          target.hit(bullet.damage);
          
          // 如果是玩家子弹，增加分数
          if (bullet.isPlayerBullet && target.isEnemy) {
            game.score += 10;
          }
        }
      });
    });
  }

  // 检测子弹与墙壁的碰撞
  checkWallCollisions(walls) {
    this.bullets.forEach(bullet => {
      if (!bullet.active) return;

      const bulletBounds = bullet.getBounds();
      
      walls.forEach(wall => {
        if (!wall.active) return;
        
        const wallBounds = wall.getBounds();
        
        // 检测碰撞
        if (bulletBounds.left < wallBounds.right &&
            bulletBounds.right > wallBounds.left &&
            bulletBounds.top < wallBounds.bottom &&
            bulletBounds.bottom > wallBounds.top) {
          
          // 处理碰撞
          bullet.active = false;
          wall.hit(bullet.damage);
        }
      });
    });
  }
}
```

## 集成到游戏主循环

```javascript
// 游戏主循环
function gameLoop() {
  // 清空画布
  ctx.fillStyle = '#111';
  ctx.fillRect(0, 0, canvas.width, canvas.height);
  
  // 更新游戏状态
  player.update();
  enemies.forEach(enemy => enemy.update());
  bulletManager.update();
  
  // 碰撞检测
  bulletManager.checkCollisions([...enemies, player]);
  bulletManager.checkWallCollisions(walls);
  
  // 绘制游戏元素
  walls.forEach(wall => wall.draw(ctx));
  player.draw(ctx);
  enemies.forEach(enemy => enemy.draw(ctx));
  bulletManager.draw(ctx);
  
  // 更新UI
  updateUI();
  
  // 继续游戏循环
  requestAnimationFrame(gameLoop);
}
```

## 总结

这个子弹系统实现了以下功能：

1. **子弹发射**：玩家和敌人都可以发射子弹，玩家有射击冷却和最大子弹数限制
2. **子弹移动**：子弹根据发射方向持续移动，直到超出边界或击中目标
3. **碰撞检测**：
   - 子弹与坦克的碰撞
   - 子弹与墙壁的碰撞
4. **生命周期管理**：子弹在击中目标或超出边界后被标记为非活跃并从数组中移除
5. **视觉区分**：玩家子弹为绿色，敌人子弹为红色

这个系统设计简洁高效，易于扩展，可以很好地集成到坦克大战游戏中。通过调整子弹的速度、大小、伤害值等参数，可以轻松平衡游戏难度。
