    <script>
        // 游戏常量
        const TILE_SIZE = 40;
        const GAME_WIDTH = 800;
        const GAME_HEIGHT = 600;
        const TANK_SPEED = 2;
        const BULLET_SPEED = 5;
        const ENEMY_BULLET_SPEED = 3;
        const ENEMY_SHOOT_INTERVAL = 2000; // 毫秒
        const ENEMY_MOVE_INTERVAL = 1000; // 毫秒
        
        // 游戏状态
        let gameState = 'playing'; // 'playing', 'paused', 'gameOver'
        let score = 0;
        let lastTime = 0;
        
        // 获取画布和上下文
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        
        // 游戏对象
        let player = null;
        let enemies = [];
        let bullets = [];
        let enemyBullets = [];
        let walls = [];
        
        // 输入状态
        const keys = {};
        
        // 坦克类
        class Tank {
            constructor(x, y, color, isPlayer = false) {
                this.x = x;
                this.y = y;
                this.width = TILE_SIZE;
                this.height = TILE_SIZE;
                this.color = color;
                this.direction = 'up'; // 'up', 'down', 'left', 'right'
                this.isPlayer = isPlayer;
                this.speed = TANK_SPEED;
                this.lastShotTime = 0;
                this.shootCooldown = isPlayer ? 500 : 1000; // 毫秒
            }
            
            draw() {
                ctx.fillStyle = this.color;
                ctx.fillRect(this.x, this.y, this.width, this.height);
                
                // 绘制坦克炮管
                ctx.fillStyle = '#000';
                let barrelX = this.x + this.width / 2;
                let barrelY = this.y + this.height / 2;
                
                ctx.beginPath();
                switch(this.direction) {
                    case 'up':
                        ctx.moveTo(barrelX, barrelY - this.height / 2);
                        ctx.lineTo(barrelX - 5, barrelY);
                        ctx.lineTo(barrelX + 5, barrelY);
                        break;
                    case 'down':
                        ctx.moveTo(barrelX, barrelY + this.height / 2);
                        ctx.lineTo(barrelX - 5, barrelY);
                        ctx.lineTo(barrelX + 5, barrelY);
                        break;
                    case 'left':
                        ctx.moveTo(barrelX - this.width / 2, barrelY);
                        ctx.lineTo(barrelX, barrelY - 5);
                        ctx.lineTo(barrelX, barrelY + 5);
                        break;
                    case 'right':
                        ctx.moveTo(barrelX + this.width / 2, barrelY);
                        ctx.lineTo(barrelX, barrelY - 5);
                        ctx.lineTo(barrelX, barrelY + 5);
                        break;
                }
                ctx.fill();
                
                // 绘制坦克细节
                ctx.fillStyle = this.isPlayer ? '#0ff' : '#f0f';
                ctx.fillRect(this.x + 5, this.y + 5, this.width - 10, this.height - 10);
            }
            
            move(direction) {
                this.direction = direction;
                let newX = this.x;
                let newY = this.y;
                
                switch(direction) {
                    case 'up':
                        newY -= this.speed;
                        break;
                    case 'down':
                        newY += this.speed;
                        break;
                    case 'left':
                        newX -= this.speed;
                        break;
                    case 'right':
                        newX += this.speed;
                        break;
                }
                
                // 边界检查
                if (newX >= 0 && newX + this.width <= GAME_WIDTH &&
                    newY >= 0 && newY + this.height <= GAME_HEIGHT) {
                    
                    // 墙壁碰撞检查
                    let canMove = true;
                    for (let wall of walls) {
                        if (this.checkCollision(newX, newY, this.width, this.height, wall)) {
                            canMove = false;
                            break;
                        }
                    }
                    
                    // 坦克碰撞检查
                    for (let enemy of enemies) {
                        if (enemy !== this && this.checkCollision(newX, newY, this.width, this.height, enemy)) {
                            canMove = false;
                            break;
                        }
                    }
                    
                    if (canMove) {
                        this.x = newX;
                        this.y = newY;
                    }
                }
            }
            
            checkCollision(x, y, width, height, other) {
                return x < other.x + other.width &&
                       x + width > other.x &&
                       y < other.y + other.height &&
                       y + height > other.y;
            }
            
            shoot() {
                const currentTime = Date.now();
                if (currentTime - this.lastShotTime < this.shootCooldown) {
                    return;
                }
                
                this.lastShotTime = currentTime;
                let bulletX = this.x + this.width / 2;
                let bulletY = this.y + this.height / 2;
                
                let bullet = new Bullet(bulletX, bulletY, this.direction, this.isPlayer);
                bullets.push(bullet);
            }
        }
        
        // 子弹类
        class Bullet {
            constructor(x, y, direction, isPlayerBullet = true) {
                this.x = x;
                this.y = y;
                this.direction = direction;
                this.speed = isPlayerBullet ? BULLET_SPEED : ENEMY_BULLET_SPEED;
                this.isPlayerBullet = isPlayerBullet;
                this.width = 8;
                this.height = 8;
                this.active = true;
            }
            
            draw() {
                ctx.fillStyle = this.isPlayerBullet ? '#0ff' : '#f0f';
                ctx.fillRect(this.x - this.width / 2, this.y - this.height / 2, this.width, this.height);
            }
            
            update() {
                switch(this.direction) {
                    case 'up':
                        this.y -= this.speed;
                        break;
                    case 'down':
                        this.y += this.speed;
                        break;
                    case 'left':
                        this.x -= this.speed;
                        break;
                    case 'right':
                        this.x += this.speed;
                        break;
                }
                
                // 检查是否超出边界
                if (this.x < 0 || this.x > GAME_WIDTH || this.y < 0 || this.y > GAME_HEIGHT) {
                    this.active = false;
                }
            }
        }
        
        // 墙壁类
        class Wall {
            constructor(x, y, destructible = true) {
                this.x = x;
                this.y = y;
                this.width = TILE_SIZE;
                this.height = TILE_SIZE;
                this.destructible = destructible;
                this.active = true;
            }
            
            draw() {
                if (this.destructible) {
                    ctx.fillStyle = '#8B4513';
                    ctx.fillRect(this.x, this.y, this.width, this.height);
                    
                    // 绘制砖块纹理
                    ctx.fillStyle = '#654321';
                    for (let i = 0; i < 4; i++) {
                        for (let j = 0; j < 4; j++) {
                            ctx.fillRect(this.x + i * 10, this.y + j * 10, 8, 8);
                        }
                    }
                } else {
                    ctx.fillStyle = '#333';
                    ctx.fillRect(this.x, this.y, this.width, this.height);
                }
            }
        }
        
        // 初始化游戏
        function initGame() {
            gameState = 'playing';
            score = 0;
            
            // 创建玩家坦克
            player = new Tank(GAME_WIDTH / 2 - TILE_SIZE / 2, GAME_HEIGHT - TILE_SIZE * 2, '#0f0', true);
            
            // 创建敌方坦克
            enemies = [
                new Tank(TILE_SIZE, TILE_SIZE, '#f00', false),
                new Tank(GAME_WIDTH - TILE_SIZE * 2, TILE_SIZE, '#f00', false),
                new Tank(GAME_WIDTH / 2 - TILE_SIZE / 2, TILE_SIZE, '#f00', false)
            ];
            
            // 创建子弹数组
            bullets = [];
            enemyBullets = [];
            
            // 创建墙壁
            walls = [];
            
            // 创建边界墙
            for (let i = 0; i < GAME_WIDTH / TILE_SIZE; i++) {
                walls.push(new Wall(i * TILE_SIZE, 0, false)); // 上边界
                walls.push(new Wall(i * TILE_SIZE, GAME_HEIGHT - TILE_SIZE, false)); // 下边界
            }
            
            for (let i = 0; i < GAME_HEIGHT / TILE_SIZE; i++) {
                walls.push(new Wall(0, i * TILE_SIZE, false)); // 左边界
                walls.push(new Wall(GAME_WIDTH - TILE_SIZE, i * TILE_SIZE, false)); // 右边界
            }
            
            // 创建内部砖墙
            for (let i = 2; i < GAME_WIDTH / TILE_SIZE - 2; i += 2) {
                for (let j = 2; j < GAME_HEIGHT / TILE_SIZE - 2; j += 2) {
                    if (Math.random() > 0.3) {
                        walls.push(new Wall(i * TILE_SIZE, j * TILE_SIZE, true));
                    }
                }
            }
            
            // 设置事件监听
            document.addEventListener('keydown', handleKeyDown);
            document.addEventListener('keyup', handleKeyUp);
            document.getElementById('restartBtn').addEventListener('click', restartGame);
            
            // 启动游戏循环
            requestAnimationFrame(gameLoop);
        }
        
        // 游戏主循环
        function gameLoop(timestamp) {
            if (gameState !== 'playing') {
                return;
            }
            
            const deltaTime = timestamp - lastTime;
            lastTime = timestamp;
            
            // 清空画布
            ctx.clearRect(0, 0, GAME_WIDTH, GAME_HEIGHT);
            
            // 绘制背景
            ctx.fillStyle = '#111';
            ctx.fillRect(0, 0, GAME_WIDTH, GAME_HEIGHT);
            
            // 处理输入
            handlePlayerInput();
            
            // 更新和绘制墙壁
            walls.forEach(wall => {
                if (wall.active) {
                    wall.draw();
                }
            });
            
            // 更新和绘制玩家
            if (player) {
                player.draw();
            }
            
            // 更新和绘制敌人
            updateEnemyTanks(deltaTime);
            enemies.forEach(enemy => {
                if (enemy.active) {
                    enemy.draw();
                }
            });
            
            // 更新和绘制子弹
            updateBullets();
            bullets.forEach(bullet => {
                if (bullet.active) {
                    bullet.draw();
                }
            });
            
            // 更新和绘制敌方子弹
            updateEnemyBullets(deltaTime);
            enemyBullets.forEach(bullet => {
                if (bullet.active) {
                    bullet.draw();
                }
            });
            
            // 碰撞检测
            checkCollisions();
            
            // 更新得分显示
            document.getElementById('score').textContent = `得分: ${score}`;
            
            // 检查游戏结束条件
            checkGameOver();
            
            requestAnimationFrame(gameLoop);
        }
        
        // 处理玩家输入
        function handlePlayerInput() {
            if (!player) return;
            
            if (keys['ArrowUp'] || keys['w'] || keys['W']) {
                player.move('up');
            }
            if (keys['ArrowDown'] || keys['s'] || keys['S']) {
                player.move('down');
            }
            if (keys['ArrowLeft'] || keys['a'] || keys['A']) {
                player.move('left');
            }
            if (keys['ArrowRight'] || keys['d'] || keys['D']) {
                player.move('right');
            }
            if (keys[' ']) {
                player.shoot();
            }
        }
        
        // 键盘按下事件
        function handleKeyDown(e) {
            keys[e.key] = true;
        }
        
        // 键盘释放事件
        function handleKeyUp(e) {
            keys[e.key] = false;
        }
        
        // 更新敌方坦克
        function updateEnemyTanks(deltaTime) {
            enemies.forEach(enemy => {
                if (!enemy.active) return;
                
                // 随机移动
                if (Math.random() < 0.01) {
                    const directions = ['up', 'down', 'left', 'right'];
                    const randomDirection = directions[Math.floor(Math.random() * directions.length)];
                    enemy.move(randomDirection);
                }
                
                // 随机射击
                if (Math.random() < 0.005) {
                    enemy.shoot();
                }
            });
        }
        
        // 更新子弹
        function updateBullets() {
            bullets = bullets.filter(bullet => {
                if (!bullet.active) return false;
                
                bullet.update();
                return bullet.active;
            });
        }
        
        // 更新敌方子弹
        function updateEnemyBullets(deltaTime) {
            enemyBullets = enemyBullets.filter(bullet => {
                if (!bullet.active) return false;
                
                bullet.update();
                return bullet.active;
            });
        }
        
        // 碰撞检测
        function checkCollisions() {
            // 玩家子弹与墙壁碰撞
            bullets.forEach(bullet => {
                if (!bullet.active) return;
                
                walls.forEach(wall => {
                    if (!wall.active) return;
                    
                    if (bullet.checkCollision(bullet.x, bullet.y, bullet.width, bullet.height, wall)) {
                        bullet.active = false;
                        if (wall.destructible) {
                            wall.active = false;
                            score += 10;
                        }
                    }
                });
            });
            
            // 敌方子弹与墙壁碰撞
            enemyBullets.forEach(bullet => {
                if (!bullet.active) return;
                
                walls.forEach(wall => {
                    if (!wall.active) return;
                    
                    if (bullet.checkCollision(bullet.x, bullet.y, bullet.width, bullet.height, wall)) {
                        bullet.active = false;
                        if (wall.destructible) {
                            wall.active = false;
                        }
                    }
                });
            });
            
            // 玩家子弹与敌人碰撞
            bullets.forEach(bullet => {
                if (!bullet.active || !player) return;
                
                enemies.forEach(enemy => {
                    if (!enemy.active) return;
                    
                    if (bullet.checkCollision(bullet.x, bullet.y, bullet.width, bullet.height, enemy)) {
                        bullet.active = false;
                        enemy.active = false;
                        score += 100;
                    }
                });
            });
            
            // 敌方子弹与玩家碰撞
            enemyBullets.forEach(bullet => {
                if (!bullet.active || !player) return;
                
                if (bullet.checkCollision(bullet.x, bullet.y, bullet.width, bullet.height, player)) {
                    bullet.active = false;
                    player = null;
                    gameState = 'gameOver';
                    document.getElementById('gameOver').style.display = 'block';
                    document.getElementById('finalScore').textContent = `最终得分: ${score}`;
                }
            });
            
            // 坦克与墙壁碰撞（已在移动函数中处理）
        }
        
        // 检查游戏结束条件
        function checkGameOver() {
            // 检查玩家是否被消灭
            if (!player) {
                gameState = 'gameOver';
                document.getElementById('gameOver').style.display = 'block';
                document.getElementById('finalScore').textContent = `最终得分: ${score}`;
                return;
            }
            
            // 检查所有敌人是否被消灭
            const allEnemiesDead = enemies.every(enemy => !enemy.active);
            if (allEnemiesDead) {
                // 生成新的敌人
                enemies
