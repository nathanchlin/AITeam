```javascript
function checkBulletCollisions() {
    // 检查玩家子弹与敌方坦克的碰撞
    playerBullets.forEach((bullet, bulletIndex) => {
        enemyTanks.forEach((tank, tankIndex) => {
            if (isColliding(bullet, tank)) {
                // 移除子弹和敌方坦克
                playerBullets.splice(bulletIndex, 1);
                enemyTanks.splice(tankIndex, 1);
                
                // 增加得分
                score += 100;
                updateScoreDisplay();
                
                // 添加爆炸效果
                createExplosion(tank.x, tank.y);
                
                // 生成新的敌方坦克
                setTimeout(() => {
                    if (gameRunning) {
                        spawnEnemyTank();
                    }
                }, 2000);
            }
        });
        
        // 检查玩家子弹与砖墙的碰撞
        walls.forEach((wall, wallIndex) => {
            if (wall.type === 'brick' && isColliding(bullet, wall)) {
                playerBullets.splice(bulletIndex, 1);
                walls.splice(wallIndex, 1);
                
                // 增加得分
                score += 10;
                updateScoreDisplay();
            }
        });
    });
    
    // 检查敌方子弹与玩家坦克的碰撞
    enemyBullets.forEach((bullet, bulletIndex) => {
        if (isColliding(bullet, playerTank)) {
            enemyBullets.splice(bulletIndex, 1);
            
            // 减少生命值
            playerHealth--;
            document.getElementById('health-display').textContent = playerHealth;
            
            // 添加爆炸效果
            createExplosion(playerTank.x, playerTank.y);
            
            // 检查游戏是否结束
            if (playerHealth <= 0) {
                gameOver();
            }
        }
        
        // 检查敌方子弹与砖墙的碰撞
        walls.forEach((wall, wallIndex) => {
            if (wall.type === 'brick' && isColliding(bullet, wall)) {
                enemyBullets.splice(bulletIndex, 1);
                walls.splice(wallIndex, 1);
            }
        });
    });
}
```

    <script>
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        ctx.imageSmoothingEnabled = false; // 保持像素风格
        
        // 游戏状态
        let score = 0;
        let playerHealth = 3;
        let gameRunning = true;
        
        // 更新得分显示
        function updateScoreDisplay() {
            document.getElementById('score-display').textContent = score;
        }
        
        // 更新生命值显示
        function updateHealthDisplay() {
            document.getElementById('health-display').textContent = playerHealth;
        }
        
        // 游戏对象
        const TILE_SIZE = 20;
        const GRID_WIDTH = canvas.width / TILE_SIZE;
        const GRID_HEIGHT = canvas.height / TILE_SIZE;
        
        // 玩家坦克
        const playerTank = {
            x: GRID_WIDTH / 2 - 1,
            y: GRID_HEIGHT - 3,
            width: 2,
            height: 2,
            speed: 0.1,
            direction: 'up',
            color: '#0f0'
        };
        
        // 敌方坦克
        const enemyTanks = [];
        const MAX_ENEMY_TANKS = 3;
        
        // 子弹
        const playerBullets = [];
        const enemyBullets = [];
        const BULLET_SPEED = 0.3;
        
        // 墙壁
        const walls = [];
        
        // 初始化墙壁
        function initWalls() {
            // 边界墙
            for (let x = 0; x < GRID_WIDTH; x++) {
                walls.push({ x, y: 0, width: 1, height: 1, type: 'steel' });
                walls.push({ x, y: GRID_HEIGHT - 1, width: 1, height: 1, type: 'steel' });
            }
            for (let y = 1; y < GRID_HEIGHT - 1; y++) {
                walls.push({ x: 0, y, width: 1, height: 1, type: 'steel' });
                walls.push({ x: GRID_WIDTH - 1, y, width: 1, height: 1, type: 'steel' });
            }
            
            // 随机砖墙
            for (let i = 0; i < 30; i++) {
                let x, y;
                do {
                    x = Math.floor(Math.random() * (GRID_WIDTH - 2)) + 1;
                    y = Math.floor(Math.random() * (GRID_HEIGHT - 2)) + 1;
                } while (
                    (x > playerTank.x - 2 && x < playerTank.x + 3 && y > playerTank.y - 2 && y < playerTank.y + 3) ||
                    walls.some(wall => wall.x === x && wall.y === y)
                );
                
                walls.push({ x, y, width: 1, height: 1, type: 'brick' });
            }
        }
        
        // 生成敌方坦克
        function spawnEnemyTank() {
            if (enemyTanks.length < MAX_ENEMY_TANKS) {
                let x, y;
                do {
                    x = Math.floor(Math.random() * (GRID_WIDTH - 2)) + 1;
                    y = Math.floor(Math.random() * 3) + 1;
                } while (
                    walls.some(wall => wall.x === x && wall.y === y) ||
                    enemyTanks.some(tank => tank.x === x && tank.y === y)
                );
                
                const directions = ['up', 'down', 'left', 'right'];
                const colors = ['#f00', '#00f', '#ff0', '#f0f'];
                
                enemyTanks.push({
                    x,
                    y,
                    width: 2,
                    height: 2,
                    speed: 0.05,
                    direction: directions[Math.floor(Math.random() * directions.length)],
                    color: colors[Math.floor(Math.random() * colors.length)],
                    lastShot: Date.now(),
                    shotCooldown: 2000 + Math.random() * 2000
                });
            }
        }
        
        // 碰撞检测
        function isColliding(obj1, obj2) {
            return obj1.x < obj2.x + obj2.width &&
                   obj1.x + obj1.width > obj2.x &&
                   obj1.y < obj2.y + obj2.height &&
                   obj1.y + obj1.height > obj2.y;
        }
        
        // 创建爆炸效果
        function createExplosion(x, y) {
            const particles = [];
            for (let i = 0; i < 10; i++) {
                particles.push({
                    x: x * TILE_SIZE + TILE_SIZE / 2,
                    y: y * TILE_SIZE + TILE_SIZE / 2,
                    vx: (Math.random() - 0.5) * 4,
                    vy: (Math.random() - 0.5) * 4,
                    life: 20
                });
            }
            
            function drawExplosion() {
                particles.forEach((particle, index) => {
                    if (particle.life <= 0) {
                        particles.splice(index, 1);
                        return;
                    }
                    
                    ctx.fillStyle = `rgba(255, ${Math.floor(255 * particle.life / 20)}, 0, ${particle.life / 20})`;
                    ctx.fillRect(
                        particle.x - particle.life / 4,
                        particle.y - particle.life / 4,
                        particle.life / 2,
                        particle.life / 2
                    );
                    
                    particle.x += particle.vx;
                    particle.y += particle.vy;
                    particle.life--;
                });
                
                if (particles.length > 0) {
                    requestAnimationFrame(drawExplosion);
                }
            }
            
            drawExplosion();
        }
        
        // 发射子弹
        function shootBullet(x, y, direction, isPlayerBullet = true) {
            const bullet = {
                x,
                y,
                width: 0.5,
                height: 0.5,
                speed: BULLET_SPEED,
                direction,
                isPlayerBullet
            };
            
            if (isPlayerBullet) {
                playerBullets.push(bullet);
            } else {
                enemyBullets.push(bullet);
            }
        }
        
        // 检查子弹碰撞
        function checkBulletCollisions() {
            // 检查玩家子弹与敌方坦克的碰撞
            playerBullets.forEach((bullet, bulletIndex) => {
                enemyTanks.forEach((tank, tankIndex) => {
                    if (isColliding(bullet, tank)) {
                        // 移除子弹和敌方坦克
                        playerBullets.splice(bulletIndex, 1);
                        enemyTanks.splice(tankIndex, 1);
                        
                        // 增加得分
                        score += 100;
                        updateScoreDisplay();
                        
                        // 添加爆炸效果
                        createExplosion(tank.x, tank.y);
                        
                        // 生成新的敌方坦克
                        setTimeout(() => {
                            if (gameRunning) {
                                spawnEnemyTank();
                            }
                        }, 2000);
                    }
                });
                
                // 检查玩家子弹与砖墙的碰撞
                walls.forEach((wall, wallIndex) => {
                    if (wall.type === 'brick' && isColliding(bullet, wall)) {
                        playerBullets.splice(bulletIndex, 1);
                        walls.splice(wallIndex, 1);
                        
                        // 增加得分
                        score += 10;
                        updateScoreDisplay();
                    }
                });
            });
            
            // 检查敌方子弹与玩家坦克的碰撞
            enemyBullets.forEach((bullet, bulletIndex) => {
                if (isColliding(bullet, playerTank)) {
                    enemyBullets.splice(bulletIndex, 1);
                    
                    // 减少生命值
                    playerHealth--;
                    updateHealthDisplay();
                    
                    // 添加爆炸效果
                    createExplosion(playerTank.x, playerTank.y);
                    
                    // 检查游戏是否结束
                    if (playerHealth <= 0) {
                        gameOver();
                    }
                }
                
                // 检查敌方子弹与砖墙的碰撞
                walls.forEach((wall, wallIndex) => {
                    if (wall.type === 'brick' && isColliding(bullet, wall)) {
                        enemyBullets.splice(bulletIndex, 1);
                        walls.splice(wallIndex, 1);
                    }
                });
            });
        }
        
        // 移动子弹
        function moveBullets() {
            // 移动玩家子弹
            playerBullets.forEach((bullet, index) => {
                switch (bullet.direction) {
                    case 'up':
                        bullet.y -= bullet.speed;
                        break;
                    case 'down':
                        bullet.y += bullet.speed;
                        break;
                    case 'left':
                        bullet.x -= bullet.speed;
                        break;
                    case 'right':
                        bullet.x += bullet.speed;
                        break;
                }
                
                // 检查是否超出边界
                if (bullet.x < 0 || bullet.x >= GRID_WIDTH || bullet.y < 0 || bullet.y >= GRID_HEIGHT) {
                    playerBullets.splice(index, 1);
                }
            });
            
            // 移动敌方子弹
            enemyBullets.forEach((bullet, index) => {
                switch (bullet.direction) {
                    case 'up':
                        bullet.y -= bullet.speed;
                        break;
                    case 'down':
                        bullet.y += bullet.speed;
                        break;
                    case 'left':
                        bullet.x -= bullet.speed;
                        break;
                    case 'right':
                        bullet.x += bullet.speed;
                        break;
                }
                
                // 检查是否超出边界
                if (bullet.x < 0 || bullet.x >= GRID_WIDTH || bullet.y < 0 || bullet.y >= GRID_HEIGHT) {
                    enemyBullets.splice(index, 1);
                }
            });
        }
        
        // 移动坦克
        function moveTank(tank) {
            let newX = tank.x;
            let newY = tank.y;
            
            switch (tank.direction) {
                case 'up':
                    newY -= tank.speed;
                    break;
                case 'down':
                    newY += tank.speed;
                    break;
                case 'left':
                    newX -= tank.speed;
                    break;
                case 'right':
                    newX += tank.speed;
                    break;
            }
            
            // 检查边界
            if (newX < 0 || newX + tank.width > GRID_WIDTH || newY < 0 || newY + tank.height > GRID_HEIGHT) {
                // 改变方向
                const directions = ['up', 'down', 'left', 'right'];
                tank.direction = directions[Math.floor(Math.random() * directions.length)];
                return;
            }
            
            // 检查墙壁碰撞
            const tempTank = { ...tank, x: newX, y: newY };
            let collision = false;
            
            for (const wall of walls) {
                if (isColliding(tempTank, wall)) {
                    collision = true;
                    break;
                }
            }
            
            // 检查与其他坦克的碰撞
            for (const otherTank of [...enemyTanks, playerTank]) {
                if (otherTank !== tank && isColliding(tempTank, otherTank)) {
                    collision = true;
                    break;
                }
            }
            
            if (!collision) {
                tank.x = newX;
                tank.y = newY;
            } else {
                // 改变方向
                const directions = ['up', 'down', 'left', 'right'];
                tank.direction = directions[Math.floor(Math.random() * directions.length)];
            }
            
            // 随机射击
            const now = Date.now();
            if (now - tank.lastShot > tank.shotCooldown) {
                shootBullet(
                    tank.x + tank.width / 2,
                    tank.y + tank.height / 2,
                    tank.direction,
                    false
                );
                tank.lastShot = now;
                tank.shotCooldown = 2000 + Math.random() * 2000;
            }
        }
        
        // 绘制坦克
        function drawTank(tank) {
            ctx.fillStyle = tank.color;
            ctx.fillRect(
                tank.x * TILE_SIZE,
                tank.y * TILE_SIZE,
                tank.width * TILE_SIZE,
                tank.height * TILE_SIZE
            );
            
            // 绘制炮管
            ctx.fillStyle =
