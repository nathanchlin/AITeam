        // 音效管理
        class SoundManager {
            constructor() {
                this.sounds = {};
                this.createSounds();
            }
            
            createSounds() {
                // 使用Web Audio API创建简单的音效
                this.audioContext = new (window.AudioContext || window.webkitAudioContext)();
                
                // 射击音效
                this.sounds.shoot = () => this.playTone(200, 0.1);
                
                // 爆炸音效
                this.sounds.explosion = () => {
                    const osc = this.audioContext.createOscillator();
                    const gain = this.audioContext.createGain();
                    
                    osc.connect(gain);
                    gain.connect(this.audioContext.destination);
                    
                    osc.type = 'sawtooth';
                    osc.frequency.value = 150;
                    gain.gain.setValueAtTime(0.3, this.audioContext.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.01, this.audioContext.currentTime + 0.3);
                    
                    osc.start(this.audioContext.currentTime);
                    osc.stop(this.audioContext.currentTime + 0.3);
                };
                
                // 击中音效
                this.sounds.hit = () => this.playTone(300, 0.05);
                
                // 移动音效
                this.sounds.move = () => this.playTone(100, 0.05);
            }
            
            playTone(frequency, duration) {
                if (!soundEnabled) return;
                
                const oscillator = this.audioContext.createOscillator();
                const gainNode = this.audioContext.createGain();
                
                oscillator.connect(gainNode);
                gainNode.connect(this.audioContext.destination);
                
                oscillator.type = 'square';
                oscillator.frequency.value = frequency;
                
                gainNode.gain.setValueAtTime(0.1, this.audioContext.currentTime);
                gainNode.gain.exponentialRampToValueAtTime(0.01, this.audioContext.currentTime + duration);
                
                oscillator.start(this.audioContext.currentTime);
                oscillator.stop(this.audioContext.currentTime + duration);
            }
            
            play(soundName) {
                if (this.sounds[soundName]) {
                    this.sounds[soundName]();
                }
            }
        }

        // 游戏对象
        class GameObject {
            constructor(x, y, width, height, color) {
                this.x = x;
                this.y = y;
                this.width = width;
                this.height = height;
                this.color = color;
                this.destroyed = false;
            }
            
            draw() {
                if (this.destroyed) return;
                
                ctx.fillStyle = this.color;
                ctx.fillRect(this.x, this.y, this.width, this.height);
                
                // 添加像素风格边框
                ctx.strokeStyle = 'rgba(0, 0, 0, 0.2)';
                ctx.lineWidth = 1;
                ctx.strokeRect(this.x, this.y, this.width, this.height);
            }
            
            collidesWith(other) {
                return !(
                    this.x + this.width < other.x ||
                    other.x + other.width < this.x ||
                    this.y + this.height < other.y ||
                    other.y + other.height < this.y
                );
            }
        }
        
        class Tank extends GameObject {
            constructor(x, y, color, isPlayer = false) {
                super(x, y, PIXEL_SIZE * 2, PIXEL_SIZE * 2, color);
                this.isPlayer = isPlayer;
                this.direction = 'up';
                this.speed = isPlayer ? 2 : 1;
                this.lastShot = 0;
                this.shootCooldown = isPlayer ? 500 : 1000;
                this.animationFrame = 0;
                this.moving = false;
            }
            
            draw() {
                if (this.destroyed) return;
                
                ctx.save();
                ctx.translate(this.x + this.width / 2, this.y + this.height / 2);
                
                // 根据方向旋转坦克
                switch (this.direction) {
                    case 'up':
                        break;
                    case 'down':
                        ctx.rotate(Math.PI);
                        break;
                    case 'left':
                        ctx.rotate(-Math.PI / 2);
                        break;
                    case 'right':
                        ctx.rotate(Math.PI / 2);
                        break;
                }
                
                // 绘制坦克主体
                ctx.fillStyle = this.color;
                ctx.fillRect(-this.width / 2, -this.height / 2, this.width, this.height);
                
                // 绘制坦克炮管
                ctx.fillStyle = this.isPlayer ? '#0a0' : '#a00';
                ctx.fillRect(-this.width / 6, -this.height / 2 - this.width / 3, this.width / 3, this.width / 3);
                
                // 绘制坦克细节（像素风格）
                ctx.fillStyle = 'rgba(0, 0, 0, 0.3)';
                ctx.fillRect(-this.width / 2 + 2, -this.height / 2 + 2, this.width - 4, 2);
                ctx.fillRect(-this.width / 2 + 2, -this.height / 2 + 2, 2, this.height - 4);
                
                // 添加履带效果
                if (this.moving) {
                    this.animationFrame = (this.animationFrame + 1) % 4;
                    ctx.fillStyle = 'rgba(0, 0, 0, 0.2)';
                    for (let i = 0; i < 3; i++) {
                        const offset = (i * this.height / 3 + this.animationFrame * 2) % this.height;
                        ctx.fillRect(-this.width / 2, -this.height / 2 + offset, this.width, 2);
                    }
                }
                
                ctx.restore();
            }
            
            move(direction, walls) {
                this.direction = direction;
                this.moving = true;
                
                let newX = this.x;
                let newY = this.y;
                
                switch (direction) {
                    case 'up':
                        newY -= this.speed;
                        break;
                    case 'down':
                        newY += this.speed;
                        break;
                    case 'left':
                        newX -= this.speed;
                        break;
                    case 'right':
                        newX += this.speed;
                        break;
                }
                
                // 边界检查
                if (newX < 0 || newX + this.width > canvas.width || 
                    newY < 0 || newY + this.height > canvas.height) {
                    return;
                }
                
                // 碰撞检测
                const tempTank = new Tank(newX, newY, this.color, this.isPlayer);
                tempTank.width = this.width;
                tempTank.height = this.height;
                
                let canMove = true;
                for (const wall of walls) {
                    if (!wall.destroyed && tempTank.collidesWith(wall)) {
                        canMove = false;
                        break;
                    }
                }
                
                // 与其他坦克碰撞检测
                for (const tank of [...tanks]) {
                    if (tank !== this && !tank.destroyed && tempTank.collidesWith(tank)) {
                        canMove = false;
                        break;
                    }
                }
                
                if (canMove) {
                    this.x = newX;
                    this.y = newY;
                    
                    if (this.isPlayer) {
                        soundManager.play('move');
                    }
                } else {
                    // 碰撞时添加震动效果
                    canvas.classList.add('shake');
                    setTimeout(() => canvas.classList.remove('shake'), 300);
                }
            }
            
            shoot() {
                const now = Date.now();
                if (now - this.lastShot < this.shootCooldown) return;
                
                this.lastShot = now;
                soundManager.play('shoot');
                
                let bulletX = this.x + this.width / 2 - PIXEL_SIZE / 2;
                let bulletY = this.y + this.height / 2 - PIXEL_SIZE / 2;
                
                switch (this.direction) {
                    case 'up':
                        bulletY = this.y - PIXEL_SIZE;
                        break;
                    case 'down':
                        bulletY = this.y + this.height;
                        break;
                    case 'left':
                        bulletX = this.x - PIXEL_SIZE;
                        break;
                    case 'right':
                        bulletX = this.x + this.width;
                        break;
                }
                
                bullets.push(new Bullet(bulletX, bulletY, this.direction, this.isPlayer));
            }
        }
