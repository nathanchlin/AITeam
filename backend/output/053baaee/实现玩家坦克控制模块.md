```html
<body>
    <div id="game-container">
        <canvas id="game-canvas" width="800" height="600"></canvas>
        <div id="score">得分: 0</div>
        <div id="game-over">
            游戏结束<br>
            <span style="font-size: 20px;">按 R 重新开始</span>
        </div>
        <div id="instructions">
            WASD/方向键: 移动 | 空格: 射击 | R: 重新开始
        </div>
    </div>

    <script>
        // 获取画布和上下文
        const canvas = document.getElementById('game-canvas');
        const ctx = canvas.getContext('2d');
        const scoreElement = document.getElementById('score');
        const gameOverElement = document.getElementById('game-over');
        
        // 游戏常量
        const TILE_SIZE = 20;
        const GAME_WIDTH = canvas.width;
        const GAME_HEIGHT = canvas.height;
        const PLAYER_SPEED = 3;
        const ENEMY_SPEED = 1.5;
        const BULLET_SPEED = 5;
        const ENEMY_BULLET_SPEED = 4;
        
        // 游戏状态
        let gameState = 'playing'; // 'playing', 'paused', 'gameOver'
        let score = 0;
        let keys = {};
        
        // 游戏对象
        let player = null;
        let enemies = [];
        let bullets = [];
        let enemyBullets = [];
        let walls = [];
        
        // 坦克类
        class Tank {
            constructor(x, y, color, isPlayer = false) {
                this.x = x;
                this.y = y;
                this.width = TILE_SIZE * 2;
                this.height = TILE_SIZE * 2;
                this.color = color;
                this.direction = 'up'; // 'up', 'down', 'left', 'right'
                this.isPlayer = isPlayer;
                this.speed = isPlayer ? PLAYER_SPEED : ENEMY_SPEED;
                this.lastShot = 0;
                this.shotCooldown = isPlayer ? 300 : 1000; // 毫秒
            }
            
            draw() {
                ctx.fillStyle = this.color;
                ctx.fillRect(this.x, this.y, this.width, this.height);
                
                // 绘制坦克炮管
                ctx.fillStyle = this.color;
                const centerX = this.x + this.width / 2;
                const centerY = this.y + this.height / 2;
                const barrelLength = this.width * 0.8;
                
                ctx.strokeStyle = this.color;
                ctx.lineWidth = 3;
                ctx.beginPath();
                
                switch(this.direction) {
                    case 'up':
                        ctx.moveTo(centerX, this.y);
                        ctx.lineTo(centerX, this.y - barrelLength);
                        break;
                    case 'down':
                        ctx.moveTo(centerX, this.y + this.height);
                        ctx.lineTo(centerX, this.y + this.height + barrelLength);
                        break;
                    case 'left':
                        ctx.moveTo(this.x, centerY);
                        ctx.lineTo(this.x - barrelLength, centerY);
                        break;
                    case 'right':
                        ctx.moveTo(this.x + this.width, centerY);
                        ctx.lineTo(this.x + this.width + barrelLength, centerY);
                        break;
                }
                
                ctx.stroke();
                
                // 绘制坦克中心
                ctx.fillStyle = this.isPlayer ? '#0ff' : '#f0f';
                ctx.fillRect(centerX - 3, centerY - 3, 6, 6);
            }
            
            move(direction) {
                this.direction = direction;
                let newX = this.x;
                let newY = this.y;
                
                switch(direction) {
                    case 'up':
                        newY -= this.speed;
                        break;
                    case 'down':
                        newY += this.speed;
                        break;
                    case 'left':
                        newX -= this.speed;
                        break;
                    case 'right':
                        newX += this.speed;
                        break;
                }
                
                // 边界检测
                if (newX >= 0 && newX + this.width <= GAME_WIDTH && 
                    newY >= 0 && newY + this.height <= GAME_HEIGHT) {
                    
                    // 碰撞检测
                    let canMove = true;
                    for (let wall of walls) {
                        if (this.checkCollision(newX, newY, this.width, this.height, wall)) {
                            canMove = false;
                            break;
                        }
                    }
                    
                    // 玩家与敌方坦克碰撞检测
                    if (this.isPlayer) {
                        for (let enemy of enemies) {
                            if (this.checkCollision(newX, newY, this.width, this.height, enemy)) {
                                canMove = false;
                                break;
                            }
                        }
                    }
                    
                    if (canMove) {
                        this.x = newX;
                        this.y = newY;
                    }
                }
            }
            
            checkCollision(x, y, width, height, other) {
                return x < other.x + other.width &&
                       x + width > other.x &&
                       y < other.y + other.height &&
                       y + height > other.y;
            }
            
            canShoot() {
                const now = Date.now();
                return now - this.lastShot > this.shotCooldown;
            }
            
            shoot() {
                if (!this.canShoot()) return;
                
                this.lastShot = Date.now();
                const centerX = this.x + this.width / 2;
                const centerY = this.y + this.height / 2;
                
                const bullet = new Bullet(
                    centerX, 
                    centerY, 
                    this.direction, 
                    this.isPlayer ? BULLET_SPEED : ENEMY_BULLET_SPEED,
                    this.isPlayer
                );
                
                if (this.isPlayer) {
                    bullets.push(bullet);
                } else {
                    enemyBullets.push(bullet);
                }
            }
        }
        
        // 子弹类
        class Bullet {
            constructor(x, y, direction, speed, isPlayerBullet = true) {
                this.x = x;
                this.y = y;
                this.direction = direction;
                this.speed = speed;
                this.width = TILE_SIZE / 2;
                this.height = TILE_SIZE / 2;
                this.isPlayerBullet = isPlayerBullet;
                this.active = true;
            }
            
            draw() {
                ctx.fillStyle = this.isPlayerBullet ? '#0f0' : '#f00';
                ctx.fillRect(this.x - this.width/2, this.y - this.height/2, this.width, this.height);
            }
            
            move() {
                switch(this.direction) {
                    case 'up':
                        this.y -= this.speed;
                        break;
                    case 'down':
                        this.y += this.speed;
                        break;
                    case 'left':
                        this.x -= this.speed;
                        break;
                    case 'right':
                        this.x += this.speed;
                        break;
                }
                
                // 检查是否超出边界
                if (this.x < 0 || this.x > GAME_WIDTH || 
                    this.y < 0 || this.y > GAME_HEIGHT) {
                    this.active = false;
                }
            }
        }
        
        // 墙壁类
        class Wall {
            constructor(x, y, destructible = true) {
                this.x = x;
                this.y = y;
                this.width = TILE_SIZE;
                this.height = TILE_SIZE;
                this.destructible = destructible;
                this.active = true;
            }
            
            draw() {
                ctx.fillStyle = this.destructible ? '#8B4513' : '#555';
                ctx.fillRect(this.x, this.y, this.width, this.height);
                
                // 添加砖块纹理
                if (this.destructible) {
                    ctx.strokeStyle = '#654321';
                    ctx.lineWidth = 1;
                    ctx.beginPath();
                    ctx.moveTo(this.x + this.width/2, this.y);
                    ctx.lineTo(this.x + this.width/2, this.y + this.height);
                    ctx.moveTo(this.x, this.y + this.height/2);
                    ctx.lineTo(this.x + this.width, this.y + this.height/2);
                    ctx.stroke();
                }
            }
        }
        
        // 初始化游戏
        function initGame() {
            gameState = 'playing';
            score = 0;
            keys = {};
            
            // 创建玩家坦克
            player = new Tank(GAME_WIDTH / 2 - TILE_SIZE, GAME_HEIGHT - TILE_SIZE * 2, '#0ff', true);
            
            // 创建敌方坦克
            enemies = [];
            for (let i = 0; i < 3; i++) {
                const x = TILE_SIZE * (i + 1);
                const y = TILE_SIZE;
                enemies.push(new Tank(x, y, '#f0f', false));
            }
            
            // 创建子弹数组
            bullets = [];
            enemyBullets = [];
            
            // 创建墙壁
            walls = [];
            
            // 创建边界墙
            for (let x = 0; x < GAME_WIDTH; x += TILE_SIZE) {
                walls.push(new Wall(x, 0, false)); // 上边界
                walls.push(new Wall(x, GAME_HEIGHT - TILE_SIZE, false)); // 下边界
            }
            
            for (let y = TILE_SIZE; y < GAME_HEIGHT - TILE_SIZE; y += TILE_SIZE) {
                walls.push(new Wall(0, y, false)); // 左边界
                walls.push(new Wall(GAME_WIDTH - TILE_SIZE, y, false)); // 右边界
            }
            
            // 创建内部障碍物
            // 水平墙
            for (let x = TILE_SIZE * 2; x < GAME_WIDTH - TILE_SIZE * 2; x += TILE_SIZE * 3) {
                for (let y = TILE_SIZE * 3; y < GAME_HEIGHT / 2; y += TILE_SIZE * 3) {
                    walls.push(new Wall(x, y, true));
                    walls.push(new Wall(x + TILE_SIZE, y, true));
                }
            }
            
            // 垂直墙
            for (let y = TILE_SIZE * 2; y < GAME_HEIGHT - TILE_SIZE * 2; y += TILE_SIZE * 3) {
                for (let x = TILE_SIZE * 3; x < GAME_WIDTH / 2; x += TILE_SIZE * 3) {
                    walls.push(new Wall(x, y, true));
                    walls.push(new Wall(x, y + TILE_SIZE, true));
                }
            }
            
            // 更新分数显示
            updateScore(0);
            
            // 隐藏游戏结束画面
            gameOverElement.style.display = 'none';
        }
        
        // 更新分数
        function updateScore(points) {
            score += points;
            scoreElement.textContent = `得分: ${score}`;
        }
        
        // 处理玩家输入
        function handlePlayerInput() {
            if (gameState !== 'playing' || !player) return;
            
            // 移动控制
            if (keys['ArrowUp'] || keys['w'] || keys['W']) {
                player.move('up');
            }
            if (keys['ArrowDown'] || keys['s'] || keys['S']) {
                player.move('down');
            }
            if (keys['ArrowLeft'] || keys['a'] || keys['A']) {
                player.move('left');
            }
            if (keys['ArrowRight'] || keys['d'] || keys['D']) {
                player.move('right');
            }
            
            // 射击控制
            if (keys[' ']) {
                player.shoot();
            }
        }
        
        // 移动敌方坦克
        function moveEnemyTanks() {
            if (gameState !== 'playing') return;
            
            enemies.forEach(enemy => {
                // 随机移动方向
                const directions = ['up', 'down', 'left', 'right'];
                const randomDirection = directions[Math.floor(Math.random() * directions.length)];
                
                // 30%的概率改变方向
                if (Math.random() < 0.3) {
                    enemy.move(randomDirection);
                }
                
                // 20%的概率射击
                if (Math.random() < 0.2) {
                    enemy.shoot();
                }
            });
        }
        
        // 移动子弹
        function moveBullets() {
            // 移动玩家子弹
            bullets = bullets.filter(bullet => {
                bullet.move();
                
                // 检查子弹是否击中墙壁
                for (let i = 0; i < walls.length; i++) {
                    const wall = walls[i];
                    if (bullet.active && bullet.checkCollision(
                        bullet.x - bullet.width/2, 
                        bullet.y - bullet.height/2, 
                        bullet.width, 
                        bullet.height, 
                        wall
                    )) {
                        bullet.active = false;
                        
                        // 如果是可破坏的墙，则移除
                        if (wall.destructible) {
                            wall.active = false;
                            walls.splice(i, 1);
                            updateScore(10);
                        }
                        break;
                    }
                }
                
                // 检查子弹是否击中敌方坦克
                for (let i = 0; i < enemies.length; i++) {
                    const enemy = enemies[i];
                    if (bullet.active && bullet.checkCollision(
                        bullet.x - bullet.width/2, 
                        bullet.y - bullet.height/2, 
                        bullet.width, 
                        bullet.height, 
                        enemy
                    )) {
                        bullet.active = false;
                        enemies.splice(i, 1);
                        updateScore(100);
                        break;
                    }
                }
                
                return bullet.active;
            });
            
            // 移动敌方子弹
            enemyBullets = enemyBullets.filter(bullet => {
                bullet.move();
                
                // 检查子弹是否击中墙壁
                for (let i = 0; i < walls.length; i++) {
                    const wall = walls[i];
                    if (bullet.active && bullet.checkCollision(
                        bullet.x - bullet.width/2, 
                        bullet.y - bullet.height/2, 
                        bullet.width, 
                        bullet.height, 
                        wall
                    )) {
                        bullet.active = false;
                        
                        // 如果是可破坏的墙，则移除
                        if (wall.destructible) {
                            wall.active = false;
                            walls.splice(i, 1);
                        }
                        break;
                    }
                }
                
                // 检查子弹是否击中玩家
                if (bullet.active && bullet.checkCollision(
                    bullet.x - bullet.width/2, 
                    bullet.y - bullet.height/2, 
                    bullet.width, 
                    bullet.height, 
                    player
                )) {
                    bullet.active = false;
                    gameState = 'gameOver';
                    gameOverElement.style.display = 'block';
                }
                
                return bullet.active;
            });
        }
        
        // 检查游戏是否结束
        function checkGameOver() {
            if (gameState === 'playing') {
                // 检查玩家是否被击中
                if (!player) {
                    gameState = 'gameOver';
                    gameOverElement.style.display = 'block';
                    return;
                }
                
                // 检查是否所有敌人都被消灭
                if (enemies.length === 0) {
                    // 生成新的敌人
                    for (let i = 0; i < 3; i++) {
                        const x = TILE_SIZE * (i + 1);
                        const y = TILE_SIZE;
                        enemies.push(new Tank(x, y, '#f0f', false));
                    }
                    updateScore(500); // 消灭所有敌人的奖励
                }
            }
        }
        
        // 渲染游戏
        function render() {
            // 清空画布
            ctx.fillStyle = '#000';
            ctx.fillRect(0, 0, GAME_WIDTH, GAME_HEIGHT);
            
            // 绘制墙壁
            walls.forEach(wall => wall.draw());
            
            // 绘制玩家
            if (player) {
                player.draw();
            }
            
            // 绘制敌人
            enemies.forEach(enemy => enemy.draw());
            
            // 绘制子弹
            bullets.forEach(bullet => bullet.draw());
            enemyBullets.forEach(bullet => bullet.draw());
        }
        
        // 游戏主循环
        function gameLoop() {
            handlePlayerInput();
            moveEnemyTanks();
            moveBullets();
            checkGameOver();
            render();
            
            requestAnimationFrame(gameLoop);
        }
        
        // 键盘事件监听
        window.addEventListener('keydown', (e) => {
            keys[e.key] = true;
            
            // 重新开始游戏
            if (e.key === 'r' || e.key === 'R') {
                initGame();
            }
        });
        
        window.addEventListener('keyup', (e) => {
            keys[e.key] = false;
        });
        
        // 启动游戏
        initGame();
        gameLoop();
    </script>
</body>
</html>
```

这个实现包含了以下功能：

1. **玩家控制**：
   - 使用WASD或方向键控制坦克移动
   - 空格键射击
   - 边界检测确保坦克不会移出游戏区域

2. **敌方AI**：
   - 3个敌方坦克自动移动
   - 随机改变方向
