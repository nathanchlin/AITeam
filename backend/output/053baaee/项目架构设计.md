## 2. 游戏循环机制

### 2.1 游戏循环流程
```javascript
// 游戏循环
function gameLoop() {
    // 1. 处理输入
    handleInput();
    
    // 2. 更新游戏状态
    updateGame();
    
    // 3. 渲染游戏画面
    renderGame();
    
    // 4. 请求下一帧
    if (!gameOver) {
        requestAnimationFrame(gameLoop);
    }
}

// 游戏状态更新
function updateGame() {
    // 更新玩家坦克
    updatePlayerTank();
    
    // 更新敌方坦克
    updateEnemyTanks();
    
    // 更新子弹
    updateBullets();
    
    // 更新障碍物
    updateObstacles();
    
    // 检测碰撞
    checkCollisions();
    
    // 检查游戏结束条件
    checkGameOverCondition();
}
```

### 2.2 游戏帧率控制

- 目标帧率：60FPS
- 基于时间的移动计算，确保不同刷新率下游戏速度一致

// 输入处理模块
const InputHandler = {
    init() {
        // 键盘事件监听
        window.addEventListener('keydown', (e) => {
            gameState.keys[e.key.toLowerCase()] = true;
            
            // 防止方向键滚动页面
            if(['ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight', ' '].includes(e.key)) {
                e.preventDefault();
            }
        });
        
        window.addEventListener('keyup', (e) => {
            gameState.keys[e.key.toLowerCase()] = false;
        });
    },
    
    handlePlayerMovement() {
        const player = gameState.player;
        let moved = false;
        
        // WASD或方向键控制
        if (gameState.keys['w'] || gameState.keys['arrowup']) {
            player.direction = 'up';
            if (player.y > 0) {
                player.y -= player.speed;
                moved = true;
            }
        }
        if (gameState.keys['s'] || gameState.keys['arrowdown']) {
            player.direction = 'down';
            if (player.y < canvas.height - player.height) {
                player.y += player.speed;
                moved = true;
            }
        }
        if (gameState.keys['a'] || gameState.keys['arrowleft']) {
            player.direction = 'left';
            if (player.x > 0) {
                player.x -= player.speed;
                moved = true;
            }
        }
        if (gameState.keys['d'] || gameState.keys['arrowright']) {
            player.direction = 'right';
            if (player.x < canvas.width - player.width) {
                player.x += player.speed;
                moved = true;
            }
        }
        
        // 空格键射击
        if (gameState.keys[' ']) {
            this.shoot(player);
        }
    },
    
    shoot(tank) {
        const now = Date.now();
        if (now - tank.lastShot > tank.shotCooldown) {
            const bullet = createBullet(tank);
            gameState.bullets.push(bullet);
            tank.lastShot = now;
        }
    }
};
```

### 3.2 渲染模块
```javascript
// 渲染模块
const Renderer = {
    // 绘制坦克
    drawTank(tank, color) {
        ctx.fillStyle = color;
        
        // 根据方向绘制坦克
        switch(tank.direction) {
            case 'up':
                ctx.fillRect(tank.x + 10, tank.y, 10, 20);
                ctx.fillRect(tank.x, tank.y + 5, 30, 15);
                ctx.fillRect(tank.x + 12, tank.y - 5, 6, 10);
                break;
            case 'down':
                ctx.fillRect(tank.x + 10, tank.y + 10, 10, 20);
                ctx.fillRect(tank.x, tank.y + 10, 30, 15);
                ctx.fillRect(tank.x + 12, tank.y + 25, 6, 10);
                break;
            case 'left':
                ctx.fillRect(tank.x, tank.y + 10, 20, 10);
                ctx.fillRect(tank.x + 5, tank.y, 15, 30);
                ctx.fillRect(tank.x - 5, tank.y + 12, 10, 6);
                break;
            case 'right':
                ctx.fillRect(tank.x + 10, tank.y + 10, 20, 10);
                ctx.fillRect(tank.x + 10, tank.y, 15, 30);
                ctx.fillRect(tank.x + 25, tank.y + 12, 10, 6);
                break;
        }
    },
    
    // 绘制子弹
    drawBullet(bullet) {
        ctx.fillStyle = bullet.color;
        ctx.fillRect(bullet.x, bullet.y, bullet.width, bullet.height);
    },
    
    // 绘制障碍物（砖墙）
    drawObstacle(obstacle) {
        ctx.fillStyle = obstacle.color;
        for (let i = 0; i < obstacle.blocks.length; i++) {
            const block = obstacle.blocks[i];
            ctx.fillRect(block.x, block.y, block.width, block.height);
            
            // 添加砖块纹理
            ctx.strokeStyle = '#8b4513';
            ctx.lineWidth = 1;
            ctx.strokeRect(block.x, block.y, block.width, block.height);
            
            // 砖块分割线
            ctx.beginPath();
            ctx.moveTo(block.x + block.width/2, block.y);
            ctx.lineTo(block.x + block.width/2, block.y + block.height);
            ctx.stroke();
        }
    },
    
    // 渲染游戏画面
    render() {
        // 清空画布
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        
        // 绘制背景网格（像素风格）
        this.drawGrid();
        
        // 绘制障碍物
        gameState.obstacles.forEach(obstacle => this.drawObstacle(obstacle));
        
        // 绘制玩家坦克
        this.drawTank(gameState.player, '#3498db');
        
        // 绘制敌方坦克
        gameState.enemies.forEach(enemy => this.drawTank(enemy, '#e74c3c'));
        
        // 绘制子弹
        gameState.bullets.forEach(bullet => this.drawBullet(bullet));
        
        // 更新得分显示
        document.getElementById('score').textContent = gameState.score;
    },
    
    // 绘制背景网格
    drawGrid() {
        ctx.strokeStyle = '#2c3e50';
        ctx.lineWidth = 0.5;
        
        // 垂直线
        for (let x = 0; x <= canvas.width; x += 20) {
            ctx.beginPath();
            ctx.moveTo(x, 0);
            ctx.lineTo(x, canvas.height);
            ctx.stroke();
        }
        
        // 水平线
        for (let y = 0; y <= canvas.height; y += 20) {
            ctx.beginPath();
            ctx.moveTo(0, y);
            ctx.lineTo(canvas.width, y);
            ctx.stroke();
        }
    }
};
```

### 3.3 游戏逻辑模块
```javascript
// 游戏逻辑模块
const GameLogic = {
    // 初始化游戏
    init() {
        // 创建玩家坦克
        gameState.player = {
            x: 400,
            y: 500,
            width: 30,
            height: 30,
            speed: 3,
            direction: 'up',
            lastShot: 0,
            shotCooldown: 300
        };
        
        // 创建敌方坦克
        gameState.enemies = [];
        for (let i = 0; i < 3; i++) {
            gameState.enemies.push(this.createEnemyTank());
        }
        
        // 创建障碍物（砖墙）
        gameState.obstacles = [];
        this.createObstacles();
        
        // 重置游戏状态
        gameState.score = 0;
        gameState.isGameOver = false;
        gameState.bullets = [];
        
        // 隐藏游戏结束界面
        document.getElementById('gameOver').style.display = 'none';
    },
    
    // 创建敌方坦克
    createEnemyTank() {
        return {
            x: Math.random() * (canvas.width - 30),
            y: Math.random() * 200,
            width: 30,
            height: 30,
            speed: 1.5,
            direction: this.getRandomDirection(),
            lastShot: 0,
            shotCooldown: 1000 + Math.random() * 2000,
            moveTimer: 0,
            moveInterval: 500 + Math.random() * 1000
        };
    },
    
    // 获取随机方向
    getRandomDirection() {
        const directions = ['up', 'down', 'left', 'right'];
        return directions[Math.floor(Math.random() * directions.length)];
    },
    
    // 创建障碍物
    createObstacles() {
        // 创建几个砖墙区域
        const obstaclePositions = [
            {x: 200, y: 200},
            {x: 500, y: 200},
            {x: 350, y: 350},
            {x: 100, y: 350},
            {x: 600, y: 350}
        ];
        
        obstaclePositions.forEach(pos => {
            const obstacle = {
                x: pos.x,
                y: pos.y,
                width: 100,
                height: 20,
                color: '#8b4513',
                blocks: []
            };
            
            // 创建砖块
            for (let i = 0; i < 5; i++) {
                for (let j = 0; j < 1; j++) {
                    obstacle.blocks.push({
                        x: pos.x + i * 20,
                        y: pos.y + j * 20,
                        width: 20,
                        height: 20
                    });
                }
            }
            
            gameState.obstacles.push(obstacle);
        });
    },
    
    // 更新游戏状态
    update() {
        // 更新玩家坦克
        this.updatePlayerTank();
        
        // 更新敌方坦克
        this.updateEnemyTanks();
        
        // 更新子弹
        this.updateBullets();
        
        // 检测碰撞
        this.checkCollisions();
        
        // 检查游戏结束条件
        this.checkGameOverCondition();
    },
    
    // 更新玩家坦克
    updatePlayerTank() {
        const player = gameState.player;
        InputHandler.handlePlayerMovement();
    },
    
    // 更新敌方坦克
    updateEnemyTanks() {
        gameState.enemies.forEach(enemy => {
            // 移动逻辑
            enemy.moveTimer += 16; // 假设60FPS，每帧约16ms
            if (enemy.moveTimer >= enemy.moveInterval) {
                enemy.moveTimer = 0;
                
                // 随机改变方向
                if (Math.random() < 0.3) {
                    enemy.direction = this.getRandomDirection();
                }
                
                // 尝试移动
                this.moveTank(enemy);
            }
            
            // 射击逻辑
            const now = Date.now();
            if (now - enemy.lastShot > enemy.shotCooldown) {
                if (Math.random() < 0.02) { // 2%的概率每帧射击
                    this.shootBullet(enemy);
                    enemy.lastShot = now;
                }
            }
        });
    },
    
    // 移动坦克
    moveTank(tank) {
        const oldX = tank.x;
        const oldY = tank.y;
        
        switch(tank.direction) {
            case 'up':
                tank.y = Math.max(0, tank.y - tank.speed);
                break;
            case 'down':
                tank.y = Math.min(canvas.height - tank.height, tank.y + tank.speed);
                break;
            case 'left':
                tank.x = Math.max(0, tank.x - tank.speed);
                break;
            case 'right':
                tank.x = Math.min(canvas.width - tank.width, tank.x + tank.speed);
                break;
        }
        
        // 检查是否与障碍物碰撞
        if (this.checkTankObstacleCollision(tank)) {
            tank.x = oldX;
            tank.y = oldY;
        }
    },
    
    // 更新子弹
    updateBullets() {
        for (let i = gameState.bullets.length - 1; i >= 0; i--) {
            const bullet = gameState.bullets[i];
            
            // 移动子弹
            switch(bullet.direction) {
                case 'up':
                    bullet.y -= bullet.speed;
                    break;
                case 'down':
                    bullet.y += bullet.speed;
                    break;
                case 'left':
                    bullet.x -= bullet.speed;
                    break;
                case 'right':
                    bullet.x += bullet.speed;
                    break;
            }
            
            // 检查子弹是否超出边界
            if (bullet.x < 0 || bullet.x > canvas.width || 
                bullet.y < 0 || bullet.y > canvas.height) {
                gameState.bullets.splice(i, 1);
            }
        }
    },
    
    // 创建子弹
    createBullet(tank) {
        const bulletSize = 4;
        let x, y;
        
        switch(tank.direction) {
            case 'up':
                x = tank.x + tank.width/2 - bulletSize/2;
                y = tank.y;
                break;
            case 'down':
                x = tank.x + tank.width/2 - bulletSize/2;
                y = tank.y + tank.height;
                break;
            case 'left':
                x = tank.x;
                y = tank.y + tank.height/2 - bulletSize/2;
                break;
            case 'right':
                x = tank.x + tank.width;
                y = tank.y + tank.height/2 - bulletSize/2;
                break;
        }
        
        return {
            x: x,
            y: y,
            width: bulletSize,
            height: bulletSize,
            speed: 5,
            direction: tank.direction,
            color: tank === gameState.player ? '#f1c40f' : '#e67e22',
            owner: tank
        };
    },
    
    // 射击子弹
    shootBullet(tank) {
        const bullet = this.createBullet(tank);
        gameState.bullets.push(bullet);
    },
    
    // 检测碰撞
    checkCollisions() {
        // 子弹与坦克碰撞
        for (let i = gameState.bullets.length - 1; i >= 0; i--) {
            const bullet = gameState.bullets[i];
            
            // 检查子弹是否击中玩家
            if (bullet.owner !== gameState.player && this.checkBulletTankCollision(bullet, gameState.player)) {
                gameState.bullets.splice(i, 1);
                this.gameOver();
                continue;
            }
            
            // 检查子弹是否击中敌方坦克
            for (let j = gameState.enemies.length - 1; j >= 0; j--) {
                const enemy = gameState.enemies[j];
                if (bullet.owner === gameState.player && this.checkBulletTankCollision(bullet, enemy)) {
                    gameState.bullets.splice(i, 1);
                    gameState.enemies.splice(j, 1);
                    gameState.score += 100;
                    
                    // 添加新敌方坦克
                    gameState.enemies.push(this.createEnemyTank());
                    break;
                }
            }
            
            // 检查子弹是否击中障碍物
            for (let k = gameState.obstacles.length - 1
