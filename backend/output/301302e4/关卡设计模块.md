import json
import pygame
from enum import Enum
from dataclasses import dataclass, asdict
from typing import List, Dict, Tuple, Optional

class ObstacleType(Enum):
    BLOCK = "block"
    CIRCLE = "circle"
    TRIANGLE = "triangle"
    PLATFORM = "platform"
    PIG = "pig"

@dataclass
class Obstacle:
    type: ObstacleType
    x: float
    y: float
    width: float
    height: float
    rotation: float = 0.0
    properties: Dict = None
    
    def __post_init__(self):
        if self.properties is None:
            self.properties = {}

class LevelEditor:
    def __init__(self, screen_width, screen_height):
        self.screen_width = screen_width
        self.screen_height = screen_height
        self.obstacles: List[Obstacle] = []
        self.selected_obstacle_type = ObstacleType.BLOCK
        self.selected_obstacle = None
        self.dragging = False
        self.grid_size = 20
        self.show_grid = True
        self.camera_x = 0
        self.camera_y = 0
        self.zoom = 1.0
        
    def add_obstacle(self, x, y):
        """添加新障碍物"""
        # 对齐到网格
        x = round(x / self.grid_size) * self.grid_size
        y = round(y / self.grid_size) * self.grid_size
        
        # 创建新障碍物
        if self.selected_obstacle_type == ObstacleType.BLOCK:
            obstacle = Obstacle(
                type=self.selected_obstacle_type,
                x=x, y=y,
                width=60, height=40,
                rotation=0.0
            )
        elif self.selected_obstacle_type == ObstacleType.CIRCLE:
            obstacle = Obstacle(
                type=self.selected_obstacle_type,
                x=x, y=y,
                width=40, height=40,
                rotation=0.0
            )
        elif self.selected_obstacle_type == ObstacleType.TRIANGLE:
            obstacle = Obstacle(
                type=self.selected_obstacle_type,
                x=x, y=y,
                width=50, height=50,
                rotation=0.0
            )
        elif self.selected_obstacle_type == ObstacleType.PLATFORM:
            obstacle = Obstacle(
                type=self.selected_obstacle_type,
                x=x, y=y,
                width=100, height=20,
                rotation=0.0
            )
        elif self.selected_obstacle_type == ObstacleType.PIG:
            obstacle = Obstacle(
                type=self.selected_obstacle_type,
                x=x, y=y,
                width=30, height=30,
                rotation=0.0,
                properties={"health": 100}
            )
        
        self.obstacles.append(obstacle)
        return obstacle
    
    def remove_obstacle(self, x, y):
        """移除指定位置的障碍物"""
        for obstacle in self.obstacles:
            if self.point_in_obstacle(x, y, obstacle):
                self.obstacles.remove(obstacle)
                return True
        return False
    
    def point_in_obstacle(self, x, y, obstacle):
        """检查点是否在障碍物内"""
        # 简化的碰撞检测，实际应根据形状实现
        return (obstacle.x <= x <= obstacle.x + obstacle.width and
                obstacle.y <= y <= obstacle.y + obstacle.height)
    
    def select_obstacle(self, x, y):
        """选择障碍物"""
        for obstacle in self.obstacles:
            if self.point_in_obstacle(x, y, obstacle):
                self.selected_obstacle = obstacle
                return obstacle
        return None
    
    def move_obstacle(self, dx, dy):
        """移动选中的障碍物"""
        if self.selected_obstacle:
            self.selected_obstacle.x += dx
            self.selected_obstacle.y += dy
    
    def rotate_obstacle(self, angle):
        """旋转选中的障碍物"""
        if self.selected_obstacle:
            self.selected_obstacle.rotation += angle
    
    def resize_obstacle(self, dw, dh):
        """调整选中障碍物的大小"""
        if self.selected_obstacle:
            self.selected_obstacle.width = max(20, self.selected_obstacle.width + dw)
            self.selected_obstacle.height = max(20, self.selected_obstacle.height + dh)
    
    def save_level(self, filename):
        """保存关卡到文件"""
        level_data = {
            "obstacles": [asdict(obs) for obs in self.obstacles],
            "screen_width": self.screen_width,
            "screen_height": self.screen_height
        }
        
        with open(filename, 'w') as f:
            json.dump(level_data, f, indent=2)
    
    def load_level(self, filename):
        """从文件加载关卡"""
        with open(filename, 'r') as f:
            level_data = json.load(f)
        
        self.obstacles = []
        for obs_data in level_data["obstacles"]:
            # 转换字符串类型的枚举值为实际枚举
            obs_data["type"] = ObstacleType(obs_data["type"])
            self.obstacles.append(Obstacle(**obs_data))
        
        self.screen_width = level_data.get("screen_width", self.screen_width)
        self.screen_height = level_data.get("screen_height", self.screen_height)
    
    def draw(self, screen):
        """绘制关卡编辑器"""
        # 绘制网格
        if self.show_grid:
            for x in range(0, self.screen_width, self.grid_size):
                pygame.draw.line(screen, (200, 200, 200), (x, 0), (x, self.screen_height))
            for y in range(0, self.screen_height, self.grid_size):
                pygame.draw.line(screen, (200, 200, 200), (0, y), (self.screen_width, y))
        
        # 绘制所有障碍物
        for obstacle in self.obstacles:
            self.draw_obstacle(screen, obstacle)
        
        # 高亮选中的障碍物
        if self.selected_obstacle:
            pygame.draw.rect(
                screen, 
                (255, 255, 0), 
                (self.selected_obstacle.x - 2, 
                 self.selected_obstacle.y - 2, 
                 self.selected_obstacle.width + 4, 
                 self.selected_obstacle.height + 4), 
                2
            )
    
    def draw_obstacle(self, screen, obstacle):
        """绘制单个障碍物"""
        if obstacle.type == ObstacleType.BLOCK:
            pygame.draw.rect(screen, (139, 69, 19), 
                            (obstacle.x, obstacle.y, obstacle.width, obstacle.height))
        elif obstacle.type == ObstacleType.CIRCLE:
            pygame.draw.circle(screen, (100, 100, 100), 
                             (int(obstacle.x + obstacle.width/2), int(obstacle.y + obstacle.height/2)), 
                             int(obstacle.width/2))
        elif obstacle.type == ObstacleType.TRIANGLE:
            points = [
                (obstacle.x + obstacle.width/2, obstacle.y),
                (obstacle.x, obstacle.y + obstacle.height),
                (obstacle.x + obstacle.width, obstacle.y + obstacle.height)
            ]
            pygame.draw.polygon(screen, (150, 150, 150), points)
        elif obstacle.type == ObstacleType.PLATFORM:
            pygame.draw.rect(screen, (105, 105, 105), 
                            (obstacle.x, obstacle.y, obstacle.width, obstacle.height))
        elif obstacle.type == ObstacleType.PIG:
            pygame.draw.circle(screen, (0, 200, 0), 
                             (int(obstacle.x + obstacle.width/2), int(obstacle.y + obstacle.height/2)), 
                             int(obstacle.width/2))

## 2. 关卡管理系统

```python
class LevelManager:
    def __init__(self):
        self.levels = {}
        self.current_level = None
        self.editor = None
    
    def create_editor(self, screen_width, screen_height):
        """创建关卡编辑器"""
        self.editor = LevelEditor(screen_width, screen_height)
        return self.editor
    
    def save_level(self, level_id, filename=None):
        """保存当前关卡"""
        if not self.editor:
            raise Exception("No editor available")
        
        if filename is None:
            filename = f"level_{level_id}.json"
        
        self.editor.save_level(filename)
        self.levels[level_id] = filename
        return filename
    
    def load_level(self, level_id):
        """加载指定关卡"""
        if level_id not in self.levels:
            raise Exception(f"Level {level_id} not found")
        
        if not self.editor:
            self.editor = LevelEditor(800, 600)
        
        self.editor.load_level(self.levels[level_id])
        self.current_level = level_id
        return self.editor.obstacles
    
    def list_levels(self):
        """列出所有已保存的关卡"""
        return list(self.levels.keys())
    
    def delete_level(self, level_id):
        """删除指定关卡"""
        if level_id in self.levels:
            del self.levels[level_id]
            if self.current_level == level_id:
                self.current_level = None
```

class GameLevelLoader:
    @staticmethod
    def load_level(filename):
        """从文件加载关卡用于游戏"""
        with open(filename, 'r') as f:
            level_data = json.load(f)
        
        obstacles = []
        for obs_data in level_data["obstacles"]:
            # 转换字符串类型的枚举值为实际枚举
            obs_data["type"] = ObstacleType(obs_data["type"])
            obstacles.append(Obstacle(**obs_data))
        
        return obstacles
    
    @staticmethod
    def create_physics_bodies(obstacles, physics_world):
        """为障碍物创建物理体"""
        bodies = []
        for obstacle in obstacles:
            if obstacle.type == ObstacleType.BLOCK:
                # 创建矩形物理体
                body = physics_world.create_box(
                    position=(obstacle.x, obstacle.y),
                    size=(obstacle.width, obstacle.height),
                    mass=5.0,
                    friction=0.5
                )
            elif obstacle.type == ObstacleType.CIRCLE:
                # 创建圆形物理体
                body = physics_world.create_circle(
                    position=(obstacle.x + obstacle.width/2, obstacle.y + obstacle.height/2),
                    radius=obstacle.width/2,
                    mass=3.0,
                    friction=0.3
                )
            elif obstacle.type == ObstacleType.TRIANGLE:
                # 创建多边形物理体
                body = physics_world.create_polygon(
                    vertices=[
                        (obstacle.x + obstacle.width/2, obstacle.y),
                        (obstacle.x, obstacle.y + obstacle.height),
                        (obstacle.x + obstacle.width, obstacle.y + obstacle.height)
                    ],
                    mass=4.0,
                    friction=0.4
                )
            elif obstacle.type == ObstacleType.PLATFORM:
                # 创建静态平台
                body = physics_world.create_box(
                    position=(obstacle.x, obstacle.y),
                    size=(obstacle.width, obstacle.height),
                    mass=0.0,  # 静态物体
                    friction=0.7
                )
            elif obstacle.type == ObstacleType.PIG:
                # 创建猪（可破坏）
                body = physics_world.create_circle(
                    position=(obstacle.x + obstacle.width/2, obstacle.y + obstacle.height/2),
                    radius=obstacle.width/2,
                    mass=1.0,
                    friction=0.2,
                    destructible=True,
                    health=obstacle.properties.get("health", 100)
                )
            
            if body:
                body.obstacle_type = obstacle.type
                bodies.append(body)
        
        return bodies

## 4. 关卡编辑器UI

```python
class EditorUI:
    def __init__(self, editor, x, y, width, height):
        self.editor = editor
        self.x = x
        self.y = y
        self.width = width
        self.height = height
        self.buttons = []
        self.create_buttons()
    
    def create_buttons(self):
        """创建UI按钮"""
        button_y = self.y + 10
        button_height = 30
        button_spacing = 5
        
        # 障碍物类型按钮
        for obstacle_type in ObstacleType:
            button = {
                "rect": pygame.Rect(self.x + 10, button_y, self.width - 20, button_height),
                "text": obstacle_type.value.capitalize(),
                "action": lambda t=obstacle_type: setattr(self.editor, 'selected_obstacle_type', t)
            }
            self.buttons.append(button)
            button_y += button_height + button_spacing
        
        # 工具按钮
        tool_y = button_y + 20
        tools = [
            ("Toggle Grid", lambda: setattr(self.editor, 'show_grid', not self.editor.show_grid)),
            ("Save Level", self.save_level),
            ("Load Level", self.load_level),
            ("Clear Level", self.clear_level)
        ]
        
        for tool_name, tool_action in tools:
            button = {
                "rect": pygame.Rect(self.x + 10, tool_y, self.width - 20, button_height),
                "text": tool_name,
                "action": tool_action
            }
            self.buttons.append(button)
            tool_y += button_height + button_spacing
    
    def draw(self, screen, font):
        """绘制UI"""
        # 绘制背景面板
        pygame.draw.rect(screen, (50, 50, 50), (self.x, self.y, self.width, self.height))
        pygame.draw.rect(screen, (200, 200, 200), (self.x, self.y, self.width, self.height), 2)
        
        # 绘制按钮
        for button in self.buttons:
            # 高亮当前选中的障碍物类型
            if button["text"] == self.editor.selected_obstacle_type.value.capitalize():
                pygame.draw.rect(screen, (100, 100, 200), button["rect"])
            else:
                pygame.draw.rect(screen, (100, 100, 100), button["rect"])
            
            pygame.draw.rect(screen, (200, 200, 200), button["rect"], 1)
            
            # 绘制文本
            text = font.render(button["text"], True, (255, 255, 255))
            text_rect = text.get_rect(center=button["rect"].center)
            screen.blit(text, text_rect)
    
    def handle_event(self, event):
        """处理UI事件"""
        if event.type == pygame.MOUSEBUTTONDOWN:
            mouse_pos = pygame.mouse.get_pos()
            for button in self.buttons:
                if button["rect"].collidepoint(mouse_pos):
                    button["action"]()
                    return True
        return False
    
    def save_level(self):
        """保存当前关卡"""
        filename = f"level_{len(self.editor.obstacles)}.json"
        self.editor.save_level(filename)
        print(f"Level saved to {filename}")
    
    def load_level(self):
        """加载关卡"""
        # 这里可以实现文件选择对话框，简化版使用固定文件名
        try:
            self.editor.load_level("level_1.json")
            print("Level loaded successfully")
        except Exception as e:
            print(f"Failed to load level: {e}")
    
    def clear_level(self):
        """清空当前关卡"""
        self.editor.obstacles.clear()
        self.editor.selected_obstacle = None
        print("Level cleared")
```

def main():
    pygame.init()
    screen_width, screen_height = 1200, 800
    screen = pygame.display.set_mode((screen_width, screen_height))
    pygame.display.set_caption("愤怒的小鸟 - 关卡编辑器")
    clock = pygame.time.Clock()
    font = pygame.font.SysFont(None, 24)
    
    # 创建关卡管理器和编辑器
    level_manager = LevelManager()
    editor = level_manager.create_editor(screen_width - 300, screen_height)
    
    # 创建UI
    ui = EditorUI(editor, screen_width - 300, 0, 300, screen_height)
    
    running = True
    while running:
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                running = False
            
            # 处理UI事件
            if ui.handle_event(event):
                continue
            
            # 处理编辑器事件
            if event.type == pygame.MOUSEBUTTONDOWN:
                if event.button == 1:  # 左键
                    mouse_x, mouse_y = pygame.mouse.get_pos()
                    if mouse_x < screen_width - 300:  # 在编辑区域
                        if pygame.key.get_mods() & pygame.KMOD_SHIFT:
                            # Shift+左键删除障碍物
                            editor.remove_obstacle(mouse_x, mouse_y)
                        else:
                            # 左键添加/选择障碍物
                            selected = editor.select_obstacle(mouse_x, mouse_y)
                            if not selected:
                                editor.add_obstacle(mouse_x, mouse_y)
                
                elif event.button == 3:  # 右键
                    editor.selected_obstacle = None
            
            elif event.type == pygame.MOUSEMOTION:
                if pygame.mouse.get_pressed()[0] and editor.selected_obstacle:
                    mouse_x, mouse_y = pygame.mouse.get_pos()
                    if mouse_x < screen_width - 300:
                        editor.move_obstacle(event.rel[0], event.rel[1])
            
            elif event.type == pygame.KEYDOWN:
                if editor.selected_obstacle:
                    if event.key == pygame.K_r:
                        editor.rotate_obstacle(15)
                    elif event.key == pygame.K_t:
                        editor.rotate_obstacle(-15)
                    elif event.key == pygame.K_PLUS or event.key == pygame.K_EQUALS:
                        editor.resize_obstacle(5, 5)
                    elif event.key == pygame.K_MINUS:
                        editor.resize_obstacle(-5, -5)
        
        # 绘制
        screen.fill((255, 255, 255))
        
        # 绘制编辑区域背景
        pygame.draw.rect(screen, (240, 240, 240), (0, 0, screen_width - 300, screen_height))
        
        # 绘制关卡编辑器
        editor.draw(screen)
        
        # 绘制UI
        ui.draw(screen, font)
        
        # 绘制说明文本
        instructions = [
            "左键: 添加/选择障碍物",
            "Shift+左键: 删除障碍物",
            "拖动: 移动障碍物",
            "R/T: 旋转障碍物",
            "+/-: 调整大小"
        ]
        y = 10
        for instruction in instructions:
            text = font.render(instruction, True, (0, 0, 0))
            screen.blit(text, (10, y))
            y += 25
        
        pygame.display.flip
