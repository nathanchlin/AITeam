## 2. 游戏引擎初始化 (game.js)

现在，让我们创建游戏引擎的核心代码：

```javascript
// 游戏常量
const GRAVITY = 0.5;
const FRICTION = 0.99;
const BOUNCE_DAMPING = 0.7;

// 游戏状态
const GameState = {
    MENU: 'menu',
    PLAYING: 'playing',
    AIMING: 'aiming',
    LAUNCHING: 'launching',
    LEVEL_COMPLETE: 'level_complete',
    GAME_OVER: 'game_over'
};

// 游戏主类
class AngryBirdsGame {
    constructor() {
        this.canvas = document.getElementById('gameCanvas');
        this.ctx = this.canvas.getContext('2d');
        this.score = 0;
        this.birdsLeft = 3;
        this.currentState = GameState.MENU;
        this.entities = [];
        this.pigs = [];
        this.obstacles = [];
        this.bird = null;
        this.slingshot = null;
        this.aimLine = null;
        this.particles = [];
        
        this.setupCanvas();
        this.init();
        this.setupEventListeners();
        this.gameLoop();
    }
    
    setupCanvas() {
        // 设置Canvas尺寸
        this.canvas.width = 1024;
        this.canvas.height = 768;
        
        // 高DPI支持
        const dpr = window.devicePixelRatio || 1;
        this.canvas.width *= dpr;
        this.canvas.height *= dpr;
        this.ctx.scale(dpr, dpr);
        this.canvas.style.width = 1024 + 'px';
        this.canvas.style.height = 768 + 'px';
    }
    
    init() {
        // 初始化弹弓
        this.slingshot = new Slingshot(150, 500);
        
        // 初始化小鸟
        this.bird = new Bird(150, 450);
        
        // 初始化障碍物和猪
        this.createLevel();
        
        // 初始化UI
        this.updateUI();
    }
    
    createLevel() {
        // 创建地面
        this.ground = new Ground(0, 650, 1024, 118);
        this.entities.push(this.ground);
        
        // 创建障碍物
        this.obstacles.push(new Block(600, 600, 30, 100, 'wood'));
        this.obstacles.push(new Block(700, 600, 30, 100, 'wood'));
        this.obstacles.push(new Block(650, 550, 100, 30, 'wood'));
        
        // 创建猪
        this.pigs.push(new Pig(650, 500));
        
        // 将所有障碍物和猪添加到实体数组
        this.entities.push(...this.obstacles);
        this.entities.push(...this.pigs);
    }
    
    setupEventListeners() {
        // 鼠标事件
        this.canvas.addEventListener('mousedown', (e) => this.handleMouseDown(e));
        this.canvas.addEventListener('mousemove', (e) => this.handleMouseMove(e));
        this.canvas.addEventListener('mouseup', (e) => this.handleMouseUp(e));
        
        // 触摸事件（移动设备支持）
        this.canvas.addEventListener('touchstart', (e) => this.handleTouchStart(e));
        this.canvas.addEventListener('touchmove', (e) => this.handleTouchMove(e));
        this.canvas.addEventListener('touchend', (e) => this.handleTouchEnd(e));
    }
    
    handleMouseDown(e) {
        const rect = this.canvas.getBoundingClientRect();
        const x = e.clientX - rect.left;
        const y = e.clientY - rect.top;
        
        if (this.currentState === GameState.AIMING && this.bird) {
            const dx = x - this.bird.x;
            const dy = y - this.bird.y;
            const distance = Math.sqrt(dx * dx + dy * dy);
            
            if (distance < 30) {
                this.currentState = GameState.LAUNCHING;
            }
        }
    }
    
    handleMouseMove(e) {
        const rect = this.canvas.getBoundingClientRect();
        const x = e.clientX - rect.left;
        const y = e.clientY - rect.top;
        
        if (this.currentState === GameState.LAUNCHING && this.bird) {
            // 限制弹弓拉伸距离
            const dx = Math.max(50, Math.min(150, x - this.slingshot.x));
            const dy = Math.max(350, Math.min(500, y - this.slingshot.y));
            
            this.bird.x = this.slingshot.x - dx;
            this.bird.y = this.slingshot.y - dy;
            
            // 更新瞄准线
            this.aimLine = {
                startX: this.slingshot.x,
                startY: this.slingshot.y,
                endX: this.bird.x,
                endY: this.bird.y
            };
        }
    }
    
    handleMouseUp(e) {
        if (this.currentState === GameState.LAUNCHING && this.bird) {
            // 计算发射速度
            const dx = this.slingshot.x - this.bird.x;
            const dy = this.slingshot.y - this.bird.y;
            const power = Math.min(15, Math.sqrt(dx * dx + dy * dy) / 10);
            
            this.bird.vx = dx * power * 0.2;
            this.bird.vy = dy * power * 0.2;
            this.bird.isLaunched = true;
            
            this.currentState = GameState.PLAYING;
            this.aimLine = null;
            
            // 创建发射特效
            this.createLaunchParticles(this.slingshot.x, this.slingshot.y);
        }
    }
    
    handleTouchStart(e) {
        e.preventDefault();
        const touch = e.touches[0];
        const mouseEvent = new MouseEvent('mousedown', {
            clientX: touch.clientX,
            clientY: touch.clientY
        });
        this.canvas.dispatchEvent(mouseEvent);
    }
    
    handleTouchMove(e) {
        e.preventDefault();
        const touch = e.touches[0];
        const mouseEvent = new MouseEvent('mousemove', {
            clientX: touch.clientX,
            clientY: touch.clientY
        });
        this.canvas.dispatchEvent(mouseEvent);
    }
    
    handleTouchEnd(e) {
        e.preventDefault();
        const mouseEvent = new MouseEvent('mouseup', {});
        this.canvas.dispatchEvent(mouseEvent);
    }
    
    createLaunchParticles(x, y) {
        for (let i = 0; i < 10; i++) {
            this.particles.push(new Particle(x, y));
        }
    }
    
    update() {
        // 更新实体
        for (let i = 0; i < this.entities.length; i++) {
            this.entities[i].update();
        }
        
        // 更新小鸟
        if (this.bird && this.bird.isLaunched) {
            this.bird.update();
            
            // 检查边界
            if (this.bird.x > this.canvas.width || this.bird.x < 0 || 
                this.bird.y > this.canvas.height) {
                this.resetBird();
            }
        }
        
        // 更新粒子
        for (let i = this.particles.length - 1; i >= 0; i--) {
            this.particles[i].update();
            if (this.particles[i].isDead()) {
                this.particles.splice(i, 1);
            }
        }
        
        // 碰撞检测
        this.checkCollisions();
        
        // 检查游戏状态
        this.checkGameState();
    }
    
    checkCollisions() {
        if (!this.bird || !this.bird.isLaunched) return;
        
        // 小鸟与障碍物碰撞
        for (let obstacle of this.obstacles) {
            if (this.bird.collidesWith(obstacle)) {
                this.handleCollision(this.bird, obstacle);
            }
        }
        
        // 小鸟与猪碰撞
        for (let i = this.pigs.length - 1; i >= 0; i--) {
            if (this.bird.collidesWith(this.pigs[i])) {
                this.handleCollision(this.bird, this.pigs[i]);
                this.pigs.splice(i, 1);
                this.score += 100;
                this.updateUI();
            }
        }
    }
    
    handleCollision(bird, entity) {
        // 简单的碰撞响应
        entity.health--;
        
        // 创建碰撞粒子效果
        for (let i = 0; i < 5; i++) {
            this.particles.push(new Particle(entity.x, entity.y));
        }
        
        // 如果障碍物被摧毁
        if (entity.health <= 0 && this.obstacles.includes(entity)) {
            const index = this.obstacles.indexOf(entity);
            if (index > -1) {
                this.obstacles.splice(index, 1);
            }
        }
    }
    
    checkGameState() {
        // 检查是否所有猪都被消灭
        if (this.pigs.length === 0) {
            this.currentState = GameState.LEVEL_COMPLETE;
            this.showMessage("关卡完成！");
            setTimeout(() => this.nextLevel(), 2000);
        }
        
        // 检查是否还有小鸟
        if (this.birdsLeft <= 0 && (!this.bird || this.bird.isLaunched)) {
            this.currentState = GameState.GAME_OVER;
            this.showMessage("游戏结束！");
        }
    }
    
    resetBird() {
        if (this.birdsLeft > 0) {
            this.birdsLeft--;
            this.updateUI();
            
            if (this.birdsLeft > 0) {
                this.bird = new Bird(150, 450);
                this.currentState = GameState.AIMING;
            } else {
                this.bird = null;
            }
        }
    }
    
    nextLevel() {
        // 重置关卡
        this.obstacles = [];
        this.pigs = [];
        this.createLevel();
        
        this.bird = new Bird(150, 450);
        this.birdsLeft = 3;
        this.currentState = GameState.AIMING;
        
        this.updateUI();
        this.hideMessage();
    }
    
    render() {
        // 清空画布
        this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);
        
        // 绘制背景
        this.drawBackground();
        
        // 绘制实体
        for (let entity of this.entities) {
            entity.render(this.ctx);
        }
        
        // 绘制弹弓
        if (this.slingshot) {
            this.slingshot.render(this.ctx);
        }
        
        // 绘制小鸟
        if (this.bird) {
            this.bird.render(this.ctx);
        }
        
        // 绘制瞄准线
        if (this.aimLine) {
            this.drawAimLine();
        }
        
        // 绘制粒子
        for (let particle of this.particles) {
            particle.render(this.ctx);
        }
    }
    
    drawBackground() {
        // 绘制天空渐变
        const gradient = this.ctx.createLinearGradient(0, 0, 0, this.canvas.height);
        gradient.addColorStop(0, '#87CEEB');
        gradient.addColorStop(1, '#70c5ce');
        this.ctx.fillStyle = gradient;
        this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);
        
        // 绘制云朵
        this.drawCloud(200, 100, 60);
        this.drawCloud(500, 80, 80);
        this.drawCloud(800, 120, 70);
    }
    
    drawCloud(x, y, size) {
        this.ctx.fillStyle = 'rgba(255, 255, 255, 0.8)';
        this.ctx.beginPath();
        this.ctx.arc(x, y, size * 0.6, 0, Math.PI * 2);
        this.ctx.arc(x + size * 0.4, y - size * 0.2, size * 0.5, 0, Math.PI * 2);
        this.ctx.arc(x + size * 0.8, y, size * 0.6, 0, Math.PI * 2);
        this.ctx.fill();
    }
    
    drawAimLine() {
        this.ctx.strokeStyle = 'rgba(0, 0, 0, 0.3)';
        this.ctx.lineWidth = 2;
        this.ctx.setLineDash([5, 5]);
        this.ctx.beginPath();
        this.ctx.moveTo(this.aimLine.startX, this.aimLine.startY);
        this.ctx.lineTo(this.aimLine.endX, this.aimLine.endY);
        this.ctx.stroke();
        this.ctx.setLineDash([]);
    }
    
    updateUI() {
        document.getElementById('score').textContent = `得分: ${this.score}`;
        document.getElementById('birds').textContent = `剩余小鸟: ${this.birdsLeft}`;
    }
    
    showMessage(text) {
        const messageEl = document.getElementById('message');
        messageEl.textContent = text;
        messageEl.style.display = 'block';
    }
    
    hideMessage() {
        document.getElementById('message').style.display = 'none';
    }
    
    gameLoop() {
        this.update();
        this.render();
        requestAnimationFrame(() => this.gameLoop());
    }
}

// 游戏实体基类
class Entity {
    constructor(x, y) {
        this.x = x;
        this.y = y;
        this.vx = 0;
        this.vy = 0;
        this.width = 50;
        this.height = 50;
        this.health = 1;
    }
    
    update() {
        // 基础更新逻辑，子类可以重写
    }
    
    render(ctx) {
        // 基础渲染逻辑，子类需要重写
    }
    
    collidesWith(other) {
        return this.x < other.x + other.width &&
               this.x + this.width > other.x &&
               this.y < other.y + other.height &&
               this.y + this.height > other.y;
    }
}

// 小鸟类
class Bird extends Entity {
    constructor(x, y) {
        super(x, y);
        this.radius = 20;
        this.isLaunched = false;
        this.color = '#FF5733';
        this.trail = [];
    }
    
    update() {
        if (this.isLaunched) {
            // 应用重力
            this.vy += GRAVITY;
            
            // 应用速度
            this.x += this.vx;
            this.y += this.vy;
            
            // 应用摩擦力
            this.vx *= FRICTION;
            
            // 添加轨迹点
            this.trail.push({ x: this.x, y: this.y });
            if (this.trail.length > 20) {
                this.trail.shift();
            }
            
            // 地面碰撞
            if (this.y + this.radius > 650) {
                this.y = 650 - this.radius;
                this.vy *= -BOUNCE_DAMPING;
                this.vx *= FRICTION;
                
                // 如果速度很小，停止运动
                if (Math.abs(this.vx) < 0.1 && Math.abs(this.vy) < 0.1) {
                    this.vx = 0;
                    this.vy = 0;
                }
            }
        }
    }
    
    render(ctx) {
        // 绘制轨迹
        ctx.strokeStyle = 'rgba(255, 87, 51, 0.3)';
        ctx.lineWidth = 3;
        ctx.beginPath();
        for (let i = 0; i < this.trail.length; i++) {
            if (i === 0) {
                ctx.moveTo(this.trail[i].x, this.trail[i].y);
            } else {
                ctx.lineTo(this.trail[i].x, this.trail[i].y);
            }
        }
        ctx.stroke();
        
        // 绘制小鸟
        ctx.fillStyle = this.color;
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
        ctx.fill();
        
        // 绘制眼睛
        ctx.fillStyle = 'white';
        ctx.beginPath();
        ctx.arc(this.x + 8, this.y - 5, 6, 0, Math.PI * 2);
        ctx.fill();
        
        ctx.fillStyle = 'black';
        ctx.beginPath();
        ctx.arc(this.x + 10, this.y - 5, 3, 0, Math.PI * 2);
        ctx.fill();
        
        // 绘制喙
        ctx.fillStyle = '#FFA500';
        ctx.beginPath();
        ctx.moveTo(this.x + this.radius, this.y);
        ctx.lineTo(this.x + this.radius + 10, this.y + 5);
        ctx.lineTo(this.x + this.radius, this.y + 10);
        ctx.closePath();
