# 弹弓机制开发方案

我将为您实现一个完整的弹弓机制，包括拖拽控制、力度指示器、发射机制和鸟类飞行轨迹。这个实现将基于Unity引擎，使用C#脚本。

## 1. 弹弓基础结构

首先，我们需要创建弹弓的基础结构：

```csharp
using UnityEngine;

public class Slingshot : MonoBehaviour
{
    [Header("Slingshot Settings")]
    public Transform leftArm; // 弹弓左臂
    public Transform rightArm; // 弹弓右臂
    public Transform restPosition; // 鸟类静止位置
    public float maxStretch = 2f; // 最大拉伸距离
    public float launchPower = 15f; // 发射力度系数
    
    private Transform bird; // 当前装载的鸟类
    private Vector2 startPos; // 拖拽起始位置
    private bool isDragging = false;
    private LineRenderer leftLine; // 左侧弹力线
    private LineRenderer rightLine; // 右侧弹力线
    
    void Start()
    {
        // 初始化弹力线
        SetupLines();
    }
    
    void SetupLines()
    {
        // 创建左侧弹力线
        leftLine = gameObject.AddComponent<LineRenderer>();
        leftLine.positionCount = 2;
        leftLine.startWidth = 0.05f;
        leftLine.endWidth = 0.05f;
        leftLine.material = new Material(Shader.Find("Unlit/Color"));
        leftLine.material.color = Color.black;
        
        // 创建右侧弹力线
        rightLine = gameObject.AddComponent<LineRenderer>();
        rightLine.positionCount = 2;
        rightLine.startWidth = 0.05f;
        rightLine.endWidth = 0.05f;
        rightLine.material = new Material(Shader.Find("Unlit/Color"));
        rightLine.material.color = Color.black;
    }
}
```

## 2. 拖拽控制机制

接下来实现拖拽控制：

```csharp
public class Slingshot : MonoBehaviour
{
    // ... 前面的代码
    
    void Update()
    {
        if (isDragging && bird != null)
        {
            Vector2 mousePos = Camera.main.ScreenToWorldPoint(Input.mousePosition);
            Vector2 direction = mousePos - (Vector2)restPosition.position;
            
            // 限制拖拽距离
            if (direction.magnitude > maxStretch)
            {
                direction = direction.normalized * maxStretch;
            }
            
            // 更新鸟类位置
            bird.position = restPosition.position + (Vector3)direction;
            
            // 更新弹力线
            UpdateLines();
            
            // 更新力度指示器
            UpdatePowerIndicator(direction.magnitude);
        }
    }
    
    void OnMouseDown()
    {
        if (bird != null && Vector2.Distance(bird.position, restPosition.position) < 0.1f)
        {
            isDragging = true;
            startPos = bird.position;
        }
    }
    
    void OnMouseUp()
    {
        if (isDragging)
        {
            Launch();
            isDragging = false;
        }
    }
    
    private void UpdateLines()
    {
        // 更新左侧弹力线
        leftLine.SetPosition(0, leftArm.position);
        leftLine.SetPosition(1, bird.position);
        
        // 更新右侧弹力线
        rightLine.SetPosition(0, rightArm.position);
        rightLine.SetPosition(1, bird.position);
    }
    
    public void LoadBird(Transform newBird)
    {
        bird = newBird;
        bird.position = restPosition.position;
        bird.GetComponent<Rigidbody2D>().isKinematic = true;
    }
}
```

```csharp
using UnityEngine.UI;

public class Slingshot : MonoBehaviour
{
    // ... 前面的代码
    
    [Header("Power Indicator")]
    public RectTransform powerBar; // 力度条UI
    public float maxPower = 1f; // 最大力度值
    
    private Image powerFill; // 力度填充图像
    
    void Start()
    {
        // ... 前面的代码
        
        // 初始化力度指示器
        if (powerBar != null)
        {
            powerFill = powerBar.GetComponent<Image>();
        }
    }
    
    private void UpdatePowerIndicator(float stretchDistance)
    {
        if (powerFill != null)
        {
            float power = Mathf.Clamp01(stretchDistance / maxStretch);
            powerFill.fillAmount = power;
        }
    }
}
```

## 4. 发射机制

实现发射机制：

```csharp
public class Slingshot : MonoBehaviour
{
    // ... 前面的代码
    
    private void Launch()
    {
        if (bird == null) return;
        
        // 计算发射方向和力度
        Vector2 launchDirection = (Vector2)restPosition.position - (Vector2)bird.position;
        float launchForce = launchDirection.magnitude * launchPower;
        
        // 归一化方向
        launchDirection.Normalize();
        
        // 应用力度
        Rigidbody2D rb = bird.GetComponent<Rigidbody2D>();
        rb.isKinematic = false;
        rb.velocity = launchDirection * launchForce;
        
        // 重置弹力线
        leftLine.SetPosition(1, leftArm.position);
        rightLine.SetPosition(1, rightArm.position);
        
        // 清空鸟类引用
        bird = null;
        
        // 重置力度指示器
        if (powerFill != null)
        {
            powerFill.fillAmount = 0;
        }
    }
}
```

## 5. 鸟类飞行轨迹

实现鸟类飞行轨迹和物理效果：

```csharp
using UnityEngine;

public class Bird : MonoBehaviour
{
    [Header("Bird Settings")]
    public float gravity = -9.8f; // 重力
    public float drag = 0.1f; // 空气阻力
    public TrailRenderer trail; // 飞行轨迹
    
    private Rigidbody2D rb;
    private bool hasLaunched = false;
    private Vector2 lastPosition;
    
    void Start()
    {
        rb = GetComponent<Rigidbody2D>();
        lastPosition = transform.position;
        
        if (trail != null)
        {
            trail.emitting = false;
        }
    }
    
    void Update()
    {
        if (hasLaunched)
        {
            // 更新轨迹
            if (trail != null)
            {
                trail.emitting = true;
            }
            
            // 应用空气阻力
            Vector2 velocity = rb.velocity;
            velocity *= (1 - drag * Time.deltaTime);
            rb.velocity = velocity;
            
            // 记录位置用于轨迹
            lastPosition = transform.position;
        }
    }
    
    public void Launch()
    {
        hasLaunched = true;
    }
    
    void OnCollisionEnter2D(Collision2D collision)
    {
        if (hasLaunched)
        {
            // 碰撞后停止轨迹
            if (trail != null)
            {
                trail.emitting = false;
            }
        }
    }
}
```
