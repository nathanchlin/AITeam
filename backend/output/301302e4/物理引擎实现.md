            checkCollision(other) {
                if (this.destroyed || other.destroyed) return false;
                
                const dx = this.x - other.x;
                const dy = this.y - other.y;
                const distance = Math.sqrt(dx * dx + dy * dy);
                const minDistance = (this.width + other.width) / 2;
                
                return distance < minDistance;
            }

            resolveCollision(other) {
                if (this.destroyed || other.destroyed) return;
                
                // 计算碰撞法线
                const dx = other.x - this.x;
                const dy = other.y - this.y;
                const distance = Math.sqrt(dx * dx + dy * dy);
                
                if (distance === 0) return; // 避免除以零
                
                // 归一化碰撞向量
                const nx = dx / distance;
                const ny = dy / distance;
                
                // 相对速度
                const dvx = other.vx - this.vx;
                const dvy = other.vy - this.vy;
                
                // 相对速度在碰撞法线上的投影
                const velocityAlongNormal = dvx * nx + dvy * ny;
                
                // 如果物体正在分离，不处理碰撞
                if (velocityAlongNormal > 0) return;
                
                // 计算冲量
                const restitution = 0.8; // 弹性系数
                const impulse = -(1 + restitution) * velocityAlongNormal;
                const totalMass = this.getMass() + other.getMass();
                
                // 应用冲量
                const impulseX = impulse * nx;
                const impulseY = impulse * ny;
                
                this.vx -= impulseX * (other.getMass() / totalMass);
                this.vy -= impulseY * (other.getMass() / totalMass);
                other.vx += impulseX * (this.getMass() / totalMass);
                other.vy += impulseY * (this.getMass() / totalMass);
                
                // 分离物体，避免重叠
                const overlap = (this.width + other.width) / 2 - distance;
                const separationX = nx * overlap * 0.5;
                const separationY = ny * overlap * 0.5;
                
                this.x -= separationX;
                this.y -= separationY;
                other.x += separationX;
                other.y += separationY;
                
                // 添加旋转效果
                if (this.type === 'bird') {
                    this.rotationSpeed = (Math.random() - 0.5) * 0.2;
                }
                
                if (other.type === 'bird') {
                    other.rotationSpeed = (Math.random() - 0.5) * 0.2;
                }
            }

            getMass() {
                switch(this.type) {
                    case 'bird': return 1;
                    case 'pig': return 1.5;
                    case 'wood': return 2;
                    case 'stone': return 3;
                    case 'ice': return 1.2;
                    default: return 1;
                }
            }

            applyExplosion(x, y, force) {
                const dx = this.x - x;
                const dy = this.y - y;
                const distance = Math.sqrt(dx * dx + dy * dy);
                
                if (distance === 0) return;
                
                // 爆炸力随距离衰减
                const strength = force / (distance * 0.1 + 1);
                
                this.vx += (dx / distance) * strength;
                this.vy += (dy / distance) * strength;
                
                // 添加旋转
                if (this.type === 'bird') {
                    this.rotationSpeed = (Math.random() - 0.5) * 0.3;
                }
            }
        }
