## 1. 得分系统设计

### 基础得分计算
```csharp
public class ScoreSystem
{
    private int currentScore = 0;
    private int comboMultiplier = 1;
    private float comboTimer = 0f;
    private float comboWindow = 3f; // 连击时间窗口
    
    public void AddScore(int baseScore, ScoreType scoreType)
    {
        // 根据得分类型计算实际得分
        int actualScore = CalculateActualScore(baseScore, scoreType);
        
        // 应用连击加成
        actualScore *= comboMultiplier;
        
        currentScore += actualScore;
        OnScoreAdded?.Invoke(actualScore);
    }
    
    private int CalculateActualScore(int baseScore, ScoreType scoreType)
    {
        switch (scoreType)
        {
            case ScoreType.DirectHit:
                return baseScore * 2;
            case ScoreType.StructureDamage:
                return baseScore;
            case ScoreType.PigElimination:
                return baseScore * 5;
            case ScoreType.CollateralDamage:
                return baseScore * 3;
            default:
                return baseScore;
        }
    }
    
    public void RegisterHit()
    {
        comboTimer = comboWindow;
        comboMultiplier = Mathf.Min(comboMultiplier + 1, 5); // 最大5倍连击
    }
    
    public void Update(float deltaTime)
    {
        if (comboTimer > 0)
        {
            comboTimer -= deltaTime;
            if (comboTimer <= 0)
            {
                comboMultiplier = 1;
            }
        }
    }
    
    public int GetCurrentScore() => currentScore;
    
    public event Action<int> OnScoreAdded;
}

public enum ScoreType
{
    DirectHit,      // 直接命中
    StructureDamage, // 建筑损坏
    PigElimination, // 消灭猪
    CollateralDamage // 连锁破坏
}
```

## 2. 星级评定系统

```csharp
public class StarRatingSystem
{
    private int starsAchieved = 0;
    private int maxStars = 3;
    
    public int CalculateStars(int score, int pigsRemaining, int birdsUsed)
    {
        // 基础星级计算逻辑
        int scoreStars = CalculateScoreStars(score);
        int efficiencyStars = CalculateEfficiencyStars(pigsRemaining, birdsUsed);
        
        starsAchieved = Mathf.Min(scoreStars + efficiencyStars, maxStars);
        return starsAchieved;
    }
    
    private int CalculateScoreStars(int score)
    {
        // 根据得分计算星级
        if (score >= 50000) return 3;
        if (score >= 30000) return 2;
        if (score >= 10000) return 1;
        return 0;
    }
    
    private int CalculateEfficiencyStars(int pigsRemaining, int birdsUsed)
    {
        // 根据效率和剩余资源计算星级
        if (pigsRemaining == 0 && birdsUsed <= 2) return 3;
        if (pigsRemaining == 0 && birdsUsed <= 3) return 2;
        if (pigsRemaining <= 1) return 1;
        return 0;
    }
    
    public int GetStarsAchieved() => starsAchieved;
}
```

## 3. 成就系统

```csharp
public class AchievementSystem
{
    private List<Achievement> achievements;
    private List<Achievement> unlockedAchievements;
    private GameProgress progress;
    
    public AchievementSystem(GameProgress progress)
    {
        this.progress = progress;
        achievements = InitializeAchievements();
        unlockedAchievements = progress.UnlockedAchievements;
    }
    
    private List<Achievement> InitializeAchievements()
    {
        return new List<Achievement>
        {
            new Achievement("First_Blood", "初次击杀", "消灭第一只猪", () => progress.TotalPigsKilled >= 1),
            new Achievement("Combo_Master", "连击大师", "达到5倍连击", () => progress.MaxComboMultiplier >= 5),
            new Achievement("Perfect_Level", "完美通关", "一关获得三星评价", () => progress.PerfectLevels >= 1),
            new Achievement("Sharpshooter", "神射手", "连续10次直接命中", () => progress.ConsecutiveDirectHits >= 10),
            new Achievement("Demolitionist", "爆破专家", "单关得分超过50000", () => progress.HighScore >= 50000),
            new Achievement("Eagle", "雄鹰", "用一只小鸟消灭所有猪", () => progress.OneBirdClears >= 1),
            new Achievement("Veteran", "老鸟", "完成50个关卡", () => progress.LevelsCompleted >= 50),
            new Achievement("Collector", "收藏家", "解锁所有成就", () => false) // 特殊成就
        };
    }
    
    public void CheckAchievements(GameStats stats)
    {
        foreach (var achievement in achievements)
        {
            if (!unlockedAchievements.Contains(achievement) && achievement.IsUnlocked(stats))
            {
                UnlockAchievement(achievement);
            }
        }
        
        // 检查收藏家成就
        if (unlockedAchievements.Count == achievements.Count - 1)
        {
            var collectorAchievement = achievements.Find(a => a.Id == "Collector");
            if (collectorAchievement != null)
            {
                UnlockAchievement(collectorAchievement);
            }
        }
    }
    
    private void UnlockAchievement(Achievement achievement)
    {
        unlockedAchievements.Add(achievement);
        progress.UnlockedAchievements = unlockedAchievements;
        
        // 保存进度
        SaveSystem.SaveGameProgress(progress);
        
        // 触发解锁事件
        OnAchievementUnlocked?.Invoke(achievement);
    }
    
    public List<Achievement> GetAllAchievements() => achievements;
    public List<Achievement> GetUnlockedAchievements() => unlockedAchievements;
    
    public event Action<Achievement> OnAchievementUnlocked;
}

public class Achievement
{
    public string Id { get; }
    public string Title { get; }
    public string Description { get; }
    public Func<bool> IsUnlockedCondition { get; }
    
    public Achievement(string id, string title, string description, Func<bool> isUnlockedCondition)
    {
        Id = id;
        Title = title;
        Description = description;
        IsUnlockedCondition = isUnlockedCondition;
    }
    
    public bool IsUnlocked(GameStats stats)
    {
        return IsUnlockedCondition();
    }
}
```

```csharp
[System.Serializable]
public class GameProgress
{
    public int CurrentLevel = 1;
    public int TotalScore = 0;
    public int HighScore = 0;
    public int LevelsCompleted = 0;
    public int TotalPigsKilled = 0;
    public int MaxComboMultiplier = 1;
    public int PerfectLevels = 0;
    public int ConsecutiveDirectHits = 0;
    public int OneBirdClears = 0;
    public List<string> UnlockedAchievements = new List<string>();
    public Dictionary<int, LevelProgress> LevelProgressData = new Dictionary<int, LevelProgress>();
}

[System.Serializable]
public class LevelProgress
{
    public int BestScore = 0;
    public int StarsAchieved = 0;
    public bool IsCompleted = false;
}

public class SaveSystem
{
    private const string SaveFileName = "AngryBirds_Save.json";
    
    public static void SaveGameProgress(GameProgress progress)
    {
        string jsonData = JsonUtility.ToJson(progress, true);
        string savePath = Path.Combine(Application.persistentDataPath, SaveFileName);
        File.WriteAllText(savePath, jsonData);
    }
    
    public static GameProgress LoadGameProgress()
    {
        string savePath = Path.Combine(Application.persistentDataPath, SaveFileName);
        
        if (File.Exists(savePath))
        {
            string jsonData = File.ReadAllText(savePath);
            return JsonUtility.FromJson<GameProgress>(jsonData);
        }
        
        return new GameProgress(); // 返回新进度
    }
    
    public static void ResetProgress()
    {
        string savePath = Path.Combine(Application.persistentDataPath, SaveFileName);
        if (File.Exists(savePath))
        {
            File.Delete(savePath);
        }
    }
}
```

## 5. 游戏统计与事件管理

```csharp
public class GameStats
{
    public int Score;
    public int PigsRemaining;
    public int BirdsUsed;
    public int DirectHits;
    public int ComboMultiplier;
    public int Level;
    public bool IsPerfectClear;
}

public class GameEventManager
{
    private ScoreSystem scoreSystem;
    private AchievementSystem achievementSystem;
    private GameProgress progress;
    
    public GameEventManager(ScoreSystem scoreSystem, AchievementSystem achievementSystem, GameProgress progress)
    {
        this.scoreSystem = scoreSystem;
        this.achievementSystem = achievementSystem;
        this.progress = progress;
    }
    
    public void OnPigEliminated()
    {
        progress.TotalPigsKilled++;
        scoreSystem.RegisterHit();
        
        var stats = GetCurrentGameStats();
        achievementSystem.CheckAchievements(stats);
    }
    
    public void OnLevelComplete(int score, int stars, bool perfectClear)
    {
        progress.LevelsCompleted++;
        progress.TotalScore += score;
        
        if (score > progress.HighScore)
        {
            progress.HighScore = score;
        }
        
        if (stars == 3)
        {
            progress.PerfectLevels++;
        }
        
        if (perfectClear)
        {
            progress.OneBirdClears++;
        }
        
        // 保存关卡进度
        if (!progress.LevelProgressData.ContainsKey(progress.CurrentLevel))
        {
            progress.LevelProgressData[progress.CurrentLevel] = new LevelProgress();
        }
        
        var levelProgress = progress.LevelProgressData[progress.CurrentLevel];
        if (score > levelProgress.BestScore)
        {
            levelProgress.BestScore = score;
        }
        
        levelProgress.StarsAchieved = Mathf.Max(levelProgress.StarsAchieved, stars);
        levelProgress.IsCompleted = true;
        
        SaveSystem.SaveGameProgress(progress);
    }
    
    public void OnDirectHit()
    {
        progress.ConsecutiveDirectHits++;
    }
    
    public void OnMissedShot()
    {
        progress.ConsecutiveDirectHits = 0;
    }
    
    private GameStats GetCurrentGameStats()
    {
        return new GameStats
        {
            Score = scoreSystem.GetCurrentScore(),
            PigsRemaining = 0, // 需要根据游戏实际情况更新
            BirdsUsed = 0,     // 需要根据游戏实际情况更新
            DirectHits = progress.ConsecutiveDirectHits,
            ComboMultiplier = scoreSystem.GetComboMultiplier(),
            Level = progress.CurrentLevel,
            IsPerfectClear = false // 需要根据游戏实际情况更新
        };
    }
}
```

## 6. UI集成示例

```csharp
public class GameUI : MonoBehaviour
{
    [SerializeField] private Text scoreText;
    [SerializeField] private Text comboText;
    [SerializeField] private Image[] starImages;
    [SerializeField] private GameObject achievementPopup;
    [SerializeField] private Text achievementTitle;
    [SerializeField] private Text achievementDescription;
    
    private ScoreSystem scoreSystem;
    private StarRatingSystem starSystem;
    private AchievementSystem achievementSystem;
    
    private void Start()
    {
        scoreSystem = FindObjectOfType<ScoreSystem>();
        starSystem = FindObjectOfType<StarRatingSystem>();
        achievementSystem = FindObjectOfType<AchievementSystem>();
        
        scoreSystem.OnScoreAdded += UpdateScoreDisplay;
        achievementSystem.OnAchievementUnlocked += ShowAchievementPopup;
    }
    
    private void UpdateScoreDisplay(int score)
    {
        scoreText.text = $"Score: {score}";
        
        // 更新连击显示
        if (scoreSystem.GetComboMultiplier() > 1)
        {
            comboText.text = $"x{scoreSystem.GetComboMultiplier()} COMBO!";
            comboText.gameObject.SetActive(true);
        }
        else
        {
            comboText.gameObject.SetActive(false);
        }
    }
    
    public void ShowStars(int stars)
    {
        for (int i = 0; i < starImages.Length; i++)
        {
            starImages[i].gameObject.SetActive(i < stars);
        }
    }
    
    private void ShowAchievementPopup(Achievement achievement)
    {
        achievementTitle.text = achievement.Title;
        achievementDescription.text = achievement.Description;
        achievementPopup.SetActive(true);
        
        // 3秒后自动隐藏
        Invoke("HideAchievementPopup", 3f);
    }
    
    private void HideAchievementPopup()
    {
        achievementPopup.SetActive(false);
    }
}
```

## 实现说明

1. **得分系统**：
   - 支持多种得分类型（直接命中、建筑损坏、消灭猪、连锁破坏）
   - 实现连击机制，连续命中可获得分数加成
   - 连击倍数最高可达5倍

2. **星级评定**：
   - 根据得分和效率（剩余小鸟和剩余猪）计算星级
   - 每关最高可获得3星

3. **成就系统**：
   - 包含多种成就类型（击杀、连击、完美通关等）
   - 成就解锁时触发事件和UI反馈
   - 支持特殊成就（如"收藏家"需要解锁所有其他成就）

4. **进度保存**：
   - 使用JSON序列化保存游戏进度
   - 保存内容包括关卡进度、成就、最高分等
   - 支持重置进度功能

5. **事件管理**：
   - 集中管理游戏事件，确保各系统间通信
   - 在适当的时候触发成就检查
