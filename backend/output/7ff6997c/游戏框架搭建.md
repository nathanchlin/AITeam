# 游戏框架搭建：忍者跑酷游戏基础环境

我将为您创建一个基于HTML5 Canvas的游戏框架，为"忍者必须死的跑酷游戏"提供基础环境。

## 完整代码实现

```html
<!DOCTYPE html>
<html lang="zh-CN">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>忍者必须死 - 跑酷游戏</title>
    <style>
        body {
            margin: 0;
            padding: 0;
            display: flex;
            justify-content: center;
            align-items: center;
            height: 100vh;
            background-color: #111;
            overflow: hidden;
            font-family: 'Arial', sans-serif;
        }
        
        #gameContainer {
            position: relative;
            box-shadow: 0 0 20px rgba(0, 255, 0, 0.3);
        }
        
        #gameCanvas {
            background-color: #000;
            border: 2px solid #0f0;
        }
        
        #gameInfo {
            position: absolute;
            top: 10px;
            left: 10px;
            color: #0f0;
            font-size: 18px;
            text-shadow: 0 0 5px rgba(0, 255, 0, 0.5);
        }
        
        #startScreen {
            position: absolute;
            top: 0;
            left: 0;
            width: 100%;
            height: 100%;
            background-color: rgba(0, 0, 0, 0.8);
            display: flex;
            flex-direction: column;
            justify-content: center;
            align-items: center;
            color: #0f0;
        }
        
        #startScreen h1 {
            font-size: 48px;
            margin-bottom: 20px;
            text-shadow: 0 0 10px rgba(0, 255, 0, 0.8);
        }
        
        #startScreen button {
            padding: 15px 30px;
            font-size: 20px;
            background-color: #0f0;
            color: #000;
            border: none;
            cursor: pointer;
            border-radius: 5px;
            transition: all 0.3s;
        }
        
        #startScreen button:hover {
            background-color: #0a0;
            box-shadow: 0 0 15px rgba(0, 255, 0, 0.8);
        }
    </style>
</head>
<body>
    <div id="gameContainer">
        <canvas id="gameCanvas"></canvas>
        <div id="gameInfo">分数: <span id="score">0</span></div>
        <div id="startScreen">
            <h1>忍者必须死</h1>
            <button id="startButton">开始游戏</button>
        </div>
    </div>

    <script>
        // 游戏引擎和基础渲染系统
        class GameEngine {
            constructor(canvasId) {
                this.canvas = document.getElementById(canvasId);
                this.ctx = this.canvas.getContext('2d');
                this.width = 800;
                this.height = 400;
                this.gameState = 'start'; // start, playing, gameover
                this.score = 0;
                this.lastTime = 0;
                this.deltaTime = 0;
                
                // 设置画布尺寸
                this.canvas.width = this.width;
                this.canvas.height = this.height;
                
                // 游戏对象集合
                this.gameObjects = [];
                this.ninja = null;
                
                // 初始化游戏
                this.init();
            }
            
            init() {
                // 创建忍者角色
                this.ninja = new Ninja(100, this.height - 100);
                this.gameObjects.push(this.ninja);
                
                // 添加事件监听
                this.setupEventListeners();
                
                // 开始游戏循环
                this.gameLoop(0);
            }
            
            setupEventListeners() {
                // 开始按钮
                document.getElementById('startButton').addEventListener('click', () => {
                    this.startGame();
                });
                
                // 键盘控制
                document.addEventListener('keydown', (e) => {
                    if (this.gameState !== 'playing') return;
                    
                    switch(e.key) {
                        case ' ':
                        case 'ArrowUp':
                            this.ninja.jump();
                            break;
                        case 'ArrowDown':
                            this.ninja.slide();
                            break;
                    }
                });
                
                document.addEventListener('keyup', (e) => {
                    if (this.gameState !== 'playing') return;
                    
                    if (e.key === 'ArrowDown') {
                        this.ninja.stopSliding();
                    }
                });
            }
            
            startGame() {
                this.gameState = 'playing';
                document.getElementById('startScreen').style.display = 'none';
                this.score = 0;
                this.updateScore();
            }
            
            gameLoop(currentTime) {
                // 计算时间差
                this.deltaTime = currentTime - this.lastTime;
                this.lastTime = currentTime;
                
                // 清空画布
                this.ctx.clearRect(0, 0, this.width, this.height);
                
                // 更新游戏状态
                if (this.gameState === 'playing') {
                    this.update();
                }
                
                // 渲染游戏
                this.render();
                
                // 继续游戏循环
                requestAnimationFrame((time) => this.gameLoop(time));
            }
            
            update() {
                // 更新所有游戏对象
                for (let i = 0; i < this.gameObjects.length; i++) {
                    this.gameObjects[i].update(this.deltaTime);
                }
                
                // 生成障碍物
                if (Math.random() < 0.02) {
                    this.generateObstacle();
                }
                
                // 增加分数
                this.score += 1;
                this.updateScore();
            }
            
            render() {
                // 渲染所有游戏对象
                for (let i = 0; i < this.gameObjects.length; i++) {
                    this.gameObjects[i].render(this.ctx);
                }
            }
            
            generateObstacle() {
                const obstacle = new Obstacle(this.width, this.height - 50);
                this.gameObjects.push(obstacle);
            }
            
            updateScore() {
                document.getElementById('score').textContent = this.score;
            }
        }
        
        // 忍者类
        class Ninja {
            constructor(x, y) {
                this.x = x;
                this.y = y;
                this.width = 30;
                this.height = 50;
                this.velocityY = 0;
                this.isJumping = false;
                this.isSliding = false;
                this.color = '#0f0'; // 绿色忍者
                this.groundY = y;
                this.jumpPower = 15;
                this.gravity = 0.8;
            }
            
            jump() {
                if (!this.isJumping && !this.isSliding) {
                    this.velocityY = -this.jumpPower;
                    this.isJumping = true;
                }
            }
            
            slide() {
                if (!this.isJumping) {
                    this.isSliding = true;
                    this.height = 25; // 滑行时高度降低
                }
            }
            
            stopSliding() {
                this.isSliding = false;
                this.height = 50; // 恢复原始高度
            }
            
            update(deltaTime) {
                // 跳跃物理
                if (this.isJumping) {
                    this.y += this.velocityY;
                    this.velocityY += this.gravity;
                    
                    // 落地检测
                    if (this.y >= this.groundY) {
                        this.y = this.groundY;
                        this.velocityY = 0;
                        this.isJumping = false;
                    }
                }
            }
            
            render(ctx) {
                ctx.fillStyle = this.color;
                
                // 绘制忍者身体（矩形）
                ctx.fillRect(this.x, this.y, this.width, this.height);
                
                // 绘制忍者头部（圆形）
                ctx.beginPath();
                ctx.arc(this.x + this.width / 2, this.y - 10, 10, 0, Math.PI * 2);
                ctx.fill();
                
                // 绘制忍者眼睛
                ctx.fillStyle = '#000';
                ctx.beginPath();
                ctx.arc(this.x + this.width / 2 - 3, this.y - 10, 2, 0, Math.PI * 2);
                ctx.arc(this.x + this.width / 2 + 3, this.y - 10, 2, 0, Math.PI * 2);
                ctx.fill();
                
                // 恢复忍者颜色
                ctx.fillStyle = this.color;
            }
        }
        
        // 障碍物类
        class Obstacle {
            constructor(x, y) {
                this.x = x;
                this.y = y;
                this.width = 20;
                this.height = 40;
                this.speed = 5;
                this.color = '#f00';
            }
            
            update(deltaTime) {
                this.x -= this.speed;
                
                // 移除屏幕外的障碍物
                if (this.x + this.width < 0) {
                    const index = gameEngine.gameObjects.indexOf(this);
                    if (index > -1) {
                        gameEngine.gameObjects.splice(index, 1);
                    }
                }
            }
            
            render(ctx) {
                ctx.fillStyle = this.color;
                ctx.fillRect(this.x, this.y, this.width, this.height);
            }
        }
        
        // 初始化游戏引擎
        const gameEngine = new GameEngine('gameCanvas');
    </script>
</body>
</html>
```

## 代码结构说明

### 1. HTML结构
- 创建了一个800x400像素的游戏画布
- 添加了游戏信息显示区域（分数）
- 设计了开始界面，包含游戏标题和开始按钮

### 2. CSS样式
- 设置了深色主题，突出绿色忍者主题
- 添加了霓虹绿色边框和发光效果，增强游戏氛围
- 实现了响应式布局，使游戏居中显示

### 3. JavaScript游戏引擎

#### GameEngine类（游戏引擎核心）
- **初始化**：设置画布尺寸，创建游戏对象
- **游戏循环**：使用requestAnimationFrame实现流畅的游戏循环
- **状态管理**：控制游戏状态（开始、游戏中、结束）
- **事件处理**：管理键盘输入和按钮点击事件
- **渲染系统**：负责清空画布和渲染所有游戏对象

#### Ninja类（忍者角色）
- **属性**：位置、尺寸、颜色、速度等
- **动作**：跳跃、滑行
- **物理系统**：实现重力和跳跃机制
- **渲染**：绘制忍者身体、头部和眼睛

#### Obstacle类（障碍物）
- **属性**：位置、尺寸、速度
- **行为**：向左移动，移出屏幕后自动删除

## 功能特点
