        class Particle {
            constructor(x, y, color) {
                this.x = x;
                this.y = y;
                this.vx = (Math.random() - 0.5) * 8;
                this.vy = (Math.random() - 0.5) * 8;
                this.radius = Math.random() * 3 + 1;
                this.color = color;
                this.alpha = 1;
                this.decay = Math.random() * 0.02 + 0.01;
            }
            
            update() {
                this.x += this.vx;
                this.y += this.vy;
                this.alpha -= this.decay;
                this.radius *= 0.98;
            }
            
            draw() {
                ctx.save();
                ctx.globalAlpha = this.alpha;
                ctx.fillStyle = this.color;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
                ctx.fill();
                ctx.restore();
            }
        }
        
        class Explosion {
            constructor(x, y, size = 30, color = '#ff6b35') {
                this.x = x;
                this.y = y;
                this.particles = [];
                this.size = size;
                this.color = color;
                this.createParticles();
            }
            
            createParticles() {
                const particleCount = Math.floor(this.size / 2);
                for (let i = 0; i < particleCount; i++) {
                    this.particles.push(new Particle(this.x, this.y, this.color));
                }
                
                // 添加一些额外的火花粒子
                for (let i = 0; i < 10; i++) {
                    const spark = new Particle(this.x, this.y, '#ffdd59');
                    spark.vx = (Math.random() - 0.5) * 12;
                    spark.vy = (Math.random() - 0.5) * 12;
                    spark.radius = Math.random() * 2 + 0.5;
                    this.particles.push(spark);
                }
            }
            
            update() {
                for (let i = this.particles.length - 1; i >= 0; i--) {
                    this.particles[i].update();
                    if (this.particles[i].alpha <= 0) {
                        this.particles.splice(i, 1);
                    }
                }
            }
            
            draw() {
                this.particles.forEach(particle => particle.draw());
            }
            
            isFinished() {
                return this.particles.length === 0;
            }
        }
        
        // 爆炸声音效果
        class ExplosionSound {
            constructor() {
                this.audioContext = new (window.AudioContext || window.webkitAudioContext)();
            }
            
            play() {
                const oscillator = this.audioContext.createOscillator();
                const gainNode = this.audioContext.createGain();
                
                oscillator.connect(gainNode);
                gainNode.connect(this.audioContext.destination);
                
                // 创建爆炸声音效果
                oscillator.type = 'sawtooth';
                oscillator.frequency.setValueAtTime(100, this.audioContext.currentTime);
                oscillator.frequency.exponentialRampToValueAtTime(0.01, this.audioContext.currentTime + 0.5);
                
                gainNode.gain.setValueAtTime(0.3, this.audioContext.currentTime);
                gainNode.gain.exponentialRampToValueAtTime(0.01, this.audioContext.currentTime + 0.5);
                
                oscillator.start(this.audioContext.currentTime);
                oscillator.stop(this.audioContext.currentTime + 0.5);
            }
        }

        class GameObject {
            constructor(x, y, width, height) {

        // 坦克类
        class Tank extends GameObject {
            constructor(x, y, color, controls) {
                super(x, y, 30, 30);
                this.color = color;
                this.controls = controls;
                this.speed = 3;
                this.direction = 0; // 0: 上, 1: 右, 2: 下, 3: 左
                this.bullets = [];
                this.canShoot = true;
                this.shootCooldown = 0;
            }
            
            draw() {
                ctx.save();
                ctx.fillStyle = this.color;
                ctx.translate(this.x + this.width / 2, this.y + this.height / 2);
                ctx.rotate(this.direction * Math.PI / 2);
                
                // 绘制坦克主体
                ctx.fillRect(-15, -10, 30, 20);
                
                // 绘制坦克炮管
                ctx.fillRect(0, -3, 20, 6);
                
                // 绘制坦克履带
                ctx.fillStyle = '#2c3e50';
                ctx.fillRect(-15, -12, 30, 4);
                ctx.fillRect(-15, 8, 30, 4);
                
                ctx.restore();
                
                // 绘制子弹
                this.bullets.forEach(bullet => bullet.draw());
            }
            
            update() {
                // 更新射击冷却
                if (this.shootCooldown > 0) {
                    this.shootCooldown--;
                } else {
                    this.canShoot = true;
                }
                
                // 更新子弹
                for (let i = this.bullets.length - 1; i >= 0; i--) {
                    this.bullets[i].update();
                    if (!this.bullets[i].alive) {
                        this.bullets.splice(i, 1);
                    }
                }
            }
            
            move(direction) {
                this.direction = direction;
                let newX = this.x;
                let newY = this.y;
                
                switch (direction) {
                    case 0: // 上
                        newY -= this.speed;
                        break;
                    case 1: // 右
                        newX += this.speed;
                        break;
                    case 2: // 下
                        newY += this.speed;
                        break;
                    case 3: // 左
                        newX -= this.speed;
                        break;
                }
                
                // 边界检查
                if (newX >= 0 && newX + this.width <= canvas.width &&
                    newY >= 0 && newY + this.height <= canvas.height) {
                    
                    // 检查与障碍物的碰撞
                    let canMove = true;
                    obstacles.forEach(obstacle => {
                        if (obstacle.alive && this.checkCollision(newX, newY, obstacle)) {
                            canMove = false;
                        }
                    });
                    
                    // 检查与其他坦克的碰撞
                    tanks.forEach(tank => {
                        if (tank !== this && tank.alive && this.checkCollision(newX, newY, tank)) {
                            canMove = false;
                        }
                    });
                    
                    if (canMove) {
                        this.x = newX;
                        this.y = newY;
                    }
                }
            }
            
            checkCollision(x, y, other) {
                return x < other.x + other.width &&
                       x + this.width > other.x &&
                       y < other.y + other.height &&
                       y + this.height > other.y;
            }
            
            shoot() {
                if (this.canShoot) {
                    const bulletX = this.x + this.width / 2;
                    const bulletY = this.y + this.height / 2;
                    
                    // 根据坦克方向计算子弹初始位置
                    let offsetX = 0;
                    let offsetY = 0;
                    
                    switch (this.direction) {
                        case 0: // 上
                            offsetY = -15;
                            break;
                        case 1: // 右
                            offsetX = 15;
                            break;
                        case 2: // 下
                            offsetY = 15;
                            break;
                        case 3: // 左
                            offsetX = -15;
                            break;
                    }
                    
                    this.bullets.push(new Bullet(bulletX + offsetX, bulletY + offsetY, this.direction, this));
                    this.canShoot = false;
                    this.shootCooldown = 20;
                }
            }
        }
        
        // 子弹类
        class Bullet extends GameObject {
            constructor(x, y, direction, owner) {
                super(x, y, 5, 5);
                this.direction = direction;
                this.owner = owner;
                this.speed = 8;
                this.alive = true;
            }
            
            draw() {
                ctx.fillStyle = '#f1c40f';
                ctx.beginPath();
                ctx.arc(this.x + this.width / 2, this.y + this.height / 2, this.width / 2, 0, Math.PI * 2);
                ctx.fill();
            }
            
            update() {
                switch (this.direction) {
                    case 0: // 上
                        this.y -= this.speed;
                        break;
                    case 1: // 右
                        this.x += this.speed;
                        break;
                    case 2: // 下
                        this.y += this.speed;
                        break;
                    case 3: // 左
                        this.x -= this.speed;
                        break;
                }
                
                // 边界检查
                if (this.x < 0 || this.x + this.width > canvas.width ||
                    this.y < 0 || this.y + this.height > canvas.height) {
                    this.alive = false;
                }
                
                // 检查与障碍物的碰撞
                obstacles.forEach(obstacle => {
                    if (obstacle.alive && this.collidesWith(obstacle)) {
                        obstacle.alive = false;

                tanks.forEach(tank => {
                    if (tank !== this.owner && tank.alive && this.collidesWith(tank)) {

            }
        }

                    ctx.moveTo(this.x, this.y + i);
