我将为您创建一个完整的坦克大战游戏，包含子弹系统。以下是完整的单文件HTML实现：

```html
<!DOCTYPE html>
<html lang="zh-CN">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>坦克大战</title>
    <style>
        body {
            margin: 0;
            padding: 0;
            display: flex;
            justify-content: center;
            align-items: center;
            min-height: 100vh;
            background-color: #2c3e50;
            font-family: Arial, sans-serif;
        }
        
        #gameContainer {
            position: relative;
            box-shadow: 0 0 20px rgba(0, 0, 0, 0.5);
        }
        
        #gameCanvas {
            background-color: #34495e;
            border: 2px solid #1a252f;
        }
        
        #gameInfo {
            position: absolute;
            top: 10px;
            left: 10px;
            color: white;
            font-size: 14px;
            background-color: rgba(0, 0, 0, 0.5);
            padding: 10px;
            border-radius: 5px;
        }
        
        #gameOver {
            position: absolute;
            top: 50%;
            left: 50%;
            transform: translate(-50%, -50%);
            background-color: rgba(0, 0, 0, 0.8);
            color: white;
            padding: 20px;
            border-radius: 10px;
            text-align: center;
            display: none;
        }
    </style>
</head>
<body>
    <div id="gameContainer">
        <canvas id="gameCanvas" width="800" height="600"></canvas>
        <div id="gameInfo">
            <div>玩家1: WASD移动, 空格射击</div>
            <div>玩家2: 方向键移动, 回车射击</div>
            <div id="score">玩家1: 0 | 玩家2: 0</div>
        </div>
        <div id="gameOver">
            <h2 id="winnerText">游戏结束</h2>
            <button id="restartBtn" onclick="restartGame()">重新开始</button>
        </div>
    </div>

    <script>
        // 获取canvas和上下文
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        
        // 游戏状态
        const game = {
            width: canvas.width,
            height: canvas.height,
            keys: {},
            bullets: [],
            explosions: [],
            tanks: [],
            obstacles: [],
            gameOver: false,
            winner: null
        };
        
        // 子弹类
        class Bullet {
            constructor(x, y, angle, owner) {
                this.x = x;
                this.y = y;
                this.angle = angle;
                this.speed = 8;
                this.radius = 4;
                this.owner = owner;
                this.active = true;
                this.color = owner === 1 ? '#3498db' : '#e74c3c';
            }
            
            update() {
                // 根据角度和速度更新子弹位置
                this.x += Math.cos(this.angle) * this.speed;
                this.y += Math.sin(this.angle) * this.speed;
                
                // 检查子弹是否超出边界
                if (this.x < 0 || this.x > game.width || 
                    this.y < 0 || this.y > game.height) {
                    this.active = false;
                }
                
                // 检查子弹是否击中障碍物
                for (const obstacle of game.obstacles) {
                    if (this.checkCollision(obstacle)) {
                        this.active = false;
                        createExplosion(this.x, this.y);
                        break;
                    }
                }
                
                // 检查子弹是否击中坦克
                for (const tank of game.tanks) {
                    if (tank.id !== this.owner && this.checkCollision(tank)) {
                        this.active = false;
                        tank.health--;
                        createExplosion(this.x, this.y);
                        
                        if (tank.health <= 0) {
                            tank.active = false;
                            if (tank.id === 1) {
                                game.winner = 2;
                            } else {
                                game.winner = 1;
                            }
                            game.gameOver = true;
                            showGameOver();
                        }
                        break;
                    }
                }
            }
            
            checkCollision(target) {
                // 简单的圆形碰撞检测
                const dx = this.x - target.x;
                const dy = this.y - target.y;
                const distance = Math.sqrt(dx * dx + dy * dy);
                return distance < this.radius + target.radius;
            }
            
            draw() {
                ctx.save();
                ctx.fillStyle = this.color;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
                ctx.fill();
                
                // 绘制子弹尾迹
                ctx.strokeStyle = this.color;
                ctx.globalAlpha = 0.5;
                ctx.beginPath();
                ctx.moveTo(this.x, this.y);
                ctx.lineTo(
                    this.x - Math.cos(this.angle) * 10,
                    this.y - Math.sin(this.angle) * 10
                );
                ctx.stroke();
                ctx.restore();
            }
        }
        
        // 爆炸效果类
        class Explosion {
            constructor(x, y) {
                this.x = x;
                this.y = y;
                this.radius = 5;
                this.maxRadius = 30;
                this.active = true;
                this.color = '#f39c12';
            }
            
            update() {
                this.radius += 2;
                if (this.radius >= this.maxRadius) {
                    this.active = false;
                }
            }
            
            draw() {
                ctx.save();
                ctx.globalAlpha = 1 - (this.radius / this.maxRadius);
                ctx.strokeStyle = this.color;
                ctx.lineWidth = 3;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
                ctx.stroke();
                ctx.restore();
            }
        }
        
        // 坦克类
        class Tank {
            constructor(x, y, id) {
                this.x = x;
                this.y = y;
                this.id = id;
                this.radius = 20;
                this.angle = id === 1 ? 0 : Math.PI;
                this.speed = 3;
                this.rotationSpeed = 0.1;
                this.health = 3;
                this.active = true;
                this.color = id === 1 ? '#3498db' : '#e74c3c';
                this.lastShot = 0;
                this.shotCooldown = 500; // 毫秒
            }
            
            update() {
                if (!this.active) return;
                
                // 玩家1控制 (WASD)
                if (this.id === 1) {
                    if (game.keys['w']) {
                        this.x += Math.cos(this.angle) * this.speed;
                        this.y += Math.sin(this.angle) * this.speed;
                    }
                    if (game.keys['s']) {
                        this.x -= Math.cos(this.angle) * this.speed;
                        this.y -= Math.sin(this.angle) * this.speed;
                    }
                    if (game.keys['a']) {
                        this.angle -= this.rotationSpeed;
                    }
                    if (game.keys['d']) {
                        this.angle += this.rotationSpeed;
                    }
                    if (game.keys[' ']) {
                        this.shoot();
                    }
                }
                
                // 玩家2控制 (方向键)
                if (this.id === 2) {
                    if (game.keys['ArrowUp']) {
                        this.x += Math.cos(this.angle) * this.speed;
                        this.y += Math.sin(this.angle) * this.speed;
                    }
                    if (game.keys['ArrowDown']) {
                        this.x -= Math.cos(this.angle) * this.speed;
                        this.y -= Math.sin(this.angle) * this.speed;
                    }
                    if (game.keys['ArrowLeft']) {
                        this.angle -= this.rotationSpeed;
                    }
                    if (game.keys['ArrowRight']) {
                        this.angle += this.rotationSpeed;
                    }
                    if (game.keys['Enter']) {
                        this.shoot();
                    }
                }
                
                // 边界检查
                this.x = Math.max(this.radius, Math.min(game.width - this.radius, this.x));
                this.y = Math.max(this.radius, Math.min(game.height - this.radius, this.y));
                
                // 障碍物碰撞检查
                for (const obstacle of game.obstacles) {
                    if (this.checkCollision(obstacle)) {
                        // 简单的碰撞响应 - 将坦克推回
                        const dx = this.x - obstacle.x;
                        const dy = this.y - obstacle.y;
                        const distance = Math.sqrt(dx * dx + dy * dy);
                        const minDistance = this.radius + obstacle.radius;
                        
                        if (distance < minDistance) {
                            const angle = Math.atan2(dy, dx);
                            this.x = obstacle.x + Math.cos(angle) * minDistance;
                            this.y = obstacle.y + Math.sin(angle) * minDistance;
                        }
                    }
                }
            }
            
            checkCollision(obstacle) {
                const dx = this.x - obstacle.x;
                const dy = this.y - obstacle.y;
                const distance = Math.sqrt(dx * dx + dy * dy);
                return distance < this.radius + obstacle.radius;
            }
            
            shoot() {
                const now = Date.now();
                if (now - this.lastShot > this.shotCooldown) {
                    // 计算子弹发射位置（坦克前端）
                    const bulletX = this.x + Math.cos(this.angle) * (this.radius + 5);
                    const bulletY = this.y + Math.sin(this.angle) * (this.radius + 5);
                    
                    game.bullets.push(new Bullet(bulletX, bulletY, this.angle, this.id));
                    this.lastShot = now;
                }
            }
            
            draw() {
                if (!this.active) return;
                
                ctx.save();
                ctx.translate(this.x, this.y);
                ctx.rotate(this.angle);
                
                // 绘制坦克主体
                ctx.fillStyle = this.color;
                ctx.fillRect(-this.radius, -this.radius/2, this.radius * 2, this.radius);
                
                // 绘制坦克炮管
                ctx.fillStyle = this.color;
                ctx.fillRect(this.radius - 5, -5, this.radius + 10, 10);
                
                // 绘制坦克中心
                ctx.fillStyle = '#2c3e50';
                ctx.beginPath();
                ctx.arc(0, 0, this.radius/2, 0, Math.PI * 2);
                ctx.fill();
                
                // 绘制生命值
                ctx.restore();
                ctx.fillStyle = 'white';
                ctx.font = '12px Arial';
                ctx.textAlign = 'center';
                ctx.fillText(`HP: ${this.health}`, this.x, this.y - this.radius - 10);
            }
        }
        
        // 障碍物类
        class Obstacle {
            constructor(x, y, radius) {
                this.x = x;
                this.y = y;
                this.radius = radius;
                this.color = '#7f8c8d';
            }
            
            draw() {
                ctx.fillStyle = this.color;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
                ctx.fill();
                
                // 添加一些纹理
                ctx.strokeStyle = '#95a5a6';
                ctx.lineWidth = 2;
                ctx.stroke();
            }
        }
        
        // 创建爆炸效果
        function createExplosion(x, y) {
            game.explosions.push(new Explosion(x, y));
        }
        
        // 初始化游戏
        function initGame() {
            // 创建坦克
            game.tanks = [
                new Tank(100, 300, 1),
                new Tank(700, 300, 2)
            ];
            
            // 创建障碍物
            game.obstacles = [
                new Obstacle(200, 200, 30),
                new Obstacle(600, 200, 30),
                new Obstacle(200, 400, 30),
                new Obstacle(600, 400, 30),
                new Obstacle(400, 300, 40)
            ];
            
            // 重置游戏状态
            game.bullets = [];
            game.explosions = [];
            game.gameOver = false;
            game.winner = null;
            
            // 更新分数显示
            updateScore();
        }
        
        // 更新分数显示
        function updateScore() {
            const scoreElement = document.getElementById('score');
            const tank1 = game.tanks[0];
            const tank2 = game.tanks[1];
            
            if (tank1 && tank2) {
                scoreElement.textContent = `玩家1: ${tank1.health} | 玩家2: ${tank2.health}`;
            }
        }
        
        // 显示游戏结束画面
        function showGameOver() {
            const gameOverElement = document.getElementById('gameOver');
            const winnerTextElement = document.getElementById('winnerText');
            
            winnerTextElement.textContent = `玩家${game.winner}获胜！`;
            gameOverElement.style.display = 'block';
        }
        
        // 重新开始游戏
        function restartGame() {
            document.getElementById('gameOver').style.display = 'none';
            initGame();
        }
        
        // 游戏主循环
        function gameLoop() {
            // 清空画布
            ctx.fillStyle = '#34495e';
            ctx.fillRect(0, 0, game.width, game.height);
            
            if (!game.gameOver) {
                // 更新和绘制坦克
                for (const tank of game.tanks) {
                    tank.update();
                    tank.draw();
                }
                
                // 更新和绘制子弹
                game.bullets = game.bullets.filter(bullet => bullet.active);
                for (const bullet of game.bullets) {
                    bullet.update();
                    bullet.draw();
                }
                
                // 更新和绘制爆炸效果
                game.explosions = game.explosions.filter(explosion => explosion.active);
                for (const explosion of game.explosions) {
                    explosion.update();
                    explosion.draw();
                }
                
                // 绘制障碍物
                for (const obstacle of game.obstacles) {
                    obstacle.draw();
                }
                
                // 更新分数
                updateScore();
            }
            
            // 继续游戏循环
            requestAnimationFrame(gameLoop);
        }
        
        // 键盘事件处理
        window.addEventListener('keydown', (e) => {
            game.keys[e.key] = true;
            
            // 防止方向键和空格键滚动页面
            if (['ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight', ' '].includes(e.key)) {
                e.preventDefault();
            }
        });
        
        window.addEventListener('keyup', (e) => {
            game.keys[e.key] = false;
        });
        
        // 初始化并开始游戏
        window.onload = () => {
            initGame();
            gameLoop();
        };
    </script>
</body>
</html>
```

这个实现包含了以下功能：

1. **子弹系统**：
   - `Bullet` 类实现了子弹的发射、移动和绘制
   - 子弹具有速度、方向和生命周期管理
   - 子弹可以击中坦克和障碍物
   - 添加了子弹尾迹效果，增强视觉效果

2. **坦克系统**：
   - 两辆坦克分别由玩家1和玩家2控制
   - 玩家1使用WASD移动，空格射击
   - 玩家2使用方向键移动，回车射击
   - 坦克具有生命值，被击中会减少

3. **游戏机制**：
   - 包含障碍物系统，子弹和坦克与障碍物碰撞检测
   - 爆炸效果系统
   - 游戏结束和重新开始功能
   - 实时显示玩家生命值

4. **视觉效果**：
   - 不同颜色的坦克和子弹区分玩家
   - 爆炸动画效果
   - 子弹尾迹效果

游戏使用原生Canvas API实现，没有依赖任何外部框架，所有代码都在一个HTML文件中，可以直接运行。
