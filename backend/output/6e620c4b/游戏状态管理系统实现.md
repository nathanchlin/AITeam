        // 游戏主类
        class Game {

                this.canvas = canvas;

            createObstacles() {

                this.obstacles.push({
                    x: this.width - wallThickness, y: 0, width: wallThickness, height: this.height

                const internalObstacles = [

                    { x: 550, y: 430, width: 100, height: 20 },
                    { x: 350, y: 250, width: 20, height: 100 },

                this.obstacles.push(...internalObstacles);
            }

                this.tanks.push(new Tank(100, 100, 0, '#3498db', 'player1'));

                        this.start();

                    document.getElementById('gameStatus').textContent = '已暂停';
                }
            }
            
            resume() {
                if (this.state === GameState.PAUSED) {
                    this.state = GameState.PLAYING;
                    document.getElementById('gameOverlay').classList.add('hidden');
                    document.getElementById('gameStatus').textContent = '进行中';
                    this.gameLoop();
                }
            }
            
            gameOver(winner) {

                    this.reset();
                    this.start();
                });
                
                document.getElementById('gameStatus').textContent = '已结束';
            }
            
            reset() {
                // 重置游戏状态
                this.score = { player1: 0, player2: 0 };
                document.getElementById('player1Score').textContent = '0';
                document.getElementById('player2Score').textContent = '0';
                
                // 清空游戏对象
                this.tanks = [];
                this.bullets = [];
                this.explosions = [];
                
                // 重新创建坦克
                this.createTanks();
            }
            
            update() {
                if (this.state !== GameState.PLAYING) return;
                
                // 更新坦克
                for (let tank of this.tanks) {
                    tank.update(this.keys, this.obstacles);
                    
                    // 射击
                    if (tank.shouldShoot(this.keys)) {
                        const bullet = tank.shoot();
                        if (bullet) {
                            this.bullets.push(bullet);
                        }
                    }
                }
                
                // 更新子弹
                for (let i = this.bullets.length - 1; i >= 0; i--) {
                    const bullet = this.bullets[i];
                    bullet.update();
                    
                    // 检查子弹是否超出边界
                    if (bullet.isOutOfBounds(this.width, this.height)) {
                        this.bullets.splice(i, 1);
                        continue;
                    }
                    
                    // 检查子弹与障碍物的碰撞
                    for (let obstacle of this.obstacles) {
                        if (this.checkCollision(bullet, obstacle)) {
                            this.bullets.splice(i, 1);
                            this.createExplosion(bullet.x, bullet.y);
                            break;
                        }
                    }
                    
                    // 检查子弹与坦克的碰撞
                    for (let tank of this.tanks) {
                        if (bullet.owner !== tank && this.checkCollision(bullet, tank)) {
                            this.bullets.splice(i, 1);
                            tank.hit();
                            this.createExplosion(bullet.x, bullet.y);
                            
                            // 更新分数
                            if (tank.health <= 0) {
                                if (bullet.owner.id === 'player1') {
                                    this.score.player1++;
                                    document.getElementById('player1Score').textContent = this.score.player1;
                                } else {
                                    this.score.player2++;
                                    document.getElementById('player2Score').textContent = this.score.player2;
                                }
                                
                                // 检查游戏是否结束
                                const aliveTanks = this.tanks.filter(t => t.health > 0);
                                if (aliveTanks.length === 1) {
                                    this.gameOver(aliveTanks[0].id === 'player1' ? '玩家1' : '玩家2');
                                }
                            }
                            break;
                        }
                    }
                }
                
                // 更新爆炸效果
                for (let i = this.explosions.length - 1; i >= 0; i--) {
                    const explosion = this.explosions[i];
                    explosion.update();
                    
                    if (explosion.isDone()) {
                        this.explosions.splice(i, 1);
                    }
                }
            }
            
            render() {
                // 清空画布
                this.ctx.fillStyle = '#34495e';
                this.ctx.fillRect(0, 0, this.width, this.height);
                
                // 绘制障碍物
                this.ctx.fillStyle = '#7f8c8d';
                for (let obstacle of this.obstacles) {
                    this.ctx.fillRect(obstacle.x, obstacle.y, obstacle.width, obstacle.height);
                }
                
                // 绘制坦克
                for (let tank of this.tanks) {
                    tank.render(this.ctx);
                }
                
                // 绘制子弹
                for (let bullet of this.bullets) {
                    bullet.render(this.ctx);
                }
                
                // 绘制爆炸效果
                for (let explosion of this.explosions) {
                    explosion.render(this.ctx);
                }
            }
            
            checkCollision(obj1, obj2) {
                return obj1.x < obj2.x + obj2.width &&
                       obj1.x + obj1.width > obj2.x &&
                       obj1.y < obj2.y + obj2.height &&
                       obj1.y + obj1.height > obj2.y;
            }
            
            createExplosion(x, y) {
                this.explosions.push(new Explosion(x, y));
            }
            
            gameLoop() {
                if (this.state !== GameState.PLAYING) return;
                
                this.update();
                this.render();
                
                requestAnimationFrame(() => this.gameLoop());
            }
        }
        
        // 坦克类
        class Tank {
            constructor(x, y, angle, color, id) {
                this.x = x;
                this.y = y;
                this.angle = angle;
                this.color = color;
                this.id = id;
                this.width = 30;
                this.height = 30;
                this.speed = 2;
                this.rotationSpeed = 0.05;
                this.lastShot = 0;
                this.shotCooldown = 500; // 毫秒
                this.health = 3;
                this.maxHealth = 3;
            }
            
            update(keys, obstacles) {
                // 移动
                let dx = 0;
                let dy = 0;
                
                if (this.id === 'player1') {
                    if (keys['w'] || keys['W']) dy = -this.speed;
                    if (keys['s'] || keys['S']) dy = this.speed;
                    if (keys['a'] || keys['A']) dx = -this.speed;
                    if (keys['d'] || keys['D']) dx = this.speed;
                } else if (this.id === 'player2') {
                    if (keys['ArrowUp']) dy = -this.speed;
                    if (keys['ArrowDown']) dy = this.speed;
                    if (keys['ArrowLeft']) dx = -this.speed;
                    if (keys['ArrowRight']) dx = this.speed;
                }
                
                // 计算新位置
                const newX = this.x + dx;
                const newY = this.y + dy;
                
                // 创建临时坦克对象用于碰撞检测
                const tempTank = {
                    x: newX,
                    y: newY,
                    width: this.width,
                    height: this.height
                };
                
                // 检查是否与障碍物碰撞
                let canMove = true;
                for (let obstacle of obstacles) {
                    if (this.checkCollision(tempTank, obstacle)) {
                        canMove = false;
                        break;
                    }
                }
                
                // 更新位置
                if (canMove) {
                    this.x = newX;
                    this.y = newY;
                }
                
                // 旋转
                if (this.id === 'player1') {
                    if (keys['q'] || keys['Q']) this.angle -= this.rotationSpeed;
                    if (keys['e'] || keys['E']) this.angle += this.rotationSpeed;
                } else if (this.id === 'player2') {
                    if (keys['n'] || keys['N']) this.angle -= this.rotationSpeed;
                    if (keys['m'] || keys['M']) this.angle += this.rotationSpeed;
                }
            }
            
            shouldShoot(keys) {
                const now = Date.now();
                if (now - this.lastShot < this.shotCooldown) return false;
                
                if (this.id === 'player1' && keys[' ']) {
                    this.lastShot = now;
                    return true;
                } else if (this.id === 'player2' && keys['Enter']) {
                    this.lastShot = now;
                    return true;
                }
                
                return false;
            }
            
            shoot() {
                const bulletX = this.x + this.width / 2 + Math.cos(this.angle) * this.width;
                const bulletY = this.y + this.height / 2 + Math.sin(this.angle) * this.height;
                
                return new Bullet(
                    bulletX,
                    bulletY,
                    this.angle,
                    this.id,
                    this.color
                );
            }
            
            hit() {
                this.health--;
            }
            
            render(ctx) {
                ctx.save();
                ctx.translate(this.x + this.width / 2, this.y + this.height / 2);
                ctx.rotate(this.angle);
                
                // 绘制坦克主体
                ctx.fillStyle = this.color;
                ctx.fillRect(-this.width / 2, -this.height / 2, this.width, this.height);
                
                // 绘制炮管
                ctx.fillStyle = this.color;
                ctx.fillRect(this.width / 2 - 5, -3, this.width / 2 + 5, 6);
                
                // 绘制生命值
                ctx.restore();
                
                // 绘制生命值条
                const barWidth = 30;
                const barHeight = 4;
                const barX = this.x;
                const barY = this.y - 10;
                
                ctx.fillStyle = '#e74c3c';
                ctx.fillRect(barX, barY, barWidth, barHeight);
                
                ctx.fillStyle = '#2ecc71';
                ctx.fillRect(barX, barY, barWidth * (this.health / this.maxHealth), barHeight);
            }
            
            checkCollision(obj1, obj2) {
                return obj1.x < obj2.x + obj2.width &&
                       obj1.x + obj1.width > obj2.x &&
                       obj1.y < obj2.y + obj2.height &&
                       obj1.y + obj1.height > obj2.y;
            }
        }
        
        // 子弹类
        class Bullet {
            constructor(x, y, angle, owner, color) {
                this.x = x;

                this.angle = angle;
