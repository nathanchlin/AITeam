        // 碰撞检测函数
        function checkCollision(obj1, obj2) {
            // 圆形碰撞检测
            if (obj1.radius && obj2.radius) {
                const dx = obj1.x - obj2.x;
                const dy = obj1.y - obj2.y;
                const distance = Math.sqrt(dx * dx + dy * dy);
                return distance < obj1.radius + obj2.radius;
            }
            
            // 矩形碰撞检测
            if (obj1.width && obj2.width) {
                return obj1.x < obj2.x + obj2.width &&
                       obj1.x + obj1.width > obj2.x &&
                       obj1.y < obj2.y + obj2.height &&
                       obj1.y + obj1.height > obj2.y;
            }
            
            // 圆形与矩形碰撞检测
            if (obj1.radius && obj2.width) {
                // 找到矩形上离圆心最近的点
                let closestX = Math.max(obj2.x, Math.min(obj1.x, obj2.x + obj2.width));
                let closestY = Math.max(obj2.y, Math.min(obj1.y, obj2.y + obj2.height));
                
                // 计算距离
                const dx = obj1.x - closestX;
                const dy = obj1.y - closestY;
                const distance = Math.sqrt(dx * dx + dy * dy);
                
                return distance < obj1.radius;
            }
            
            // 矩形与圆形碰撞检测
            if (obj1.width && obj2.radius) {
                // 找到矩形上离圆心最近的点
                let closestX = Math.max(obj1.x, Math.min(obj2.x, obj1.x + obj1.width));
                let closestY = Math.max(obj1.y, Math.min(obj2.y, obj1.y + obj1.height));
                
                // 计算距离
                const dx = obj2.x - closestX;
                const dy = obj2.y - closestY;
                const distance = Math.sqrt(dx * dx + dy * dy);
                
                return distance < obj2.radius;
            }
            
            return false;
        }
        
        // 处理坦克与障碍物的碰撞
        function handleTankObstacleCollision(tank, obstacle) {
            // 计算坦克中心与障碍物中心的向量
            const dx = tank.x - (obstacle.x + obstacle.width/2);
            const dy = tank.y - (obstacle.y + obstacle.height/2);
            
            // 计算重叠量
            const overlapX = (tank.width/2 + obstacle.width/2) - Math.abs(dx);
            const overlapY = (tank.height/2 + obstacle.height/2) - Math.abs(dy);
            
            // 如果有重叠，将坦克推离障碍物
            if (overlapX > 0 && overlapY > 0) {
                if (overlapX < overlapY) {
                    // 水平方向碰撞
                    tank.x += (dx > 0 ? overlapX : -overlapX);
                } else {
                    // 垂直方向碰撞
                    tank.y += (dy > 0 ? overlapY : -overlapY);
                }
            }
        }
        
        // 处理子弹与坦克的碰撞
        function handleBulletTankCollision(bullet, tank) {
            if (bullet.owner === tank.playerNum) return; // 子弹不能伤害发射它的坦克
            
            // 创建子弹的圆形碰撞体
            const bulletCircle = {
                x: bullet.x,
                y: bullet.y,
                radius: bullet.width/2
            };
            
            // 创建坦克的矩形碰撞体
            const tankRect = {
                x: tank.x - tank.width/2,
                y: tank.y - tank.height/2,
                width: tank.width,
                height: tank.height
            };
            
            // 检查碰撞
            if (checkCollision(bulletCircle, tankRect)) {
                bullet.active = false;
                tank.takeDamage(20);
                explosions.push(new Explosion(bullet.x, bullet.y, 15));
            }
        }

        function handleBulletObstacleCollision(bullet, obstacle) {
            // 创建子弹的圆形碰撞体

        // 处理坦克与坦克的碰撞
        function handleTankTankCollision(tank1, tank2) {
            // 创建两个坦克的矩形碰撞体
            const rect1 = {
                x: tank1.x - tank1.width/2,
                y: tank1.y - tank1.height/2,
                width: tank1.width,
                height: tank1.height
            };
            
            const rect2 = {
                x: tank2.x - tank2.width/2,
                y: tank2.y - tank2.height/2,
                width: tank2.width,
                height: tank2.height
            };
            
            // 检查碰撞
            if (checkCollision(rect1, rect2)) {
                // 计算重叠量
                const dx = tank1.x - tank2.x;
                const dy = tank1.y - tank2.y;
                const distance = Math.sqrt(dx * dx + dy * dy);
                
                if (distance === 0) return; // 避免除以零
                
                // 归一化方向向量
                const nx = dx / distance;
                const ny = dy / distance;
                
                // 计算重叠量
                const overlap = (tank1.width/2 + tank2.width/2) - distance;
                
                // 将两个坦克推开
                tank1.x += nx * overlap/2;
                tank1.y += ny * overlap/2;
                tank2.x -= nx * overlap/2;
                tank2.y -= ny * overlap/2;
            }
        }
