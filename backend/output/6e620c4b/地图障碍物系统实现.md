    <script>
        // 游戏常量
        const CELL_SIZE = 40;
        const GRID_WIDTH = 20;
        const GRID_HEIGHT = 15;
        const TANK_SIZE = 30;
        const BULLET_SIZE = 6;
        const TANK_SPEED = 3;
        const BULLET_SPEED = 7;
        const EXPLOSION_DURATION = 20;
        
        // 障碍物类型
        const OBSTACLE_TYPES = {
            DESTRUCTIBLE: 1,
            INDESTRUCTIBLE: 2,
            BASE: 3
        };
        
        // 游戏状态
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        
        // 键盘状态
        const keys = {};
        
        // 游戏对象数组
        let tanks = [];
        let bullets = [];
        let obstacles = [];
        let explosions = [];
        
        // 玩家分数
        const scores = {
            player1: 0,
            player2: 0
        };
        
        // 障碍物类
        class Obstacle {
            constructor(x, y, type) {
                this.x = x;
                this.y = y;
                this.type = type;
                this.width = CELL_SIZE;
                this.height = CELL_SIZE;
                this.health = type === OBSTACLE_TYPES.DESTRUCTIBLE ? 1 : Infinity;
            }
            
            draw() {
                ctx.save();
                
                if (this.type === OBSTACLE_TYPES.DESTRUCTIBLE) {
                    // 可破坏障碍物 - 砖块样式
                    ctx.fillStyle = '#8B4513';
                    ctx.fillRect(this.x, this.y, this.width, this.height);
                    
                    // 砖块纹理
                    ctx.strokeStyle = '#654321';
                    ctx.lineWidth = 1;
                    ctx.beginPath();
                    ctx.moveTo(this.x + this.width / 2, this.y);
                    ctx.lineTo(this.x + this.width / 2, this.y + this.height);
                    ctx.moveTo(this.x, this.y + this.height / 2);
                    ctx.lineTo(this.x + this.width, this.y + this.height / 2);
                    ctx.stroke();
                } else if (this.type === OBSTACLE_TYPES.INDESTRUCTIBLE) {
                    // 不可破坏障碍物 - 钢筋混凝土样式
                    ctx.fillStyle = '#555';
                    ctx.fillRect(this.x, this.y, this.width, this.height);
                    
                    // 混凝土纹理
                    ctx.strokeStyle = '#777';
                    ctx.lineWidth = 1;
                    for (let i = 0; i < 4; i++) {
                        for (let j = 0; j < 3; j++) {
                            ctx.strokeRect(
                                this.x + i * this.width / 4, 
                                this.y + j * this.height / 3, 
                                this.width / 4, 
                                this.height / 3
                            );
                        }
                    }
                } else if (this.type === OBSTACLE_TYPES.BASE) {
                    // 基地 - 特殊样式
                    ctx.fillStyle = '#FF0000';
                    ctx.fillRect(this.x, this.y, this.width, this.height);
                    
                    // 基地标示
                    ctx.fillStyle = '#FFFF00';
                    ctx.font = 'bold 16px Arial';
                    ctx.textAlign = 'center';
                    ctx.textBaseline = 'middle';
                    ctx.fillText('基地', this.x + this.width / 2, this.y + this.height / 2);
                }
                
                ctx.restore();
            }
            
            hit() {
                if (this.type === OBSTACLE_TYPES.DESTRUCTIBLE) {
                    this.health--;
                    if (this.health <= 0) {
                        return true; // 障碍物被摧毁
                    }
                }
                return false;
            }
        }
        
        // 坦克类
        class Tank {
            constructor(x, y, color, controls, isPlayer1) {
                this.x = x;
                this.y = y;
                this.width = TANK_SIZE;
                this.height = TANK_SIZE;
                this.color = color;
                this.controls = controls;
                this.isPlayer1 = isPlayer1;
                this.direction = 0; // 0: 上, 1: 右, 2: 下, 3: 左
                this.speed = TANK_SPEED;
                this.lastShot = 0;
                this.shotCooldown = 500; // 毫秒
                this.alive = true;
            }
            
            update() {
                if (!this.alive) return;
                
                // 移动
                let newX = this.x;
                let newY = this.y;
                
                if (keys[this.controls.up]) {
                    newY -= this.speed;
                    this.direction = 0;
                }
                if (keys[this.controls.down]) {
                    newY += this.speed;
                    this.direction = 2;
                }
                if (keys[this.controls.left]) {
                    newX -= this.speed;
                    this.direction = 3;
                }
                if (keys[this.controls.right]) {
                    newX += this.speed;
                    this.direction = 1;
                }
                
                // 边界检查
                newX = Math.max(0, Math.min(canvas.width - this.width, newX));
                newY = Math.max(0, Math.min(canvas.height - this.height, newY));
                
                // 障碍物碰撞检查
                const tankRect = {
                    x: newX,
                    y: newY,
                    width: this.width,
                    height: this.height
                };
                
                let canMove = true;
                for (const obstacle of obstacles) {
                    if (this.checkCollision(tankRect, obstacle)) {
                        canMove = false;
                        break;
                    }
                }
                
                // 坦克之间的碰撞检查
                for (const tank of tanks) {
                    if (tank !== this && tank.alive && this.checkCollision(tankRect, tank)) {
                        canMove = false;
                        break;
                    }
                }
                
                if (canMove) {
                    this.x = newX;
                    this.y = newY;
                }
                
                // 射击
                if (keys[this.controls.shoot]) {
                    this.shoot();
                }
            }
            
            shoot() {
                const now = Date.now();
                if (now - this.lastShot < this.shotCooldown) return;
                
                this.lastShot = now;
                
                let bulletX = this.x + this.width / 2 - BULLET_SIZE / 2;
                let bulletY = this.y + this.height / 2 - BULLET_SIZE / 2;
                
                // 根据坦克方向调整子弹初始位置
                switch (this.direction) {
                    case 0: // 上
                        bulletY = this.y - BULLET_SIZE;
                        break;
                    case 1: // 右
                        bulletX = this.x + this.width;
                        break;
                    case 2: // 下
                        bulletY = this.y + this.height;
                        break;
                    case 3: // 左
                        bulletX = this.x - BULLET_SIZE;
                        break;
                }
                
                bullets.push(new Bullet(
                    bulletX, 
                    bulletY, 
                    this.direction, 
                    this.color,
                    this.isPlayer1
                ));
            }
            
            draw() {
                if (!this.alive) return;
                
                ctx.save();
                ctx.translate(this.x + this.width / 2, this.y + this.height / 2);
                ctx.rotate(this.direction * Math.PI / 2);
                
                // 坦克主体
                ctx.fillStyle = this.color;
                ctx.fillRect(-this.width / 2, -this.height / 2, this.width, this.height);
                
                // 坦克炮管
                ctx.fillStyle = '#333';
                ctx.fillRect(-3, -this.height / 2 - 10, 6, 15);
                
                // 坦克细节
                ctx.fillStyle = '#222';
                ctx.fillRect(-this.width / 2 + 5, -this.height / 2 + 5, this.width - 10, 5);
                ctx.fillRect(-this.width / 2 + 5, this.height / 2 - 10, this.width - 10, 5);
                
                ctx.restore();
            }
            
            checkCollision(rect1, rect2) {
                return rect1.x < rect2.x + rect2.width &&
                       rect1.x + rect1.width > rect2.x &&
                       rect1.y < rect2.y + rect2.height &&
                       rect1.y + rect1.height > rect2.y;
            }
            
            hit() {
                this.alive = false;
                explosions.push(new Explosion(this.x + this.width / 2, this.y + this.height / 2));
                
                // 更新分数
                if (this.isPlayer1) {
                    scores.player2++;
                    document.getElementById('player2Score').textContent = scores.player2;
                } else {
                    scores.player1++;
                    document.getElementById('player1Score').textContent = scores.player1;
                }
            }
        }
        
        // 子弹类
        class Bullet {
            constructor(x, y, direction, color, shotByPlayer1) {
                this.x = x;
                this.y = y;
                this.width = BULLET_SIZE;
                this.height = BULLET_SIZE;
                this.direction = direction;
                this.color = color;
                this.speed = BULLET_SPEED;
                this.shotByPlayer1 = shotByPlayer1;
                this.active = true;
            }
            
            update() {
                if (!this.active) return;
                
                switch (this.direction) {
                    case 0: // 上
                        this.y -= this.speed;
                        break;
                    case 1: // 右
                        this.x += this.speed;
                        break;
                    case 2: // 下
                        this.y += this.speed;
                        break;
                    case 3: // 左
                        this.x -= this.speed;
                        break;
                }
                
                // 边界检查
                if (this.x < 0 || this.x > canvas.width || 
                    this.y < 0 || this.y > canvas.height) {
                    this.active = false;
                    return;
                }
                
                // 障碍物碰撞检查
                const bulletRect = {
                    x: this.x,
                    y: this.y,
                    width: this.width,
                    height: this.height
                };
                
                for (let i = 0; i < obstacles.length; i++) {
                    const obstacle = obstacles[i];
                    if (this.checkCollision(bulletRect, obstacle)) {
                        if (obstacle.hit()) {
                            // 障碍物被摧毁
                            explosions.push(new Explosion(
                                obstacle.x + obstacle.width / 2, 
                                obstacle.y + obstacle.height / 2
                            ));
                            obstacles.splice(i, 1);
                        }
                        this.active = false;
                        return;
                    }
                }
                
                // 坦克碰撞检查
                for (const tank of tanks) {
                    if (tank.alive && tank.isPlayer1 !== this.shotByPlayer1 && 
                        this.checkCollision(bulletRect, tank)) {
                        tank.hit();
                        this.active = false;
                        return;
                    }
                }
            }
            
            draw() {
                if (!this.active) return;
                
                ctx.fillStyle = this.color;
                ctx.beginPath();
                ctx.arc(
                    this.x + this.width / 2, 
                    this.y + this.height / 2, 
                    this.width / 2, 
                    0, 
                    Math.PI * 2
                );
                ctx.fill();
            }
            
            checkCollision(rect1, rect2) {
                return rect1.x < rect2.x + rect2.width &&
                       rect1.x + rect1.width > rect2.x &&
                       rect1.y < rect2.y + rect2.height &&
                       rect1.y + rect1.height > rect2.y;
            }
        }
        
        // 爆炸效果类
        class Explosion {
            constructor(x, y) {
                this.x = x;
                this.y = y;
                this.radius = 5;
                this.maxRadius = 25;
                this.frame = 0;
                this.totalFrames = EXPLOSION_DURATION;
            }
            
            update() {
                this.frame++;
                this.radius = (this.frame / this.totalFrames) * this.maxRadius;
                return this.frame < this.totalFrames;
            }
            
            draw() {
                const alpha = 1 - (this.frame / this.totalFrames);
                
                ctx.save();
                ctx.globalAlpha = alpha;
                
                // 外圈
                ctx.fillStyle = '#FF4500';
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
                ctx.fill();
                
                // 中圈
                ctx.fillStyle = '#FFD700';
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius * 0.7, 0, Math.PI * 2);
                ctx.fill();
                
                // 内圈
                ctx.fillStyle = '#FFFFFF';
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius * 0.4, 0, Math.PI * 2);
                ctx.fill();
                
                ctx.restore();
            }
        }
        
        // 生成地图障碍物
        function generateObstacles() {
            obstacles = [];
            
            // 生成边界障碍物（不可破坏）
            for (let x = 0; x < GRID_WIDTH; x++) {
                // 上边界
                obstacles.push(new Obstacle(x * CELL_SIZE, 0, OBSTACLE_TYPES.INDESTRUCTIBLE));
                // 下边界
                obstacles.push(new Obstacle(x * CELL_SIZE, (GRID_HEIGHT - 1) * CELL_SIZE, OBSTACLE_TYPES.INDESTRUCTIBLE));
            }
            
            for (let y = 1; y < GRID_HEIGHT - 1; y++) {
                // 左边界
                obstacles.push(new Obstacle(0, y * CELL_SIZE, OBSTACLE_TYPES.INDESTRUCTIBLE));
                // 右边界
                obstacles.push(new Obstacle((GRID_WIDTH - 1) * CELL_SIZE, y * CELL_SIZE, OBSTACLE_TYPES.INDESTRUCTIBLE));
            }
            
            // 生成随机障碍物
            const obstacleCount = 30;
            for (let i = 0; i < obstacleCount; i++) {
                let x, y;
                let validPosition = false;
                let attempts = 0;
                
                // 尝试找到一个不与玩家初始位置重叠的位置
                while (!validPosition && attempts < 100) {
                    x = Math.floor(Math.random() * (GRID_WIDTH - 2)) + 1;
                    y = Math.floor(Math.random() * (GRID_HEIGHT - 2)) + 1;
                    
                    // 检查是否与玩家初始位置重叠
                    const player1Area = {
                        x: 1 * CELL_SIZE,
                        y: 1 * CELL_SIZE,
                        width: 2 * CELL_SIZE,
                        height: 2 * CELL_SIZE
                    };
                    
                    const player2Area = {
                        x: (GRID_WIDTH - 3) * CELL_SIZE,
                        y: (GRID_HEIGHT - 3) * CELL_SIZE,
                        width: 2 * CELL_SIZE,
                        height: 2 * CELL_SIZE
                    };
                    
                    const newObstacle = {
                        x: x * CELL_SIZE,
                        y: y * CELL_SIZE,
                        width: CELL_SIZE,
                        height: CELL_SIZE
                    };
                    
                    if (!checkAreaCollision(newObstacle, player1Area) && 
                        !checkAreaCollision(newObstacle, player2Area)) {
                        validPosition = true;
                    }
                    
                    attempts++;
                }
                
                if (validPosition) {
                    // 随机选择障碍物类型
                    const type = Math.random() > 0.3 ? OBSTACLE_TYPES.DESTRUCTIBLE : OBSTACLE_TYPES.INDESTRUCTIBLE;
                    obstacles.push(new Obstacle(x * CELL_SIZE, y * CELL_SIZE, type));
                }
            }
            
            // 添加基地（不可破坏）
            obstacles.push(new Obstacle((GRID_WIDTH / 2 - 1) * CELL_SIZE, 0, OBSTACLE_TYPES.BASE));
            obstacles.push(new Obstacle((GRID_WIDTH / 2 - 1) * CELL_SIZE, (GRID_HEIGHT - 1) * CELL_SIZE, OBSTACLE_TYPES.BASE));
        }
        
        // 检查区域碰撞
        function checkAreaCollision(rect1, rect2) {
            return rect1.x < rect2.x + rect2.width &&
                   rect1.x + rect1.width > rect2.x &&
                   rect1.y < rect2.y + rect2.height &&
                   rect1.y + rect1.height
