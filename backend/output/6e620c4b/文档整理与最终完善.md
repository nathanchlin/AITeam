## 游戏说明

### 游戏概述
这是一个基于Canvas渲染的双人对战坦克游戏，两名玩家各自控制一辆坦克，在游戏地图上进行战斗。

### 操作说明
- **玩家1（蓝色坦克）**：
  - W/A/S/D：移动坦克
  - 空格键：发射子弹

- **玩家2（红色坦克）**：
  - 方向键：移动坦克
  - 回车键：发射子弹

### 游戏特性
1. 双人对战模式
2. 地图障碍物系统
3. 坦克移动与碰撞检测
4. 子弹发射与碰撞系统
5. 爆炸效果
6. 生命值系统
7. 得分系统
8. 游戏结束判定

        // 坦克类
        class Tank {
            constructor(x, y, color, controls, isPlayer1) {
                this.x = x;
                this.y = y;
                this.width = 40;
                this.height = 40;
                this.color = color;
                this.speed = 3;
                this.angle = 0;
                this.controls = controls;
                this.isPlayer1 = isPlayer1;
                this.health = 100;
                this.score = 0;
                this.lastShot = 0;
                this.shootCooldown = 500; // 射击冷却时间（毫秒）
                this.alive = true;
            }
            
            // 更新坦克位置
            update() {
                if (!this.alive) return;
                
                // 根据控制键移动坦克
                if (this.controls.up && this.y > 0) {
                    this.y -= this.speed;
                    this.angle = -Math.PI / 2;
                }
                if (this.controls.down && this.y < game.height - this.height) {
                    this.y += this.speed;
                    this.angle = Math.PI / 2;
                }
                if (this.controls.left && this.x > 0) {
                    this.x -= this.speed;
                    this.angle = Math.PI;
                }
                if (this.controls.right && this.x < game.width - this.width) {
                    this.x += this.speed;
                    this.angle = 0;
                }
                
                // 边界检查
                this.x = Math.max(0, Math.min(game.width - this.width, this.x));
                this.y = Math.max(0, Math.min(game.height - this.height, this.y));
                
                // 检查与障碍物的碰撞
                for (const obstacle of game.obstacles) {
                    if (this.checkCollision(obstacle)) {
                        // 简单的碰撞响应 - 将坦克推回
                        if (this.controls.up) this.y += this.speed;
                        if (this.controls.down) this.y -= this.speed;
                        if (this.controls.left) this.x += this.speed;
                        if (this.controls.right) this.x -= this.speed;
                    }
                }
            }
            
            // 检查碰撞
            checkCollision(obj) {
                return this.x < obj.x + obj.width &&
                       this.x + this.width > obj.x &&
                       this.y < obj.y + obj.height &&
                       this.y + this.height > obj.y;
            }

            shoot() {
                const now = Date.now();
                if (now - this.lastShot < this.shootCooldown) return;
                
                this.lastShot = now;

                const bulletX = this.x + this.width / 2 - 5;
                const bulletY = this.y + this.height / 2 - 5;
                
                // 创建新子弹
                game.bullets.push(new Bullet(
                    bulletX,
                    bulletY,
                    this.angle,
                    this.color,
                    this.isPlayer1
                ));
            }

        class Bullet {
            constructor(x, y, angle, color, fromPlayer1) {
                this.x = x;
                this.y = y;
                this.width = 10;
                this.height = 10;
                this.angle = angle;
                this.speed = 8;
                this.color = color;
                this.fromPlayer1 = fromPlayer1;
                this.active = true;
            }
            
            // 更新子弹位置
            update() {
                this.x += Math.cos(this.angle) * this.speed;
                this.y += Math.sin(this.angle) * this.speed;
                
                // 检查是否超出边界
                if (this.x < 0 || this.x > game.width || 
                    this.y < 0 || this.y > game.height) {
                    this.active = false;
                }
                
                // 检查与障碍物的碰撞
                for (const obstacle of game.obstacles) {
                    if (this.checkCollision(obstacle)) {
                        this.active = false;
                        game.explosions.push(new Explosion(
                            this.x + this.width / 2,
                            this.y + this.height / 2,
                            20
                        ));
                        break;
                    }

                    this.active = false;

                        this.x + this.width / 2,
                        this.y + this.height / 2,

            checkCollision(obj) {
                return this.x < obj.x + obj.width &&
                       this.x + this.width > obj.x &&
                       this.y < obj.y + obj.height &&
                       this.y + this.height > obj.y;
            }
            
            // 绘制子弹
            draw() {
                if (!this.active) return;
                
                ctx.fillStyle = this.color;
                ctx.beginPath();
                ctx.arc(this.x + this.width / 2, this.y + this.height / 2, 5, 0, Math.PI * 2);
                ctx.fill();

                    this.y + this.height / 2 - Math.sin(this.angle) * 5,
                    3,
                    0,
                    Math.PI * 2
                );
                ctx.fill();
            }
        }
        
        // 爆炸效果类
        class Explosion {
            constructor(x, y, maxRadius) {
                this.x = x;
                this.y = y;
                this.radius = 0;
                this.maxRadius = maxRadius;
                this.active = true;
                this.growthRate = 2;
            }
            
            // 更新爆炸效果
            update() {
                this.radius += this.growthRate;
                if (this.radius >= this.maxRadius) {
                    this.active = false;
                }
            }
            
            // 绘制爆炸效果
            draw() {
                if (!this.active) return;
                
                const alpha = 1 - (this.radius / this.maxRadius);
                ctx.fillStyle = `rgba(255, 100, 0, ${alpha})`;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
                ctx.fill();
                
                ctx.fillStyle = `rgba(255, 200, 0, ${alpha * 0.7})`;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius * 0.7, 0, Math.PI * 2);
                ctx.fill();
            }
        }
        
        // 粒子效果类
        class Particle {
            constructor(x, y, angle, speed, color) {
                this.x = x;
                this.y = y;
                this.angle = angle;
                this.speed = speed;
                this.color = color;
                this.size = Math.random() * 3 + 1;
                this.life = 1;
                this.decay = Math.random() * 0.02 + 0.01;
                this.active = true;
            }
            
            // 更新粒子
            update() {
                this.x += Math.cos(this.angle) * this.speed;
                this.y += Math.sin(this.angle) * this.speed;
                this.life -= this.decay;
                this.speed *= 0.98;
                
                if (this.life <= 0) {
                    this.active = false;
                }
            }
            
            // 绘制粒子
            draw() {
                if (!this.active) return;
                
                ctx.fillStyle = this.color;
                ctx.globalAlpha = this.life;
                ctx.fillRect(this.x, this.y, this.size, this.size);
                ctx.globalAlpha = 1;
            }
        }
        
        // 障碍物类
        class Obstacle {
            constructor(x, y, width, height) {
                this.x = x;
                this.y = y;
                this.width = width;
                this.height = height;
                this.color = '#7f8c8d';
            }
            
            // 绘制障碍物
            draw() {
                ctx.fillStyle = this.color;
                ctx.fillRect(this.x, this.y, this.width, this.height);
                
                // 添加纹理效果
                ctx.fillStyle = '#95a5a6';
                for (let i = 0; i < this.width; i += 10) {

                        if ((i + j) % 20 === 0) {
                            ctx.fillRect(this.x + i, this.y + j, 5, 5);
                        }
                    }
                }
            }
        }
