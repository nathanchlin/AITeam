<body>
    <div class="game-container">
        <canvas id="gameCanvas" width="400" height="600"></canvas>

    <script>
        // 获取画布和上下文
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        
        // 游戏状态
        let gameStarted = false;
        let gameOver = false;
        let score = 0;
        let highScore = localStorage.getItem('flappyHighScore') || 0;
        document.getElementById('highScore').textContent = highScore;
        
        // 小鸟对象
        const bird = {
            x: 50,
            y: 250,
            width: 34,
            height: 24,
            velocity: 0,
            gravity: 0.5,
            jump: -8,
            rotation: 0,
            frame: 0,
            animationSpeed: 5,
            
            draw: function() {
                ctx.save();
                ctx.translate(this.x + this.width/2, this.y + this.height/2);
                ctx.rotate(this.rotation);
                
                // 绘制小鸟身体
                ctx.fillStyle = '#FFD700';
                ctx.beginPath();
                ctx.ellipse(0, 0, this.width/2, this.height/2, 0, 0, Math.PI * 2);
                ctx.fill();
                
                // 绘制小鸟眼睛
                ctx.fillStyle = 'white';
                ctx.beginPath();
                ctx.arc(8, -5, 5, 0, Math.PI * 2);
                ctx.fill();
                
                ctx.fillStyle = 'black';
                ctx.beginPath();
                ctx.arc(10, -5, 2, 0, Math.PI * 2);
                ctx.fill();
                
                // 绘制小鸟嘴巴
                ctx.fillStyle = '#FF8C00';
                ctx.beginPath();
                ctx.moveTo(15, 0);
                ctx.lineTo(25, 0);
                ctx.lineTo(15, 5);
                ctx.closePath();
                ctx.fill();
                
                // 绘制小鸟翅膀（动画效果）
                ctx.fillStyle = '#FFA500';
                ctx.beginPath();
                const wingOffset = Math.sin(this.frame / this.animationSpeed) * 5;
                ctx.ellipse(-5, wingOffset, 10, 6, Math.PI/4, 0, Math.PI * 2);
                ctx.fill();
                
                ctx.restore();
            },
            
            update: function() {
                this.velocity += this.gravity;
                this.y += this.velocity;
                
                // 更新旋转角度
                this.rotation = Math.min(Math.max(this.velocity * 0.05, -0.5), 0.5);
                
                // 更新动画帧
                this.frame++;
                
                // 边界检查
                if (this.y + this.height > canvas.height || this.y < 0) {
                    gameOver = true;
                }
            },
            
            jump: function() {
                this.velocity = this.jump;
            }
        };
        
        // 管道对象
        function Pipe(x, gapY) {
            this.width = 52;
            this.gap = 150;
            this.x = x;
            this.gapY = gapY;
            this.speed = 2;
            this.passed = false;
            
            this.draw = function() {
                // 绘制上管道
                ctx.fillStyle = '#2ecc71';
                ctx.fillRect(this.x, 0, this.width, this.gapY);
                
                // 绘制上管道边缘
                ctx.fillStyle = '#27ae60';
                ctx.fillRect(this.x - 5, this.gapY - 30, this.width + 10, 30);
                
                // 绘制下管道
                ctx.fillStyle = '#2ecc71';
                ctx.fillRect(this.x, this.gapY + this.gap, this.width, canvas.height - this.gapY - this.gap);
                
                // 绘制下管道边缘
                ctx.fillStyle = '#27ae60';
                ctx.fillRect(this.x - 5, this.gapY + this.gap, this.width + 10, 30);
            };
            
            this.update = function() {
                this.x -= this.speed;
                
                // 检查是否通过管道
                if (!this.passed && this.x + this.width < bird.x) {
                    this.passed = true;
                    score++;
                    document.getElementById('score').textContent = score;
                }
            };
            
            this.offscreen = function() {
                return this.x + this.width < 0;
            };
            
            this.collidesWith = function(bird) {
                // 检查小鸟是否与管道碰撞
                if (bird.x < this.x + this.width &&
                    bird.x + bird.width > this.x) {
                    if (bird.y < this.gapY || bird.y + bird.height > this.gapY + this.gap) {
                        return true;
                    }
                }
                return false;
            };
        }
        
        // 管道数组
        let pipes = [];
        let pipeTimer = 0;
        const pipeInterval = 90; // 每90帧生成一个新管道
        
        // 背景元素
        const background = {
            clouds: [],
            
            init: function() {
                // 初始化云朵
                for (let i = 0; i < 3; i++) {
                    this.clouds.push({
                        x: Math.random() * canvas.width,
                        y: Math.random() * 200,
                        width: 60,
                        height: 30,
                        speed: 0.5
                    });
                }
            },
            
            draw: function() {
                // 绘制云朵
                ctx.fillStyle = 'rgba(255, 255, 255, 0.8)';
                this.clouds.forEach(cloud => {
                    ctx.beginPath();
                    ctx.arc(cloud.x, cloud.y, cloud.width/2, 0, Math.PI * 2);
                    ctx.arc(cloud.x + cloud.width/3, cloud.y - 5, cloud.width/3, 0, Math.PI * 2);
                    ctx.arc(cloud.x - cloud.width/3, cloud.y - 5, cloud.width/3, 0, Math.PI * 2);
                    ctx.fill();
                });
            },
            
            update: function() {
                // 更新云朵位置
                this.clouds.forEach(cloud => {
                    cloud.x -= cloud.speed;
                    if (cloud.x + cloud.width < 0) {
                        cloud.x = canvas.width + cloud.width;
                        cloud.y = Math.random() * 200;
                    }
                });
            }
        };
        
        // 地面
        const ground = {
            y: canvas.height - 50,
            height: 50,
            
            draw: function() {
                // 绘制地面
                ctx.fillStyle = '#8B4513';
                ctx.fillRect(0, this.y, canvas.width, this.height);
                
                // 绘制草地
                ctx.fillStyle = '#2ecc71';
                ctx.fillRect(0, this.y, canvas.width, 10);
            }
        };
        
        // 游戏循环
        function gameLoop() {
            if (!gameStarted || gameOver) return;
            
            // 清除画布
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            // 更新和绘制背景
            background.update();
            background.draw();
            
            // 更新和绘制小鸟
            bird.update();
            bird.draw();
            
            // 更新和绘制管道
            pipeTimer++;
            if (pipeTimer >= pipeInterval) {
                const gapY = 100 + Math.random() * (canvas.height - ground.height - 100 - 150);
                pipes.push(new Pipe(canvas.width, gapY));
                pipeTimer = 0;
            }
            
            // 更新和绘制所有管道
            for (let i = pipes.length - 1; i >= 0; i--) {
                pipes[i].update();
                pipes[i].draw();
                
                // 检查碰撞
                if (pipes[i].collidesWith(bird)) {
                    gameOver = true;
                }
                
                // 移除离开屏幕的管道
                if (pipes[i].offscreen()) {
                    pipes.splice(i, 1);
                }
            }
            
            // 绘制地面
            ground.draw();
            
            // 检查是否撞到地面
            if (bird.y + bird.height > ground.y) {
                gameOver = true;
            }
            
            requestAnimationFrame(gameLoop);
        }
        
        // 事件监听
        function handleJump() {
            if (!gameStarted) {
                startGame();
            } else if (!gameOver) {
                bird.jump();
            }
        }
        
        document.addEventListener('keydown', (e) => {
            if (e.key === ' ' || e.key === 'Spacebar') {
                e.preventDefault();
                handleJump();
            }
        });
        
        canvas.addEventListener('click', handleJump);
        
        // 开始游戏
        function startGame() {
            gameStarted = true;
            gameOver = false;
            score = 0;
            document.getElementById('score').textContent = score;
            bird.y = 250;
            bird.velocity = 0;
            bird.frame = 0;
            pipes = [];
            pipeTimer = 0;
            
            document.getElementById('startScreen').classList.add('hidden');
            document.getElementById('gameOverScreen').classList.add('hidden');
            
            gameLoop();
        }
        
        // 结束游戏
        function endGame() {
            gameOver = true;
            document.getElementById('finalScore').textContent = score;
            document.getElementById('gameOverScreen').classList.remove('hidden');
            
            // 更新最高分
            if (score > highScore) {
                highScore = score;
                localStorage.setItem('flappyHighScore', highScore);
                document.getElementById('highScore').textContent = highScore;
            }
        }
        
        // 按钮事件
        document.getElementById('startBtn').addEventListener('click', startGame);
        document.getElementById('restartBtn').addEventListener('click', startGame);
        
        // 初始化背景
        background.init();
        
        // 定期检查游戏状态
        setInterval(() => {
            if (gameStarted && gameOver) {
                endGame();
            }
        }, 100);
    </script>
</body>
</html>
