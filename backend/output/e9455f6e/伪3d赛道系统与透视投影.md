    class MathUtils {
        static lerp(a, b, t) {
            return a + (b - a) * t;
        }
        
        static clamp(value, min, max) {
            return Math.max(min, Math.min(max, value));
        }
        
        static randomRange(min, max) {
            return Math.random() * (max - min) + min;
        }
        
        static randomInt(min, max) {
            return Math.floor(Math.random() * (max - min + 1)) + min;
        }
        
        static easeInOutQuad(t) {
            return t < 0.5 ? 2 * t * t : 1 - Math.pow(-2 * t + 2, 2) / 2;
        }
        
        static easeOutCubic(t) {
            return 1 - Math.pow(1 - t, 3);
        }
        
        static easeInOut(t) {
            return t < 0.5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2;
        }
        
        static exponentialFog(distance, density) {
            return 1 / Math.pow(Math.E, (distance * distance * density));
        }
        
        static project(p, cameraX, cameraY, cameraZ, cameraDepth, width, height, roadWidth) {
            const translatedZ = p.z - cameraZ;
            p.camera = {
                x: p.x - cameraX,
                y: p.y - cameraY,
                z: translatedZ
            };
            
            if (translatedZ <= 0) {
                p.screen = { x: 0, y: 0, w: 0, scale: 0 };
                return p;
            }
            
            const scale = cameraDepth / translatedZ;
            p.screen = {
                x: Math.round(width / 2 + scale * p.camera.x * width / 2),
                y: Math.round(height / 2 - scale * p.camera.y * height / 2),
                w: Math.round(scale * roadWidth * width / 2),
                scale: scale
            };
            
            return p;
        }
    }

    class ResourceLoader {
        constructor() {
            this.resources = new Map();
            this.loadingProgress = 0;
            this.totalResources = 0;
            this.loadedResources = 0;
        }
        
        loadImages(imageList) {
            return new Promise((resolve) => {
                this.totalResources = imageList.length;
                this.loadedResources = 0;
                
                if (imageList.length === 0) {
                    resolve();
                    return;
                }
                
                imageList.forEach(({ name, src }) => {
                    const img = new Image();
                    img.onload = () => {
                        this.resources.set(name, img);
                        this.loadedResources++;
                        this.loadingProgress = this.loadedResources / this.totalResources;
                        
                        if (this.loadedResources === this.totalResources) {
                            resolve();
                        }
                    };
                    img.onerror = () => {
                        console.warn(`Failed to load: ${src}`);
                        this.loadedResources++;
                        this.loadingProgress = this.loadedResources / this.totalResources;
                        
                        if (this.loadedResources === this.totalResources) {
                            resolve();
                        }
                    };
                    img.src = src;
                });
            });
        }
        
        getImage(name) {
            return this.resources.get(name);
        }
        
        getProgress() {
            return this.loadingProgress;
        }
    }

    class PerformanceMonitor {
        constructor() {
            this.fps = 0;
            this.frameTime = 0;
            this.drawCalls = 0;
            this.objectCount = 0;
            
            this.frameCount = 0;
            this.lastTime = performance.now();

                fps: document.getElementById('fpsValue'),
                frameTime: document.getElementById('frameTimeValue'),
                drawCalls: document.getElementById('drawCallsValue'),
                objectCount: document.getElementById('segmentsValue'),
                state: document.getElementById('stateValue')
            };
        }
        
        beginFrame() {
            this.frameStartTime = performance.now();
            this.drawCalls = 0;
            this.objectCount = 0;
        }
        
        endFrame() {
            const now = performance.now();
            const frameTime = now - this.frameStartTime;
            
            this.frameTimeHistory.push(frameTime);
            if (this.frameTimeHistory.length > this.maxHistoryLength) {
                this.frameTimeHistory.shift();
            }
            
            this.frameCount++;
            
            if (now - this.fpsUpdateTime >= 500) {
                this.fps = Math.round(this.frameCount * 1000 / (now - this.fpsUpdateTime));
                this.frameCount = 0;
                this.fpsUpdateTime = now;
                
                const sum = this.frameTimeHistory.reduce((a, b) => a + b, 0);
                this.frameTime = (sum / this.frameTimeHistory.length).toFixed(2);
                
                this.updateDisplay();
            }
        }
        
        addDrawCall(count = 1) {
            this.drawCalls += count;
        }
        
        addObject(count = 1) {
            this.objectCount += count;
        }
        
        updateDisplay() {
            if (this.elements.fps) this.elements.fps.textContent = this.fps;
            if (this.elements.frameTime) this.elements.frameTime.textContent = this.frameTime + 'ms';
            if (this.elements.drawCalls) this.elements.drawCalls.textContent = this.drawCalls;
            if (this.elements.objectCount) this.elements.objectCount.textContent = this.objectCount;
            
            if (this.elements.fps) {
                if (this.fps >= 55) {
                    this.elements.fps.style.color = CONFIG.COLORS.neonGreen;

                    this.elements.fps.style.color = CONFIG.COLORS.neonYellow;

                }
            }

    // 道路构建器
    // ============================================
    class RoadBuilder {
        constructor() {
            this.segments = [];
            this.segmentLength = CONFIG.SEGMENT_LENGTH;
            this.trackLength = 0;
        }
        
        build() {
            this.segments = [];
            
            // 构建一个包含各种元素的赛道
            this.addStraight(50);
            this.addCurve(50, 2);
            this.addHill(30, 30);
            this.addCurve(40, -3);
            this.addStraight(30);
            this.addHill(20, -20);
            this.addCurve(60, 4);
            this.addStraight(40);
            this.addCurve(30, -2);
            this.addHill(40, 40);
            this.addCurve(50, 3);
            this.addStraight(30);
            this.addCurve(40, -4);
            this.addHill(30, -30);
            this.addStraight(50);
            this.addCurve(60, 2);
            this.addHill(25, 25);
            this.addStraight(40);
            this.addSCurve();
            this.addStraight(50);
            
            this.trackLength = this.segments.length * this.segmentLength;
            
            console.log(`Track built: ${this.segments.length} segments, ${(this.trackLength / 1000).toFixed(1)}km`);
            return this.segments;
        }
        
        addSegment(curve, y) {
            const n = this.segments.length;
            const prevY = n > 0 ? this.segments[n - 1].p2.y : 0;
            
            this.segments.push(new RoadSegment(
                n,
                curve,
                prevY,
                false
            ));
            
            // 设置p2的y值为当前目标y
            this.segments[n].p2.y = y;
        }
        
        addStraight(numSegments) {
            for (let i = 0; i < numSegments; i++) {
                this.addSegment(0, 0);
            }
        }
        
        addCurve(numSegments, curve) {
            const enterCurve = numSegments / 3;
            const holdCurve = numSegments / 3;
            const exitCurve = numSegments - enterCurve - holdCurve;
            
            for (let i = 0; i < numSegments; i++) {
                let c = 0;
                if (i < enterCurve) {
                    c = curve * MathUtils.easeInOut(i / enterCurve);
                } else if (i < enterCurve + holdCurve) {
                    c = curve;
                } else {
                    c = curve * MathUtils.easeInOut(1 - (i - enterCurve - holdCurve) / exitCurve);
                }
                this.addSegment(c, 0);
            }
        }
        
        addHill(numSegments, height) {
            for (let i = 0; i < numSegments; i++) {
                const y = MathUtils.easeInOut(i / numSegments) * height;
                this.addSegment(0, y);
            }
        }
        
        addSCurve() {
            this.addCurve(15, -3);
            this.addCurve(15, 3);
            this.addCurve(15, -3);
            this.addCurve(15, 3);
        }
        
        findSegment(z) {
            if (this.segments.length === 0) return null;
            
            // 处理循环赛道
            const trackLength = this.trackLength;
            let zPos = z;
            while (zPos < 0) zPos += trackLength;
            while (zPos >= trackLength) zPos -= trackLength;
            
            const index = Math.floor(zPos / this.segmentLength) % this.segments.length;
            return this.segments[index];
        }
    }

    class Pseudo3DRenderer {
        constructor(canvas) {
            this.canvas = canvas;
            this.ctx = canvas.getContext('2d');
            this.width = CONFIG.CANVAS_WIDTH;
            this.height = CONFIG.CANVAS_HEIGHT;
            
            this.canvas.width = this.width;
            this.canvas.height = this.height;
            this.ctx.imageSmoothingEnabled = false;
            
            this.drawDistance = CONFIG.DRAW_DISTANCE;
            this.cameraDepth = CONFIG.CAMERA_DEPTH;
            
            // 渲染统计
            this.drawCalls = 0;
            this.renderedSegments = 0;
        }
        
        renderBackground(skyOffset) {
            const ctx = this.ctx;
            const w = this.width;
            const h = this.height;
            
            // 绘制渐变天空
            const skyGradient = ctx.createLinearGradient(0, 0, 0, h / 2);
            skyGradient.addColorStop(0, CONFIG.COLORS.sky.top);
            skyGradient.addColorStop(1, CONFIG.COLORS.sky.bottom);
            ctx.fillStyle = skyGradient;
            ctx.fillRect(0, 0, w, h / 2);
            
            // 绘制星星
            ctx.fillStyle = 'rgba(255, 255, 255, 0.8)';
            for (let i = 0; i < 100; i++) {
                const x = (i * 137.5 + skyOffset * 0.1) % w;
                const y = (i * 73.7) % (h / 2);
                const size = (i % 3) + 1;
                ctx.beginPath();
                ctx.arc(x, y, size * 0.5, 0, Math.PI * 2);
                ctx.fill();
            }
            
            // 绘制霓虹网格地平线
            this.drawNeonHorizon(skyOffset);
        }
        
        drawNeonHorizon(offset) {
            const ctx = this.ctx;
            const w = this.width;
            const h = this.height;
            const horizonY = h / 2;
            
            // 远处山脉轮廓
            ctx.fillStyle = '#0a0a1a';
            ctx.beginPath();
            ctx.moveTo(0, horizonY);
            
            for (let x = 0; x <= w; x += 20) {
                const mountainHeight = 
                    Math.sin((x + offset * 0.2) * 0.01) * 30 +
                    Math.sin((x + offset * 0.3) * 0.02) * 20 +
                    Math.sin((x + offset * 0.1) * 0.005) * 50;
                ctx.lineTo(x, horizonY - mountainHeight - 20);
            }
            ctx.lineTo(w, horizonY);
            ctx.closePath();
            ctx.fill();
            
            // 霓虹城市剪影
            ctx.fillStyle = '#0f0f1f';
            for (let i = 0; i < 30; i++) {
                const x = ((i * 50 + offset * 0.05) % (w + 100)) - 50;
                const buildingWidth = 20 + Math.random() * 40;
                const buildingHeight = 40 + Math.sin(i * 1.5) * 60;
                
                ctx.fillRect(x, horizonY - buildingHeight, buildingWidth, buildingHeight);
                
                // 窗户灯光
                if (i % 3 === 0) {
                    const lightColor = i % 2 === 0 ? CONFIG.COLORS.neonCyan : CONFIG.COLORS.neonPink;
                    ctx.fillStyle = lightColor;
                    ctx.globalAlpha = 0.6;
                    for (let wy = horizonY - buildingHeight + 10; wy < horizonY - 10; wy += 15) {
                        for (let wx = x + 5; wx < x + buildingWidth - 5; wx += 10) {
                            if (Math.random() > 0.3) {
                                ctx.fillRect(wx, wy, 5, 8);
                            }
                        }
                    }
                    ctx.globalAlpha = 1;
                    ctx.fillStyle = '#0f0f1f';
                }
            }
        }
        
        renderRoad(segments, playerZ, cameraX, cameraY, baseSegment) {
            if (!segments || segments.length === 0) return;
            
            this.drawCalls = 0;
            this.renderedSegments = 0;
            
            const ctx = this.ctx;
            const width = this.width;
            const height = this.height;
            const segmentLength = CONFIG.SEGMENT_LENGTH;
            const roadWidth = CONFIG.ROAD_WIDTH;
            
            const baseIndex = baseSegment.index;
            const playerPercent = (playerZ % segmentLength) / segmentLength;
            
            let maxY = height;
            let x = 0;
            let dx = 0;
            
            // 从远到近渲染道路段
            for (let n = 0; n < this.drawDistance; n++) {
                const segIndex = (baseIndex + n) % segments.length;
                const segment = segments[segIndex];
                const looped = (baseIndex + n) >= segments.length;
                const fog = MathUtils.exponentialFog(n / this.drawDistance, 2);
                
                // 计算camZ偏移以实现循环
                const camZ = playerZ - (looped ? segments.length * segmentLength : 0);
                
                // 投影道路段的两个端点
                const p1 = segment.p1;
                const p2 = segment.p2;
                
                // 设置世界坐标
                p1.x = x;
                p1.z = segment.z + (looped ? segments.length * segmentLength : 0);
                
                p2.x = x + dx;
                p2.z = p1.z + segmentLength;
                
                // 投影到屏幕
                MathUtils.project(p1, cameraX, cameraY, camZ, this.cameraDepth, width, height, roadWidth);
                MathUtils.project(p2, cameraX, cameraY, camZ, this.cameraDepth, width, height, roadWidth);
                
                x += dx;
                dx += segment.curve;
                
                // 跳过被遮挡的段
                if (p1.camera.z <= this.cameraDepth || p2.screen.y >= maxY || p2.screen.y >= p1.screen.y) {
                    continue;
                }
                
                // 渲染道路段
                this.renderSegment(segment, p1, p2, n, fog);
                
                maxY = p2.screen.y;
                this.renderedSegments++;
                this.drawCalls++;
            }
            
            return this.drawCalls;
        }
        
        renderSegment(segment, p1, p2, n, fog) {
            const ctx = this.ctx;
            
            // 确定颜色（交替条纹效果）
            const isLight = Math.floor(segment.index / 3) % 2 === 0;
            
            const asphaltColor = isLight ? CONFIG.COLORS.asphalt.light : CONFIG.COLORS.asphalt.dark;
            const grassColor = isLight ? CONFIG.COLORS.grass.light : CONFIG.COLORS.grass.dark;
            const rumbleColor = isLight ? CONFIG.COLORS.rumble.light : CONFIG.COLORS.rumble.dark;
            
            // 应用雾效
            const fogAsphalt = this.applyFog(asphaltColor, fog);
            const fogGrass = this.applyFog(grassColor, fog);
            const fogRumble = this.applyFog(rumbleColor, fog);
            
            // 绘制草地（道路两侧）
            this.drawTrapezoid(
                ctx,
                0, p2.screen.y, this.width, p2.screen.y,
                0, p1.screen.y, this.width, p1.screen.y,
                fogGrass
            );
            
            // 绘制路肩
            const rumbleW1 = p1.screen.w * 1.15;
            const rumbleW2 = p2.screen.w * 1.15;
            
            this.drawTrapezoid(
                ctx,
                p1.screen.x - rumbleW1, p1.screen.y, p1.screen.x + rumbleW1, p1.screen.y,
                p2.screen.x - rumbleW2, p2.screen.y, p2.screen.x + rumbleW2, p2.screen.y,
                fogRumble
            );
            
            // 绘制沥青路面
            this.drawTrapezoid(
                ctx,
                p1.screen.x - p1.screen.w, p1.screen.y, p1.screen.x + p1.screen.w, p1.screen.y,
                p2.screen.x - p2.screen.w, p2.screen.y, p2.screen.x + p2.screen.w, p2.screen.y,
                fogAsphalt
            );
            
            // 绘制车道线（每3段绘制一次）
            if (segment.index % 6 < 3) {
                const laneW1 = p1.screen.w / 20;
                const laneW2 = p2.screen.w / 20;
                
                // 中央虚线
                ctx.fillStyle = this.applyFog(CONFIG.COLORS.lane, fog);
                ctx.globalAlpha = 0.8;
                
                this.drawTrapezoid(
                    ctx,
                    p1.screen.x - laneW1 / 2, p1.screen.y, p1.screen.x + laneW1 / 2, p1.screen.y,
                    p2.screen.x - laneW2 / 2, p2.screen.y, p2.screen.x + laneW2 / 2, p2.screen.y,
                    this.applyFog(CONFIG.COLORS.neonCyan, fog)
                );
                
                ctx.globalAlpha = 1;
            }
            
            // 边缘发光线
            const glowW1 = p1.screen.w / 30;
            const glowW2 = p2.screen.w / 30;
            
            // 左边缘
            this.drawTrapezoid(
                ctx,
                p1.screen.x - p1.screen.w - glowW1, p1.screen.y, p1.screen.x - p1.screen.w, p1.screen.y,
                p2.screen.x - p2.screen.w - glowW2, p2.screen.y, p2.screen.x - p2.screen.w, p2.screen.y,
                this.applyFog(CONFIG.COLORS.neonCyan, fog * 0.7)
            );
            
            // 右边缘
            this.drawTrapezoid(
                ctx,
                p1.screen.x + p1.screen.w, p1.screen.y, p1.screen.x + p1.screen.w + glowW1, p1.screen.y,
                p2.screen.x + p2.screen.w, p2.screen.y, p2.screen.x + p2.screen.w + glowW2, p2.screen.y,
                this.applyFog(CONFIG.COLORS.neonCyan, fog * 0.7)
            );
            
            this.drawCalls += 5;
        }
        
        drawTrapezoid(ctx, x1a, y1a, x1b, y1b, x2a, y2a, x2b, y2b, color) {
            ctx.fillStyle = color;
            ctx.beginPath();
            ctx.moveTo(x1a, y1a);
            ctx.lineTo(x1b, y1b);
            ctx.lineTo(x2b, y2b);
            ctx.lineTo(x2a, y2a);
            ctx.closePath();
            ctx.fill();
        }
        
        applyFog(color, fog) {
            if (fog >= 1) return color;
            
            // 解析颜色
            const r = parseInt(color.slice(1, 3), 16);
            const g = parseInt(color.slice(3, 5), 16);
            const b = parseInt(color.slice(5, 7), 16);
            
            // 雾色
            const fogColor = CONFIG.COLORS.horizon;
            const fogR = parseInt(fogColor.slice(1, 3), 16);
            const fogG = parseInt(fogColor.slice(3, 5), 16);
            const fogB = parseInt(fogColor.slice(5, 7), 16);
            
            // 混合
            const finalR = Math.round(MathUtils.lerp(r, fogR, 1 - fog));
            const finalG = Math.round(MathUtils.lerp(g, fogG, 1 - fog));
            const finalB = Math.round(MathUtils.lerp(b, fogB, 1 - fog));
            
            return `#${finalR.toString(16).padStart(2, '0')}${finalG.toString(16).padStart(2, '0')}${finalB.toString(16).padStart(2, '0')}`;
        }
        
        getDrawCalls() {
            return this.drawCalls;
        }
        
        getRenderedSegments() {
            return this.renderedSegments;
        }
    }

    // ============================================
    // 玩家类
    // ============================================
    class Player {
        constructor() {
            this.x = 0;
            this.z = 0;
            this.speed = 0;
            this.maxSpeed = CONFIG.SEGMENT_LENGTH * 60;
            this.acceleration = this.maxSpeed / 3;
            this.braking = -this.maxSpeed;
            this.deceleration = -this.maxSpeed / 5;
            this.offRoadDeceleration = -this.maxSpeed / 2;
            this.offRoadLimit = this.maxSpeed / 4;
            this.centripetal = 0.3;
            
            this.position = 0;
        }
        
        update(deltaTime, input, segments, trackLength) {
            const segment = segments[Math.floor(this.z / CONFIG.SEGMENT_LENGTH) % segments.length];
            
            // 加速/刹车
            if (input.isKeyDown('KeyW') || input.isKeyDown('ArrowUp')) {
                this.speed += this.acceleration * deltaTime;
            } else if (input.isKeyDown('KeyS') || input.isKeyDown('ArrowDown')) {
                this.speed += this.braking * deltaTime;
            } else {
                this.speed += this.deceleration * deltaTime;
            }
            
            // 转向
            if (input.isKeyDown('KeyA') || input.isKeyDown('ArrowLeft')) {
                this.x -= 0.02 * (this.speed / this.maxSpeed) * deltaTime * 60;
            }
            if (input.isKeyDown('KeyD') || input.isKeyDown('ArrowRight')) {
                this.x += 0.02 * (this.speed / this.maxSpeed) * deltaTime * 60;
            }
            
            // 弯道离心力
            if (segment) {
                this.x -= segment.curve * this.centripetal * (this.speed / this.maxSpeed) * deltaTime;
            }
            
            // 限制在道路范围内
            this.x = MathUtils.clamp(this.x, -2, 2);
            
            // 离开道路减速
            if (Math.abs(this.x) > 1) {
                this.speed += this.offRoadDeceleration * deltaTime;
                if (this.speed > this.offRoadLimit) {
                    this.speed = MathUtils.clamp(this.speed, 0, this.offRoadLimit);
                }
            }
            
            // 速度限制
            this.speed = MathUtils.clamp(this.speed, 0, this.maxSpeed);
            
            // 更新位置
            this.z += this.speed * deltaTime;
            while (this.z < 0) this.z += trackLength;
            while (this.z >= trackLength) this.z -= trackLength;
            
            this.position = this.z;
        }
        
        getSpeedKMH() {
            return Math.round(this.speed / CONFIG.SEGMENT_LENGTH * 3.6);
        }
    }

    class GameStateMachine {
        constructor(game) {
            this.game = game;
            this.currentState = null;
            this.states = new Map();
            this.stateStartTime = 0;
        }
        
        registerState(name, state) {
            this.states.set(name, state);
        }
        
        changeState(name) {
            if (this.currentState && this.currentState.exit) {
                this.currentState.exit();
            }
            
            this.currentState = this.states.get(name);
            this.stateStartTime = performance.now();
            
            if (this.currentState && this.currentState.enter) {
                this.currentState.enter();
            }
            
            this.game.performanceMonitor.updateState(name);
            console.log(`State changed to: ${name}`);
        }
        
        update(deltaTime) {
            if (this.currentState && this.currentState.update) {
                this.currentState.update(deltaTime);
            }
        }
        
        render(renderer) {
            if (this.currentState && this.currentState.render) {
                this.currentState.render(renderer);
            }
        }
        
        getStateTime() {
            return performance.now() - this.stateStartTime;
        }
    }

    // ============================================
    // 游戏主类
    // ============================================
    class Game {
        constructor() {
            this.canvas = document.getElementById('gameCanvas');
            this.renderer = new Pseudo3DRenderer(this.canvas);
            this.performanceMonitor = new PerformanceMonitor();
            this.inputManager = new InputManager();
            this.resourceLoader = new ResourceLoader();
            this.stateMachine = new GameStateMachine(this);
            
            this.roadBuilder = new RoadBuilder();
            this.segments = [];
            this.player = new Player();
            this.trackLength = 0;
            
            this.running = false;
            this.lastTime = 0;
            this.skyOffset = 0;
            
            this.setupStates();
        }
        
        setupStates() {
            this.stateMachine.registerState('LOADING', {
                enter: () => this.showLoading(),
                update: (dt) => this.updateLoading(dt),
                render: (r) => this.renderLoading(r)
            });
            
            this.stateMachine.registerState('PLAYING', {
                enter: () => this.startPlaying(),
                update: (dt) => this.updatePlaying(dt),
                render: (r) => this.renderPlaying(r)
            });
        }
        
        async init() {
            this.stateMachine.changeState('LOADING');
            
            // 构建赛道
            this.updateLoadingText('Building track...');
            await this.sleep(100);
            this.segments = this.roadBuilder.build();
            this.trackLength = this.roadBuilder.trackLength;
            this.updateLoadingProgress(0.5);
            
            this.updateLoadingText('Initializing renderer...');
            await this.sleep(100);
            this.updateLoadingProgress(1);
            
            await this.sleep(500);
            
            this.stateMachine.changeState('PLAYING');
            this.running = true;
            this.lastTime = performance.now();
            this.gameLoop();
        }
        
        sleep(ms) {
            return new Promise(resolve => setTimeout(resolve, ms));
        }
        
        showLoading() {
            const loadingBar = document.getElementById('loadingBar');
            if (loadingBar) loadingBar.style.width = '0%';
        }
        
        updateLoading(dt) {}
        
        renderLoading(renderer) {}
        
        updateLoadingProgress(progress) {
            const loadingBar = document.getElementById('loadingBar');
            if (loadingBar) {
                loadingBar.style.width = (progress * 100) + '%';
            }
        }
        
        updateLoadingText(text) {
            const loadingText = document.getElementById('loadingText');
            if (loadingText) {
                loadingText.textContent = text;
            }
        }
        
        startPlaying() {
            this.player = new Player();
        }
        
        updatePlaying(deltaTime) {
            this.player.update(deltaTime, this.inputManager, this.segments, this.trackLength);
            this.skyOffset += this.player.speed * deltaTime * 0.01;
            
            // 更新HUD
            document.getElementById('speedValue').textContent = this.player.getSpeedKMH();
            document.getElementById('posValue').textContent = Math.floor(this.player.z / 1000) + 'km';
            
            this.inputManager.clearJustPressed();
        }
        
        renderPlaying(renderer) {
            // 获取玩家所在段
            const playerSegment = this.roadBuilder.findSegment(this.player.z);
            
            // 渲染背景
            renderer.renderBackground(this.skyOffset);
            
            // 计算相机Y位置（基于道路高度）
            const cameraY = CONFIG.CAMERA_HEIGHT + playerSegment.p1.y;
            
            // 渲染道路
            renderer.renderRoad(
                this.segments,
                this.player.z,
                this.player.x * CONFIG.ROAD_WIDTH,
                cameraY,
                playerSegment
            );
            
            this.performanceMonitor.addObject(renderer.getRenderedSegments());
            this.performanceMonitor.addDrawCall(renderer.getDrawCalls());
        }
        
        gameLoop() {
            if (!this.running) return;
            
            const now = performance.now();
            const deltaTime = Math.min((now - this.lastTime) / 1000, 0.1);
            this.lastTime = now;
            
            this.performanceMonitor.beginFrame();
            
            // 更新
            this.stateMachine.update(deltaTime);
            
            // 渲染
            this.stateMachine.render(this.renderer);
            
            this.performanceMonitor.endFrame();
            
            requestAnimationFrame(() => this.gameLoop());
        }
    }

    // ============================================

        const game = new Game();

    });

## 实现说明

### 核心技术点

1. **透视投影系统**
   - 使用 `MathUtils.project()` 方法将3D世界坐标投影到2D屏幕
   - 基于FOV计算相机深度：`cameraDepth = 1 / tan(FOV/2)`
   - 缩放公式：`scale = cameraDepth / translatedZ`

2. **道路分段系统**
   - `RoadSegment` 类存储每段道路的曲率和高度信息
   - `RoadBuilder` 负责生成包含直道、弯道、坡道的完整赛道
   - 使用缓和函数实现曲线的平滑过渡

3. **弯道曲线实现**
   - 通过累积 `dx` 偏移量实现水平弯曲
   - 弯道值越大，道路偏移越明显
   - 支持S形弯道组合

4. **坡道起伏效果**
   - 每段道路有独立的 `y` 高度值
   - 投影时自动产生上下起伏的视觉效果
   - 支持正坡（上坡）和负坡（下坡）

5. **赛道纹理**
   - 沥青路面：深色交替条纹
   - 草地：绿色交替区域
   - 路肩：粉白相间的霓虹边框
   - 中央虚线：霓虹青色发光

6. **雾效系统**
   - 指数雾效公式：`1 / e^(distance² × density)`
   - 远处道路逐渐融入地平线颜色
   - 增强距离感和氛围

7. **无限循环赛道**
   - 使用取模运算处理超出赛道长度的位置
   - 渲染时正确处理首尾衔接

### 控制方式
- **W/↑**：加速
- **S/↓**：刹车
- **A/←** 和 **D/→**：转向
- 驶出道路会自动减速
