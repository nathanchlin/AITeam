        BEST_LAPS: 'neonRacer_bestLaps'
    },
    
    init() {
        // 初始化默认设置
        if (!localStorage.getItem(this.KEYS.SETTINGS)) {
            this.saveSettings(SettingsManager.defaultSettings);
        }
        if (!localStorage.getItem(this.KEYS.RECORDS)) {
            localStorage.setItem(this.KEYS.RECORDS, JSON.stringify({}));
        }
        if (!localStorage.getItem(this.KEYS.BEST_LAPS)) {
            localStorage.setItem(this.KEYS.BEST_LAPS, JSON.stringify({}));
        }
    },
    
    // 保存游戏设置
    saveSettings(settings) {
        try {
            localStorage.setItem(this.KEYS.SETTINGS, JSON.stringify(settings));
            return true;
        } catch (e) {
            console.warn('无法保存设置:', e);
            return false;
        }
    },
    
    // 加载游戏设置
    loadSettings() {
        try {
            const data = localStorage.getItem(this.KEYS.SETTINGS);
            return data ? JSON.parse(data) : SettingsManager.defaultSettings;
        } catch (e) {
            console.warn('无法加载设置:', e);
            return SettingsManager.defaultSettings;
        }
    },
    
    // 保存比赛记录
    saveRecord(trackId, record) {
        try {
            const records = this.getAllRecords();
            if (!records[trackId]) {
                records[trackId] = [];
            }
            
            records[trackId].push({
                ...record,
                timestamp: Date.now(),
                date: new Date().toLocaleDateString('zh-CN')
            });
            
            // 按总时间排序，保留前10名
            records[trackId].sort((a, b) => a.totalTime - b.totalTime);
            records[trackId] = records[trackId].slice(0, 10);
            
            localStorage.setItem(this.KEYS.RECORDS, JSON.stringify(records));
            return true;
        } catch (e) {
            console.warn('无法保存记录:', e);
            return false;
        }
    },
    
    // 获取所有记录
    getAllRecords() {
        try {
            const data = localStorage.getItem(this.KEYS.RECORDS);
            return data ? JSON.parse(data) : {};
        } catch (e) {
            return {};
        }
    },
    
    // 获取特定赛道记录
    getTrackRecords(trackId) {
        const records = this.getAllRecords();
        return records[trackId] || [];
    },
    
    // 保存最佳单圈
    saveBestLap(trackId, lapTime, vehicleId) {
        try {
            const bestLaps = this.getAllBestLaps();
            const key = `${trackId}_${vehicleId}`;
            
            if (!bestLaps[key] || lapTime < bestLaps[key].time) {
                bestLaps[key] = {
                    time: lapTime,
                    vehicleId: vehicleId,
                    timestamp: Date.now()
                };
                localStorage.setItem(this.KEYS.BEST_LAPS, JSON.stringify(bestLaps));
                return true; // 新纪录
            }
            return false;
        } catch (e) {
            console.warn('无法保存最佳单圈:', e);
            return false;
        }
    },
    
    // 获取所有最佳单圈
    getAllBestLaps() {
        try {
            const data = localStorage.getItem(this.KEYS.BEST_LAPS);
            return data ? JSON.parse(data) : {};
        } catch (e) {
            return {};
        }
    },
    
    // 获取赛道最佳单圈
    getTrackBestLap(trackId, vehicleId = null) {
        const bestLaps = this.getAllBestLaps();
        if (vehicleId) {
            const key = `${trackId}_${vehicleId}`;
            return bestLaps[key] ? bestLaps[key].time : Infinity;
        } else {
            // 返回该赛道所有车辆中最好的
            let best = Infinity;
            for (const key in bestLaps) {
                if (key.startsWith(trackId + '_')) {
                    best = Math.min(best, bestLaps[key].time);
                }
            }
            return best;
        }
    },
    
    // 检查是否是新纪录
    isNewRecord(trackId, totalTime, lapTime) {
        const records = this.getTrackRecords(trackId);
        if (records.length === 0) return { total: true, lap: true };
        
        const bestTotal = records[0].totalTime;
        const bestLap = this.getTrackBestLap(trackId);
        
        return {
            total: totalTime < bestTotal,
            lap: lapTime < bestLap
        };
    },
    
    // 清除所有记录
    clearAllRecords() {
        if (confirm('确定要清除所有游戏记录吗？此操作不可撤销。')) {
            localStorage.removeItem(this.KEYS.RECORDS);
            localStorage.removeItem(this.KEYS.BEST_LAPS);
            UIManager.renderLeaderboard(TRACKS[0].id);
            alert('所有记录已清除');
        }
    }

const AudioManager = {
    context: null,
    masterGain: null,
    sfxGain: null,
    musicGain: null,
    
    sounds: {},
    oscillators: {},
    isInitialized: false,
    
    init() {
        if (this.isInitialized) return;
        
        try {
            this.context = new (window.AudioContext || window.webkitAudioContext)();
            
            // 创建增益节点
            this.masterGain = this.context.createGain();
            this.sfxGain = this.context.createGain();
            this.musicGain = this.context.createGain();
            
            // 连接节点
            this.sfxGain.connect(this.masterGain);
            this.musicGain.connect(this.masterGain);
            this.masterGain.connect(this.context.destination);
            
            // 从设置加载音量
            const settings = SettingsManager.currentSettings;
            this.setMasterVolume(settings.masterVolume / 100);
            this.setSFXVolume(settings.sfxVolume / 100);
            this.setMusicVolume(settings.musicVolume / 100);
            
            this.isInitialized = true;
            this.createSounds();
        } catch (e) {
            console.warn('Web Audio API 不可用:', e);
        }
    },
    
    resume() {
        if (this.context && this.context.state === 'suspended') {
            this.context.resume();
        }
    },
    
    createSounds() {
        // 使用振荡器合成各种音效
        this.soundConfigs = {
            engine: { type: 'sawtooth', frequency: 80, duration: 0.05, loop: true },
            nitro: { type: 'sine', frequency: 400, duration: 0.3 },
            collision: { type: 'square', frequency: 150, duration: 0.15 },
            countdown: { type: 'sine', frequency: 880, duration: 0.15 },
            go: { type: 'sine', frequency: 1320, duration: 0.3 },
            lap: { type: 'sine', frequency: 660, duration: 0.2 },
            finish: { type: 'sine', frequency: 523, duration: 0.5 },
            click: { type: 'sine', frequency: 1000, duration: 0.05 },
            wind: { type: 'sine', frequency: 100, duration: 0.1, loop: true }
        };
    },
    
    play(soundName, options = {}) {
        if (!this.isInitialized || !this.context) return;
        
        this.resume();
        
        const config = this.soundConfigs[soundName];
        if (!config) return;
        
        const osc = this.context.createOscillator();
        const gain = this.context.createGain();
        
        osc.type = config.type;
        osc.frequency.setValueAtTime(config.frequency * (options.pitch || 1), this.context.currentTime);
        
        if (options.frequency) {
            osc.frequency.setValueAtTime(options.frequency, this.context.currentTime);
        }
        
        gain.gain.setValueAtTime(0, this.context.currentTime);
        gain.gain.linearRampToValueAtTime(options.volume || 0.3, this.context.currentTime + 0.01);
        gain.gain.linearRampToValueAtTime(0, this.context.currentTime + (options.duration || config.duration));
        
        osc.connect(gain);
        gain.connect(this.sfxGain);
        
        osc.start(this.context.currentTime);
        osc.stop(this.context.currentTime + (options.duration || config.duration));
        
        return osc;
    },
    
    playEngine(speedRatio) {
        if (!this.isInitialized || !this.context) return;
        
        this.resume();
        
        const baseFreq = 60;
        const freq = baseFreq + speedRatio * 150;
        
        if (!this.oscillators.engine) {
            const osc = this.context.createOscillator();
            const gain = this.context.createGain();
            
            osc.type = 'sawtooth';
            osc.connect(gain);
            gain.connect(this.sfxGain);
            gain.gain.setValueAtTime(0, this.context.currentTime);
            
            osc.start();
            this.oscillators.engine = { osc, gain };
        }
        
        const e = this.oscillators.engine;
        e.osc.frequency.setTargetAtTime(freq, this.context.currentTime, 0.1);
        e.gain.gain.setTargetAtTime(Math.min(0.15, speedRatio * 0.2), this.context.currentTime, 0.1);
    },
    
    stopEngine() {
        if (this.oscillators.engine) {
            this.oscillators.engine.gain.gain.setTargetAtTime(0, this.context.currentTime, 0.1);
        }
    },
    
    playWind(speedRatio) {
        if (!this.isInitialized || !this.context) return;
        
        const volume = Math.min(0.1, speedRatio * 0.15);
        
        if (!this.oscillators.wind) {
            const osc = this.context.createOscillator();
            const gain = this.context.createGain();
            const filter = this.context.createBiquadFilter();
            
            osc.type = 'sine';
            filter.type = 'lowpass';
            filter.frequency.setValueAtTime(200, this.context.currentTime);
            
            osc.connect(filter);
            filter.connect(gain);
            gain.connect(this.sfxGain);
            gain.gain.setValueAtTime(0, this.context.currentTime);
            
            osc.start();
            this.oscillators.wind = { osc, gain, filter };
        }
        
        const w = this.oscillators.wind;
        w.filter.frequency.setTargetAtTime(100 + speedRatio * 300, this.context.currentTime, 0.1);
        w.gain.gain.setTargetAtTime(volume, this.context.currentTime, 0.1);
    },
    
    stopWind() {
        if (this.oscillators.wind) {
            this.oscillators.wind.gain.gain.setTargetAtTime(0, this.context.currentTime, 0.2);
        }
    },
    
    setMasterVolume(value) {
        if (this.masterGain) {
            this.masterGain.gain.setValueAtTime(value, this.context.currentTime);
        }
    },
    
    setSFXVolume(value) {
        if (this.sfxGain) {
            this.sfxGain.gain.setValueAtTime(value, this.context.currentTime);
        }
    },
    
    setMusicVolume(value) {
        if (this.musicGain) {
            this.musicGain.gain.setValueAtTime(value, this.context.currentTime);
        }
    }
};

const ParticleSystem = {
    particles: [],
    maxParticles: 200,
    
    emit(x, y, config) {
        if (!SettingsManager.getSetting('particles')) return;
        if (this.particles.length >= this.maxParticles) return;
        
        const count = config.count || 1;
        for (let i = 0; i < count; i++) {
            if (this.particles.length < this.maxParticles) {
                this.particles.push({
                    x: x + (Math.random() - 0.5) * (config.spreadX || 0),
                    y: y + (Math.random() - 0.5) * (config.spreadY || 0),
                    vx: (config.vx || 0) + (Math.random() - 0.5) * (config.randomVx || 0),
                    vy: (config.vy || 0) + (Math.random() - 0.5) * (config.randomVy || 0),
                    size: config.size || 3,
                    decay: config.decay || 0.02,
                    color: config.color || '#00f5ff',
                    alpha: 1,
                    life: config.life || 1
                });
            }
        }
    },
    
    update(dt) {
        for (let i = this.particles.length - 1; i >= 0; i--) {
            const p = this.particles[i];
            p.x += p.vx * dt;
            p.y += p.vy * dt;
            p.alpha -= p.decay * dt * 60;
            p.life -= dt;
            
            if (p.alpha <= 0 || p.life <= 0) {
                this.particles.splice(i, 1);
            }
        }
    },
    
    draw(ctx) {
        for (const p of this.particles) {
            ctx.globalAlpha = p.alpha;
            ctx.fillStyle = p.color;
            ctx.beginPath();
            ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
            ctx.fill();
        }
        ctx.globalAlpha = 1;
    },
    
    clear() {
        this.particles = [];
    }
};

class Player {
    constructor(index, vehicleId) {
        this.index = index;
        this.vehicle = VEHICLES[vehicleId];
        this.reset();
    }
    
    reset() {
        this.x = 0;
        this.z = 0;
        this.speed = 0;
        this.position = CONFIG.TRACK_LENGTH * 10;
        this.lap = 1;
        this.finished = false;
        this.nitro = 100;
        this.nitroActive = false;
        this.currentLapStart = 0;
        this.bestLapTime = Infinity;
        this.lapTimes = [];
        this.topSpeed = 0;
        this.checkpoint = 0;
    }
    
    update(dt, input, segments) {
        if (this.finished) return;
        
        const config = this.vehicle.config;
        const trackLength = CONFIG.TRACK_LENGTH * CONFIG.SEGMENT_LENGTH;
        
        // 获取当前路段
        const segmentIndex = Math.floor(this.position / CONFIG.SEGMENT_LENGTH) % CONFIG.TRACK_LENGTH;
        const segment = segments[segmentIndex];
        
        // 加速/减速
        if (input.up) {
            const accel = this.nitroActive ? config.acceleration * 1.5 : config.acceleration;
            this.speed = Math.min(this.speed + accel * dt * 60, this.nitroActive ? config.maxSpeed * 1.2 : config.maxSpeed);
        } else if (input.down) {
            this.speed = Math.max(this.speed - config.acceleration * 2 * dt * 60, -config.maxSpeed * 0.3);
        } else {
            this.speed *= 0.98;
        }
        
        // 转向
        if (input.left) {
            this.x -= 0.02 * this.speed * dt * 60 * config.driftFactor;
        }
        if (input.right) {
            this.x += 0.02 * this.speed * dt * 60 * config.driftFactor;
        }
        
        // 弯道影响
        if (segment) {
            this.x += segment.curve * this.speed * 0.00001 * dt * 60;
        }
        
        // 边界限制
        this.x = Math.max(-1, Math.min(1, this.x));
        
        // 草地减速
        if (Math.abs(this.x) > 0.8) {
            this.speed *= 0.95;
        }
        
        // 氮气
        if (input.nitro && this.nitro > 0) {
            this.nitroActive = true;
            this.nitro = Math.max(0, this.nitro - dt * 30);
            GameData.nitroUsedCount++;
        } else {
            this.nitroActive = false;
            this.nitro = Math.min(100, this.nitro + dt * 5);
        }
        
        // 更新位置
        this.position += this.speed * dt;
        this.z = this.position;
        
        // 圈数检测
        const prevLap = this.lap;
        if (this.position >= trackLength * this.lap) {
            this.lap++;
            const lapTime = GameData.currentTime - this.currentLapStart;
            this.lapTimes.push(lapTime);
            if (lapTime < this.bestLapTime) {
                this.bestLapTime = lapTime;
            }
            this.currentLapStart = GameData.currentTime;
            
            if (this.lap > CONFIG.TOTAL_LAPS) {
                this.finished = true;
                this.lap = CONFIG.TOTAL_LAPS;
            }
        }
        
        // 记录最高速度
        if (this.speed > this.topSpeed) {
            this.topSpeed = this.speed;
        }
    }
    
    getGear() {
        if (this.speed < 0) return 'R';
        if (this.speed === 0) return 'N';
        if (this.speed < 80) return '1';
        if (this.speed < 160) return '2';
        if (this.speed < 240) return '3';
        if (this.speed < 320) return '4';
        return '5';
    }
    
    getDisplaySpeed() {
        return Math.abs(Math.round(this.speed));
    }
}

class AIOpponent {
    constructor(index, vehicleId) {
        this.index = index;
        this.vehicle = VEHICLES[vehicleId];
        this.reset();
    }
    
    reset() {
        this.x = 0;
        this.position = CONFIG.TRACK_LENGTH * 10 + (this.index + 1) * 500;
        this.speed = 0;
        this.targetSpeed = this.vehicle.config.maxSpeed * (0.7 + Math.random() * 0.25);
        this.lap = 1;
        this.finished = false;
    }
    
    update(dt, segments, playerPosition) {
        if (this.finished) return;
        
        const config = this.vehicle.config;
        const segmentIndex = Math.floor(this.position / CONFIG.SEGMENT_LENGTH) % CONFIG.TRACK_LENGTH;
        const segment = segments[segmentIndex];
        
        // 基础AI速度调整
        const speedVariation = Math.sin(GameData.currentTime * 2 + this.index) * 20;
        const targetSpeed = this.targetSpeed + speedVariation;
        
        // 加速到目标速度
        if (this.speed < targetSpeed) {
            this.speed += config.acceleration * 0.8 * dt * 60;
        } else {
            this.speed -= config.acceleration * 0.5 * dt * 60;
        }
        
        // 弯道减速
        if (segment && Math.abs(segment.curve) > 2) {
            this.speed *= 0.995;
        }
        
        // 更新位置
        this.position += this.speed * dt;
        
        // 随机偏移
        this.x += (Math.random() - 0.5) * 0.01;
        this.x = Math.max(-0.7, Math.min(0.7, this.x));
        
        // 圈数检测
        const trackLength = CONFIG.TRACK_LENGTH * CONFIG.SEGMENT_LENGTH;
        if (this.position >= trackLength * this.lap) {
            this.lap++;
            if (this.lap > CONFIG.TOTAL_LAPS) {
                this.finished = true;
                this.lap = CONFIG.TOTAL_LAPS;
            }
        }
    }
}

    currentFPS: 60,
    
    screenShake: { x: 0, y: 0, intensity: 0 },
    nitroOverlay: null,
    
    inputP1: { up: false, down: false, left: false, right: false, nitro: false },
    inputP2: { up: false, down: false, left: false, right: false, nitro: false },
    
    init() {
        this.canvas = document.getElementById('gameCanvas');
        this.ctx = this.canvas.getContext('2d');
        this.vfxCanvas = document.getElementById('vfxCanvas');
        this.vfxCtx = this.vfxCanvas.getContext('2d');
        this.weatherCanvas = document.getElementById('weatherOverlay');
        this.weatherCtx = this.weatherCanvas.getContext('2d');
        this.nitroOverlay = document.getElementById('nitroOverlay');
        
        this.resize();
        window.addEventListener('resize', () => this.resize());
        
        this.createSegments();
        this.setupInput();
        
        // 初始化设置和音频
        SettingsManager.init();
        AudioManager.init();
        
        requestAnimationFrame((t) => this.gameLoop(t));
    },
    
    resize() {
        const container = document.getElementById('gameContainer');
        const rect = container.getBoundingClientRect();
        
        const dpr = Math.min(window.devicePixelRatio || 1, 2);
        
        this.canvas.width = rect.width * dpr;
        this.canvas.height = rect.height * dpr;
        this.vfxCanvas.width = rect.width * dpr;
        this.vfxCanvas.height = rect.height * dpr;
        this.weatherCanvas.width = rect.width * dpr;
        this.weatherCanvas.height = rect.height * dpr;
        
        this.ctx.scale(dpr, dpr);
        this.vfxCtx.scale(dpr, dpr);
        this.weatherCtx.scale(dpr, dpr);
        
        this.displayWidth = rect.width;
        this.displayHeight = rect.height;
    },
    
    createSegments() {
        const track = TRACKS[GameData.selectedTrack];
        this.segments = [];
        
        for (let i = 0; i < CONFIG.TRACK_LENGTH; i++) {
            const segmentIndex = i % track.segments.length;
            const s = track.segments[segmentIndex];
            this.segments.push(new RoadSegment(i, s.curve, s.hill));
        }
    },
    
    setupInput() {
        document.addEventListener('keydown', (e) => {
            // 初始化音频上下文
            AudioManager.init();
            AudioManager.resume();
            
            // P1 controls
            if (e.key === 'w' || e.key === 'W') this.inputP1.up = true;
            if (e.key === 's' || e.key === 'S') this.inputP1.down = true;
            if (e.key === 'a' || e.key === 'A') this.inputP1.left = true;
            if (e.key === 'd' || e.key === 'D') this.inputP1.right = true;
            if (e.key === ' ') { this.inputP1.nitro = true; e.preventDefault(); }
            
            // P2 controls
            if (e.key === 'ArrowUp') { this.inputP2.up = true; e.preventDefault(); }
            if (e.key === 'ArrowDown') { this.inputP2.down = true; e.preventDefault(); }
            if (e.key === 'ArrowLeft') { this.inputP2.left = true; e.preventDefault(); }
            if (e.key === 'ArrowRight') { this.inputP2.right = true; e.preventDefault(); }
            if (e.key === 'Control') this.inputP2.nitro = true;
            
            // Pause
            if (e.key === 'Escape') {
                if (GameData.state === GameState.PLAYING) {
                    this.pause();
                } else if (GameData.state === GameState.PAUSED) {
                    this.resume();
                }
            }
            
            // 音量快捷键
            if (e.key === 'm' || e.key === 'M') {
                const master = SettingsManager.currentSettings.masterVolume;
                SettingsManager.updateVolume('master', master > 0 ? 0 : 80);
                document.getElementById('masterVolume').value = SettingsManager.currentSettings.masterVolume;
            }
        });
        
        document.addEventListener('keyup', (e) => {
            if (e.key === 'w' || e.key === 'W') this.inputP1.up = false;
            if (e.key === 's' || e.key === 'S') this.inputP1.down = false;
            if (e.key === 'a' || e.key === 'A') this.inputP1.left = false;
            if (e.key === 'd' || e.key === 'D') this.inputP1.right = false;
            if (e.key === ' ') this.inputP1.nitro = false;
            
            if (e.key === 'ArrowUp') this.inputP2.up = false;
            if (e.key === 'ArrowDown') this.inputP2.down = false;
            if (e.key === 'ArrowLeft') this.inputP2.left = false;
            if (e.key === 'ArrowRight') this.inputP2.right = false;
            if (e.key === 'Control') this.inputP2.nitro = false;
        });
    },
    
    startRace() {
        AudioManager.init();
        AudioManager.resume();
        
        this.createSegments();
        
        // 创建玩家
        this.players = [];
        this.players.push(new Player(0, GameData.selectedVehicles[0]));
        if (GameData.isTwoPlayer) {
            this.players.push(new Player(1, GameData.selectedVehicles[1]));
        }
        
        // 创建AI对手
        this.aiOpponents = [];
        const usedVehicles = GameData.selectedVehicles;
        for (let i = 0; i < 3; i++) {
            let vehicleId = (i + 2) % VEHICLES.length;
            this.aiOpponents.push(new AIOpponent(i, vehicleId));
        }
        
        // 重置数据
        GameData.raceStartTime = 0;
        GameData.currentTime = 0;
        GameData.bestLapTime = Infinity;
        GameData.topSpeed = 0;
        GameData.nitroUsedCount = 0;
        GameData.lapTimes = [];
        GameData.state = GameState.COUNTDOWN;
        
        UIManager.hideAll();
        this.startCountdown();
    },
    
    startCountdown() {
        const countdownEl = document.getElementById('countdown');
        const numberEl = document.getElementById('countdownNumber');
        countdownEl.classList.remove('hidden');
        
        let count = 3;
        numberEl.textContent = count;
        numberEl.className = 'countdown-number';
        AudioManager.play('countdown');
        
        const interval = setInterval(() => {
            count--;
            if (count > 0) {
                numberEl.textContent = count;
                AudioManager.play('countdown');
            } else if (count === 0) {
                numberEl.textContent = 'GO!';
                numberEl.className = 'countdown-text';
                AudioManager.play('go');
            } else {
                clearInterval(interval);
                countdownEl.classList.add('hidden');
                GameData.state = GameState.PLAYING;
                GameData.raceStartTime = performance.now();
                UIManager.showHUD();
            }
        }, 1000);
    },
    
    pause() {
        GameData.state = GameState.PAUSED;
        document.getElementById('pauseMenu').classList.remove('hidden');
        AudioManager.stopEngine();
        AudioManager.stopWind();
    },
    
    resume() {
        GameData.state = GameState.PLAYING;
        document.getElementById('pauseMenu').classList.add('hidden');
    },
    
    restart() {
        document.getElementById('pauseMenu').classList.add('hidden');
        document.getElementById('resultsPanel').classList.add('hidden');
        this.startRace();
    },
    
    quit() {
        GameData.state = GameState.MENU;
        this.running = false;
        AudioManager.stopEngine();
        AudioManager.stopWind();
        UIManager.showMainMenu();
    },
    
    stop() {
        this.running = false;
        AudioManager.stopEngine();
        AudioManager.stopWind();
    },
    
    finishRace() {
        GameData.state = GameState.FINISHED;
        
        const player = this.players[0];
        const track = TRACKS[GameData.selectedTrack];
        
        // 计算排名
        const allRacers = [
            { type: 'player', index: 0, lap: player.lap, position: player.position }
        ];
        
        this.aiOpponents.forEach((ai, i) => {
            allRacers.push({ type: 'ai', index: i, lap: ai.lap, position: ai.position });
        });
        
        if (this.players[1]) {
            allRacers.push({ type: 'player', index: 1, lap: this.players[1].lap, position: this.players[1].position });
        }
        
        allRacers.sort((a, b) => {
            if (b.lap !== a.lap) return b.lap - a.lap;
            return b.position - a.position;
        });
        
        const playerRank = allRacers.findIndex(r => r.type === 'player' && r.index === 0) + 1;
        
        // 保存记录
        const totalTime = GameData.currentTime;
        const bestLapTime = player.bestLapTime === Infinity ? totalTime : player.bestLapTime;
        
        const isNewRecord = StorageManager.saveBestLap(track.id, bestLapTime, VEHICLES[GameData.selectedVehicles[0]].id);
        
        StorageManager.saveRecord(track.id, {
            totalTime: totalTime,
            bestLap: bestLapTime,
            vehicle: VEHICLES[GameData.selectedVehicles[0]].name,
            rank: playerRank
        });
        
        // 检查新纪录
        const recordCheck = StorageManager.isNewRecord(track.id, totalTime, bestLapTime);
        
        // 更新UI
        const titleEl = document.getElementById('resultsTitle');
        titleEl.textContent = playerRank === 1 ? '胜利!' : '比赛结束';
        titleEl.className = 'results-title ' + (playerRank === 1 ? 'winner' : 'loser');
        
        document.getElementById('resultsPosition').textContent = this.getOrdinal(playerRank);
        document.getElementById('resultTime').textContent = this.formatTime(totalTime);
        document.getElementById('resultBestLap').textContent = this.formatTime(bestLapTime);
        document.getElementById('resultTopSpeed').textContent = Math.round(player.topSpeed) + ' KM/H';
        
        document.getElementById('newTimeRecord').style.display = recordCheck.total ? 'inline' : 'none';
        document.getElementById('newLapRecord').style.display = recordCheck.lap ? 'inline' : 'none';
        
        UIManager.hideAll();
        document.getElementById('resultsPanel').classList.remove('hidden');
        
        AudioManager.play('finish');
    },
    
    getOrdinal(n) {
        const s = ['th', 'st', 'nd', 'rd'];
        const v = n % 100;
        return n + (s[(v - 20) % 10] || s[v] || s[0]);
    },
    
    formatTime(seconds) {
        const mins = Math.floor(seconds / 60);
        const secs = Math.floor(seconds % 60);
        const ms = Math.floor((seconds % 1) * 100);
        return `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}.${ms.toString().padStart(2, '0')}`;
    },
    
    toggleFullscreen() {
        if (!document.fullscreenElement) {
            document.documentElement.requestFullscreen().then(() => {
                document.body.classList.add('fullscreen-mode');
            }).catch(err => {
                console.warn('无法进入全屏模式:', err);
            });
        } else {
            document.exitFullscreen().then(() => {
                document.body.classList.remove('fullscreen-mode');
            });
        }
    },
    
    gameLoop(timestamp) {
        const dt = Math.min((timestamp - this.lastTime) / 1000, 0.1);
        this.lastTime = timestamp;
        
        // FPS计算
        this.frameCount++;
        this.fpsTime += dt;
        if (this.fpsTime >= 1) {
            this.currentFPS = this.frameCount;
            this.frameCount = 0;
            this.fpsTime = 0;
            
            document.getElementById('fpsValue').textContent = this.currentFPS;
            document.getElementById('frameTime').textContent = (1000 / this.currentFPS).toFixed(1) + 'ms';
        }
        
        if (GameData.state === GameState.PLAYING) {
            GameData.currentTime = (performance.now() - GameData.raceStartTime) / 1000;
            this.update(dt);
        }
        
        this.render();
        
        requestAnimationFrame((t) => this.gameLoop(t));
    },
    
    update(dt) {
        // 更新玩家
        this.players[0].update(dt, this.inputP1, this.segments);
        if (this.players[1]) {
            this.players[1].update(dt, this.inputP2, this.segments);
        }
        
        // 更新AI
        this.aiOpponents.forEach(ai => ai.update(dt, this.segments, this.players[0].position));
        
        // 更新相机
        this.cameraX = this.players[0].x * CONFIG.ROAD_WIDTH;
        this.cameraZ = this.players[0].position;
        
        // 音效
        const speedRatio = this.players[0].speed / this.players[0].vehicle.config.maxSpeed;
        AudioManager.playEngine(speedRatio);
        AudioManager.playWind(speedRatio);
        
        // 氮气加速粒子
        if (this.players[0].nitroActive) {
            this.nitroOverlay.style.opacity = '0.3';
            this.emitNitroParticles();
        } else {
            this.nitroOverlay.style.opacity = '0';
        }
        
        // 轮胎烟尘
        if (Math.abs(this.inputP1.left) || Math.abs(this.inputP1.right)) {
            if (this.players[0].speed > 100) {
                this.emitTireSmoke();
            }
        }
        
        // 更新粒子
        ParticleSystem.update(dt);
        
        // 屏幕震动
        if (SettingsManager.getSetting('screenShake')) {
            this.screenShake.x = (Math.random() - 0.5) * this.screenShake.intensity;
            this.screenShake.y = (Math.random() - 0.5) * this.screenShake.intensity;
            this.screenShake.intensity *= 0.9;
        }
        
        // 检查比赛结束
        if (this.players[0].finished) {
            this.finishRace();
        }
        
        // 更新HUD
        this.updateHUD();
    },
    
    emitNitroParticles() {
        const cx = this.displayWidth / 2;
        const cy = this.displayHeight - 150;
        
        ParticleSystem.emit(cx, cy, {
            count: 2,
            vx: (Math.random() - 0.5) * 5,
            vy: Math.random() * 3 + 2,
            size: Math.random() * 4 + 2,
            color: Math.random() > 0.5 ? '#00f5ff' : '#b347ea',
            decay: 0.03,
            life: 0.5,
            spreadX: 30
        });
    },
    
    emitTireSmoke() {
        const cx = this.displayWidth / 2;
        const cy = this.displayHeight - 120;
        
        ParticleSystem.emit(cx + (this.inputP1.left ? 30 : -30), cy, {
            count: 1,
            vx: (Math.random() - 0.5) * 2,
            vy: -Math.random() * 2,
            size: Math.random() * 6 + 3,
            color: 'rgba(255, 255, 255, 0.5)',
            decay: 0.02,
            life: 0.8,
            spreadX: 20
        });
    },
    
    updateHUD() {
        const player = this.players[0];
        
        // 计算排名
        const allRacers = [
            { type: 'player', index: 0, lap: player.lap, position: player.position }
        ];
        this.aiOpponents.forEach((ai, i) => {
            allRacers.push({ type: 'ai', index: i, lap: ai.lap, position: ai.position });
        });
        if (this.players[1]) {
            allRacers.push({ type: 'player', index: 1, lap: this.players[1].lap, position: this.players[1].position });
        }
        allRacers.sort((a, b) => {
            if (b.lap !== a.lap) return b.lap - a.lap;
            return b.position - a.position;
        });
        
        const playerRank = allRacers.findIndex(r => r.type === 'player' && r.index === 0) + 1;
        
        document.getElementById('positionValue').textContent = playerRank;
        document.getElementById('currentLap').textContent = Math.min(player.lap, CONFIG.TOTAL_LAPS);
        document.getElementById('totalLaps').textContent = CONFIG.TOTAL_LAPS;
        document.getElementById('raceTimer').textContent = this.formatTime(GameData.currentTime);
        document.getElementById('speedValue').textContent = player.getDisplaySpeed();
        document.getElementById('gearValue').textContent = player.getGear();
        document.getElementById('nitroFill').style.width = player.nitro + '%';
        document.getElementById('nitroPercent').textContent = Math.round(player.nitro) + '%';
        
        // P2 HUD
        if (this.players[1]) {
            document.getElementById('positionValueP2').textContent = 
                allRacers.findIndex(r => r.type === 'player' && r.index === 1) + 1;
            document.getElementById('currentLapP2').textContent = Math.min(this.players[1].lap, CONFIG.TOTAL_LAPS);
        }
    },
    
    render() {
        const ctx = this.ctx;
        const width = this.displayWidth;
        const height = this.displayHeight;
        
        // 清空画布
        ctx.fillStyle = CONFIG.COLORS.bgDark;
        ctx.fillRect(0, 0, width, height);
        
        if (GameData.state === GameState.MENU || 
            GameData.state === GameState.VEHICLE_SELECT || 
            GameData.state === GameState.TRACK_SELECT) {
            this.renderMenuBackground(ctx, width, height);
            return;
        }
        
        // 应用屏幕震动
        ctx.save();
        if (SettingsManager.getSetting('screenShake')) {
            ctx.translate(this.screenShake.x, this.screenShake.y);
        }
        
        // 渲染天空
        this.renderSky(ctx, width, height);
        
        // 渲染道路
        this.renderRoad(ctx, width, height);
        
        // 渲染车辆
        this.renderVehicles(ctx, width, height);
        
        ctx.restore();
        
        // 渲染粒子
        this.vfxCtx.clearRect(0, 0, width, height);
        ParticleSystem.draw(this.vfxCtx);
    },
    
    renderMenuBackground(ctx, width, height) {
        // 动态背景
        const time = performance.now() * 0.001;
        
        // 渐变背景
        const gradient = ctx.createLinearGradient(0, 0, 0, height);
        gradient.addColorStop(0, '#0a0a1a');
        gradient.addColorStop(1, '#1a0a2e');
        ctx.fillStyle = gradient;
        ctx.fillRect(0, 0, width, height);
        
        // 动态网格
        ctx.strokeStyle = 'rgba(0, 245, 255, 0.1)';
        ctx.lineWidth = 1;
        
        const gridSize = 50;
        const offsetY = (time * 30) % gridSize;
        
        for (let y = -gridSize + offsetY; y < height + gridSize; y += gridSize) {
            ctx.beginPath();
            ctx.moveTo(0, y);
            ctx.lineTo(width, y);
            ctx.stroke();
        }
        
        for (let x = 0; x < width; x += gridSize) {
            ctx.beginPath();
            ctx.moveTo(x, 0);
            ctx.lineTo(x, height);
            ctx.stroke();
        }
    },
    
    renderSky(ctx, width, height) {
        const gradient = ctx.createLinearGradient(0, 0, 0, height * 0.5);
        gradient.addColorStop(0, CONFIG.COLORS.sky.top);
        gradient.addColorStop(1, CONFIG.COLORS.sky.bottom);
        ctx.fillStyle = gradient;
        ctx.fillRect(0, 0, width, height * 0.5);
        
        // 星星
        ctx.fillStyle = '#ffffff';
        for (let i = 0; i < 50; i++) {
            const x = (i * 73 + performance.now() * 0.01) % width;
            const y = (i * 37) % (height * 0.4);
            const size = (i % 3) + 1;
            ctx.globalAlpha = 0.3 + Math.sin(performance.now() * 0.002 + i) * 0.3;
            ctx.fillRect(x, y, size, size);
        }
        ctx.globalAlpha = 1;
    },
    
    renderRoad(ctx, width, height) {
        const baseSegment = Math.floor(this.cameraZ / CONFIG.SEGMENT_LENGTH);
        const basePercent = (this.cameraZ % CONFIG.SEGMENT_LENGTH) / CONFIG.SEGMENT_LENGTH;
        
        let maxy = height;
        
        const trackLength = CONFIG.TRACK_LENGTH * CONFIG.SEGMENT_LENGTH;
        
        // 从远到近渲染路段
        for (let n = CONFIG.DRAW_DISTANCE - 1; n >= 0; n--) {
            const segmentIndex = (baseSegment + n) % CONFIG.TRACK_LENGTH;
            const segment = this.segments[segmentIndex];
            if (!segment) continue;
            
            const looped = (baseSegment + n) >= CONFIG.TRACK_LENGTH;
            const segmentZ = (baseSegment + n) * CONFIG.SEGMENT_LENGTH - (looped ? trackLength : 0);
            
            // 投影计算
            const scale = CONFIG.CAMERA_DEPTH / (segmentZ - this.cameraZ + CONFIG.SEGMENT_LENGTH);
            if (scale <= 0) continue;
            
            const x = width / 2 + (this.cameraX * scale);
            const y = height / 2 + (CONFIG.CAMERA_HEIGHT * scale);
            const w = CONFIG.ROAD_WIDTH * scale;
            
            segment.p1.screen.x = x;
            segment.p1.screen.y = y;
            segment.p1.screen.w = w;
            
            if (n < CONFIG.DRAW_DISTANCE - 1) {
                const prevSegment = this.segments[(baseSegment + n + 1) % CONFIG.TRACK_LENGTH];
                if (prevSegment) {
                    // 草地
                    const grassColor = segment.color === 'dark' ? CONFIG.COLORS.grass.dark : CONFIG.COLORS.grass.light;
                    ctx.fillStyle = grassColor;
                    ctx.fillRect(0, prevSegment.p1.screen.y, width, segment.p1.screen.y - prevSegment.p1.screen.y + 1);
                    
                    // 道路
                    const roadColor = segment.color === 'dark' ? CONFIG.COLORS.asphalt.dark : CONFIG.COLORS.asphalt.light;
                    ctx.fillStyle = roadColor;
                    this.drawTrapezoid(ctx, 
                        segment.p1.screen.x, segment.p1.screen.y, segment.p1.screen.w,
                        prevSegment.p1.screen.x, prevSegment.p1.screen.y, prevSegment.p1.screen.w
                    );
                    
                    // 路肩
                    const rumbleColor = segment.color === 'dark' ? CONFIG.COLORS.rumble.dark : CONFIG.COLORS.rumble.light;
                    ctx.fillStyle = rumbleColor;
                    const rumbleW = w * 0.1;
                    const prevRumbleW = prevSegment.p1.screen.w * 0.1;
                    
                    // 左路肩
                    this.drawTrapezoid(ctx,
                        segment.p1.screen.x - segment.p1.screen.w / 2 - rumbleW, segment.p1.screen.y, rumbleW,
                        prevSegment.p1.screen.x - prevSegment.p1.screen.w / 2 - prevRumbleW, prevSegment.p1.screen.y, prevRumbleW
                    );
                    
                    // 右路肩
                    this.drawTrapezoid(ctx,
                        segment.p1.screen.x + segment.p1.screen.w / 2, segment.p1.screen.y, rumbleW,
                        prevSegment.p1.screen.x + prevSegment.p1.screen.w / 2, prevSegment.p1.screen.y, prevRumbleW
                    );
                    
                    // 车道线
                    if (segment.color === 'dark') {
                        ctx.fillStyle = CONFIG.COLORS.lane;
                        const laneW = w * 0.02;
                        const prevLaneW = prevSegment.p1.screen.w * 0.02;
                        
                        for (let lane = -1; lane <= 1; lane++) {
                            if (lane !== 0) {
                                this.drawTrapezoid(ctx,
                                    segment.p1.screen.x + lane * segment.p1.screen.w / 3 - laneW / 2, segment.p1.screen.y, laneW,
                                    prevSegment.p1.screen.x + lane * prevSegment.p1.screen.w / 3 - prevLaneW / 2, prevSegment.p1.screen.y, prevLaneW
                                );
                            }
                        }
                    }
                }
            }
            
            if (segment.p1.screen.y >= maxy) continue;
            maxy = segment.p1.screen.y;
        }
    },
    
    drawTrapezoid(ctx, x1, y1, w1, x2, y2, w2) {
        ctx.beginPath();
        ctx.moveTo(x1 - w1 / 2, y1);
        ctx.lineTo(x1 + w1 / 2, y1);
        ctx.lineTo(x2 + w2 / 2, y2);
        ctx.lineTo(x2 - w2 / 2, y2);
        ctx.closePath();
        ctx.fill();
    },
    
    renderVehicles(ctx, width, height) {
        const player = this.players[0];
        const playerVehicle = player.vehicle;
        
        // 渲染玩家车辆
        const carWidth = 80;
        const carHeight = 50;
        const carX = width / 2 + player.x * 100;
        const carY = height - 100;
        
        // 车身
        ctx.fillStyle = playerVehicle.color;
        ctx.shadowColor = playerVehicle.color;
        ctx.shadowBlur = 20;
        
        ctx.beginPath();
        ctx.moveTo(carX - carWidth / 2, carY);
        ctx.lineTo(carX - carWidth / 3, carY - carHeight);
        ctx.lineTo(carX + carWidth / 3, carY - carHeight);
        ctx.lineTo(carX + carWidth / 2, carY);
        ctx.closePath();
        ctx.fill();
        
        ctx.shadowBlur = 0;
        
        // 车窗
        ctx.fillStyle = 'rgba(0, 0, 0, 0.5)';
        ctx.beginPath();
        ctx.moveTo(carX - carWidth / 4, carY - carHeight * 0.4);
        ctx.lineTo(carX - carWidth / 5, carY - carHeight * 0.85);
        ctx.lineTo(carX + carWidth / 5, carY - carHeight * 0.85);
        ctx.lineTo(carX + carWidth / 4, carY - carHeight * 0.4);
        ctx.closePath();
        ctx.fill();
        
        // 氮气火焰
        if (player.nitroActive) {
            ctx.fillStyle = '#00f5ff';
            ctx.shadowColor = '#00f5ff';
            ctx.shadowBlur = 30;
            
            const flameLength = 30 + Math.random() * 20;
            ctx.beginPath();
            ctx.moveTo(carX - 15, carY);
            ctx.lineTo(carX, carY + flameLength);
            ctx.lineTo(carX + 15, carY);
            ctx.closePath();
            ctx.fill();
            
            ctx.shadowBlur = 0;
        }
        
        // 渲染AI车辆
        this.aiOpponents.forEach((ai, i) => {
            const relativeZ = ai.position - player.position;
            if (relativeZ < 0 || relativeZ > CONFIG.DRAW_DISTANCE * CONFIG.SEGMENT_LENGTH) return;
            
            const scale = CONFIG.CAMERA_DEPTH / (relativeZ + CONFIG.SEGMENT_LENGTH);
            if (scale <= 0) return;
            
            const aiX = width / 2 + ai.x * CONFIG.ROAD_WIDTH * scale;
            const aiY = height / 2 - CONFIG.CAMERA_HEIGHT * scale;
            const aiW = 60 * scale;
            const aiH = 35 * scale;
            
            if (aiY < 0 || aiY > height) return;
            
            ctx.fillStyle = ai.vehicle.color;
            ctx.shadowColor = ai.vehicle.color;
            ctx.shadowBlur = 10 * scale;
            
            ctx.beginPath();
            ctx.moveTo(aiX - aiW / 2, aiY);
            ctx.lineTo(aiX - aiW / 3, aiY - aiH);
            ctx.lineTo(aiX + aiW / 3, aiY - aiH);
            ctx.lineTo(aiX + aiW / 2, aiY);
            ctx.closePath();
            ctx.fill();
            
            ctx.shadowBlur = 0;
        });
        
        // 渲染P2车辆
        if (this.players[1]) {
            const p2 = this.players[1];
            const relativeZ = p2.position - player.position;
            
            if (Math.abs(relativeZ) < CONFIG.DRAW_DISTANCE * CONFIG.SEGMENT_LENGTH) {
                const scale = CONFIG.CAMERA_DEPTH / (Math.abs(relativeZ) + CONFIG.SEGMENT_LENGTH);
                const p2X = width / 2 + p2.x * CONFIG.ROAD_WIDTH * scale;
                let p2Y;
                
                if (relativeZ >= 0) {
                    p2Y = height / 2 - CONFIG.CAMERA_HEIGHT * scale;
                } else {
                    // 在玩家后面，显示在后视镜区域或简化显示
                    return;
                }
                
                const p2W = 60 * scale;
                const p2H = 35 * scale;
                
                ctx.fillStyle = p2.vehicle.color;
                ctx.shadowColor = p2.vehicle.color;
                ctx.shadowBlur = 15 * scale;
                
                ctx.beginPath();
                ctx.moveTo(p2X - p2W / 2, p2Y);
                ctx.lineTo(p2X - p2W / 3, p2Y - p2H);
                ctx.lineTo(p2X + p2W / 3, p2Y - p2H);
                ctx.lineTo(p2X + p2W / 2, p2Y);
                ctx.closePath();
                ctx.fill();
                
                ctx.shadowBlur = 0;
            }
        }
    }
};

## 主要功能实现说明

### 1. 本地存储系统

- **最佳单圈记录**: 按赛道+车辆组合存储
- **新纪录检测**: 完赛时自动检测并提示

### 2. Web Audio API 音效系统
- **引擎声音**: 根据速度动态调整频率
- **氮气音效**: 加速时的反馈音
- **UI点击音**: 菜单操作反馈
- **倒计时音效**: 3-2-1-GO 提示音
- **音量控制**: 主音量/音效/音乐三通道独立控制

- **音频设置**: 三通道音量滑块控制
- **画质设置**: 低/中/高/超高四档

- **对象池**: 粒子系统复用
- **FPS监控**: 实时显示帧率和帧时间
