    class MathUtils {
        static lerp(a, b, t) {
            return a + (b - a) * t;
        }
        
        static clamp(value, min, max) {
            return Math.max(min, Math.min(max, value));
        }
        
        static randomRange(min, max) {
            return Math.random() * (max - min) + min;
        }
        
        static easeOutCubic(t) {
            return 1 - Math.pow(1 - t, 3);
        }
        
        static exponentialFog(distance, density) {
            return 1 / Math.pow(Math.E, (distance * distance * density));
        }
        
        static project(p, cameraX, cameraY, cameraZ, cameraDepth, width, height, roadWidth) {
            const translatedZ = p.z - cameraZ;
            p.camera = {
                x: p.x - cameraX,
                y: p.y - cameraY,
                z: translatedZ
            };
            
            if (translatedZ <= 0) {
                p.screen = { x: 0, y: 0, w: 0, scale: 0 };
                return p;
            }
            
            const scale = cameraDepth / translatedZ;
            p.screen = {
                x: Math.round(width / 2 + scale * p.camera.x * width / 2),
                y: Math.round(height / 2 - scale * p.camera.y * height / 2),
                w: Math.round(scale * roadWidth * width / 2),
                scale: scale
            };
            
            return p;
        }
    }

    class Particle {
        constructor(x, y, vx, vy, life, size, color, type = 'default') {
            this.x = x;
            this.y = y;
            this.vx = vx;
            this.vy = vy;
            this.life = life;
            this.maxLife = life;
            this.size = size;
            this.color = color;
            this.type = type;
            this.alpha = 1;
            this.rotation = Math.random() * Math.PI * 2;
            this.rotationSpeed = (Math.random() - 0.5) * 0.2;
        }
        
        update(dt) {
            this.x += this.vx * dt;
            this.y += this.vy * dt;
            this.life -= dt;
            this.alpha = Math.max(0, this.life / this.maxLife);
            this.rotation += this.rotationSpeed * dt;
            
            if (this.type === 'smoke') {
                this.size *= 1.02;
                this.vy -= 0.5 * dt;
            } else if (this.type === 'spark') {
                this.vy += 0.3 * dt;
            } else if (this.type === 'speedline') {
                this.size *= 0.98;
            }
            
            return this.life > 0;
        }
    }

    class ParticleSystem {
        constructor() {
            this.particles = [];
            this.maxParticles = 500;
        }
        
        emit(x, y, count, config) {
            for (let i = 0; i < count && this.particles.length < this.maxParticles; i++) {
                const angle = config.angle + (Math.random() - 0.5) * config.spread;
                const speed = config.speed * (0.5 + Math.random() * 0.5);
                const particle = new Particle(
                    x + (Math.random() - 0.5) * (config.offsetX || 0),
                    y + (Math.random() - 0.5) * (config.offsetY || 0),
                    Math.cos(angle) * speed,
                    Math.sin(angle) * speed,
                    config.life * (0.5 + Math.random() * 0.5),
                    config.size * (0.5 + Math.random() * 0.5),
                    config.color,
                    config.type
                );
                this.particles.push(particle);
            }
        }
        
        emitSpeedLines(x, y, count, speedRatio) {
            const config = {
                angle: Math.PI,
                spread: 0.3,
                speed: 15 + speedRatio * 25,
                life: 0.3 + speedRatio * 0.3,
                size: 50 + speedRatio * 100,
                color: `rgba(0, 245, 255, ${0.3 + speedRatio * 0.4})`,
                type: 'speedline',
                offsetX: 100,
                offsetY: 50
            };
            this.emit(x, y, count, config);
        }
        
        emitTireSmoke(x, y, intensity) {
            const config = {
                angle: -Math.PI / 2,
                spread: Math.PI / 3,
                speed: 2 + intensity * 3,
                life: 0.8 + Math.random() * 0.4,
                size: 8 + intensity * 12,
                color: `rgba(180, 180, 180, ${0.3 + intensity * 0.3})`,
                type: 'smoke',
                offsetX: 30,
                offsetY: 10
            };
            this.emit(x, y, Math.floor(2 + intensity * 4), config);
        }
        
        emitSparks(x, y, intensity) {
            const config = {
                angle: -Math.PI / 2,
                spread: Math.PI,
                speed: 8 + intensity * 12,
                life: 0.4 + Math.random() * 0.3,
                size: 2 + intensity * 3,
                color: CONFIG.COLORS.neonOrange,
                type: 'spark',
                offsetX: 20,
                offsetY: 10
            };
            this.emit(x, y, Math.floor(3 + intensity * 5), config);
        }
        
        emitNitroFlame(x, y) {
            const colors = [CONFIG.COLORS.neonCyan, CONFIG.COLORS.neonPurple, CONFIG.COLORS.neonPink];
            const config = {
                angle: Math.PI,
                spread: 0.5,
                speed: 10 + Math.random() * 5,
                life: 0.2 + Math.random() * 0.2,
                size: 15 + Math.random() * 15,
                color: colors[Math.floor(Math.random() * colors.length)],
                type: 'flame',
                offsetX: 40,
                offsetY: 15
            };
            this.emit(x, y, 4, config);
        }
        
        update(dt) {
            this.particles = this.particles.filter(p => p.update(dt));
        }
        
        render(ctx, camera) {
            ctx.save();
            
            for (const p of this.particles) {
                ctx.globalAlpha = p.alpha;
                
                if (p.type === 'speedline') {
                    ctx.strokeStyle = p.color;
                    ctx.lineWidth = 2;
                    ctx.beginPath();
                    const lineLength = p.size * p.alpha;
                    ctx.moveTo(p.x, p.y);
                    ctx.lineTo(p.x - Math.cos(p.rotation) * lineLength, p.y - Math.sin(p.rotation) * lineLength);
                    ctx.stroke();
                } else if (p.type === 'smoke') {
                    const gradient = ctx.createRadialGradient(p.x, p.y, 0, p.x, p.y, p.size);
                    gradient.addColorStop(0, `rgba(150, 150, 150, ${p.alpha * 0.5})`);
                    gradient.addColorStop(1, `rgba(100, 100, 100, 0)`);
                    ctx.fillStyle = gradient;
                    ctx.beginPath();
                    ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
                    ctx.fill();
                } else if (p.type === 'spark') {
                    ctx.fillStyle = p.color;
                    ctx.shadowColor = p.color;
                    ctx.shadowBlur = 10;
                    ctx.beginPath();
                    ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
                    ctx.fill();
                    ctx.shadowBlur = 0;
                } else if (p.type === 'flame') {
                    const gradient = ctx.createRadialGradient(p.x, p.y, 0, p.x, p.y, p.size);
                    gradient.addColorStop(0, p.color);
                    gradient.addColorStop(0.5, `rgba(255, 255, 255, ${p.alpha * 0.5})`);
                    gradient.addColorStop(1, `rgba(0, 245, 255, 0)`);
                    ctx.fillStyle = gradient;
                    ctx.beginPath();
                    ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
                    ctx.fill();
                } else {
                    ctx.fillStyle = p.color;
                    ctx.beginPath();
                    ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
                    ctx.fill();
                }
            }
            
            ctx.restore();
        }
        
        get count() {
            return this.particles.length;
        }
    }

    class CameraShake {
        constructor() {
            this.shakeX = 0;
            this.shakeY = 0;
            this.shakeRotation = 0;
            this.trauma = 0;
            this.maxOffset = 30;
            this.maxRotation = 0.1;
        }
        
        addTrauma(amount) {
            this.trauma = Math.min(1, this.trauma + amount);
        }
        
        shake(intensity) {
            this.addTrauma(intensity);
        }
        
        update(dt) {
            if (this.trauma > 0) {
                const shake = Math.pow(this.trauma, 2);
                const seed = performance.now() * 0.001;
                
                this.shakeX = (Math.sin(seed * 50) * this.maxOffset * shake);
                this.shakeY = (Math.cos(seed * 47) * this.maxOffset * shake);
                this.shakeRotation = (Math.sin(seed * 53) * this.maxRotation * shake);
                
                this.trauma = Math.max(0, this.trauma - dt * 2);
            } else {
                this.shakeX = 0;
                this.shakeY = 0;
                this.shakeRotation = 0;
            }
        }
        
        apply(ctx, centerX, centerY) {
            ctx.translate(centerX + this.shakeX, centerY + this.shakeY);
            ctx.rotate(this.shakeRotation);
        }
    }

            // 速度线强度
            this.speedLinesIntensity = MathUtils.lerp(
                this.speedLinesIntensity,
                speedRatio > 0.5 ? (speedRatio - 0.5) * 2 : 0,
                dt * 5
            );
            
            // 动态模糊
            this.motionBlurAmount = MathUtils.lerp(
                this.motionBlurAmount,
                speedRatio * 0.5,
                dt * 3
            );
            
            // FOV拉伸
            this.fovStretch = MathUtils.lerp(
                this.fovStretch,
                speedRatio * 0.3,
                dt * 4
            );
            
            // 氮气效果
            this.nitroActive = player.nitroActive;
            
            // 漂移指示器
            if (player.drifting) {
                this.driftIndicator.classList.add('active');
            } else {
                this.driftIndicator.classList.remove('active');
            }
            
            // 边缘暗角
            this.vignetteIntensity = MathUtils.lerp(
                this.vignetteIntensity,
                speedRatio * 0.6 + (player.drifting ? 0.2 : 0),
                dt * 3
            );
            
            // 发射粒子
            this.emitParticles(player, speedRatio);
            
            // 更新氮气叠加层
            if (this.nitroActive) {
                this.nitroOverlay.style.opacity = '0.4';
                this.nitroOverlay.style.boxShadow = 'inset 0 0 150px rgba(0, 245, 255, 0.8)';
            } else {
                this.nitroOverlay.style.opacity = '0';
            }
        }
        
        emitParticles(player, speedRatio) {
            const canvasWidth = this.canvas.width;
            const canvasHeight = this.canvas.height;
            
            // 速度线
            if (this.speedLinesIntensity > 0.3 && Math.random() < this.speedLinesIntensity) {
                const side = Math.random() > 0.5 ? 1 : -1;
                this.particleSystem.emitSpeedLines(
                    canvasWidth / 2 + side * (canvasWidth * 0.3),
                    canvasHeight * 0.3 + Math.random() * canvasHeight * 0.4,
                    Math.floor(1 + this.speedLinesIntensity * 2),
                    speedRatio
                );
            }
            
            // 轮胎烟尘
            if (player.drifting || player.handbraking) {
                const intensity = player.drifting ? 0.8 : 0.5;
                this.particleSystem.emitTireSmoke(
                    canvasWidth / 2 - 50,
                    canvasHeight * 0.75,
                    intensity
                );
                this.particleSystem.emitTireSmoke(
                    canvasWidth / 2 + 50,
                    canvasHeight * 0.75,
                    intensity
                );
            }
            
            // 火花（高速过弯）
            if (player.drifting && speedRatio > 0.6) {
                this.particleSystem.emitSparks(
                    canvasWidth / 2 + (Math.random() > 0.5 ? 60 : -60),
                    canvasHeight * 0.7,
                    (speedRatio - 0.6) * 2
                );
            }
            
            // 氮气火焰
            if (this.nitroActive) {
                this.particleSystem.emitNitroFlame(canvasWidth / 2 - 30, canvasHeight * 0.72);
                this.particleSystem.emitNitroFlame(canvasWidth / 2 + 30, canvasHeight * 0.72);
            }
        }
        
        render(player) {
            this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);
            
            // 应用相机震动
            this.ctx.save();
            this.cameraShake.apply(this.ctx, this.canvas.width / 2, this.canvas.height / 2);
            
            // 渲染粒子
            this.particleSystem.render(this.ctx);
            
            this.ctx.restore();
            
            // 渲染后处理效果
            this.renderPostEffects();
        }
        
        renderPostEffects() {
            const ctx = this.ctx;
            const w = this.canvas.width;
            const h = this.canvas.height;
            
            // 边缘暗角
            if (this.vignetteIntensity > 0.1) {
                const gradient = ctx.createRadialGradient(
                    w / 2, h / 2, h * 0.3,
                    w / 2, h / 2, h * 0.8
                );
                gradient.addColorStop(0, 'transparent');
                gradient.addColorStop(1, `rgba(0, 0, 0, ${this.vignetteIntensity * 0.7})`);
                ctx.fillStyle = gradient;
                ctx.fillRect(0, 0, w, h);
            }
            
            // 速度边缘发光
            if (this.speedLinesIntensity > 0.5) {
                const edgeGradient = ctx.createLinearGradient(0, 0, w, 0);
                edgeGradient.addColorStop(0, `rgba(0, 245, 255, ${this.speedLinesIntensity * 0.2})`);
                edgeGradient.addColorStop(0.1, 'transparent');
                edgeGradient.addColorStop(0.9, 'transparent');
                edgeGradient.addColorStop(1, `rgba(0, 245, 255, ${this.speedLinesIntensity * 0.2})`);
                ctx.fillStyle = edgeGradient;
                ctx.fillRect(0, 0, w, h);
            }
            
            // 动态模糊条纹
            if (this.motionBlurAmount > 0.2) {
                ctx.globalAlpha = this.motionBlurAmount * 0.3;
                for (let i = 0; i < 5; i++) {
                    const y = h * 0.2 + Math.random() * h * 0.6;
                    ctx.fillStyle = 'rgba(255, 255, 255, 0.1)';
                    ctx.fillRect(0, y, w, 1);
                }
                ctx.globalAlpha = 1;
            }
        }
        
        addCollisionShake(intensity) {
            this.cameraShake.shake(intensity);
        }
        
        get particleCount() {
            return this.particleSystem.count;
        }
    }

    class HUDManager {
        constructor() {
            this.speedValue = document.getElementById('speedValue');
            this.speedUnit = document.getElementById('speedUnit');
            this.gearIndicator = document.getElementById('gearIndicator');
            this.tachBar = document.getElementById('tachBar');
            this.nitroFill = document.getElementById('nitroFill');
            this.speedArc = document.getElementById('speedArc');
            
            this.displaySpeed = 0;
            this.displayNitro = 100;
            
            this.initSpeedometer();
        }
        
        initSpeedometer() {
            const centerX = 90;
            const centerY = 95;
            const radius = 75;
            const startAngle = Math.PI * 0.8;
            const endAngle = Math.PI * 2.2;
            
            const x1 = centerX + Math.cos(startAngle) * radius;
            const y1 = centerY + Math.sin(startAngle) * radius;
            const x2 = centerX + Math.cos(endAngle) * radius;
            const y2 = centerY + Math.sin(endAngle) * radius;
            
            const arcPath = `M ${x1} ${y1} A ${radius} ${radius} 0 1 1 ${x2} ${y2}`;
            this.speedArcBg = document.getElementById('speedArcBg');
            if (this.speedArcBg) {
                this.speedArcBg.setAttribute('d', arcPath);
            }
            
            this.arcStartAngle = startAngle;
            this.arcEndAngle = endAngle;
            this.arcRadius = radius;
            this.arcCenterX = centerX;
            this.arcCenterY = centerY;
        }
        
        update(player, dt) {
            const targetSpeed = Math.abs(player.speed);
            this.displaySpeed = MathUtils.lerp(this.displaySpeed, targetSpeed, dt * 8);
            
            const displaySpeedInt = Math.round(this.displaySpeed);
            this.speedValue.innerHTML = `${displaySpeedInt}<span id="speedUnit">KM/H</span>`;
            
            // 更新速度弧
            this.updateSpeedArc(targetSpeed / PLAYER_CONFIG.maxSpeed);
            
            // 档位指示
            const gear = this.getCurrentGear(targetSpeed);
            this.gearIndicator.textContent = gear;
            this.gearIndicator.style.color = player.nitroActive ? CONFIG.COLORS.neonCyan : CONFIG.COLORS.neonYellow;
            this.gearIndicator.style.textShadow = `0 0 15px ${player.nitroActive ? CONFIG.COLORS.neonCyan : CONFIG.COLORS.neonYellow}`;
            
            // 转速表（基于当前档位进度）
            const rpm = this.calculateRPM(targetSpeed);
            this.tachBar.style.width = `${rpm * 100}%`;
            
            // 氮气条
            this.displayNitro = MathUtils.lerp(this.displayNitro, player.nitro, dt * 5);
            this.nitroFill.style.width = `${this.displayNitro}%`;
            
            if (player.nitroActive) {
                this.nitroFill.style.background = 'linear-gradient(90deg, #00f5ff, #b347ea, #ff2d75)';
            } else {
                this.nitroFill.style.background = 'linear-gradient(90deg, #00f5ff, #b347ea)';
            }
            
            // 速度数字颜色
            if (targetSpeed > 280) {
                this.speedValue.style.color = CONFIG.COLORS.neonPink;
                this.speedValue.style.textShadow = `0 0 30px ${CONFIG.COLORS.neonPink}`;
            } else if (targetSpeed > 200) {
                this.speedValue.style.color = CONFIG.COLORS.neonOrange;
                this.speedValue.style.textShadow = `0 0 25px ${CONFIG.COLORS.neonOrange}`;
            } else {
                this.speedValue.style.color = CONFIG.COLORS.neonCyan;
                this.speedValue.style.textShadow = `0 0 20px ${CONFIG.COLORS.neonCyan}`;
            }
        }
        
        updateSpeedArc(ratio) {
            const clampedRatio = MathUtils.clamp(ratio, 0, 1);
            const currentAngle = this.arcStartAngle + (this.arcEndAngle - this.arcStartAngle) * clampedRatio;
            
            const x = this.arcCenterX + Math.cos(currentAngle) * this.arcRadius;
            const y = this.arcCenterY + Math.sin(currentAngle) * this.arcRadius;
            
            const startX = this.arcCenterX + Math.cos(this.arcStartAngle) * this.arcRadius;
            const startY = this.arcCenterY + Math.sin(this.arcStartAngle) * this.arcRadius;
            
            const largeArc = clampedRatio > 0.5 ? 1 : 0;
            const arcPath = `M ${startX} ${startY} A ${this.arcRadius} ${this.arcRadius} 0 ${largeArc} 1 ${x} ${y}`;
            
            if (this.speedArc) {
                this.speedArc.setAttribute('d', arcPath);
            }
        }
        
        getCurrentGear(speed) {
            if (speed < 5) return 'N';
            for (let i = PLAYER_CONFIG.gears.length - 1; i >= 0; i--) {
                if (speed >= PLAYER_CONFIG.gears[i].minSpeed) {
                    return PLAYER_CONFIG.gears[i].name;
                }
            }
            return '1';
        }
        
        calculateRPM(speed) {
            for (let i = PLAYER_CONFIG.gears.length - 1; i >= 0; i--) {
                const gear = PLAYER_CONFIG.gears[i];
                if (speed >= gear.minSpeed) {
                    const range = gear.maxSpeed - gear.minSpeed;
                    const position = speed - gear.minSpeed;
                    return Math.min(1, position / range);
                }
            }
            return speed / 60;
        }
    }

    class PerformanceMonitor {
        constructor() {
            this.fps = 0;
            this.frameTime = 0;
            this.drawCalls = 0;
            
            this.frameCount = 0;
            this.lastTime = performance.now();
            this.fpsUpdateTime = 0;
            this.frameTimeHistory = [];
            
            this.elements = {
                fps: document.getElementById('fpsValue'),
                frameTime: document.getElementById('frameTimeValue'),
                particles: document.getElementById('particlesValue'),
                state: document.getElementById('stateValue')
            };
        }
        
        beginFrame() {
            this.frameStartTime = performance.now();
            this.drawCalls = 0;
        }
        
        endFrame() {
            const now = performance.now();
            const frameTime = now - this.frameStartTime;
            
            this.frameTimeHistory.push(frameTime);
            if (this.frameTimeHistory.length > 60) {
                this.frameTimeHistory.shift();
            }
            
            this.frameCount++;
            
            if (now - this.fpsUpdateTime >= 500) {
                this.fps = Math.round(this.frameCount * 1000 / (now - this.fpsUpdateTime));
                this.frameCount = 0;
                this.fpsUpdateTime = now;
                
                const sum = this.frameTimeHistory.reduce((a, b) => a + b, 0);
                this.frameTime = (sum / this.frameTimeHistory.length).toFixed(2);
                
                this.updateDisplay();
            }
        }
        
        updateDisplay() {
            if (this.elements.fps) {
                this.elements.fps.textContent = this.fps;
                this.elements.fps.style.color = this.fps >= 55 ? CONFIG.COLORS.neonGreen :
                    this.fps >= 30 ? CONFIG.COLORS.neonYellow : CONFIG.COLORS.neonPink;
            }
            if (this.elements.frameTime) this.elements.frameTime.textContent = this.frameTime + 'ms';
        }
        
        updateParticles(count) {
            if (this.elements.particles) this.elements.particles.textContent = count;
        }
        
        updateState(stateName) {
            if (this.elements.state) {
                this.elements.state.textContent = stateName;
                const stateColors = {
                    'MENU': CONFIG.COLORS.neonCyan,
                    'PLAYING': CONFIG.COLORS.neonGreen,
                    'PAUSED': CONFIG.COLORS.neonYellow
                };
                this.elements.state.style.color = stateColors[stateName] || CONFIG.COLORS.neonCyan;
            }
        }
    }

    class RoadBuilder {
        constructor() {
            this.segments = [];
            this.segmentLength = CONFIG.SEGMENT_LENGTH;
            this.trackLength = 0;
        }
        
        build() {
            this.segments = [];
            
            this.addStraight(50);
            this.addCurve(50, 2);
            this.addHill(30, 30);
            this.addCurve(40, -3);
            this.addStraight(30);
            this.addHill(20, -20);
            this.addCurve(60, 4);
            this.addStraight(40);
            this.addCurve(30, -2);
            this.addHill(40, 40);
            this.addCurve(50, 3);
            this.addStraight(30);
            this.addCurve(40, -4);
            this.addHill(30, -30);
            this.addStraight(50);
            this.addCurve(60, -2);
            this.addHill(25, 20);
            this.addStraight(40);
            
            this.trackLength = this.segments.length * this.segmentLength;
            return this;
        }
        
        addStraight(num) { this.addRoad(num, 0, 0); }
        addCurve(num, curve) { this.addRoad(num, curve, 0); }
        addHill(num, height) { this.addRoad(num, 0, height); }
        
        addRoad(num, curve, height) {
            const totalSegments = this.segments.length;
            for (let i = 0; i < num; i++) {
                const segCurve = curve * Math.sin((i / num) * Math.PI);
                const prevY = this.segments.length > 0 ? this.segments[this.segments.length - 1].y : 0;
                this.segments.push(new RoadSegment(totalSegments + i, segCurve, prevY + height / num));
            }
        }
        
        findSegment(z) {
            if (this.segments.length === 0) return null;
            const index = Math.floor(z / this.segmentLength) % this.segments.length;
            return this.segments[index >= 0 ? index : index + this.segments.length];
        }
    }

    // 玩家车辆类
    // ============================================
    class PlayerCar {
        constructor() {
            this.x = 0;
            this.y = 0;
            this.z = 0;
            
            this.speed = 0;
            this.velocityX = 0;
            
            this.steer = 0;
            this.targetSteer = 0;
            
            this.drifting = false;
            this.driftAngle = 0;
            this.driftIntensity = 0;
            
            this.handbraking = false;
            
            this.nitro = PLAYER_CONFIG.nitroMax;
            this.nitroActive = false;
            
            this.currentGear = 0;
            this.offRoad = false;
        }
        
        update(dt, input, road) {
            const segment = road.findSegment(this.z);
            const speedPercent = this.speed / PLAYER_CONFIG.maxSpeed;
            
            // 氮气控制
            if (input.isKeyDown('ShiftLeft') && this.nitro > 0 && this.speed > 50) {
                this.nitroActive = true;
                this.nitro = Math.max(0, this.nitro - PLAYER_CONFIG.nitroDrain);
            } else {
                this.nitroActive = false;
                if (this.speed > 10) {
                    this.nitro = Math.min(PLAYER_CONFIG.nitroMax, this.nitro + PLAYER_CONFIG.nitroRecharge);
                }
            }
            
            // 加速/刹车
            let accel = 0;
            if (input.isKeyDown('KeyW') || input.isKeyDown('ArrowUp')) {
                accel = PLAYER_CONFIG.acceleration;
            }
            if (input.isKeyDown('KeyS') || input.isKeyDown('ArrowDown')) {
                accel = PLAYER_CONFIG.braking;
            }
            
            // 氮气加速
            if (this.nitroActive) {
                accel *= PLAYER_CONFIG.nitroBoost;
            }
            
            // 档位加速度修正
            const gear = this.getCurrentGear();
            accel *= gear.accel;
            
            // 转向
            this.targetSteer = 0;
            if (input.isKeyDown('KeyA') || input.isKeyDown('ArrowLeft')) {
                this.targetSteer = -PLAYER_CONFIG.steerMax;
            }
            if (input.isKeyDown('KeyD') || input.isKeyDown('ArrowRight')) {
                this.targetSteer = PLAYER_CONFIG.steerMax;
            }
            
            // 手刹
            this.handbraking = input.isKeyDown('Space');
            
            // 转向速度调整
            let steerSpeed = PLAYER_CONFIG.steerSpeed;
            if (this.speed > 200) steerSpeed *= 0.7;
            if (this.handbraking) {
                steerSpeed *= PLAYER_CONFIG.handbrakeTurnBoost;
                this.targetSteer *= PLAYER_CONFIG.handbrakeTurnBoost;
            }
            
            this.steer = MathUtils.lerp(this.steer, this.targetSteer, steerSpeed);
            
            if (!this.handbraking && Math.abs(this.targetSteer) < 0.1) {
                this.steer = MathUtils.lerp(this.steer, 0, PLAYER_CONFIG.steerReturn);
            }
            
            // 漂移检测
            const driftThreshold = this.handbraking ? 0.15 : PLAYER_CONFIG.driftThreshold;
            this.drifting = Math.abs(this.steer) > driftThreshold && this.speed > 80;
            
            if (this.drifting) {
                this.driftIntensity = Math.min(1, this.driftIntensity + dt * 3);
                this.driftAngle = MathUtils.lerp(this.driftAngle, this.steer * 0.5, dt * 5);
                this.speed *= PLAYER_CONFIG.driftSpeedPenalty;
            } else {
                this.driftIntensity = Math.max(0, this.driftIntensity - dt * 2);
                this.driftAngle = MathUtils.lerp(this.driftAngle, 0, dt * 4);
            }
            
            // 离心力
            if (segment) {
                this.velocityX += segment.curve * speedPercent * PLAYER_CONFIG.centrifugal * dt * 60;
            }
            
            // 横向移动
            let grip = this.handbraking ? PLAYER_CONFIG.handbrakeGrip : 1;
            this.velocityX += this.steer * speedPercent * grip * dt * 60;
            this.velocityX *= Math.pow(this.drifting ? PLAYER_CONFIG.driftFactor : 0.9, dt * 60);
            this.x += this.velocityX * dt * 60;
            
            // 速度更新
            this.speed += accel * dt * 60;
            
            // 减速
            if (accel === 0) {
                this.speed += PLAYER_CONFIG.deceleration * dt * 60;
            }
            
            // 离开道路减速
            this.offRoad = Math.abs(this.x) > 1;
            if (this.offRoad) {
                this.speed += PLAYER_CONFIG.offRoadDeceleration * dt * 60;
                this.speed = Math.min(this.speed, PLAYER_CONFIG.maxSpeed * PLAYER_CONFIG.offRoadLimit);
            }
            
            // 速度限制
            const maxSpeed = this.nitroActive ? PLAYER_CONFIG.maxSpeed * 1.2 : PLAYER_CONFIG.maxSpeed;
            this.speed = MathUtils.clamp(this.speed, 0, maxSpeed);
            
            // 位置更新
            this.z += this.speed * dt * 60;
            
            // 赛道循环
            if (this.z >= road.trackLength) {
                this.z -= road.trackLength;
            }
            if (this.z < 0) {
                this.z += road.trackLength;
            }
            
            // 横向边界
            this.x = MathUtils.clamp(this.x, -2.5, 2.5);
        }
        
        getCurrentGear() {
            for (let i = PLAYER_CONFIG.gears.length - 1; i >= 0; i--) {
                if (this.speed >= PLAYER_CONFIG.gears[i].minSpeed) {
                    this.currentGear = i;
                    return PLAYER_CONFIG.gears[i];
                }
            }
            return PLAYER_CONFIG.gears[0];
        }
    }

    class Renderer {
        constructor(canvas) {
            this.canvas = canvas;
            this.ctx = canvas.getContext('2d');
            this.canvas.width = CONFIG.CANVAS_WIDTH;
            this.canvas.height = CONFIG.CANVAS_HEIGHT;
        }
        
        clear() {
            this.ctx.fillStyle = CONFIG.COLORS.bgDark;
            this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);
        }
        
        renderSky(fovStretch) {
            const ctx = this.ctx;
            const w = this.canvas.width;
            const h = this.canvas.height;
            
            // 天空渐变
            const skyGradient = ctx.createLinearGradient(0, 0, 0, h * 0.5);
            skyGradient.addColorStop(0, CONFIG.COLORS.sky.top);
            skyGradient.addColorStop(1, CONFIG.COLORS.sky.bottom);
            ctx.fillStyle = skyGradient;
            ctx.fillRect(0, 0, w, h * 0.5);
            
            // 星星
            ctx.fillStyle = 'rgba(255, 255, 255, 0.5)';
            for (let i = 0; i < 100; i++) {
                const x = (i * 137.5) % w;
                const y = (i * 73.7) % (h * 0.4);
                const size = Math.sin(i) * 1 + 1;
                ctx.beginPath();
                ctx.arc(x, y, size, 0, Math.PI * 2);
                ctx.fill();
            }
            
            // 地平线发光
            const horizonGlow = ctx.createRadialGradient(w / 2, h * 0.5, 0, w / 2, h * 0.5, w * 0.5);
            horizonGlow.addColorStop(0, 'rgba(0, 245, 255, 0.1)');
            horizonGlow.addColorStop(0.5, 'rgba(180, 70, 234, 0.05)');
            horizonGlow.addColorStop(1, 'transparent');
            ctx.fillStyle = horizonGlow;
            ctx.fillRect(0, 0, w, h);
        }
        
        renderRoad(road, player, fovStretch) {
            const ctx = this.ctx;
            const w = this.canvas.width;
            const h = this.canvas.height;
            
            const baseSegment = road.findSegment(player.z);
            if (!baseSegment) return;
            
            const cameraDepth = CONFIG.CAMERA_DEPTH * (1 + fovStretch);
            const playerY = MathUtils.lerp(CONFIG.CAMERA_HEIGHT, baseSegment.y + CONFIG.CAMERA_HEIGHT, 0.5);
            
            let maxy = h;
            let x = 0;
            let dx = 0;
            
            // 从远到近渲染道路分段
            for (let n = 0; n < CONFIG.DRAW_DISTANCE; n++) {
                const segmentIndex = (baseSegment.index + n) % road.segments.length;
                const segment = road.segments[segmentIndex];
                
                const camZ = player.z - (n > 0 ? 0 : 0);
                
                MathUtils.project(segment.p1, player.x * CONFIG.ROAD_WIDTH - x, playerY, camZ, cameraDepth, w, h, CONFIG.ROAD_WIDTH);
                MathUtils.project(segment.p2, player.x * CONFIG.ROAD_WIDTH - x - dx, playerY, camZ, cameraDepth, w, h, CONFIG.ROAD_WIDTH);
                
                x += dx;
                dx += segment.curve;
                
                if (segment.p1.camera.z <= 0 || segment.p2.screen.y >= maxy) continue;
                
                this.renderSegment(segment, n, maxy);
                maxy = segment.p2.screen.y;
            }
            
            // 渲染玩家车辆
            this.renderPlayerCar(player);
        }
        
        renderSegment(segment, index, maxy) {
            const ctx = this.ctx;
            const p1 = segment.p1.screen;
            const p2 = segment.p2.screen;
            
            if (p2.y >= maxy) return;
            
            const isLight = Math.floor(segment.index / 3) % 2 === 0;
            
            // 草地
            ctx.fillStyle = isLight ? CONFIG.COLORS.grass.light : CONFIG.COLORS.grass.dark;
            ctx.fillRect(0, p2.y, this.canvas.width, p1.y - p2.y);
            
            // 路肩
            const rumbleW1 = p1.w * 1.15;
            const rumbleW2 = p2.w * 1.15;
            ctx.fillStyle = isLight ? CONFIG.COLORS.rumble.light : CONFIG.COLORS.rumble.dark;
            ctx.beginPath();
            ctx.moveTo(p1.x - rumbleW1, p1.y);
            ctx.lineTo(p1.x + rumbleW1, p1.y);
            ctx.lineTo(p2.x + rumbleW2, p2.y);
            ctx.lineTo(p2.x - rumbleW2, p2.y);
            ctx.closePath();
            ctx.fill();
            
            // 道路
            ctx.fillStyle = isLight ? CONFIG.COLORS.asphalt.light : CONFIG.COLORS.asphalt.dark;
            ctx.beginPath();
            ctx.moveTo(p1.x - p1.w, p1.y);
            ctx.lineTo(p1.x + p1.w, p1.y);
            ctx.lineTo(p2.x + p2.w, p2.y);
            ctx.lineTo(p2.x - p2.w, p2.y);
            ctx.closePath();
            ctx.fill();
            
            // 车道线
            if (isLight && p1.w > 10) {
                ctx.strokeStyle = CONFIG.COLORS.lane;
                ctx.lineWidth = 2;
                ctx.setLineDash([20, 30]);
                ctx.beginPath();
                ctx.moveTo(p1.x, p1.y);
                ctx.lineTo(p2.x, p2.y);
                ctx.stroke();
                ctx.setLineDash([]);
            }
            
            // 边缘发光线
            if (p1.w > 5) {
                ctx.strokeStyle = CONFIG.COLORS.roadLineGlow;
                ctx.lineWidth = 3;
                ctx.shadowColor = CONFIG.COLORS.neonCyan;
                ctx.shadowBlur = 10;
                
                ctx.beginPath();
                ctx.moveTo(p1.x - p1.w, p1.y);
                ctx.lineTo(p2.x - p2.w, p2.y);
                ctx.stroke();
                
                ctx.beginPath();
                ctx.moveTo(p1.x + p1.w, p1.y);
                ctx.lineTo(p2.x + p2.w, p2.y);
                ctx.stroke();
                
                ctx.shadowBlur = 0;
            }
        }
        
        renderPlayerCar(player) {
            const ctx = this.ctx;
            const w = this.canvas.width;
            const h = this.canvas.height;
            
            const carX = w / 2 + player.velocityX * 50;
            const carY = h * 0.78;
            const carWidth = 80;
            const carHeight = 40;
            
            ctx.save();
            ctx.translate(carX, carY);
            ctx.rotate(player.driftAngle * 0.3);
            
            // 车身阴影
            ctx.shadowColor = 'rgba(0, 0, 0, 0.5)';
            ctx.shadowBlur = 20;
            ctx.shadowOffsetY = 10;
            
            // 车身
            const bodyGradient = ctx.createLinearGradient(-carWidth/2, 0, carWidth/2, 0);
            bodyGradient.addColorStop(0, '#1a1a2e');
            bodyGradient.addColorStop(0.5, '#2a2a4e');
            bodyGradient.addColorStop(1, '#1a1a2e');
            ctx.fillStyle = bodyGradient;
            
            ctx.beginPath();
            ctx.moveTo(-carWidth/2, carHeight/2);
            ctx.lineTo(-carWidth/2 + 10, -carHeight/2);
            ctx.lineTo(carWidth/2 - 10, -carHeight/2);
            ctx.lineTo(carWidth/2, carHeight/2);
            ctx.closePath();
            ctx.fill();
            
            ctx.shadowBlur = 0;
            ctx.shadowOffsetY = 0;
            
            // 车灯发光
            ctx.shadowColor = CONFIG.COLORS.neonCyan;
            ctx.shadowBlur = 15;
            ctx.fillStyle = CONFIG.COLORS.neonCyan;
            ctx.fillRect(-carWidth/2 + 5, carHeight/2 - 5, 15, 3);
            ctx.fillRect(carWidth/2 - 20, carHeight/2 - 5, 15, 3);
            
            // 尾灯
            ctx.shadowColor = CONFIG.COLORS.neonPink;
            ctx.fillStyle = CONFIG.COLORS.neonPink;
            ctx.fillRect(-carWidth/2 + 5, -carHeight/2 + 2, 10, 3);
            ctx.fillRect(carWidth/2 - 15, -carHeight/2 + 2, 10, 3);
            
            // 漂移时侧裙发光
            if (player.drifting) {
                ctx.shadowColor = CONFIG.COLORS.neonOrange;
                ctx.shadowBlur = 20;
                ctx.strokeStyle = CONFIG.COLORS.neonOrange;
                ctx.lineWidth = 3;
                ctx.beginPath();
                ctx.moveTo(-carWidth/2, 0);
                ctx.lineTo(carWidth/2, 0);
                ctx.stroke();
            }
            
            // 氮气尾焰发光
            if (player.nitroActive) {
                ctx.shadowColor = CONFIG.COLORS.neonCyan;
                ctx.shadowBlur = 30;
                ctx.fillStyle = CONFIG.COLORS.neonCyan;
                ctx.beginPath();
                ctx.arc(-20, -carHeight/2 - 10, 8, 0, Math.PI * 2);
                ctx.arc(20, -carHeight/2 - 10, 8, 0, Math.PI * 2);
                ctx.fill();
            }
            
            ctx.restore();
        }
    }

    // ============================================
    // 游戏状态机
    // ============================================
    class GameStateMachine {
        constructor() {
            this.state = 'MENU';
            this.transitions = {
                MENU: ['PLAYING'],
                PLAYING: ['PAUSED', 'MENU'],
                PAUSED: ['PLAYING', 'MENU']
            };
        }
        
        canTransition(to) {
            return this.transitions[this.state]?.includes(to);
        }
        
        transition(to) {
            if (this.canTransition(to)) {
                this.state = to;
                return true;
            }
            return false;
        }
    }

    // ============================================
    // 主游戏类
    // ============================================
    class Game {
        constructor() {
            this.canvas = document.getElementById('gameCanvas');
            this.vfxCanvas = document.getElementById('vfxCanvas');
            
            this.renderer = new Renderer(this.canvas);
            this.vfxManager = new VisualEffectsManager(this.vfxCanvas);
            this.hud = new HUDManager();
            this.performance = new PerformanceMonitor();
            this.input = new InputManager();
            this.stateMachine = new GameStateMachine();
            
            this.road = new RoadBuilder().build();
            this.player = new PlayerCar();
            
            this.lastTime = 0;
            this.running = false;
            
            this.init();
        }
        
        init() {
            this.stateMachine.transition('PLAYING');
            this.running = true;
            requestAnimationFrame((t) => this.gameLoop(t));
        }
        
        gameLoop(timestamp) {
            if (!this.running) return;
            
            const dt = Math.min((timestamp - this.lastTime) / 1000, 0.05);
            this.lastTime = timestamp;
            
            this.performance.beginFrame();
            
            this.update(dt);
            this.render();
            
            this.performance.endFrame();
            this.performance.updateState(this.stateMachine.state);
            this.performance.updateParticles(this.vfxManager.particleCount);
            
            this.input.clearJustPressed();
            
            requestAnimationFrame((t) => this.gameLoop(t));
        }
        
        update(dt) {
            if (this.stateMachine.state === 'PLAYING') {
                this.player.update(dt, this.input, this.road);
                this.vfxManager.update(dt, this.player, this.stateMachine.state);
                this.hud.update(this.player, dt);
                
                // 高速震动
                if (this.player.speed > 280) {
                    this.vfxManager.cameraShake.addTrauma(0.005);
                }
                
                // 漂移震动
                if (this.player.drifting) {
                    this.vfxManager.cameraShake.addTrauma(0.01);
                }
            }
        }
        
        render() {
            this.renderer.clear();
            this.renderer.renderSky(this.vfxManager.fovStretch);
            this.renderer.renderRoad(this.road, this.player, this.vfxManager.fovStretch);
            this.vfxManager.render(this.player);
        }
    }

    // ============================================
    // 启动游戏
    // ============================================
    window.addEventListener('DOMContentLoaded', () => {
        new Game();
    });
    </script>

## 实现的视觉特效功能

### 1. 粒子系统
- **速度线粒子**：高速时从屏幕两侧发射的流动线条，增强速度感
- **轮胎烟尘**：漂移/手刹时产生的渐变烟雾效果
- **火花飞溅**：高速漂移时轮胎摩擦产生的火花
- **氮气火焰**：使用氮气时的蓝紫色尾焰效果

### 2. 动态反馈
- **FOV视野拉伸**：速度越快，视野越宽，模拟真实驾驶感
- **相机震动**：高速行驶(>280km/h)和漂移时的震动反馈
- **边缘暗角**：高速时屏幕边缘变暗，聚焦视线中心
- **氮气屏幕光效**：使用氮气时屏幕边缘发光

### 3. UI仪表动画
- **速度表弧形动画**：SVG绘制的动态速度弧线
- **档位指示器**：实时显示当前档位，氮气时变色
- **转速条**：渐变色进度条，接近红线时变红
- **氮气条**：显示氮气储备，使用时有特殊效果
- **漂移指示器**：漂移时屏幕中央显示"DRIFT!"动画

### 4. 颜色动态变化
- 速度数字根据速度变色（青→橙→粉）
- 仪表发光效果随状态变化
- 氮气激活时整体色调偏蓝紫
