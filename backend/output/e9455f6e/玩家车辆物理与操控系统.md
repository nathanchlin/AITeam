    class MathUtils {
        static lerp(a, b, t) {
            return a + (b - a) * t;
        }
        
        static clamp(value, min, max) {
            return Math.max(min, Math.min(max, value));
        }
        
        static randomRange(min, max) {
            return Math.random() * (max - min) + min;
        }
        
        static easeOutCubic(t) {
            return 1 - Math.pow(1 - t, 3);
        }
        
        static exponentialFog(distance, density) {
            return 1 / Math.pow(Math.E, (distance * distance * density));
        }
        
        static project(p, cameraX, cameraY, cameraZ, cameraDepth, width, height, roadWidth) {
            const translatedZ = p.z - cameraZ;
            p.camera = {
                x: p.x - cameraX,
                y: p.y - cameraY,
                z: translatedZ
            };
            
            if (translatedZ <= 0) {
                p.screen = { x: 0, y: 0, w: 0, scale: 0 };
                return p;
            }
            
            const scale = cameraDepth / translatedZ;
            p.screen = {
                x: Math.round(width / 2 + scale * p.camera.x * width / 2),
                y: Math.round(height / 2 - scale * p.camera.y * height / 2),
                w: Math.round(scale * roadWidth * width / 2),
                scale: scale
            };
            
            return p;
        }
    }

                const sum = this.frameTimeHistory.reduce((a, b) => a + b, 0);
                this.frameTime = (sum / this.frameTimeHistory.length).toFixed(2);
                
                this.updateDisplay();
            }

    // 道路构建器

    class RoadBuilder {
        constructor() {
            this.segments = [];
            this.segmentLength = CONFIG.SEGMENT_LENGTH;
            this.trackLength = 0;
        }
        
        build() {
            this.segments = [];
            
            this.addStraight(50);
            this.addCurve(50, 2);
            this.addHill(30, 30);
            this.addCurve(40, -3);
            this.addStraight(30);
            this.addHill(20, -20);
            this.addCurve(60, 4);
            this.addStraight(40);
            this.addCurve(30, -2);
            this.addHill(40, 40);
            this.addCurve(50, 3);
            this.addStraight(30);
            this.addCurve(40, -4);
            this.addHill(30, -30);
            this.addStraight(50);
            this.addCurve(60, -2);
            this.addHill(25, 20);
            this.addStraight(40);
            
            this.trackLength = this.segments.length * this.segmentLength;
            
            return this;
        }
        
        addStraight(num) {
            this.addRoad(num, 0, 0);
        }
        
        addCurve(num, curve) {
            this.addRoad(num, curve, 0);
        }
        
        addHill(num, height) {
            this.addRoad(num, 0, height);
        }
        
        addRoad(num, curve, height) {
            const totalSegments = this.segments.length;
            
            for (let i = 0; i < num; i++) {
                const segCurve = this.enterCurve(i, num, curve);
                const segHeight = this.interpolateHeight(i, num, height);
                
                const prevY = this.segments.length > 0 ? 
                    this.segments[this.segments.length - 1].y : 0;
                    
                this.segments.push(new RoadSegment(
                    totalSegments + i,
                    segCurve,
                    prevY + segHeight
                ));
            }
        }
        
        enterCurve(i, num, curve) {
            return curve * Math.sin((i / num) * Math.PI);
        }
        
        interpolateHeight(i, num, height) {
            return height / num;
        }
        
        findSegment(z) {
            if (this.segments.length === 0) return null;
            const index = Math.floor(z / this.segmentLength) % this.segments.length;
            return this.segments[index >= 0 ? index : index + this.segments.length];
        }
    }

        constructor() {

            this.exhaustParticles = [];
        }
        
        update(dt, input, road) {
            // 1. 处理输入
            this.processInput(input);
            
            // 2. 更新转向
            this.updateSteering(dt);
            
            // 3. 更新速度和档位
            this.updateSpeedAndGear(dt);
            
            // 4. 更新漂移
            this.updateDrift(dt);
            
            // 5. 更新位置
            this.updatePosition(dt, road);
            
            // 6. 检查边界
            this.checkBounds(road);
            
            // 7. 更新粒子效果
            this.updateEffects(dt);
        }
        
        processInput(input) {
            // 手刹
            this.handbrake = input.isKeyDown('Space');
            
            // 转向输入
            let steerInput = 0;
            if (input.isKeyDown('ArrowLeft') || input.isKeyDown('KeyA')) {
                steerInput = -1;
            } else if (input.isKeyDown('ArrowRight') || input.isKeyDown('KeyD')) {
                steerInput = 1;
            }
            
            // 手刹时增强转向
            if (this.handbrake && Math.abs(this.speed) > 50) {
                steerInput *= PLAYER_CONFIG.handbrakeTurnBoost;
            }
            
            this.targetSteer = steerInput * PLAYER_CONFIG.steerMax;
            
            // 加速/刹车
            if (input.isKeyDown('ArrowUp') || input.isKeyDown('KeyW')) {
                this.accelerate();
            } else if (input.isKeyDown('ArrowDown') || input.isKeyDown('KeyS')) {
                this.brake();
            } else {
                this.decelerate();
            }
        }
        
        accelerate() {
            const gear = this.getCurrentGear();
            let accel = PLAYER_CONFIG.acceleration * gear.accel;
            
            // 离开道路时减速
            if (!this.onRoad) {
                accel *= 0.5;
            }
            
            this.speed += accel;
            
            // 根据档位限制最高速度
            const maxForGear = gear.maxSpeed;
            this.speed = Math.min(this.speed, maxForGear);
        }
        
        brake() {
            if (this.speed > 0) {
                this.speed += PLAYER_CONFIG.braking;
                this.speed = Math.max(this.speed, 0);
            } else {
                // 倒车
                this.speed -= PLAYER_CONFIG.acceleration * 0.3;
                this.speed = Math.max(this.speed, -40);
            }
        }
        
        decelerate() {
            // 自然减速（摩擦力）
            const friction = this.onRoad ? 
                PLAYER_CONFIG.deceleration : 
                PLAYER_CONFIG.offRoadDeceleration;
            
            if (this.speed > 0) {
                this.speed += friction;
                this.speed = Math.max(this.speed, 0);
            } else if (this.speed < 0) {
                this.speed -= friction;
                this.speed = Math.min(this.speed, 0);
            }
        }
        
        getCurrentGear() {
            const absSpeed = Math.abs(this.speed);
            for (let i = PLAYER_CONFIG.gears.length - 1; i >= 0; i--) {
                const gear = PLAYER_CONFIG.gears[i];
                if (absSpeed >= gear.minSpeed) {
                    this.currentGear = i + 1;
                    this.gearProgress = (absSpeed - gear.minSpeed) / (gear.maxSpeed - gear.minSpeed);
                    return gear;
                }
            }
            this.currentGear = 1;
            this.gearProgress = absSpeed / PLAYER_CONFIG.gears[0].maxSpeed;
            return PLAYER_CONFIG.gears[0];
        }
        
        updateSteering(dt) {
            const speedFactor = Math.abs(this.speed) / PLAYER_CONFIG.maxSpeed;
            
            // 低速时转向更灵敏，高速时转向更稳
            const steerResponsiveness = MathUtils.lerp(
                PLAYER_CONFIG.steerSpeed * 1.5,
                PLAYER_CONFIG.steerSpeed * 0.6,
                speedFactor
            );
            
            // 平滑过渡到目标转向
            if (Math.abs(this.targetSteer) > Math.abs(this.steer)) {
                this.steer = MathUtils.lerp(this.steer, this.targetSteer, steerResponsiveness);
            } else {
                this.steer = MathUtils.lerp(this.steer, this.targetSteer, PLAYER_CONFIG.steerReturn);
            }
            
            // 漂移时增加转向角度
            if (this.drifting) {
                this.steer = MathUtils.lerp(this.steer, this.targetSteer * PLAYER_CONFIG.driftSteerMultiplier, 0.1);
            }
        }
        
        updateDrift(dt) {
            const speedPercent = Math.abs(this.speed) / PLAYER_CONFIG.maxSpeed;
            const isTurning = Math.abs(this.steer) > 0.2;
            const isFastEnough = speedPercent > PLAYER_CONFIG.driftThreshold;
            
            // 检测漂移条件
            if (isTurning && isFastEnough) {
                if (!this.drifting) {
                    this.startDrift();
                }
                this.updateDriftPhysics();
            } else {
                this.endDrift();
            }
            
            // 手刹漂移
            if (this.handbrake && isFastEnough) {
                this.driftMomentum *= PLAYER_CONFIG.handbrakeGrip;
                this.speed *= PLAYER_CONFIG.driftSpeedPenalty;
            }
        }
        
        startDrift() {
            this.drifting = true;
            this.driftMomentum = this.steer * 0.8;
        }
        
        updateDriftPhysics() {
            // 漂移角度随转向累积
            const targetDriftAngle = this.steer * 0.5;
            this.driftAngle = MathUtils.lerp(this.driftAngle, targetDriftAngle, 0.08);
            
            // 漂移动量衰减
            this.driftMomentum = MathUtils.lerp(
                this.driftMomentum + this.steer * 0.02,
                0,
                0.02
            );
            this.driftMomentum *= PLAYER_CONFIG.driftFactor;
            
            // 计算漂移强度
            this.driftIntensity = Math.abs(this.driftAngle) + Math.abs(this.driftMomentum);
            
            // 漂移时轻微减速
            this.speed *= PLAYER_CONFIG.driftSpeedPenalty;
        }
        
        endDrift() {
            this.drifting = false;
            this.driftAngle = MathUtils.lerp(this.driftAngle, 0, 0.12);
            this.driftMomentum = MathUtils.lerp(this.driftMomentum, 0, 0.15);
            this.driftIntensity = MathUtils.lerp(this.driftIntensity, 0, 0.1);
        }
        
        updatePosition(dt, road) {
            // 更新Z位置
            this.z += this.speed * dt;
            
            // 赛道循环
            while (this.z >= road.trackLength) {
                this.z -= road.trackLength;
            }
            while (this.z < 0) {
                this.z += road.trackLength;
            }
            
            // 获取当前分段
            const segment = road.findSegment(this.z);
            if (!segment) return;
            
            // 离心力效果
            const centrifugalForce = segment.curve * PLAYER_CONFIG.centrifugal * 
                (this.speed / PLAYER_CONFIG.maxSpeed);
            
            // 横向移动 = 转向 + 漂移 + 离心力
            const lateralSpeed = (
                this.steer + 
                this.driftMomentum + 
                centrifugalForce
            ) * Math.abs(this.speed) * dt * 0.003;
            
            this.velocityX = MathUtils.lerp(this.velocityX, lateralSpeed, 0.2);
            this.x -= this.velocityX;
            
            // 更新高度
            this.y = MathUtils.lerp(this.y, segment.y, 0.15);
        }
        
        checkBounds(road) {
            // 检查是否在道路上
            this.onRoad = Math.abs(this.x) < 0.75;
            
            // 边界碰撞
            if (this.x < -2.2) {
                this.x = -2.2;
                this.velocityX *= -0.3;
                this.speed *= 0.7;
            } else if (this.x > 2.2) {
                this.x = 2.2;
                this.velocityX *= -0.3;
                this.speed *= 0.7;
            }
        }
        
        updateEffects(dt) {
            // 漂移时添加轮胎痕迹
            if (this.drifting && Math.abs(this.driftAngle) > 0.1) {
                this.tireTracks.push({
                    x: this.x,
                    z: this.z,
                    alpha: 1,
                    angle: this.driftAngle
                });
            }
            
            // 限制轮胎痕迹数量
            if (this.tireTracks.length > 100) {
                this.tireTracks.shift();
            }
            
            // 衰减轮胎痕迹
            this.tireTracks.forEach(track => {
                track.alpha *= 0.995;
            });
            this.tireTracks = this.tireTracks.filter(track => track.alpha > 0.1);
        }
        
        getSpeedPercent() {
            return Math.abs(this.speed) / PLAYER_CONFIG.maxSpeed;
        }
        
        getDisplaySpeed() {
            return Math.round(Math.abs(this.speed));
        }
    }

            const baseSegment = road.findSegment(player.z);
            if (!baseSegment) return;

            let x = 0;
            let dx = -(baseSegment.curve * basePercent);
            
            // 从远到近渲染
            for (let n = 0; n < CONFIG.DRAW_DISTANCE; n++) {
                const segIndex = (baseSegment.index + n) % road.segments.length;
                const segment = road.segments[segIndex];
                
                const camZ = player.z - (n > 0 ? 0 : basePercent * CONFIG.SEGMENT_LENGTH);
                
                // 投影
                MathUtils.project(segment.p1, 
                    player.x * CONFIG.ROAD_WIDTH - x, 
                    CONFIG.CAMERA_HEIGHT + player.y, 
                    camZ,
                    CONFIG.CAMERA_DEPTH,
                    this.width, this.height,
                    CONFIG.ROAD_WIDTH
                );
                
                MathUtils.project(segment.p2,
                    player.x * CONFIG.ROAD_WIDTH - x - dx,
                    CONFIG.CAMERA_HEIGHT + player.y,
                    camZ,
                    CONFIG.CAMERA_DEPTH,
                    this.width, this.height,
                    CONFIG.ROAD_WIDTH
                );
                
                x += dx;
                dx += segment.curve;
                
                if (segment.p1.camera.z <= 0) continue;
                
                this.renderSegment(segment, n);
                perf.addDrawCall();
                perf.addObject();
            }

            this.ctx.lineTo(p2.x + rumbleW2, p2.y);
            this.ctx.lineTo(p2.x - rumbleW2, p2.y);

## 实现说明

### 核心物理系统

1. **加速度/最高速度/摩擦力模型**
   - `PLAYER_CONFIG` 定义了所有物理参数
   - `accelerate()` - 根据当前档位应用不同加速度
   - `brake()` - 刹车减速，低速时允许倒车
   - `decelerate()` - 自然摩擦力减速，离开道路时额外减速

2. **速度档位系统**
   - 5个档位，每个档位有不同的加速度和最高速度
   - 低档位加速快但极速低，高档位相反
   - HUD显示当前档位和转速进度

3. **转向物理**
   - 转向响应随速度变化（低速灵敏，高速稳定）
   - 平滑的转向过渡，避免突兀的方向改变

### 漂移系统

1. **漂移触发条件**
   - 速度超过阈值 + 转向角度足够大
   - 手刹（空格键）可增强漂移效果

2. **漂移物理**
   - `driftMomentum` - 漂移动量，累积后逐渐衰减
   - `driftAngle` - 车身倾斜角度
   - 漂移时有轻微的速度损失

3. **视觉反馈**
   - 车身倾斜 + 粉色霓虹轮廓
   - 火花粒子效果
   - HUD显示 "DRIFT!" 指示

### 控制映射

- **W/↑** - 加速
- **S/↓** - 刹车/倒车
- **A/←** - 左转
- **D/→** - 右转
- **Space** - 手刹（增强漂移）
