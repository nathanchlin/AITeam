    class MathUtils {
        static lerp(a, b, t) {
            return a + (b - a) * t;
        }
        
        static clamp(value, min, max) {
            return Math.max(min, Math.min(max, value));
        }
        
        static randomRange(min, max) {
            return Math.random() * (max - min) + min;
        }
        
        static easeOutCubic(t) {
            return 1 - Math.pow(1 - t, 3);
        }
        
        static exponentialFog(distance, density) {
            return 1 / Math.pow(Math.E, (distance * distance * density));
        }
        
        static project(p, cameraX, cameraY, cameraZ, cameraDepth, width, height, roadWidth) {
            const translatedZ = p.z - cameraZ;
            p.camera = {
                x: p.x - cameraX,
                y: p.y - cameraY,
                z: translatedZ
            };
            
            if (translatedZ <= 0) {
                p.screen = { x: 0, y: 0, w: 0, scale: 0 };
                return p;
            }
            
            const scale = cameraDepth / translatedZ;
            p.screen = {
                x: Math.round(width / 2 + scale * p.camera.x * width / 2),
                y: Math.round(height / 2 - scale * p.camera.y * height / 2),
                w: Math.round(scale * roadWidth * width / 2),
                scale: scale
            };
            
            return p;
        }
        
        static distance2D(x1, z1, x2, z2) {
            const dx = x2 - x1;
            const dz = z2 - z1;
            return Math.sqrt(dx * dx + dz * dz);
        }
        
        static wrap(value, max) {
            return ((value % max) + max) % max;
        }
    }

            });
            
            window.addEventListener('keyup', (e) => {
                this.keys[e.code] = false;
            });
        }
        
        isKeyDown(code) {
            return !!this.keys[code];

        isKeyJustPressed(code) {
            return !!this.keysJustPressed[code];
        }

    class Particle {
        constructor(x, y, vx, vy, life, size, color, type = 'default') {
            this.x = x;
            this.y = y;
            this.vx = vx;
            this.vy = vy;
            this.life = life;
            this.maxLife = life;
            this.size = size;
            this.color = color;
            this.type = type;
            this.alpha = 1;
            this.rotation = Math.random() * Math.PI * 2;
            this.rotationSpeed = (Math.random() - 0.5) * 0.2;
        }
        
        update(dt) {
            this.x += this.vx * dt;
            this.y += this.vy * dt;
            this.life -= dt;
            this.alpha = Math.max(0, this.life / this.maxLife);
            this.rotation += this.rotationSpeed * dt;
            
            if (this.type === 'smoke') {
                this.size *= 1.02;
                this.vy -= 0.5 * dt;
            } else if (this.type === 'spark') {
                this.vy += 0.3 * dt;
            } else if (this.type === 'speedline') {
                this.size *= 0.98;
            }
            
            return this.life > 0;
        }
    }

                    y + (Math.random() - 0.5) * (config.offsetY || 0),
                    Math.cos(angle) * speed,
                    Math.sin(angle) * speed,
                    config.life * (0.5 + Math.random() * 0.5),
                    config.size * (0.5 + Math.random() * 0.5),
                    config.color,
                    config.type
                );
                this.particles.push(particle);
            }
        }
        
        emitSpeedLines(x, y, count, speedRatio) {
            const config = {
                angle: Math.PI,
                spread: 0.3,
                speed: 15 + speedRatio * 25,
                life: 0.3 + speedRatio * 0.3,
                size: 50 + speedRatio * 100,
                color: `rgba(0, 245, 255, ${0.3 + speedRatio * 0.4})`,
                type: 'speedline',
                offsetX: 100,
                offsetY: 50
            };
            this.emit(x, y, count, config);
        }
        
        emitTireSmoke(x, y, intensity) {
            const config = {
                angle: -Math.PI / 2,
                spread: Math.PI / 3,
                speed: 2 + intensity * 3,
                life: 0.8 + Math.random() * 0.4,
                size: 8 + intensity * 12,
                color: `rgba(180, 180, 180, ${0.3 + intensity * 0.3})`,
                type: 'smoke',
                offsetX: 30,
                offsetY: 10
            };
            this.emit(x, y, Math.floor(2 + intensity * 4), config);
        }
        
        emitSparks(x, y, intensity) {
            const config = {
                angle: -Math.PI / 2,
                spread: Math.PI,
                speed: 8 + intensity * 12,
                life: 0.4 + Math.random() * 0.3,
                size: 2 + intensity * 3,
                color: CONFIG.COLORS.neonOrange,
                type: 'spark',
                offsetX: 20,
                offsetY: 10
            };
            this.emit(x, y, Math.floor(3 + intensity * 5), config);
        }
        
        emitNitroFlame(x, y) {
            const colors = [CONFIG.COLORS.neonCyan, CONFIG.COLORS.neonPurple, CONFIG.COLORS.neonPink];
            const config = {
                angle: Math.PI,
                spread: 0.5,
                speed: 10 + Math.random() * 5,
                life: 0.2 + Math.random() * 0.2,
                size: 15 + Math.random() * 15,
                color: colors[Math.floor(Math.random() * colors.length)],
                type: 'flame',
                offsetX: 40,
                offsetY: 15
            };
            this.emit(x, y, 4, config);
        }
        
        emitCollisionSparks(x, y) {
            const config = {
                angle: 0,
                spread: Math.PI * 2,
                speed: 15,
                life: 0.5,
                size: 4,
                color: CONFIG.COLORS.neonYellow,
                type: 'spark',
                offsetX: 0,
                offsetY: 0
            };
            this.emit(x, y, 15, config);
        }
        
        update(dt) {
            this.particles = this.particles.filter(p => p.update(dt));
        }
        
        render(ctx, camera) {
            ctx.save();
            
            for (const p of this.particles) {
                ctx.globalAlpha = p.alpha;
                
                if (p.type === 'speedline') {
                    ctx.strokeStyle = p.color;
                    ctx.lineWidth = 2;
                    ctx.beginPath();
                    const lineLength = p.size * p.alpha;
                    ctx.moveTo(p.x, p.y);
                    ctx.lineTo(p.x - Math.cos(p.rotation) * lineLength, p.y - Math.sin(p.rotation) * lineLength);
                    ctx.stroke();
                } else if (p.type === 'smoke') {
                    const gradient = ctx.createRadialGradient(p.x, p.y, 0, p.x, p.y, p.size);
                    gradient.addColorStop(0, `rgba(150, 150, 150, ${p.alpha * 0.5})`);
                    gradient.addColorStop(1, `rgba(100, 100, 100, 0)`);
                    ctx.fillStyle = gradient;
                    ctx.beginPath();
                    ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
                    ctx.fill();
                } else if (p.type === 'spark' || p.type === 'flame') {
                    ctx.fillStyle = p.color;
                    ctx.shadowColor = p.color;
                    ctx.shadowBlur = 10;
                    ctx.beginPath();
                    ctx.arc(p.x, p.y, p.size * p.alpha, 0, Math.PI * 2);
                    ctx.fill();
                }
            }
            
            ctx.restore();
        }
        
        getCount() {
            return this.particles.length;
        }
    }

    class WeatherSystem {
        constructor(canvas) {
            this.canvas = canvas;
            this.ctx = canvas.getContext('2d');
            this.currentWeather = 'clear';
            this.particles = [];
            this.resize();
        }
        
        resize() {
            this.canvas.width = CONFIG.CANVAS_WIDTH;
            this.canvas.height = CONFIG.CANVAS_HEIGHT;
        }
        
        setWeather(weather) {
            this.currentWeather = weather;
            this.particles = [];
            
            if (weather === 'rain') {
                this.initRain();
            } else if (weather === 'snow') {
                this.initSnow();
            }
        }
        
        initRain() {
            const config = WEATHER_CONFIG.rain;
            for (let i = 0; i < config.dropCount; i++) {
                this.particles.push({
                    x: Math.random() * this.canvas.width,
                    y: Math.random() * this.canvas.height,
                    speed: config.dropSpeed * (0.8 + Math.random() * 0.4),
                    length: config.dropLength * (0.8 + Math.random() * 0.4),
                    alpha: 0.3 + Math.random() * 0.4
                });
            }
        }
        
        initSnow() {
            const config = WEATHER_CONFIG.snow;
            for (let i = 0; i < config.flakeCount; i++) {
                this.particles.push({
                    x: Math.random() * this.canvas.width,
                    y: Math.random() * this.canvas.height,
                    speed: config.fallSpeed * (0.5 + Math.random() * 0.5),
                    size: config.size.min + Math.random() * (config.size.max - config.size.min),
                    wobble: Math.random() * Math.PI * 2,
                    wobbleSpeed: 1 + Math.random() * 2,
                    alpha: 0.5 + Math.random() * 0.5
                });
            }
        }
        
        update(dt, playerSpeed) {
            const speedFactor = 1 + playerSpeed * 0.005;
            
            if (this.currentWeather === 'rain') {
                const config = WEATHER_CONFIG.rain;
                for (const p of this.particles) {
                    p.y += p.speed * speedFactor;
                    p.x += config.windFactor * speedFactor;
                    
                    if (p.y > this.canvas.height) {
                        p.y = -p.length;
                        p.x = Math.random() * this.canvas.width;
                    }
                    if (p.x > this.canvas.width) {
                        p.x = 0;
                    }
                }
            } else if (this.currentWeather === 'snow') {
                const config = WEATHER_CONFIG.snow;
                for (const p of this.particles) {
                    p.y += p.speed * speedFactor;
                    p.wobble += p.wobbleSpeed * dt;
                    p.x += Math.sin(p.wobble) * config.windFactor;
                    
                    if (p.y > this.canvas.height) {
                        p.y = -p.size;
                        p.x = Math.random() * this.canvas.width;
                    }
                    if (p.x < 0) p.x = this.canvas.width;
                    if (p.x > this.canvas.width) p.x = 0;
                }
            }
        }
        
        render(ctx, timeOfDay) {
            if (this.currentWeather === 'clear') return;
            
            ctx.save();
            
            // 根据时间调整透明度
            let alphaMod = 1;
            if (timeOfDay > 0.3 && timeOfDay < 0.7) {
                alphaMod = 0.7; // 白天稍微透明
            }
            
            if (this.currentWeather === 'rain') {
                const config = WEATHER_CONFIG.rain;
                ctx.strokeStyle = config.color;
                ctx.lineWidth = 1;
                
                for (const p of this.particles) {
                    ctx.globalAlpha = p.alpha * alphaMod;
                    ctx.beginPath();
                    ctx.moveTo(p.x, p.y);
                    ctx.lineTo(p.x + config.windFactor * 2, p.y + p.length);
                    ctx.stroke();
                }
            } else if (this.currentWeather === 'snow') {
                const config = WEATHER_CONFIG.snow;
                ctx.fillStyle = config.color;
                
                for (const p of this.particles) {
                    ctx.globalAlpha = p.alpha * alphaMod;
                    ctx.beginPath();
                    ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
                    ctx.fill();
                }
            }
            
            ctx.restore();
        }
    }

    class DayNightCycle {
        constructor() {
            this.timeMode = 'night'; // 'day', 'night', 'auto'
            this.currentTime = 21; // 24小时制
            this.autoTime = 0;
            this.transitionDuration = 2; // 过渡时间（秒）
        }
        
        setTimeMode(mode) {
            this.timeMode = mode;
            if (mode === 'day') {
                this.currentTime = 12;
            } else if (mode === 'night') {
                this.currentTime = 21;
            }
        }
        
        update(dt) {
            if (this.timeMode === 'auto') {
                this.autoTime += dt;
                // 加速日夜循环，120秒 = 24小时
                this.currentTime = (this.autoTime / TIME_CONFIG.dayDuration * 24) % 24;
            }
        }
        
        getTimeOfDay() {
            // 返回0-1的值，0=午夜，0.5=中午
            return this.currentTime / 24;
        }
        
        getSkyColors() {
            const time = this.currentTime;
            const colors = CONFIG.COLORS;
            
            // 夜晚 (20-5)
            if (time >= 20 || time < 5) {
                return { top: colors.sky.top, bottom: colors.sky.bottom };
            }
            // 黎明 (5-7)
            else if (time >= 5 && time < 7) {
                const t = (time - 5) / 2;
                return {
                    top: this.lerpColor(colors.sky.top, colors.skySunset.top, t),
                    bottom: this.lerpColor(colors.sky.bottom, colors.skySunset.bottom, t)
                };
            }
            // 早晨到中午 (7-12)
            else if (time >= 7 && time < 12) {
                const t = (time - 7) / 5;
                return {
                    top: this.lerpColor(colors.skySunset.top, colors.skyDay.top, t),
                    bottom: this.lerpColor(colors.skySunset.bottom, colors.skyDay.bottom, t)
                };
            }
            // 中午到黄昏 (12-18)
            else if (time >= 12 && time < 18) {
                const t = (time - 12) / 6;
                return {
                    top: this.lerpColor(colors.skyDay.top, colors.skySunset.top, t),
                    bottom: this.lerpColor(colors.skyDay.bottom, colors.skySunset.bottom, t)
                };
            }
            // 黄昏到夜晚 (18-20)
            else {
                const t = (time - 18) / 2;
                return {
                    top: this.lerpColor(colors.skySunset.top, colors.sky.top, t),
                    bottom: this.lerpColor(colors.skySunset.bottom, colors.sky.bottom, t)
                };
            }
        }
        
        getGrassColors() {
            const time = this.currentTime;
            const colors = CONFIG.COLORS;
            
            // 白天使用亮色草地
            if (time >= 6 && time < 19) {
                const t = Math.min(1, Math.abs(time - 12) / 6);
                return {
                    light: this.lerpColor(colors.grassDay.light, colors.grass.light, t * 0.5),
                    dark: this.lerpColor(colors.grassDay.dark, colors.grass.dark, t * 0.5)
                };
            }
            return { light: colors.grass.light, dark: colors.grass.dark };
        }
        
        getAmbientLight() {
            const time = this.currentTime;
            // 返回0-1的环境光强度
            if (time >= 6 && time < 19) {
                return 1;
            } else if (time >= 19 && time < 20) {
                return 1 - (time - 19);
            } else if (time >= 5 && time < 6) {
                return time - 5;
            }
            return 0;
        }
        
        lerpColor(color1, color2, t) {
            const c1 = this.hexToRgb(color1);
            const c2 = this.hexToRgb(color2);
            const r = Math.round(c1.r + (c2.r - c1.r) * t);
            const g = Math.round(c1.g + (c2.g - c1.g) * t);
            const b = Math.round(c1.b + (c2.b - c1.b) * t);
            return `rgb(${r}, ${g}, ${b})`;
        }
        
        hexToRgb(hex) {
            const result = /^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i.exec(hex);
            return result ? {
                r: parseInt(result[1], 16),
                g: parseInt(result[2], 16),
                b: parseInt(result[3], 16)
            } : { r: 0, g: 0, b: 0 };
        }
        
        getFormattedTime() {
            const hours = Math.floor(this.currentTime);
            const minutes = Math.floor((this.currentTime % 1) * 60);
            return `${hours.toString().padStart(2, '0')}:${minutes.toString().padStart(2, '0')}`;
        }
    }

    class Decoration {
        constructor(type, segment, side, offset = 0) {
            this.type = type;

            this.offset = offset;
            this.config = DECOR_CONFIG.types[type];

            this.scale = 0.8 + Math.random() * 0.4;
            this.variation = Math.random();

    // ============================================
    // 装饰物管理器
    // ============================================
    class DecorationManager {
        constructor() {
            this.decorations = [];
            this.visibleCount = 0;
        }
        
        generate(trackLength) {
            this.decorations = [];
            
            // 生成各种装饰物
            for (const [type, config] of Object.entries(DECOR_CONFIG.types)) {
                for (let i = 0; i < trackLength; i += config.repeat) {
                    // 两侧都放置
                    if (Math.random() > 0.3) {
                        this.decorations.push(new Decoration(type, i, -1, Math.random() * 100 - 50));
                    }
                    if (Math.random() > 0.3) {
                        this.decorations.push(new Decoration(type, i + config.repeat / 2, 1, Math.random() * 100 - 50));
                    }
                }
            }
            
            // 按segment排序
            this.decorations.sort((a, b) => a.segment - b.segment);
        }
        
        render(ctx, camera, segments, playerZ, dayNight) {
            this.visibleCount = 0;
            const ambientLight = dayNight.getAmbientLight();
            
            // 收集需要渲染的装饰物及其屏幕位置
            const toRender = [];
            
            for (const decor of this.decorations) {
                const segment = segments[decor.segment % segments.length];
                if (!segment) continue;
                
                const relZ = decor.segment * CONFIG.SEGMENT_LENGTH - playerZ;
                
                // 只渲染可见范围内的
                if (relZ < 0 || relZ > CONFIG.SEGMENT_LENGTH * DECOR_CONFIG.drawDistance) continue;
                
                const point = {
                    x: decor.side * (CONFIG.ROAD_WIDTH * decor.config.offset + decor.offset),
                    y: segment.p2.y,
                    z: decor.segment * CONFIG.SEGMENT_LENGTH
                };
                
                MathUtils.project(point, camera.x, camera.y - CONFIG.CAMERA_HEIGHT, playerZ,
                    CONFIG.CAMERA_DEPTH, CONFIG.CANVAS_WIDTH, CONFIG.CANVAS_HEIGHT, CONFIG.ROAD_WIDTH);
                
                if (point.screen.scale > 0) {
                    toRender.push({ decor, point, segment, relZ });
                }
            }
            
            // 按距离排序（远到近）
            toRender.sort((a, b) => b.relZ - a.relZ);
            
            // 渲染
            for (const { decor, point, segment, relZ } of toRender) {
                this.renderDecoration(ctx, decor, point, segment, ambientLight, relZ);
                this.visibleCount++;
            }
        }
        
        renderDecoration(ctx, decor, point, segment, ambientLight, distance) {
            const scale = point.screen.scale * decor.scale;
            const w = decor.config.width * scale;
            const h = decor.config.height * scale;
            const x = point.screen.x;
            const y = point.screen.y;
            
            // 雾效
            const fogFactor = Math.min(1, distance / (CONFIG.SEGMENT_LENGTH * DECOR_CONFIG.drawDistance * DECOR_CONFIG.fogStart));
            const fogAlpha = MathUtils.exponentialFog(distance / CONFIG.SEGMENT_LENGTH, 0.0005);
            
            ctx.save();
            ctx.globalAlpha = fogAlpha;
            
            switch (decor.type) {
                case 'tree':
                    this.renderTree(ctx, x, y, w, h, decor, ambientLight);
                    break;
                case 'billboard':
                    this.renderBillboard(ctx, x, y, w, h, decor, ambientLight);
                    break;
                case 'building':
                    this.renderBuilding(ctx, x, y, w, h, decor, ambientLight);
                    break;
                case 'lamp':
                    this.renderLamp(ctx, x, y, w, h, decor, ambientLight);
                    break;
                case 'rock':
                    this.renderRock(ctx, x, y, w, h, decor);
                    break;
                case 'barrier':
                    this.renderBarrier(ctx, x, y, w, h, decor);
                    break;
            }
            
            ctx.restore();
        }
        
        renderTree(ctx, x, y, w, h, decor, ambientLight) {
            const colors = decor.colors;
            
            // 树干
            ctx.fillStyle = colors.trunk;
            ctx.fillRect(x - w * 0.1, y - h * 0.4, w * 0.2, h * 0.4);
            
            // 树冠（使用多边形）
            const leafColor = ambientLight > 0.5 ? colors.leaves : colors.neonLeaves;
            ctx.fillStyle = leafColor;
            ctx.globalAlpha *= (0.5 + ambientLight * 0.5);
            
            ctx.beginPath();
            ctx.moveTo(x, y - h);
            ctx.lineTo(x - w * 0.5, y - h * 0.35);
            ctx.lineTo(x + w * 0.5, y - h * 0.35);
            ctx.closePath();
            ctx.fill();
            
            // 霓虹边缘效果（夜间）
            if (ambientLight < 0.3) {
                ctx.strokeStyle = colors.neonLeaves;
                ctx.lineWidth = 2;
                ctx.shadowColor = colors.neonLeaves;
                ctx.shadowBlur = 10;
                ctx.stroke();
            }
        }
        
        renderBillboard(ctx, x, y, w, h, decor, ambientLight) {
            const colors = decor.colors;
            
            // 支架
            ctx.fillStyle = '#2a2a3e';
            ctx.fillRect(x - w * 0.05, y - h, w * 0.1, h);
            
            // 广告牌主体
            ctx.fillStyle = colors.frame;
            ctx.fillRect(x - w * 0.5, y - h * 1.3, w, h * 0.6);
            
            // 屏幕
            ctx.fillStyle = colors.screen;
            ctx.fillRect(x - w * 0.45, y - h * 1.25, w * 0.9, h * 0.5);
            
            // 霓虹内容（随机图案）
            if (ambientLight < 0.5) {
                ctx.fillStyle = colors.glow;
                ctx.shadowColor = colors.glow;
                ctx.shadowBlur = 15;
                
                // 简单的广告图案
                const pattern = Math.floor(decor.variation * 3);
                if (pattern === 0) {
                    // 线条
                    for (let i = 0; i < 3; i++) {
                        ctx.fillRect(x - w * 0.35, y - h * 1.2 + i * h * 0.12, w * 0.7, h * 0.05);
                    }
                } else if (pattern === 1) {
                    // 方块
                    ctx.fillRect(x - w * 0.3, y - h * 1.18, w * 0.25, h * 0.15);
                    ctx.fillRect(x + w * 0.05, y - h * 1.18, w * 0.25, h * 0.15);
                } else {
                    // 圆形
                    ctx.beginPath();
                    ctx.arc(x, y - h * 1.1, w * 0.2, 0, Math.PI * 2);
                    ctx.fill();
                }
            }
        }
        
        renderBuilding(ctx, x, y, w, h, decor, ambientLight) {
            const colors = decor.colors;
            
            // 建筑主体
            ctx.fillStyle = colors.base;
            ctx.fillRect(x - w * 0.5, y - h, w, h);
            
            // 窗户网格
            const windowRows = 6;
            const windowCols = 3;
            const windowW = w * 0.2;
            const windowH = h * 0.08;
            const gapX = w * 0.1;
            const gapY = h * 0.05;
            
            for (let row = 0; row < windowRows; row++) {
                for (let col = 0; col < windowCols; col++) {
                    const wx = x - w * 0.35 + col * (windowW + gapX);
                    const wy = y - h * 0.9 + row * (windowH + gapY);
                    
                    // 随机窗户亮灭
                    const isLit = ((row * windowCols + col + Math.floor(decor.variation * 100)) % 3) !== 0;
                    
                    if (isLit && ambientLight < 0.5) {
                        ctx.fillStyle = colors.window;
                        ctx.shadowColor = colors.window;
                        ctx.shadowBlur = 8;
                    } else {
                        ctx.fillStyle = colors.windowOff;
                        ctx.shadowBlur = 0;
                    }
                    
                    ctx.fillRect(wx, wy, windowW, windowH);
                }
            }
            
            // 顶部霓虹灯（夜间）
            if (ambientLight < 0.3) {
                ctx.fillStyle = CONFIG.COLORS.neonPink;
                ctx.shadowColor = CONFIG.COLORS.neonPink;
                ctx.shadowBlur = 20;
                ctx.fillRect(x - w * 0.4, y - h * 0.98, w * 0.8, h * 0.02);
            }
        }
        
        renderLamp(ctx, x, y, w, h, decor, ambientLight) {
            const colors = decor.colors;
            
            // 灯杆
            ctx.fillStyle = colors.pole;
            ctx.fillRect(x - w * 0.2, y - h, w * 0.4, h);
            
            // 灯头
            ctx.fillStyle = colors.pole;
            ctx.fillRect(x - w * 0.6, y - h, w * 1.2, h * 0.1);
            
            // 灯光（夜间更亮）
            if (ambientLight < 0.5) {
                const intensity = 1 - ambientLight * 2;
                
                // 光晕
                const gradient = ctx.createRadialGradient(x, y - h * 0.95, 0, x, y - h * 0.95, h * 0.8);
                gradient.addColorStop(0, `rgba(255, 204, 0, ${0.4 * intensity})`);
                gradient.addColorStop(0.5, `rgba(255, 204, 0, ${0.15 * intensity})`);
                gradient.addColorStop(1, 'rgba(255, 204, 0, 0)');
                ctx.fillStyle = gradient;
                ctx.beginPath();
                ctx.arc(x, y - h * 0.95, h * 0.8, 0, Math.PI * 2);
                ctx.fill();
                
                // 灯泡
                ctx.fillStyle = colors.light;
                ctx.shadowColor = colors.light;
                ctx.shadowBlur = 20;
                ctx.beginPath();
                ctx.arc(x, y - h * 0.95, w * 0.3, 0, Math.PI * 2);
                ctx.fill();
            }
        }
        
        renderRock(ctx, x, y, w, h, decor) {
            const colors = decor.colors;
            
            // 简单的岩石形状
            ctx.fillStyle = colors.main;
            ctx.beginPath();
            ctx.moveTo(x - w * 0.5, y);
            ctx.lineTo(x - w * 0.3, y - h * 0.7);
            ctx.lineTo(x, y - h);
            ctx.lineTo(x + w * 0.4, y - h * 0.6);
            ctx.lineTo(x + w * 0.5, y);
            ctx.closePath();
            ctx.fill();
            
            // 高光
            ctx.fillStyle = colors.highlight;
            ctx.beginPath();
            ctx.moveTo(x - w * 0.2, y - h * 0.5);
            ctx.lineTo(x + w * 0.1, y - h * 0.8);
            ctx.lineTo(x + w * 0.2, y - h * 0.5);
            ctx.closePath();
            ctx.fill();
        }
        
        renderBarrier(ctx, x, y, w, h, decor) {
            const colors = decor.colors;
            
            // 护栏主体
            ctx.fillStyle = colors.main;

    class CheckpointSystem {
        constructor(trackLength) {
            this.checkpoints = [];
            this.passedCheckpoints = new Set();
            this.currentLap = 1;
            this.totalLaps = CONFIG.TOTAL_LAPS;
            this.trackLength = trackLength;
            this.lastPlayerSegment = 0;
            this.lapCompleted = false;
            
            this.initCheckpoints();
        }
        
        initCheckpoints() {
            const spacing = this.trackLength / CONFIG.CHECKPOINT_COUNT;
            for (let i = 0; i < CONFIG.CHECKPOINT_COUNT; i++) {
                this.checkpoints.push({
                    index: i,
                    segment: Math.floor(spacing * (i + 0.5)),
                    passed: false
                });
            }
        }
        
        update(playerSegment) {
            const wrappedSegment = MathUtils.wrap(playerSegment, this.trackLength);
            
            // 检测是否通过检查点
            for (const cp of this.checkpoints) {
                if (!cp.passed) {
                    // 玩家是否刚刚通过了检查点
                    const prevDist = MathUtils.wrap(this.lastPlayerSegment - cp.segment, this.trackLength);
                    const currDist = MathUtils.wrap(wrappedSegment - cp.segment, this.trackLength);
                    
                    if (prevDist > this.trackLength * 0.4 && currDist < this.trackLength * 0.1) {
                        cp.passed = true;
                        this.passedCheckpoints.add(cp.index);
                        this.showCheckpointNotification(cp.index);
                    }
                }
            }
            
            // 检测是否完成一圈（必须通过所有检查点并回到起点）
            const startZone = this.trackLength * 0.05;
            const crossedStart = this.lastPlayerSegment > this.trackLength - startZone && 
                                 wrappedSegment < startZone;
            
            if (crossedStart && this.passedCheckpoints.size === CONFIG.CHECKPOINT_COUNT) {
                if (this.currentLap < this.totalLaps) {
                    this.currentLap++;
                    this.resetCheckpoints();
                    this.showLapNotification();
                } else if (!this.lapCompleted) {
                    this.lapCompleted = true;
                    this.showRaceCompleteNotification();
                }
            }
            
            this.lastPlayerSegment = wrappedSegment;
        }
        
        resetCheckpoints() {
            for (const cp of this.checkpoints) {
                cp.passed = false;
            }
            this.passedCheckpoints.clear();
        }
        
        showCheckpointNotification(index) {
            const notification = document.getElementById('checkpointNotification');
            notification.textContent = `CHECKPOINT ${index + 1}!`;
            notification.style.opacity = '1';
            setTimeout(() => {
                notification.style.opacity = '0';
            }, 1000);
        }
        
        showLapNotification() {
            const notification = document.getElementById('lapNotification');
            notification.textContent = `LAP ${this.currentLap}`;
            notification.style.opacity = '1';
            notification.style.transform = 'translate(-50%, -50%) scale(1.2)';
            
            setTimeout(() => {
                notification.style.opacity = '0';
                notification.style.transform = 'translate(-50%, -50%) scale(1)';
            }, 2000);
        }
        
        showRaceCompleteNotification() {
            const notification = document.getElementById('lapNotification');
            notification.textContent = 'RACE COMPLETE!';
            notification.style.color = CONFIG.COLORS.neonGreen;
            notification.style.opacity = '1';
        }
        
        renderStartFinish(ctx, camera, segments, playerZ) {
            const startSegment = segments[0];
            if (!startSegment) return;
            
            const relZ = 0 - playerZ;
            if (relZ < -CONFIG.SEGMENT_LENGTH * 5 || relZ > CONFIG.SEGMENT_LENGTH * 20) return;
            
            // 渲染起跑线
            const point = {
                x: 0,
                y: startSegment.p1.y,
                z: 0
            };
            
            MathUtils.project(point, camera.x, camera.y - CONFIG.CAMERA_HEIGHT, playerZ,
                CONFIG.CAMERA_DEPTH, CONFIG.CANVAS_WIDTH, CONFIG.CANVAS_HEIGHT, CONFIG.ROAD_WIDTH);
            
            if (point.screen.scale > 0) {
                ctx.save();
                
                // 棋盘格起跑线
                const lineWidth = point.screen.w * 2;
                const lineHeight = 20 * point.screen.scale;
                const squareSize = lineWidth / 10;
                
                for (let i = 0; i < 10; i++) {
                    for (let j = 0; j < 2; j++) {
                        ctx.fillStyle = (i + j) % 2 === 0 ? '#ffffff' : '#000000';
                        ctx.fillRect(
                            point.screen.x - lineWidth / 2 + i * squareSize,
                            point.screen.y - lineHeight / 2 + j * (lineHeight / 2),
                            squareSize,
                            lineHeight / 2
                        );
                    }
                }
                
                // 霓虹边框
                ctx.strokeStyle = CONFIG.COLORS.neonCyan;
                ctx.lineWidth = 3 * point.screen.scale;
                ctx.shadowColor = CONFIG.COLORS.neonCyan;
                ctx.shadowBlur = 10;
                ctx.strokeRect(point.screen.x - lineWidth / 2, point.screen.y - lineHeight / 2, lineWidth, lineHeight);
                
                ctx.restore();
            }
        }
        
        updateUI() {
            document.getElementById('currentLap').textContent = this.currentLap;
            document.getElementById('totalLaps').textContent = this.totalLaps;
            
            // 更新检查点指示器
            const bar = document.getElementById('checkpointBar');
            bar.innerHTML = '';
            for (let i = 0; i < CONFIG.CHECKPOINT_COUNT; i++) {
                const dot = document.createElement('div');
                dot.className = 'checkpoint-dot';
                if (this.passedCheckpoints.has(i)) {
                    dot.classList.add('passed');
                } else if (!this.passedCheckpoints.size || i === [...Array(CONFIG.CHECKPOINT_COUNT).keys()].find(n => !this.passedCheckpoints.has(n))) {
                    dot.classList.add('next');
                }
                bar.appendChild(dot);
            }
        }
    }

    class AIVehicle {
        constructor(id, startSegment, lane, color) {
            this.id = id;
            this.segment = startSegment;
            this.lane = lane; // -1 to 1
            this.x = lane * CONFIG.ROAD_WIDTH * 0.35;
            this.z = startSegment * CONFIG.SEGMENT_LENGTH;
            this.y = 0;
            this.speed = MathUtils.randomRange(AI_CONFIG.minSpeed, AI_CONFIG.maxSpeed);
            this.targetSpeed = this.speed;
            this.targetLane = lane;
            this.color = color;
            this.width = 60;
            this.height = 30;
            this.laneChangeTimer = 0;
        }
        
        update(dt, playerZ, segments, trackLength, playerX) {
            // 更换车道逻辑
            this.laneChangeTimer -= dt;
            if (this.laneChangeTimer <= 0) {
                this.laneChangeTimer = AI_CONFIG.laneChangeInterval + Math.random() * 2;
                
                if (Math.random() < AI_CONFIG.laneChangeChance) {
                    this.targetLane = MathUtils.clamp(this.lane + (Math.random() > 0.5 ? 0.3 : -0.3), -0.8, 0.8);
                }
            }
            
            // 检测玩家距离
            const playerDist = this.z - playerZ;
            if (playerDist > 0 && playerDist < AI_CONFIG.avoidDistance) {
                // 躲避玩家
                if (Math.abs(this.x - playerX) < CONFIG.ROAD_WIDTH * 0.3) {
                    this.targetLane = playerX > 0 ? -0.5 : 0.5;
                    this.targetSpeed = this.speed * 0.9;
                }
            }
            
            // 平滑切换车道
            const targetX = this.targetLane * CONFIG.ROAD_WIDTH * 0.35;
            this.x += (targetX - this.x) * AI_CONFIG.reactionTime * dt * 10;
            
            // 速度调整
            this.speed += (this.targetSpeed - this.speed) * AI_CONFIG.acceleration * dt;
            
            // 更新位置
            this.z += this.speed * dt;
            
            // 循环赛道
            if (this.z >= trackLength * CONFIG.SEGMENT_LENGTH) {
                this.z -= trackLength * CONFIG.SEGMENT_LENGTH;
            }
            
            // 更新segment
            this.segment = Math.floor(this.z / CONFIG.SEGMENT_LENGTH) % trackLength;
            
            // 获取道路高度
            const segment = segments[this.segment];
            if (segment) {
                this.y = segment.p1.y;
            }
        }
        
        render(ctx, camera, playerZ) {
            const point = {
                x: this.x,
                y: this.y,
                z: this.z
            };
            
            MathUtils.project(point, camera.x, camera.y - CONFIG.CAMERA_HEIGHT, playerZ,
                CONFIG.CAMERA_DEPTH, CONFIG.CANVAS_WIDTH, CONFIG.CANVAS_HEIGHT, CONFIG.ROAD_WIDTH);
            
            if (point.screen.scale <= 0 || point.camera.z < 0) return;
            
            const scale = point.screen.scale;
            const w = this.width * scale;
            const h = this.height * scale;
            
            ctx.save();
            
            // 阴影
            ctx.fillStyle = 'rgba(0, 0, 0, 0.4)';
            ctx.beginPath();
            ctx.ellipse(point.screen.x, point.screen.y + h * 0.1, w, h * 0.3, 0, 0, Math.PI * 2);
            ctx.fill();
            
            // 车身
            ctx.fillStyle = this.color;
            ctx.shadowColor = this.color;
            ctx.shadowBlur = 15;
            
            // 简化的车辆形状
            ctx.beginPath();
            ctx.moveTo(point.screen.x - w, point.screen.y);
            ctx.lineTo(point.screen.x - w * 0.8, point.screen.y - h);
            ctx.lineTo(point.screen.x + w * 0.8, point.screen.y - h);
            ctx.lineTo(point.screen.x + w, point.screen.y);
            ctx.closePath();
            ctx.fill();
            
            // 车窗
            ctx.fillStyle = 'rgba(0, 0, 0, 0.6)';
            ctx.fillRect(point.screen.x - w * 0.5, point.screen.y - h * 0.85, w, h * 0.4);
            
            // 车灯
            ctx.fillStyle = '#ff0000';
            ctx.shadowColor = '#ff0000';
            ctx.shadowBlur = 10;
            ctx.fillRect(point.screen.x - w * 0.8, point.screen.y - h * 0.2, w * 0.2, h * 0.15);
            ctx.fillRect(point.screen.x + w * 0.6, point.screen.y - h * 0.2, w * 0.2, h * 0.15);
            
            ctx.restore();
        }
    }

    class PlayerVehicle {
        constructor() {
            this.x = 0;
            this.z = 0;
            this.y = 0;
            this.speed = 0;
            this.steer = 0;
            this.driftAngle = 0;
            this.nitro = PLAYER_CONFIG.nitroMax;
            this.nitroActive = false;
            this.offRoad = false;
        }
        
        update(dt, input, segments, trackLength) {
            // 获取当前路段信息
            const segmentIndex = Math.floor(this.z / CONFIG.SEGMENT_LENGTH) % trackLength;
            const segment = segments[segmentIndex];
            
            if (!segment) return;
            
            // 检测是否在道路上
            this.offRoad = Math.abs(this.x) > CONFIG.ROAD_WIDTH * 0.5;
            
            // 加速/刹车
            let accel = 0;
            if (input.isKeyDown('ArrowUp') || input.isKeyDown('KeyW')) {
                accel = PLAYER_CONFIG.acceleration;
            }
            if (input.isKeyDown('ArrowDown') || input.isKeyDown('KeyS')) {
                accel = PLAYER_CONFIG.braking;
            }
            
            // 档位影响
            const gear = PLAYER_CONFIG.gears.find(g => this.speed >= g.minSpeed && this.speed < g.maxSpeed) || PLAYER_CONFIG.gears[0];
            accel *= gear.accel;
            
            // 氮气加速
            this.nitroActive = input.isKeyDown('ShiftLeft') && this.nitro > 0 && this.speed > 50;
            if (this.nitroActive) {
                accel *= PLAYER_CONFIG.nitroBoost;
                this.nitro = Math.max(0, this.nitro - PLAYER_CONFIG.nitroDrain);
            } else {
                this.nitro = Math.min(PLAYER_CONFIG.nitroMax, this.nitro + PLAYER_CONFIG.nitroRecharge);
            }
            
            // 减速
            let decel = PLAYER_CONFIG.deceleration;
            if (this.offRoad) {
                decel = PLAYER_CONFIG.offRoadDeceleration;
            }
            
            // 手刹
            const handbrake = input.isKeyDown('Space');
            const driftMultiplier = handbrake ? PLAYER_CONFIG.handbrakeTurnBoost : 1;
            const gripFactor = handbrake ? PLAYER_CONFIG.handbrakeGrip : 1;
            
            // 转向
            const steerInput = (input.isKeyDown('ArrowLeft') || input.isKeyDown('KeyA') ? -1 : 0) +
                              (input.isKeyDown('ArrowRight') || input.isKeyDown('KeyD') ? 1 : 0);
            
            if (steerInput !== 0) {
                this.steer += steerInput * PLAYER_CONFIG.steerSpeed * driftMultiplier;
                this.steer = MathUtils.clamp(this.steer, -PLAYER_CONFIG.steerMax, PLAYER_CONFIG.steerMax);
            } else {
                this.steer *= (1 - PLAYER_CONFIG.steerReturn);
            }
            
            // 应用速度变化
            this.speed += accel * dt * 60;
            this.speed += decel * (this.speed > 0 ? 1 : -0.5) * dt * 60;
            
            // 离路限制
            if (this.offRoad) {
                this.speed = Math.min(this.speed, PLAYER_CONFIG.maxSpeed * PLAYER_CONFIG.offRoadLimit);
            }
            
            this.speed = MathUtils.clamp(this.speed, 0, PLAYER_CONFIG.maxSpeed);
            
            // 弯道离心力
            const centrifugalForce = segment.curve * PLAYER_CONFIG.centrifugal * (this.speed / PLAYER_CONFIG.maxSpeed);
            
            // 更新位置
            this.x += (this.steer * this.speed * 0.02 + centrifugalForce) * dt * 60;
            this.x = MathUtils.clamp(this.x, -CONFIG.ROAD_WIDTH * 1.2, CONFIG.ROAD_WIDTH * 1.2);
            
            // 漂移角度
            if (handbrake && Math.abs(this.steer) > 0.1 && this.speed > 80) {
                this.driftAngle += this.steer * 0.1;
                this.speed *= Math.pow(PLAYER_CONFIG.driftSpeedPenalty, dt * 60);
            } else {
                this.driftAngle *= PLAYER_CONFIG.driftFactor;
            }
            this.driftAngle = MathUtils.clamp(this.driftAngle, -PLAYER_CONFIG.steerMax * 1.5, PLAYER_CONFIG.steerMax * 1.5);
            
            // 前进
            this.z += this.speed * dt;
            
            // 循环赛道
            if (this.z >= trackLength * CONFIG.SEGMENT_LENGTH) {
                this.z -= trackLength * CONFIG.SEGMENT_LENGTH;
            }
            
            // 高度跟随
            this.y = segment.p1.y;
        }
        
        render(ctx) {
            const screenX = CONFIG.CANVAS_WIDTH / 2;
            const screenY = CONFIG.CANVAS_HEIGHT - 100;
            const scale = 1;
            
            ctx.save();
            ctx.translate(screenX, screenY);
            ctx.rotate(-this.driftAngle * 0.5);
            
            // 车身
            ctx.fillStyle = CONFIG.COLORS.neonCyan;
            ctx.shadowColor = CONFIG.COLORS.neonCyan;
            ctx.shadowBlur = 20;
            
            ctx.beginPath();
            ctx.moveTo(-40 * scale, 0);
            ctx.lineTo(-35 * scale, -50 * scale);
            ctx.lineTo(35 * scale, -50 * scale);
            ctx.lineTo(40 * scale, 0);
            ctx.closePath();
            ctx.fill();
            
            // 车窗
            ctx.fillStyle = 'rgba(0, 0, 0, 0.6)';
            ctx.fillRect(-25 * scale, -45 * scale, 50 * scale, 20 * scale);
            
            // 氮气效果
            if (this.nitroActive) {
                ctx.fillStyle = CONFIG.COLORS.neonPurple;
                ctx.shadowColor = CONFIG.COLORS.neonPurple;
                ctx.shadowBlur = 30;
                ctx.beginPath();
                ctx.moveTo(-20 * scale, 0);
                ctx.lineTo(-15 * scale, 30 * scale);
                ctx.lineTo(15 * scale, 30 * scale);
                ctx.lineTo(20 * scale, 0);
                ctx.closePath();
                ctx.fill();
            }
            
            ctx.restore();
        }
    }

    class Track {
        constructor(length) {
            this.length = length;
            this.segments = [];
            this.generate();
        }
        
        generate() {
            for (let i = 0; i < this.length; i++) {
                let curve = 0;
                let hill = 0;
                
                // 弯道
                if (i > 50 && i < 100) curve = 2;
                if (i > 150 && i < 200) curve = -3;
                if (i > 250 && i < 300) curve = 4;
                if (i > 350 && i < 400) curve = -2;
                if (i > 420 && i < 470) curve = 3;
                
                // 坡道
                if (i > 20 && i < 40) hill = 30;
                if (i > 60 && i < 80) hill = -20;
                if (i > 120 && i < 140) hill = 25;
                if (i > 200 && i < 230) hill = -35;
                if (i > 300 && i < 330) hill = 40;
                if (i > 400 && i < 430) hill = -25;
                
                const p1 = { x: 0, y: this.getLastY() + hill, z: i * CONFIG.SEGMENT_LENGTH };
                const p2 = { x: 0, y: this.getLastY() + hill * 2, z: (i + 1) * CONFIG.SEGMENT_LENGTH };
                
                this.segments.push({
                    index: i,
                    p1, p2,
                    curve,
                    color: i % 2 === 0 ? 'light' : 'dark'
                });
            }
        }
        
        getLastY() {
            if (this.segments.length === 0) return 0;
            return this.segments[this.segments.length - 1].p2.y;
        }
    }

    class Renderer {
        constructor(canvas) {
            this.canvas = canvas;
            this.ctx = canvas.getContext('2d');
            this.resize();
        }
        
        resize() {
            this.canvas.width = CONFIG.CANVAS_WIDTH;
            this.canvas.height = CONFIG.CANVAS_HEIGHT;
        }
        
        clear() {
            this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);
        }
        
        renderSky(skyColors) {
            const gradient = this.ctx.createLinearGradient(0, 0, 0, this.canvas.height * 0.5);
            gradient.addColorStop(0, skyColors.top);
            gradient.addColorStop(1, skyColors.bottom);
            this.ctx.fillStyle = gradient;
            this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height * 0.5);
            
            // 星星（夜间）
            const timeOfDay = skyColors.top === CONFIG.COLORS.sky.top;
            if (timeOfDay) {
                this.ctx.fillStyle = '#ffffff';
                for (let i = 0; i < 50; i++) {
                    const x = (i * 137) % this.canvas.width;
                    const y = (i * 89) % (this.canvas.height * 0.4);
                    const size = 1 + (i % 3);
                    this.ctx.globalAlpha = 0.3 + (i % 5) * 0.1;
                    this.ctx.fillRect(x, y, size, size);
                }
                this.ctx.globalAlpha = 1;
            }
        }
        
        renderRoad(player, track, camera, grassColors) {
            const segments = track.segments;
            const baseSegment = Math.floor(player.z / CONFIG.SEGMENT_LENGTH);
            const playerPercent = (player.z % CONFIG.SEGMENT_LENGTH) / CONFIG.SEGMENT_LENGTH;
            
            let maxY = this.canvas.height;
            let x = 0;
            let dx = 0;
            
            // 从远到近渲染
            for (let n = 0; n < CONFIG.DRAW_DISTANCE; n++) {
                const segmentIndex = (baseSegment + n) % track.length;
                const segment = segments[segmentIndex];
                
                if (!segment) continue;
                
                // 投影
                const p1 = MathUtils.project(
                    { x: x, y: segment.p1.y, z: segment.p1.z },
                    camera.x, camera.y - CONFIG.CAMERA_HEIGHT, player.z,
                    CONFIG.CAMERA_DEPTH, this.canvas.width, this.canvas.height, CONFIG.ROAD_WIDTH
                );
                
                const p2 = MathUtils.project(
                    { x: x + dx, y: segment.p2.y, z: segment.p2.z },
                    camera.x, camera.y - CONFIG.CAMERA_HEIGHT, player.z,
                    CONFIG.CAMERA_DEPTH, this.canvas.width, this.canvas.height, CONFIG.ROAD_WIDTH
                );
                
                x += dx;
                dx += segment.curve;
                
                // 跳过屏幕外的段落
                if (p1.screen.y >= maxY || p2.screen.y >= p1.screen.y) continue;
                
                // 草地
                const grassColor = segment.color === 'light' ? grassColors.light : grassColors.dark;
                this.ctx.fillStyle = grassColor;
                this.ctx.fillRect(0, p2.screen.y, this.canvas.width, p1.screen.y - p2.screen.y);
                
                // 路肩
                const rumbleColor = segment.color === 'light' ? CONFIG.COLORS.rumble.light : CONFIG.COLORS.rumble.dark;
                const rumbleW1 = p1.screen.w * 1.15;
                const rumbleW2 = p2.screen.w * 1.15;
                
                this.ctx.fillStyle = rumbleColor;
                this.ctx.beginPath();
                this.ctx.moveTo(p1.screen.x - rumbleW1, p1.screen.y);
                this.ctx.lineTo(p1.screen.x - p1.screen.w, p1.screen.y);
                this.ctx.lineTo(p2.screen.x - p2.screen.w, p2.screen.y);
                this.ctx.lineTo(p2.screen.x - rumbleW2, p2.screen.y);
                this.ctx.fill();
                
                this.ctx.beginPath();
                this.ctx.moveTo(p1.screen.x + p1.screen.w, p1.screen.y);
                this.ctx.lineTo(p1.screen.x + rumbleW1, p1.screen.y);
                this.ctx.lineTo(p2.screen.x + rumbleW2, p2.screen.y);
                this.ctx.lineTo(p2.screen.x + p2.screen.w, p2.screen.y);
                this.ctx.fill();
                
                // 道路
                const asphaltColor = segment.color === 'light' ? CONFIG.COLORS.asphalt.light : CONFIG.COLORS.asphalt.dark;
                this.ctx.fillStyle = asphaltColor;
                this.ctx.beginPath();
                this.ctx.moveTo(p1.screen.x - p1.screen.w, p1.screen.y);
                this.ctx.lineTo(p1.screen.x + p1.screen.w, p1.screen.y);
                this.ctx.lineTo(p2.screen.x + p2.screen.w, p2.screen.y);
                this.ctx.lineTo(p2.screen.x - p2.screen.w, p2.screen.y);
                this.ctx.fill();
                
                // 车道线
                if (segment.color === 'light') {
                    this.ctx.strokeStyle = CONFIG.COLORS.lane;
                    this.ctx.lineWidth = 2;
                    this.ctx.setLineDash([20, 20]);
                    this.ctx.beginPath();
                    this.ctx.moveTo(p1.screen.x, p1.screen.y);
                    this.ctx.lineTo(p2.screen.x, p2.screen.y);
                    this.ctx.stroke();
                    this.ctx.setLineDash([]);
                }
                
                // 边缘线
                this.ctx.strokeStyle = CONFIG.COLORS.roadLine;
                this.ctx.shadowColor = CONFIG.COLORS.roadLineGlow;
                this.ctx.shadowBlur = 5;
                this.ctx.lineWidth = 2;
                this.ctx.beginPath();
                this.ctx.moveTo(p1.screen.x - p1.screen.w, p1.screen.y);
                this.ctx.lineTo(p2.screen.x - p2.screen.w, p2.screen.y);
                this.ctx.stroke();
                this.ctx.beginPath();
                this.ctx.moveTo(p1.screen.x + p1.screen.w, p1.screen.y);
                this.ctx.lineTo(p2.screen.x + p2.screen.w, p2.screen.y);
                this.ctx.stroke();
                this.ctx.shadowBlur = 0;
                
                maxY = p2.screen.y;
            }
        }
    }

    class Game {
        constructor() {
            this.canvas = document.getElementById('gameCanvas');

            this.weatherCanvas = document.getElementById('weatherOverlay');
            
            this.renderer = new Renderer(this.canvas);
            this.vfxCtx = this.vfxCanvas.getContext('2d');
            
            this.input = new InputManager();
            this.particleSystem = new ParticleSystem();
            this.weatherSystem = new WeatherSystem(this.weatherCanvas);
            this.dayNight = new DayNightCycle();
            this.decorationManager = new DecorationManager();
            
            this.track = new Track(CONFIG.TRACK_LENGTH);
            this.player = new PlayerVehicle();
            this.checkpointSystem = new CheckpointSystem(CONFIG.TRACK_LENGTH);
            
            // 生成AI车辆
            this.aiVehicles = [];
            const aiColors = [CONFIG.COLORS.neonPink, CONFIG.COLORS.neonOrange, CONFIG.COLORS.neonPurple, CONFIG.COLORS.neonGreen];
            for (let i = 0; i < AI_CONFIG.count; i++) {
                this.aiVehicles.push(new AIVehicle(
                    i,
                    Math.floor(Math.random() * CONFIG.TRACK_LENGTH),
                    (Math.random() - 0.5) * 1.6,
                    aiColors[i]
                ));
            }
            
            // 生成装饰物
            this.decorationManager.generate(CONFIG.TRACK_LENGTH);
            
            // 调整VFX画布大小
            this.vfxCanvas.width = CONFIG.CANVAS_WIDTH;
            this.vfxCanvas.height = CONFIG.CANVAS_HEIGHT;
            
            this.camera = { x: 0, y: 0 };
            this.frameCount = 0;
            this.lastTime = 0;
            this.fps = 0;
            this.fpsUpdateTimer = 0;
            
            this.screenShake = { x: 0, y: 0, intensity: 0 };
            
            this.bindWeatherControls();
            this.init();
        }
        
        bindWeatherControls() {
            document.querySelectorAll('.weather-btn[data-weather]').forEach(btn => {
                btn.addEventListener('click', () => {
                    document.querySelectorAll('.weather-btn[data-weather]').forEach(b => b.classList.remove('active'));
                    btn.classList.add('active');
                    this.weatherSystem.setWeather(btn.dataset.weather);
                });
            });
            
            document.querySelectorAll('.weather-btn[data-time]').forEach(btn => {
                btn.addEventListener('click', () => {
                    document.querySelectorAll('.weather-btn[data-time]').forEach(b => b.classList.remove('active'));
                    btn.classList.add('active');
                    this.dayNight.setTimeMode(btn.dataset.time);
                });
            });
        }
        
        init() {
            this.lastTime = performance.now();
            requestAnimationFrame((t) => this.gameLoop(t));
        }
        
        gameLoop(timestamp) {
            const currentTime = timestamp;
            const dt = Math.min((currentTime - this.lastTime) / 1000, 0.05);
            this.lastTime = currentTime;
            
            this.update(dt);
            this.render();
            
            this.frameCount++;
            this.updatePerformancePanel(dt);
            
            requestAnimationFrame((t) => this.gameLoop(t));
        }
        
        update(dt) {
            // 更新日夜循环
            this.dayNight.update(dt);
            
            // 更新玩家
            this.player.update(dt, this.input, this.track.segments, CONFIG.TRACK_LENGTH);
            
            // 更新摄像机
            this.camera.x = this.player.x * 0.5;
            this.camera.y = this.player.y + CONFIG.CAMERA_HEIGHT;
            
            // 更新AI车辆
            for (const ai of this.aiVehicles) {
                ai.update(dt, this.player.z, this.track.segments, CONFIG.TRACK_LENGTH, this.player.x);
            }
            
            // 更新检查点
            const playerSegment = Math.floor(this.player.z / CONFIG.SEGMENT_LENGTH);
            this.checkpointSystem.update(playerSegment);
            this.checkpointSystem.updateUI();
            
            // 更新天气
            this.weatherSystem.update(dt, this.player.speed);
            
            // 粒子效果
            this.updateParticles(dt);
            
            // 屏幕震动衰减
            this.screenShake.intensity *= 0.9;
            this.screenShake.x = (Math.random() - 0.5) * this.screenShake.intensity;
            this.screenShake.y = (Math.random() - 0.5) * this.screenShake.intensity;
            
            // 氮气视觉反馈
            const nitroOverlay = document.getElementById('nitroOverlay');
            nitroOverlay.style.opacity = this.player.nitroActive ? '0.3' : '0';
            
            // 清除按键状态
            this.input.clearJustPressed();
        }
        
        updateParticles(dt) {
            const speedRatio = this.player.speed / PLAYER_CONFIG.maxSpeed;
            const isDrifting = Math.abs(this.player.driftAngle) > PLAYER_CONFIG.driftThreshold;
            
            // 速度线
            if (this.player.speed > 150) {
                this.particleSystem.emitSpeedLines(
                    CONFIG.CANVAS_WIDTH / 2,
                    CONFIG.CANVAS_HEIGHT / 2,
                    Math.floor(speedRatio * 2),
                    speedRatio
                );
            }
            
            // 轮胎烟
            if (isDrifting) {
                this.particleSystem.emitTireSmoke(
                    CONFIG.CANVAS_WIDTH / 2 - this.player.driftAngle * 50,
                    CONFIG.CANVAS_HEIGHT - 80,
                    Math.abs(this.player.driftAngle)
                );
            }
            
            // 氮气火焰
            if (this.player.nitroActive) {
                this.particleSystem.emitNitroFlame(CONFIG.CANVAS_WIDTH / 2, CONFIG.CANVAS_HEIGHT - 70);
            }
            
            // 离路火花
            if (this.player.offRoad && this.player.speed > 100) {
                this.particleSystem.emitSparks(
                    CONFIG.CANVAS_WIDTH / 2,
                    CONFIG.CANVAS_HEIGHT - 80,
                    0.5
                );
            }
            
            this.particleSystem.update(dt);
        }
        
        render() {
            const skyColors = this.dayNight.getSkyColors();
            const grassColors = this.dayNight.getGrassColors();
            
            // 应用屏幕震动
            this.renderer.ctx.save();
            this.renderer.ctx.translate(this.screenShake.x, this.screenShake.y);
            
            // 清空画布
            this.renderer.clear();
            
            // 渲染天空
            this.renderer.renderSky(skyColors);
            
            // 渲染道路
            this.renderer.renderRoad(this.player, this.track, this.camera, grassColors);
            
            // 渲染起跑线
            this.checkpointSystem.renderStartFinish(this.renderer.ctx, this.camera, this.track.segments, this.player.z);
            
            // 渲染装饰物
            this.decorationManager.render(this.renderer.ctx, this.camera, this.track.segments, this.player.z, this.dayNight);
            
            // 渲染AI车辆（按距离排序）
            const sortedAI = [...this.aiVehicles].sort((a, b) => {
                const distA = a.z - this.player.z;
                const distB = b.z - this.player.z;
                return distB - distA;
            });
            
            for (const ai of sortedAI) {
                ai.render(this.renderer.ctx, this.camera, this.player.z);
            }
            
            // 渲染玩家车辆
            this.player.render(this.renderer.ctx);
            
            this.renderer.ctx.restore();
            
            // 渲染VFX
            this.vfxCtx.clearRect(0, 0, CONFIG.CANVAS_WIDTH, CONFIG.CANVAS_HEIGHT);
            this.particleSystem.render(this.vfxCtx, this.camera);
            
            // 渲染天气
            this.weatherSystem.render(this.vfxCtx, this.dayNight.getTimeOfDay());
            
            // 更新HUD
            this.updateHUD();
        }
        
        updateHUD() {
            document.getElementById('speedValue').textContent = Math.floor(this.player.speed);
            document.getElementById('speedFill').style.width = (this.player.speed / PLAYER_CONFIG.maxSpeed * 100) + '%';
            document.getElementById('nitroFill').style.width = this.player.nitro + '%';
            document.getElementById('timeValue').textContent = this.dayNight.getFormattedTime();
        }
        
        updatePerformancePanel(dt) {
            this.fpsUpdateTimer += dt;
            if (this.fpsUpdateTimer >= 0.5) {
                this.fps = Math.round(1 / dt);
                this.fpsUpdateTimer = 0;
            }
            
            document.getElementById('fpsValue').textContent = this.fps;
            document.getElementById('frameValue').textContent = this.frameCount;
            document.getElementById('particleValue').textContent = this.particleSystem.getCount();
            document.getElementById('aiValue').textContent = AI_CONFIG.count;
            document.getElementById('decorValue').textContent = this.decorationManager.getCount();
        }
    }

    // ============================================
    // 启动游戏
    // ============================================
    window.addEventListener('DOMContentLoaded', () => {
        new Game();
    });
</script>
</body>
</html>

## 实现说明

### 1. 装饰物系统 (`DecorationManager` 和 `Decoration` 类)

- **树木**：带有霓虹效果的赛博朋克风格树木，夜间会发出绿色霓虹光
- **广告牌**：随机显示不同图案的霓虹广告牌
- **建筑**：带窗户网格的建筑物，夜间窗户随机亮灯
- **路灯**：投射光晕效果的路灯，夜间更亮
- **岩石**：路边的自然装饰
- **护栏**：带条纹的霓虹护栏

所有装饰物都有**伪3D投影**和**视差滚动**效果，距离越远雾效越明显。

### 2. 检查点与圈数系统 (`CheckpointSystem` 类)

- **4个检查点**均匀分布在赛道上
- 通过检查点时有视觉提示（通知动画 + HUD更新）
- 完成3圈后显示"RACE COMPLETE"
- **起跑线**使用棋盘格样式渲染

### 3. 天气系统 (`WeatherSystem` 类)

- **Clear**：晴天无效果
- **Rain**：雨滴效果，速度随车速变化
- **Snow**：雪花飘落效果，带有飘摇动画

### 4. 日夜循环系统 (`DayNightCycle` 类)

- **Day/Night/Auto** 三种模式
- Auto模式下120秒完成一个日夜循环
- 天空颜色、草地颜色、环境光都会随时间变化
- 装饰物在不同时间显示不同效果（如路灯夜间更亮）
