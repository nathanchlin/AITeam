const canvas = document.getElementById('gameCanvas');
const ctx = canvas.getContext('2d');
const scale = 32;
canvas.width = 800;
canvas.height = 600;

// 玩家对象
const player = {
  x: 100,
  y: 100,
  width: 32,
  height: 32,
  speed: 5,
  level: 1,
  experience: 0,
  money: 100,
  inventory: []
};

// 物品类
class Item {
  constructor(name, cost) {
    this.name = name;
    this.cost = cost;
  }

  sell() {
    player.money += this.cost;
    const itemIndex = player.inventory.indexOf(this);
    if (itemIndex > -1) {
      player.inventory.splice(itemIndex, 1);
    }
    updateInventory();
    updateTrade();
  }
}

// 更新状态信息
function updateStatus() {
  const statusDiv = document.getElementById('status');
  statusDiv.textContent = `玩家等级：${player.level}，经验：${player.experience}，金币：${player.money}`;
}

// 更新物品列表
function updateInventory() {
  const inventoryDiv = document.getElementById('inventory');
  inventoryDiv.innerHTML = '';
  player.inventory.forEach(item => {
    const itemDiv = document.createElement('div');
    itemDiv.className = 'item';
    itemDiv.textContent = `${item.name}（售价：${item.cost}金币）`;
    itemDiv.onclick = () => item.sell();
    inventoryDiv.appendChild(itemDiv);
  });
}

// 更新交易界面
function updateTrade() {
  const tradeDiv = document.getElementById('trade');
  tradeDiv.style.display = 'block';
  const itemDiv = tradeDiv.querySelector('.item');
  itemDiv.textContent = `金币：${player.money}`;
}

// 添加物品到库存
function addItem(item) {
  player.inventory.push(item);
  updateInventory();
  updateTrade();
}

// 交易系统
function trade() {
  const tradeDiv = document.getElementById('trade');
  tradeDiv.style.display = 'block';
}

// 卖出物品
function sellItem() {
  const item = player.inventory[0]; // 假设玩家只想卖出第一个物品
  if (item) {
    item.sell();
  } else {
    alert('没有物品可以卖出！');
  }
}
