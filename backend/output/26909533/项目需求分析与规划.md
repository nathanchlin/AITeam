# 俄罗斯方块游戏项目需求分析与规划

## 1. 核心需求分析

### 1.1 功能性需求
- **游戏核心机制**：经典的7种方块形状（I、O、T、S、Z、J、L）
- **控制功能**：键盘控制（左右移动、旋转、加速下落）
- **计分系统**：消行计分机制
- **游戏状态管理**：游戏结束检测、开始/暂停功能
- **用户界面**：下一个方块预览、美观的界面设计

### 1.2 非功能性需求
- **响应式设计**：适配不同屏幕尺寸
- **性能优化**：流畅的游戏体验
- **用户体验**：直观的操作和清晰的视觉反馈

## 2. 技术实现方案

### 2.1 前端技术栈
- **HTML5**：语义化标签构建游戏结构
- **CSS3**：游戏样式和动画效果
- **JavaScript**：游戏逻辑和交互控制
- **Canvas API**：游戏区域渲染

### 2.2 架构设计
- **MVC模式**：分离数据模型、视图和控制器
- **模块化设计**：功能模块化，便于维护和扩展

## 3. 开发步骤与里程碑

### 3.1 开发步骤
1. **项目初始化**：创建HTML结构，设置基本样式
2. **游戏核心模块**：实现方块形状、游戏逻辑
3. **用户交互**：添加键盘控制功能
4. **计分系统**：实现消行计分逻辑
5. **游戏状态管理**：添加开始/暂停、游戏结束检测
6. **UI优化**：美化界面，添加动画效果
7. **测试与调试**：全面测试游戏功能

### 3.2 里程碑
1. **里程碑1**：完成基本游戏框架和方块渲染

    <script>
        // 游戏常量
        const COLS = 10;
        const ROWS = 20;
        const BLOCK_SIZE = 30;
        const COLORS = [
            '#FF0D72', '#0DC2FF', '#0DFF72', '#F538FF',
            '#FF8E0D', '#FFE138', '#3877FF'
        ];

        // 方块形状定义
        const SHAPES = [
            [[1, 1, 1, 1]], // I
            [[1, 1], [1, 1]], // O
            [[0, 1, 0], [1, 1, 1]], // T
            [[0, 1, 1], [1, 1, 0]], // S
            [[1, 1, 0], [0, 1, 1]], // Z
            [[1, 0, 0], [1, 1, 1]], // J
            [[0, 0, 1], [1, 1, 1]]  // L
        ];

        // 游戏状态
        let board = [];
        let currentPiece = null;
        let nextPiece = null;
        let score = 0;
        let gameRunning = false;
        let gamePaused = false;
        let dropInterval = null;
        let dropSpeed = 1000;

        // 获取Canvas上下文
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        const nextCanvas = document.getElementById('nextCanvas');
        const nextCtx = nextCanvas.getContext('2d');

        // 方块类
        class Piece {
            constructor(shape, color) {
                this.shape = shape;
                this.color = color;
                this.x = Math.floor(COLS / 2) - Math.floor(shape[0].length / 2);
                this.y = 0;
            }

            // 旋转方块
            rotate() {
                const rotated = [];
                const rows = this.shape.length;
                const cols = this.shape[0].length;
                
                for (let i = 0; i < cols; i++) {
                    rotated[i] = [];
                    for (let j = rows - 1; j >= 0; j--) {
                        rotated[i].push(this.shape[j][i]);
                    }
                }
                
                return rotated;
            }
        }

        // 初始化游戏板
        function initBoard() {
            board = [];
            for (let r = 0; r < ROWS; r++) {
                board[r] = [];
                for (let c = 0; c < COLS; c++) {
                    board[r][c] = 0;
                }
            }
        }

        // 绘制方块
        function drawBlock(ctx, x, y, color) {
            ctx.fillStyle = color;
            ctx.fillRect(x * BLOCK_SIZE, y * BLOCK_SIZE, BLOCK_SIZE, BLOCK_SIZE);
            
            // 添加边框效果
            ctx.strokeStyle = '#333';
            ctx.lineWidth = 1;
            ctx.strokeRect(x * BLOCK_SIZE, y * BLOCK_SIZE, BLOCK_SIZE, BLOCK_SIZE);
        }

        // 绘制游戏板
        function drawBoard() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            // 绘制已固定的方块
            for (let r = 0; r < ROWS; r++) {
                for (let c = 0; c < COLS; c++) {
                    if (board[r][c]) {
                        drawBlock(ctx, c, r, board[r][c]);
                    }
                }
            }
            
            // 绘制当前方块
            if (currentPiece) {
                for (let r = 0; r < currentPiece.shape.length; r++) {
                    for (let c = 0; c < currentPiece.shape[r].length; c++) {
                        if (currentPiece.shape[r][c]) {
                            drawBlock(ctx, currentPiece.x + c, currentPiece.y + r, currentPiece.color);
                        }
                    }
                }
            }
        }

        // 绘制下一个方块
        function drawNext() {
            nextCtx.clearRect(0, 0, nextCanvas.width, nextCanvas.height);
            
            if (nextPiece) {
                const blockSize = 20;
                const offsetX = (nextCanvas.width - nextPiece.shape[0].length * blockSize) / 2;
                const offsetY = (nextCanvas.height - nextPiece.shape.length * blockSize) / 2;
                
                for (let r = 0; r < nextPiece.shape.length; r++) {
                    for (let c = 0; c < nextPiece.shape[r].length; c++) {
                        if (nextPiece.shape[r][c]) {
                            nextCtx.fillStyle = nextPiece.color;
                            nextCtx.fillRect(
                                offsetX + c * blockSize,
                                offsetY + r * blockSize,
                                blockSize,
                                blockSize
                            );
                            
                            nextCtx.strokeStyle = '#333';
                            nextCtx.lineWidth = 1;
                            nextCtx.strokeRect(
                                offsetX + c * blockSize,
                                offsetY + r * blockSize,
                                blockSize,
                                blockSize
                            );
                        }
                    }
                }
            }
        }

        // 检查碰撞
        function checkCollision(piece, dx = 0, dy = 0, newShape = null) {
            const shape = newShape || piece.shape;
            
            for (let r = 0; r < shape.length; r++) {
                for (let c = 0; c < shape[r].length; c++) {
                    if (shape[r][c]) {
                        const newX = piece.x + c + dx;
                        const newY = piece.y + r + dy;
                        
                        if (newX < 0 || newX >= COLS || newY >= ROWS) {
                            return true;
                        }
                        
                        if (newY >= 0 && board[newY][newX]) {
                            return true;
                        }
                    }
                }
            }
            
            return false;
        }

        // 固定方块
        function lockPiece() {
            for (let r = 0; r < currentPiece.shape.length; r++) {
                for (let c = 0; c < currentPiece.shape[r].length; c++) {
                    if (currentPiece.shape[r][c]) {
                        const y = currentPiece.y + r;
                        const x = currentPiece.x + c;
                        
                        if (y >= 0) {
                            board[y][x] = currentPiece.color;
                        }
                    }
                }
            }
            
            clearLines();
            spawnNewPiece();
        }

        // 清除完整的行
        function clearLines() {
            let linesCleared = 0;
            
            for (let r = ROWS - 1; r >= 0; r--) {
                if (board[r].every(cell => cell !== 0)) {
                    board.splice(r, 1);
                    board.unshift(new Array(COLS).fill(0));
                    linesCleared++;
                    r++; // 重新检查当前行
                }
            }
            
            if (linesCleared > 0) {
                // 计分系统
                const points = [0, 100, 300, 500, 800];
                score += points[linesCleared];
                document.getElementById('score').textContent = score;
                
                // 增加游戏速度
                if (score > 0 && score % 500 === 0) {
                    dropSpeed = Math.max(100, dropSpeed - 100);
                    clearInterval(dropInterval);
                    dropInterval = setInterval(drop, dropSpeed);
                }
            }
        }

        // 生成新方块
        function spawnNewPiece() {
            const shapeIndex = Math.floor(Math.random() * SHAPES.length);
            
            if (nextPiece) {
                currentPiece = nextPiece;
                nextPiece = new Piece(SHAPES[shapeIndex], COLORS[shapeIndex]);
            } else {
                currentPiece = new Piece(SHAPES[shapeIndex], COLORS[shapeIndex]);
                nextPiece = new Piece(SHAPES[Math.floor(Math.random() * SHAPES.length)], COLORS[Math.floor(Math.random() * COLORS.length)]);
            }
            
            drawNext();
            
            if (checkCollision(currentPiece)) {
                gameOver();
            }
        }

        // 方块下落
        function drop() {
            if (!gameRunning || gamePaused) return;
            
            if (!checkCollision(currentPiece, 0, 1)) {
                currentPiece.y++;
            } else {
                lockPiece();
            }
            
            drawBoard();
        }

        // 移动方块
        function movePiece(dx) {
            if (!gameRunning || gamePaused) return;
            
            if (!checkCollision(currentPiece, dx, 0)) {
                currentPiece.x += dx;
                drawBoard();
            }
        }

        // 旋转方块
        function rotatePiece() {
            if (!gameRunning || gamePaused) return;
            
            const rotated = currentPiece.rotate();
            if (!checkCollision(currentPiece, 0, 0, rotated)) {
                currentPiece.shape = rotated;
                drawBoard();
            }
        }

        // 硬降（直接落到底部）
        function hardDrop() {
            if (!gameRunning || gamePaused) return;
            
            while (!checkCollision(currentPiece, 0, 1)) {
                currentPiece.y++;
                score += 2;
            }
            
            document.getElementById('score').textContent = score;
            drawBoard();
            lockPiece();
        }

        // 开始游戏
        function startGame() {
            if (gameRunning) return;
            
            initBoard();
            score = 0;
            dropSpeed = 1000;
            document.getElementById('score').textContent = score;
            document.getElementById('gameOver').style.display = 'none';
            document.getElementById('startBtn').textContent = '重新开始';
            
            gameRunning = true;
            gamePaused = false;
            spawnNewPiece();
            drawBoard();
            
            clearInterval(dropInterval);
            dropInterval = setInterval(drop, dropSpeed);
        }

        // 暂停/继续游戏
        function togglePause() {
            if (!gameRunning) return;
            
            gamePaused = !gamePaused;
            document.getElementById('pauseBtn').textContent = gamePaused ? '继续' : '暂停';
        }

        // 游戏结束
        function gameOver() {
            gameRunning = false;
            clearInterval(dropInterval);
            document.getElementById('finalScore').textContent = score;
            document.getElementById('gameOver').style.display = 'block';
        }

        // 重置游戏
        function resetGame() {
            gameRunning = false;
            gamePaused = false;
            clearInterval(dropInterval);
            initBoard();
            currentPiece = null;
            nextPiece = null;
            score = 0;
            document.getElementById('score').textContent = score;
            document.getElementById('gameOver').style.display = 'none';
            document.getElementById('startBtn').textContent = '开始游戏';
