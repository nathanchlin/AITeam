    <script>
        // 游戏配置
        const COLS = 10;
        const ROWS = 20;
        const BLOCK_SIZE = 30;
        const COLORS = [
            '#FF0D72', '#0DC2FF', '#0DFF72', '#F538FF',
            '#FF8E0D', '#FFE138', '#3877FF'
        ];

        // 方块形状定义
        const SHAPES = [
            [[1, 1, 1, 1]], // I
            [[1, 1], [1, 1]], // O
            [[0, 1, 0], [1, 1, 1]], // T
            [[1, 1, 0], [0, 1, 1]], // S
            [[0, 1, 1], [1, 1, 0]], // Z
            [[1, 0, 0], [1, 1, 1]], // J
            [[0, 0, 1], [1, 1, 1]]  // L
        ];

        // 游戏状态
        let board = [];
        let currentPiece = null;
        let nextPiece = null;
        let score = 0;
        let lines = 0;
        let level = 1;
        let gameRunning = false;
        let gamePaused = false;
        let dropInterval = 1000;
        let lastDropTime = 0;
        let animationId = null;
        let clearingLines = [];

        // 获取画布
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        const nextCanvas = document.getElementById('nextPieceCanvas');
        const nextCtx = nextCanvas.getContext('2d');

        // 初始化游戏板
        function initBoard() {
            board = Array(ROWS).fill().map(() => Array(COLS).fill(0));
        }

        // 方块类
        class Piece {
            constructor(shape, color) {
                this.shape = shape;
                this.color = color;
                this.x = Math.floor((COLS - shape[0].length) / 2);
                this.y = 0;
            }

            // 旋转方块
            rotate() {
                const rotated = this.shape[0].map((_, i) =>
                    this.shape.map(row => row[i]).reverse()
                );
                
                // 检查旋转后是否有效
                if (this.isValidMove(this.x, this.y, rotated)) {
                    this.shape = rotated;
                }
            }

            // 检查移动是否有效
            isValidMove(newX, newY, newShape = this.shape) {
                for (let y = 0; y < newShape.length; y++) {
                    for (let x = 0; x < newShape[y].length; x++) {
                        if (newShape[y][x] !== 0) {
                            const boardX = newX + x;
                            const boardY = newY + y;
                            
                            if (boardX < 0 || boardX >= COLS || boardY >= ROWS) {
                                return false;
                            }
                            
                            if (boardY >= 0 && board[boardY][boardX] !== 0) {
                                return false;
                            }
                        }
                    }
                }
                return true;
            }

            // 移动方块
            move(dx, dy) {
                if (this.isValidMove(this.x + dx, this.y + dy)) {
                    this.x += dx;
                    this.y += dy;
                    return true;
                }
                return false;
            }

            // 硬降落（直接落到底部）
            hardDrop() {
                while (this.move(0, 1)) {
                    score += 2;
                }
                updateScore();
            }

            // 固定方块到游戏板
            lock() {
                for (let y = 0; y < this.shape.length; y++) {
                    for (let x = 0; x < this.shape[y].length; x++) {
                        if (this.shape[y][x] !== 0) {
                            const boardY = this.y + y;
                            const boardX = this.x + x;
                            if (boardY >= 0) {
                                board[boardY][boardX] = this.color;
                            }
                        }
                    }
                }
            }
        }

        // 创建新方块
        function createPiece() {
            const shapeIndex = Math.floor(Math.random() * SHAPES.length);
            return new Piece(SHAPES[shapeIndex], COLORS[shapeIndex]);
        }

        // 绘制游戏板
        function drawBoard() {
            // 清空画布
            ctx.fillStyle = '#111';
            ctx.fillRect(0, 0, canvas.width, canvas.height);
            
            // 绘制网格线
            ctx.strokeStyle = '#333';
            ctx.lineWidth = 0.5;
            
            for (let i = 1; i < COLS; i++) {
                ctx.beginPath();
                ctx.moveTo(i * BLOCK_SIZE, 0);
                ctx.lineTo(i * BLOCK_SIZE, canvas.height);
                ctx.stroke();
            }
            
            for (let i = 1; i < ROWS; i++) {
                ctx.beginPath();
                ctx.moveTo(0, i * BLOCK_SIZE);
                ctx.lineTo(canvas.width, i * BLOCK_SIZE);
                ctx.stroke();
            }
            
            // 绘制已固定的方块
            for (let y = 0; y < ROWS; y++) {
                for (let x = 0; x < COLS; x++) {
                    if (board[y][x] !== 0) {
                        drawBlock(x, y, board[y][x]);
                    }
                }
            }
        }

        // 绘制单个方块
        function drawBlock(x, y, color, isClearing = false) {
            const size = isClearing ? BLOCK_SIZE * 1.1 : BLOCK_SIZE;
            const offset = isClearing ? (BLOCK_SIZE - size) / 2 : 0;
            
            // 绘制方块主体
            ctx.fillStyle = color;
            ctx.fillRect(
                x * BLOCK_SIZE + offset, 
                y * BLOCK_SIZE + offset, 
                size, 
                size
            );
            
            // 绘制高光效果
            ctx.fillStyle = 'rgba(255, 255, 255, 0.3)';
            ctx.fillRect(
                x * BLOCK_SIZE + offset, 
                y * BLOCK_SIZE + offset, 
                size * 0.3, 
                size * 0.3
            );
            
            // 绘制阴影效果
            ctx.fillStyle = 'rgba(0, 0, 0, 0.3)';
            ctx.fillRect(
                x * BLOCK_SIZE + size - offset - size * 0.3, 
                y * BLOCK_SIZE + offset, 
                size * 0.3, 
                size
            );
            ctx.fillRect(
                x * BLOCK_SIZE + offset, 
                y * BLOCK_SIZE + size - offset - size * 0.3, 
                size, 
                size * 0.3
            );
        }

        // 绘制当前方块
        function drawPiece(piece) {
            for (let y = 0; y < piece.shape.length; y++) {
                for (let x = 0; x < piece.shape[y].length; x++) {
                    if (piece.shape[y][x] !== 0) {
                        drawBlock(piece.x + x, piece.y + y, piece.color);
                    }
                }
            }
        }

        // 绘制下一个方块
        function drawNextPiece() {
            nextCtx.fillStyle = '#111';
            nextCtx.fillRect(0, 0, nextCanvas.width, nextCanvas.height);
            
            if (nextPiece) {
                const blockSize = 20;
                const offsetX = (nextCanvas.width - nextPiece.shape[0].length * blockSize) / 2;
                const offsetY = (nextCanvas.height - nextPiece.shape.length * blockSize) / 2;
                
                for (let y = 0; y < nextPiece.shape.length; y++) {
                    for (let x = 0; x < nextPiece.shape[y].length; x++) {
                        if (nextPiece.shape[y][x] !== 0) {
                            nextCtx.fillStyle = nextPiece.color;
                            nextCtx.fillRect(
                                offsetX + x * blockSize,
                                offsetY + y * blockSize,
                                blockSize,
                                blockSize
                            );
                            
                            // 绘制高光
                            nextCtx.fillStyle = 'rgba(255, 255, 255, 0.3)';
                            nextCtx.fillRect(
                                offsetX + x * blockSize,
                                offsetY + y * blockSize,
                                blockSize * 0.3,
                                blockSize * 0.3
                            );
                        }
                    }
                }
            }
        }

        // 检查并清除完整的行
        function clearLines() {
            let linesCleared = 0;
            clearingLines = [];
            
            for (let y = ROWS - 1; y >= 0; y--) {
                if (board[y].every(cell => cell !== 0)) {
                    clearingLines.push(y);
                    board.splice(y, 1);
                    board.unshift(Array(COLS).fill(0));
                    linesCleared++;
                    y++; // 重新检查当前行，因为上面的行下移了
                }
            }
            
            if (linesCleared > 0) {
                // 计算得分
                const points = [40, 100, 300, 1200];
                score += points[linesCleared - 1] * level;
                lines += linesCleared;
                
                // 更新等级
                level = Math.floor(lines / 10) + 1;
                dropInterval = Math.max(100, 1000 - (level - 1) * 100);
                
                updateScore();
                
                // 添加清除动画
                animateLineClear();
            }
        }

        // 行清除动画
        function animateLineClear() {
            let animationFrame = 0;
            const maxFrames = 30;
            
            function animate() {
                drawBoard();
                
                // 绘制正在清除的行
                clearingLines.forEach(y => {
                    ctx.fillStyle = 'rgba(255, 255, 255, 0.8)';
                    ctx.fillRect(0, y * BLOCK_SIZE, canvas.width, BLOCK_SIZE);
                });
                
                animationFrame++;
                
                if (animationFrame < maxFrames) {
                    requestAnimationFrame(animate);
                } else {
                    clearingLines = [];
                    drawBoard();
                }
            }
            
            animate();
        }

        // 更新分数显示
        function updateScore() {
            document.getElementById('score').textContent = score;
            document.getElementById('level').textContent = level;
            document.getElementById('lines').textContent = lines;
        }

        // 游戏结束检测
        function checkGameOver() {
            if (currentPiece && !currentPiece.isValidMove(currentPiece.x, currentPiece.y)) {
                gameOver();
                return true;
            }
            return false;
        }

        // 游戏结束
        function gameOver() {
            gameRunning = false;
            document.getElementById('finalScore').textContent = score;
            document.getElementById('gameOverOverlay').classList.add('show');
            cancelAnimationFrame(animationId);
        }

        // 重置游戏
        function resetGame() {
            initBoard();
            score = 0;
            lines = 0;
            level = 1;
            dropInterval = 1000;
            updateScore();
            currentPiece = createPiece();
            nextPiece = createPiece();
            document.getElementById('gameOverOverlay').classList.remove('show');
            drawBoard();
            drawNextPiece();
        }

        function toggleGame() {
            if (!gameRunning) {
