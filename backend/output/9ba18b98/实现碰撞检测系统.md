    <script>
        // 游戏常量
        const COLS = 10;
        const ROWS = 20;
        const BLOCK_SIZE = 30;
        const COLORS = [
            '#FF0D72', '#0DC2FF', '#0DFF72', '#F538FF',
            '#FF8E0D', '#FFE138', '#3877FF'
        ];

        // 方块形状定义
        const SHAPES = [
            [[1, 1, 1, 1]], // I
            [[1, 1], [1, 1]], // O
            [[0, 1, 0], [1, 1, 1]], // T
            [[1, 1, 0], [0, 1, 1]], // S
            [[0, 1, 1], [1, 1, 0]], // Z
            [[1, 0, 0], [1, 1, 1]], // J
            [[0, 0, 1], [1, 1, 1]]  // L
        ];

        // 游戏状态
        let board = [];
        let currentPiece = null;
        let nextPiece = null;
        let score = 0;
        let level = 1;
        let lines = 0;
        let gameRunning = false;
        let gamePaused = false;
        let dropInterval = null;
        let currentSpeed = 1000; // 初始下落速度（毫秒）

        // 获取DOM元素
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        const nextCanvas = document.getElementById('nextCanvas');
        const nextCtx = nextCanvas.getContext('2d');
        const scoreEl = document.getElementById('score');
        const levelEl = document.getElementById('level');
        const linesEl = document.getElementById('lines');
        const gameOverEl = document.getElementById('gameOver');
        const finalScoreEl = document.getElementById('finalScore');
        const startBtn = document.getElementById('startBtn');
        const pauseBtn = document.getElementById('pauseBtn');
        restartBtn = document.getElementById('restartBtn');

        // 初始化游戏板
        function initBoard() {
            board = Array(ROWS).fill().map(() => Array(COLS).fill(0));
        }

        // 方块类
        class Piece {
            constructor(shape, color) {
                this.shape = shape;
                this.color = color;
                this.x = Math.floor((COLS - shape[0].length) / 2);
                this.y = 0;
            }

            // 旋转方块
            rotate() {
                const rows = this.shape.length;
                const cols = this.shape[0].length;
                const rotated = Array(cols).fill().map(() => Array(rows).fill(0));
                
                for (let r = 0; r < rows; r++) {
                    for (let c = 0; c < cols; c++) {
                        rotated[c][rows - 1 - r] = this.shape[r][c];
                    }
                }
                
                return rotated;
            }

            // 检查碰撞
            collide(board, dx = 0, dy = 0, newShape = null) {
                const shape = newShape || this.shape;
                const newX = this.x + dx;
                const newY = this.y + dy;
                
                for (let r = 0; r < shape.length; r++) {
                    for (let c = 0; c < shape[r].length; c++) {
                        if (shape[r][c]) {
                            const boardX = newX + c;
                            const boardY = newY + r;
                            
                            // 检查边界
                            if (boardX < 0 || boardX >= COLS || boardY >= ROWS) {
                                return true;
                            }
                            
                            // 检查与已放置方块的碰撞
                            if (boardY >= 0 && board[boardY][boardX]) {
                                return true;
                            }
                        }
                    }
                }
                
                return false;
            }

            // 移动方块
            move(dx, dy) {
                if (!this.collide(board, dx, dy)) {
                    this.x += dx;
                    this.y += dy;
                    return true;
                }
                return false;
            }

            // 旋转方块（带碰撞检测）
            tryRotate() {
                const rotated = this.rotate();
                if (!this.collide(board, 0, 0, rotated)) {
                    this.shape = rotated;
                    return true;
                }
                
                // 尝试墙踢（Wall Kick）
                const kicks = [
                    [1, 0], [-1, 0], [0, -1], // 右、左、上
                    [-2, 0], [2, 0]           // 更大的左右偏移
                ];
                
                for (const [dx, dy] of kicks) {
                    if (!this.collide(board, dx, dy, rotated)) {
                        this.x += dx;
                        this.y += dy;
                        this.shape = rotated;
                        return true;
                    }
                }
                
                return false;
            }

            // 固定方块到游戏板
            lock() {
                for (let r = 0; r < this.shape.length; r++) {
                    for (let c = 0; c < this.shape[r].length; c++) {
                        if (this.shape[r][c]) {
                            const boardY = this.y + r;
                            const boardX = this.x + c;
                            
                            if (boardY >= 0) {
                                board[boardY][boardX] = this.color;
                            }
                        }
                    }
                }
            }
        }

        // 创建新方块
        function newPiece() {
            const index = Math.floor(Math.random() * SHAPES.length);
            const shape = SHAPES[index];
            const color = COLORS[index];
            
            if (nextPiece) {
                currentPiece = nextPiece;
                nextPiece = new Piece(shape, color);
            } else {
                currentPiece = new Piece(shape, color);
                nextPiece = newPiece();
            }
            
            // 检查游戏是否结束
            if (currentPiece.collide(board)) {
                gameOver();
            }
        }

        // 清除完整的行
        function clearLines() {
            let linesCleared = 0;
            
            for (let r = ROWS - 1; r >= 0; r--) {
                if (board[r].every(cell => cell !== 0)) {
                    board.splice(r, 1);
                    board.unshift(Array(COLS).fill(0));
                    linesCleared++;
                    r++; // 重新检查当前行
                }
            }
            
            if (linesCleared > 0) {
                // 计算得分
                const points = [0, 40, 100, 300, 1200];
                score += points[linesCleared] * level;
                lines += linesCleared;
                
                // 更新等级
                level = Math.floor(lines / 10) + 1;
                currentSpeed = Math.max(100, 1000 - (level - 1) * 100);
                
                // 更新显示
                updateDisplay();
            }
        }

        // 更新显示
        function updateDisplay() {
            scoreEl.textContent = score;
            levelEl.textContent = level;
            linesEl.textContent = lines;
        }

        // 绘制游戏
        function draw() {
            // 清空画布
            ctx.fillStyle = '#111';
            ctx.fillRect(0, 0, canvas.width, canvas.height);
            
            // 绘制网格
            ctx.strokeStyle = '#333';
            ctx.lineWidth = 0.5;
            
            for (let r = 0; r < ROWS; r++) {
                for (let c = 0; c < COLS; c++) {
                    ctx.strokeRect(c * BLOCK_SIZE, r * BLOCK_SIZE, BLOCK_SIZE, BLOCK_SIZE);
                }
            }
            
            // 绘制已固定的方块
            for (let r = 0; r < ROWS; r++) {
                for (let c = 0; c < COLS; c++) {
                    if (board[r][c]) {
                        drawBlock(ctx, c, r, board[r][c]);
                    }
                }
            }
            
            // 绘制当前方块
            if (currentPiece) {
                for (let r = 0; r < currentPiece.shape.length; r++) {
                    for (let c = 0; c < currentPiece.shape[r].length; c++) {
                        if (currentPiece.shape[r][c]) {
                            drawBlock(ctx, currentPiece.x + c, currentPiece.y + r, currentPiece.color);
                        }
                    }
                }
            }
            
            // 绘制下一个方块
            nextCtx.fillStyle = '#111';
            nextCtx.fillRect(0, 0, nextCanvas.width, nextCanvas.height);
            
            if (nextPiece) {
                const blockSize = 20;
                const offsetX = (nextCanvas.width - nextPiece.shape[0].length * blockSize) / 2;
                const offsetY = (nextCanvas.height - nextPiece.shape.length * blockSize) / 2;
                
                for (let r = 0; r < nextPiece.shape.length; r++) {
                    for (let c = 0; c < nextPiece.shape[r].length; c++) {
                        if (nextPiece.shape[r][c]) {
                            drawBlock(nextCtx, offsetX + c * blockSize, offsetY + r * blockSize, 
                                     nextPiece.color, blockSize);
                        }
                    }
                }
            }
        }

        // 绘制单个方块
        function drawBlock(context, x, y, color, size = BLOCK_SIZE) {
            context.fillStyle = color;
            context.fillRect(x * size, y * size, size, size);
            
            // 添加高光效果
            context.fillStyle = 'rgba(255, 255, 255, 0.3)';
            context.fillRect(x * size, y * size, size, 4);
            context.fillRect(x * size, y * size, 4, size);
            
            // 添加阴影效果
            context.fillStyle = 'rgba(0, 0, 0, 0.3)';
            context.fillRect(x * size + size - 4, y * size, 4, size);
            context.fillRect(x * size, y * size + size - 4, size, 4);
        }

        // 游戏循环
        function gameLoop() {
            if (!gameRunning || gamePaused) return;
            
            if (!currentPiece.move(0, 1)) {
                currentPiece.lock();
                clearLines();
                newPiece();
            }
            
            draw();
        }

        // 开始游戏
        function startGame() {
            if (gameRunning) return;
            
            initBoard();
            score = 0;
            level = 1;
            lines = 0;
            currentSpeed = 1000;
            gameRunning = true;
            gamePaused = false;
            
            updateDisplay();
            newPiece();
            
            // 清除现有的下落间隔
            if (dropInterval) {
                clearInterval(dropInterval);
            }
            
            // 设置新的下落间隔
            dropInterval = setInterval(gameLoop, currentSpeed);
            
            draw();
            
            startBtn.textContent = '重新开始';
            gameOverEl.style.display = 'none';
        }

        // 暂停/继续游戏
        function togglePause() {
            if (!gameRunning) return;
            
            gamePaused = !gamePaused;
            pauseBtn.textContent = gamePaused ? '继续' : '暂停';
            
            if (!gamePaused) {
                dropInterval = setInterval(gameLoop, currentSpeed);
            } else {
                clearInterval(dropInterval);
            }
        }

        // 游戏结束
        function gameOver() {
            gameRunning = false;
            clearInterval(dropInterval);
            
            finalScoreEl.textContent = score;
            gameOverEl.style.display = 'block';
            
            // 保存最高分
            const highScore = localStorage.getItem('tetrisHighScore') || 0;
            if (score > highScore) {
                localStorage.setItem('tetrisHighScore', score);
            }
        }

        // 键盘控制
        document
