```html
    <script>
        // 游戏常量定义
        const COLS = 10;
        const ROWS = 20;
        const BLOCK_SIZE = 30;
        const COLORS = [
            '#FF0D72', '#0DC2FF', '#0DFF72', '#F538FF',
            '#FF8E0D', '#FFE138', '#3877FF'
        ];
        
        // 获取Canvas和上下文
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        
        // 游戏状态
        let board = [];
        let score = 0;
        let level = 1;
        let lines = 0;
        let gameRunning = true;
        let isPaused = false;
        let currentPiece = null;
        let dropCounter = 0;
        let lastTime = 0;
        let dropInterval = 1000; // 初始下落间隔1秒
        
        // 方块形状定义
        const SHAPES = [
            [[1, 1, 1, 1]], // I
            [[1, 1], [1, 1]], // O
            [[1, 1, 1], [0, 1, 0]], // T
            [[1, 1, 1], [1, 0, 0]], // L
            [[1, 1, 1], [0, 0, 1]], // J
            [[0, 1, 1], [1, 1, 0]], // S
            [[1, 1, 0], [0, 1, 1]]  // Z
        ];
        
        // 方块类
        class Piece {
            constructor(shape, color) {
                this.shape = shape;
                this.color = color;
                this.x = Math.floor((COLS - shape[0].length) / 2);
                this.y = 0;
            }
            
            // 旋转方块
            rotate() {
                const rows = this.shape.length;
                const cols = this.shape[0].length;
                const rotated = Array(cols).fill().map(() => Array(rows).fill(0));
                
                for (let r = 0; r < rows; r++) {
                    for (let c = 0; c < cols; c++) {
                        rotated[c][rows - 1 - r] = this.shape[r][c];
                    }
                }
                
                return rotated;
            }
        }
        
        // 初始化游戏板
        function initBoard() {
            board = Array(ROWS).fill().map(() => Array(COLS).fill(0));
        }
        
        // 随机生成新方块
        function randomPiece() {
            const shapeIndex = Math.floor(Math.random() * SHAPES.length);
            return new Piece(SHAPES[shapeIndex], COLORS[shapeIndex]);
        }
        
        // 绘制单个方块
        function drawBlock(x, y, color) {
            ctx.fillStyle = color;
            ctx.fillRect(x * BLOCK_SIZE, y * BLOCK_SIZE, BLOCK_SIZE, BLOCK_SIZE);
            
            // 绘制方块边框
            ctx.strokeStyle = '#000';
            ctx.strokeRect(x * BLOCK_SIZE, y * BLOCK_SIZE, BLOCK_SIZE, BLOCK_SIZE);
        }
        
        // 绘制游戏板
        function drawBoard() {
            // 清空画布
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            // 绘制网格线
            ctx.strokeStyle = '#ddd';
            ctx.lineWidth = 0.5;
            
            for (let i = 1; i < COLS; i++) {
                ctx.beginPath();
                ctx.moveTo(i * BLOCK_SIZE, 0);
                ctx.lineTo(i * BLOCK_SIZE, canvas.height);
                ctx.stroke();
            }
            
            for (let i = 1; i < ROWS; i++) {
                ctx.beginPath();
                ctx.moveTo(0, i * BLOCK_SIZE);
                ctx.lineTo(canvas.width, i * BLOCK_SIZE);
                ctx.stroke();
            }
            
            // 绘制已固定的方块
            for (let r = 0; r < ROWS; r++) {
                for (let c = 0; c < COLS; c++) {
                    if (board[r][c]) {
                        drawBlock(c, r, board[r][c]);
                    }
                }
            }
            
            // 绘制当前方块
            if (currentPiece) {
                drawPiece(currentPiece);
            }
        }
        
        // 绘制当前方块
        function drawPiece(piece) {
            piece.shape.forEach((row, r) => {
                row.forEach((value, c) => {
                    if (value) {
                        drawBlock(piece.x + c, piece.y + r, piece.color);
                    }
                });
            });
        }
        
        // 检查碰撞
        function collision(piece, board, moveX = 0, moveY = 0) {
            for (let r = 0; r < piece.shape.length; r++) {
                for (let c = 0; c < piece.shape[r].length; c++) {
                    if (piece.shape[r][c]) {
                        const newX = piece.x + c + moveX;
                        const newY = piece.y + r + moveY;
                        
                        // 检查边界
                        if (newX < 0 || newX >= COLS || newY >= ROWS) {
                            return true;
                        }
                        
                        // 检查与已固定方块的碰撞
                        if (newY >= 0 && board[newY][newX]) {
                            return true;
                        }
                    }
                }
            }
            return false;
        }
        
        // 固定方块到游戏板
        function merge(piece, board) {
            piece.shape.forEach((row, r) => {
                row.forEach((value, c) => {
                    if (value) {
                        const boardY = piece.y + r;
                        const boardX = piece.x + c;
                        if (boardY >= 0) {
                            board[boardY][boardX] = piece.color;
                        }
                    }
                });
            });
        }
        
        // 清除完整的行
        function clearLines() {
            let linesCleared = 0;
            
            for (let r = ROWS - 1; r >= 0; r--) {
                if (board[r].every(cell => cell !== 0)) {
                    board.splice(r, 1);
                    board.unshift(Array(COLS).fill(0));
                    linesCleared++;
                    r++; // 重新检查当前行，因为上面的行下移了
                }
            }
            
            if (linesCleared > 0) {
                lines += linesCleared;
                score += linesCleared * 100 * level;
                
                // 更新等级
                level = Math.floor(lines / 10) + 1;
                dropInterval = Math.max(100, 1000 - (level - 1) * 100);
                
                // 更新UI
                updateScore();
            }
        }
        
        // 更新分数显示
        function updateScore() {
            document.getElementById('score').textContent = score;
            document.getElementById('level').textContent = level;
            document.getElementById('lines').textContent = lines;
        }
        
        // 游戏结束检查
        function gameOver() {
            for (let c = 0; c < COLS; c++) {
                if (board[0][c]) {
                    gameRunning = false;
                    document.getElementById('finalScore').textContent = score;
                    document.getElementById('gameOver').style.display = 'block';
                    return true;
                }
            }
            return false;
        }
        
        // 游戏主循环
        function gameLoop(time = 0) {
            if (!gameRunning || isPaused) {
                requestAnimationFrame(gameLoop);
                return;
            }
            
            const deltaTime = time - lastTime;
            lastTime = time;
            
            dropCounter += deltaTime;
            if (dropCounter > dropInterval) {
                moveDown();
                dropCounter = 0;
            }
            
            drawBoard();
            requestAnimationFrame(gameLoop);
        }
        
        // 移动方块
        function moveDown() {
            if (!collision(currentPiece, board, 0, 1)) {
                currentPiece.y++;
            } else {
                merge(currentPiece, board);
                clearLines();
                
                currentPiece = randomPiece();
                if (collision(currentPiece, board)) {
                    gameOver();
                }
            }
        }
        
        // 左右移动
        function moveHorizontal(dir) {
            if (!collision(currentPiece, board, dir, 0)) {
                currentPiece.x += dir;
            }
        }
        
        // 旋转方块
        function rotatePiece() {
            const rotated = currentPiece.rotate();
            const previousShape = currentPiece.shape;
            currentPiece.shape = rotated;
            
            if (collision(currentPiece, board)) {
                currentPiece.shape = previousShape;
            }
        }
        
        // 硬降落（直接落到底部）
        function hardDrop() {
            while (!collision(currentPiece, board, 0, 1)) {
                currentPiece.y++;
                score += 2;
            }
            updateScore();
            moveDown();
        }
        
        // 键盘控制
        document.addEventListener('keydown', event => {
            if (!gameRunning || isPaused) return;
            
            switch (event.key) {
                case 'ArrowLeft':
                    moveHorizontal(-1);
                    break;
                case 'ArrowRight':
                    moveHorizontal(1);
                    break;
                case 'ArrowDown':
                    moveDown();
                    break;
                case 'ArrowUp':
                    rotatePiece();
                    break;
                case ' ':
                    hardDrop();
                    break;
                case 'p':
                case 'P':
                    isPaused = !isPaused;
                    break;
            }
        });
        
        // 重新开始游戏
        document.getElementById('restartBtn').addEventListener('click', () => {
            initBoard();
            score = 0;
            level = 1;
            lines = 0;
            dropInterval = 1000;
            gameRunning = true;
            isPaused = false;
            currentPiece = randomPiece();
            updateScore();
            document.getElementById('gameOver').style.display = 'none';
        });
        
        // 初始化游戏
        function init() {
            initBoard();
            currentPiece = randomPiece();
            updateScore();
            gameLoop();
        }
        
        // 启动游戏
        init();
    </script>
</body>
</html>
```

## 代码结构说明

### 1. HTML结构
- 创建了主游戏容器，包含Canvas画布和信息面板
- 添加了游戏结束弹窗和重新开始按钮
- 设计了响应式布局，使游戏在不同屏幕上都能良好显示

### 2. CSS样式
- 使用Flexbox布局实现居中显示
- 为Canvas添加了边框和阴影效果
- 设计了信息面板的样式，包括分数、等级和消除行数的显示
- 添加了游戏结束弹窗的样式

### 3. JavaScript游戏逻辑

#### 常量定义
- 定义了游戏板尺寸(COLS, ROWS)、方块大小(BLOCK_SIZE)
- 定义了7种不同颜色用于不同形状的方块

#### 游戏状态
- `board`: 存储游戏板上已固定的方块
- `score`, `level`, `lines`: 游戏分数、等级和消除行数
- `gameRunning`, `isPaused`: 游戏运行和暂停状态
- `currentPiece`: 当前正在下落的方块
- `dropCounter`, `lastTime`: 控制方块下落速度

#### 方块类(Piece)
- 封装了方块的形状、颜色和位置信息
- 提供旋转方法

#### 核心游戏函数
- `initBoard()`: 初始化游戏板
- `randomPiece()`: 随机生成新方块
- `drawBlock()`: 绘制单个方块
- `drawBoard()`: 绘制整个游戏板
- `collision()`: 检查碰撞
- `merge()`: 将方块固定到游戏板
- `clearLines()`: 清除完整的行并更新分数
- `gameLoop()`: 游戏主循环
- `moveDown()`: 方块下落
- `moveHorizontal()`: 左右移动
- `rotatePiece()`: 旋转方块
- `hardDrop()`: 硬降落

#### 事件处理
- 键盘事件：方向键控制移动和旋转，空格键硬降落，P键暂停
- 重新开始按钮：重置游戏状态

## 功能特点

1. **完整的游戏界面**：包含游戏区域、信息面板和操作说明
2. **基础游戏机制**：方块生成、移动、旋转和固定
3. **碰撞检测**：防止方块移出边界或重叠
4. **行消除**：自动检测并消除完整的行
5. **计分系统**：根据消除行数计算分数
6. **等级系统**：随着消除行数增加，游戏速度加快
