        // 游戏常量
        const COLS = 10;
        const ROWS = 20;
        const BLOCK_SIZE = 30;
        const COLORS = [
            '#FF0D72', '#0DC2FF', '#0DFF72', '#F538FF',
            '#FF8E0D', '#FFE138', '#3877FF'
        ];
        
        // 方块形状定义
        const SHAPES = [
            [[1, 1, 1, 1]], // I
            [[1, 1], [1, 1]], // O
            [[1, 1, 1], [0, 1, 0]], // T
            [[1, 1, 1], [1, 0, 0]], // L
            [[1, 1, 1], [0, 0, 1]], // J
            [[0, 1, 1], [1, 1, 0]], // S
            [[1, 1, 0], [0, 1, 1]]  // Z
        ];
        
        // 游戏变量
        let canvas, ctx, nextCanvas, nextCtx;
        let board = [];
        let currentPiece = null;
        let nextPiece = null;
        let score = 0;
        let level = 1;
        let lines = 0;
        let gameRunning = false;
        let gamePaused = false;
        let dropInterval;
        let lastDropTime = 0;
        let dropSpeed = 1000; // 初始下落速度（毫秒）
        
        // 方块类
        class Piece {
            constructor(shape, color) {
                this.shape = shape;
                this.color = color;
                this.x = Math.floor((COLS - shape[0].length) / 2);
                this.y = 0;
            }
            
            // 旋转方块
            rotate() {
                const rows = this.shape.length;
                const cols = this.shape[0].length;
                const rotated = Array(cols).fill().map(() => Array(rows).fill(0));
                
                for (let r = 0; r < rows; r++) {
                    for (let c = 0; c < cols; c++) {
                        rotated[c][rows - 1 - r] = this.shape[r][c];
                    }
                }
                
                return rotated;
            }
        }
        
        // 初始化游戏
        function initGame() {
            canvas = document.getElementById('gameCanvas');
            ctx = canvas.getContext('2d');
            nextCanvas = document.getElementById('nextPieceCanvas');
            nextCtx = nextCanvas.getContext('2d');
            
            // 设置画布大小
            canvas.width = COLS * BLOCK_SIZE;
            canvas.height = ROWS * BLOCK_SIZE;
            
            // 初始化游戏板
            board = Array(ROWS).fill().map(() => Array(COLS).fill(0));
            
            // 重置游戏状态
            score = 0;
            level = 1;
            lines = 0;
            gameRunning = false;
            gamePaused = false;
            dropSpeed = 1000;
            
            // 更新UI
            updateScore();
            updateLevel();
            document.getElementById('gameOverOverlay').style.display = 'none';
            
            // 绘制初始游戏板
            drawBoard();
            drawNextPiece();
        }
        
        // 开始游戏
        function startGame() {
            if (gameRunning) return;
            
            initGame();
            gameRunning = true;
            
            // 生成第一个方块
            nextPiece = createRandomPiece();
            spawnNewPiece();
            
            // 开始游戏循环
            gameLoop();
        }
        
        // 游戏循环
        function gameLoop(timestamp) {
            if (!gameRunning || gamePaused) return;
            
            // 方块自动下落
            if (timestamp - lastDropTime > dropSpeed) {
                moveDown();
                lastDropTime = timestamp;
            }
            
            // 绘制游戏
            draw();
            
            // 继续游戏循环
            requestAnimationFrame(gameLoop);
        }
        
        // 创建随机方块
        function createRandomPiece() {
            const shapeIndex = Math.floor(Math.random() * SHAPES.length);
            return new Piece(SHAPES[shapeIndex], COLORS[shapeIndex]);
        }
        
        // 生成新方块
        function spawnNewPiece() {
            currentPiece = nextPiece;
            nextPiece = createRandomPiece();
            
            // 检查游戏是否结束
            if (collision(currentPiece, 0, 0)) {
                gameOver();
            }
            
            // 绘制下一个方块
            drawNextPiece();
        }
        
        // 碰撞检测
        function collision(piece, dx, dy) {
            for (let r = 0; r < piece.shape.length; r++) {
                for (let c = 0; c < piece.shape[r].length; c++) {
                    if (piece.shape[r][c] !== 0) {
                        const newX = piece.x + c + dx;
                        const newY = piece.y + r + dy;
                        
                        // 检查边界
                        if (newX < 0 || newX >= COLS || newY >= ROWS) {
                            return true;
                        }
                        
                        // 检查与其他方块的碰撞
                        if (newY >= 0 && board[newY][newX] !== 0) {
                            return true;
                        }
                    }
                }
            }
            return false;
        }
        
        // 将方块固定到游戏板上
        function lockPiece() {
            for (let r = 0; r < currentPiece.shape.length; r++) {
                for (let c = 0; c < currentPiece.shape[r].length; c++) {
                    if (currentPiece.shape[r][c] !== 0) {
                        const y = currentPiece.y + r;
                        const x = currentPiece.x + c;
                        if (y >= 0) {
                            board[y][x] = currentPiece.color;
                        }
                    }
                }
            }
            
            // 检查是否有完整的行
            clearLines();
            
            // 生成新方块
            spawnNewPiece();
        }
        
        // 清除完整的行
        function clearLines() {
            let linesCleared = 0;
            
            for (let r = ROWS - 1; r >= 0; r--) {
                if (board[r].every(cell => cell !== 0)) {
                    board.splice(r, 1);
                    board.unshift(Array(COLS).fill(0));
                    linesCleared++;
                    r++; // 重新检查当前行
                }
            }
            
            if (linesCleared > 0) {
                // 更新分数
                const points = [0, 100, 300, 500, 800];
                score += points[linesCleared] * level;
                lines += linesCleared;
                
                // 更新等级
                level = Math.floor(lines / 10) + 1;
                dropSpeed = Math.max(100, 1000 - (level - 1) * 100);
                
                updateScore();
                updateLevel();
            }
        }
        
        // 移动方块
        function moveLeft() {
            if (!gameRunning || gamePaused || !currentPiece) return;
            if (!collision(currentPiece, -1, 0)) {
                currentPiece.x--;
            }
        }
        
        function moveRight() {
            if (!gameRunning || gamePaused || !currentPiece) return;
            if (!collision(currentPiece, 1, 0)) {
                currentPiece.x++;
            }
        }
        
        function moveDown() {
            if (!gameRunning || gamePaused || !currentPiece) return;
            if (!collision(currentPiece, 0, 1)) {
                currentPiece.y++;
            } else {
                lockPiece();
            }
        }
        
        // 旋转方块
        function rotatePiece() {
            if (!gameRunning || gamePaused || !currentPiece) return;
            
            const originalShape = currentPiece.shape;
            currentPiece.shape = currentPiece.rotate();
            
            // 碰撞后尝试调整位置
            if (collision(currentPiece, 0, 0)) {
                // 尝试墙踢
                const kicks = [
                    [0, 0], [-1, 0], [1, 0], [0, -1], [-2, 0], [2, 0]
                ];
                
                for (const [dx, dy] of kicks) {
                    if (!collision(currentPiece, dx, dy)) {
                        currentPiece.x += dx;
                        currentPiece.y += dy;
                        return;
                    }
                }
                
                // 如果所有墙踢都失败，恢复原始形状
                currentPiece.shape = originalShape;
            }
        }
        
        // 硬降（直接落下）
        function hardDrop() {
            if (!gameRunning || gamePaused || !currentPiece) return;
            
            while (!collision(currentPiece, 0, 1)) {
                currentPiece.y++;
                score += 2; // 硬降加分
            }
            
            lockPiece();
            updateScore();
        }
        
        // 暂停/继续游戏
        function togglePause() {
            if (!gameRunning) return;
            
            gamePaused = !gamePaused;
            if (!gamePaused) {
                lastDropTime = performance.now();
                gameLoop();
            }
        }
        
        // 游戏结束
        function gameOver() {
            gameRunning = false;
            document.getElementById('finalScore').textContent = `最终得分: ${score}`;
            document.getElementById('gameOverOverlay').style.display = 'flex';
            
            // 保存最高分（使用localStorage）
            const highScore = localStorage.getItem('tetrisHighScore') || 0;
            if (score > highScore) {
                localStorage.setItem('tetrisHighScore', score);
            }
        }
        
        // 更新分数显示
        function updateScore() {
            document.getElementById('score').textContent = score;
        }
        
        // 更新等级显示
        function updateLevel() {
            document.getElementById('level').textContent = level;
        }
        
        // 绘制游戏板
        function drawBoard() {
            // 清空画布
            ctx.fillStyle = '#111';
            ctx.fillRect(0, 0, canvas.width, canvas.height);
            
            // 绘制网格线
            ctx.strokeStyle = '#333';
            ctx.lineWidth = 0.5;
            
            for (let r = 0; r <= ROWS; r++) {
                ctx.beginPath();
                ctx.moveTo(0, r * BLOCK_SIZE);
                ctx.lineTo(canvas.width, r * BLOCK_SIZE);
                ctx.stroke();
            }
            
            for (let c = 0; c <= COLS; c++) {
                ctx.beginPath();
                ctx.moveTo(c * BLOCK_SIZE, 0);
                ctx.lineTo(c * BLOCK_SIZE, canvas.height);
                ctx.stroke();
            }
            
            // 绘制已固定的方块
            for (let r = 0; r < ROWS; r++) {
                for (let c = 0; c < COLS; c++) {
                    if (board[r][c] !== 0) {
                        drawBlock(c, r, board[r][c]);
                    }
                }
            }
        }

            ctx.lineWidth = 2;

            ctx.fillStyle = 'rgba(255, 255, 255, 0.3)';
            ctx.fillRect(x * BLOCK_SIZE + 2, y
