            this.STORAGE_KEYS = {

            if (!localStorage.getItem(this.STORAGE_KEYS.STATS)) {
                this.saveStats({
                    totalGames: 0,
                    wins: 0,
                    losses: 0,
                    draws: 0,
                    currentStreak: 0,

                        cards: []
                    }
                });
            }

        getStats() {
            try {
                return JSON.parse(localStorage.getItem(this.STORAGE_KEYS.STATS));
            } catch (e) {
                return this.getDefaultStats();
            }
        }
        
        saveStats(stats) {
            localStorage.setItem(this.STORAGE_KEYS.STATS, JSON.stringify(stats));
        }
        
        updateStats(updates) {
            const stats = this.getStats();
            Object.assign(stats, updates);
            this.saveStats(stats);
            return stats;
        }
        
        recordGameResult(won, roundsWon, roundsLost, playTime, cardsPlayed) {
            const stats = this.getStats();
            stats.totalGames++;
            
            if (won) {
                stats.wins++;
                stats.currentStreak++;
                stats.bestStreak = Math.max(stats.bestStreak, stats.currentStreak);
            } else {
                stats.losses++;
                stats.currentStreak = 0;
            }
            
            stats.roundsWon += roundsWon;
            stats.roundsLost += roundsLost;
            stats.totalPlayTime += playTime;
            stats.cardsPlayed += cardsPlayed;
            
            this.saveStats(stats);
            return stats;
        }

    // ==================== 游戏状态管理器 ====================
    class GameStateManager {
        constructor(persistenceManager) {
            this.persistence = persistenceManager;
            this.settings = persistenceManager.getSettings();
            this.reset();
        }
        
        reset() {
            // 游戏阶段
            this.phase = GamePhase.INIT;
            this.currentRound = 1;
            this.maxRounds = 3;
            
            // 回合权
            this.isPlayerTurn = true;
            this.playerPassed = false;
            this.aiPassed = false;
            
            // 系列赛比分
            this.playerRoundWins = 0;
            this.aiRoundWins = 0;
            
            // 换牌阶段
            this.mulliganCount = this.settings.mulliganCount;
            this.selectedForMulligan = [];
            
            // 玩家数据
            this.playerHand = [];
            this.playerDeck = [];
            this.playerBoard = { melee: [], ranged: [], siege: [] };
            this.playerGrave = [];
            this.playerLeaderUsed = false;
            
            // AI数据
            this.aiHand = [];
            this.aiDeck = [];
            this.aiBoard = { melee: [], ranged: [], siege: [] };
            this.aiGrave = [];
            this.aiLeaderUsed = false;
            
            // 天气
            this.weather = { melee: null, ranged: null, siege: null };
            
            // 游戏计时
            this.gameStartTime = Date.now();
            this.cardsPlayedThisGame = 0;
            
            // 动画状态
            this.animations = [];
            this.floatingTexts = [];
        }
        
        // 初始化游戏
        initializeGame() {
            this.reset();
            this.buildDecks();
            this.drawStartingHands();
            this.phase = GamePhase.MULLIGAN;
            this.mulliganCount = this.settings.mulliganCount;
            return this.getState();
        }
        
        // 构建卡组
        buildDecks() {
            const playerFaction = this.settings.playerFaction;
            const aiFaction = this.settings.aiFaction;
            
            // 玩家卡组
            this.playerDeck = this.shuffleArray([...CardDatabase[playerFaction]]);
            
            // AI卡组
            this.aiDeck = this.shuffleArray([...CardDatabase[aiFaction]]);
        }
        
        // 抽起始手牌
        drawStartingHands() {
            const handSize = this.settings.startingHand;
            
            for (let i = 0; i < handSize; i++) {
                if (this.playerDeck.length > 0) {
                    this.playerHand.push(this.playerDeck.pop());
                }
                if (this.aiDeck.length > 0) {
                    this.aiHand.push(this.aiDeck.pop());
                }
            }
        }
        
        // 换牌操作
        mulliganCard(cardIndex) {
            if (this.phase !== GamePhase.MULLIGAN || this.mulliganCount <= 0) return false;
            if (cardIndex < 0 || cardIndex >= this.playerHand.length) return false;
            
            const card = this.playerHand[cardIndex];
            
            // 把卡放回卡组
            this.playerDeck.push(card);
            this.playerDeck = this.shuffleArray(this.playerDeck);
            
            // 抽一张新卡
            this.playerHand[cardIndex] = this.playerDeck.pop();
            
            this.mulliganCount--;
            
            return true;
        }
        
        // 确认换牌完成
        confirmMulligan() {
            if (this.phase !== GamePhase.MULLIGAN) return false;
            
            this.phase = GamePhase.ROUND_PLAY;
            this.isPlayerTurn = true; // 玩家先手
            this.playerPassed = false;
            this.aiPassed = false;
            
            return true;
        }
        
        // 打出卡牌
        playCard(cardIndex, targetRow, isPlayer = true) {
            const hand = isPlayer ? this.playerHand : this.aiHand;
            const board = isPlayer ? this.playerBoard : this.aiBoard;
            
            if (cardIndex < 0 || cardIndex >= hand.length) return null;
            
            const card = hand.splice(cardIndex, 1)[0];
            
            // 确定放置位置
            let row = targetRow || card.row;
            row = row.toLowerCase();
            
            if (board[row]) {
                board[row].push(card);
            }
            
            this.cardsPlayedThisGame++;
            
            // 触发卡牌能力
            this.triggerCardAbility(card, isPlayer);
            
            return card;
        }
        
        // 触发卡牌能力
        triggerCardAbility(card, isPlayer) {
            // 简化的能力触发
            if (card.abilities && card.abilities.length > 0) {
                card.abilities.forEach(ability => {
                    switch (ability) {
                        case 'SPY':
                            // 间谍：给对手抽2张牌
                            const deck = isPlayer ? this.playerDeck : this.aiDeck;
                            const hand = isPlayer ? this.playerHand : this.aiHand;
                            for (let i = 0; i < 2 && deck.length > 0; i++) {
                                hand.push(deck.pop());
                            }
                            break;
                        case 'MUSTER':
                            // 集结：从卡组召唤同名卡
                            this.summonSameCards(card, isPlayer);
                            break;
                        // 其他能力...
                    }
                });
            }
        }
        
        // 集结能力
        summonSameCards(card, isPlayer) {
            const deck = isPlayer ? this.playerDeck : this.aiDeck;
            const board = isPlayer ? this.playerBoard : this.aiBoard;
            const row = card.row.toLowerCase();
            
            const sameCards = deck.filter(c => c.name === card.name);
            sameCards.forEach(c => {
                const index = deck.indexOf(c);
                if (index > -1) {
                    deck.splice(index, 1);
                    board[row].push(c);
                }
            });
        }
        
        // Pass操作
        pass(isPlayer = true) {
            if (isPlayer) {
                this.playerPassed = true;
            } else {
                this.aiPassed = true;
            }
            
            // 检查回合是否结束
            if (this.playerPassed && this.aiPassed) {
                this.endRound();
            } else {
                // 切换回合权
                this.switchTurn();
            }
        }
        
        // 切换回合权
        switchTurn() {
            // 如果一方已pass，回合权不切换
            if (this.playerPassed && !this.aiPassed) {
                this.isPlayerTurn = false;
            } else if (this.aiPassed && !this.playerPassed) {
                this.isPlayerTurn = true;
            } else {
                this.isPlayerTurn = !this.isPlayerTurn;
            }
        }
        
        // 结算回合
        endRound() {
            const playerPower = this.calculateTotalPower(this.playerBoard, this.weather);
            const aiPower = this.calculateTotalPower(this.aiBoard, this.weather);
            
            let roundWinner = null;
            if (playerPower > aiPower) {
                this.playerRoundWins++;
                roundWinner = 'player';
            } else if (aiPower > playerPower) {
                this.aiRoundWins++;
                roundWinner = 'ai';
            } else {
                // 平局，双方各得一分
                this.playerRoundWins++;
                this.aiRoundWins++;
                roundWinner = 'draw';
            }
            
            // 检查游戏是否结束
            if (this.playerRoundWins >= 2 || this.aiRoundWins >= 2) {
                this.endGame();
                return { gameOver: true, winner: this.playerRoundWins >= 2 ? 'player' : 'ai' };
            }
            
            // 准备下一回合
            this.prepareNextRound();
            
            return { 
                gameOver: false, 
                roundWinner, 
                playerPower, 
                aiPower,
                playerRoundWins: this.playerRoundWins,
                aiRoundWins: this.aiRoundWins
            };
        }
        
        // 准备下一回合
        prepareNextRound() {
            // 清空战场，卡牌进入墓地
            ['melee', 'ranged', 'siege'].forEach(row => {
                this.playerGrave.push(...this.playerBoard[row]);
                this.aiGrave.push(...this.aiBoard[row]);
                this.playerBoard[row] = [];
                this.aiBoard[row] = [];
            });
            
            // 重置天气
            this.weather = { melee: null, ranged: null, siege: null };
            
            // 重置pass状态
            this.playerPassed = false;
            this.aiPassed = false;
            
            // 上一回合输家先手
            this.isPlayerTurn = this.playerRoundWins <= this.aiRoundWins;
            
            // 增加回合数
            this.currentRound++;
            
            // 抽2张牌
            for (let i = 0; i < 2; i++) {
                if (this.playerDeck.length > 0) {
                    this.playerHand.push(this.playerDeck.pop());
                }
                if (this.aiDeck.length > 0) {
                    this.aiHand.push(this.aiDeck.pop());
                }
            }
            
            // 进入换牌阶段
            this.phase = GamePhase.MULLIGAN;
            this.mulliganCount = 1; // 后续回合只有1次换牌机会
        }
        
        // 结束游戏
        endGame() {
            this.phase = GamePhase.GAME_OVER;
            
            const playTime = Math.floor((Date.now() - this.gameStartTime) / 1000);
            const playerWon = this.playerRoundWins >= 2;
            
            // 记录统计
            this.persistence.recordGameResult(
                playerWon,
                this.playerRoundWins,
                this.aiRoundWins,
                playTime,
                this.cardsPlayedThisGame
            );
        }
        
        // 计算总战力
        calculateTotalPower(board, weather) {
            let total = 0;
            
            ['melee', 'ranged', 'siege'].forEach(row => {
                board[row].forEach(card => {
                    if (card.type === 'UNIT') {
                        let power = card.power;
                        
                        // 天气效果
                        if (weather[row]) {
                            power = 1;
                        }
                        
                        total += power;
                    }
                });
            });
            
            return total;
        }
        
        // 获取完整状态
        getState() {
            return {
                phase: this.phase,
                currentRound: this.currentRound,
                isPlayerTurn: this.isPlayerTurn,
                playerPassed: this.playerPassed,
                aiPassed: this.aiPassed,
                playerRoundWins: this.playerRoundWins,
                aiRoundWins: this.aiRoundWins,
                mulliganCount: this.mulliganCount,
                playerHand: this.playerHand,
                playerDeck: this.playerDeck,
                playerBoard: this.playerBoard,
                playerGrave: this.playerGrave,
                playerLeaderUsed: this.playerLeaderUsed,
                aiHand: this.aiHand,
                aiDeck: this.aiDeck,
                aiBoard: this.aiBoard,
                aiGrave: this.aiGrave,
                aiLeaderUsed: this.aiLeaderUsed,
                weather: this.weather,
                playerPower: this.calculateTotalPower(this.playerBoard, this.weather),
                aiPower: this.calculateTotalPower(this.aiBoard, this.weather)
            };
        }
        
        // 工具函数
        shuffleArray(array) {
            const newArray = [...array];
            for (let i = newArray.length - 1; i > 0; i--) {
                const j = Math.floor(Math.random() * (i + 1));
                [newArray[i], newArray[j]] = [newArray[j], newArray[i]];
            }
            return newArray;
        }
    }
    
    // ==================== AI决策系统 ====================
    class AIDecisionSystem {
        constructor(difficulty = 'NORMAL') {
            this.difficulty = difficulty;
            this.config = this.getDifficultyConfig(difficulty);
            this.decisionHistory = [];
        }
        
        getDifficultyConfig(difficulty) {
            const configs = {
                EASY: {
                    randomFactor: 0.4,
                    passThreshold: -20,
                    passLeadThreshold: 25,
                    mistakeChance: 0.35,
                    thinkTime: 600
                },
                NORMAL: {
                    randomFactor: 0.2,
                    passThreshold: -12,
                    passLeadThreshold: 18,
                    mistakeChance: 0.15,
                    thinkTime: 1000
                },
                HARD: {
                    randomFactor: 0.05,
                    passThreshold: -6,
                    passLeadThreshold: 12,
                    mistakeChance: 0.02,
                    thinkTime: 1500
                }
            };
            return configs[difficulty] || configs.NORMAL;
        }
        
        setDifficulty(difficulty) {
            this.difficulty = difficulty;
            this.config = this.getDifficultyConfig(difficulty);
        }
        
        makeDecision(gameState) {
            const { aiHand, aiBoard, playerBoard, round, playerPassed, aiPassed,
                    aiRoundWins, playerRoundWins, weather } = gameState;
            
            const aiPower = this.calculateTotalPower(aiBoard, weather);
            const playerPower = this.calculateTotalPower(playerBoard, weather);
            const powerGap = aiPower - playerPower;
            const cardAdvantage = aiHand.length - gameState.playerHand.length;
            
            // 检查是否应该pass
            if (this.shouldPass(gameState, powerGap, cardAdvantage)) {
                return { action: 'PASS' };
            }
            
            // 评估卡牌
            const cardScores = aiHand.map((card, index) => ({
                card,
                index,
                score: this.evaluateCard(card, gameState)
            }));
            
            // 排序选择最佳
            cardScores.sort((a, b) => b.score - a.score);
            
            // 随机因素
            if (Math.random() < this.config.mistakeChance) {
                const randomIndex = Math.floor(Math.random() * cardScores.length);
                return {
                    action: 'PLAY',
                    cardIndex: cardScores[randomIndex].index,
                    targetRow: cardScores[randomIndex].card.row
                };
            }
            
            const selected = cardScores[0];
            return {
                action: 'PLAY',
                cardIndex: selected.index,
                targetRow: selected.card.row
            };
        }
        
        shouldPass(gameState, powerGap, cardAdvantage) {
            const { round, playerPassed, aiPassed, aiRoundWins, playerRoundWins } = gameState;
            
            if (aiPassed) return false;
            
            if (playerPassed) {
                return powerGap > 0;
            }
            
            // 第一回合大幅领先
            if (round === 1 && powerGap > this.config.passLeadThreshold && cardAdvantage >= 0) {
                return Math.random() < 0.7;
            }
            
            // 落后太多
            if (powerGap < this.config.passThreshold) {
                return Math.random() < 0.6;
            }
            
            return false;
        }
        
        evaluateCard(card, gameState) {
            let score = card.power || 5;
            
            // 特殊卡评估
            if (card.type === 'SPECIAL') {
                if (card.abilityType === 'SCORCH') {
                    score = this.evaluateScorch(gameState);
                } else if (card.abilityType === 'COMMANDERS_HORN') {
                    score = 15;
                }
            }
            
            // 间谍卡高价值（给对手牌但获得情报）
            if (card.abilities && card.abilities.includes('SPY')) {
                score += 10;
            }
            
            // 医疗卡在后期更有价值
            if (card.abilities && card.abilities.includes('MEDIC')) {
                if (gameState.aiGrave && gameState.aiGrave.length > 0) {
                    score += 8;
                }
            }
            
            return score;
        }
        
        evaluateScorch(gameState) {
            const maxPlayerPower = this.getMaxPower(gameState.playerBoard);
            const maxAiPower = this.getMaxPower(gameState.aiBoard);
            
            // 如果会烧到自己最强单位，降低价值
            if (maxAiPower >= maxPlayerPower) {
                return 2;
            }
            return maxPlayerPower > 8 ? 20 : 5;
        }
        
        getMaxPower(board) {
            let max = 0;
            ['melee', 'ranged', 'siege'].forEach(row => {
                board[row].forEach(card => {
                    if (card.power > max) max = card.power;
                });
            });
            return max;
        }
        
        calculateTotalPower(board, weather) {
            let total = 0;
            ['melee', 'ranged', 'siege'].forEach(row => {
                board[row].forEach(card => {
                    if (card.type === 'UNIT') {
                        total += weather[row] ? 1 : card.power;
                    }
                });
            });
            return total;
        }
    }

    class GameRenderer {
        constructor(canvas, ctx) {
            this.canvas = canvas;
            this.ctx = ctx;
            this.particles = [];
            this.animationFrame = null;
        }
        
        clear() {
            this.ctx.fillStyle = CONFIG.COLORS.BG_PRIMARY;
            this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);
        }
        
        // 绘制战场背景
        drawBattlefield() {
            const ctx = this.ctx;
            const w = this.canvas.width;
            const h = this.canvas.height;
            
            // 渐变背景
            const gradient = ctx.createLinearGradient(0, 0, 0, h);
            gradient.addColorStop(0, '#0a0a12');
            gradient.addColorStop(0.5, '#0d0d15');
            gradient.addColorStop(1, '#080810');
            ctx.fillStyle = gradient;
            ctx.fillRect(0, 0, w, h);
            
            // 战场区域
            const rowHeight = h / 7;
            
            // AI区域 (上方)
            ctx.fillStyle = 'rgba(30, 20, 35, 0.5)';
            ctx.fillRect(50, 50, w - 100, rowHeight * 3);
            
            // 玩家区域 (下方)
            ctx.fillStyle = 'rgba(20, 30, 35, 0.5)';
            ctx.fillRect(50, h - 50 - rowHeight * 3, w - 100, rowHeight * 3);
            
            // 中间区域
            ctx.fillStyle = 'rgba(25, 25, 30, 0.3)';
            ctx.fillRect(50, 50 + rowHeight * 3, w - 100, rowHeight);
            
            // 绘制网格线
            ctx.strokeStyle = 'rgba(60, 60, 70, 0.3)';
            ctx.lineWidth = 1;
            
            for (let i = 1; i < 7; i++) {
                ctx.beginPath();
                ctx.moveTo(50, 50 + i * rowHeight);
                ctx.lineTo(w - 50, 50 + i * rowHeight);
                ctx.stroke();
            }
            
            // 区域标签
            ctx.font = '12px Cinzel';
            ctx.fillStyle = '#5a5a65';
            ctx.textAlign = 'left';
            
            const labels = ['攻城', '远程', '近战', '天气', '近战', '远程', '攻城'];
            labels.forEach((label, i) => {
                ctx.fillText(label, 10, 65 + i * rowHeight);
            });
        }
        
        // 绘制卡牌
        drawCard(card, x, y, scale = 1, isSelected = false, isHovered = false) {
            const ctx = this.ctx;
            const w = CONFIG.CARD_WIDTH * scale;
            const h = CONFIG.CARD_HEIGHT * scale;
            
            ctx.save();
            ctx.translate(x, y);
            
            // 卡牌背景
            let bgColor = '#1a1a20';
            let borderColor = CONFIG.COLORS.CARD_BORDER;
            
            if (card.color === 'GOLD') {
                bgColor = '#2a2515';
                borderColor = CONFIG.COLORS.GOLD_PRIMARY;
            } else if (card.color === 'SPECIAL') {
                bgColor = '#1a2520';
                borderColor = '#356a55';
            }
            
            if (isSelected) {
                borderColor = CONFIG.COLORS.CARD_SELECTED;
                ctx.shadowColor = CONFIG.COLORS.CARD_SELECTED;
                ctx.shadowBlur = 15;
            }
            
            if (isHovered) {
                ctx.shadowColor = borderColor;
                ctx.shadowBlur = 20;
            }
            
            // 卡牌主体
            ctx.fillStyle = bgColor;
            ctx.strokeStyle = borderColor;
            ctx.lineWidth = 2;
            
            // 圆角矩形
            this.roundRect(-w/2, -h/2, w, h, 5);
            ctx.fill();
            ctx.stroke();
            
            ctx.shadowBlur = 0;
            
            // 卡牌图像区域
            ctx.fillStyle = '#0d0d10';
            ctx.fillRect(-w/2 + 5, -h/2 + 5, w - 10, h * 0.5);
            
            // 战力数值
            if (card.type === 'UNIT') {
                const powerY = -h/2 + h * 0.7;
                
                ctx.font = `bold ${16 * scale}px Cinzel`;
                ctx.textAlign = 'center';
                ctx.fillStyle = card.color === 'GOLD' ? 
                    CONFIG.COLORS.GOLD_LIGHT : '#d4c4a8';
                ctx.fillText(card.power.toString(), 0, powerY);
            }
            
            // 卡牌名称
            ctx.font = `${8 * scale}px Cinzel`;
            ctx.textAlign = 'center';
            ctx.fillStyle = '#a0a0a0';
            ctx.fillText(card.name.substring(0, 6), 0, h/2 - 8);
            
            ctx.restore();
            
            return { x: x - w/2, y: y - h/2, w, h };
        }
        
        roundRect(x, y, w, h, r) {
            this.ctx.beginPath();
            this.ctx.moveTo(x + r, y);
            this.ctx.lineTo(x + w - r, y);
            this.ctx.quadraticCurveTo(x + w, y, x + w, y + r);
            this.ctx.lineTo(x + w, y + h - r);
            this.ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
            this.ctx.lineTo(x + r, y + h);
            this.ctx.quadraticCurveTo(x, y + h, x, y + h - r);
            this.ctx.lineTo(x, y + r);
            this.ctx.quadraticCurveTo(x, y, x + r, y);
            this.ctx.closePath();
        }
        
        // 绘制手牌
        drawHand(hand, y, isPlayer = true, selectedIndex = -1, hoveredIndex = -1) {
            const ctx = this.ctx;
            const w = this.canvas.width;
            const cardSpacing = Math.min(80, (w - 200) / hand.length);
            const startX = w / 2 - (hand.length - 1) * cardSpacing / 2;
            
            const bounds = [];
            
            hand.forEach((card, i) => {
                const x = startX + i * cardSpacing;
                const scale = i === hoveredIndex ? CONFIG.HOVER_SCALE : 1;
                const offsetY = isPlayer ? 0 : 0;
                
                const bound = this.drawCard(
                    card, 
                    x, 
                    y + offsetY, 
                    scale,
                    i === selectedIndex,
                    i === hoveredIndex
                );
                
                bounds.push({ index: i, ...bound });
            });
            
            return bounds;
        }
        
        // 绘制战场上的卡牌
        drawBoardCards(board, weather, isPlayer = true) {
            const ctx = this.ctx;
            const w = this.canvas.width;
            const h = this.canvas.height;
            const rowHeight = h / 7;
            
            const rows = ['siege', 'ranged', 'melee'];
            const baseY = isPlayer ? h - 50 - rowHeight * 1.5 : 50 + rowHeight * 1.5;
            
            rows.forEach((row, rowIndex) => {
                const cards = board[row];
                const y = isPlayer ? 
                    baseY - rowIndex * rowHeight :
                    baseY + rowIndex * rowHeight;
                
                const cardSpacing = Math.min(60, (w - 200) / Math.max(cards.length, 1));
                const startX = w / 2 - (cards.length - 1) * cardSpacing / 2;
                
                cards.forEach((card, i) => {
                    const x = startX + i * cardSpacing;
                    let displayPower = card.power;
                    
                    // 天气效果
                    if (weather[row]) {
                        displayPower = 1;
                    }
                    
                    this.drawCard(card, x, y, 0.8);
                });
            });
        }
        
        // 绘制战力显示
        drawPowerDisplay(playerPower, aiPower) {
            const ctx = this.ctx;
            const w = this.canvas.width;
            const h = this.canvas.height;
            
            // 玩家战力
            ctx.font = 'bold 36px Cinzel';
            ctx.textAlign = 'center';
            ctx.fillStyle = CONFIG.COLORS.PLAYER_POWER;
            ctx.shadowColor = CONFIG.COLORS.PLAYER_POWER;
            ctx.shadowBlur = 10;
            ctx.fillText(playerPower.toString(), w - 80, h - 100);
            
            // AI战力
            ctx.fillStyle = CONFIG.COLORS.ENEMY_POWER;
            ctx.shadowColor = CONFIG.COLORS.ENEMY_POWER;
            ctx.fillText(aiPower.toString(), w - 80, 130);
            
            ctx.shadowBlur = 0;
        }
        
        // 绘制回合指示
        drawTurnIndicator(isPlayerTurn, playerPassed, aiPassed) {
            const ctx = this.ctx;
            const w = this.canvas.width;
            const h = this.canvas.height;
            
            ctx.font = '16px Cinzel';
            ctx.textAlign = 'center';
            
            if (playerPassed && !aiPassed) {
                ctx.fillStyle = '#f87171';
                ctx.fillText('AI回合', w/2, 30);
            } else if (aiPassed && !playerPassed) {
                ctx.fillStyle = '#4ade80';
                ctx.fillText('你的回合', w/2, h - 20);
            } else if (isPlayerTurn) {
                ctx.fillStyle = '#4ade80';
                ctx.fillText('你的回合', w/2, h - 20);
            } else {
                ctx.fillStyle = '#f87171';
                ctx.fillText('AI思考中...', w/2, 30);
            }
        }
        
        // 添加粒子
        addParticle(x, y, color, type = 'normal') {
            this.particles.push({
                x, y,
                vx: (Math.random() - 0.5) * 4,
                vy: -Math.random() * 3 - 1,
                life: 1,
                color,
                size: Math.random() * 3 + 2,
                type
            });
        }
        
        // 更新粒子
        updateParticles() {
            this.particles = this.particles.filter(p => {
                p.x += p.vx;
                p.y += p.vy;
                p.vy += 0.1;
                p.life -= 0.02;
                return p.life > 0;
            });
        }
        
        // 绘制粒子
        drawParticles() {
            const ctx = this.ctx;
            this.particles.forEach(p => {
                ctx.globalAlpha = p.life;
                ctx.fillStyle = p.color;
                ctx.beginPath();
                ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
                ctx.fill();
            });
            ctx.globalAlpha = 1;
        }
        
        // 绘制浮动文字
        drawFloatingTexts() {
            const ctx = this.ctx;
            this.floatingTexts = this.floatingTexts.filter(ft => {
                ft.y -= 1;
                ft.life -= 0.02;
                
                if (ft.life > 0) {
                    ctx.globalAlpha = ft.life;
                    ctx.font = `bold ${ft.size}px Cinzel`;
                    ctx.fillStyle = ft.color;
                    ctx.textAlign = 'center';
                    ctx.fillText(ft.text, ft.x, ft.y);
                    ctx.globalAlpha = 1;
                }
                
                return ft.life > 0;
            });
        }
        
        addFloatingText(x, y, text, color, size = 20) {
            this.floatingTexts.push({ x, y, text, color, size, life: 1 });
        }
    }
    
    // ==================== 主游戏类 ====================
    class GwentGame {
        constructor() {
            this.canvas = document.getElementById('gameCanvas');
            this.ctx = this.canvas.getContext('2d');
            
            this.persistence = new PersistenceManager();
            this.stateManager = new GameStateManager(this.persistence);
            this.ai = new AIDecisionSystem(this.persistence.getSettings().difficulty);
            this.renderer = new GameRenderer(this.canvas, this.ctx);
            
            this.selectedCardIndex = -1;
            this.hoveredCardIndex = -1;
            this.cardBounds = [];
            this.isDragging = false;
            this.dragPos = { x: 0, y: 0 };
            
            this.aiThinking = false;
            this.roundEndAnimation = null;
            
            this.init();
        }
        
        init() {
            this.resizeCanvas();
            window.addEventListener('resize', () => this.resizeCanvas());
            
            this.setupEventListeners();
            this.updateSettingsUI();
            this.updateStatsUI();
            
            this.startNewGame();
            this.gameLoop();
        }
        
        resizeCanvas() {
            this.canvas.width = window.innerWidth;
            this.canvas.height = window.innerHeight;
        }
        
        setupEventListeners() {
            // 鼠标事件
            this.canvas.addEventListener('mousemove', (e) => this.handleMouseMove(e));
            this.canvas.addEventListener('mousedown', (e) => this.handleMouseDown(e));
            this.canvas.addEventListener('mouseup', (e) => this.handleMouseUp(e));
            
            // 按钮事件
            document.getElementById('btnPass').addEventListener('click', () => this.playerPass());
            document.getElementById('btnLeader').addEventListener('click', () => this.useLeaderAbility());
            document.getElementById('btnConfirm').addEventListener('click', () => this.confirmMulligan());
            document.getElementById('btnNewGame').addEventListener('click', () => this.startNewGame());
            document.getElementById('btnSettings').addEventListener('click', () => this.toggleSettings());
            document.getElementById('btnSaveSettings').addEventListener('click', () => this.saveSettings());
            document.getElementById('btnCloseSettings').addEventListener('click', () => this.toggleSettings());
            document.getElementById('btnPlayAgain').addEventListener('click', () => {
                document.getElementById('gameOverModal').classList.remove('show');
                this.startNewGame();
            });
            
            // 难度按钮
            document.querySelectorAll('.difficulty-btn').forEach(btn => {
                btn.addEventListener('click', (e) => {
                    document.querySelectorAll('.difficulty-btn').forEach(b => b.classList.remove('active'));
                    e.target.classList.add('active');
                    const difficulty = e.target.dataset.difficulty;
                    this.ai.setDifficulty(difficulty);
                    this.persistence.updateStats({}); // 触发保存
                });
            });
            
            // 天气按钮
            document.querySelectorAll('.weather-btn').forEach(btn => {
                btn.addEventListener('click', (e) => {
                    document.querySelectorAll('.weather-btn').forEach(b => b.classList.remove('active'));
                    e.target.classList.add('active');
                });
            });
        }
        
        handleMouseMove(e) {
            const rect = this.canvas.getBoundingClientRect();
            const x = e.clientX - rect.left;
            const y = e.clientY - rect.top;
            
            if (this.isDragging) {
                this.dragPos = { x, y };
                return;
            }
            
            // 检测手牌悬停
            this.hoveredCardIndex = -1;
            this.cardBounds.forEach((bound, i) => {
                if (x >= bound.x && x <= bound.x + bound.w &&
                    y >= bound.y && y <= bound.y + bound.h) {
                    this.hoveredCardIndex = i;
                }
            });
        }
        
        handleMouseDown(e) {
            const state = this.stateManager.getState();
            
            if (state.phase === GamePhase.MULLIGAN) {
                // 换牌阶段 - 点击选择换牌
                if (this.hoveredCardIndex >= 0) {
                    this.stateManager.mulliganCard(this.hoveredCardIndex);
                    this.renderer.addFloatingText(
                        this.canvas.width / 2,
                        this.canvas.height - 200,
                        '换牌!',
                        '#60c0e0',
                        24
                    );
                    this.updateMulliganUI();
                }
            } else if (state.phase === GamePhase.ROUND_PLAY && state.isPlayerTurn && !state.playerPassed) {
                // 游戏阶段 - 选择卡牌
                if (this.hoveredCardIndex >= 0) {
                    this.selectedCardIndex = this.hoveredCardIndex;
                    this.isDragging = true;
                    const rect = this.canvas.getBoundingClientRect();
                    this.dragPos = {
                        x: e.clientX - rect.left,
                        y: e.clientY - rect.top
                    };
                }
            }
        }
        
        handleMouseUp(e) {
            if (!this.isDragging || this.selectedCardIndex < 0) {
                this.isDragging = false;
                return;
            }
            
            const rect = this.canvas.getBoundingClientRect();
            const x = e.clientX - rect.left;
            const y = e.clientY - rect.top;
            
            // 检测放置区域
            const row = this.detectTargetRow(y);
            
            if (row) {
                this.playPlayerCard(this.selectedCardIndex, row);
            }
            
            this.isDragging = false;
            this.selectedCardIndex = -1;
        }
        
        detectTargetRow(y) {
            const h = this.canvas.height;
            const rowHeight = h / 7;
            
            if (y > h - 50 - rowHeight && y < h - 50) return 'siege';
            if (y > h - 50 - rowHeight * 2 && y < h - 50 - rowHeight) return 'ranged';
            if (y > h - 50 - rowHeight * 3 && y < h - 50 - rowHeight * 2) return 'melee';
            
            return null;
        }
        
        playPlayerCard(cardIndex, targetRow) {
            const state = this.stateManager.getState();
            
            if (state.phase !== GamePhase.ROUND_PLAY || !state.isPlayerTurn || state.playerPassed) {
                return;
            }
            
            const card = this.stateManager.playCard(cardIndex, targetRow, true);
            
            if (card) {
                // 特效
                for (let i = 0; i < 15; i++) {
                    this.renderer.addParticle(
                        this.canvas.width / 2,
                        this.canvas.height / 2,
                        card.color === 'GOLD' ? '#f0d060' : '#4ade80'
                    );
                }
                
                this.renderer.addFloatingText(
                    this.canvas.width / 2,
                    this.canvas.height / 2,
                    `+${card.power || 0}`,
                    '#4ade80',
                    28
                );
                
                // 切换回合
                this.stateManager.switchTurn();
                
                // 触发AI
                setTimeout(() => this.aiTurn(), this.ai.config.thinkTime);
            }
        }
        
        playerPass() {
            const state = this.stateManager.getState();
            
            if (state.phase !== GamePhase.ROUND_PLAY || state.playerPassed) return;
            
            this.stateManager.pass(true);
            
            this.renderer.addFloatingText(
                this.canvas.width / 2,
                this.canvas.height / 2,
                'PASS',
                '#c9a227',
                32
            );
            
            // 检查回合结束
            if (state.aiPassed) {
                this.handleRoundEnd();
            } else {
                setTimeout(() => this.aiTurn(), this.ai.config.thinkTime);
            }
        }
        
        aiTurn() {
            const state = this.stateManager.getState();
            
            if (state.phase !== GamePhase.ROUND_PLAY || state.aiPassed) return;
            if (state.isPlayerTurn && !state.playerPassed) return;
            
            this.aiThinking = true;
            
            const decision = this.ai.makeDecision(state);
            
            setTimeout(() => {
                if (decision.action === 'PASS') {
                    this.stateManager.pass(false);
                    
                    this.renderer.addFloatingText(
                        this.canvas.width / 2,
                        150,
                        'AI PASS',
                        '#c9a227',
                        28
                    );
                    
                    if (state.playerPassed) {
                        this.handleRoundEnd();
                    }
                } else {
                    const card = this.stateManager.playCard(decision.cardIndex, decision.targetRow, false);
                    
                    if (card) {
                        for (let i = 0; i < 15; i++) {
                            this.renderer.addParticle(
                                this.canvas.width / 2,
                                150,
                                '#f87171'
                            );
                        }
                        
                        this.renderer.addFloatingText(
                            this.canvas.width / 2,
                            150,
                            `AI: ${card.name}`,
                            '#f87171',
                            18
                        );
                    }
                    
                    this.stateManager.switchTurn();
                }
                
                this.aiThinking = false;
                
                // 继续AI回合（如果玩家已pass）
                const newState = this.stateManager.getState();
                if (!newState.isPlayerTurn && !newState.aiPassed && newState.playerPassed) {
                    setTimeout(() => this.aiTurn(), this.ai.config.thinkTime);
                }
            }, this.ai.config.thinkTime);
        }
        
        handleRoundEnd() {
            const result = this.stateManager.endRound();
            
            this.updateScoreDisplay();
            
            if (result.gameOver) {
                this.showGameOver(result.winner === 'player');
            } else {
                // 回合结束动画
                this.renderer.addFloatingText(
                    this.canvas.width / 2,
                    this.canvas.height / 2,
                    `回合 ${this.stateManager.currentRound - 1} 结束`,
                    '#c9a227',
                    36
                );
                
                // 延迟后进入换牌阶段
                setTimeout(() => {
                    this.updateMulliganUI();
                }, 1500);
            }
        }
        
        confirmMulligan() {
            if (this.stateManager.phase === GamePhase.MULLIGAN) {
                this.stateManager.confirmMulligan();
                document.getElementById('mulliganUI').style.display = 'none';
                document.getElementById('phaseIndicator').textContent = `第 ${this.stateManager.currentRound} 回合`;
                
                // AI也确认换牌
                setTimeout(() => this.aiTurn(), 500);
            }
        }
        
        useLeaderAbility() {
            const state = this.stateManager.getState();
            
            if (state.playerLeaderUsed) return;
            
            this.stateManager.playerLeaderUsed = true;
            document.getElementById('btnLeader').classList.add('used');
            
            this.renderer.addFloatingText(
                this.canvas.width / 2,
                this.canvas.height / 2,
                '领袖技能!',
                '#f0d060',
                32
            );
        }
        
        startNewGame() {
            this.stateManager.initializeGame();
            this.selectedCardIndex = -1;
            this.hoveredCardIndex = -1;
            this.aiThinking = false;
            
            this.updateScoreDisplay();
            this.updateMulliganUI();
            this.updateStatsUI();
            
            document.getElementById('gameOverModal').classList.remove('show');
            document.getElementById('btnLeader').classList.remove('used');
            document.getElementById('phaseIndicator').textContent = '换牌阶段';
        }
        
        updateMulliganUI() {
            const state = this.stateManager.getState();
            
            if (state.phase === GamePhase.MULLIGAN) {
                document.getElementById('mulliganUI').style.display = 'block';
                document.getElementById('mulliganCount').textContent = state.mulliganCount;
                document.getElementById('phaseIndicator').textContent = '换牌阶段';
                document.getElementById('btnConfirm').style.display = 'block';
            } else {
                document.getElementById('mulliganUI').style.display = 'none';
                document.getElementById('btnConfirm').style.display = 'none';
            }
        }
        
        updateScoreDisplay() {
            const state = this.stateManager.getState();
            
            // 玩家宝石
            for (let i = 1; i <= 2; i++) {
                const gem = document.getElementById(`playerGem${i}`);
                if (i <= state.playerRoundWins) {
                    gem.classList.remove('empty');
                    gem.textContent = '2';
                } else {
                    gem.classList.add('empty');
                    gem.textContent = '';
                }
            }
            
            // AI宝石
            for (let i = 1; i <= 2; i++) {
                const gem = document.getElementById(`enemyGem${i}`);
                if (i <= state.aiRoundWins) {
                    gem.classList.remove('empty');
                    gem.textContent = '2';
                } else {
                    gem.classList.add('empty');
                    gem.textContent = '';
                }
            }
        }
        
        updateStatsUI() {
            const stats = this.persistence.getStats();
            
            document.getElementById('statTotal').textContent = stats.totalGames;
            document.getElementById('statWins').textContent = stats.wins;
            document.getElementById('statLosses').textContent = stats.losses;
            
            const winRate = stats.totalGames > 0 ? 
                Math.round((stats.wins / stats.totalGames) * 100) : 0;
            document.getElementById('statWinRate').textContent = `${winRate}%`;
            document.getElementById('statStreak').textContent = stats.currentStreak;
        }
        
        updateSettingsUI() {
            const settings = this.persistence.getSettings();
            
            document.getElementById('playerFaction').value = settings.playerFaction;
            document.getElementById('aiFaction').value = settings.aiFaction;
            document.getElementById('startingHand').value = settings.startingHand;
            document.getElementById('mulliganCount').value = settings.mulliganCount;
        }
        
        toggleSettings() {
            const panel = document.getElementById('settingsPanel');
            panel.classList.toggle('show');
        }
        
        saveSettings() {
            const settings = {
                playerFaction: document.getElementById('playerFaction').value,
                aiFaction: document.getElementById('aiFaction').value,
                startingHand: parseInt(document.getElementById('startingHand').value),
                mulliganCount: parseInt(document.getElementById('mulliganCount').value),
                difficulty: this.ai.difficulty,
                soundEnabled: true,
                animationsEnabled: true
            };
            
            this.persistence.saveSettings(settings);
            this.settings = settings;
            
            this.toggleSettings();
            this.startNewGame();
        }
        
        showGameOver(playerWon) {
            const modal = document.getElementById('gameOverModal');
            const title = document.getElementById('modalTitle');
            const stats = this.persistence.getStats();
            
            title.textContent = playerWon ? '胜利' : '失败';
            title.className = `modal-title ${playerWon ? 'victory' : 'defeat'}`;
            
            const state = this.stateManager.getState();
            document.getElementById('finalScore').textContent = 
                `${state.playerRoundWins} - ${state.aiRoundWins}`;
            
            const playTime = Math.floor((Date.now() - this.stateManager.gameStartTime) / 1000);
            const minutes = Math.floor(playTime / 60);
            const seconds = playTime % 60;
            document.getElementById('gameTime').textContent = 
                `${minutes}:${seconds.toString().padStart(2, '0')}`;
            
            document.getElementById('powerDiff').textContent = 
                `${playerWon ? '+' : '-'}${Math.abs(state.playerPower - state.aiPower)}`;
            
            modal.classList.add('show');
            this.updateStatsUI();
        }
        
        gameLoop() {
            const state = this.stateManager.getState();
            
            // 清空画布
            this.renderer.clear();
            
            // 绘制战场
            this.renderer.drawBattlefield();
            
            // 绘制AI手牌（背面）
            const aiHandY = 60;
            state.aiHand.forEach((_, i) => {
                const x = this.canvas.width / 2 - (state.aiHand.length - 1) * 30 / 2 + i * 30;
                this.ctx.fillStyle = '#1a1a20';
                this.ctx.strokeStyle = '#3a3a40';
                this.ctx.lineWidth = 1;
                this.ctx.fillRect(x - 20, aiHandY - 30, 40, 60);
                this.ctx.strokeRect(x - 20, aiHandY - 30, 40, 60);
            });
            
            // 绘制战场卡牌
            this.renderer.drawBoardCards(state.aiBoard, state.weather, false);
            this.renderer.drawBoardCards(state.playerBoard, state.weather, true);
            
            // 绘制手牌
            this.cardBounds = this.renderer.drawHand(
                state.playerHand,
                this.canvas.height - 70,
                true,
                this.selectedCardIndex,
                this.hoveredCardIndex
            );
            
            // 绘制战力
            this.renderer.drawPowerDisplay(state.playerPower, state.aiPower);
            
            // 绘制回合指示
            this.renderer.drawTurnIndicator(
                state.isPlayerTurn,
                state.playerPassed,
                state.aiPassed
            );
            
            // 绘制拖拽中的卡牌
            if (this.isDragging && this.selectedCardIndex >= 0) {
                const card = state.playerHand[this.selectedCardIndex];
                if (card) {
                    this.ctx.globalAlpha = 0.7;
                    this.renderer.drawCard(card, this.dragPos.x, this.dragPos.y, 1.2, true);
                    this.ctx.globalAlpha = 1;
                }
            }
            
            // 更新和绘制粒子
            this.renderer.updateParticles();
            this.renderer.drawParticles();
            
            // 绘制浮动文字
            this.renderer.drawFloatingTexts();
            
            // 更新阶段指示器
            if (state.phase === GamePhase.ROUND_PLAY) {
                document.getElementById('phaseIndicator').textContent = 
                    `第 ${state.currentRound} 回合`;
            } else if (state.phase === GamePhase.GAME_OVER) {
                document.getElementById('phaseIndicator').textContent = '游戏结束';
            }
            
            requestAnimationFrame(() => this.gameLoop());
        }
    }
    
    // 启动游戏
    window.addEventListener('DOMContentLoaded', () => {
        new GwentGame();
    });

## 实现说明

### 1. 游戏流程控制

**状态机设计：**
- `INIT` - 初始化阶段，构建卡组
- `MULLIGAN` - 换牌阶段，玩家可以替换手牌
- `ROUND_PLAY` - 对战阶段，双方交替打出卡牌
- `ROUND_END` - 回合结算
- `GAME_OVER` - 游戏结束

**换牌阶段：**
- 起始手牌数可在设置中配置（10-11张）
- 换牌次数可配置（2-3次）
- 点击手牌中的卡牌进行更换
- 点击"确认"按钮结束换牌

**回合权传递：**
- 玩家和AI交替行动
- 一方PASS后，另一方继续行动
- 双方都PASS后结算回合

**胜负判定：**
- 比较双方战力，高者获得1分
- 三局两胜制
- 平局时双方各得1分

### 2. 数据持久化

**保存到localStorage：**
- 玩家卡组配置（阵营、领袖等）
- 游戏设置（起始手牌数、换牌次数等）
- 胜负统计数据

**统计面板显示：**
- 总场次、胜利、失败
- 胜率、当前连胜

### 3. UI增强

**比分宝石显示：**
- 顶部中央显示双方回合得分
- 获得的回合显示为彩色，未获得为灰色

**游戏结束弹窗：**
- 显示胜利/失败
- 显示最终比分、用时、战力差
- "再来一局"按钮
