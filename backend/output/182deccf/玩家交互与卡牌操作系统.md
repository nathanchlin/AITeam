        // ==================== 核心规则引擎 ====================
        class GwentRuleEngine {
            constructor() {
                this.state = this.getInitialState();
                this.listeners = [];
            }
            
            getInitialState() {
                return {
                    phase: 'GAME_START',
                    currentRound: 1,
                    turnCount: 0,
                    currentPlayer: 'player',
                    startingPlayer: 'player',
                    
                    players: {
                        player: {
                            roundWins: 0,
                            totalPower: 0,
                            hasPassed: false,
                            hand: [],
                            deck: [],
                            graveyard: [],
                            leader: null,
                            leaderUsed: false
                        },
                        enemy: {
                            roundWins: 0,
                            totalPower: 0,
                            hasPassed: false,
                            hand: [],
                            deck: [],
                            graveyard: [],
                            leader: null,
                            leaderUsed: false
                        }
                    },
                    
                    board: {
                        player: {
                            melee: { cards: [], power: 0, hasHorn: false },
                            ranged: { cards: [], power: 0, hasHorn: false },
                            siege: { cards: [], power: 0, hasHorn: false }
                        },
                        enemy: {
                            melee: { cards: [], power: 0, hasHorn: false },
                            ranged: { cards: [], power: 0, hasHorn: false },
                            siege: { cards: [], power: 0, hasHorn: false }
                        }
                    },
                    
                    weather: {
                        melee: null,
                        ranged: null,
                        siege: null
                    },
                    
                    eventLog: [],
                    gameOver: false,
                    winner: null
                };
            }
            
            subscribe(callback) {
                this.listeners.push(callback);
                return () => {
                    this.listeners = this.listeners.filter(l => l !== callback);
                };
            }
            
            emit(event) {
                this.listeners.forEach(cb => cb(event, this.state));
            }
            
            log(message, type = 'system') {
                const entry = {
                    time: Date.now(),
                    message,
                    type
                };
                this.state.eventLog.push(entry);
                this.emit({ type: 'LOG', entry });
            }
            
            initNewGame() {
                this.state = this.getInitialState();
                
                const playerDeck = this.buildDeck('monsters');
                const enemyDeck = this.buildDeck('northern');
                
                this.state.players.player.deck = playerDeck;
                this.state.players.enemy.deck = enemyDeck;
                this.state.players.player.leader = CardDatabase.leaders[0];
                this.state.players.enemy.leader = CardDatabase.leaders[2];
                
                this.shuffleDeck('player');
                this.shuffleDeck('enemy');
                
                this.drawCards('player', 10);
                this.drawCards('enemy', 10);
                
                this.state.phase = 'ROUND_PLAY';
                this.state.currentPlayer = 'player';
                this.state.startingPlayer = 'player';
                
                this.log('新游戏开始！第1回合', 'system');
                this.emit({ type: 'GAME_INIT' });
            }
            
            buildDeck(faction) {
                let deck = [];
                const factionCards = CardDatabase[faction] || [];
                
                factionCards.forEach(card => {
                    deck.push({ ...card, instanceId: `${card.id}_${Math.random().toString(36).substr(2, 9)}` });
                });
                
                CardDatabase.monsters.filter(c => c.faction === 'NEUTRAL').forEach(card => {
                    deck.push({ ...card, instanceId: `${card.id}_${Math.random().toString(36).substr(2, 9)}` });
                });
                
                return deck;
            }
            
            shuffleDeck(player) {
                const deck = this.state.players[player].deck;
                for (let i = deck.length - 1; i > 0; i--) {
                    const j = Math.floor(Math.random() * (i + 1));
                    [deck[i], deck[j]] = [deck[j], deck[i]];
                }
            }
            
            drawCards(player, count) {
                const playerData = this.state.players[player];
                const drawn = [];
                for (let i = 0; i < count && playerData.deck.length > 0; i++) {
                    const card = playerData.deck.pop();
                    playerData.hand.push(card);
                    drawn.push(card);
                }
                return drawn;
            }
            
            canPlayCard(player) {
                if (this.state.phase !== 'ROUND_PLAY') return false;
                if (this.state.currentPlayer !== player) return false;
                if (this.state.players[player].hasPassed) return false;
                if (this.state.gameOver) return false;
                return true;
            }
            
            getValidRows(card, player) {
                if (card.type === 'WEATHER') return ['weather'];
                if (card.type === 'SPECIAL') return ['special'];
                if (card.abilities && card.abilities.includes('SPY')) {
                    return card.row ? [card.row.toLowerCase()] : ['melee', 'ranged', 'siege'];
                }
                if (card.row) {
                    return [card.row.toLowerCase()];
                }
                return ['melee', 'ranged', 'siege'];
            }
            
            playCard(player, card, targetRow) {
                if (!this.canPlayCard(player)) {
                    return { success: false, reason: '无法出牌' };
                }
                
                const playerData = this.state.players[player];
                
                const cardIndex = playerData.hand.findIndex(c => c.instanceId === card.instanceId);
                if (cardIndex === -1) {
                    return { success: false, reason: '卡牌不在手中' };
                }
                playerData.hand.splice(cardIndex, 1);
                
                if (card.type === 'UNIT') {
                    this.placeUnit(player, card, targetRow);
                } else if (card.type === 'WEATHER') {
                    this.applyWeatherCard(card);
                } else if (card.type === 'SPECIAL') {
                    this.playSpecialCard(player, card, targetRow);
                }
                
                this.log(`${player === 'player' ? '玩家' : '对手'}打出 ${card.name}`, player === 'player' ? 'player-action' : 'enemy-action');
                
                this.recalculateAllPower();
                this.emit({ type: 'CARD_PLAYED', player, card, row: targetRow });
                this.switchTurn();
                
                return { success: true };
            }
            
            placeUnit(player, card, targetRow) {
                let actualPlayer = player;
                let row = targetRow;
                
                // SPY能力：放到对方战场
                if (card.abilities && card.abilities.includes('SPY')) {
                    actualPlayer = player === 'player' ? 'enemy' : 'player';
                    // 给玩家抽牌
                    this.drawCards(player, 1);
                }
                
                const boardRow = this.state.board[actualPlayer][row];
                if (boardRow) {
                    boardRow.cards.push(card);
                }
            }
            
            applyWeatherCard(card) {
                if (card.abilityType === 'BITING_FROST') {
                    this.state.weather.melee = card;
                    this.log('霜冻笼罩了近战排', 'system');
                } else if (card.abilityType === 'IMPENETRABLE_FOG') {
                    this.state.weather.ranged = card;
                    this.log('迷雾笼罩了远程排', 'system');
                }
            }
            
            playSpecialCard(player, card, targetRow) {
                if (card.abilityType === 'COMMANDERS_HORN') {
                    if (targetRow && this.state.board.player[targetRow]) {
                        this.state.board.player[targetRow].hasHorn = true;
                        this.log(`指挥号角激活了${targetRow}排`, 'system');
                    }
                } else if (card.abilityType === 'SCORCH') {
                    this.scorchStrongest();
                }
            }
            
            scorchStrongest() {
                let maxPower = 0;
                const allCards = [];
                
                ['player', 'enemy'].forEach(player => {
                    ['melee', 'ranged', 'siege'].forEach(row => {
                        this.state.board[player][row].cards.forEach(card => {
                            if (card.power > maxPower) {
                                maxPower = card.power;
                                allCards.length = 0;
                                allCards.push({ card, player, row });
                            } else if (card.power === maxPower) {
                                allCards.push({ card, player, row });
                            }
                        });
                    });
                });
                
                allCards.forEach(({ card, player, row }) => {
                    const rowCards = this.state.board[player][row].cards;
                    const idx = rowCards.findIndex(c => c.instanceId === card.instanceId);
                    if (idx !== -1) {
                        rowCards.splice(idx, 1);
                        this.state.players[player].graveyard.push(card);
                    }
                });
                
                this.log(`灼烧摧毁了${allCards.length}张战力${maxPower}的卡牌`, 'system');
            }
            
            recalculateAllPower() {
                ['player', 'enemy'].forEach(player => {
                    let totalPower = 0;
                    ['melee', 'ranged', 'siege'].forEach(rowName => {
                        const row = this.state.board[player][rowName];
                        let rowPower = 0;
                        
                        row.cards.forEach(card => {
                            let cardPower = card.power;
                            
                            // 天气效果
                            if (this.state.weather[rowName] && card.type === 'UNIT') {
                                cardPower = 1;
                            }
                            
                            rowPower += cardPower;
                        });
                        
                        // 指挥号角
                        if (row.hasHorn) {
                            rowPower *= 2;
                        }
                        
                        row.power = rowPower;
                        totalPower += rowPower;
                    });
                    
                    this.state.players[player].totalPower = totalPower;
                });
            }
            
            switchTurn() {
                const player = this.state.players.player;
                const enemy = this.state.players.enemy;
                
                // 检查双方是否都放弃
                if (player.hasPassed && enemy.hasPassed) {
                    this.endRound();
                    return;
                }
                
                // 如果当前玩家已放弃，切换到对方
                if (this.state.currentPlayer === 'player') {
                    if (enemy.hasPassed) {
                        // 对手已放弃，玩家继续
                    } else {
                        this.state.currentPlayer = 'enemy';
                        this.emit({ type: 'TURN_SWITCH', player: 'enemy' });
                        // AI行动
                        setTimeout(() => this.aiTurn(), 1000);
                    }
                } else {
                    if (player.hasPassed) {
                        // 玩家已放弃，对手继续
                        setTimeout(() => this.aiTurn(), 1000);
                    } else {
                        this.state.currentPlayer = 'player';
                        this.emit({ type: 'TURN_SWITCH', player: 'player' });
                    }
                }
                
                this.state.turnCount++;
            }
            
            pass(player) {
                if (this.state.players[player].hasPassed) return;
                
                this.state.players[player].hasPassed = true;
                this.log(`${player === 'player' ? '玩家' : '对手'}选择放弃`, player === 'player' ? 'player-action' : 'enemy-action');
                this.emit({ type: 'PLAYER_PASSED', player });
                this.switchTurn();
            }
            
            useLeaderAbility(player) {
                const playerData = this.state.players[player];
                if (playerData.leaderUsed) {
                    return { success: false, reason: '领袖技能已使用' };
                }
                
                playerData.leaderUsed = true;
                this.log(`${player === 'player' ? '玩家' : '对手'}使用了领袖技能`, player === 'player' ? 'player-action' : 'enemy-action');
                this.emit({ type: 'LEADER_USED', player });
                return { success: true };
            }
            
            endRound() {
                const playerPower = this.state.players.player.totalPower;
                const enemyPower = this.state.players.enemy.totalPower;
                
                this.log(`回合结束！玩家${playerPower} vs 对手${enemyPower}`, 'system');
                
                if (playerPower > enemyPower) {
                    this.state.players.player.roundWins++;
                    this.log('玩家赢得本回合！', 'player-action');
                } else if (enemyPower > playerPower) {
                    this.state.players.enemy.roundWins++;
                    this.log('对手赢得本回合！', 'enemy-action');
                } else {
                    this.state.players.player.roundWins++;
                    this.state.players.enemy.roundWins++;
                    this.log('本回合平局！', 'system');
                }
                
                // 检查游戏结束
                if (this.state.players.player.roundWins >= 2 || this.state.players.enemy.roundWins >= 2) {
                    this.endGame();
                    return;
                }
                
                // 开始新回合
                this.startNewRound();
            }
            
            startNewRound() {
                this.state.currentRound++;
                
                // 清空战场
                ['player', 'enemy'].forEach(player => {
                    ['melee', 'ranged', 'siege'].forEach(row => {
                        this.state.board[player][row].cards.forEach(card => {
                            this.state.players[player].graveyard.push(card);
                        });
                        this.state.board[player][row].cards = [];
                        this.state.board[player][row].power = 0;
                        this.state.board[player][row].hasHorn = false;
                    });
                    this.state.players[player].hasPassed = false;
                });
                
                // 清除天气
                this.state.weather = { melee: null, ranged: null, siege: null };
                
                // 抽2张牌
                this.drawCards('player', 2);
                this.drawCards('enemy', 2);
                
                // 交换先手
                this.state.startingPlayer = this.state.startingPlayer === 'player' ? 'enemy' : 'player';
                this.state.currentPlayer = this.state.startingPlayer;
                
                this.log(`第${this.state.currentRound}回合开始！`, 'system');
                this.emit({ type: 'ROUND_START', round: this.state.currentRound });
                
                if (this.state.currentPlayer === 'enemy') {
                    setTimeout(() => this.aiTurn(), 1000);
                }
            }
            
            endGame() {
                this.state.gameOver = true;
                this.state.phase = 'GAME_OVER';
                
                if (this.state.players.player.roundWins > this.state.players.enemy.roundWins) {
                    this.state.winner = 'player';
                    this.log('恭喜！你赢得了比赛！', 'player-action');
                } else {
                    this.state.winner = 'enemy';
                    this.log('你输了这场比赛...', 'enemy-action');
                }
                
                this.emit({ type: 'GAME_OVER', winner: this.state.winner });
            }
            
            aiTurn() {
                if (this.state.gameOver || this.state.currentPlayer !== 'enemy') return;
                if (this.state.players.enemy.hasPassed) return;
                
                const hand = this.state.players.enemy.hand;
                
                // 简单AI：如果手牌少于2张或落后很多，考虑放弃
                const powerDiff = this.state.players.enemy.totalPower - this.state.players.player.totalPower;
                if (hand.length === 0 || (this.state.players.player.hasPassed && powerDiff > 20)) {
                    this.pass('enemy');
                    return;
                }
                
                // 打出第一张可用的单位卡
                for (let i = 0; i < hand.length; i++) {
                    const card = hand[i];
                    if (card.type === 'UNIT') {
                        const validRows = this.getValidRows(card, 'enemy');
                        // SPY卡放到对方战场
                        const targetPlayer = card.abilities && card.abilities.includes('SPY') ? 'player' : 'enemy';
                        const row = validRows[0];
                        
                        this.playCard('enemy', card, row);
                        return;
                    }
                }
                
                // 没有可打的牌，放弃
                this.pass('enemy');
            }
        }
        
        // ==================== 渲染系统 ====================
        class GameRenderer {
            constructor(canvas, ruleEngine) {
                this.canvas = canvas;
                this.ctx = canvas.getContext('2d');
                this.engine = ruleEngine;
                
                this.width = 0;
                this.height = 0;
                
                // 交互状态
                this.hoveredCard = null;
                this.selectedCard = null;
                this.draggingCard = null;
                this.dragOffset = { x: 0, y: 0 };
                this.dragPosition = { x: 0, y: 0 };
                this.highlightedRows = [];
                
                // 卡牌位置缓存
                this.cardPositions = new Map();
                this.rowRects = {};
                
                // 粒子系统
                this.particles = [];
                
                // 天气效果
                this.currentWeather = 'clear';
                this.weatherParticles = [];
                
                // 动画
                this.animations = [];
                
                this.init();
            }
            
            init() {
                this.resize();
                window.addEventListener('resize', () => this.resize());
                
                this.engine.subscribe((event, state) => this.handleGameEvent(event, state));
                
                this.startRenderLoop();
            }
            
            resize() {
                this.width = window.innerWidth;
                this.height = window.innerHeight;
                this.canvas.width = this.width;
                this.canvas.height = this.height;
                
                this.calculateLayout();
            }
            
            calculateLayout() {
                const centerY = this.height / 2;
                const rowHeight = 120;
                const boardWidth = Math.min(800, this.width - 100);
                const boardLeft = (this.width - boardWidth) / 2;
                
                // 战场区域
                this.rowRects = {
                    enemy: {
                        siege: { x: boardLeft, y: centerY - rowHeight * 3 - 30, w: boardWidth, h: rowHeight },
                        ranged: { x: boardLeft, y: centerY - rowHeight * 2 - 20, w: boardWidth, h: rowHeight },
                        melee: { x: boardLeft, y: centerY - rowHeight - 10, w: boardWidth, h: rowHeight }
                    },
                    player: {
                        melee: { x: boardLeft, y: centerY + 10, w: boardWidth, h: rowHeight },
                        ranged: { x: boardLeft, y: centerY + rowHeight + 20, w: boardWidth, h: rowHeight },
                        siege: { x: boardLeft, y: centerY + rowHeight * 2 + 30, w: boardWidth, h: rowHeight }
                    },
                    weather: { x: boardLeft - 100, y: centerY - 60, w: 80, h: 120 }
                };
            }
            
            handleGameEvent(event, state) {
                if (event.type === 'LOG') {
                    this.addLogEntry(event.entry);
                } else if (event.type === 'GAME_OVER') {
                    this.showToast(event.winner === 'player' ? '胜利！' : '失败...', 
                                   event.winner === 'player' ? 'success' : 'error');
                } else if (event.type === 'TURN_SWITCH') {
                    this.showTurnIndicator(event.player);
                } else if (event.type === 'ROUND_START') {
                    this.showToast(`第${event.round}回合开始`, 'info');
                }
            }
            
            addLogEntry(entry) {
                const logContainer = document.getElementById('gameLog');
                if (!logContainer) return;
                
                const div = document.createElement('div');
                div.className = `log-entry ${entry.type}`;
                div.textContent = entry.message;
                logContainer.appendChild(div);
                logContainer.scrollTop = logContainer.scrollHeight;
            }
            
            showToast(message, type = 'info') {
                const container = document.getElementById('toastContainer');
                if (!container) return;
                
                const toast = document.createElement('div');
                toast.className = `toast ${type}`;
                toast.textContent = message;
                container.appendChild(toast);
                
                setTimeout(() => toast.classList.add('show'), 10);
                setTimeout(() => {
                    toast.classList.remove('show');
                    setTimeout(() => toast.remove(), 300);
                }, 2000);
            }
            
            showTurnIndicator(player) {
                const indicator = document.getElementById('turnIndicator');
                if (!indicator) return;
                
                indicator.textContent = player === 'player' ? '你的回合' : '对手回合';
                indicator.classList.toggle('enemy-turn', player !== 'player');
                indicator.classList.add('show');
                
                setTimeout(() => indicator.classList.remove('show'), 1500);
            }
            
            // 粒子效果
            addParticle(x, y, color, velocity = { x: 0, y: -1 }, life = 60) {
                this.particles.push({
                    x, y,
                    vx: velocity.x + (Math.random() - 0.5) * 2,
                    vy: velocity.y + (Math.random() - 0.5) * 2,
                    color,
                    life,
                    maxLife: life,
                    size: Math.random() * 3 + 1
                });
            }
            
            updateParticles() {
                for (let i = this.particles.length - 1; i >= 0; i--) {
                    const p = this.particles[i];
                    p.x += p.vx;
                    p.y += p.vy;
                    p.life--;
                    
                    if (p.life <= 0) {
                        this.particles.splice(i, 1);
                    }
                }
            }
            
            renderParticles() {
                this.particles.forEach(p => {
                    const alpha = p.life / p.maxLife;
                    this.ctx.beginPath();
                    this.ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
                    this.ctx.fillStyle = p.color.replace(')', `, ${alpha})`).replace('rgb', 'rgba');
                    this.ctx.fill();
                });
            }
            
            // 主渲染循环
            startRenderLoop() {
                const render = () => {
                    this.update();
                    this.render();
                    requestAnimationFrame(render);
                };
                render();
            }
            
            update() {
                this.updateParticles();
                this.updateWeatherParticles();
            }
            
            render() {
                const ctx = this.ctx;
                
                // 清空画布
                ctx.fillStyle = CONFIG.COLORS.BG_PRIMARY;
                ctx.fillRect(0, 0, this.width, this.height);
                
                // 绘制背景
                this.drawBackground();
                
                // 绘制战场区域
                this.drawBoard();
                
                // 绘制天气效果
                this.drawWeatherEffects();
                
                // 绘制战场上的卡牌
                this.drawBoardCards();
                
                // 绘制手牌
                this.drawHand();
                
                // 绘制高亮区域
                this.drawHighlights();
                
                // 绘制拖拽中的卡牌
                if (this.draggingCard) {
                    this.drawDraggingCard();
                }
                
                // 绘制悬停预览
                if (this.hoveredCard && !this.draggingCard) {
                    this.drawCardPreview();
                }
                
                // 绘制粒子
                this.renderParticles();
                
                // 绘制战力显示
                this.drawPowerDisplay();
                
                // 更新UI
                this.updateUI();
            }
            
            drawBackground() {
                const ctx = this.ctx;
                
                // 渐变背景
                const gradient = ctx.createRadialGradient(
                    this.width / 2, this.height / 2, 0,
                    this.width / 2, this.height / 2, this.width / 2
                );
                gradient.addColorStop(0, '#1a1a25');
                gradient.addColorStop(1, '#0a0a0f');
                ctx.fillStyle = gradient;
                ctx.fillRect(0, 0, this.width, this.height);
                
                // 网格线
                ctx.strokeStyle = 'rgba(42, 42, 53, 0.3)';
                ctx.lineWidth = 1;
                const gridSize = 50;
                
                for (let x = 0; x < this.width; x += gridSize) {
                    ctx.beginPath();
                    ctx.moveTo(x, 0);
                    ctx.lineTo(x, this.height);
                    ctx.stroke();
                }
                
                for (let y = 0; y < this.height; y += gridSize) {
                    ctx.beginPath();
                    ctx.moveTo(0, y);
                    ctx.lineTo(this.width, y);
                    ctx.stroke();
                }
            }
            
            drawBoard() {
                const ctx = this.ctx;
                
                // 敌方区域
                ['siege', 'ranged', 'melee'].forEach(row => {
                    const rect = this.rowRects.enemy[row];
                    ctx.fillStyle = 'rgba(26, 21, 32, 0.6)';
                    ctx.fillRect(rect.x, rect.y, rect.w, rect.h);
                    ctx.strokeStyle = 'rgba(248, 113, 113, 0.2)';
                    ctx.strokeRect(rect.x, rect.y, rect.w, rect.h);
                });
                
                // 玩家区域
                ['melee', 'ranged', 'siege'].forEach(row => {
                    const rect = this.rowRects.player[row];
                    ctx.fillStyle = 'rgba(21, 26, 32, 0.6)';
                    ctx.fillRect(rect.x, rect.y, rect.w, rect.h);
                    ctx.strokeStyle = 'rgba(74, 222, 128, 0.2)';
                    ctx.strokeRect(rect.x, rect.y, rect.w, rect.h);
                });
                
                // 行标签
                ctx.font = '11px Cinzel, serif';
                ctx.fillStyle = CONFIG.COLORS.TEXT_SECONDARY;
                ctx.textAlign = 'left';
                
                const rowLabels = {
                    siege: '攻城',
                    ranged: '远程',
                    melee: '近战'
                };
                
                ['siege', 'ranged', 'melee'].forEach(row => {
                    const rect = this.rowRects.enemy[row];
                    ctx.fillText(rowLabels[row], rect.x + 5, rect.y + 15);
                });
                
                ['melee', 'ranged', 'siege'].forEach(row => {
                    const rect = this.rowRects.player[row];
                    ctx.fillText(rowLabels[row], rect.x + 5, rect.y + 15);
                });
            }
            
            drawHighlights() {
                const ctx = this.ctx;
                
                this.highlightedRows.forEach(({ rect, valid }) => {
                    ctx.fillStyle = valid ? CONFIG.COLORS.HIGHLIGHT_VALID : CONFIG.COLORS.HIGHLIGHT_INVALID;
                    ctx.fillRect(rect.x, rect.y, rect.w, rect.h);
                    
                    // 边框闪烁
                    ctx.strokeStyle = valid ? CONFIG.COLORS.CARD_SELECTED : '#f87171';
                    ctx.lineWidth = 3;
                    ctx.setLineDash([10, 5]);
                    ctx.strokeRect(rect.x, rect.y, rect.w, rect.h);
                    ctx.setLineDash([]);
                });
            }
            
            drawBoardCards() {
                const state = this.engine.state;
                if (!state.board) return;
                
                this.cardPositions.clear();
                
                ['player', 'enemy'].forEach(player => {
                    ['melee', 'ranged', 'siege'].forEach(row => {
                        const cards = state.board[player][row].cards;
                        const rect = this.rowRects[player][row];
                        
                        cards.forEach((card, i) => {
                            const x = rect.x + 20 + i * (CONFIG.CARD_WIDTH + 5);
                            const y = rect.y + (rect.h - CONFIG.CARD_HEIGHT) / 2;
                            
                            this.drawCard(card, x, y, 1, false);
                            
                            // 缓存卡牌位置
                            this.cardPositions.set(card.instanceId, { x, y, card, player, row });
                        });
                    });
                });
            }
            
            drawHand() {
                const state = this.engine.state;
                if (!state.players || !state.players.player) return;
                
                const hand = state.players.player.hand;
                if (hand.length === 0) return;
                
                const centerX = this.width / 2;
                const baseY = this.height - 80;
                const totalWidth = hand.length * (CONFIG.CARD_WIDTH + 10);
                const startX = centerX - totalWidth / 2;
                
                hand.forEach((card, i) => {
                    const spreadAngle = CONFIG.HAND_FAN_ANGLE;
                    const angle = (i - (hand.length - 1) / 2) * spreadAngle;
                    
                    const x = startX + i * (CONFIG.CARD_WIDTH + 10);
                    const y = baseY + Math.abs(angle) * 50;
                    
                    const isHovered = this.hoveredCard && this.hoveredCard.instanceId === card.instanceId;
                    const isSelected = this.selectedCard && this.selectedCard.instanceId === card.instanceId;
                    const scale = isHovered ? CONFIG.HAND_HOVER_SCALE : 1;
                    const drawY = isHovered ? y - 40 : y;
                    
                    if (!this.draggingCard || this.draggingCard.instanceId !== card.instanceId) {
                        this.drawCard(card, x, drawY, scale, isHovered || isSelected);
                    }
                    
                    // 缓存手牌位置
                    this.cardPositions.set(card.instanceId, { 
                        x, y: baseY, 
                        width: CONFIG.CARD_WIDTH * scale, 
                        height: CONFIG.CARD_HEIGHT * scale, 
                        card, 
                        inHand: true 
                    });
                });
            }
            
            drawCard(card, x, y, scale = 1, highlighted = false) {
                const ctx = this.ctx;
                const w = CONFIG.CARD_WIDTH * scale;
                const h = CONFIG.CARD_HEIGHT * scale;
                
                ctx.save();
                ctx.translate(x, y);
                
                // 阴影
                if (highlighted) {
                    ctx.shadowColor = CONFIG.COLORS.CARD_SELECTED;
                    ctx.shadowBlur = 20;
                }
                
                // 卡牌背景
                let bgColor, borderColor;
                if (card.color === 'GOLD') {
                    bgColor = 'linear-gradient(180deg, #3a3020 0%, #2a2010 100%)';
                    borderColor = CONFIG.COLORS.GOLD_PRIMARY;
                } else if (card.color === 'SPECIAL' || card.color === 'WEATHER') {
                    bgColor = 'linear-gradient(180deg, #1a2a3a 0%, #101820 100%)';
                    borderColor = '#4a8aba';
                } else {
                    bgColor = 'linear-gradient(180deg, #2a2015 0%, #1a1510 100%)';
                    borderColor = '#8b5a2b';
                }
                
                // 绘制背景
                ctx.fillStyle = card.color === 'GOLD' ? '#3a3020' : 
                               (card.color === 'SPECIAL' || card.color === 'WEATHER') ? '#1a2a3a' : '#2a2015';
                ctx.beginPath();
                ctx.roundRect(0, 0, w, h, 5);
                ctx.fill();
                
                // 边框
                ctx.strokeStyle = borderColor;
                ctx.lineWidth = highlighted ? 3 : 2;
                ctx.stroke();
                
                // 战力
                if (card.type === 'UNIT') {
                    ctx.fillStyle = '#1a1a20';
                    ctx.beginPath();
                    ctx.arc(w / 2, 15 * scale, 12 * scale, 0, Math.PI * 2);
                    ctx.fill();
                    ctx.strokeStyle = borderColor;
                    ctx.lineWidth = 2;
                    ctx.stroke();
                    
                    ctx.fillStyle = card.color === 'GOLD' ? CONFIG.COLORS.GOLD_LIGHT : '#f4a460';
                    ctx.font = `bold ${14 * scale}px Cinzel, serif`;
                    ctx.textAlign = 'center';
                    ctx.textBaseline = 'middle';
                    ctx.fillText(card.power.toString(), w / 2, 15 * scale);
                }
                
                // 卡牌图标区域
                const iconY = card.type === 'UNIT' ? 35 * scale : 15 * scale;
                const iconH = card.type === 'UNIT' ? h - 55 * scale : h - 35 * scale;
                
                ctx.fillStyle = 'rgba(0, 0, 0, 0.4)';
                ctx.fillRect(5 * scale, iconY, w - 10 * scale, iconH);
                
                // 卡牌名称
                ctx.fillStyle = CONFIG.COLORS.TEXT_PRIMARY;
                ctx.font = `${9 * scale}px Cinzel, serif`;
                ctx.textAlign = 'center';
                ctx.textBaseline = 'bottom';
                
                const name = card.name.length > 6 ? card.name.substring(0, 5) + '..' : card.name;
                ctx.fillText(name, w / 2, h - 5 * scale);
                
                // 能力标记
                if (card.abilities && card.abilities.length > 0) {
                    ctx.fillStyle = CONFIG.COLORS.GOLD_PRIMARY;
                    ctx.font = `bold ${8 * scale}px Arial`;
                    ctx.fillText('★', w - 10 * scale, 12 * scale);
                }
                
                ctx.restore();
            }
            
            drawDraggingCard() {
                if (!this.draggingCard) return;
                
                const ctx = this.ctx;
                ctx.save();
                ctx.globalAlpha = CONFIG.COLORS.DRAG_ALPHA || 0.8;
                
                this.drawCard(
                    this.draggingCard, 
                    this.dragPosition.x - CONFIG.CARD_WIDTH / 2, 
                    this.dragPosition.y - CONFIG.CARD_HEIGHT / 2, 
                    1.1, 
                    true
                );
                
                ctx.restore();
            }
            
            drawCardPreview() {
                if (!this.hoveredCard) return;
                
                const card = this.hoveredCard;
                const pos = this.cardPositions.get(card.instanceId);
                if (!pos) return;
                
                // 放大预览
                const previewScale = 1.8;
                const previewX = pos.x - (CONFIG.CARD_WIDTH * (previewScale - 1)) / 2;
                const previewY = pos.y - CONFIG.CARD_HEIGHT * (previewScale - 1) - 20;
                
                this.drawCard(card, Math.max(10, previewX), Math.max(10, previewY), previewScale, true);
            }
            
            drawPowerDisplay() {
                const ctx = this.ctx;
                const state = this.engine.state;
                
                if (!state.players) return;
                
                const playerPower = state.players.player.totalPower || 0;
                const enemyPower = state.players.enemy.totalPower || 0;
                
                // 敌方总战力
                ctx.fillStyle = 'rgba(248, 113, 113, 0.2)';
                ctx.beginPath();
                ctx.arc(this.width / 2, 60, 35, 0, Math.PI * 2);
                ctx.fill();
                
                ctx.fillStyle = CONFIG.COLORS.ENEMY_POWER;
                ctx.font = 'bold 24px Cinzel, serif';
                ctx.textAlign = 'center';
                ctx.textBaseline = 'middle';
                ctx.fillText(enemyPower.toString(), this.width / 2, 60);
                
                ctx.font = '10px Cinzel, serif';
                ctx.fillText('对手', this.width / 2, 100);
                
                // 玩家总战力
                ctx.fillStyle = 'rgba(74, 222, 128, 0.2)';
                ctx.beginPath();
                ctx.arc(this.width / 2, this.height - 160, 35, 0, Math.PI * 2);
                ctx.fill();
                
                ctx.fillStyle = CONFIG.COLORS.PLAYER_POWER;
                ctx.font = 'bold 24px Cinzel, serif';
                ctx.fillText(playerPower.toString(), this.width / 2, this.height - 160);
                
                ctx.font = '10px Cinzel, serif';
                ctx.fillText('我方', this.width / 2, this.height - 120);
                
                // 各行战力
                ['player', 'enemy'].forEach(player => {
                    ['melee', 'ranged', 'siege'].forEach(row => {
                        const rect = this.rowRects[player][row];
                        const power = state.board[player][row].power || 0;
                        
                        ctx.fillStyle = player === 'player' ? CONFIG.COLORS.PLAYER_POWER : CONFIG.COLORS.ENEMY_POWER;
                        ctx.font = 'bold 16px Cinzel, serif';
                        ctx.textAlign = 'right';
                        ctx.fillText(power.toString(), rect.x + rect.w - 10, rect.y + rect.h / 2 + 5);
                    });
                });
            }
            
            // 天气效果
            updateWeatherParticles() {
                // 更新现有粒子
                for (let i = this.weatherParticles.length - 1; i >= 0; i--) {
                    const p = this.weatherParticles[i];
                    p.y += p.vy;
                    p.x += p.vx;
                    
                    if (p.y > this.height) {
                        this.weatherParticles.splice(i, 1);
                    }
                }
                
                // 生成新粒子
                if (this.currentWeather === 'rain') {
                    for (let i = 0; i < 3; i++) {
                        this.weatherParticles.push({
                            x: Math.random() * this.width,
                            y: -10,
                            vx: -1,
                            vy: 8 + Math.random() * 4,
                            type: 'rain'
                        });
                    }
                } else if (this.currentWeather === 'frost') {
                    for (let i = 0; i < 1; i++) {
                        this.weatherParticles.push({
                            x: Math.random() * this.width,
                            y: -10,
                            vx: Math.random() * 2 - 1,
                            vy: 1 + Math.random() * 2,
                            size: Math.random() * 3 + 1,
                            type: 'snow'
                        });
                    }
                }
            }
            
            drawWeatherEffects() {
                const ctx = this.ctx;
                
                this.weatherParticles.forEach(p => {
                    if (p.type === 'rain') {
                        ctx.strokeStyle = 'rgba(150, 180, 255, 0.5)';
                        ctx.lineWidth = 1;
                        ctx.beginPath();
                        ctx.moveTo(p.x, p.y);
                        ctx.lineTo(p.x + p.vx * 2, p.y + p.vy * 2);
                        ctx.stroke();
                    } else if (p.type === 'snow') {
                        ctx.fillStyle = 'rgba(255, 255, 255, 0.7)';
                        ctx.beginPath();
                        ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
                        ctx.fill();
                    }
                });
            }
            
            updateUI() {
                const state = this.engine.state;
                
                // 更新回合宝石
                const playerGem1 = document.getElementById('playerGem1');
                const playerGem2 = document.getElementById('playerGem2');
                const enemyGem1 = document.getElementById('enemyGem1');

            // 交互处理
            getCardAt(x, y) {
                // 从后往前检查（后绘制的在上面）
                const entries = Array.from(this.cardPositions.entries()).reverse();
                
                for (const [id, pos] of entries) {
                    const w = pos.width || CONFIG.CARD_WIDTH;
                    const h = pos.height || CONFIG.CARD_HEIGHT;
                    
                    if (x >= pos.x && x <= pos.x + w && y >= pos.y && y <= pos.y + h) {
                        return pos;
                    }
                }
                
                return null;
            }
            
            getRowAt(x, y) {
                for (const [name, rect] of Object.entries(this.rowRects.player)) {
                    if (x >= rect.x && x <= rect.x + rect.w && y >= rect.y && y <= rect.y + rect.h) {
                        return { name, rect, player: 'player' };
                    }
                }
                
                for (const [name, rect] of Object.entries(this.rowRects.enemy)) {
                    if (x >= rect.x && x <= rect.x + rect.w && y >= rect.y && y <= rect.y + rect.h) {
                        return { name, rect, player: 'enemy' };
                    }
                }
                
                return null;
            }
            
            startDrag(card, x, y) {
                this.draggingCard = card;
                this.dragPosition = { x, y };
                
                const pos = this.cardPositions.get(card.instanceId);
                if (pos) {
                    this.dragOffset = {
                        x: x - pos.x,
                        y: y - pos.y
                    };
                }
                
                // 计算有效放置区域
                this.updateHighlights(x, y);
            }
            
            updateDrag(x, y) {
                if (!this.draggingCard) return;
                
                this.dragPosition = { x, y };
                this.updateHighlights(x, y);
            }
            
            updateHighlights(x, y) {
                this.highlightedRows = [];
                
                if (!this.draggingCard) return;
                
                const validRows = this.engine.getValidRows(this.draggingCard, 'player');
                
                // 高亮所有有效行
                Object.entries(this.rowRects.player).forEach(([name, rect]) => {
                    const isValid = validRows.includes(name);
                    this.highlightedRows.push({ name, rect, valid: isValid });
                });
            }
            
            endDrag(x, y) {
                if (!this.draggingCard) return;
                
                const row = this.getRowAt(x, y);
                
                if (row && row.player === 'player') {
                    const validRows = this.engine.getValidRows(this.draggingCard, 'player');
                    
                    if (validRows.includes(row.name)) {
                        const result = this.engine.playCard('player', this.draggingCard, row.name);
                        
                        if (result.success) {
                            // 成功出牌特效
                            for (let i = 0; i < 10; i++) {
                                this.addParticle(x, y, 'rgb(74, 222, 128)', { x: 0, y: -2 }, 40);
                            }
                        } else {
                            this.showToast(result.reason, 'error');
                        }
                    } else {
                        this.showToast('无法将此卡放置在该排', 'error');
                    }
                }
                
                this.draggingCard = null;
                this.highlightedRows = [];
            }
            
            setHoveredCard(card) {
                this.hoveredCard = card;
            }
            
            showCardDetail(card) {
                const panel = document.getElementById('cardDetailPanel');
                if (!panel) return;
                
                // 更新面板内容
                document.getElementById('detailName').textContent = card.name;
                document.getElementById('detailFaction').textContent = 
                    card.faction === 'MONSTERS' ? '怪物' : 
                    card.faction === 'NORTHERN' ? '北方领域' : 
                    card.faction === 'NEUTRAL' ? '中立' : card.faction;
                
                document.getElementById('detailPower').textContent = 
                    card.type === 'UNIT' ? card.power : '-';
                
                document.getElementById('detailType').textContent = 
                    card.row === 'MELEE' ? '近战' : 
                    card.row === 'RANGED' ? '远程' : 
                    card.row === 'SIEGE' ? '攻城' : 
                    card.type === 'WEATHER' ? '天气' : '特殊';
                
                const abilitiesContainer = document.getElementById('detailAbilities');
                if (card.abilities && card.abilities.length > 0) {
                    abilitiesContainer.innerHTML = card.abilities.map(a => 
                        `<span class="ability-tag">${this.getAbilityName(a)}</span>`
                    ).join('');
                } else {
                    abilitiesContainer.innerHTML = '<span class="ability-tag">无</span>';
                }
                
                document.getElementById('detailDesc').textContent = card.desc || '这是一张基础卡牌。';
                
                // 更新图标
                const icon = document.getElementById('detailIcon');
                icon.className = 'card-detail-icon ' + card.color.toLowerCase();
                icon.textContent = card.type === 'UNIT' ? card.power : '★';
                
                panel.classList.add('visible');
            }
            
            getAbilityName(ability) {

                    'MEDIC': '医师',
                    'SPY': '间谍',

                };

            }
            
            hideCardDetail() {
                const panel = document.getElementById('cardDetailPanel');
                if (panel) {
                    panel.classList.remove('visible');
                }
            }
            
            setWeather(weather) {
                this.currentWeather = weather;
                this.weatherParticles = [];
            }
        }
        
        // ==================== 输入管理器 ====================
        class InputManager {
            constructor(renderer, ruleEngine) {
                this.renderer = renderer;
                this.engine = ruleEngine;
                this.canvas = renderer.canvas;
                
                this.init();
            }
            
            init() {
                // 鼠标事件
                this.canvas.addEventListener('mousemove', (e) => this.onMouseMove(e));
                this.canvas.addEventListener('mousedown', (e) => this.onMouseDown(e));
                this.canvas.addEventListener('mouseup', (e) => this.onMouseUp(e));
                this.canvas.addEventListener('contextmenu', (e) => this.onRightClick(e));
                
                // 触摸事件
                this.canvas.addEventListener('touchstart', (e) => this.onTouchStart(e));
                this.canvas.addEventListener('touchmove', (e) => this.onTouchMove(e));
                this.canvas.addEventListener('touchend', (e) => this.onTouchEnd(e));
                
                // UI按钮
                document.getElementById('newGameBtn')?.addEventListener('click', () => this.startNewGame());
                document.getElementById('passBtn')?.addEventListener('click', () => this.pass());
                document.getElementById('leaderBtn')?.addEventListener('click', () => this.useLeader());
                document.getElementById('closeDetailBtn')?.addEventListener('click', () => this.renderer.hideCardDetail());
                
                // 天气按钮
                document.querySelectorAll('.weather-btn').forEach(btn => {
                    btn.addEventListener('click', (e) => {
                        document.querySelectorAll('.weather-btn').forEach(b => b.classList.remove('active'));
                        btn.classList.add('active');
                        this.renderer.setWeather(btn.dataset.weather);
                    });
                });
                
                // 开始新游戏
                this.startNewGame();
            }
            
            startNewGame() {
                this.engine.initNewGame();
                this.renderer.showToast('新游戏开始！', 'success');
            }
            
            pass() {
                if (this.engine.state.currentPlayer !== 'player') return;
                if (this.engine.state.players.player.hasPassed) return;
                
                this.engine.pass('player');
            }
            
            useLeader() {
                const result = this.engine.useLeaderAbility('player');
                if (result.success) {
                    this.renderer.showToast('领袖技能已激活！', 'success');
                } else {
                    this.renderer.showToast(result.reason, 'error');
                }
            }
            
            onMouseMove(e) {
                const rect = this.canvas.getBoundingClientRect();
                const x = e.clientX - rect.left;
                const y = e.clientY - rect.top;
                
                if (this.renderer.draggingCard) {
                    this.renderer.updateDrag(x, y);
                } else {
                    const cardPos = this.renderer.getCardAt(x, y);
                    
                    if (cardPos && cardPos.inHand) {
                        this.renderer.setHoveredCard(cardPos.card);
                        this.canvas.style.cursor = 'grab';
                    } else {
                        this.renderer.setHoveredCard(null);
                        this.canvas.style.cursor = 'default';
                    }
                }
            }
            
            onMouseDown(e) {
                if (e.button !== 0) return;
                
                const rect = this.canvas.getBoundingClientRect();
                const x = e.clientX - rect.left;
                const y = e.clientY - rect.top;
                
                const cardPos = this.renderer.getCardAt(x, y);
                
                if (cardPos && cardPos.inHand) {
                    // 检查是否可以出牌
                    if (!this.engine.canPlayCard('player')) {
                        this.renderer.showToast('现在不是你的回合', 'error');
                        return;
                    }
                    
                    this.renderer.startDrag(cardPos.card, x, y);
                    this.canvas.style.cursor = 'grabbing';
                }
            }
            
            onMouseUp(e) {
                if (e.button !== 0) return;
                
                const rect = this.canvas.getBoundingClientRect();
                const x = e.clientX - rect.left;
                const y = e.clientY - rect.top;
                
                if (this.renderer.draggingCard) {
                    this.renderer.endDrag(x, y);
                    this.canvas.style.cursor = 'default';
                }
            }
            
            onRightClick(e) {
                e.preventDefault();
                
                const rect = this.canvas.getBoundingClientRect();
                const x = e.clientX - rect.left;
                const y = e.clientY - rect.top;
                
                const cardPos = this.renderer.getCardAt(x, y);
                
                if (cardPos) {
                    this.renderer.showCardDetail(cardPos.card);
                } else {
                    this.renderer.hideCardDetail();
                }
            }
            
            onTouchStart(e) {
                e.preventDefault();
                const touch = e.touches[0];
                const rect = this.canvas.getBoundingClientRect();
                const x = touch.clientX - rect.left;
                const y = touch.clientY - rect.top;
                
                const cardPos = this.renderer.getCardAt(x, y);
                
                if (cardPos && cardPos.inHand && this.engine.canPlayCard('player')) {
                    this.renderer.startDrag(cardPos.card, x, y);
                }
            }
            
            onTouchMove(e) {
                e.preventDefault();
                const touch = e.touches[0];
                const rect = this.canvas.getBoundingClientRect();
                const x = touch.clientX - rect.left;
                const y = touch.clientY - rect.top;
                
                this.renderer.updateDrag(x, y);
            }
            
            onTouchEnd(e) {
                e.preventDefault();
                const touch = e.changedTouches[0];
                const rect = this.canvas.getBoundingClientRect();
                const x = touch.clientX - rect.left;
                const y = touch.clientY - rect.top;
                
                this.renderer.endDrag(x, y);
            }
        }
        
        // ==================== 初始化 ====================
        function init() {
            const canvas = document.getElementById('gameCanvas');
            if (!canvas) {
                console.error('Canvas not found');
                return;
            }
            
            const engine = new GwentRuleEngine();
            const renderer = new GameRenderer(canvas, engine);
            const input = new InputManager(renderer, engine);
        }
        
        // 页面加载完成后初始化

            document.addEventListener('DOMContentLoaded', init);
        } else {
            init();

    })();
