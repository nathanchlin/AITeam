        // ==================== 动画系统 ====================
        class AnimationManager {
            constructor() {
                this.animations = [];
                this.powerPopups = [];
                this.shockwaves = [];
                this.trailParticles = [];
                this.hologramEffects = [];
                this.cameraShake = { x: 0, y: 0, intensity: 0, decay: 0.9 };
                this.cameraZoom = { current: 1, target: 1, speed: 0.05 };
                this.cameraPan = { x: 0, y: 0, targetX: 0, targetY: 0, speed: 0.08 };
            }
            
            // 卡牌打出飞行动画
            playCardFly(card, startX, startY, endX, endY, isGold = false, callback) {
                const animation = {
                    type: 'cardFly',
                    card: card,
                    startX: startX,
                    startY: startY,
                    endX: endX,
                    endY: endY,
                    x: startX,
                    y: startY,
                    progress: 0,
                    duration: 600,
                    startTime: Date.now(),
                    isGold: isGold,
                    callback: callback,
                    trailPoints: [],
                    easing: (t) => {
                        // 弹性缓动
                        const c4 = (2 * Math.PI) / 3;
                        return t === 0 ? 0 : t === 1 ? 1 :
                            Math.pow(2, -10 * t) * Math.sin((t * 10 - 0.75) * c4) + 1;
                    }
                };
                this.animations.push(animation);
                
                // 相机震动
                if (isGold) {
                    this.cameraShake.intensity = 8;
                } else {
                    this.cameraShake.intensity = 3;
                }
            }
            
            // 更新飞行动画拖尾粒子
            updateCardFlyTrail(animation) {
                if (animation.progress < 1) {
                    // 添加拖尾点
                    animation.trailPoints.push({
                        x: animation.x,
                        y: animation.y,
                        alpha: 1,
                        size: animation.isGold ? 8 : 5
                    });
                    
                    // 限制拖尾长度
                    if (animation.trailPoints.length > 20) {
                        animation.trailPoints.shift();
                    }
                }
                
                // 衰减拖尾
                animation.trailPoints.forEach((point, index) => {
                    point.alpha *= 0.92;
                    point.size *= 0.96;
                });
                
                // 移除消失的粒子
                animation.trailPoints = animation.trailPoints.filter(p => p.alpha > 0.05);
            }

                    });
                }
                
                this.animations.push(animation);
            }
            
            // 更新所有动画
            update() {
                const now = Date.now();
                
                // 更新卡牌飞行动画
                this.animations = this.animations.filter(anim => {
                    if (anim.type === 'cardFly') {
                        const elapsed = now - anim.startTime;
                        anim.progress = Math.min(elapsed / anim.duration, 1);
                        const easedProgress = anim.easing(anim.progress);
                        
                        anim.x = anim.startX + (anim.endX - anim.startX) * easedProgress;
                        anim.y = anim.startY + (anim.endY - anim.startY) * easedProgress;
                        
                        // 添加旋转效果
                        anim.rotation = (1 - anim.progress) * Math.PI * 0.5;
                        
                        this.updateCardFlyTrail(anim);
                        
                        if (anim.progress >= 1) {
                            // 到达终点，触发冲击波
                            if (anim.isGold) {
                                this.addShockwave(anim.endX, anim.endY, '#f0d060', 200);
                                this.addHologramEffect(anim.endX - 40, anim.endY - 55, 80, 110, 3000);
                            } else {
                                this.addShockwave(anim.endX, anim.endY, '#5a4a35', 80);
                            }
                            if (anim.callback) anim.callback();
                            return false;
                        }
                        return true;
                    }
                    
                    if (anim.type === 'roundEndClash') {
                        const elapsed = now - anim.startTime;
                        anim.progress = Math.min(elapsed / anim.duration, 1);
                        
                        // 闪光效果
                        if (anim.progress < 0.3) {
                            anim.flashIntensity = Math.sin(anim.progress / 0.3 * Math.PI);
                        } else {
                            anim.flashIntensity *= 0.9;
                        }
                        
                        // 更新粒子
                        anim.particles.forEach(p => {
                            p.x += p.vx;
                            p.y += p.vy;
                            p.vy += 0.2; // 重力
                            p.alpha *= 0.96;
                        });
                        
                        if (anim.progress >= 1) {
                            if (anim.callback) anim.callback();
                            return false;
                        }
                        return true;
                    }
                    
                    return true;
                });
                
                // 更新战力弹窗
                this.powerPopups = this.powerPopups.filter(popup => {
                    const elapsed = now - popup.startTime;
                    const progress = elapsed / popup.duration;
                    
                    // 弹跳效果
                    popup.bouncePhase += 0.15;
                    popup.offsetY = Math.sin(popup.bouncePhase) * 5 - (progress * 40);
                    
                    // 缩放弹跳
                    if (progress < 0.2) {
                        popup.scale = 0.5 + (progress / 0.2) * 0.8;
                    } else {
                        popup.scale = 1.3 - ((progress - 0.2) / 0.8) * 0.3;
                    }
                    
                    popup.alpha = 1 - Math.pow(progress, 2);
                    
                    return progress < 1;
                });
                
                // 更新冲击波
                this.shockwaves = this.shockwaves.filter(sw => {
                    sw.radius += sw.speed;
                    sw.alpha = 1 - (sw.radius / sw.maxRadius);
                    sw.lineWidth = Math.max(1, 4 * sw.alpha);
                    return sw.radius < sw.maxRadius;
                });
                
                // 更新全息效果
                this.hologramEffects = this.hologramEffects.filter(holo => {
                    const elapsed = now - holo.startTime;
                    holo.phase = elapsed / holo.duration;
                    return holo.phase < 1;
                });
                
                // 更新相机震动
                if (this.cameraShake.intensity > 0.1) {
                    this.cameraShake.x = (Math.random() - 0.5) * this.cameraShake.intensity;
                    this.cameraShake.y = (Math.random() - 0.5) * this.cameraShake.intensity;
                    this.cameraShake.intensity *= this.cameraShake.decay;
                } else {
                    this.cameraShake.x = 0;
                    this.cameraShake.y = 0;
                    this.cameraShake.intensity = 0;
                }
                
                // 更新相机缩放
                this.cameraZoom.current += (this.cameraZoom.target - this.cameraZoom.current) * this.cameraZoom.speed;
                
                // 更新相机平移
                this.cameraPan.x += (this.cameraPan.targetX - this.cameraPan.x) * this.cameraPan.speed;
                this.cameraPan.y += (this.cameraPan.targetY - this.cameraPan.y) * this.cameraPan.speed;
            }
            
            // 绘制所有动画
            draw(ctx, canvas) {
                const centerX = canvas.width / 2;
                const centerY = canvas.height / 2;
                
                // 应用相机变换
                ctx.save();
                ctx.translate(centerX + this.cameraPan.x + this.cameraShake.x, 
                             centerY + this.cameraPan.y + this.cameraShake.y);
                ctx.scale(this.cameraZoom.current, this.cameraZoom.current);
                ctx.translate(-centerX, -centerY);
                
                // 绘制卡牌飞行拖尾
                this.animations.forEach(anim => {
                    if (anim.type === 'cardFly' && anim.trailPoints.length > 0) {
                        anim.trailPoints.forEach((point, index) => {
                            const gradient = ctx.createRadialGradient(
                                point.x, point.y, 0,
                                point.x, point.y, point.size
                            );
                            
                            if (anim.isGold) {
                                gradient.addColorStop(0, `rgba(240, 208, 96, ${point.alpha * 0.8})`);
                                gradient.addColorStop(0.5, `rgba(201, 162, 39, ${point.alpha * 0.4})`);
                                gradient.addColorStop(1, `rgba(138, 112, 32, 0)`);
                            } else {
                                gradient.addColorStop(0, `rgba(90, 74, 53, ${point.alpha * 0.6})`);
                                gradient.addColorStop(1, `rgba(90, 74, 53, 0)`);
                            }
                            
                            ctx.fillStyle = gradient;
                            ctx.beginPath();
                            ctx.arc(point.x, point.y, point.size, 0, Math.PI * 2);
                            ctx.fill();
                        });
                    }
                });
                
                // 绘制冲击波
                this.shockwaves.forEach(sw => {
                    ctx.strokeStyle = sw.color;
                    ctx.globalAlpha = sw.alpha;
                    ctx.lineWidth = sw.lineWidth;
                    ctx.beginPath();
                    ctx.arc(sw.x, sw.y, sw.radius, 0, Math.PI * 2);
                    ctx.stroke();
                    ctx.globalAlpha = 1;
                });
                
                // 绘制全息效果
                this.hologramEffects.forEach(holo => {
                    const alpha = (1 - holo.phase) * 0.6;
                    
                    // 彩虹边缘光
                    for (let i = 0; i < 3; i++) {
                        const hue = (holo.phase * 360 + i * 120) % 360;
                        const offset = Math.sin(holo.phase * Math.PI * 4 + i) * 3;
                        
                        ctx.strokeStyle = `hsla(${hue}, 70%, 60%, ${alpha * 0.5})`;
                        ctx.lineWidth = 2;
                        ctx.strokeRect(
                            holo.x - offset, 
                            holo.y - offset, 
                            holo.width + offset * 2, 
                            holo.height + offset * 2
                        );
                    }
                    
                    // 闪烁高光
                    if (Math.sin(holo.phase * Math.PI * 8) > 0.7) {
                        const glowGradient = ctx.createLinearGradient(
                            holo.x, holo.y,
                            holo.x + holo.width, holo.y + holo.height
                        );
                        glowGradient.addColorStop(0, `rgba(255, 255, 255, 0)`);
                        glowGradient.addColorStop(0.5, `rgba(255, 255, 255, ${alpha * 0.3})`);
                        glowGradient.addColorStop(1, `rgba(255, 255, 255, 0)`);
                        
                        ctx.fillStyle = glowGradient;
                        ctx.fillRect(holo.x, holo.y, holo.width, holo.height);
                    }
                });
                
                // 绘制战力弹窗
                ctx.textAlign = 'center';
                ctx.textBaseline = 'middle';
                this.powerPopups.forEach(popup => {
                    ctx.save();
                    ctx.translate(popup.x, popup.y + popup.offsetY);
                    ctx.scale(popup.scale, popup.scale);
                    ctx.globalAlpha = popup.alpha;
                    
                    // 文字阴影
                    ctx.font = 'bold 24px Cinzel, serif';
                    const text = (popup.isPositive ? '+' : '') + popup.value;
                    const color = popup.isPositive ? '#4ade80' : '#f87171';
                    
                    // 外发光
                    ctx.shadowColor = color;
                    ctx.shadowBlur = 10;
                    ctx.fillStyle = color;
                    ctx.fillText(text, 0, 0);
                    
                    // 描边
                    ctx.shadowBlur = 0;
                    ctx.strokeStyle = 'rgba(0, 0, 0, 0.5)';
                    ctx.lineWidth = 3;
                    ctx.strokeText(text, 0, 0);
                    ctx.fillText(text, 0, 0);
                    
                    ctx.restore();
                });
                
                // 绘制回合结束对冲特效
                this.animations.forEach(anim => {
                    if (anim.type === 'roundEndClash') {
                        // 闪光
                        if (anim.flashIntensity > 0) {
                            ctx.fillStyle = `rgba(255, 255, 255, ${anim.flashIntensity * 0.3})`;
                            ctx.fillRect(0, 0, canvas.width, canvas.height);
                        }
                        
                        // 粒子
                        anim.particles.forEach(p => {
                            ctx.globalAlpha = p.alpha;
                            ctx.fillStyle = p.color;
                            ctx.beginPath();
                            ctx.arc(
                                centerX + p.x + this.cameraShake.x, 
                                centerY + p.y + this.cameraShake.y, 
                                p.size, 0, Math.PI * 2
                            );
                            ctx.fill();
                        });
                        ctx.globalAlpha = 1;
                    }
                });
                
                ctx.restore();
            }
            
            // 绘制飞行中的卡牌（需要在卡牌绘制后调用）
            drawFlyingCards(ctx, cardRenderer) {
                this.animations.forEach(anim => {
                    if (anim.type === 'cardFly' && anim.progress < 1) {
                        ctx.save();
                        ctx.translate(anim.x, anim.y);
                        ctx.rotate(anim.rotation || 0);
                        ctx.globalAlpha = 1 - anim.progress * 0.3;
                        
                        // 绘制卡牌
                        cardRenderer.drawCard(ctx, anim.card, 0, 0, 1, true);
                        
                        // 金卡特效增强
                        if (anim.isGold) {
                            const glowIntensity = Math.sin(anim.progress * Math.PI * 4) * 0.5 + 0.5;
                            ctx.shadowColor = '#f0d060';
                            ctx.shadowBlur = 20 * glowIntensity;
                        }
                        
                        ctx.restore();
                    }
                });
            }
        }
        
        // ==================== 粒子系统 ====================
        class ParticleSystem {
            constructor() {
                this.particles = [];
            }
            
            emit(x, y, count, options = {}) {
                for (let i = 0; i < count; i++) {
                    this.particles.push({
                        x: x,
                        y: y,
                        vx: (Math.random() - 0.5) * (options.speed || 2),
                        vy: (Math.random() - 0.5) * (options.speed || 2) - (options.gravity ? 2 : 0),
                        life: 1,
                        decay: options.decay || 0.02,
                        color: options.color || '#c9a227',
                        size: options.size || 3
                    });
                }
            }
            
            update() {
                this.particles = this.particles.filter(p => {
                    p.x += p.vx;
                    p.y += p.vy;
                    p.life -= p.decay;
                    return p.life > 0;
                });
            }
            
            draw(ctx) {
                this.particles.forEach(p => {
                    ctx.globalAlpha = p.life;
                    ctx.fillStyle = p.color;
                    ctx.beginPath();
                    ctx.arc(p.x, p.y, p.size * p.life, 0, Math.PI * 2);
                    ctx.fill();
                });
                ctx.globalAlpha = 1;
            }
        }
        
        // ==================== 核心规则引擎 ====================
        class GwentRuleEngine {
            constructor() {
                this.state = this.getInitialState();
                this.listeners = [];
            }
            
            getInitialState() {
                return {
                    phase: 'GAME_START',
                    currentRound: 1,
                    turnCount: 0,
                    currentPlayer: 'player',
                    startingPlayer: 'player',
                    
                    players: {
                        player: {
                            roundWins: 0,
                            totalPower: 0,
                            hasPassed: false,
                            hand: [],
                            deck: [],
                            graveyard: [],
                            leader: null,
                            leaderUsed: false
                        },
                        enemy: {
                            roundWins: 0,
                            totalPower: 0,
                            hasPassed: false,
                            hand: [],
                            deck: [],
                            graveyard: [],
                            leader: null,
                            leaderUsed: false
                        }
                    },
                    
                    board: {
                        player: {
                            melee: { cards: [], power: 0, hasHorn: false },
                            ranged: { cards: [], power: 0, hasHorn: false },
                            siege: { cards: [], power: 0, hasHorn: false }
                        },
                        enemy: {
                            melee: { cards: [], power: 0, hasHorn: false },
                            ranged: { cards: [], power: 0, hasHorn: false },
                            siege: { cards: [], power: 0, hasHorn: false }
                        }
                    },
                    
                    weather: {
                        melee: null,
                        ranged: null,
                        siege: null
                    },
                    
                    eventLog: [],
                    gameOver: false,
                    winner: null
                };
            }
            
            subscribe(callback) {
                this.listeners.push(callback);
                return () => {
                    this.listeners = this.listeners.filter(l => l !== callback);
                };
            }
            
            emit(event) {
                this.listeners.forEach(cb => cb(event, this.state));
            }
            
            log(message, type = 'system') {
                const entry = {
                    time: Date.now(),
                    message,
                    type
                };
                this.state.eventLog.push(entry);
                this.emit({ type: 'LOG', entry });
            }
            
            initNewGame() {
                this.state = this.getInitialState();
                
                const playerDeck = this.buildDeck('monsters');
                const enemyDeck = this.buildDeck('northern');
                
                this.state.players.player.deck = playerDeck;
                this.state.players.enemy.deck = enemyDeck;
                this.state.players.player.leader = CardDatabase.leaders[0];
                this.state.players.enemy.leader = CardDatabase.leaders[2];
                
                this.shuffleDeck('player');
                this.shuffleDeck('enemy');
                
                this.drawCards('player', 10);
                this.drawCards('enemy', 10);
                
                this.state.phase = 'ROUND_PLAY';
                this.state.currentPlayer = 'player';
                this.state.startingPlayer = 'player';
                
                this.log('新游戏开始！第1回合', 'system');
                this.emit({ type: 'GAME_INIT' });
            }
            
            buildDeck(faction) {
                let deck = [];
                const factionCards = CardDatabase[faction] || [];
                
                factionCards.forEach(card => {
                    deck.push({ ...card, instanceId: `${card.id}_${Math.random().toString(36).substr(2, 9)}` });
                });
                
                CardDatabase.monsters.filter(c => c.faction === 'NEUTRAL').forEach(card => {
                    deck.push({ ...card, instanceId: `${card.id}_${Math.random().toString(36).substr(2, 9)}` });
                });
                
                return deck;
            }
            
            shuffleDeck(player) {
                const deck = this.state.players[player].deck;
                for (let i = deck.length - 1; i > 0; i--) {
                    const j = Math.floor(Math.random() * (i + 1));
                    [deck[i], deck[j]] = [deck[j], deck[i]];
                }
            }
            
            drawCards(player, count) {
                const playerData = this.state.players[player];
                const drawn = [];
                for (let i = 0; i < count && playerData.deck.length > 0; i++) {
                    const card = playerData.deck.pop();
                    playerData.hand.push(card);
                    drawn.push(card);
                }
                return drawn;
            }
            
            canPlayCard(player) {
                if (this.state.phase !== 'ROUND_PLAY') return false;
                if (this.state.currentPlayer !== player) return false;
                if (this.state.players[player].hasPassed) return false;
                if (this.state.gameOver) return false;
                return true;
            }
            
            getValidRows(card, player) {
                if (card.type === 'WEATHER') return ['weather'];
                if (card.type === 'SPECIAL') return ['special'];
                if (card.abilities && card.abilities.includes('SPY')) {
                    return card.row ? [card.row.toLowerCase()] : ['melee', 'ranged', 'siege'];
                }
                if (card.row) {
                    return [card.row.toLowerCase()];
                }
                return ['melee', 'ranged', 'siege'];
            }
            
            playCard(player, card, targetRow) {
                if (!this.canPlayCard(player)) {
                    this.log(`${player} 无法出牌`, 'system');
                    return false;
                }
                
                const hand = this.state.players[player].hand;
                const cardIndex = hand.findIndex(c => c.instanceId === card.instanceId);
                if (cardIndex === -1) {
                    this.log('卡牌不在手牌中', 'system');
                    return false;
                }
                
                hand.splice(cardIndex, 1);
                
                // 处理天气卡
                if (card.type === 'WEATHER') {
                    this.playWeatherCard(player, card);
                    this.emit({ type: 'CARD_PLAYED', player, card, row: 'weather' });
                    this.switchTurn();
                    return true;
                }
                
                // 处理特殊卡
                if (card.type === 'SPECIAL') {
                    this.playSpecialCard(player, card, targetRow);
                    this.emit({ type: 'CARD_PLAYED', player, card, row: 'special' });
                    this.switchTurn();
                    return true;
                }
                
                // 处理单位卡
                const actualTarget = player === 'player' ? 'player' : 'enemy';
                if (card.abilities && card.abilities.includes('SPY')) {
                    // 间谍卡放到对方场地
                    const targetPlayer = player === 'player' ? 'enemy' : 'player';
                    this.state.board[targetPlayer][targetRow].cards.push({ ...card, owner: player });
                    this.log(`${player} 将间谍 ${card.name} 放置到 ${targetPlayer} 的 ${targetRow} 排`, 'system');
                } else {
                    this.state.board[actualTarget][targetRow].cards.push({ ...card, owner: player });
                }
                
                this.log(`${player} 打出 ${card.name} 到 ${targetRow} 排`, player);
                this.emit({ type: 'CARD_PLAYED', player, card, row: targetRow });
                
                // 触发能力
                this.triggerAbilities(card, player, targetRow);
                
                this.calculateAllPower();
                this.switchTurn();
                return true;
            }
            
            playWeatherCard(player, card) {
                const weatherMap = {
                    'BITING_FROST': 'melee',
                    'IMPENETRABLE_FOG': 'ranged',
                    'TORRENTIAL_RAIN': 'siege'
                };
                
                const targetRow = weatherMap[card.abilityType];
                if (targetRow) {
                    this.state.weather[targetRow] = card;
                    this.log(`${player} 打出天气卡 ${card.name}`, player);
                }
            }
            
            playSpecialCard(player, card, targetRow) {
                if (card.abilityType === 'SCORCH') {
                    this.scorchAbility();
                } else if (card.abilityType === 'COMMANDERS_HORN') {
                    if (targetRow && this.state.board.player[targetRow]) {
                        this.state.board.player[targetRow].hasHorn = true;
                        this.state.board.enemy[targetRow].hasHorn = true;
                    }
                }
                this.log(`${player} 打出特殊卡 ${card.name}`, player);
            }
            
            scorchAbility() {
                let maxPower = 0;
                ['player', 'enemy'].forEach(p => {
                    ['melee', 'ranged', 'siege'].forEach(r => {
                        this.state.board[p][r].cards.forEach(c => {
                            if (c.power > maxPower && c.color !== 'GOLD') {
                                maxPower = c.power;
                            }
                        });
                    });
                });
                
                ['player', 'enemy'].forEach(p => {
                    ['melee', 'ranged', 'siege'].forEach(r => {
                        const row = this.state.board[p][r];
                        const toRemove = row.cards.filter(c => c.power === maxPower && c.color !== 'GOLD');
                        toRemove.forEach(c => {
                            this.state.players[c.owner || p].graveyard.push(c);
                        });
                        row.cards = row.cards.filter(c => !(c.power === maxPower && c.color !== 'GOLD'));
                    });
                });
                
                this.log(`灼烧! 摧毁了所有战力为 ${maxPower} 的非金卡单位`, 'system');
            }
            
            triggerAbilities(card, player, row) {
                if (!card.abilities) return;
                
                card.abilities.forEach(ability => {
                    switch (ability) {
                        case 'MUSTER':
                            this.musterAbility(card, player, row);
                            break;
                        case 'BOND':
                            this.bondAbility(card, player, row);
                            break;
                        case 'MEDIC':
                            this.medicAbility(player);
                            break;
                    }
                });
            }
            
            musterAbility(card, player, row) {
                const hand = this.state.players[player].hand;
                const deck = this.state.players[player].deck;
                const matching = [...hand, ...deck].filter(c => c.name === card.name);
                
                matching.slice(0, 2).forEach(c => {
                    const handIdx = hand.findIndex(h => h.instanceId === c.instanceId);
                    if (handIdx !== -1) {
                        hand.splice(handIdx, 1);
                        this.state.board[player][row].cards.push({ ...c, owner: player });
                    }
                });
                
                if (matching.length > 0) {
                    this.log(`${card.name} 召唤了 ${Math.min(matching.length, 2)} 个同名牌`, 'system');
                }
            }
            
            bondAbility(card, player, row) {
                const rowCards = this.state.board[player][row].cards;
                const sameCards = rowCards.filter(c => c.name === card.name);
                if (sameCards.length > 1) {
                    sameCards.forEach(c => {
                        c.bondBonus = (sameCards.length - 1) * card.power;
                    });
                }
            }
            
            medicAbility(player) {
                const graveyard = this.state.players[player].graveyard;
                const nonHero = graveyard.filter(c => c.type === 'UNIT' && c.color !== 'GOLD');
                if (nonHero.length > 0) {
                    const revived = nonHero[nonHero.length - 1];
                    graveyard.splice(graveyard.indexOf(revived), 1);
                    this.state.players[player].hand.push(revived);
                    this.log(`医疗兵复活了 ${revived.name}`, player);
                }
            }
            
            calculateAllPower() {
                ['player', 'enemy'].forEach(p => {
                    let total = 0;
                    ['melee', 'ranged', 'siege'].forEach(r => {
                        const row = this.state.board[p][r];
                        row.power = 0;
                        
                        row.cards.forEach(card => {
                            let power = card.power || 0;
                            
                            // 天气效果
                            if (this.state.weather[r] && card.color !== 'GOLD') {
                                power = 1;
                            }
                            
                            // 羁绊加成
                            if (card.bondBonus) {
                                power += card.bondBonus;
                            }
                            
                            card.currentPower = power;
                            row.power += power;
                        });
                        
                        // 指挥号角
                        if (row.hasHorn) {
                            row.power *= 2;
                        }
                        
                        total += row.power;
                    });
                    
                    this.state.players[p].totalPower = total;
                });
            }
            
            switchTurn() {
                const player = this.state.players.player;
                const enemy = this.state.players.enemy;
                
                // 检查双方是否都放弃
                if (player.hasPassed && enemy.hasPassed) {
                    this.endRound();
                    return;
                }
                
                // 切换玩家
                if (this.state.currentPlayer === 'player') {
                    if (!enemy.hasPassed) {
                        this.state.currentPlayer = 'enemy';
                    }
                } else {
                    if (!player.hasPassed) {
                        this.state.currentPlayer = 'player';
                    }
                }
                
                this.state.turnCount++;
                
                // AI回合
                if (this.state.currentPlayer === 'enemy' && !enemy.hasPassed) {
                    setTimeout(() => this.aiPlay(), 800);
                }
            }
            
            aiPlay() {
                const enemy = this.state.players.enemy;
                if (enemy.hand.length === 0) {
                    this.pass('enemy');
                    return;
                }
                
                // 简单AI：根据战力差距决定策略
                const powerDiff = this.state.players.player.totalPower - enemy.totalPower;
                
                if (powerDiff > 30 && enemy.hand.length > 2) {
                    // 差距太大，考虑放弃
                    if (Math.random() > 0.3) {
                        this.pass('enemy');
                        return;
                    }
                }
                
                // 选择一张牌打出
                const card = enemy.hand[Math.floor(Math.random() * enemy.hand.length)];
                const validRows = this.getValidRows(card, 'enemy');
                const targetRow = validRows[Math.floor(Math.random() * validRows.length)];
                
                this.playCard('enemy', card, targetRow);
            }
            
            pass(player) {
                this.state.players[player].hasPassed = true;
                this.log(`${player} 放弃跟牌`, player);
                this.emit({ type: 'PLAYER_PASSED', player });
                
                // 检查回合是否结束
                if (this.state.players.player.hasPassed && this.state.players.enemy.hasPassed) {
                    this.endRound();
                } else {
                    this.switchTurn();
                }
            }
            
            endRound() {
                const playerPower = this.state.players.player.totalPower;
                const enemyPower = this.state.players.enemy.totalPower;
                
                this.log(`回合 ${this.state.currentRound} 结束! 玩家: ${playerPower}, 敌方: ${enemyPower}`, 'system');
                
                // 触发回合结束特效
                this.emit({ 
                    type: 'ROUND_END', 
                    playerPower, 
                    enemyPower,
                    round: this.state.currentRound
                });
                
                let winner = null;
                if (playerPower > enemyPower) {
                    this.state.players.player.roundWins++;
                    winner = 'player';
                    this.log('玩家赢得此回合!', 'player');
                } else if (enemyPower > playerPower) {
                    this.state.players.enemy.roundWins++;
                    winner = 'enemy';
                    this.log('敌方赢得此回合!', 'enemy');
                } else {
                    this.state.players.player.roundWins++;
                    this.state.players.enemy.roundWins++;
                    this.log('平局! 双方各得一分', 'system');
                }
                
                // 检查游戏是否结束
                if (this.state.players.player.roundWins >= 2) {
                    this.state.gameOver = true;
                    this.state.winner = 'player';
                    this.log('玩家赢得比赛!', 'player');
                    this.emit({ type: 'GAME_OVER', winner: 'player' });
                    return;
                }
                
                if (this.state.players.enemy.roundWins >= 2) {
                    this.state.gameOver = true;
                    this.state.winner = 'enemy';
                    this.log('敌方赢得比赛!', 'enemy');
                    this.emit({ type: 'GAME_OVER', winner: 'enemy' });
                    return;
                }
                
                // 开始新回合
                setTimeout(() => this.startNewRound(), 2000);
            }
            
            startNewRound() {
                this.state.currentRound++;
                this.state.turnCount = 0;
                
                // 重置状态
                this.state.players.player.hasPassed = false;
                this.state.players.enemy.hasPassed = false;
                
                // 清空战场
                ['player', 'enemy'].forEach(p => {
                    ['melee', 'ranged', 'siege'].forEach(r => {
                        const row = this.state.board[p][r];
                        row.cards.forEach(c => {
                            this.state.players[c.owner || p].graveyard.push(c);
                        });
                        row.cards = [];
                        row.power = 0;
                        row.hasHorn = false;
                    });
                });
                
                // 清除天气
                this.state.weather = { melee: null, ranged: null, siege: null };
                
                // 抽2张牌
                this.drawCards('player', 2);
                this.drawCards('enemy', 2);
                
                this.log(`第 ${this.state.currentRound} 回合开始!`, 'system');
                this.state.phase = 'ROUND_PLAY';
                
                // 决定先手
                this.state.currentPlayer = this.state.startingPlayer === 'player' ? 'enemy' : 'player';
                this.state.startingPlayer = this.state.currentPlayer;
                
                this.emit({ type: 'ROUND_START', round: this.state.currentRound });
                
                if (this.state.currentPlayer === 'enemy') {
                    setTimeout(() => this.aiPlay(), 500);
                }
            }
        }

                            p.y += p.speed * 0.5;

            roundRect(ctx, x, y, w, h, r) {
                ctx.beginPath();
                ctx.moveTo(x + r, y);
                ctx.lineTo(x + w - r, y);
                ctx.quadraticCurveTo(x + w, y, x + w, y + r);
                ctx.lineTo(x + w, y + h - r);
                ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
                ctx.lineTo(x + r, y + h);
                ctx.quadraticCurveTo(x, y + h, x, y + h - r);
                ctx.lineTo(x, y + r);
                ctx.quadraticCurveTo(x, y, x + r, y);
                ctx.closePath();

                const ctx = this.ctx;

                const cardWidth = CONFIG.CARD_WIDTH;
                const cardHeight = CONFIG.CARD_HEIGHT;
                const maxSpread = Math.min(600, this.canvas.width * 0.7);
                const spacing = Math.min(cardWidth + 5, maxSpread / hand.length);
                const totalWidth = (hand.length - 1) * spacing + cardWidth;
                const startX = centerX - totalWidth / 2;
                const baseY = this.canvas.height - 40;
                
                hand.forEach((card, i) => {
                    const x = startX + i * spacing;
                    const angle = (i - (hand.length - 1) / 2) * 0.05;
                    const yOffset = Math.abs(i - (hand.length - 1) / 2) * 5;
                    
                    ctx.save();
                    ctx.translate(x + cardWidth / 2, baseY);
                    ctx.rotate(angle);
                    ctx.translate(-cardWidth / 2, -cardHeight / 2);
                    
                    const isHovered = this.hoveredCard && this.hoveredCard.instanceId === card.instanceId;
                    const isSelected = this.selectedCard && this.selectedCard.instanceId === card.instanceId;
                    
                    if (isHovered) {
                        ctx.translate(0, -20);
                        ctx.scale(1.2, 1.2);
                    }
                    
                    if (isSelected) {
                        ctx.translate(0, -30);
                        ctx.scale(1.15, 1.15);
                    }
                    
                    // 不绘制正在拖拽的卡牌（会在drawDraggingCard中绘制）
                    if (this.draggingCard && this.draggingCard.instanceId === card.instanceId) {
                        ctx.restore();
                        return;
                    }
                    
                    this.drawCard(ctx, card, 0, 0, 1, isHovered || isSelected);
                    
                    // 选中高亮
                    if (isSelected) {
                        ctx.strokeStyle = CONFIG.COLORS.CARD_SELECTED;
                        ctx.lineWidth = 3;
                        this.roundRect(ctx, 0, 0, cardWidth, cardHeight, 5);
                        ctx.stroke();
                        
                        // 发光效果
                        ctx.shadowColor = CONFIG.COLORS.CARD_SELECTED;
                        ctx.shadowBlur = 15;
                        ctx.stroke();
                    }
                    
                    ctx.restore();
                    
                    // 存储卡牌位置用于点击检测
                    card._bounds = {
                        x: x,
                        y: baseY - cardHeight / 2 - (isHovered ? 20 : 0) - (isSelected ? 30 : 0),
                        w: cardWidth * (isHovered ? 1.2 : (isSelected ? 1.15 : 1)),
                        h: cardHeight * (isHovered ? 1.2 : (isSelected ? 1.15 : 1))
                    };
                });

            // 绘制拖拽中的卡牌
            drawDraggingCard() {

                ctx.save();
                ctx.globalAlpha = CONFIG.DRAG_ALPHA;
                
                this.drawCard(ctx, this.draggingCard, 
                    this.dragPosition.x - CONFIG.CARD_WIDTH / 2, 
                    this.dragPosition.y - CONFIG.CARD_HEIGHT / 2, 
                    1.1, true);
                
                ctx.restore();
            }

                this.particles.update();
            }

                this.particles.draw(this.ctx);

                this.drawHand(state.players.player.hand, state);
                this.drawDraggingCard();
                this.drawPowerBars(state);
                this.drawRoundInfo(state);
                this.drawAnimations();
            }

        class GwentGame {
            constructor() {

            setupEventListeners() {

            setupGameListeners() {
                this.ruleEngine.subscribe((event, state) => {
                    switch (event.type) {
                        case 'LOG':
                            this.addLogEntry(event.entry);
                            break;
                            
                        case 'CARD_PLAYED':
                            // 触发卡牌打出动画
                            if (event.player === 'player') {
                                const card = event.card;
                                const isGold = card.color === 'GOLD';
                                
                                // 获取起始位置（手牌位置）
                                const hand = state.players.player.hand;
                                const cardIndex = hand.findIndex(c => c.instanceId === card.instanceId);
                                const startX = this.canvas.width / 2;
                                const startY = this.canvas.height - 100;
                                
                                // 获取目标位置
                                const targetPos = this.getRowPosition(event.row, 'player');
                                
                                // 播放飞行动画
                                this.renderer.animationManager.playCardFly(
                                    card,
                                    startX, startY,
                                    targetPos.x, targetPos.y,
                                    isGold,
                                    () => {
                                        // 添加战力弹窗
                                        if (card.power > 0) {
                                            this.renderer.animationManager.addPowerPopup(
                                                targetPos.x, targetPos.y - 50,
                                                card.power, true
                                            );
                                        }
                                    }
                                );
                            }
                            break;
                            
                        case 'ROUND_END':
                            // 播放回合结束特效
                            this.showRoundResult(event.playerPower, event.enemyPower);
                            this.renderer.animationManager.playRoundEndClash(
                                event.playerPower,

                            // 回合切换镜头动画
                            this.renderer.animationManager.playTurnTransition(
                                this.ruleEngine.state.currentPlayer,
                                this.ruleEngine.state.currentPlayer === 'player' ? 'enemy' : 'player'
                            );
                            document.getElementById('roundResultOverlay').classList.remove('show');
                            break;
                            
                        case 'GAME_OVER':
                            alert(event.winner === 'player' ? '恭喜你赢得比赛!' : '你输掉了比赛!');
                            break;
                    }
                });

            // 获取排的位置（用于动画）
            getRowPosition(rowName, player) {
                const centerX = this.canvas.width / 2;
                const boardWidth = Math.min(800, this.canvas.width - 100);
                const rowHeight = 100;
                const startX = centerX - boardWidth / 2;
                
                let y;
                if (player === 'player') {
                    const offsets = { melee: 2, ranged: 1, siege: 0 };
                    y = this.canvas.height - 80 - offsets[rowName] * rowHeight + 30;
                } else {
                    const offsets = { melee: 0, ranged: 1, siege: 2 };
                    y = 80 + offsets[rowName] * rowHeight + 30;

                return { x: centerX, y: y };

            handleMouseMove(e) {

                        this.renderer.draggingCard, 'player'
                    ).filter(r => r !== 'weather' && r !== 'special');

                this.renderer.hoveredCard = null;
                const hand = this.ruleEngine.state.players.player.hand;
                
                hand.forEach(card => {
                    if (card._bounds) {
                        const b = card._bounds;
                        if (x >= b.x && x <= b.x + b.w && y >= b.y && y <= b.y + b.h) {
                            this.renderer.hoveredCard = card;
                        }
                    }
                });
            }
            
            handleClick(e) {
                // 如果正在拖拽，忽略点击
                if (this.renderer.draggingCard) return;
                
                const hand = this.ruleEngine.state.players.player.hand;
                
                // 检测点击手牌
                hand.forEach(card => {
                    if (card._bounds && this.renderer.hoveredCard && 
                        this.renderer.hoveredCard.instanceId === card.instanceId) {
                        
                        if (this.renderer.selectedCard && 
                            this.renderer.selectedCard.instanceId === card.instanceId) {
                            // 取消选择
                            this.renderer.selectedCard = null;
                        } else {
                            this.renderer.selectedCard = card;
                        }
                    }
                });
            }
            
            handleMouseDown(e) {
                if (this.renderer.hoveredCard && this.ruleEngine.canPlayCard('player')) {
                    this.renderer.draggingCard = this.renderer.hoveredCard;
                    this.renderer.dragPosition = {
                        x: e.clientX - this.canvas.getBoundingClientRect().left,
                        y: e.clientY - this.canvas.getBoundingClientRect().top
                    };
                    this.renderer.selectedCard = null;
                }
            }
            
            handleMouseUp(e) {
                if (!this.renderer.draggingCard) return;
                
                const rect = this.canvas.getBoundingClientRect();
                const x = e.clientX - rect.left;
                const y = e.clientY - rect.top;
                
                // 检测放置位置
                const dropRow = this.getDropRow(x, y);
                
                if (dropRow && this.renderer.validDropZones.includes(dropRow)) {
                    // 打出卡牌
                    const card = this.renderer.draggingCard;
                    const success = this.ruleEngine.playCard('player', card, dropRow);
                    
                    if (success) {
                        // 触发卡牌打出动画

                            card,

                                    );
                                }
                            }
                        );

                this.renderer.validDropZones = [];
            }
            
            handleRightClick(e) {

                const x = e.clientX - rect.left;
                const y = e.clientY - rect.top;

                            targetCard = card;

            }
            
            getDropRow(x, y) {
                const centerX = this.canvas.width / 2;
                const boardWidth = Math.min(800, this.canvas.width - 100);
                const startX = centerX - boardWidth / 2;
                const rowHeight = 100;
                
                // 检查是否在区域内
                if (x < startX || x > startX + boardWidth) return null;
                
                // 玩家区域
                const rows = ['siege', 'ranged', 'melee'];
                for (let i = 0; i < 3; i++) {
                    const rowY = this.canvas.height - 80 - (3 - i) * rowHeight;
                    if (y >= rowY && y < rowY + rowHeight) {
                        return rows[i];
                    }
                }
                
                return null;
            }
            
            showCardDetail(card) {

## 新增的动画系统功能说明

### 1. 卡牌打出飞行动画
- 使用弹性缓动函数实现自然的飞行轨迹
- 金卡和铜卡有不同的拖尾粒子颜色
- 拖尾粒子会逐渐衰减消失

### 2. 金卡/传奇卡特效
- **全息边缘光**：彩虹色的边框渐变，随时间变化
- **冲击波**：打出时产生扩散的圆形冲击波
- **高光闪烁**：随机的白色高光扫过效果

### 3. 战力数值弹跳动画
- 绿色表示增加战力，红色表示减少
- 数字会上下弹跳并逐渐上浮消失
- 带有外发光和描边效果

### 4. 回合切换镜头推拉
- 相机平滑缩放和平移
- 配合轻微的震动效果增强冲击感

### 5. 回合结束战力条对冲
- 双色粒子对冲爆炸效果
- 屏幕闪光增强史诗感
- 显示回合结算面板
