        // ==================== 核心规则引擎 ====================
        class GwentRuleEngine {
            constructor() {
                this.state = this.getInitialState();
                this.listeners = [];
            }
            
            getInitialState() {
                return {
                    phase: 'GAME_START',
                    currentRound: 1,
                    turnCount: 0,
                    currentPlayer: 'player',
                    startingPlayer: 'player',
                    
                    players: {
                        player: {

                            totalPower: 0,

                        },

                            roundWins: 0,
                            totalPower: 0,
                            hasPassed: false,
                            hand: [],
                            deck: [],
                            graveyard: [],
                            leader: null,
                            leaderUsed: false
                        }
                    },
                    
                    board: {
                        player: {
                            melee: { cards: [], power: 0, hasHorn: false },
                            ranged: { cards: [], power: 0, hasHorn: false },
                            siege: { cards: [], power: 0, hasHorn: false }
                        },
                        enemy: {
                            melee: { cards: [], power: 0, hasHorn: false },
                            ranged: { cards: [], power: 0, hasHorn: false },
                            siege: { cards: [], power: 0, hasHorn: false }
                        }
                    },
                    
                    weather: {
                        melee: null,
                        ranged: null,
                        siege: null
                    },
                    
                    eventLog: [],
                    gameOver: false,
                    winner: null
                };
            }
            
            // 订阅状态变化
            subscribe(callback) {
                this.listeners.push(callback);
                return () => {
                    this.listeners = this.listeners.filter(l => l !== callback);
                };
            }
            
            emit(event) {
                this.listeners.forEach(cb => cb(event, this.state));
            }
            
            // 日志记录
            log(message, type = 'system') {
                const entry = {
                    time: Date.now(),
                    message,
                    type
                };
                this.state.eventLog.push(entry);
                this.emit({ type: 'LOG', entry });
            }
            
            // 初始化新游戏
            initNewGame() {
                this.state = this.getInitialState();
                
                // 构建牌组
                const playerDeck = this.buildDeck('monsters');
                const enemyDeck = this.buildDeck('northern');
                
                this.state.players.player.deck = playerDeck;
                this.state.players.enemy.deck = enemyDeck;
                this.state.players.player.leader = CardDatabase.leaders[0];
                this.state.players.enemy.leader = CardDatabase.leaders[2];
                
                // 洗牌
                this.shuffleDeck('player');
                this.shuffleDeck('enemy');
                
                // 抽起始手牌
                this.drawCards('player', 10);
                this.drawCards('enemy', 10);
                
                this.state.phase = 'ROUND_PLAY';
                this.state.currentPlayer = 'player';
                this.state.startingPlayer = 'player';
                
                this.log('新游戏开始！第1回合', 'system');
                this.emit({ type: 'GAME_INIT' });
            }
            
            // 构建牌组
            buildDeck(faction) {
                let deck = [];
                const factionCards = CardDatabase[faction] || [];
                
                factionCards.forEach(card => {
                    if (card.type === 'UNIT' || card.type === 'SPECIAL' || card.type === 'WEATHER') {
                        deck.push({ ...card, instanceId: `${card.id}_${Math.random().toString(36).substr(2, 9)}` });
                    }
                });
                
                // 添加中立卡
                CardDatabase.monsters.filter(c => c.faction === 'NEUTRAL').forEach(card => {
                    deck.push({ ...card, instanceId: `${card.id}_${Math.random().toString(36).substr(2, 9)}` });
                });
                
                return deck;
            }
            
            // 洗牌
            shuffleDeck(player) {
                const deck = this.state.players[player].deck;
                for (let i = deck.length - 1; i > 0; i--) {
                    const j = Math.floor(Math.random() * (i + 1));
                    [deck[i], deck[j]] = [deck[j], deck[i]];
                }
            }
            
            // 抽牌
            drawCards(player, count) {
                const playerData = this.state.players[player];
                const drawn = [];
                for (let i = 0; i < count && playerData.deck.length > 0; i++) {
                    const card = playerData.deck.pop();
                    playerData.hand.push(card);
                    drawn.push(card);
                }
                return drawn;
            }
            
            // 出牌
            playCard(player, card, targetRow) {
                if (!this.canPlayCard(player)) {
                    return { success: false, reason: '无法出牌' };
                }
                
                const playerData = this.state.players[player];
                
                // 从手牌移除
                const cardIndex = playerData.hand.findIndex(c => c.instanceId === card.instanceId);
                if (cardIndex === -1) {
                    return { success: false, reason: '卡牌不在手中' };
                }
                playerData.hand.splice(cardIndex, 1);
                
                // 处理不同卡牌类型
                if (card.type === 'UNIT') {
                    this.placeUnit(player, card, targetRow);
                } else if (card.type === 'WEATHER') {
                    this.applyWeatherCard(card);
                } else if (card.type === 'SPECIAL') {
                    this.playSpecialCard(player, card, targetRow);
                }
                
                this.log(`${player === 'player' ? '玩家' : '对手'}打出 ${card.name}`, player === 'player' ? 'player-action' : 'enemy-action');
                
                this.recalculateAllPower();
                this.emit({ type: 'CARD_PLAYED', player, card, row: targetRow });
                this.switchTurn();
                
                return { success: true };
            }
            
            // 放置单位
            placeUnit(player, card, row) {
                const boardRow = this.state.board[player][row];
                const unitCard = {
                    ...card,
                    currentPower: card.power,
                    basePower: card.power,
                    originalPlayer: player
                };
                
                // 处理间谍
                if (card.abilities && card.abilities.includes('SPY')) {
                    const opponent = player === 'player' ? 'enemy' : 'player';
                    this.state.board[opponent][row].cards.push(unitCard);
                    this.drawCards(player, 2);
                    this.log(`间谍效果：抽2张牌`, 'system');
                } else {
                    boardRow.cards.push(unitCard);
                }
                
                // 处理部署能力
                this.triggerDeployAbilities(player, unitCard, row);
            }
            
            // 触发部署能力
            triggerDeployAbilities(player, card, row) {
                if (!card.abilities) return;
                
                card.abilities.forEach(ability => {
                    switch(ability) {
                        case 'MEDIC':
                            this.medicAbility(player);
                            break;
                        case 'MUSTER':
                            this.musterAbility(player, card);
                            break;
                        case 'BOND':
                            this.bondAbility(player, card, row);
                            break;
                        case 'MORALE_BOOST':
                            this.moraleBoostAbility(player, row, card);
                            break;
                    }
                });
            }
            
            // 医生能力
            medicAbility(player) {
                const playerData = this.state.players[player];
                const units = playerData.graveyard.filter(c => c.type === 'UNIT');
                if (units.length > 0) {
                    const revived = units[units.length - 1];
                    playerData.graveyard = playerData.graveyard.filter(c => c.instanceId !== revived.instanceId);
                    playerData.hand.push(revived);
                    this.log(`医生复活了 ${revived.name}`, 'system');
                }
            }
            
            // 集结能力
            musterAbility(player, card) {
                const playerData = this.state.players[player];
                const musterName = card.name.split(' ')[0];
                
                const musterCards = playerData.deck.filter(c => c.name.includes(musterName));
                musterCards.slice(0, 2).forEach(c => {
                    playerData.hand.push(c);
                    playerData.deck = playerData.deck.filter(d => d.instanceId !== c.instanceId);
                });
                
                if (musterCards.length > 0) {
                    this.log(`集结合并召唤了 ${Math.min(musterCards.length, 2)} 张卡`, 'system');
                }
            }
            
            // 紧密bond
            bondAbility(player, card, row) {
                const rowCards = this.state.board[player][row].cards;
                const sameCards = rowCards.filter(c => c.name === card.name);
                if (sameCards.length >= 2) {
                    sameCards.forEach(c => {
                        c.currentPower = c.basePower * sameCards.length;
                    });
                    this.log(`紧密bond激活：${sameCards.length}张卡战力翻倍`, 'system');
                }
            }
            
            // 士气提升
            moraleBoostAbility(player, row, sourceCard) {
                const rowCards = this.state.board[player][row].cards;
                rowCards.forEach(c => {
                    if (c.instanceId !== sourceCard.instanceId) {
                        c.currentPower = (c.currentPower || c.basePower) + 1;
                    }
                });
                this.log(`士气提升：同行其他单位+1`, 'system');
            }
            
            // 应用天气卡
            applyWeatherCard(card) {
                const weatherMap = {
                    'BITING_FROST': 'melee',
                    'IMPENETRABLE_FOG': 'ranged',
                    'TORRENTIAL_RAIN': 'siege'
                };
                
                const targetRow = weatherMap[card.abilityType];
                if (targetRow) {
                    this.state.weather[targetRow] = card.abilityType;
                    this.log(`天气 ${card.name} 已应用到 ${targetRow} 排`, 'system');
                }
            }
            
            // 播放特殊卡
            playSpecialCard(player, card, targetRow) {
                switch(card.abilityType) {
                    case 'SCORCH':
                        this.scorchAbility();
                        break;
                    case 'COMMANDERS_HORN':
                        if (targetRow) {
                            this.state.board[player][targetRow].hasHorn = true;
                            this.log(`号角应用到 ${targetRow} 排`, 'system');
                        }
                        break;
                }
            }
            
            // 灼烧能力
            scorchAbility() {
                let maxPower = 0;
                let toDestroy = [];
                
                ['player', 'enemy'].forEach(side => {
                    ['melee', 'ranged', 'siege'].forEach(rowName => {
                        this.state.board[side][rowName].cards.forEach(card => {
                            if (card.color !== 'GOLD' && card.currentPower > maxPower) {
                                maxPower = card.currentPower;
                                toDestroy = [card];
                            } else if (card.color !== 'GOLD' && card.currentPower === maxPower && maxPower > 0) {
                                toDestroy.push(card);
                            }
                        });
                    });
                });
                
                toDestroy.forEach(card => this.destroyCard(card));
                this.log(`灼烧！消灭了 ${toDestroy.length} 张战力 ${maxPower} 的卡`, 'system');
                this.recalculateAllPower();
            }
            
            // 摧毁卡牌
            destroyCard(card) {
                ['player', 'enemy'].forEach(side => {
                    ['melee', 'ranged', 'siege'].forEach(rowName => {
                        const row = this.state.board[side][rowName];
                        const index = row.cards.findIndex(c => c.instanceId === card.instanceId);
                        if (index !== -1) {
                            row.cards.splice(index, 1);
                            this.state.players[side].graveyard.push(card);
                        }
                    });
                });
            }
            
            // 清除天气
            clearWeather() {
                this.state.weather = { melee: null, ranged: null, siege: null };
                this.recalculateAllPower();
                this.log('天气已清除', 'system');
            }
            
            // 重新计算所有战力
            recalculateAllPower() {
                ['player', 'enemy'].forEach(side => {
                    let totalPower = 0;
                    
                    ['melee', 'ranged', 'siege'].forEach(rowName => {
                        const row = this.state.board[side][rowName];
                        let rowPower = 0;
                        
                        row.cards.forEach(card => {
                            let cardPower = card.currentPower || card.basePower;
                            
                            // 天气效果（金卡免疫）
                            if (this.state.weather[rowName] && card.color !== 'GOLD') {
                                cardPower = 1;
                            }
                            
                            rowPower += cardPower;
                        });
                        
                        // 号角翻倍
                        if (row.hasHorn) {
                            rowPower *= 2;
                        }
                        
                        row.power = rowPower;
                        totalPower += rowPower;
                    });
                    
                    this.state.players[side].totalPower = totalPower;
                });
                
                this.emit({ type: 'POWER_UPDATED' });
            }
            
            // 检查是否可以出牌
            canPlayCard(player) {
                if (this.state.phase !== 'ROUND_PLAY') return false;
                if (this.state.currentPlayer !== player) return false;
                if (this.state.players[player].hasPassed) return false;
                if (this.state.gameOver) return false;
                return true;
            }
            
            // 玩家放弃
            playerPass(player) {
                if (this.state.players[player].hasPassed) return false;
                
                this.state.players[player].hasPassed = true;
                this.log(`${player === 'player' ? '玩家' : '对手'}放弃本回合`, player === 'player' ? 'player-action' : 'enemy-action');
                
                this.emit({ type: 'PLAYER_PASSED', player });
                this.switchTurn();
                return true;
            }
            
            // 切换回合
            switchTurn() {
                const playerPassed = this.state.players.player.hasPassed;
                const enemyPassed = this.state.players.enemy.hasPassed;
                
                // 双方都pass，回合结束
                if (playerPassed && enemyPassed) {
                    this.endRound();
                    return;
                }
                
                // 交替
                if (this.state.currentPlayer === 'player') {
                    if (!enemyPassed) {
                        this.state.currentPlayer = 'enemy';
                    }
                } else {
                    if (!playerPassed) {
                        this.state.currentPlayer = 'player';
                    }
                }
                
                this.state.turnCount++;
                this.emit({ type: 'TURN_SWITCH', currentPlayer: this.state.currentPlayer });
                
                // AI回合
                if (this.state.currentPlayer === 'enemy' && !this.state.gameOver) {
                    setTimeout(() => this.aiTurn(), 800);
                }
            }
            
            // AI回合
            aiTurn() {
                if (this.state.gameOver || this.state.phase !== 'ROUND_PLAY') return;
                
                const enemy = this.state.players.enemy;
                
                // 简单AI：如果战力落后太多且有牌就打，否则考虑pass
                const powerDiff = this.state.players.enemy.totalPower - this.state.players.player.totalPower;
                
                if (enemy.hand.length === 0 || (powerDiff > 20 && enemy.hand.length < 3 && Math.random() > 0.3)) {
                    this.playerPass('enemy');
                    return;
                }
                
                // 打出一张牌
                const playableCards = enemy.hand.filter(c => c.type === 'UNIT');
                if (playableCards.length > 0) {
                    const card = playableCards[Math.floor(Math.random() * playableCards.length)];
                    const row = card.row ? card.row.toLowerCase() : 'melee';
                    this.playCard('enemy', card, row);
                } else {
                    this.playerPass('enemy');
                }
            }
            
            // 回合结束
            endRound() {
                this.state.phase = 'ROUND_END';
                
                const playerPower = this.state.players.player.totalPower;
                const enemyPower = this.state.players.enemy.totalPower;
                
                let result;
                if (playerPower > enemyPower) {
                    this.state.players.player.roundWins++;
                    result = 'player';
                    this.log(`玩家赢得第 ${this.state.currentRound} 回合！(${playerPower} vs ${enemyPower})`, 'system');
                } else if (enemyPower > playerPower) {
                    this.state.players.enemy.roundWins++;
                    result = 'enemy';
                    this.log(`对手赢得第 ${this.state.currentRound} 回合！(${enemyPower} vs ${playerPower})`, 'system');
                } else {
                    result = 'tie';
                    this.state.players.player.roundWins++;
                    this.state.players.enemy.roundWins++;
                    this.log(`第 ${this.state.currentRound} 回合平局！(${playerPower} vs ${enemyPower})`, 'system');
                }
                
                this.emit({ type: 'ROUND_END', result, playerPower, enemyPower });
                
                // 检查游戏结束
                if (this.state.players.player.roundWins >= 2 || this.state.players.enemy.roundWins >= 2) {
                    this.state.gameOver = true;
                    this.state.phase = 'GAME_OVER';
                    this.state.winner = this.state.players.player.roundWins >= 2 ? 'player' : 'enemy';
                    this.log(this.state.winner === 'player' ? '玩家赢得比赛！' : '对手赢得比赛！', 'system');
                    this.emit({ type: 'GAME_OVER', winner: this.state.winner });
                } else {
                    // 准备下一回合
                    setTimeout(() => this.startNextRound(result), 1500);
                }
            }
            
            // 开始下一回合
            startNextRound(lastResult) {
                this.clearBoard();
                this.state.currentRound++;
                this.state.phase = 'ROUND_PLAY';
                this.state.players.player.hasPassed = false;
                this.state.players.enemy.hasPassed = false;
                this.state.turnCount = 0;
                
                // 输方先手
                if (lastResult === 'player') {
                    this.state.startingPlayer = 'enemy';
                } else if (lastResult === 'enemy') {
                    this.state.startingPlayer = 'player';
                }
                
                this.state.currentPlayer = this.state.startingPlayer;
                
                // 抽2张牌
                this.drawCards('player', 2);
                this.drawCards('enemy', 2);
                
                this.log(`第 ${this.state.currentRound} 回合开始！`, 'system');
                this.emit({ type: 'ROUND_START', round: this.state.currentRound });
                
                if (this.state.currentPlayer === 'enemy') {
                    setTimeout(() => this.aiTurn(), 800);
                }
            }
            
            // 清空战场
            clearBoard() {
                ['player', 'enemy'].forEach(side => {
                    ['melee', 'ranged', 'siege'].forEach(rowName => {
                        const row = this.state.board[side][rowName];
                        row.cards.forEach(card => {
                            this.state.players[side].graveyard.push(card);
                        });
                        row.cards = [];
                        row.power = 0;
                        row.hasHorn = false;
                    });
                });
                
                this.state.weather = { melee: null, ranged: null, siege: null };
            }

            }

        // ==================== 粒子系统 ====================
        class Particle {
            constructor(type) {
                this.type = type;
                this.reset();
            }
            
            reset() {
                this.x = Math.random() * canvas.width;
                this.y = -20;
                this.vx = (Math.random() - 0.5) * 2;
                this.vy = Math.random() * 2 + 1;
                this.size = Math.random() * 3 + 1;
                this.alpha = Math.random() * 0.5 + 0.3;
                this.life = 1;
                this.decay = Math.random() * 0.005 + 0.002;
                
                if (this.type === 'fog') {
                    this.x = Math.random() * canvas.width;
                    this.y = Math.random() * canvas.height;
                    this.size = Math.random() * 100 + 50;
                    this.alpha = Math.random() * 0.15 + 0.05;
                    this.vx = (Math.random() - 0.5) * 0.5;
                    this.vy = (Math.random() - 0.5) * 0.2;
                } else if (this.type === 'rain') {
                    this.vx = -2;
                    this.vy = Math.random() * 15 + 10;
                    this.size = Math.random() * 2 + 1;
                    this.length = Math.random() * 20 + 10;
                    this.alpha = Math.random() * 0.4 + 0.2;
                } else if (this.type === 'snow') {
                    this.vx = (Math.random() - 0.5) * 1;
                    this.vy = Math.random() * 2 + 0.5;
                    this.size = Math.random() * 3 + 1;
                    this.alpha = Math.random() * 0.6 + 0.4;
                } else if (this.type === 'ambient') {
                    this.x = Math.random() * canvas.width;
                    this.y = Math.random() * canvas.height;
                    this.size = Math.random() * 2 + 0.5;
                    this.alpha = 0;
                    this.targetAlpha = Math.random() * 0.6 + 0.2;
                    this.pulse = Math.random() * Math.PI * 2;
                    this.color = Math.random() > 0.5 ? CONFIG.COLORS.GOLD_LIGHT : '#80a0c0';
                }
            }
            
            update() {
                if (this.type === 'fog') {
                    this.x += this.vx;
                    this.y += this.vy;
                    if (this.x < -this.size) this.x = canvas.width + this.size;
                    if (this.x > canvas.width + this.size) this.x = -this.size;
                    if (this.y < -this.size) this.y = canvas.height + this.size;
                    if (this.y > canvas.height + this.size) this.y = -this.size;
                } else if (this.type === 'ambient') {
                    this.pulse += 0.05;
                    this.alpha = (Math.sin(this.pulse) + 1) * 0.5 * this.targetAlpha;
                    this.y -= 0.2;
                    if (this.y < 0) {
                        this.y = canvas.height;
                        this.x = Math.random() * canvas.width;
                    }
                } else if (this.type === 'rain') {
                    this.x += this.vx;
                    this.y += this.vy;
                    if (this.y > canvas.height) {
                        this.reset();
                    }
                } else if (this.type === 'snow') {
                    this.x += this.vx + Math.sin(this.y * 0.02) * 0.5;
                    this.y += this.vy;
                    if (this.y > canvas.height) {
                        this.reset();
                    }
                } else {
                    this.x += this.vx;
                    this.y += this.vy;
                    this.life -= this.decay;
                    if (this.life <= 0 || this.y > canvas.height + 20) {
                        this.reset();
                    }
                }
            }
            
            draw(ctx) {
                ctx.save();
                
                if (this.type === 'fog') {
                    const gradient = ctx.createRadialGradient(
                        this.x, this.y, 0,
                        this.x, this.y, this.size
                    );
                    gradient.addColorStop(0, `rgba(50, 60, 80, ${this.alpha})`);
                    gradient.addColorStop(1, 'rgba(50, 60, 80, 0)');
                    ctx.fillStyle = gradient;
                    ctx.beginPath();
                    ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                    ctx.fill();
                } else if (this.type === 'rain') {
                    ctx.strokeStyle = `rgba(150, 170, 200, ${this.alpha})`;
                    ctx.lineWidth = 1;
                    ctx.beginPath();
                    ctx.moveTo(this.x, this.y);
                    ctx.lineTo(this.x + this.vx * 2, this.y + this.length);
                    ctx.stroke();
                } else if (this.type === 'snow') {
                    ctx.fillStyle = `rgba(200, 220, 255, ${this.alpha})`;
                    ctx.beginPath();
                    ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                    ctx.fill();
                } else if (this.type === 'ambient') {
                    ctx.globalAlpha = this.alpha;
                    ctx.fillStyle = this.color;
                    ctx.shadowBlur = 10;
                    ctx.shadowColor = this.color;
                    ctx.beginPath();
                    ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                    ctx.fill();
                }
                
                ctx.restore();
            }
        }

        function drawBoardZone(x, y, width, height, side, state) {
            const rows = ['siege', 'ranged', 'melee'];

                const rowY = y + i * rowHeight;
                const rowData = state.board[side][rowName];
                
                // 行背景
                const isPlayer = side === 'player';
                const bgColor = isPlayer ? 'rgba(20, 40, 30, 0.3)' : 'rgba(40, 20, 30, 0.3)';
                ctx.fillStyle = bgColor;
                ctx.fillRect(x, rowY, width, rowHeight);
                
                // 行边框
                ctx.strokeStyle = isPlayer ? 'rgba(74, 222, 128, 0.3)' : 'rgba(248, 113, 113, 0.3)';
                ctx.lineWidth = 1;
                ctx.strokeRect(x, rowY, width, rowHeight);
                
                // 天气效果指示
                if (state.weather[rowName]) {
                    ctx.fillStyle = 'rgba(100, 150, 200, 0.2)';
                    ctx.fillRect(x, rowY, width, rowHeight);
                }
                
                // 绘制该行的卡牌
                drawRowCards(rowData.cards, x + 10, rowY + 5, width - 20, rowHeight - 10);

                ctx.font = 'bold 18px Cinzel';
                ctx.textAlign = 'center';

            });

        function drawRowCards(cards, x, y, width, height) {

        function drawCard(card, x, y, width, height, isHand) {
            const isHovered = GameState.hoveredCard && GameState.hoveredCard.instanceId === card.instanceId;
            const isSelected = GameState.selectedCard && GameState.selectedCard.instanceId === card.instanceId;

            if (isSelected) {
                ctx.shadowColor = CONFIG.COLORS.GOLD_LIGHT;
                ctx.shadowBlur = 20;
            } else if (isHovered && isHand) {
                ctx.translate(x + width/2, y - 20);
                ctx.scale(1.1, 1.1);
                ctx.translate(-(x + width/2), -(y - 20));
            }
            
            // 卡牌背景
            let bgColor;
            if (card.color === 'GOLD') {
                bgColor = 'linear-gradient(180deg, #3a3020 0%, #2a2015 100%)';
            } else if (card.color === 'SPECIAL') {
                bgColor = 'linear-gradient(180deg, #203030 0%, #152020 100%)';
            } else {
                bgColor = 'linear-gradient(180deg, #252525 0%, #1a1a1a 100%)';
            }
            
            // 绘制渐变背景
            const gradient = ctx.createLinearGradient(x, y, x, y + height);
            if (card.color === 'GOLD') {
                gradient.addColorStop(0, '#3a3020');
                gradient.addColorStop(1, '#2a2015');
            } else if (card.color === 'SPECIAL') {
                gradient.addColorStop(0, '#203030');
                gradient.addColorStop(1, '#152020');
            } else {
                gradient.addColorStop(0, '#252525');
                gradient.addColorStop(1, '#1a1a1a');
            }
            
            ctx.fillStyle = gradient;
            ctx.fillRect(x, y, width, height);
            
            // 卡牌边框
            ctx.strokeStyle = card.color === 'GOLD' ? CONFIG.COLORS.GOLD_PRIMARY : CONFIG.COLORS.CARD_BORDER;
            ctx.lineWidth = card.color === 'GOLD' ? 2 : 1;
            ctx.strokeRect(x, y, width, height);
            
            // 战力值
            if (card.type === 'UNIT') {
                const power = card.currentPower || card.power;
                const powerColor = card.color === 'GOLD' ? CONFIG.COLORS.GOLD_LIGHT : '#d4c4a8';
                
                ctx.fillStyle = powerColor;
                ctx.font = `bold ${Math.floor(height * 0.2)}px Cinzel`;
                ctx.textAlign = 'center';
                ctx.textBaseline = 'top';
                ctx.fillText(power.toString(), x + width/2, y + 5);
            }
            
            // 卡牌名称
            ctx.fillStyle = '#a0a0a8';
            ctx.font = `${Math.floor(Math.min(width * 0.12, 10))}px Cinzel`;
            ctx.textAlign = 'center';
            ctx.textBaseline = 'bottom';
            
            const name = card.name.length > 8 ? card.name.substring(0, 7) + '...' : card.name;
            ctx.fillText(name, x + width/2, y + height - 5);

                ctx.fillStyle = CONFIG.COLORS.GOLD_PRIMARY;
                ctx.font = '10px Arial';
                ctx.fillText('★', x + width - 10, y + 10);
            }
            
            ctx.restore();
        }

            drawEnemyHand(enemyHand.length, enemyHandY);
        }
        
        function drawHandCards(cards, y, side) {
            if (cards.length === 0) return;
            
            const centerX = canvas.width / 2;
            const totalWidth = cards.length * (CONFIG.CARD_WIDTH + 10);
            const startX = centerX - totalWidth / 2;
            
            cards.forEach((card, i) => {
                const angle = (i - (cards.length - 1) / 2) * CONFIG.HAND_FAN_ANGLE * 0.1;
                const offsetX = (i - (cards.length - 1) / 2) * (CONFIG.CARD_WIDTH + 10);
                const offsetY = Math.abs(i - (cards.length - 1) / 2) * 5;
                
                const cardX = centerX + offsetX - CONFIG.CARD_WIDTH / 2;
                const cardY = y + offsetY;
                
                ctx.save();
                ctx.translate(cardX + CONFIG.CARD_WIDTH / 2, cardY + CONFIG.CARD_HEIGHT);
                ctx.rotate(angle);
                ctx.translate(-(cardX + CONFIG.CARD_WIDTH / 2), -(cardY + CONFIG.CARD_HEIGHT));
                
                drawCard(card, cardX, cardY, CONFIG.CARD_WIDTH, CONFIG.CARD_HEIGHT, true);
                
                ctx.restore();
                
                // 存储卡牌位置用于点击检测
                card._renderX = cardX;
                card._renderY = cardY;
            });
        }
        
        function drawEnemyHand(count, y) {
            const centerX = canvas.width / 2;
            const cardWidth = 50;
            const spacing = cardWidth + 5;
            const startX = centerX - (count * spacing) / 2;
            
            for (let i = 0; i < count; i++) {
                const x = startX + i * spacing;
                
                // 绘制卡背
                ctx.fillStyle = '#1a1a25';
                ctx.fillRect(x, y, cardWidth, 70);
                ctx.strokeStyle = '#3a3a45';
                ctx.lineWidth = 1;
                ctx.strokeRect(x, y, cardWidth, 70);
                
                // 卡背图案
                ctx.fillStyle = '#2a2a35';
                ctx.beginPath();
                ctx.arc(x + cardWidth/2, y + 35, 15, 0, Math.PI * 2);
                ctx.fill();
            }
        }
        
        function drawPowerDisplay(x, y, power, side) {
            const isPlayer = side === 'player';
            const color = isPlayer ? CONFIG.COLORS.PLAYER_POWER : CONFIG.COLORS.ENEMY_POWER;
            
            ctx.save();
            ctx.fillStyle = color;
            ctx.font = 'bold 36px Cinzel';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            ctx.shadowColor = color;
            ctx.shadowBlur = 15;

            ctx.restore();

        function drawWeatherEffects() {

            // 绘制天气粒子
            if (state.weather.melee === 'BITING_FROST' || 
                state.weather.ranged === 'IMPENETRABLE_FOG' ||

                if (state.weather.melee === 'BITING_FROST') {

                    GameState.particles.filter(p => p.type === 'snow').forEach(p => {
                        p.update();
                        p.draw(ctx);
                    });
                }
                
                if (state.weather.ranged === 'IMPENETRABLE_FOG') {
                    // 浓雾效果
                    GameState.fogParticles.forEach(p => {
                        p.update();
                        p.draw(ctx);
                    });
                }
                
                if (state.weather.siege === 'TORRENTIAL_RAIN') {
                    // 暴雨效果
                    GameState.rainDrops.forEach(p => {
                        p.update();
                        p.draw(ctx);
                    });
                }
            }
        }
        
        function drawParticles() {
            // 环境粒子
            GameState.particles.filter(p => p.type === 'ambient').forEach(p => {
                p.update();
                p.draw(ctx);
            });
        }
        
        // ==================== 事件处理 ====================
        function handleMouseMove(e) {
            const rect = canvas.getBoundingClientRect();
            GameState.mousePos.x = e.clientX - rect.left;
            GameState.mousePos.y = e.clientY - rect.top;
            
            // 检查悬停的卡牌
            const state = ruleEngine.getState();
            GameState.hoveredCard = null;
            
            state.players.player.hand.forEach(card => {
                if (card._renderX !== undefined) {
                    if (GameState.mousePos.x >= card._renderX && 
                        GameState.mousePos.x <= card._renderX + CONFIG.CARD_WIDTH &&
                        GameState.mousePos.y >= card._renderY && 
                        GameState.mousePos.y <= card._renderY + CONFIG.CARD_HEIGHT) {
                        GameState.hoveredCard = card;
                    }
                }
            });
        }
        
        function handleClick(e) {
            const state = ruleEngine.getState();
            
            if (state.phase !== 'ROUND_PLAY' || state.currentPlayer !== 'player') {
                return;
            }
            
            // 检查点击的手牌
            if (GameState.hoveredCard) {
                const card = GameState.hoveredCard;
                
                if (card.type === 'UNIT') {
                    // 确定放置的行
                    let targetRow = card.row ? card.row.toLowerCase() : 'melee';
                    
                    // 间谍卡特殊处理
                    if (card.abilities && card.abilities.includes('SPY')) {
                        targetRow = card.row ? card.row.toLowerCase() : 'melee';
                    }
                    
                    ruleEngine.playCard('player', card, targetRow);
                } else if (card.type === 'SPECIAL') {
                    if (card.abilityType === 'COMMANDERS_HORN') {
                        // 简单处理：应用到近战排
                        ruleEngine.playCard('player', card, 'melee');
                    } else {
                        ruleEngine.playCard('player', card, null);
                    }
                } else if (card.type === 'WEATHER') {
                    ruleEngine.playCard('player', card, null);
                }
            }
        }
        
        function handleKeyPress(e) {
            if (e.code === 'Space') {
                e.preventDefault();
                const state = ruleEngine.getState();
                if (state.phase === 'ROUND_PLAY' && state.currentPlayer === 'player') {
                    ruleEngine.playerPass('player');
                }
            }
        }
        
        // ==================== UI更新 ====================
        function updateUI() {
            const state = ruleEngine.getState();
            
            // 更新回合宝石
            for (let i = 1; i <= 2; i++) {

                const enemyGem = document.getElementById(`enemyGem${i}`);
                
                playerGem.className = 'round-gem';
                enemyGem.className = 'round-gem';
                
                if (state.players.player.roundWins >= i) {
                    playerGem.classList.add('won');
                }
                if (state.players.enemy.roundWins >= i) {
                    enemyGem.classList.add('lost');
                }
            }
            
            // 更新回合指示器
            const turnIndicator = document.getElementById('turnIndicator');
            if (state.phase === 'GAME_OVER') {
                turnIndicator.textContent = state.winner === 'player' ? '你赢了！' : '你输了！';
                turnIndicator.className = 'turn-indicator';
            } else if (state.phase === 'ROUND_PLAY') {
                turnIndicator.textContent = state.currentPlayer === 'player' ? '你的回合' : '对手回合';
                turnIndicator.className = `turn-indicator ${state.currentPlayer === 'player' ? 'player-turn' : 'enemy-turn'}`;
            } else {
                turnIndicator.textContent = `第 ${state.currentRound} 回合`;
                turnIndicator.className = 'turn-indicator';
            }
            
            // 更新日志
            const logContainer = document.getElementById('gameLog');
            const recentLogs = state.eventLog.slice(-10);
            logContainer.innerHTML = recentLogs.map(entry => 
                `<div class="log-entry ${entry.type}">${entry.message}</div>`
            ).join('');
            logContainer.scrollTop = logContainer.scrollHeight;
        }
        
        // ==================== 初始化 ====================
        function init() {
            initParticles();
            
            // 初始化规则引擎订阅
            ruleEngine.subscribe((event, state) => {
                updateUI();
            });
            
            // 开始新游戏
            ruleEngine.initNewGame();
            updateUI();
            
            // 事件监听
            canvas.addEventListener('mousemove', handleMouseMove);
            canvas.addEventListener('click', handleClick);
            document.addEventListener('keypress', handleKeyPress);
            
            // UI按钮
            document.getElementById('btnNewGame').addEventListener('click', () => {
                ruleEngine.initNewGame();
                updateUI();
            });
            
            document.getElementById('btnPass').addEventListener('click', () => {
                const state = ruleEngine.getState();
                if (state.phase === 'ROUND_PLAY' && state.currentPlayer === 'player') {
                    ruleEngine.playerPass('player');
                }
            });
            
            document.getElementById('btnScorch').addEventListener('click', () => {
                const state = ruleEngine.getState();
                if (state.phase === 'ROUND_PLAY' && state.currentPlayer === 'player') {
                    ruleEngine.scorchAbility();
                }
            });
            
            // 天气按钮
            document.querySelectorAll('.weather-btn').forEach(btn => {
                btn.addEventListener('click', () => {
                    const weather = btn.dataset.weather;
                    
                    document.querySelectorAll('.weather-btn').forEach(b => b.classList.remove('active'));
                    btn.classList.add('active');
                    
                    ruleEngine.setWeather(weather);
                });
            });
            
            // 开始游戏循环
            gameLoop();
        }
        
        function gameLoop() {
            GameState.time++;
            
            // 清空画布
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            // 绘制背景
            drawBackground();
            
            // 绘制粒子
            drawParticles();
            
            // 绘制战场
            drawBoard();
            
            // 绘制天气效果
            drawWeatherEffects();
            
            requestAnimationFrame(gameLoop);
        }
        
        // 启动
        init();
        
    })();
    </script>
</body>
</html>

## 核心规则引擎实现说明

我为你实现了一个完整的昆特牌核心规则引擎 `GwentRuleEngine` 类，包含以下功能：

### 1. 三局两胜赛制
- 跟踪双方 `roundWins`，任意一方达到2胜时触发 `GAME_OVER`
- 回合间自动清空战场、抽2张牌

### 2. 先手后手轮换
- 每回合开始时，输方获得先手权
- 通过 `startingPlayer` 和 `currentPlayer` 管理

### 3. 每回合出牌机制
- `canPlayCard()` 验证出牌合法性
- `playerPass()` 处理放弃操作
- 双方都pass后自动进入回合结算

### 4. 战力实时计算
- `recalculateAllPower()` 计算每排和总战力
- 考虑天气效果（铜卡战力变1）、号角翻倍

### 5. 特殊能力系统
- **间谍(SPYY)**：放到对方战场，抽2张牌
- **医生(MEDIC)**：复活墓场一张单位
- **集结(MUSTER)**：从牌组召唤同名卡
- **紧密bond(BOND)**：同名卡在场时战力翻倍
- **士气提升(MORALE_BOOST)**：同行其他单位+1
- **灼烧(SCORCH)**：消灭场上战力最高的铜卡
- **号角(COMMANDERS_HORN)**：该排战力翻倍
- **天气卡**：对相应排造成削弱效果
