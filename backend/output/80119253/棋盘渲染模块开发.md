# 棋盘渲染模块开发

我将为您开发一个基于HTML5 Canvas的五子棋棋盘渲染模块，包含网格线、坐标标记和棋子渲染功能。

## 模块设计

```html
<!DOCTYPE html>
<html lang="zh-CN">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>五子棋游戏</title>
    <style>
        body {
            margin: 0;
            padding: 20px;
            display: flex;
            flex-direction: column;
            align-items: center;
            background-color: #f5f5f5;
            font-family: 'Arial', sans-serif;
        }
        
        h1 {
            color: #333;
            margin-bottom: 20px;
        }
        
        #gameContainer {
            background-color: #e8c88f;
            padding: 20px;
            border-radius: 8px;
            box-shadow: 0 4px 8px rgba(0, 0, 0, 0.2);
        }
        
        #canvas {
            display: block;
            cursor: pointer;
        }
        
        .controls {
            margin-top: 20px;
            display: flex;
            gap: 10px;
        }
        
        button {
            padding: 8px 16px;
            background-color: #4CAF50;
            color: white;
            border: none;
            border-radius: 4px;
            cursor: pointer;
            font-size: 16px;
            transition: background-color 0.3s;
        }
        
        button:hover {
            background-color: #45a049;
        }
        
        .status {
            margin-top: 15px;
            font-size: 18px;
            color: #333;
        }
    </style>
</head>
<body>
    <h1>五子棋游戏</h1>
    <div id="gameContainer">
        <canvas id="canvas" width="600" height="600"></canvas>
        <div class="controls">
            <button id="restartBtn">重新开始</button>
            <button id="undoBtn">悔棋</button>
        </div>
        <div class="status" id="status">黑方先行</div>
    </div>

    <script>
        // 棋盘渲染模块
        class BoardRenderer {
            constructor(canvasId, gridSize = 15, cellSize = 40) {
                this.canvas = document.getElementById(canvasId);
                this.ctx = this.canvas.getContext('2d');
                this.gridSize = gridSize; // 棋盘大小 (15x15)
                this.cellSize = cellSize; // 每个格子的大小
                this.margin = 20; // 边距
                this.board = Array(gridSize).fill(null).map(() => Array(gridSize).fill(0));
                this.history = []; // 存储落子历史
                this.currentPlayer = 1; // 1代表黑棋，2代表白棋
                this.lastMove = null; // 记录最后一步
                
                // 计算实际画布大小
                this.canvas.width = this.margin * 2 + (gridSize - 1) * cellSize;
                this.canvas.height = this.margin * 2 + (gridSize - 1) * cellSize;
                
                // 初始化事件监听
                this.initEventListeners();
            }
            
            // 初始化事件监听
            initEventListeners() {
                this.canvas.addEventListener('click', (e) => this.handleClick(e));
                document.getElementById('restartBtn').addEventListener('click', () => this.restart());
                document.getElementById('undoBtn').addEventListener('click', () => this.undo());
            }
            
            // 绘制棋盘
            drawBoard() {
                // 清空画布
                this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);
                
                // 绘制棋盘背景
                this.ctx.fillStyle = '#e8c88f';
                this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);
                
                // 绘制网格线
                this.drawGridLines();
                
                // 绘制坐标标记
                this.drawCoordinates();
                
                // 绘制星位（天元和四个角的星位）
                this.drawStarPoints();
                
                // 绘制所有棋子
                this.drawPieces();
            }
            
            // 绘制网格线
            drawGridLines() {
                this.ctx.strokeStyle = '#000';
                this.ctx.lineWidth = 1;
                
                // 绘制横线
                for (let i = 0; i < this.gridSize; i++) {
                    const y = this.margin + i * this.cellSize;
                    this.ctx.beginPath();
                    this.ctx.moveTo(this.margin, y);
                    this.ctx.lineTo(this.canvas.width - this.margin, y);
                    this.ctx.stroke();
                }
                
                // 绘制竖线
                for (let i = 0; i < this.gridSize; i++) {
                    const x = this.margin + i * this.cellSize;
                    this.ctx.beginPath();
                    this.ctx.moveTo(x, this.margin);
                    this.ctx.lineTo(x, this.canvas.height - this.margin);
                    this.ctx.stroke();
                }
            }
            
            // 绘制坐标标记
            drawCoordinates() {
                this.ctx.fillStyle = '#000';
                this.ctx.font = '14px Arial';
                this.textAlign = 'center';
                this.textBaseline = 'middle';
                
                // 绘制列标记 (A-O)
                for (let i = 0; i < this.gridSize; i++) {
                    const letter = String.fromCharCode(65 + i); // A, B, C, ...
                    const x = this.margin + i * this.cellSize;
                    this.ctx.fillText(letter, x, this.margin - 10);
                    this.ctx.fillText(letter, x, this.canvas.height - this.margin + 10);
                }
                
                // 绘制行标记 (1-15)
                for (let i = 0; i < this.gridSize; i++) {
                    const y = this.margin + i * this.cellSize;
                    this.ctx.fillText((i + 1).toString(), this.margin - 10, y);
                    this.ctx.fillText((i + 1).toString(), this.canvas.width - this.margin + 10, y);
                }
            }
            
            // 绘制星位
            drawStarPoints() {
                this.ctx.fillStyle = '#000';
                const starPoints = [];
                
                // 标准五子棋的星位位置
                if (this.gridSize === 15) {
                    starPoints.push([3, 3], [3, 11], [7, 7], [11, 3], [11, 11]);
                }
                
                // 绘制星位
                starPoints.forEach(([row, col]) => {
                    const x = this.margin + col * this.cellSize;
                    const y = this.margin + row * this.cellSize;
                    this.ctx.beginPath();
                    this.ctx.arc(x, y, 4, 0, Math.PI * 2);
                    this.ctx.fill();
                });
            }
            
            // 绘制棋子
            drawPieces() {
                for (let row = 0; row < this.gridSize; row++) {
                    for (let col = 0; col < this.gridSize; col++) {
                        if (this.board[row][col] !== 0) {
                            this.drawPiece(row, col, this.board[row][col]);
                        }
                    }
                }
                
                // 高亮显示最后一步
                if (this.lastMove) {
                    const [row, col] = this.lastMove;
                    const x = this.margin + col * this.cellSize;
                    const y = this.margin + row * this.cellSize;
                    
                    this.ctx.strokeStyle = '#ff0000';
                    this.ctx.lineWidth = 2;
                    this.ctx.beginPath();
                    this.ctx.arc(x, y, this.cellSize / 2 - 2, 0, Math.PI * 2);
                    this.ctx.stroke();
                }
            }
            
            // 绘制单个棋子
            drawPiece(row, col, player) {
                const x = this.margin + col * this.cellSize;
                const y = this.margin + row * this.cellSize;
                const radius = this.cellSize / 2 - 2;
                
                // 绘制阴影
                this.ctx.fillStyle = 'rgba(0, 0, 0, 0.3)';
                this.ctx.beginPath();
                this.ctx.arc(x + 2, y + 2, radius, 0, Math.PI * 2);
                this.ctx.fill();
                
                // 绘制棋子
                if (player === 1) {
                    // 黑棋
                    this.ctx.fillStyle = '#000';
                } else {
                    // 白棋
                    this.ctx.fillStyle = '#fff';
                }
                
                this.ctx.beginPath();
                this.ctx.arc(x, y, radius, 0, Math.PI * 2);
                this.ctx.fill();
                
                // 白棋添加边框
                if (player === 2) {
                    this.ctx.strokeStyle = '#000';
                    this.ctx.lineWidth = 1;
                    this.ctx.stroke();
                }
            }
            
            // 处理点击事件
            handleClick(event) {
                const rect = this.canvas.getBoundingClientRect();
                const x = event.clientX - rect.left;
                const y = event.clientY - rect.top;
                
                // 计算点击的格子位置
                const col = Math.round((x - this.margin) / this.cellSize);
                const row = Math.round((y - this.margin) / this.cellSize);
                
                // 检查是否在有效范围内
                if (row >= 0 && row < this.gridSize && col >= 0 && col < this.gridSize) {
                    // 检查该位置是否已经有棋子
                    if (this.board[row][col] === 0) {
                        // 放置棋子
                        this.board[row][col] = this.currentPlayer;
                        this.lastMove = [row, col];
                        this.history.push({row, col, player: this.currentPlayer});
                        
                        // 切换玩家
                        this.currentPlayer = this.currentPlayer === 1 ? 2 : 1;
                        this.updateStatus();
                        
                        // 重新绘制棋盘
                        this.drawBoard();
                    }
                }
            }
            
            // 更新状态显示
            updateStatus() {
                const statusElement = document.getElementById('status');
                if (this.currentPlayer === 1) {
                    statusElement.textContent = '黑方回合';
                } else {
                    statusElement.textContent = '白方回合';
                }
            }
            
            // 重新开始游戏
            restart() {
                this.board = Array(this.gridSize).fill(null).map(() => Array(this.gridSize).fill(0));
                this.history = [];
                this.currentPlayer = 1;
                this.lastMove = null;
                this.updateStatus();
                this.drawBoard();
            }
            
            // 悔棋
            undo() {
                if (this.history.length > 0) {
                    const lastMove = this.history.pop();
                    this.board[lastMove.row][lastMove.col] = 0;
                    this.currentPlayer = lastMove.player;
                    this.lastMove = this.history.length > 0 ? 
                        [this.history[this.history.length - 1].row, this.history[this.history.length - 1].col] : 
                        null;
                    this.updateStatus();
                    this.drawBoard();
                }
            }
        }
        
        // 初始化棋盘
        document.addEventListener('DOMContentLoaded', () => {
            const board = new BoardRenderer('canvas');
            board.drawBoard();
        });
    </script>
</body>
</html>
```

## 功能说明

这个棋盘渲染模块实现了以下功能：

1. **棋盘绘制**：
   - 使用Canvas绘制15×15的标准五子棋棋盘
   - 添加了坐标标记（A-O列，1-15行）
   - 标记了星位（天元和四个角的星位）

2. **棋子渲染**：
   - 黑棋和白棋的渲染，带有阴影效果
   - 最后一步棋子用红色圆圈标记
   - 棋子大小根据格子大小自动调整

3. **交互功能**：
   - 点击棋盘放置棋子
   - 重新开始按钮
   - 悔棋功能

4. **状态管理**：
   - 记录游戏历史
   - 跟踪当前玩家
   - 显示当前回合状态

## 技术亮点

1. **模块化设计**：使用ES6类封装棋盘渲染逻辑，便于维护和扩展
2. **响应式布局**：棋盘大小根据格子数和格子大小自动计算
3. **视觉效果**：添加了阴影、高亮等视觉效果，提升用户体验
4. **事件处理**：精确处理鼠标点击，将坐标转换为棋盘位置
