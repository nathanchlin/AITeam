        // 音效管理
        class SoundManager {
            constructor() {
                this.enabled = true;
                this.sounds = {
                    place: this.createSound(300, 0.1, 'square'),
                    win: this.createSound(523.25, 0.2, 'sine'),
                    invalid: this.createSound(200, 0.1, 'sawtooth')
                };
            }
            
            createSound(frequency, duration, type = 'sine') {
                return () => {
                    if (!this.enabled) return;
                    
                    const audioContext = new (window.AudioContext || window.webkitAudioContext)();
                    const oscillator = audioContext.createOscillator();
                    const gainNode = audioContext.createGain();
                    
                    oscillator.connect(gainNode);
                    gainNode.connect(audioContext.destination);
                    
                    oscillator.type = type;
                    oscillator.frequency.value = frequency;
                    gainNode.gain.value = 0.1;
                    
                    oscillator.start();
                    oscillator.stop(audioContext.currentTime + duration);
                };
            }
            
            play(soundName) {
                if (this.sounds[soundName]) {
                    this.sounds[soundName]();
                }
            }
            
            toggle() {
                this.enabled = !this.enabled;
            }
        }

        // 游戏状态管理
        class GameState {
            constructor() {
                this.board = Array(15).fill(null).map(() => Array(15).fill(null));
                this.currentPlayer = 'black';
                this.gameOver = false;
                this.history = [];
                this.winner = null;
                this.winningLine = [];
            }
            
            reset() {
                this.board = Array(15).fill(null).map(() => Array(15).fill(null));
                this.currentPlayer = 'black';
                this.gameOver = false;
                this.history = [];
                this.winner = null;
                this.winningLine = [];
            }
            
            placePiece(row, col) {
                if (this.gameOver || this.board[row][col] !== null) {
                    return false;
                }
                
                this.board[row][col] = this.currentPlayer;
                this.history.push({
                    row,
                    col,
                    player: this.currentPlayer
                });
                
                // 检查是否获胜
                if (this.checkWin(row, col)) {
                    this.gameOver = true;
                    this.winner = this.currentPlayer;
                } else {
                    // 切换玩家
                    this.currentPlayer = this.currentPlayer === 'black' ? 'white' : 'black';
                }
                
                return true;
            }
            
            undo() {
                if (this.history.length === 0 || this.gameOver) {
                    return false;
                }
                
                const lastMove = this.history.pop();
                this.board[lastMove.row][lastMove.col] = null;
                this.currentPlayer = lastMove.player;
                
                // 如果之前有获胜者，清除获胜状态
                if (this.winner) {
                    this.gameOver = false;
                    this.winner = null;
                    this.winningLine = [];
                }
                
                return true;
            }
            
            checkWin(row, col) {
                const player = this.board[row][col];
                if (!player) return false;
                
                // 检查方向: 水平、垂直、对角线1、对角线2
                const directions = [
                    [[0, 1], [0, -1]],  // 水平
                    [[1, 0], [-1, 0]],  // 垂直
                    [[1, 1], [-1, -1]], // 对角线1
                    [[1, -1], [-1, 1]]  // 对角线2
                ];
                
                for (const direction of directions) {
                    const line = [[row, col]];
                    
                    // 检查两个方向
                    for (const [dr, dc] of direction) {
                        let r = row + dr;
                        let c = col + dc;
                        
                        while (
                            r >= 0 && r < 15 && 
                            c >= 0 && c < 15 && 
                            this.board[r][c] === player
                        ) {
                            line.push([r, c]);
                            r += dr;
                            c += dc;
                        }
                    }
                    
                    // 如果连成5个或以上，则获胜
                    if (line.length >= 5) {
                        this.winningLine = line;
                        return true;
                    }
                }
                
                return false;
            }
        }

        // 游戏渲染器
        class BoardRenderer {
            constructor(gameState, soundManager) {
                this.gameState = gameState;
                this.soundManager = soundManager;
                this.boardElement = document.getElementById('game-board');
                this.statusElement = document.getElementById('game-status');
                this.playerIndicator = document.getElementById('player-indicator');
                this.historyList = document.getElementById('history-list');
                this.winnerMessage = document.getElementById('winner-message');
                this.overlay = document.getElementById('overlay');
                this.winnerText = document.getElementById('winner-text');
                
                this.init();
            }
            
            init() {
                // 创建棋盘格子
                for (let row = 0; row < 15; row++) {
                    for (let col = 0; col < 15; col++) {
                        const cell = document.createElement('div');
                        cell.className = 'cell';
                        cell.dataset.row = row;
                        cell.dataset.col = col;
                        
                        cell.addEventListener('click', () => this.handleCellClick(row, col));
                        
                        this.boardElement.appendChild(cell);
                    }
                }
                
                // 绑定按钮事件
                document.getElementById('restart-btn').addEventListener('click', () => this.restart());
                document.getElementById('undo-btn').addEventListener('click', () => this.undo());
                document.getElementById('winner-restart-btn').addEventListener('click', () => this.restart());
                document.getElementById('sound-toggle').addEventListener('change', () => this.toggleSound());
                
                // 初始渲染
                this.render();
            }
            
            handleCellClick(row, col) {
                if (this.gameState.gameOver) {
                    this.soundManager.play('invalid');
                    return;
                }
                
                if (this.gameState.placePiece(row, col)) {
                    this.soundManager.play('place');
                    this.render();
                } else {
                    this.soundManager.play('invalid');
                }
            }
            
            render() {
                // 清空棋盘
                this.boardElement.innerHTML = '';
                
                // 重新创建棋盘格子
                for (let row = 0; row < 15; row++) {
                    for (let col = 0; col < 15; col++) {
                        const cell = document.createElement('div');
                        cell.className = 'cell';
                        cell.dataset.row = row;
                        cell.dataset.col = col;
                        
                        // 添加棋子
                        if (this.gameState.board[row][col]) {
                            const piece = document.createElement('div');
                            piece.className = `piece ${this.gameState.board[row][col]}`;
                            cell.appendChild(piece);
                        }
                        
                        // 标记获胜线
                        if (this.gameState.winningLine.some(([r, c]) => r === row && c === col)) {
                            cell.style.backgroundColor = 'rgba(255, 215, 0, 0.5)';
                        }
                        
                        cell.addEventListener('click', () => this.handleCellClick(row, col));
                        
                        this.boardElement.appendChild(cell);
                    }
                }
                
                // 更新状态显示
                if (this.gameState.gameOver) {
                    this.statusElement.textContent = `游戏结束 - ${this.gameState.winner === 'black' ? '黑棋' : '白棋'}获胜!`;
                    this.soundManager.play('win');
                    this.showWinnerMessage();
                } else {
                    this.statusElement.textContent = '游戏进行中';
                }
                
                // 更新当前玩家指示器
                this.playerIndicator.className = `player-indicator ${this.gameState.currentPlayer}`;
                
                // 更新悔棋按钮状态
                document.getElementById('undo-btn').disabled = this.gameState.history.length === 0 || this.gameState.gameOver;
                
                // 更新历史记录
                this.updateHistory();
            }
            
            updateHistory() {
                this.historyList.innerHTML = '';
                
                this.gameState.history.forEach((move, index) => {
                    const historyItem = document.createElement('div');
                    historyItem.className = 'history-item';
                    
                    const moveText = document.createElement('span');
                    moveText.textContent = `第${index + 1}步: ${move.player === 'black' ? '黑棋' : '白棋'} (${move.row + 1}, ${String.fromCharCode(65 + move.col)})`;
                    
                    const timestamp = document.createElement('span');
                    timestamp.textContent = new Date().toLocaleTimeString();
                    
                    historyItem.appendChild(moveText);
                    historyItem.appendChild(timestamp);
                    this.historyList.appendChild(historyItem);
                });
            }
            
            showWinnerMessage() {
                this.overlay.style.display = 'block';
                this.winnerMessage.style.display = 'block';
                this.winnerText.textContent = `${this.gameState.winner === 'black' ? '黑棋' : '白棋'}获胜!`;
            }
            
            hideWinnerMessage() {
                this.overlay.style.display = 'none';
                this.winnerMessage.style.display = 'none';
            }
            
            restart() {
                this.gameState.reset();
                this.hideWinnerMessage();
                this.render();
            }
            
            undo() {
                if (this.gameState.undo()) {
                    this.render();
                }
            }
            
            toggleSound() {
                this.soundManager.toggle();
            }
        }

        // 初始化游戏
        document.addEventListener('DOMContentLoaded', () => {
            const gameState = new GameState();
            const soundManager = new SoundManager();
            const renderer = new BoardRenderer(gameState, soundManager);
        });
    </script>
</body>
</html>
