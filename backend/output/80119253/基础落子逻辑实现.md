```html
    <script>
        // 游戏配置
        const BOARD_SIZE = 15; // 棋盘大小 15x15
        const CELL_SIZE = 30; // 每个格子的大小
        const PIECE_RADIUS = 14; // 棋子半径
        const WIN_COUNT = 5; // 连成5子获胜
        
        // 游戏状态
        const GameState = {
            PLAYING: 'playing',
            BLACK_WIN: 'black_win',
            WHITE_WIN: 'white_win',
            DRAW: 'draw'
        };
        
        // 游戏类
        class GomokuGame {
            constructor() {
                this.board = Array(BOARD_SIZE).fill(null).map(() => Array(BOARD_SIZE).fill(0));
                this.currentPlayer = 1; // 1表示黑棋，2表示白棋
                this.gameState = GameState.PLAYING;
                this.history = []; // 历史记录，用于悔棋
                this.init();
            }
            
            init() {
                // 初始化棋盘
                this.boardElement = document.getElementById('board');
                this.boardElement.style.width = `${BOARD_SIZE * CELL_SIZE}px`;
                this.boardElement.style.height = `${BOARD_SIZE * CELL_SIZE}px`;
                
                // 绘制网格
                this.drawGrid();
                
                // 添加点击事件
                this.boardElement.addEventListener('click', (e) => this.handleBoardClick(e));
                
                // 添加按钮事件
                document.getElementById('undo-btn').addEventListener('click', () => this.undo());
                document.getElementById('restart-btn').addEventListener('click', () => this.restart());
                
                // 更新UI
                this.updateUI();
            }
            
            drawGrid() {
                const gridCanvas = document.getElementById('grid');
                const ctx = gridCanvas.getContext('2d');
                
                // 设置画布大小
                gridCanvas.width = BOARD_SIZE * CELL_SIZE;
                gridCanvas.height = BOARD_SIZE * CELL_SIZE;
                
                // 绘制网格线
                ctx.strokeStyle = '#8d6e63';
                ctx.lineWidth = 1;
                
                // 绘制横线
                for (let i = 0; i < BOARD_SIZE; i++) {
                    ctx.beginPath();
                    ctx.moveTo(CELL_SIZE / 2, i * CELL_SIZE + CELL_SIZE / 2);
                    ctx.lineTo(BOARD_SIZE * CELL_SIZE - CELL_SIZE / 2, i * CELL_SIZE + CELL_SIZE / 2);
                    ctx.stroke();
                }
                
                // 绘制竖线
                for (let i = 0; i < BOARD_SIZE; i++) {
                    ctx.beginPath();
                    ctx.moveTo(i * CELL_SIZE + CELL_SIZE / 2, CELL_SIZE / 2);
                    ctx.lineTo(i * CELL_SIZE + CELL_SIZE / 2, BOARD_SIZE * CELL_SIZE - CELL_SIZE / 2);
                    ctx.stroke();
                }
                
                // 绘制星位点
                const starPoints = [
                    [3, 3], [3, 11], [11, 3], [11, 11], [7, 7]
                ];
                
                ctx.fillStyle = '#8d6e63';
                starPoints.forEach(([x, y]) => {
                    ctx.beginPath();
                    ctx.arc(
                        x * CELL_SIZE + CELL_SIZE / 2, 
                        y * CELL_SIZE + CELL_SIZE / 2, 
                        3, 
                        0, 
                        Math.PI * 2
                    );
                    ctx.fill();
                });
            }
            
            handleBoardClick(e) {
                if (this.gameState !== GameState.PLAYING) {
                    this.showMessage('游戏已结束，请点击"重新开始"开始新游戏', 'error');
                    return;
                }
                
                // 计算点击位置对应的棋盘坐标
                const rect = this.boardElement.getBoundingClientRect();
                const x = e.clientX - rect.left;
                const y = e.clientY - rect.top;
                
                const col = Math.floor(x / CELL_SIZE);
                const row = Math.floor(y / CELL_SIZE);
                
                // 检查坐标是否有效
                if (row < 0 || row >= BOARD_SIZE || col < 0 || col >= BOARD_SIZE) {
                    this.showMessage('点击位置超出棋盘范围', 'error');
                    return;
                }
                
                // 尝试落子
                this.placePiece(row, col);
            }
            
            placePiece(row, col) {
                // 检查该位置是否已有棋子
                if (this.board[row][col] !== 0) {
                    this.showMessage('该位置已有棋子，请选择其他位置', 'error');
                    return;
                }
                
                // 记录历史
                this.history.push({
                    row,
                    col,
                    player: this.currentPlayer,
                    boardState: this.board.map(row => [...row])
                });
                
                // 放置棋子
                this.board[row][col] = this.currentPlayer;
                
                // 渲染棋子
                this.renderPiece(row, col, this.currentPlayer);
                
                // 检查是否获胜
                if (this.checkWin(row, col)) {
                    this.gameState = this.currentPlayer === 1 ? GameState.BLACK_WIN : GameState.WHITE_WIN;
                    const winner = this.currentPlayer === 1 ? '黑棋' : '白棋';
                    this.showMessage(`恭喜！${winner}获胜！`, 'win');
                    
                    // 禁用悔棋按钮
                    document.getElementById('undo-btn').disabled = true;
                    return;
                }
                
                // 检查是否平局
                if (this.checkDraw()) {
                    this.gameState = GameState.DRAW;
                    this.showMessage('游戏平局！', 'win');
                    
                    // 禁用悔棋按钮
                    document.getElementById('undo-btn').disabled = true;
                    return;
                }
                
                // 切换玩家
                this.currentPlayer = this.currentPlayer === 1 ? 2 : 1;
                
                // 更新UI
                this.updateUI();
            }
            
            renderPiece(row, col, player) {
                const piece = document.createElement('div');
                piece.className = `piece ${player === 1 ? 'black' : 'white'}`;
                piece.style.left = `${col * CELL_SIZE + CELL_SIZE / 2}px`;
                piece.style.top = `${row * CELL_SIZE + CELL_SIZE / 2}px`;
                
                this.boardElement.appendChild(piece);
            }
            
            checkWin(row, col) {
                const player = this.board[row][col];
                
                // 检查四个方向：横向、纵向、左上到右下对角线、右上到左下对角线
                const directions = [
                    [[0, 1], [0, -1]], // 横向
                    [[1, 0], [-1, 0]], // 纵向
                    [[1, 1], [-1, -1]], // 左上到右下对角线
                    [[1, -1], [-1, 1]] // 右上到左下对角线
                ];
                
                for (const direction of directions) {
                    let count = 1; // 包括当前位置
                    
                    // 检查正方向
                    for (let i = 1; i < WIN_COUNT; i++) {
                        const newRow = row + direction[0][0] * i;
                        const newCol = col + direction[0][1] * i;
                        
                        if (newRow < 0 || newRow >= BOARD_SIZE || 
                            newCol < 0 || newCol >= BOARD_SIZE || 
                            this.board[newRow][newCol] !== player) {
                            break;
                        }
                        
                        count++;
                    }
                    
                    // 检查反方向
                    for (let i = 1; i < WIN_COUNT; i++) {
                        const newRow = row + direction[1][0] * i;
                        const newCol = col + direction[1][1] * i;
                        
                        if (newRow < 0 || newRow >= BOARD_SIZE || 
                            newCol < 0 || newCol >= BOARD_SIZE || 
                            this.board[newRow][newCol] !== player) {
                            break;
                        }
                        
                        count++;
                    }
                    
                    // 如果连成足够数量的棋子，则获胜
                    if (count >= WIN_COUNT) {
                        return true;
                    }
                }
                
                return false;
            }
            
            checkDraw() {
                // 检查是否还有空位
                for (let row = 0; row < BOARD_SIZE; row++) {
                    for (let col = 0; col < BOARD_SIZE; col++) {
                        if (this.board[row][col] === 0) {
                            return false;
                        }
                    }
                }
                return true;
            }
            
            undo() {
                if (this.history.length === 0) {
                    this.showMessage('没有可撤销的步骤', 'error');
                    return;
                }
                
                // 获取上一步的状态
                const lastMove = this.history.pop();
                
                // 恢复棋盘状态
                this.board = lastMove.boardState;
                
                // 移除最后一步的棋子
                const pieces = this.boardElement.querySelectorAll('.piece');
                if (pieces.length > 0) {
                    pieces[pieces.length - 1].remove();
                }
                
                // 恢复玩家
                this.currentPlayer = lastMove.player;
                
                // 更新游戏状态
                if (this.gameState !== GameState.PLAYING) {
                    this.gameState = GameState.PLAYING;
                    document.getElementById('undo-btn').disabled = false;
                }
                
                // 更新UI
                this.updateUI();
            }
            
            restart() {
                // 清空棋盘
                this.board = Array(BOARD_SIZE).fill(null).map(() => Array(BOARD_SIZE).fill(0));
                this.currentPlayer = 1;
                this.gameState = GameState.PLAYING;
                this.history = [];
                
                // 移除所有棋子
                const pieces = this.boardElement.querySelectorAll('.piece');
                pieces.forEach(piece => piece.remove());
                
                // 启用悔棋按钮
                document.getElementById('undo-btn').disabled = false;
                
                // 更新UI
                this.updateUI();
                this.showMessage('');
            }
            
            updateUI() {
                const currentPlayerText = this.currentPlayer === 1 ? '黑棋' : '白棋';
                const currentPlayerIndicator = this.currentPlayer === 1 ? 'black' : 'white';
                
                document.getElementById('current-player').textContent = currentPlayerText;
                document.getElementById('current-player-indicator').className = `player-indicator ${currentPlayerIndicator}`;
            }
            
            showMessage(text, type = '') {
                const messageElement = document.getElementById('message');
                messageElement.textContent = text;
                
                // 移除所有消息类型类
                messageElement.classList.remove('win-message', 'error-message');
                
                // 添加新的消息类型类
                if (type === 'win') {
                    messageElement.classList.add('win-message');
                } else if (type === 'error') {
                    messageElement.classList.add('error-message');
                }
            }
        }
        
        // 初始化游戏
        const game = new GomokuGame();
    </script>
</body>
</html>
```

## 功能实现说明

### 1. 基础落子逻辑

- **落子验证**：检查点击位置是否在棋盘范围内，以及该位置是否已有棋子
- **落子渲染**：在棋盘上渲染对应的黑棋或白棋
- **交替落子**：每次落子后自动切换当前玩家

### 2. 胜负判断逻辑

- **检查方向**：横向、纵向、两个对角线方向
- **连子计数**：从当前棋子位置向两个方向延伸，计算相同颜色棋子的数量
- **获胜条件**：任意方向连成5个相同颜色的棋子即获胜

### 3. 悔棋功能

- **历史记录**：每次落子前记录当前棋盘状态
- **状态恢复**：撤销时恢复上一步的棋盘状态和玩家
- **棋子移除**：从UI中移除最后一步的棋子

### 4. 重新开始功能

- **状态重置**：重置棋盘状态、当前玩家、游戏历史
- **UI重置**：清除棋盘上的所有棋子，重置游戏状态显示

### 5. 错误处理和UI反馈

- **错误提示**：对无效点击、重复落子等错误情况提供明确的提示
- **游戏状态提示**：显示当前玩家、游戏结果等信息
- **按钮状态管理**：游戏结束时禁用悔棋按钮

### 6. 游戏状态管理

- **游戏状态枚举**：定义了游戏中可能的状态（进行中、黑棋胜、白棋胜、平局）
- **状态转换**：根据游戏进展更新游戏状态
