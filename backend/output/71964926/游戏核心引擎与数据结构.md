        class Tetromino {
            constructor(type) {
                this.type = type;
                this.shape = TETROMINOES[type].shape;
                this.color = TETROMINOES[type].color;
                this.rotation = 0;
                this.x = Math.floor(COLS / 2) - Math.floor(this.shape[0][0].length / 2);
                this.y = 0;
            }

            getCurrentShape() {
                return this.shape[this.rotation];
            }

            rotate(direction = 1) {
                const oldRotation = this.rotation;
                if (direction === 1) {
                    this.rotation = (this.rotation + 1) % 4;
                } else {
                    this.rotation = (this.rotation + 3) % 4;
                }
                return oldRotation;
            }

            clone() {
                const cloned = new Tetromino(this.type);
                cloned.rotation = this.rotation;
                cloned.x = this.x;
                cloned.y = this.y;
                return cloned;
            }
        }

            init() {

                this.nextPiece = this.getNextFromBag();
                this.spawnPiece();
                this.updateStats();
            }

            // 7-bag随机系统实现
            getNextFromBag() {
                if (this.bag.length === 0) {
                    this.bag = ['I', 'O', 'T', 'S', 'Z', 'J', 'L'];
                    // Fisher-Yates洗牌
                    for (let i = this.bag.length - 1; i > 0; i--) {
                        const j = Math.floor(Math.random() * (i + 1));
                        [this.bag[i], this.bag[j]] = [this.bag[j], this.bag[i]];
                    }
                }
                return new Tetromino(this.bag.pop());
            }

            spawnPiece() {
                this.currentPiece = this.nextPiece;
                this.nextPiece = this.getNextFromBag();
                this.canHold = true;
                this.isLocking = false;

                // 重置位置
                this.currentPiece.x = Math.floor(COLS / 2) - Math.floor(this.currentPiece.getCurrentShape()[0].length / 2);
                this.currentPiece.y = 0;

                // 检查游戏结束
                if (this.checkCollision(this.currentPiece, 0, 0)) {
                    this.gameOver = true;
                }

                this.drawNext();
            }

            checkCollision(piece, offsetX, offsetY, shape = null) {
                const testShape = shape || piece.getCurrentShape();
                for (let y = 0; y < testShape.length; y++) {
                    for (let x = 0; x < testShape[y].length; x++) {
                        if (testShape[y][x]) {
                            const newX = piece.x + x + offsetX;
                            const newY = piece.y + y + offsetY;

                            if (newX < 0 || newX >= COLS || newY >= ROWS) {
                                return true;
                            }
                            if (newY >= 0 && this.board[newY][newX] !== EMPTY) {
                                return true;
                            }
                        }
                    }
                }
                return false;
            }

            // SRS旋转系统实现
            rotatePiece(direction = 1) {
                if (!this.currentPiece || this.currentPiece.type === 'O') {
                    if (this.currentPiece && this.currentPiece.type === 'O') {
                        // O方块旋转无效果但仍需处理
                    }
                    return false;
                }

                const oldRotation = this.currentPiece.rotation;
                this.currentPiece.rotate(direction);
                const newRotation = this.currentPiece.rotation;

                // 获取正确的踢墙数据
                let kickData;
                if (this.currentPiece.type === 'I') {
                    kickData = SRS_WALL_KICK_DATA_I;
                } else {
                    kickData = SRS_WALL_KICK_DATA;
                }

                const kickKey = `${oldRotation}->${newRotation}`;
                const kicks = kickData[kickKey] || [[0, 0]];

                // 尝试每个踢墙偏移
                for (const [kickX, kickY] of kicks) {
                    if (!this.checkCollision(this.currentPiece, kickX, -kickY)) {
                        this.currentPiece.x += kickX;
                        this.currentPiece.y -= kickY;
                        
                        // 旋转成功后重置锁定延迟
                        if (this.isLocking) {
                            this.resetLockDelay();
                        }
                        return true;
                    }
                }

                // 所有踢墙都失败，恢复原旋转状态
                this.currentPiece.rotation = oldRotation;
                return false;
            }

            moveLeft() {
                if (!this.currentPiece) return false;
                if (!this.checkCollision(this.currentPiece, -1, 0)) {
                    this.currentPiece.x--;
                    if (this.isLocking) this.resetLockDelay();
                    return true;
                }
                return false;
            }

            moveRight() {
                if (!this.currentPiece) return false;
                if (!this.checkCollision(this.currentPiece, 1, 0)) {
                    this.currentPiece.x++;
                    if (this.isLocking) this.resetLockDelay();
                    return true;
                }
                return false;
            }

            moveDown() {
                if (!this.currentPiece) return false;
                if (!this.checkCollision(this.currentPiece, 0, 1)) {
                    this.currentPiece.y++;
                    this.score += 1;
                    return true;
                }
                return false;
            }

            hardDrop() {
                if (!this.currentPiece) return 0;
                let dropDistance = 0;
                while (!this.checkCollision(this.currentPiece, 0, 1)) {
                    this.currentPiece.y++;
                    dropDistance++;
                }
                this.score += dropDistance * 2;
                this.lockPiece();
                return dropDistance;
            }

            softDrop() {
                if (this.moveDown()) {
                    this.score += 1;
                    return true;
                }
                return false;
            }

            // Hold功能实现
            holdCurrentPiece() {
                if (!this.canHold || !this.currentPiece) return false;

                this.canHold = false;

                if (this.holdPiece) {
                    // 交换当前方块和hold方块
                    const temp = this.holdPiece;
                    this.holdPiece = new Tetromino(this.currentPiece.type);
                    this.currentPiece = temp;
                    this.currentPiece.x = Math.floor(COLS / 2) - Math.floor(this.currentPiece.getCurrentShape()[0].length / 2);
                    this.currentPiece.y = 0;
                } else {
                    // 第一次hold
                    this.holdPiece = new Tetromino(this.currentPiece.type);
                    this.spawnPiece();
                }

                this.drawHold();
                return true;
            }

            // 获取Ghost位置
            getGhostY() {
                if (!this.currentPiece) return 0;
                let ghostY = this.currentPiece.y;
                while (!this.checkCollision(this.currentPiece, 0, ghostY - this.currentPiece.y + 1)) {
                    ghostY++;
                }
                return ghostY;
            }

            lockPiece() {
                if (!this.currentPiece) return;

                const shape = this.currentPiece.getCurrentShape();
                for (let y = 0; y < shape.length; y++) {
                    for (let x = 0; x < shape[y].length; x++) {
                        if (shape[y][x]) {
                            const boardY = this.currentPiece.y + y;
                            const boardX = this.currentPiece.x + x;
                            if (boardY >= 0 && boardY < ROWS && boardX >= 0 && boardX < COLS) {
                                this.board[boardY][boardX] = this.currentPiece.type;
                            }
                        }
                    }
                }

                this.clearLines();
                this.spawnPiece();
            }

            // 消行判定与得分计算
            clearLines() {
                let linesCleared = 0;
                const linesToClear = [];

                for (let y = ROWS - 1; y >= 0; y--) {
                    if (this.board[y].every(cell => cell !== EMPTY)) {
                        linesToClear.push(y);
                        linesCleared++;
                    }
                }

                if (linesCleared > 0) {
                    // 删除满行
                    for (const lineY of linesToClear) {
                        this.board.splice(lineY, 1);
                        this.board.unshift(Array(COLS).fill(EMPTY));
                    }

                    // 计算得分 (经典得分系统)
                    const lineScores = [0, 100, 300, 500, 800];
                    this.score += lineScores[linesCleared] * this.level;
                    this.lines += linesCleared;

                    // 等级提升 (每10行升一级)
                    const newLevel = Math.floor(this.lines / 10) + 1;
                    if (newLevel > this.level) {
                        this.level = newLevel;
                        this.updateDropSpeed();
                    }

                    this.updateStats();
                    this.flashBoard();
                }
            }

            updateDropSpeed() {
                // 速度随等级递增
                const speeds = [1000, 900, 800, 700, 600, 500, 400, 300, 200, 100];
                this.dropInterval = speeds[Math.min(this.level - 1, speeds.length - 1)];
            }

            flashBoard() {
                this.canvas.classList.add('flash');
                setTimeout(() => this.canvas.classList.remove('flash'), 100);
            }

            resetLockDelay() {
                if (this.lockTimer) {
                    clearTimeout(this.lockTimer);
                    this.lockTimer = null;
                }
                this.isLocking = true;
                this.lockTimer = setTimeout(() => {
                    if (!this.checkCollision(this.currentPiece, 0, 1)) {
                        this.isLocking = false;
                    }
                }, this.lockDelay);
            }

            update(timestamp) {
                if (this.gameOver) return;

                if (timestamp - this.lastDrop > this.dropInterval) {
                    if (!this.moveDown()) {
                        // 触底锁定
                        if (!this.isLocking) {
                            this.isLocking = true;
                            this.resetLockDelay();
                        }
                        setTimeout(() => {
                            if (this.isLocking && !this.gameOver) {
                                this.lockPiece();
                            }
                        }, this.lockDelay);
                    }
                    this.lastDrop = timestamp;
                }

                this.updateStats();
            }

            updateStats() {
                if (this.scoreEl) this.scoreEl.textContent = this.score.toLocaleString();
                if (this.levelEl) this.levelEl.textContent = this.level;
                if (this.linesEl) this.linesEl.textContent = this.lines;
            }

            // 渲染系统
            draw() {
                // 清空画布
                this.ctx.fillStyle = '#0a0a0f';
                this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);

                // 绘制网格
                this.drawGrid();

                // 绘制已锁定的方块
                this.drawBoard();

                // 绘制Ghost方块
                if (this.currentPiece && !this.gameOver) {
                    this.drawGhost();
                }

                // 绘制当前方块
                if (this.currentPiece && !this.gameOver) {
                    this.drawPiece(this.currentPiece, this.currentPiece.x, this.currentPiece.y);
                }
            }

            drawGrid() {
                this.ctx.strokeStyle = COLORS.gridLine;
                this.ctx.lineWidth = 0.5;

                for (let x = 0; x <= COLS; x++) {
                    this.ctx.beginPath();
                    this.ctx.moveTo(x * BLOCK_SIZE, 0);
                    this.ctx.lineTo(x * BLOCK_SIZE, ROWS * BLOCK_SIZE);
                    this.ctx.stroke();
                }

                for (let y = 0; y <= ROWS; y++) {
                    this.ctx.beginPath();
                    this.ctx.moveTo(0, y * BLOCK_SIZE);
                    this.ctx.lineTo(COLS * BLOCK_SIZE, y * BLOCK_SIZE);
                    this.ctx.stroke();
                }
            }

            drawBoard() {
                for (let y = 0; y < ROWS; y++) {
                    for (let x = 0; x < COLS; x++) {
                        if (this.board[y][x] !== EMPTY) {
                            this.drawBlock(x, y, TETROMINOES[this.board[y][x]].color);
                        }
                    }
                }
            }

            drawPiece(piece, px, py) {
                const shape = piece.getCurrentShape();
                for (let y = 0; y < shape.length; y++) {
                    for (let x = 0; x < shape[y].length; x++) {
                        if (shape[y][x]) {
                            this.drawBlock(px + x, py + y, piece.color);
                        }
                    }
                }
            }

            drawGhost() {
                const ghostY = this.getGhostY();
                if (ghostY !== this.currentPiece.y) {
                    const shape = this.currentPiece.getCurrentShape();
                    for (let y = 0; y < shape.length; y++) {
                        for (let x = 0; x < shape[y].length; x++) {
                            if (shape[y][x]) {
                                this.drawGhostBlock(this.currentPiece.x + x, ghostY + y);
                            }
                        }
                    }
                }
            }

            drawBlock(x, y, color) {
                const padding = 1;
                const size = BLOCK_SIZE - padding * 2;

                // 主色块
                this.ctx.fillStyle = color;
                this.ctx.fillRect(x * BLOCK_SIZE + padding, y * BLOCK_SIZE + padding, size, size);

                // 高光效果
                this.ctx.fillStyle = 'rgba(255, 255, 255, 0.3)';
                this.ctx.fillRect(x * BLOCK_SIZE + padding, y * BLOCK_SIZE + padding, size, 3);
                this.ctx.fillRect(x * BLOCK_SIZE + padding, y * BLOCK_SIZE + padding, 3, size);

                // 阴影效果
                this.ctx.fillStyle = 'rgba(0, 0, 0, 0.3)';
                this.ctx.fillRect(x * BLOCK_SIZE + padding, y * BLOCK_SIZE + size - 2, size, 3);
                this.ctx.fillRect(x * BLOCK_SIZE + size - 2, y * BLOCK_SIZE + padding, 3, size);
            }

            drawGhostBlock(x, y) {
                const padding = 2;
                const size = BLOCK_SIZE - padding * 2;
                this.ctx.strokeStyle = COLORS.ghost;
                this.ctx.lineWidth = 2;
                this.ctx.strokeRect(
                    x * BLOCK_SIZE + padding,
                    y * BLOCK_SIZE + padding,
                    size,
                    size
                );
            }

            drawNext() {
                if (!this.nextPiece) return;

                this.nextCtx.fillStyle = '#12121a';
                this.nextCtx.fillRect(0, 0, this.nextCanvas.width, this.nextCanvas.height);

                const shape = this.nextPiece.shape[0];
                const blockSize = 18;
                const offsetX = (this.nextCanvas.width - shape[0].length * blockSize) / 2;
                const offsetY = (this.nextCanvas.height - shape.length * blockSize) / 2;

                for (let y = 0; y < shape.length; y++) {
                    for (let x = 0; x < shape[y].length; x++) {
                        if (shape[y][x]) {
                            this.drawPreviewBlock(this.nextCtx, offsetX + x * blockSize, offsetY + y * blockSize, blockSize, this.nextPiece.color);
                        }
                    }
                }
            }

            drawHold() {
                this.holdCtx.fillStyle = '#12121a';
                this.holdCtx.fillRect(0, 0, this.holdCanvas.width, this.holdCanvas.height);

                if (!this.holdPiece) return;

                const shape = this.holdPiece.shape[0];
                const blockSize = 18;
                const offsetX = (this.holdCanvas.width - shape[0].length * blockSize) / 2;
                const offsetY = (this.holdCanvas.height - shape.length * blockSize) / 2;

                const color = this.canHold ? this.holdPiece.color : '#444444';

                for (let y = 0; y < shape.length; y++) {
                    for (let x = 0; x < shape[y].length; x++) {
                        if (shape[y][x]) {
                            this.drawPreviewBlock(this.holdCtx, offsetX + x * blockSize, offsetY + y * blockSize, blockSize, color);
                        }
                    }
                }
            }

            drawPreviewBlock(ctx, x, y, size, color) {
                const padding = 1;
                const blockSize = size - padding * 2;

                ctx.fillStyle = color;
                ctx.fillRect(x + padding, y + padding, blockSize, blockSize);

                ctx.fillStyle = 'rgba(255, 255, 255, 0.25)';
                ctx.fillRect(x + padding, y + padding, blockSize, 2);
                ctx.fillRect(x + padding, y + padding, 2, blockSize);
            }

            reset() {
                this.board = this.createBoard();
                this.score = 0;
                this.level = 1;
                this.lines = 0;
                this.bag = [];
                this.holdPiece = null;
                this.canHold = true;
                this.gameOver = false;
                this.lastDrop = 0;
                this.dropInterval = 1000;
                this.isLocking = false;
                if (this.lockTimer) {
                    clearTimeout(this.lockTimer);
                    this.lockTimer = null;
                }
                this.init();
                this.drawHold();
            }

            // 对战模式：添加垃圾行
            addGarbageLines(count) {
                for (let i = 0; i < count; i++) {
                    // 移除顶部行
                    this.board.shift();
                    // 在底部添加垃圾行（有一个随机空洞）
                    const garbageLine = Array(COLS).fill('G');
                    const hole = Math.floor(Math.random() * COLS);
                    garbageLine[hole] = EMPTY;
                    this.board.push(garbageLine);
                }
            }
        }
