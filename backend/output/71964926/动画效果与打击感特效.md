        class InputManager {
            constructor() {
                this.keys = {};
                this.dasState = {};
                
                this.keyBindings = {
                    p1: {
                        left: 'KeyA',
                        right: 'KeyD',
                        down: 'KeyS',
                        rotateCW: 'KeyE',
                        rotateCCW: 'KeyQ',
                        hardDrop: 'Space',
                        hold: 'KeyW'
                    },
                    p2: {
                        left: 'ArrowLeft',
                        right: 'ArrowRight',
                        down: 'ArrowDown',
                        rotateCW: 'Comma',
                        rotateCCW: 'Period',
                        hardDrop: 'Enter',
                        hold: 'ArrowUp'
                    }
                };
                
                this.setupListeners();
            }
            
            setupListeners() {
                document.addEventListener('keydown', (e) => {
                    const code = e.code;
                    if (!this.keys[code] || !this.keys[code].pressed) {
                        this.keys[code] = {
                            pressed: true,
                            justPressed: true,
                            timestamp: performance.now()
                        };
                        this.dasState[code] = null;
                    }
                    
                    if (['Space', 'ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight'].includes(code)) {
                        e.preventDefault();
                    }
                });
                
                document.addEventListener('keyup', (e) => {
                    const code = e.code;
                    if (this.keys[code]) {
                        this.keys[code].pressed = false;
                        this.keys[code].justPressed = false;
                    }
                    this.dasState[code] = null;
                });
            }
            
            isJustPressed(player, action) {
                const keyCode = this.keyBindings[player][action];
                const keyState = this.keys[keyCode];
                if (keyState && keyState.justPressed) {
                    keyState.justPressed = false;
                    return true;
                }
                return false;
            }
            
            getDasAction(player, action, now, isSoftDrop = false) {
                const keyCode = this.keyBindings[player][action];
                const keyState = this.keys[keyCode];
                
                if (!keyState || !keyState.pressed) {
                    this.dasState[keyCode] = null;
                    return false;
                }
                
                if (!this.dasState[keyCode]) {
                    this.dasState[keyCode] = { 
                        triggered: true, 
                        lastTrigger: now,
                        initialFired: true
                    };
                    return true;
                }
                
                const dasData = this.dasState[keyCode];
                const elapsed = now - keyState.timestamp;
                const repeatInterval = isSoftDrop ? DAS_CONFIG.softDropRepeat : DAS_CONFIG.repeat;
                
                if (elapsed >= DAS_CONFIG.delay) {
                    const sinceLastTrigger = now - dasData.lastTrigger;
                    if (sinceLastTrigger >= repeatInterval) {
                        dasData.lastTrigger = now;
                        return true;
                    }
                }
                
                return false;
            }
            
            isPressed(player, action) {
                const keyCode = this.keyBindings[player][action];
                return this.keys[keyCode]?.pressed || false;
            }
            
            reset() {
                this.keys = {};
                this.dasState = {};
            }
        }

        class TetrisGame {
            constructor(player, manager) {
                this.player = player;
                this.manager = manager;
                this.canvas = document.getElementById(`canvas-p${player}`);
                this.ctx = this.canvas.getContext('2d');
                this.holdCanvas = document.getElementById(`hold-p${player}`);
                this.holdCtx = this.holdCanvas.getContext('2d');
                this.previewCanvas = document.getElementById(`preview-p${player}`);
                this.previewCtx = this.previewCanvas.getContext('2d');
                
                this.board = [];
                this.currentPiece = null;
                this.holdPiece = null;
                this.canHold = true;
                this.bag = [];
                this.preview = [];
                this.score = 0;
                this.totalAttack = 0;
                this.combo = -1;
                this.pendingGarbage = 0;
                this.gameOver = false;
                this.lockDelay = 0;
                this.lockDelayMax = 500;
                this.lastDropTime = 0;
                this.dropInterval = 1000;
                this.softDropping = false;
                
                this.init();
            }

            init() {
                this.board = Array(ROWS).fill(null).map(() => Array(COLS).fill(EMPTY));
                this.bag = [];
                this.preview = [];
                this.fillPreview();
                this.spawnPiece();
                this.updateStats();
                this.updatePendingDisplay();
            }

            fillPreview() {
                while (this.preview.length < PREVIEW_COUNT) {
                    if (this.bag.length === 0) {
                        this.bag = Object.keys(TETROMINOES);
                        // Fisher-Yates shuffle
                        for (let i = this.bag.length - 1; i > 0; i--) {
                            const j = Math.floor(Math.random() * (i + 1));
                            [this.bag[i], this.bag[j]] = [this.bag[j], this.bag[i]];
                        }
                    }
                    this.preview.push(this.bag.pop());
                }
            }

            spawnPiece() {
                const type = this.preview.shift();
                this.fillPreview();
                this.currentPiece = new Tetromino(type);
                this.canHold = true;
                this.lockDelay = 0;
                
                if (!this.isValidPosition(this.currentPiece.x, this.currentPiece.y, this.currentPiece.getCurrentShape())) {
                    this.gameOver = true;
                    this.manager.declareGameOver(this.player);
                }
                
                this.drawPreview();
            }

            isValidPosition(x, y, shape) {
                for (let row = 0; row < shape.length; row++) {
                    for (let col = 0; col < shape[row].length; col++) {
                        if (shape[row][col]) {
                            const newX = x + col;
                            const newY = y + row;
                            if (newX < 0 || newX >= COLS || newY >= ROWS) return false;
                            if (newY >= 0 && this.board[newY][newX] !== EMPTY) return false;
                        }
                    }
                }
                return true;
            }

            move(dx, dy) {
                const newX = this.currentPiece.x + dx;
                const newY = this.currentPiece.y + dy;
                
                if (this.isValidPosition(newX, newY, this.currentPiece.getCurrentShape())) {
                    this.currentPiece.x = newX;
                    this.currentPiece.y = newY;
                    return true;
                }
                return false;
            }

            rotate(dir = 1) {
                const oldRotation = this.currentPiece.rotation;
                const oldShape = this.currentPiece.getCurrentShape();
                this.currentPiece.rotate(dir);
                const newShape = this.currentPiece.getCurrentShape();
                
                const kickTable = this.currentPiece.type === 'O' ? [[0,0]] :
                    this.currentPiece.type === 'I' ? SRS_KICKS.I : SRS_KICKS.JLSTZ;
                
                const kickKey = `${oldRotation}>${this.currentPiece.rotation}`;
                const kicks = kickTable[kickKey] || [[0,0]];
                
                for (const [kx, ky] of kicks) {
                    if (this.isValidPosition(this.currentPiece.x + kx, this.currentPiece.y - ky, newShape)) {
                        this.currentPiece.x += kx;
                        this.currentPiece.y -= ky;
                        return true;
                    }
                }
                
                this.currentPiece.rotation = oldRotation;
                return false;
            }

            hardDrop() {
                let dropDistance = 0;
                while (this.move(0, 1)) {
                    dropDistance++;
                }
                this.score += dropDistance * 2;
                
                // 硬降特效
                this.manager.effects.createHardDropImpact(this.player, this.currentPiece.y);
                
                this.lockPiece();
            }

            getGhostY() {
                let ghostY = this.currentPiece.y;
                while (this.isValidPosition(this.currentPiece.x, ghostY + 1, this.currentPiece.getCurrentShape())) {
                    ghostY++;
                }
                return ghostY;
            }

            lockPiece() {
                const shape = this.currentPiece.getCurrentShape();
                const color = this.currentPiece.color;
                
                for (let row = 0; row < shape.length; row++) {
                    for (let col = 0; col < shape[row].length; col++) {
                        if (shape[row][col]) {
                            const boardY = this.currentPiece.y + row;
                            const boardX = this.currentPiece.x + col;
                            if (boardY >= 0 && boardY < ROWS) {
                                this.board[boardY][boardX] = this.currentPiece.type;
                            }
                        }
                    }
                }
                
                const clearedLines = this.clearLines();
                if (clearedLines > 0) {
                    const baseAttack = ATTACK_TABLE[clearedLines] || 0;
                    this.combo++;
                    const comboBonus = COMBO_BONUS[Math.min(this.combo, COMBO_BONUS.length - 1)] || 0;
                    const totalAttack = baseAttack + comboBonus;
                    
                    // 消行特效
                    const clearedRows = [];
                    for (let row = ROWS - 1; row >= 0; row--) {
                        if (this.board[row].every(cell => cell !== EMPTY)) {
                            clearedRows.push(row);
                        }
                    }
                    this.manager.effects.createLineClearEffect(this.player, clearedRows, this.combo);
                    this.manager.effects.showComboPopup(this.player, this.combo);
                    
                    if (totalAttack > 0) {
                        this.manager.sendAttack(this.player, totalAttack);
                    }
                    
                    // 消行分数
                    const lineScores = { 1: 100, 2: 300, 3: 500, 4: 800 };
                    this.score += (lineScores[clearedLines] || 0) * (1 + this.combo * 0.5);
                } else {
                    this.combo = -1;
                    // 添加垃圾行
                    this.addGarbage();
                }
                
                this.updateStats();
                this.spawnPiece();
            }

            clearLines() {
                let linesCleared = 0;
                
                for (let row = ROWS - 1; row >= 0; row--) {
                    if (this.board[row].every(cell => cell !== EMPTY)) {
                        this.board.splice(row, 1);
                        this.board.unshift(Array(COLS).fill(EMPTY));
                        linesCleared++;
                        row++;
                    }
                }
                
                return linesCleared;
            }

            addGarbage() {
                if (this.pendingGarbage <= 0) return;
                
                const lines = Math.min(this.pendingGarbage, ROWS - 4);
                this.pendingGarbage -= lines;
                
                // 受击特效
                this.manager.effects.showElectricShock(this.player);
                
                for (let i = 0; i < lines; i++) {
                    this.board.shift();
                    const garbageLine = Array(COLS).fill('G');
                    const hole = Math.floor(Math.random() * COLS);
                    garbageLine[hole] = EMPTY;
                    this.board.push(garbageLine);
                }
                
                this.updatePendingDisplay();
            }

            hold() {
                if (!this.canHold) return;
                
                if (this.holdPiece === null) {
                    this.holdPiece = this.currentPiece.type;
                    this.spawnPiece();
                } else {
                    const temp = this.holdPiece;
                    this.holdPiece = this.currentPiece.type;
                    this.currentPiece = new Tetromino(temp);
                }
                
                this.canHold = false;
                this.drawHold();
            }

            updateStats() {
                document.getElementById(`score-p${this.player}`).textContent = Math.floor(this.score);
                document.getElementById(`attack-p${this.player}`).textContent = this.totalAttack;
            }

            updatePendingDisplay() {
                const fill = document.getElementById(`garbage-p${this.player}`);
                const percent = Math.min((this.pendingGarbage / ROWS) * 100, 100);
                fill.style.height = `${percent}%`;
            }

            checkDangerZone() {
                // 检查是否进入危险区域（顶部4行有方块）
                let danger = false;
                for (let row = 0; row < 4; row++) {
                    if (this.board[row].some(cell => cell !== EMPTY)) {
                        danger = true;
                        break;
                    }
                }
                this.manager.effects.setDangerPulse(this.player, danger);
            }

            update(now, input) {
                if (this.gameOver) return;

                const playerKey = `p${this.player}`;

                // Hold
                if (input.isJustPressed(playerKey, 'hold')) {
                    this.hold();
                }

                // 左移
                if (input.getDasAction(playerKey, 'left', now)) {
                    this.move(-1, 0);
                }

                // 右移
                if (input.getDasAction(playerKey, 'right', now)) {
                    this.move(1, 0);
                }

                // 软降
                if (input.getDasAction(playerKey, 'down', now, true)) {
                    if (this.move(0, 1)) {
                        this.score += 1;
                        this.softDropping = true;
                    }
                } else {
                    this.softDropping = false;
                }

                // 旋转
                if (input.isJustPressed(playerKey, 'rotateCW')) {
                    this.rotate(1);
                }
                if (input.isJustPressed(playerKey, 'rotateCCW')) {
                    this.rotate(-1);
                }

                // 硬降
                if (input.isJustPressed(playerKey, 'hardDrop')) {
                    this.hardDrop();
                }

                // 自动下落
                const dropSpeed = this.softDropping ? this.dropInterval / 20 : this.dropInterval;
                if (now - this.lastDropTime >= dropSpeed) {
                    if (!this.move(0, 1)) {
                        this.lockDelay += dropSpeed;
                        if (this.lockDelay >= this.lockDelayMax) {
                            this.lockPiece();
                        }
                    } else {
                        this.lockDelay = 0;
                    }
                    this.lastDropTime = now;
                }

                this.checkDangerZone();
            }

            draw() {
                this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);
                
                // 绘制网格
                this.ctx.strokeStyle = COLORS.gridLine;
                this.ctx.lineWidth = 0.5;
                for (let x = 0; x <= COLS; x++) {
                    this.ctx.beginPath();
                    this.ctx.moveTo(x * BLOCK_SIZE, 0);
                    this.ctx.lineTo(x * BLOCK_SIZE, ROWS * BLOCK_SIZE);
                    this.ctx.stroke();
                }
                for (let y = 0; y <= ROWS; y++) {
                    this.ctx.beginPath();
                    this.ctx.moveTo(0, y * BLOCK_SIZE);
                    this.ctx.lineTo(COLS * BLOCK_SIZE, y * BLOCK_SIZE);
                    this.ctx.stroke();
                }

                // 绘制已放置的方块
                for (let row = 0; row < ROWS; row++) {
                    for (let col = 0; col < COLS; col++) {
                        if (this.board[row][col] !== EMPTY) {
                            const type = this.board[row][col];
                            const color = type === 'G' ? COLORS.G : TETROMINOES[type]?.color;
                            if (color) {
                                this.drawBlock(col, row, color);
                            }
                        }
                    }
                }

                // 绘制Ghost piece
                if (this.currentPiece && !this.gameOver) {
                    const ghostY = this.getGhostY();
                    const shape = this.currentPiece.getCurrentShape();
                    
                    this.ctx.fillStyle = COLORS.ghost;
                    this.ctx.strokeStyle = 'rgba(255, 255, 255, 0.2)';
                    this.ctx.lineWidth = 1;
                    
                    for (let row = 0; row < shape.length; row++) {
                        for (let col = 0; col < shape[row].length; col++) {
                            if (shape[row][col]) {
                                const x = (this.currentPiece.x + col) * BLOCK_SIZE;
                                const y = (ghostY + row) * BLOCK_SIZE;
                                
                                this.ctx.fillRect(x + 2, y + 2, BLOCK_SIZE - 4, BLOCK_SIZE - 4);
                                this.ctx.strokeRect(x + 2, y + 2, BLOCK_SIZE - 4, BLOCK_SIZE - 4);
                            }
                        }
                    }
                }

                // 绘制当前方块
                if (this.currentPiece && !this.gameOver) {
                    const shape = this.currentPiece.getCurrentShape();
                    const color = this.currentPiece.color;
                    
                    for (let row = 0; row < shape.length; row++) {
                        for (let col = 0; col < shape[row].length; col++) {
                            if (shape[row][col]) {
                                this.drawBlock(
                                    this.currentPiece.x + col,
                                    this.currentPiece.y + row,
                                    color
                                );
                            }
                        }
                    }
                }
            }

            drawBlock(x, y, color) {
                const px = x * BLOCK_SIZE;
                const py = y * BLOCK_SIZE;
                
                // 发光效果
                this.ctx.shadowColor = color.glow;
                this.ctx.shadowBlur = 8;
                
                // 主体
                this.ctx.fillStyle = color.main;
                this.ctx.fillRect(px + 1, py + 1, BLOCK_SIZE - 2, BLOCK_SIZE - 2);
                
                // 高光
                this.ctx.fillStyle = 'rgba(255, 255, 255, 0.3)';
                this.ctx.fillRect(px + 2, py + 2, BLOCK_SIZE - 6, 3);
                this.ctx.fillRect(px + 2, py + 2, 3, BLOCK_SIZE - 6);
                
                // 暗部
                this.ctx.fillStyle = color.dark;
                this.ctx.fillRect(px + BLOCK_SIZE - 4, py + 4, 2, BLOCK_SIZE - 6);
                this.ctx.fillRect(px + 4, py + BLOCK_SIZE - 4, BLOCK_SIZE - 6, 2);
                
                this.ctx.shadowBlur = 0;
            }

            drawPreview() {
                this.previewCtx.clearRect(0, 0, this.previewCanvas.width, this.previewCanvas.height);
                
                const previewSize = 12;
                const gap = 8;
                
                for (let i = 0; i < Math.min(this.preview.length, PREVIEW_COUNT); i++) {
                    const type = this.preview[i];
                    const shape = TETROMINOES[type].shape[0];
                    const color = TETROMINOES[type].color;
                    
                    const offsetY = i * (60 + gap) + 15;
                    const offsetX = (this.previewCanvas.width - shape[0].length * previewSize) / 2;
                    
                    for (let row = 0; row < shape.length; row++) {
                        for (let col = 0; col < shape[row].length; col++) {
                            if (shape[row][col]) {
                                this.previewCtx.fillStyle = color.main;
                                this.previewCtx.shadowColor = color.glow;
                                this.previewCtx.shadowBlur = 4;
                                this.previewCtx.fillRect(
                                    offsetX + col * previewSize + 1,
                                    offsetY + row * previewSize + 1,
                                    previewSize - 2,
                                    previewSize - 2
                                );
                                this.previewCtx.shadowBlur = 0;
                            }
                        }
                    }
                }
            }

            drawHold() {
                this.holdCtx.clearRect(0, 0, this.holdCanvas.width, this.holdCanvas.height);
                
                if (!this.holdPiece) return;
                
                const shape = TETROMINOES[this.holdPiece].shape[0];
                const color = TETROMINOES[this.holdPiece].color;
                const previewSize = 14;
                
                const offsetX = (this.holdCanvas.width - shape[0].length * previewSize) / 2;
                const offsetY = (this.holdCanvas.height - shape.length * previewSize) / 2;
                
                for (let row = 0; row < shape.length; row++) {
                    for (let col = 0; col < shape[row].length; col++) {
                        if (shape[row][col]) {
                            this.holdCtx.fillStyle = this.canHold ? color.main : 'rgba(100, 100, 100, 0.5)';
                            this.holdCtx.shadowColor = this.canHold ? color.glow : 'transparent';
                            this.holdCtx.shadowBlur = this.canHold ? 4 : 0;
                            this.holdCtx.fillRect(
                                offsetX + col * previewSize + 1,
                                offsetY + row * previewSize + 1,
                                previewSize - 2,
                                previewSize - 2
                            );
                            this.holdCtx.shadowBlur = 0;
                        }
                    }
                }
            }
        }

        // ============================================
        // 主程序
        // ============================================
        const input = new InputManager();
        const gameManager = new GameManager();
        let game1, game2;
        let gameStarted = false;
        let lastTime = 0;

        function init() {
            game1 = new TetrisGame(1, gameManager);
            game2 = new TetrisGame(2, gameManager);
            gameManager.setGames(game1, game2);
            
            // 绘制初始状态
            game1.draw();
            game2.draw();
            game1.drawPreview();
            game2.drawPreview();
            game1.drawHold();
            game2.drawHold();
        }

        function gameLoop(timestamp) {
            if (!gameStarted) {
                requestAnimationFrame(gameLoop);
                return;
            }

            if (!gameManager.gameOver) {
                game1.update(timestamp, input);
                game2.update(timestamp, input);
            }

            game1.draw();
            game2.draw();

            requestAnimationFrame(gameLoop);
        }

        function startGame() {
            document.getElementById('startOverlay').classList.add('hidden');
            gameStarted = true;
            lastTime = performance.now();
        }

        function restartGame() {
            document.getElementById('gameOverOverlay').classList.remove('active');
            gameManager.reset();
            input.reset();
            
            game1 = new TetrisGame(1, gameManager);
            game2 = new TetrisGame(2, gameManager);
            gameManager.setGames(game1, game2);
            
            gameStarted = true;
        }

        // 事件绑定
        document.getElementById('startBtn').addEventListener('click', startGame);
        document.getElementById('restartBtn').addEventListener('click', restartGame);

        // 初始化
        init();
        requestAnimationFrame(gameLoop);
    </script>
</body>
</html>

## 主要特效功能说明

### 1. 硬降特效系统
- **冲击波环**：硬降时在落点位置生成扩散的圆环动画
- **屏幕微震**：根据玩家位置产生不同方向的震动效果
- **能量波纹**：从落点向外扩散的视觉冲击

### 2. 消行特效系统
- **能量爆发**：每行清除时产生横向能量光束
- **粒子爆炸**：消行位置的方块爆散成粒子
- **连击弹窗**：显示连击数，字体大小随连击数增长

### 3. 受击特效
- **电流闪烁**：场地边缘出现红色电流效果
- **屏幕震动**：受到垃圾行攻击时的震动反馈

### 4. 危机预警系统
- **脉动发光**：当方块堆叠到顶部4行时，场地边框开始脉动
- **渐变警告**：使用动态渐变背景增强紧迫感

### 5. Ghost Piece（落点预览）
- 半透明显示方块最终落点位置
- 带有微弱边框的视觉效果

### 6. 连击增强
- 连击数越高，粒子数量越多、弹窗越大
- 颜色强度和发光效果随连击提升
