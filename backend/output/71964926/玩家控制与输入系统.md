        // ============================================
        // 输入管理器 - DAS系统
        // ============================================
        class InputManager {
            constructor() {
                this.keys = {};
                this.dasState = {};
                
                // 玩家1: WASD移动 + QE旋转 + 空格硬降 + W hold
                // 玩家2: 方向键移动 + < >旋转 + Enter硬降 + ↑ hold
                this.keyBindings = {
                    p1: {
                        left: 'KeyA',
                        right: 'KeyD',
                        down: 'KeyS',
                        rotateCW: 'KeyE',
                        rotateCCW: 'KeyQ',
                        hardDrop: 'Space',
                        hold: 'KeyW'
                    },
                    p2: {
                        left: 'ArrowLeft',
                        right: 'ArrowRight',
                        down: 'ArrowDown',
                        rotateCW: 'Comma',      // < 键
                        rotateCCW: 'Period',   // > 键
                        hardDrop: 'Enter',
                        hold: 'ArrowUp'
                    }
                };
                
                this.setupListeners();
            }
            
            setupListeners() {
                document.addEventListener('keydown', (e) => {
                    const code = e.code;
                    if (!this.keys[code] || !this.keys[code].pressed) {
                        this.keys[code] = {
                            pressed: true,
                            justPressed: true,
                            timestamp: performance.now()
                        };
                        // 重置DAS状态
                        this.dasState[code] = null;
                    }
                    
                    // 阻止某些按键的默认行为
                    if (['Space', 'ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight'].includes(code)) {
                        e.preventDefault();
                    }
                });
                
                document.addEventListener('keyup', (e) => {
                    const code = e.code;
                    if (this.keys[code]) {
                        this.keys[code].pressed = false;
                        this.keys[code].justPressed = false;
                    }
                    this.dasState[code] = null;
                });
            }
            
            // 检查是否刚刚按下（用于单次触发的动作如旋转、硬降）
            isJustPressed(player, action) {
                const keyCode = this.keyBindings[player][action];
                const keyState = this.keys[keyCode];
                if (keyState && keyState.justPressed) {
                    keyState.justPressed = false;
                    return true;
                }
                return false;
            }
            
            // DAS处理 - 用于左右移动和软降
            getDasAction(player, action, now, isSoftDrop = false) {
                const keyCode = this.keyBindings[player][action];
                const keyState = this.keys[keyCode];
                
                if (!keyState || !keyState.pressed) {
                    this.dasState[keyCode] = null;
                    return false;
                }
                
                // 首次触发
                if (!this.dasState[keyCode]) {
                    this.dasState[keyCode] = { 
                        triggered: true, 
                        lastTrigger: now,
                        initialFired: true
                    };
                    return true;
                }
                
                const dasData = this.dasState[keyCode];
                const elapsed = now - keyState.timestamp;
                const repeatInterval = isSoftDrop ? DAS_CONFIG.softDropRepeat : DAS_CONFIG.repeat;
                
                // DAS延迟后开始重复触发
                if (elapsed >= DAS_CONFIG.delay) {
                    const sinceLastTrigger = now - dasData.lastTrigger;
                    if (sinceLastTrigger >= repeatInterval) {
                        dasData.lastTrigger = now;
                        return true;
                    }
                }
                
                return false;
            }
            
            // 检查按键是否按下（用于硬降等即时动作）
            isPressed(player, action) {
                const keyCode = this.keyBindings[player][action];
                return this.keys[keyCode]?.pressed || false;
            }
            
            // 重置所有状态
            reset() {
                this.keys = {};
                this.dasState = {};
            }
        }

        // 游戏主类
        // ============================================
        class TetrisGame {
            constructor(boardId, nextId, holdId, scoreId, levelId, linesId, playerNum) {
                this.canvas = document.getElementById(boardId);
                this.ctx = this.canvas.getContext('2d');
                this.nextCanvas = document.getElementById(nextId);
                this.nextCtx = this.nextCanvas.getContext('2d');
                this.holdCanvas = document.getElementById(holdId);
                this.holdCtx = this.holdCanvas.getContext('2d');

                this.scoreEl = document.getElementById(scoreId);
                this.levelEl = document.getElementById(levelId);
                this.linesEl = document.getElementById(linesId);

                this.playerNum = playerNum;
                this.playerKey = playerNum === 1 ? 'p1' : 'p2';

                this.init();
            }

            init() {
                this.board = this.createBoard();
                this.score = 0;
                this.level = 1;
                this.lines = 0;
                this.currentPiece = null;
                this.nextPieces = [];
                this.holdPiece = null;
                this.canHold = true;
                this.gameOver = false;
                this.lastDrop = 0;
                this.dropInterval = 800;
                this.bag = [];

                // 预生成方块
                this.nextPieces = [];
                for (let i = 0; i < PREVIEW_COUNT; i++) {
                    this.nextPieces.push(this.getNextFromBag());
                }

                this.spawnPiece();
                this.updateStats();
                this.draw();
            }

            createBoard() {
                return Array.from({ length: ROWS }, () => Array(COLS).fill(EMPTY));
            }

            getNextFromBag() {
                if (this.bag.length === 0) {
                    this.bag = ['I', 'O', 'T', 'S', 'Z', 'J', 'L'];
                    // Fisher-Yates洗牌
                    for (let i = this.bag.length - 1; i > 0; i--) {
                        const j = Math.floor(Math.random() * (i + 1));
                        [this.bag[i], this.bag[j]] = [this.bag[j], this.bag[i]];
                    }
                }
                return new Tetromino(this.bag.pop());
            }

            spawnPiece() {
                this.currentPiece = this.nextPieces.shift();
                this.nextPieces.push(this.getNextFromBag());
                this.canHold = true;

                this.currentPiece.x = Math.floor(COLS / 2) - Math.floor(this.currentPiece.getCurrentShape()[0].length / 2);
                this.currentPiece.y = 0;

                if (this.checkCollision(this.currentPiece, 0, 0)) {
                    this.gameOver = true;
                }

                this.drawNext();
                this.drawHold();
            }

            checkCollision(piece, ox, oy, shape = null) {
                const s = shape || piece.getCurrentShape();
                for (let y = 0; y < s.length; y++) {
                    for (let x = 0; x < s[y].length; x++) {
                        if (s[y][x]) {
                            const nx = piece.x + x + ox;
                            const ny = piece.y + y + oy;
                            if (nx < 0 || nx >= COLS || ny >= ROWS) return true;
                            if (ny >= 0 && this.board[ny][nx] !== EMPTY) return true;
                        }
                    }
                }
                return false;
            }

            rotatePiece(dir = 1) {
                if (!this.currentPiece || this.currentPiece.type === 'O') return false;

                const oldRot = this.currentPiece.rotation;
                this.currentPiece.rotate(dir);
                const newRot = this.currentPiece.rotation;

                const kicks = this.currentPiece.type === 'I' ? SRS_KICKS.I : SRS_KICKS.JLSTZ;
                const kickKey = `${oldRot}>${newRot}`;
                const kickData = kicks[kickKey] || [[0, 0]];

                for (const [kx, ky] of kickData) {
                    if (!this.checkCollision(this.currentPiece, kx, -ky)) {
                        this.currentPiece.x += kx;
                        this.currentPiece.y -= ky;
                        return true;
                    }
                }

                this.currentPiece.rotation = oldRot;
                return false;
            }

            moveLeft() {
                if (!this.currentPiece) return false;
                if (!this.checkCollision(this.currentPiece, -1, 0)) {
                    this.currentPiece.x--;
                    return true;
                }
                return false;
            }

            moveRight() {
                if (!this.currentPiece) return false;
                if (!this.checkCollision(this.currentPiece, 1, 0)) {
                    this.currentPiece.x++;
                    return true;
                }
                return false;
            }

            moveDown() {
                if (!this.currentPiece) return false;
                if (!this.checkCollision(this.currentPiece, 0, 1)) {
                    this.currentPiece.y++;
                    this.score += 1;
                    return true;
                }
                return false;
            }

            hardDrop() {
                if (!this.currentPiece) return;
                let dropDist = 0;
                while (!this.checkCollision(this.currentPiece, 0, 1)) {
                    this.currentPiece.y++;
                    dropDist++;
                }
                this.score += dropDist * 2;
                this.lockPiece();
            }

            holdPiece_() {
                if (!this.canHold || !this.currentPiece) return;

                this.canHold = false;
                const type = this.currentPiece.type;

                if (this.holdPiece) {
                    const holdType = this.holdPiece.type;
                    this.holdPiece = new Tetromino(type);
                    this.currentPiece = new Tetromino(holdType);
                    this.currentPiece.x = Math.floor(COLS / 2) - Math.floor(this.currentPiece.getCurrentShape()[0].length / 2);
                    this.currentPiece.y = 0;
                } else {
                    this.holdPiece = new Tetromino(type);
                    this.spawnPiece();
                }

                this.drawHold();
            }

            getGhostY() {
                if (!this.currentPiece) return 0;
                let ghostY = this.currentPiece.y;
                while (!this.checkCollision(this.currentPiece, 0, ghostY - this.currentPiece.y + 1)) {
                    ghostY++;
                }
                return ghostY;
            }

            lockPiece() {
                if (!this.currentPiece) return;

                const shape = this.currentPiece.getCurrentShape();
                for (let y = 0; y < shape.length; y++) {
                    for (let x = 0; x < shape[y].length; x++) {
                        if (shape[y][x]) {
                            const boardY = this.currentPiece.y + y;
                            const boardX = this.currentPiece.x + x;
                            if (boardY >= 0 && boardY < ROWS && boardX >= 0 && boardX < COLS) {
                                this.board[boardY][boardX] = this.currentPiece.type;
                            }
                        }
                    }
                }

                this.clearLines();
                this.spawnPiece();
            }

            clearLines() {
                let linesCleared = 0;
                
                for (let y = ROWS - 1; y >= 0; y--) {
                    if (this.board[y].every(cell => cell !== EMPTY)) {
                        this.board.splice(y, 1);
                        this.board.unshift(Array(COLS).fill(EMPTY));
                        linesCleared++;
                        y++; // 重新检查当前行
                    }
                }

                if (linesCleared > 0) {
                    // 计分系统
                    const lineScores = [0, 100, 300, 500, 800];
                    this.score += lineScores[linesCleared] * this.level;
                    this.lines += linesCleared;
                    
                    // 升级
                    const newLevel = Math.floor(this.lines / 10) + 1;
                    if (newLevel > this.level) {
                        this.level = newLevel;
                        this.dropInterval = Math.max(100, 800 - (this.level - 1) * 70);
                    }
                    
                    this.updateStats();
                }
            }

            updateStats() {
                this.scoreEl.textContent = this.score.toLocaleString();
                this.levelEl.textContent = this.level;
                this.linesEl.textContent = this.lines;
            }

            update(now, input) {
                if (this.gameOver) return;

                const player = this.playerKey;

                // 处理输入 - DAS移动
                if (input.getDasAction(player, 'left', now)) {
                    this.moveLeft();
                }
                if (input.getDasAction(player, 'right', now)) {
                    this.moveRight();
                }
                if (input.getDasAction(player, 'down', now, true)) {
                    this.moveDown();
                }

                // 单次触发动作
                if (input.isJustPressed(player, 'rotateCW')) {
                    this.rotatePiece(1);
                }
                if (input.isJustPressed(player, 'rotateCCW')) {
                    this.rotatePiece(-1);
                }
                if (input.isJustPressed(player, 'hardDrop')) {
                    this.hardDrop();
                }
                if (input.isJustPressed(player, 'hold')) {
                    this.holdPiece_();
                }

                // 自动下落
                if (now - this.lastDrop >= this.dropInterval) {
                    if (!this.moveDown()) {
                        this.lockPiece();
                    }
                    this.lastDrop = now;
                }
            }

            draw() {
                // 清空画布
                this.ctx.fillStyle = COLORS.grid;
                this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);

                // 绘制网格线
                this.ctx.strokeStyle = COLORS.gridLine;
                this.ctx.lineWidth = 0.5;
                for (let x = 0; x <= COLS; x++) {
                    this.ctx.beginPath();
                    this.ctx.moveTo(x * BLOCK_SIZE, 0);
                    this.ctx.lineTo(x * BLOCK_SIZE, ROWS * BLOCK_SIZE);
                    this.ctx.stroke();
                }
                for (let y = 0; y <= ROWS; y++) {
                    this.ctx.beginPath();
                    this.ctx.moveTo(0, y * BLOCK_SIZE);
                    this.ctx.lineTo(COLS * BLOCK_SIZE, y * BLOCK_SIZE);
                    this.ctx.stroke();
                }

                // 绘制已锁定的方块
                for (let y = 0; y < ROWS; y++) {
                    for (let x = 0; x < COLS; x++) {
                        if (this.board[y][x] !== EMPTY) {
                            this.drawBlock(this.ctx, x, y, TETROMINOES[this.board[y][x]].color);
                        }
                    }
                }

                // 绘制幽灵方块
                if (this.currentPiece && !this.gameOver) {
                    const ghostY = this.getGhostY();
                    const shape = this.currentPiece.getCurrentShape();
                    for (let y = 0; y < shape.length; y++) {
                        for (let x = 0; x < shape[y].length; x++) {
                            if (shape[y][x]) {
                                this.ctx.fillStyle = COLORS.ghost;
                                this.ctx.fillRect(
                                    (this.currentPiece.x + x) * BLOCK_SIZE + 1,
                                    (ghostY + y) * BLOCK_SIZE + 1,
                                    BLOCK_SIZE - 2,
                                    BLOCK_SIZE - 2
                                );
                            }
                        }
                    }
                }

                // 绘制当前方块
                if (this.currentPiece && !this.gameOver) {
                    const shape = this.currentPiece.getCurrentShape();
                    for (let y = 0; y < shape.length; y++) {
                        for (let x = 0; x < shape[y].length; x++) {
                            if (shape[y][x]) {
                                this.drawBlock(
                                    this.ctx,
                                    this.currentPiece.x + x,
                                    this.currentPiece.y + y,
                                    this.currentPiece.color
                                );
                            }
                        }
                    }
                }

                // 游戏结束遮罩
                if (this.gameOver) {
                    this.ctx.fillStyle = 'rgba(0, 0, 0, 0.7)';
                    this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);
                    
                    this.ctx.fillStyle = '#ff006e';
                    this.ctx.font = 'bold 16px Orbitron';
                    this.ctx.textAlign = 'center';
                    this.ctx.fillText('GAME OVER', this.canvas.width / 2, this.canvas.height / 2);
                }
            }

            drawBlock(ctx, x, y, color) {
                const px = x * BLOCK_SIZE;
                const py = y * BLOCK_SIZE;
                const size = BLOCK_SIZE - 1;

                // 发光效果
                ctx.shadowColor = color.glow;
                ctx.shadowBlur = 8;

                // 主体颜色
                ctx.fillStyle = color.main;
                ctx.fillRect(px + 1, py + 1, size - 1, size - 1);

                // 高光
                ctx.fillStyle = 'rgba(255, 255, 255, 0.3)';
                ctx.fillRect(px + 2, py + 2, size - 4, 3);

                // 阴影
                ctx.fillStyle = color.dark;
                ctx.fillRect(px + 2, py + size - 4, size - 4, 2);

                ctx.shadowBlur = 0;
            }

            drawNext() {
                const ctx = this.nextCtx;
                ctx.fillStyle = COLORS.grid;
                ctx.fillRect(0, 0, this.nextCanvas.width, this.nextCanvas.height);

                let offsetY = 5;
                for (let i = 0; i < this.nextPieces.length && i < 3; i++) {
                    const piece = this.nextPieces[i];
                    const shape = piece.getCurrentShape();
                    const color = piece.color;

                    // 计算居中偏移
                    const pieceWidth = shape[0].length * 16;
                    const pieceHeight = shape.length * 16;
                    const offsetX = (this.nextCanvas.width - pieceWidth) / 2;

                    for (let y = 0; y < shape.length; y++) {
                        for (let x = 0; x < shape[y].length; x++) {
                            if (shape[y][x]) {
                                ctx.fillStyle = color.main;
                                ctx.shadowColor = color.glow;
                                ctx.shadowBlur = 4;
                                ctx.fillRect(offsetX + x * 16 + 1, offsetY + y * 16 + 1, 14, 14);
                            }
                        }
                    }
                    ctx.shadowBlur = 0;
                    offsetY += pieceHeight + 10;
                }
            }

            drawHold() {
                const ctx = this.holdCtx;
                ctx.fillStyle = COLORS.grid;
                ctx.fillRect(0, 0, this.holdCanvas.width, this.holdCanvas.height);

                if (!this.holdPiece) return;

                const shape = this.holdPiece.getCurrentShape();
                const color = this.holdPiece.color;

                const pieceWidth = shape[0].length * 16;
                const offsetX = (this.holdCanvas.width - pieceWidth) / 2;
                const offsetY = (this.holdCanvas.height - shape.length * 16) / 2;

                for (let y = 0; y < shape.length; y++) {
                    for (let x = 0; x < shape[y].length; x++) {
                        if (shape[y][x]) {
                            ctx.fillStyle = this.canHold ? color.main : 'rgba(100, 100, 100, 0.5)';
                            ctx.shadowColor = this.canHold ? color.glow : 'transparent';
                            ctx.shadowBlur = 4;
                            ctx.fillRect(offsetX + x * 16 + 1, offsetY + y * 16 + 1, 14, 14);
                        }
                    }
                }
                ctx.shadowBlur = 0;
            }
        }

        // ============================================
        // 游戏管理器
        // ============================================
        class GameManager {
            constructor() {
                this.input = new InputManager();
                this.game1 = null;
                this.game2 = null;
                this.isRunning = false;
                this.isPaused = false;
                this.lastFrame = 0;

                this.gameStatusEl = document.getElementById('gameStatus');
                this.statusTitleEl = document.getElementById('statusTitle');
                this.statusMessageEl = document.getElementById('statusMessage');
                this.restartBtn = document.getElementById('restartBtn');
                this.pauseOverlay = document.getElementById('pauseOverlay');

                this.setupEventListeners();
                this.start();
            }

            setupEventListeners() {
                this.restartBtn.addEventListener('click', () => this.restart());
                
                // ESC暂停
                document.addEventListener('keydown', (e) => {
                    if (e.code === 'Escape' && this.isRunning) {
                        this.togglePause();
                    }
                });
            }

            start() {
                this.game1 = new TetrisGame('board1', 'next1', 'hold1', 'score1', 'level1', 'lines1', 1);
                this.game2 = new TetrisGame('board2', 'next2', 'hold2', 'score2', 'level2', 'lines2', 2);
                this.isRunning = true;
                this.isPaused = false;
                this.lastFrame = performance.now();
                this.gameLoop();
            }

            restart() {
                this.input.reset();
                this.gameStatusEl.classList.remove('show');
                this.pauseOverlay.classList.remove('show');
                this.game1.init();
                this.game2.init();
                this.isRunning = true;
                this.isPaused = false;
                this.lastFrame = performance.now();
            }

            togglePause() {
                this.isPaused = !this.isPaused;
                if (this.isPaused) {
                    this.pauseOverlay.classList.add('show');
                } else {
                    this.pauseOverlay.classList.remove('show');
                    this.lastFrame = performance.now();
                    this.gameLoop();
                }
            }

            gameLoop() {
                if (!this.isRunning || this.isPaused) return;

                const now = performance.now();

                // 更新两个游戏
                this.game1.update(now, this.input);
                this.game2.update(now, this.input);

                // 绘制
                this.game1.draw();
                this.game2.draw();

                // 检查游戏结束
                if (this.game1.gameOver || this.game2.gameOver) {
                    this.endGame();
                    return;
                }

                // 60fps
                requestAnimationFrame(() => this.gameLoop());
            }

            endGame() {
                this.isRunning = false;

                let title, message;
                if (this.game1.gameOver && this.game2.gameOver) {
                    title = 'DRAW';
                    this.statusTitleEl.className = 'status-title draw';
                    message = 'Both players game over!';
                } else if (this.game1.gameOver) {
                    title = 'PLAYER 2 WINS';
                    this.statusTitleEl.className = 'status-title p2-win';
                    message = `Score: ${this.game2.score.toLocaleString()}`;
                } else {
                    title = 'PLAYER 1 WINS';
                    this.statusTitleEl.className = 'status-title p1-win';
                    message = `Score: ${this.game1.score.toLocaleString()}`;
                }

                this.statusTitleEl.textContent = title;
                this.statusMessageEl.textContent = message;
                this.gameStatusEl.classList.add('show');
            }
        }

        // 启动游戏
        const gameManager = new GameManager();
    </script>
</body>

## 实现说明

### 1. 双人键盘控制方案

**玩家1 (P1)**：
- `A/D` - 左右移动
- `W` - Hold暂存方块
- `S` - 软降（加速下落）
- `Q/E` - 逆时针/顺时针旋转
- `Space` - 硬降（瞬间落底）

**玩家2 (P2)**：
- `←/→` - 左右移动
- `↑` - Hold暂存方块
- `↓` - 软降（加速下落）
- `,/.` - 逆时针/顺时针旋转（对应 `<` 和 `>` 键）
- `Enter` - 硬降（瞬间落底）

### 2. DAS (Delayed Auto Shift) 系统

```javascript
const DAS_CONFIG = {
    delay: 170,           // 首次触发延迟 - 按住170ms后开始自动移动
    repeat: 50,           // 重复间隔 - 每50ms移动一次（20次/秒）
    softDropRepeat: 50    // 软降速度
};
```

DAS系统工作原理：
1. **首次按键**：立即触发一次移动
2. **延迟阶段**：等待170ms
3. **自动重复**：之后每50ms触发一次，实现流畅的连续移动

### 3. 60fps 线性移动

- 使用 `requestAnimationFrame` 确保流畅的60fps渲染
- 移动无缓动/插值，确保精确的落点判断
- 输入处理与渲染分离，避免输入延迟

- **ESC** 暂停游戏
- **Ghost方块** 显示落点预览
- **Hold功能** 暂存/交换当前方块
- **3个预览方块** 显示接下来将出现的方块
- **SRS旋转系统** 支持踢墙
