                '0>3': [[0,0],[1,0],[1,1],[0,-2],[1,-2]]
            },
            I: {
                '0>1': [[0,0],[-2,0],[1,0],[-2,-1],[1,2]],
                '1>0': [[0,0],[2,0],[-1,0],[2,1],[-1,-2]],
                '1>2': [[0,0],[-1,0],[2,0],[-1,2],[2,-1]],
                '2>1': [[0,0],[1,0],[-2,0],[1,-2],[-2,1]],
                '2>3': [[0,0],[2,0],[-1,0],[2,1],[-1,-2]],
                '3>2': [[0,0],[-2,0],[1,0],[-2,-1],[1,2]],
                '3>0': [[0,0],[1,0],[-2,0],[1,-2],[-2,1]],
                '0>3': [[0,0],[-1,0],[2,0],[-1,2],[2,-1]]
            }
        };

        // ============================================
        // 方块类
        // ============================================
        class Tetromino {
            constructor(type) {
                this.type = type;
                this.shape = TETROMINOES[type].shape;
                this.color = TETROMINOES[type].color;
                this.rotation = 0;
                this.x = Math.floor(COLS / 2) - Math.floor(this.shape[0][0].length / 2);
                this.y = 0;
            }

            getCurrentShape() {
                return this.shape[this.rotation];
            }

            rotate(dir = 1) {
                const old = this.rotation;
                this.rotation = dir === 1 ? (this.rotation + 1) % 4 : (this.rotation + 3) % 4;
                return old;
            }
        }

        // ============================================
        // 输入管理器 - DAS系统
        // ============================================
        class InputManager {
            constructor() {
                this.keys = {};
                this.dasState = {};
                
                this.keyBindings = {
                    p1: {
                        left: 'KeyA',
                        right: 'KeyD',
                        down: 'KeyS',
                        rotateCW: 'KeyE',
                        rotateCCW: 'KeyQ',
                        hardDrop: 'Space',
                        hold: 'KeyW'
                    },
                    p2: {
                        left: 'ArrowLeft',
                        right: 'ArrowRight',
                        down: 'ArrowDown',
                        rotateCW: 'Comma',
                        rotateCCW: 'Period',
                        hardDrop: 'Enter',
                        hold: 'ArrowUp'
                    }
                };
                
                this.setupListeners();
            }
            
            setupListeners() {
                document.addEventListener('keydown', (e) => {
                    const code = e.code;
                    if (!this.keys[code] || !this.keys[code].pressed) {
                        this.keys[code] = {
                            pressed: true,
                            justPressed: true,
                            timestamp: performance.now()
                        };
                        this.dasState[code] = null;
                    }
                    
                    if (['Space', 'ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight'].includes(code)) {
                        e.preventDefault();
                    }
                });
                
                document.addEventListener('keyup', (e) => {
                    const code = e.code;
                    if (this.keys[code]) {
                        this.keys[code].pressed = false;
                        this.keys[code].justPressed = false;
                    }
                    this.dasState[code] = null;
                });
            }
            
            isJustPressed(player, action) {
                const keyCode = this.keyBindings[player][action];
                const keyState = this.keys[keyCode];
                if (keyState && keyState.justPressed) {
                    keyState.justPressed = false;
                    return true;
                }
                return false;
            }
            
            getDasAction(player, action, now, isSoftDrop = false) {
                const keyCode = this.keyBindings[player][action];
                const keyState = this.keys[keyCode];
                
                if (!keyState || !keyState.pressed) {
                    this.dasState[keyCode] = null;
                    return false;
                }
                
                if (!this.dasState[keyCode]) {
                    this.dasState[keyCode] = { 
                        triggered: true, 
                        lastTrigger: now,
                        initialFired: true
                    };
                    return true;
                }
                
                const dasData = this.dasState[keyCode];
                const elapsed = now - keyState.timestamp;
                const repeatInterval = isSoftDrop ? DAS_CONFIG.softDropRepeat : DAS_CONFIG.repeat;
                
                if (elapsed >= DAS_CONFIG.delay) {
                    const sinceLastTrigger = now - dasData.lastTrigger;
                    if (sinceLastTrigger >= repeatInterval) {
                        dasData.lastTrigger = now;
                        return true;
                    }
                }
                
                return false;
            }
            
            isPressed(player, action) {
                const keyCode = this.keyBindings[player][action];
                return this.keys[keyCode]?.pressed || false;
            }
            
            reset() {
                this.keys = {};
                this.dasState = {};
            }
        }

        // ============================================
        // 游戏管理器 - 处理攻击传递
        // ============================================
        class GameManager {
            constructor() {
                this.game1 = null;
                this.game2 = null;
                this.gameOver = false;
                this.winner = null;
            }

            setGames(game1, game2) {
                this.game1 = game1;
                this.game2 = game2;
            }

            // 发送攻击到对手
            sendAttack(fromPlayer, lines) {
                if (this.gameOver || lines <= 0) return;
                
                const targetGame = fromPlayer === 1 ? this.game2 : this.game1;
                const fromGame = fromPlayer === 1 ? this.game1 : this.game2;
                
                // 显示攻击弹道动画
                this.showAttackProjectile(fromPlayer, lines);
                
                // 将攻击添加到对手的待处理队列
                targetGame.pendingGarbage += lines;
                targetGame.updatePendingDisplay();
                
                // 更新攻击统计
                fromGame.totalAttack += lines;
                fromGame.updateStats();
            }

            // 显示攻击弹道动画
            showAttackProjectile(fromPlayer, lines) {
                const fromBoard = document.getElementById(`board-p${fromPlayer}`);
                const toBoard = document.getElementById(`board-p${fromPlayer === 1 ? 2 : 1}`);
                
                const fromRect = fromBoard.getBoundingClientRect();
                const toRect = toBoard.getBoundingClientRect();
                
                const projectile = document.createElement('div');
                projectile.className = `attack-projectile p${fromPlayer}-to-p${fromPlayer === 1 ? 2 : 1}`;
                projectile.textContent = `+${lines}`;
                
                const startX = fromRect.left + fromRect.width / 2;
                const startY = fromRect.top + fromRect.height / 2;
                const targetX = toRect.left + toRect.width / 2;
                const targetY = toRect.top + toRect.height / 2;
                
                projectile.style.left = `${startX}px`;
                projectile.style.top = `${startY}px`;
                projectile.style.setProperty('--start-x', `${startX}px`);
                projectile.style.setProperty('--start-y', `${startY}px`);
                projectile.style.setProperty('--target-x', `${targetX}px`);
                projectile.style.setProperty('--target-y', `${targetY}px`);
                
                document.body.appendChild(projectile);
                
                setTimeout(() => {
                    projectile.remove();
                }, 500);
            }

            // 检查胜负
            checkGameOver(loser) {
                if (this.gameOver) return;
                
                this.gameOver = true;
                this.winner = loser === 1 ? 2 : 1;
                
                // 显示胜负弹窗
                this.showGameOverModal();
            }

            showGameOverModal() {
                const modal = document.getElementById('game-over-modal');
                const winnerText = document.getElementById('winner-text');
                
                winnerText.textContent = `PLAYER ${this.winner}`;
                winnerText.className = `winner-text p${this.winner}-wins`;
                
                modal.classList.remove('hidden');
            }

            reset() {
                this.gameOver = false;
                this.winner = null;
                
                const modal = document.getElementById('game-over-modal');
                modal.classList.add('hidden');
            }
        }

        // ============================================
        // 游戏主类
        // ============================================
        class TetrisGame {
            constructor(boardId, nextId, holdId, scoreId, levelId, linesId, comboId, attackId, playerNum, gameManager) {
                this.canvas = document.getElementById(boardId);
                this.ctx = this.canvas.getContext('2d');
                this.nextCanvas = document.getElementById(nextId);
                this.nextCtx = this.nextCanvas.getContext('2d');
                this.holdCanvas = document.getElementById(holdId);
                this.holdCtx = this.holdCanvas.getContext('2d');

                this.scoreEl = document.getElementById(scoreId);
                this.levelEl = document.getElementById(levelId);
                this.linesEl = document.getElementById(linesId);
                this.comboEl = document.getElementById(comboId);
                this.attackEl = document.getElementById(attackId);

                this.playerNum = playerNum;
                this.playerKey = playerNum === 1 ? 'p1' : 'p2';
                this.gameManager = gameManager;

                this.boardWrapper = this.canvas.parentElement;
                this.gameBoard = document.getElementById(boardId);
                this.pendingContainer = document.getElementById(`pending-p${playerNum}`);
                this.garbageWarning = document.getElementById(`garbage-warning-p${playerNum}`);

                this.init();
            }

            init() {
                this.board = this.createBoard();
                this.score = 0;
                this.level = 1;
                this.lines = 0;
                this.combo = 0;
                this.maxCombo = 0;
                this.totalAttack = 0;
                this.currentPiece = null;
                this.nextPieces = [];
                this.holdPiece = null;
                this.canHold = true;
                this.gameOver = false;
                this.lastDrop = 0;
                this.dropInterval = 800;
                this.bag = [];
                this.pendingGarbage = 0;
                this.lockDelay = 0;
                this.lockDelayMax = 500;
                this.lockMoves = 0;
                this.lockMovesMax = 15;

                this.nextPieces = [];
                for (let i = 0; i < PREVIEW_COUNT; i++) {
                    this.nextPieces.push(this.getNextFromBag());
                }

                this.spawnPiece();
                this.updateStats();
                this.updatePendingDisplay();
                this.draw();
            }

            createBoard() {
                return Array.from({ length: ROWS }, () => Array(COLS).fill(EMPTY));
            }

            getNextFromBag() {
                if (this.bag.length === 0) {
                    this.bag = ['I', 'O', 'T', 'S', 'Z', 'J', 'L'];
                    for (let i = this.bag.length - 1; i > 0; i--) {
                        const j = Math.floor(Math.random() * (i + 1));
                        [this.bag[i], this.bag[j]] = [this.bag[j], this.bag[i]];
                    }
                }
                return this.bag.pop();
            }

            spawnPiece() {
                // 先处理待处理的垃圾行
                if (this.pendingGarbage > 0) {
                    this.receiveGarbage();
                }

                const type = this.nextPieces.shift();
                this.nextPieces.push(this.getNextFromBag());
                this.currentPiece = new Tetromino(type);
                this.canHold = true;
                this.lockDelay = 0;
                this.lockMoves = 0;

                if (this.checkCollision(this.currentPiece.x, this.currentPiece.y, this.currentPiece.getCurrentShape())) {
                    this.gameOver = true;
                    this.gameManager.checkGameOver(this.playerNum);
                }
            }

            // 接收垃圾行
            receiveGarbage() {
                const lines = this.pendingGarbage;
                this.pendingGarbage = 0;
                this.updatePendingDisplay();

                // 将现有棋盘向上推
                for (let i = 0; i < lines; i++) {
                    // 移除最顶行
                    this.board.shift();
                    
                    // 在底部添加垃圾行
                    const garbageLine = this.createGarbageLine();
                    this.board.push(garbageLine);
                }

                // 闪烁效果
                this.gameBoard.classList.add('damage-flash');
                setTimeout(() => {
                    this.gameBoard.classList.remove('damage-flash');
                }, 300);

                // 更新垃圾行警告
                this.updateGarbageWarning();
            }

            // 创建垃圾行（带1格随机空洞）
            createGarbageLine() {
                const line = new Array(COLS).fill('G');
                const hole = Math.floor(Math.random() * COLS);
                line[hole] = EMPTY;
                return line;
            }

            // 更新待处理攻击显示
            updatePendingDisplay() {
                this.pendingContainer.innerHTML = '';
                for (let i = 0; i < this.pendingGarbage; i++) {
                    const bar = document.createElement('div');
                    bar.className = 'attack-bar';
                    this.pendingContainer.appendChild(bar);
                }
                this.updateGarbageWarning();
            }

            // 更新垃圾行警告
            updateGarbageWarning() {
                const warningHeight = Math.min(this.pendingGarbage * BLOCK_SIZE, ROWS * BLOCK_SIZE);
                this.garbageWarning.style.height = `${warningHeight}px`;
            }

            checkCollision(x, y, shape) {
                for (let row = 0; row < shape.length; row++) {
                    for (let col = 0; col < shape[row].length; col++) {
                        if (shape[row][col]) {
                            const newX = x + col;
                            const newY = y + row;
                            
                            if (newX < 0 || newX >= COLS || newY >= ROWS) {
                                return true;
                            }
                            
                            if (newY >= 0 && this.board[newY][newX] !== EMPTY) {
                                return true;
                            }
                        }
                    }
                }
                return false;
            }

            moveLeft() {
                if (!this.checkCollision(this.currentPiece.x - 1, this.currentPiece.y, this.currentPiece.getCurrentShape())) {
                    this.currentPiece.x--;
                    this.resetLockDelay();
                }
            }

            moveRight() {
                if (!this.checkCollision(this.currentPiece.x + 1, this.currentPiece.y, this.currentPiece.getCurrentShape())) {
                    this.currentPiece.x++;
                    this.resetLockDelay();
                }
            }

            moveDown() {
                if (!this.checkCollision(this.currentPiece.x, this.currentPiece.y + 1, this.currentPiece.getCurrentShape())) {
                    this.currentPiece.y++;
                    this.score += 1;
                    return true;
                }
                return false;
            }

            hardDrop() {
                let dropped = 0;
                while (!this.checkCollision(this.currentPiece.x, this.currentPiece.y + 1, this.currentPiece.getCurrentShape())) {
                    this.currentPiece.y++;
                    dropped++;
                }
                this.score += dropped * 2;
                this.lockPiece();
            }

            rotate(dir) {
                const type = this.currentPiece.type;
                const oldRotation = this.currentPiece.rotation;
                this.currentPiece.rotate(dir);
                const newRotation = this.currentPiece.rotation;
                
                const kickData = type === 'O' ? [[0,0]] : 
                    type === 'I' ? SRS_KICKS.I[`${oldRotation}>${newRotation}`] :
                    SRS_KICKS.JLSTZ[`${oldRotation}>${newRotation}`];
                
                for (const [kickX, kickY] of kickData) {
                    if (!this.checkCollision(this.currentPiece.x + kickX, this.currentPiece.y - kickY, this.currentPiece.getCurrentShape())) {
                        this.currentPiece.x += kickX;
                        this.currentPiece.y -= kickY;
                        this.resetLockDelay();
                        return true;
                    }
                }
                
                this.currentPiece.rotation = oldRotation;
                return false;
            }

            resetLockDelay() {
                if (this.lockMoves < this.lockMovesMax) {
                    this.lockDelay = 0;
                    this.lockMoves++;
                }
            }

            hold() {
                if (!this.canHold) return;
                
                const currentType = this.currentPiece.type;
                
                if (this.holdPiece) {
                    this.currentPiece = new Tetromino(this.holdPiece);
                } else {
                    this.spawnPiece();
                }
                
                this.holdPiece = currentType;
                this.canHold = false;
                this.lockDelay = 0;
                this.lockMoves = 0;
            }

            getGhostY() {
                let ghostY = this.currentPiece.y;
                while (!this.checkCollision(this.currentPiece.x, ghostY + 1, this.currentPiece.getCurrentShape())) {
                    ghostY++;
                }
                return ghostY;
            }

            lockPiece() {
                const shape = this.currentPiece.getCurrentShape();
                for (let row = 0; row < shape.length; row++) {
                    for (let col = 0; col < shape[row].length; col++) {
                        if (shape[row][col]) {
                            const y = this.currentPiece.y + row;
                            const x = this.currentPiece.x + col;
                            if (y >= 0 && y < ROWS && x >= 0 && x < COLS) {
                                this.board[y][x] = this.currentPiece.type;
                            }
                        }
                    }
                }
                
                const linesCleared = this.clearLines();
                this.calculateAttack(linesCleared);
                this.spawnPiece();
            }

            clearLines() {
                let linesCleared = 0;
                
                for (let row = ROWS - 1; row >= 0; row--) {
                    if (this.board[row].every(cell => cell !== EMPTY)) {
                        this.board.splice(row, 1);
                        this.board.unshift(Array(COLS).fill(EMPTY));
                        linesCleared++;
                        row++;
                    }
                }
                
                if (linesCleared > 0) {
                    this.lines += linesCleared;
                    
                    // 计分
                    const points = [0, 100, 300, 500, 800];
                    this.score += points[linesCleared] * this.level;
                    
                    // 升级
                    const newLevel = Math.floor(this.lines / 10) + 1;
                    if (newLevel > this.level) {
                        this.level = newLevel;
                        this.dropInterval = Math.max(100, 800 - (this.level - 1) * 50);
                    }
                    
                    // 增加连击
                    this.combo++;
                    this.maxCombo = Math.max(this.maxCombo, this.combo);
                } else {
                    // 重置连击
                    this.combo = 0;
                }
                
                this.updateStats();
                return linesCleared;
            }

            // 计算并发送攻击
            calculateAttack(linesCleared) {
                if (linesCleared === 0) return;
                
                // 基础攻击
                let attackLines = ATTACK_TABLE[linesCleared] || 0;
                
                // 连击加成
                if (this.combo > 1) {
                    const comboBonus = COMBO_BONUS[Math.min(this.combo, COMBO_BONUS.length - 1)];
                    attackLines += comboBonus;
                    
                    // 显示连击提示
                    this.showComboPopup();
                }
                
                // 发送攻击
                if (attackLines > 0) {
                    this.gameManager.sendAttack(this.playerNum, attackLines);
                }
            }

            showComboPopup() {
                if (this.combo < 2) return;
                
                const popup = document.createElement('div');
                popup.className = 'combo-popup';
                popup.textContent = `${this.combo} COMBO!`;
                this.boardWrapper.appendChild(popup);
                
                setTimeout(() => popup.remove(), 800);
            }

            updateStats() {
                this.scoreEl.textContent = this.score.toLocaleString();
                this.levelEl.textContent = this.level;
                this.linesEl.textContent = this.lines;
                this.comboEl.textContent = this.combo;
                this.attackEl.textContent = this.totalAttack;
            }

            update(now, input) {
                if (this.gameOver || this.gameManager.gameOver) return;

                // 处理输入
                if (input.isJustPressed(this.playerKey, 'rotateCW')) {
                    this.rotate(1);
                }
                if (input.isJustPressed(this.playerKey, 'rotateCCW')) {
                    this.rotate(-1);
                }
                if (input.isJustPressed(this.playerKey, 'hardDrop')) {
                    this.hardDrop();
                }
                if (input.isJustPressed(this.playerKey, 'hold')) {
                    this.hold();
                }
                
                if (input.getDasAction(this.playerKey, 'left', now)) {
                    this.moveLeft();
                }
                if (input.getDasAction(this.playerKey, 'right', now)) {
                    this.moveRight();
                }
                if (input.getDasAction(this.playerKey, 'down', now, true)) {
                    this.moveDown();
                }

                // 自动下落
                if (now - this.lastDrop > this.dropInterval) {
                    if (!this.moveDown()) {
                        // 锁定延迟
                        this.lockDelay += this.dropInterval;
                        if (this.lockDelay >= this.lockDelayMax) {
                            this.lockPiece();
                        }
                    }
                    this.lastDrop = now;
                }

                this.draw();
            }

            draw() {
                const ctx = this.ctx;
                ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);

                // 绘制网格背景
                ctx.fillStyle = COLORS.grid;
                ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);
                
                ctx.strokeStyle = COLORS.gridLine;
                ctx.lineWidth = 1;
                for (let x = 0; x <= COLS; x++) {
                    ctx.beginPath();
                    ctx.moveTo(x * BLOCK_SIZE, 0);
                    ctx.lineTo(x * BLOCK_SIZE, ROWS * BLOCK_SIZE);
                    ctx.stroke();
                }
                for (let y = 0; y <= ROWS; y++) {
                    ctx.beginPath();
                    ctx.moveTo(0, y * BLOCK_SIZE);
                    ctx.lineTo(COLS * BLOCK_SIZE, y * BLOCK_SIZE);
                    ctx.stroke();
                }

                // 绘制已锁定方块
                for (let row = 0; row < ROWS; row++) {
                    for (let col = 0; col < COLS; col++) {
                        const cell = this.board[row][col];
                        if (cell !== EMPTY) {
                            const color = cell === 'G' ? COLORS.G : TETROMINOES[cell].color;
                            this.drawBlock(ctx, col, row, color, cell === 'G');
                        }
                    }
                }

                // 绘制幽灵方块
                if (this.currentPiece) {
                    const ghostY = this.getGhostY();
                    const shape = this.currentPiece.getCurrentShape();
                    
                    ctx.globalAlpha = 0.3;
                    for (let row = 0; row < shape.length; row++) {
                        for (let col = 0; col < shape[row].length; col++) {
                            if (shape[row][col]) {
                                this.drawBlock(ctx, this.currentPiece.x + col, ghostY + row, this.currentPiece.color, false, true);
                            }
                        }
                    }
                    ctx.globalAlpha = 1;

                    // 绘制当前方块
                    for (let row = 0; row < shape.length; row++) {
                        for (let col = 0; col < shape[row].length; col++) {
                            if (shape[row][col]) {
                                this.drawBlock(ctx, this.currentPiece.x + col, this.currentPiece.y + row, this.currentPiece.color);
                            }
                        }
                    }
                }

                // 绘制预览方块
                this.drawNext();
                this.drawHold();
            }

            drawBlock(ctx, x, y, color, isGarbage = false, isGhost = false) {
                const bx = x * BLOCK_SIZE;
                const by = y * BLOCK_SIZE;
                const size = BLOCK_SIZE - 1;

                if (isGhost) {
                    ctx.strokeStyle = color.main;
                    ctx.lineWidth = 2;
                    ctx.strokeRect(bx + 1, by + 1, size - 1, size - 1);
                    return;
                }

                // 发光效果
                ctx.shadowColor = color.glow;
                ctx.shadowBlur = isGarbage ? 0 : 8;
                
                // 主体
                ctx.fillStyle = color.main;
                ctx.fillRect(bx, by, size, size);
                
                // 高光
                ctx.fillStyle = 'rgba(255, 255, 255, 0.3)';
                ctx.fillRect(bx, by, size, 3);
                ctx.fillRect(bx, by, 3, size);
                
                // 暗边
                ctx.fillStyle = color.dark;
                ctx.fillRect(bx + size - 3, by, 3, size);
                ctx.fillRect(bx, by + size - 3, size, 3);
                
                ctx.shadowBlur = 0;
            }

            drawNext() {
                const ctx = this.nextCtx;
                ctx.clearRect(0, 0, this.nextCanvas.width, this.nextCanvas.height);
                
                let offsetY = 5;
                for (let i = 0; i < this.nextPieces.length; i++) {
                    const type = this.nextPieces[i];
                    const shape = TETROMINOES[type].shape[0];
                    const color = TETROMINOES[type].color;
                    
                    const pieceWidth = shape[0].length * 14;
                    const pieceHeight = shape.length * 14;
                    const offsetX = (this.nextCanvas.width - pieceWidth) / 2;
                    
                    for (let row = 0; row < shape.length; row++) {
                        for (let col = 0; col < shape[row].length; col++) {
                            if (shape[row][col]) {
                                const bx = offsetX + col * 14;
                                const by = offsetY + row * 14;
                                
                                ctx.shadowColor = color.glow;
                                ctx.shadowBlur = 4;
                                ctx.fillStyle = color.main;
                                ctx.fillRect(bx, by, 13, 13);
                                ctx.shadowBlur = 0;
                            }
                        }
                    }
                    
                    offsetY += pieceHeight + 12;
                }
            }

            drawHold() {
                const ctx = this.holdCtx;
                ctx.clearRect(0, 0, this.holdCanvas.width, this.holdCanvas.height);
                
                if (!this.holdPiece) return;
                
                const type = this.holdPiece;
                const shape = TETROMINOES[type].shape[0];
                const color = TETROMINOES[type].color;
                
                const pieceWidth = shape[0].length * 14;
                const pieceHeight = shape.length * 14;
                const offsetX = (this.holdCanvas.width - pieceWidth) / 2;
                const offsetY = (this.holdCanvas.height - pieceHeight) / 2;
                
                ctx.globalAlpha = this.canHold ? 1 : 0.4;
                
                for (let row = 0; row < shape.length; row++) {
                    for (let col = 0; col < shape[row].length; col++) {
                        if (shape[row][col]) {
                            const bx = offsetX + col * 14;
                            const by = offsetY + row * 14;
                            
                            ctx.shadowColor = color.glow;
                            ctx.shadowBlur = 4;
                            ctx.fillStyle = color.main;
                            ctx.fillRect(bx, by, 13, 13);
                            ctx.shadowBlur = 0;
                        }
                    }
                }
                
                ctx.globalAlpha = 1;
            }
        }

        const game2 = new TetrisGame('canvas-p2', 'next-p2', 'hold-p2', 'score-p2', 'level-p2', 'lines-p2', 'combo-p2', 'attack-p2', 2, gameManager);

## 实现说明

### 核心对战功能

1. **攻击系统**：
   - Single (1行) = 0行攻击
   - Double (2行) = 1行攻击
   - Triple (3行) = 2行攻击
   - Tetris (4行) = 4行攻击

2. **连击系统**：
   - 连续消行会累积连击数
   - 连击提供额外攻击加成（2连击+1行，3连击+1行，4连击+2行...）
   - 未消行会重置连击
   - 连击时显示 COMBO 弹出提示

3. **垃圾行机制**：
   - 待接收的垃圾行显示在场地右侧的红色条带中
   - 底部有渐变警告条显示即将到来的垃圾量
   - 垃圾行在下一个方块生成前推入场地下方
   - 每行垃圾有1格随机空洞

4. **攻击弹道动画**：
   - 发送攻击时显示飞行弹道
   - 从攻击方飞向受击方
   - 显示攻击行数

5. **胜负判定**：
   - 方块生成时检测是否与现有方块重叠
   - 触顶则判定该玩家失败
   - 显示胜负弹窗，可点击 REMATCH 重新开始
