# 游戏界面与基础渲染

```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>Backpack War Game</title>
<style>
  body {
    margin: 0;
    overflow: hidden;
  }
  canvas {
    display: block;
  }
</style>
</head>
<body>
<canvas id="gameCanvas"></canvas>
<script>
  const canvas = document.getElementById('gameCanvas');
  const ctx = canvas.getContext('2d');
  canvas.width = 800;
  canvas.height = 600;

  let gameLoop;

  // Game objects
  const player = {
    x: 50,
    y: 50,
    width: 30,
    height: 30,
    color: 'blue'
  };

  const backpack = {
    x: 50,
    y: 100,
    width: 100,
    height: 50,
    color: 'green'
  };

  const map = {
    width: canvas.width,
    height: canvas.height,
    color: 'grey'
  };

  // Draw functions
  function drawRect(x, y, width, height, color) {
    ctx.fillStyle = color;
    ctx.fillRect(x, y, width, height);
  }

  function drawPlayer() {
    drawRect(player.x, player.y, player.width, player.height, player.color);
  }

  function drawBackpack() {
    drawRect(backpack.x, backpack.y, backpack.width, backpack.height, backpack.color);
  }

  function drawMap() {
    drawRect(map.x, map.y, map.width, map.height, map.color);
  }

  function gameLoop() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    drawMap();
    drawPlayer();
    drawBackpack();
  }

  // Event listeners
  window.addEventListener('keydown', (e) => {
    if (e.key === 'ArrowLeft') {
      player.x -= 5;
    } else if (e.key === 'ArrowRight') {
      player.x += 5;
    } else if (e.key === 'ArrowUp') {
      player.y -= 5;
    } else if (e.key === 'ArrowDown') {
      player.y += 5;
    }
  });

  // Initialization
  window.onload = () => {
    gameLoop = () => {
      requestAnimationFrame(gameLoop);
    };
    requestAnimationFrame(gameLoop);
  };
</script>
</body>
</html>
```

这段代码创建了一个简单的Web游戏界面，使用Canvas API绘制了一个地图、一个玩家角色和一个背包。游戏循环通过`requestAnimationFrame`实现，并且通过键盘事件监听器控制玩家的移动。这个HTML文件是一个自包含的单文件，可以直接在浏览器中打开运行。
