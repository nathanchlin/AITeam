```html
<style>
  body {
    margin: 0;
    overflow: hidden;
  }

  }
  #inventory {
    position: absolute;
    top: 10px;

    background-color: rgba(0, 0, 0, 0.7);
    color: white;
    padding: 5px;
    border-radius: 5px;
    pointer-events: none;
    display: none;
  }
  .item {
    margin: 5px;
    padding: 5px;
    border-radius: 5px;
    cursor: pointer;
  }
  .item.selected {
    background-color: rgba(255, 255, 0, 0.5);
  }
</style>
</head>

<canvas id="gameCanvas"></canvas>
<div id="inventory"></div>
<script>
  const canvas = document.getElementById('gameCanvas');
  const ctx = canvas.getContext('2d');
  canvas.width = 800;
  canvas.height = 600;

  let inventory = [];
  let selectedItemIndex = -1;
  let playerLevel = 1;
  let playerExperience = 0;
  const experienceToNextLevel = 100;

  const player = {
    x: 50,
    y: 50,
    width: 30,
    height: 30,
    color: 'blue',
    strength: 1,
    defense: 1
  };

  const backpack = {
    x: 50,
    y: 100,
    width: 100,
    height: 50,
    color: 'green'
  };

  const ground = {
    x: 0,
    y: canvas.height - 20,
    width: canvas.width,
    height: 20,
    color: 'grey'
  };

  // Draw functions
  function drawRect(x, y, width, height, color) {
    ctx.fillStyle = color;
    ctx.fillRect(x, y, width, height);
  }

  function drawPlayer() {
    drawRect(player.x, player.y, player.width, player.height, player.color);
  }

  function drawBackpack() {
    drawRect(backpack.x, backpack.y, backpack.width, backpack.height, backpack.color);
  }

  function drawGround() {
    drawRect(ground.x, ground.y, ground.width, ground.height, ground.color);
  }

  function drawInventory() {
    const inventoryDiv = document.getElementById('inventory');
    inventoryDiv.innerHTML = ''; // Clear inventory display
    inventory.forEach((item, index) => {
      const itemDiv = document.createElement('div');
      itemDiv.className = 'item';
      itemDiv.style.backgroundColor = item.color;
      itemDiv.innerText = item.name;
      itemDiv.dataset.index = index;
      itemDiv.addEventListener('click', () => {
        selectedItemIndex = index;
        inventory.forEach((i, idx) => {
          itemDiv.classList.toggle('selected', idx === selectedItemIndex);
        });
      });
      inventoryDiv.appendChild(itemDiv);
    });
    inventoryDiv.style.display = 'block';
  }

  function drawLevelUp() {
    ctx.fillStyle = 'black';
    ctx.font = '20px Arial';
    ctx.fillText(`Level Up! Level ${playerLevel}`, 10, 60);
  }

  function updatePlayerExperience() {
    if (playerExperience >= experienceToNextLevel) {
      playerLevel++;
      playerExperience -= experienceToNextLevel;
      player.strength += 1;
      player.defense += 1;
      drawLevelUp();
    }
  }

  function gameLoop() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    drawGround();
    drawPlayer();
    drawBackpack();
    drawInventory();
    updatePlayerExperience();
  }

  // Event listeners
  window.addEventListener('keydown', (e) => {
    if (e.key === 'i') {
      inventory = [{ name: 'Sword', color: 'red' }, { name: 'Shield', color: 'blue' }]; // Example items
      drawInventory();
    }
  });

  window.addEventListener('keydown', (e) => {
    if (e.key === 'e') {
      if (selectedItemIndex !== -1) {
        const selectedItem = inventory[selectedItemIndex];
        if (selectedItem.name === 'Sword') {
          player.strength += 2;
        } else if (selectedItem.name === 'Shield') {
          player.defense += 2;
        }
        inventory.splice(selectedItemIndex, 1);
        drawInventory();
      }
    }
  });

  window.addEventListener('keydown', (e) => {
    if (e.key === ' ') {
      playerExperience += 10; // Gain experience
    }
  });

  // Initialization
  window.onload = () => {
    let gameLoop = () => {
      requestAnimationFrame(gameLoop);
    };
    requestAnimationFrame(gameLoop);
  };
</script>
</body>
</html>
```

这个代码实现了一个简单的背包战争游戏，其中包括一个角色成长系统。玩家可以通过按空格键获得经验值，达到一定经验值后可以升级，增加角色的力量和防御力。背包系统允许玩家存储和装备物品，例如剑和盾牌。游戏界面包含一个简单的物品栏，玩家可以通过点击物品来装备它们。
