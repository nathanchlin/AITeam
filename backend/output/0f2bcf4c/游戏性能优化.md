                this.x = Math.floor((COLS - shape[0].length) / 2);

            rotate() {

                for (let i = 0; i < cols; i++) {

                    for (let x = 0; x < shape[y].length; x++) {

                                return false;
                            }

            getPossibleMoves(board, piece) {

                const rotations = this.getRotations(piece);

                            y++;
                        }

                                x: x,

                            });
                        }
                    }
                }

            // 获取所有可能的旋转
            getRotations(piece) {
                const rotations = [piece.shape];
                let current = piece.shape;
                
                // 生成所有旋转状态（最多4种）
                for (let i = 0; i < 3; i++) {
                    current = this.rotateMatrix(current);
                    rotations.push(current);
                }
                
                return rotations;

            rotateMatrix(matrix) {

            // 模拟移动并返回新的游戏板状态
            simulateMove(board, piece, x, y, shape) {
                const newBoard = board.map(row => [...row]);
                
                for (let py = 0; py < shape.length; py++) {
                    for (let px = 0; px < shape[py].length; px++) {
                        if (shape[py][px]) {
                            const boardY = y + py;
                            const boardX = x + px;
                            
                            if (boardY >= 0 && boardY < ROWS && boardX >= 0 && boardX < COLS) {
                                newBoard[boardY][boardX] = piece.color;
                            }
                        }
                    }
                }
                
                return newBoard;
            }
            
            // 评估游戏板状态
            evaluateBoard(board) {
                // 使用缓存提高性能
                const boardKey = JSON.stringify(board);
                if (this.cache.has(boardKey)) {
                    return this.cache.get(boardKey);
                }
                
                const height = this.calculateHeight(board);
                const holes = this.calculateHoles(board);
                const bumpiness = this.calculateBumpiness(board);
                const lines = this.calculateLines(board);
                
                // 计算加权分数
                const score = 
                    this.weights.height * height +
                    this.weights.holes * holes +
                    this.weights.bumpiness * bumpiness +
                    this.weights.lines * lines;
                
                // 限制缓存大小
                if (this.cache.size >= this.maxCacheSize) {
                    // 删除最旧的缓存项
                    const firstKey = this.cache.keys().next().value;
                    this.cache.delete(firstKey);
                }
                
                this.cache.set(boardKey, score);
                return score;
            }
            
            // 计算最高列高度
            calculateHeight(board) {
                let height = 0;
                for (let x = 0; x < COLS; x++) {
                    let columnHeight = 0;
                    for (let y = ROWS - 1; y >= 0; y--) {
                        if (board[y][x]) {
                            columnHeight = ROWS - y;
                            break;
                        }
                    }
                    height += columnHeight;
                }
                return height;
            }
            
            // 计算孔洞数量
            calculateHoles(board) {
                let holes = 0;
                for (let x = 0; x < COLS; x++) {
                    let blockFound = false;
                    for (let y = ROWS - 1; y >= 0; y--) {
                        if (board[y][x]) {
                            blockFound = true;
                        } else if (blockFound) {
                            holes++;
                        }
                    }
                }
                return holes;
            }
            
            // 计算表面不平整度
            calculateBumpiness(board) {
                let bumpiness = 0;
                const heights = [];
                
                // 计算每列的高度
                for (let x = 0; x < COLS; x++) {
                    let height = 0;
                    for (let y = ROWS - 1; y >= 0; y--) {
                        if (board[y][x]) {
                            height = ROWS - y;
                            break;
                        }
                    }
                    heights.push(height);
                }
                
                // 计算相邻列高度差
                for (let i = 0; i < heights.length - 1; i++) {
                    bumpiness += Math.abs(heights[i] - heights[i + 1]);
                }
                
                return bumpiness;
            }
            
            // 计算可消除的行数
            calculateLines(board) {
                let lines = 0;
                for (let y = 0; y < ROWS; y++) {
                    if (board[y].every(cell => cell !== 0)) {
                        lines++;
                    }
                }
                return lines;
            }

            findBestMove(board, piece) {
                const startTime = performance.now();
                
                const moves = this.getPossibleMoves(board, piece);
                let bestMove = null;
                let bestScore = -Infinity;
                
                // 评估每个可能的移动
                for (const move of moves) {
                    const score = this.evaluateBoard(move.board);
                    
                    if (score > bestScore) {
                        bestScore = score;
                        bestMove = move;
                    }
                }
                
                // 记录AI计算时间
                aiCalculationTime = performance.now() - startTime;
                
                return bestMove;
            }
        }
