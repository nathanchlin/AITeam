    <script>
        // 游戏常量
        const COLS = 10;
        const ROWS = 20;
        const BLOCK_SIZE = 30;
        const INITIAL_SPEED = 500; // 初始下落速度(毫秒)
        const SPEED_INCREMENT = 50; // 每升一级速度减少的毫秒数
        const LINE_SCORE = 100; // 消除一行的基础分数
        const MULTIPLIER = [0, 100, 300, 500, 800]; // 消除多行的分数倍数
        
        // 方块形状定义
        const SHAPES = [
            // I
            [[1, 1, 1, 1]],
            // O
            [[1, 1],
             [1, 1]],
            // T
            [[0, 1, 0],
             [1, 1, 1]],
            // S
            [[0, 1, 1],
             [1, 1, 0]],
            // Z
            [[1, 1, 0],
             [0, 1, 1]],
            // J
            [[1, 0, 0],
             [1, 1, 1]],
            // L
            [[0, 0, 1],
             [1, 1, 1]]
        ];
        
        // 方块颜色
        const COLORS = [
            '#00f0f0', // I - 青色
            '#f0f000', // O - 黄色
            '#a000f0', // T - 紫色
            '#00f000', // S - 绿色
            '#f00000', // Z - 红色
            '#0000f0', // J - 蓝色
            '#f0a000'  // L - 橙色
        ];
        
        // 游戏状态
        class GameState {
            constructor(canvasId, nextCanvasId, isAI = false) {
                this.canvas = document.getElementById(canvasId);
                this.ctx = this.canvas.getContext('2d');
                this.nextCanvas = document.getElementById(nextCanvasId);
                this.nextCtx = this.nextCanvas.getContext('2d');
                this.isAI = isAI;
                
                this.board = this.createBoard();
                this.score = 0;
                this.level = 1;
                this.lines = 0;
                this.gameOver = false;
                this.currentPiece = null;
                this.nextPiece = null;
                this.dropInterval = INITIAL_SPEED;
                this.lastDropTime = 0;
                
                this.scoreElement = document.getElementById(`score${canvasId.slice(-1)}`);
                this.levelElement = document.getElementById(`level${canvasId.slice(-1)}`);
                this.linesElement = document.getElementById(`lines${canvasId.slice(-1)}`);
                
                this.generateNextPiece();
                this.spawnPiece();
            }
            
            createBoard() {
                return Array(ROWS).fill().map(() => Array(COLS).fill(0));
            }
            
            generateNextPiece() {
                const typeId = Math.floor(Math.random() * SHAPES.length);
                this.nextPiece = {
                    shape: SHAPES[typeId],
                    color: COLORS[typeId],
                    x: Math.floor((COLS - SHAPES[typeId][0].length) / 2),
                    y: 0,
                    typeId: typeId
                };
            }
            
            spawnPiece() {
                this.currentPiece = this.nextPiece;
                this.generateNextPiece();
                
                // 检查游戏是否结束
                if (this.collision(this.currentPiece)) {
                    this.gameOver = true;
                    return false;
                }
                
                return true;
            }
            
            collision(piece, board = this.board) {
                for (let y = 0; y < piece.shape.length; y++) {
                    for (let x = 0; x < piece.shape[y].length; x++) {
                        if (piece.shape[y][x]) {
                            const boardX = piece.x + x;
                            const boardY = piece.y + y;
                            
                            if (boardX < 0 || boardX >= COLS || boardY >= ROWS) {
                                return true;
                            }
                            
                            if (boardY >= 0 && board[boardY][boardX]) {
                                return true;
                            }
                        }
                    }
                }
                return false;
            }
            
            merge() {
                for (let y = 0; y < this.currentPiece.shape.length; y++) {
                    for (let x = 0; x < this.currentPiece.shape[y].length; x++) {
                        if (this.currentPiece.shape[y][x]) {
                            const boardY = this.currentPiece.y + y;
                            const boardX = this.currentPiece.x + x;
                            
                            if (boardY >= 0) {
                                this.board[boardY][boardX] = this.currentPiece.typeId + 1;
                            }
                        }
                    }
                }
            }
            
            clearLines() {
                let linesCleared = 0;
                
                for (let y = ROWS - 1; y >= 0; y--) {
                    if (this.board[y].every(cell => cell !== 0)) {
                        this.board.splice(y, 1);
                        this.board.unshift(Array(COLS).fill(0));
                        linesCleared++;
                        y++; // 重新检查当前行
                    }
                }
                
                if (linesCleared > 0) {
                    // 更新分数
                    this.lines += linesCleared;
                    this.score += MULTIPLIER[linesCleared] * this.level;
                    
                    // 更新等级
                    this.level = Math.floor(this.lines / 10) + 1;
                    this.dropInterval = Math.max(50, INITIAL_SPEED - (this.level - 1) * SPEED_INCREMENT);
                    
                    // 更新UI
                    this.scoreElement.textContent = this.score;
                    this.levelElement.textContent = this.level;
                    this.linesElement.textContent = this.lines;
                }
                
                return linesCleared;
            }
            
            move(dir) {
                const newPiece = {
                    ...this.currentPiece,
                    x: this.currentPiece.x + dir
                };
                
                if (!this.collision(newPiece)) {
                    this.currentPiece.x = newPiece.x;
                    return true;
                }
                return false;
            }
            
            rotate() {
                const rotated = this.rotateMatrix(this.currentPiece.shape);
                const originalShape = this.currentPiece.shape;
                
                this.currentPiece.shape = rotated;
                
                // 尝试旋转墙踢
                const kicks = [
                    [0, 0], // 无偏移
                    [-1, 0], // 左
                    [1, 0], // 右
                    [0, -1], // 上
                    [-1, -1], // 左上
                    [1, -1] // 右上
                ];
                
                for (const [kickX, kickY] of kicks) {
                    const newPiece = {
                        ...this.currentPiece,
                        x: this.currentPiece.x + kickX,
                        y: this.currentPiece.y + kickY
                    };
                    
                    if (!this.collision(newPiece)) {
                        this.currentPiece.x = newPiece.x;
                        this.currentPiece.y = newPiece.y;
                        return true;
                    }
                }
                
                // 所有旋转尝试都失败，恢复原始形状
                this.currentPiece.shape = originalShape;
                return false;
            }
            
            rotateMatrix(matrix) {
                const rows = matrix.length;
                const cols = matrix[0].length;
                const rotated = Array(cols).fill().map(() => Array(rows).fill(0));
                
                for (let y = 0; y < rows; y++) {
                    for (let x = 0; x < cols; x++) {
                        rotated[x][rows - 1 - y] = matrix[y][x];
                    }
                }
                
                return rotated;
            }
            
            drop() {
                const newPiece = {
                    ...this.currentPiece,
                    y: this.currentPiece.y + 1
                };
                
                if (!this.collision(newPiece)) {
                    this.currentPiece.y = newPiece.y;
                    return true;
                }
                return false;
            }
            
            hardDrop() {
                let dropped = 0;
                while (this.drop()) {
                    dropped++;
                }
                this.score += dropped * 2;
                this.scoreElement.textContent = this.score;
            }
            
            update(timestamp) {
                if (this.gameOver) return;
                
                if (timestamp - this.lastDropTime > this.dropInterval) {
                    if (!this.drop()) {
                        this.merge();
                        this.clearLines();
                        if (!this.spawnPiece()) {
                            this.gameOver = true;
                        }
                    }
                    this.lastDropTime = timestamp;
                }
            }
            
            draw() {
                // 清空画布
                this.ctx.fillStyle = '#0f3460';
                this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);
                
                // 绘制网格线
                this.ctx.strokeStyle = '#1a4d7a';
                this.ctx.lineWidth = 0.5;
                
                for (let i = 0; i <= COLS; i++) {
                    this.ctx.beginPath();
                    this.ctx.moveTo(i * BLOCK_SIZE, 0);
                    this.ctx.lineTo(i * BLOCK_SIZE, this.canvas.height);
                    this.ctx.stroke();
                }
                
                for (let i = 0; i <= ROWS; i++) {
                    this.ctx.beginPath();
                    this.ctx.moveTo(0, i * BLOCK_SIZE);
                    this.ctx.lineTo(this.canvas.width, i * BLOCK_SIZE);
                    this.ctx.stroke();
                }
                
                // 绘制已固定的方块
                for (let y = 0; y < ROWS; y++) {
                    for (let x = 0; x < COLS; x++) {
                        if (this.board[y][x]) {
                            this.drawBlock(x, y, COLORS[this.board[y][x] - 1]);
                        }
                    }
                }
                
                // 绘制当前方块
                if (this.currentPiece) {
                    for (let y = 0; y < this.currentPiece.shape.length; y++) {
                        for (let x = 0; x < this.currentPiece.shape[y].length; x++) {
                            if (this.currentPiece.shape[y][x]) {
                                this.drawBlock(
                                    this.currentPiece.x + x,
                                    this.currentPiece.y + y,
                                    this.currentPiece.color
                                );
                            }
                        }
                    }
                }
                
                // 绘制下一个方块
                this.nextCtx.fillStyle = '#0f3460';
                this.nextCtx.fillRect(0, 0, this.nextCanvas.width, this.nextCanvas.height);
                
                if (this.nextPiece) {
                    const offsetX = (this.nextCanvas.width - this.nextPiece.shape[0].length * BLOCK_SIZE / 2) / 2;
                    const offsetY = (this.nextCanvas.height - this.nextPiece.shape.length * BLOCK_SIZE / 2) / 2;
                    const blockSize = BLOCK_SIZE / 2;
                    
                    for (let y = 0; y < this.nextPiece.shape.length; y++) {
                        for (let x = 0; x < this.nextPiece.shape[y].length; x++) {
                            if (this.nextPiece.shape[y][x]) {
                                this.nextCtx.fillStyle = this.nextPiece.color;
                                this.nextCtx.fillRect(
                                    offsetX + x * blockSize,
                                    offsetY + y * blockSize,
                                    blockSize - 1,
                                    blockSize - 1
                                );
                            }
                        }
                    }
                }
            }
            
            drawBlock(x, y, color) {
                this.ctx.fillStyle = color;
                this.ctx.fillRect(x * BLOCK_SIZE, y * BLOCK_SIZE, BLOCK_SIZE - 1, BLOCK_SIZE - 1);
                
                // 添加高光效果
                this.ctx.fillStyle = 'rgba(255, 255, 255, 0.3)';
                this.ctx.fillRect(x * BLOCK_SIZE, y * BLOCK_SIZE, BLOCK_SIZE - 1, 4);
                this.ctx.fillRect(x * BLOCK_SIZE, y * BLOCK_SIZE, 4, BLOCK_SIZE - 1);
                
                // 添加阴影效果
                this.ctx.fillStyle = 'rgba(0, 0, 0, 0.3)';
                this.ctx.fillRect(x * BLOCK_SIZE + BLOCK_SIZE - 5, y * BLOCK_SIZE, 4, BLOCK_SIZE - 1);
                this.ctx.fillRect(x * BLOCK_SIZE, y * BLOCK_SIZE + BLOCK_SIZE - 5, BLOCK_SIZE - 1, 4);
            }
        }
        
        // AI 玩家类
        class AIPlayer extends GameState {
            constructor(canvasId, nextCanvasId) {
                super(canvasId, nextCanvasId, true);
                this.moveQueue = [];
                this.thinkInterval = 100; // AI思考间隔(毫秒)
                this.lastThinkTime = 0;
            }
            
            think(timestamp) {
                if (this.gameOver || timestamp - this.lastThinkTime < this.thinkInterval) return;
                this.lastThinkTime = timestamp;
