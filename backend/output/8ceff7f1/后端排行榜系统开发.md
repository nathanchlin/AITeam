### 3. 数据模型 (models/Score.js)

```javascript
const mongoose = require('mongoose');

const ScoreSchema = new mongoose.Schema({
  username: {
    type: String,
    required: true,
    trim: true,
    maxlength: 50
  },
  score: {
    type: Number,
    required: true,
    min: 0
  },
  level: {
    type: Number,
    required: true,
    min: 1
  },
  timeSpent: {
    type: Number, // 单位：秒
    required: true,
    min: 0
  },
  completedAt: {
    type: Date,
    default: Date.now
  },
  avatar: {
    type: String,
    default: 'default'
  }
}, {
  timestamps: true
});

// 创建索引以提高查询性能
ScoreSchema.index({ score: -1, completedAt: -1 });
ScoreSchema.index({ username: 1 });

module.exports = mongoose.model('Score', ScoreSchema);
```

### 4. API 路由 (routes/scores.js)

```javascript
const express = require('express');
const router = express.Router();
const Score = require('../models/Score');

// 获取排行榜
router.get('/', async (req, res) => {
  try {
    const { limit = 10, page = 1 } = req.query;
    const skip = (page - 1) * limit;
    
    const leaderboard = await Score.find()
      .sort({ score: -1, completedAt: -1 })
      .skip(skip)
      .limit(parseInt(limit));
    
    const total = await Score.countDocuments();
    
    res.json({
      leaderboard,
      pagination: {
        total,
        page: parseInt(page),
        limit: parseInt(limit),
        totalPages: Math.ceil(total / limit)
      }
    });
  } catch (err) {
    res.status(500).json({ message: 'Server error' });
  }
});

// 提交分数
router.post('/', async (req, res) => {
  try {
    const { username, score, level, timeSpent, avatar } = req.body;
    
    // 基本验证
    if (!username || !score || !level || !timeSpent) {
      return res.status(400).json({ message: 'Missing required fields' });
    }
    
    const newScore = new Score({
      username,
      score,
      level,
      timeSpent,
      avatar: avatar || 'default'
    });
    
    await newScore.save();
    
    // 获取更新后的排名
    const rank = await Score.countDocuments({ score: { $gt: score } }) + 1;
    
    res.status(201).json({
      message: 'Score submitted successfully',
      score: newScore,
      rank
    });
  } catch (err) {
    res.status(500).json({ message: 'Server error' });
  }
});

// 获取用户排名
router.get('/user/:username', async (req, res) => {
  try {
    const { username } = req.params;
    
    // 获取用户所有记录的最佳成绩
    const userBestScore = await Score.findOne({ username })
      .sort({ score: -1, completedAt: -1 });
    
    if (!userBestScore) {
      return res.status(404).json({ message: 'User not found' });
    }
    
    // 获取用户排名
    const rank = await Score.countDocuments({ score: { $gt: userBestScore.score } }) + 1;
    
    // 获取用户所有记录
    const userScores = await Score.find({ username })
      .sort({ score: -1, completedAt: -1 });
    
    res.json({
      bestScore: userBestScore,
      rank,
      allScores: userScores
    });
  } catch (err) {
    res.status(500).json({ message: 'Server error' });
  }
});

module.exports = router;
```

### 5. 完整的服务器配置 (server.js)

```javascript
require('dotenv').config();
const express = require('express');
const mongoose = require('mongoose');
const cors = require('cors');
const scoreRoutes = require('./routes/scores');

const app = express();
const PORT = process.env.PORT || 5000;

// 中间件
app.use(cors());
app.use(express.json());

// 数据库连接
mongoose.connect(process.env.MONGODB_URI || 'mongodb://localhost:27017/sheep-sheep', {
  useNewUrlParser: true,
  useUnifiedTopology: true,
})
.then(() => console.log('MongoDB connected'))
.catch(err => console.error('MongoDB connection error:', err));

// 路由
app.get('/', (req, res) => {
  res.send('Sheep Sheep Leaderboard API');
});

app.use('/api/scores', scoreRoutes);

// 错误处理中间件
app.use((err, req, res, next) => {
  console.error(err.stack);
  res.status(500).json({ message: 'Something went wrong!' });
});

app.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
});
```

### 3. 前端API调用示例 (React)

```javascript
import axios from 'axios';

// 获取排行榜
export const getLeaderboard = async (page = 1, limit = 10) => {
  try {
    const response = await axios.get(`/api/scores?page=${page}&limit=${limit}`);
    return response.data;
  } catch (error) {
    console.error('Error fetching leaderboard:', error);
    throw error;
  }
};

// 提交分数
export const submitScore = async (scoreData) => {
  try {
    const response = await axios.post('/api/scores', scoreData);
    return response.data;
  } catch (error) {
    console.error('Error submitting score:', error);
    throw error;
  }
};

// 获取用户排名
export const getUserRank = async (username) => {
  try {
    const response = await axios.get(`/api/scores/user/${username}`);
    return response.data;
  } catch (error) {
    console.error('Error fetching user rank:', error);
    throw error;
  }
};
```
