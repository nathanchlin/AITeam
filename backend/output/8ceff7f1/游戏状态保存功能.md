## 4. 存储功能 (js/storage.js)

```javascript
/**
 * 游戏存储模块
 * 负责保存和加载游戏进度
 */
class GameStorage {
    constructor() {
        this.storageKey = 'yang_le_le_game_data';
        this.highScoreKey = 'yang_le_le_high_score';
        this.gamesPlayedKey = 'yang_le_le_games_played';
        this.initializeStorage();
    }

    /**
     * 初始化存储
     */
    initializeStorage() {
        if (!this.isLocalStorageAvailable()) {
            console.warn('Local storage is not available. Game progress cannot be saved.');
            return;
        }

        // 初始化统计数据
        if (!localStorage.getItem(this.highScoreKey)) {
            localStorage.setItem(this.highScoreKey, '0');
        }

        if (!localStorage.getItem(this.gamesPlayedKey)) {
            localStorage.setItem(this.gamesPlayedKey, '0');
        }
    }

    /**
     * 检查本地存储是否可用
     */
    isLocalStorageAvailable() {
        try {
            const testKey = '__test__';
            localStorage.setItem(testKey, testKey);
            localStorage.removeItem(testKey);
            return true;
        } catch (e) {
            return false;
        }
    }

    /**
     * 保存游戏状态
     * @param {Object} gameState - 游戏状态对象
     */
    saveGameState(gameState) {
        if (!this.isLocalStorageAvailable()) {
            console.error('Cannot save game state: Local storage is not available');
            return false;
        }

        try {
            const dataToSave = {
                ...gameState,
                timestamp: Date.now(),
                version: '1.0'
            };
            
            localStorage.setItem(this.storageKey, JSON.stringify(dataToSave));
            console.log('Game state saved successfully');
            return true;
        } catch (e) {
            console.error('Failed to save game state:', e);
            return false;
        }
    }

    /**
     * 加载游戏状态
     * @returns {Object|null} 游戏状态对象，如果没有保存的游戏则返回null
     */
    loadGameState() {
        if (!this.isLocalStorageAvailable()) {
            console.error('Cannot load game state: Local storage is not available');
            return null;
        }

        try {
            const savedData = localStorage.getItem(this.storageKey);
            if (!savedData) {
                console.log('No saved game state found');
                return null;
            }

            const gameState = JSON.parse(savedData);
            
            // 检查数据版本
            if (gameState.version !== '1.0') {
                console.warn('Game state version mismatch, using saved data anyway');
            }

            // 检查数据是否过期（7天）
            const now = Date.now();
            const savedTime = gameState.timestamp || 0;
            const maxAge = 7 * 24 * 60 * 60 * 1000; // 7天
            
            if (now - savedTime > maxAge) {
                console.log('Saved game state is too old, removing');
                this.clearGameState();
                return null;
            }

            console.log('Game state loaded successfully');
            return gameState;
        } catch (e) {
            console.error('Failed to load game state:', e);
            return null;
        }
    }

    /**
     * 清除保存的游戏状态
     */
    clearGameState() {
        if (!this.isLocalStorageAvailable()) {
            return;
        }

        try {
            localStorage.removeItem(this.storageKey);
            console.log('Game state cleared');
        } catch (e) {
            console.error('Failed to clear game state:', e);
        }
    }

    /**
     * 更新最高分
     * @param {number} score - 当前得分
     */
    updateHighScore(score) {
        if (!this.isLocalStorageAvailable()) {
            return;
        }

        try {
            const currentHighScore = parseInt(localStorage.getItem(this.highScoreKey), 10) || 0;
            
            if (score > currentHighScore) {
                localStorage.setItem(this.highScoreKey, score.toString());
                console.log(`High score updated to: ${score}`);
                return true;
            }
            
            return false;
        } catch (e) {
            console.error('Failed to update high score:', e);
            return false;
        }
    }

    /**
     * 获取
