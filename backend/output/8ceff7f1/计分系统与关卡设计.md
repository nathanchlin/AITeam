## 计分系统设计

### 计分规则
```javascript
class ScoreSystem {
  constructor() {
    this.score = 0;
    this.combo = 0;
    this.maxCombo = 0;
    this.moves = 0;
    this.matchedPairs = 0;
  }

  // 消除一对卡片时的计分
  addMatchedPair(pairValue, isSpecialCard = false) {
    this.moves++;
    this.matchedPairs++;
    
    // 基础分数根据卡片价值
    let points = pairValue * 10;
    
    // 连击加分
    this.combo++;
    if (this.combo > 1) {
      points *= (1 + (this.combo - 1) * 0.1); // 每次连击增加10%分数
    }
    
    // 特殊卡片额外加分
    if (isSpecialCard) {
      points *= 2;
    }
    
    // 完成关卡奖励
    if (this.matchedPairs % 6 === 0) { // 每消除6对卡片
      points += 100;
    }
    
    this.score += Math.floor(points);
    this.maxCombo = Math.max(this.maxCombo, this.combo);
    
    return Math.floor(points);
  }

  // 错误匹配时重置连击
  resetCombo() {
    this.combo = 0;
  }

  // 计算最终得分
  calculateFinalScore(level, timeUsed) {
    // 时间奖励
    const timeBonus = Math.max(0, 300 - timeUsed) * 2; // 5分钟内完成有时间奖励
    
    // 关卡奖励
    const levelBonus = level * 500;
    
    // 移动次数奖励（越少移动次数得分越高）
    const moveBonus = Math.max(0, (50 - this.moves) * 20);
    
    this.score += timeBonus + levelBonus + moveBonus;
    
    return this.score;
  }

  // 获取当前得分信息
  getScoreInfo() {
    return {
      score: this.score,
      combo: this.combo,
      maxCombo: this.maxCombo,
      moves: this.moves,
      matchedPairs: this.matchedPairs
    };
  }
}
```

### 计分规则说明
1. **基础得分**：每对卡片根据其价值获得基础分数（价值×10）
2. **连击系统**：连续正确匹配会获得额外加分，每次连击增加10%分数
3. **特殊卡片**：特殊卡片（如金卡）匹配时获得双倍分数
4. **关卡奖励**：每消除6对卡片获得100分奖励
5. **最终得分**：包括时间奖励、关卡奖励和移动次数奖励

1. **渐进式难度**：从4×4的简单布局逐渐增加到10×10的复杂布局
2. **多样化卡片**：卡片类型从6种逐渐增加到24种
3. **特殊卡片**：随关卡增加特殊卡片数量
4. **时间限制**：高级关卡有时间限制，增加紧张感
5. **目标分数**：每个关卡设置不同的目标分数
6. **提示系统**：高级关卡减少提示次数，增加挑战性

## 游戏流程整合

```javascript
class SheepSheepGame {
  constructor() {
    this.scoreSystem = new ScoreSystem();
    this.levelSystem = new LevelSystem();
    this.gameState = 'menu'; // menu, playing, paused, completed
    this.selectedCards = [];
    this.board = [];
    this.isProcessing = false;
  }

  // 开始新游戏
  startNewGame() {
    this.scoreSystem = new ScoreSystem();
    this.levelSystem = new LevelSystem();
    this.gameState = 'playing';
    this.initializeBoard();
  }

  // 初始化游戏板
  initializeBoard() {
    const level = this.levelSystem.getCurrentLevel();
    const totalCards = level.rows * level.cols;
    const pairCount = totalCards / 2;
    
    // 创建卡片对
    let cards = [];
    for (let i = 0; i < pairCount; i++) {
      const cardType = i % level.cardTypes;
      const isSpecial = cardType < level.specialCards;
      cards.push({ id: i * 2, type: cardType, isSpecial: isSpecial });
      cards.push({ id: i * 2 + 1, type: cardType, isSpecial: isSpecial });
    }
    
    // 洗牌
    for (let i = 0; i < level.shuffleTimes; i++) {
      this.shuffleArray(cards);
    }
    
    // 创建游戏板
    this.board = [];
    for (let row = 0; row < level.rows; row++) {
      this.board[row] = [];
      for (let col = 0; col < level.cols; col++) {
        this.board[row][col] = cards[row * level.cols + col];
      }
    }
  }

  // 洗牌算法
  shuffleArray(array) {
    for (let i = array.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [array[i], array[j]] = [array[j], array[i]];
    }
  }

  // 选择卡片
  selectCard(row, col) {
    if (this.gameState !== 'playing' || this.isProcessing) return;
    
    const card = this.board[row][col];
    if (!card || card.matched) return;
    
    // 如果已经选择了两张卡片，清除选择
    if (this.selectedCards.length >= 2) {
      this.clearSelection();
    }
    
    // 添加到选中列表
    this.selectedCards.push({ row, col, card });
    
    // 如果选中了两张卡片，检查是否匹配
    if (this.selectedCards.length === 2) {
      this.checkMatch();
    }
  }

  // 检查匹配
  checkMatch() {
    this.isProcessing = true;
    const [first, second] = this.selectedCards;
    
    if (first.card.type === second.card.type) {
      // 匹配成功
      setTimeout(() => {
        this.handleMatch();
      }, 500);
    } else {
      // 匹配失败
      setTimeout(() => {
        this.clearSelection();
        this.scoreSystem.resetCombo();
        this.isProcessing = false;
      }, 1000);
    }
  }

  // 处理匹配成功
  handleMatch() {
    const [first, second] = this.selectedCards;
    const level = this.levelSystem.getCurrentLevel();
    
    // 标记卡片为已匹配
    first.card.matched = true;
    second.card.matched = true;
    
    // 更新分数
    const points = this.scoreSystem.addMatchedPair(first.card.type, first.card.isSpecial);
    
    // 清除选择
    this.clearSelection();
    this.isProcessing = false;
    
    // 检查是否完成关卡
    const matchedPairs = this.scoreSystem.matchedPairs;
    const totalPairs = (level.rows * level.cols) / 2;
    
    if (matchedPairs === totalPairs) {
      this.completeLevel();
    }
  }

  // 完成关卡
  completeLevel() {
    const timeUsed = (Date.now() - this.levelSystem.levelStartTime) / 1000;
    const finalScore = this.scoreSystem.calculateFinalScore(
      this.levelSystem.currentLevel,
      timeUsed
    );
    
    // 检查是否还有下一关
    if (this.levelSystem.nextLevel()) {
      // 进入下一关
      this.initializeBoard();
      this.gameState = 'playing';
    } else {
      // 游戏通关
      this.gameState = 'completed';
    }
  }

  // 清除选择
  clearSelection() {
    this.selectedCards = [];
  }

  // 使用提示
  useHint() {
    const level = this.levelSystem.getCurrentLevel();
    if (level.hintCount <= 0) return;
    
    // 寻找可匹配的卡片对
    for (let row1 = 0; row1 < level.rows; row1++) {
      for (let col1 = 0; col1 < level.cols; col1++) {
        const card1 = this.board[row1][col1];
        if (card1 && !card1.matched) {
          for (let row2 = 0; row2 < level.rows; row2++) {
            for (let col2 = 0; col2 < level.cols; col2++) {
              if (row1 === row2 && col1 === col2) continue;
              
              const card2 = this.board[row2][col2];
              if (card2 && !card2.matched && card1.type === card2.type) {
                // 高亮显示这对卡片
                this.highlightCards(row1, col1, row2, col2);
                level.hintCount--;
                return;
              }
            }
          }
        }
      }
    }
  }

  // 高亮显示卡片
  highlightCards(row1, col1, row2, col2) {
    // 实现高亮效果
    console.log(`提示: (${row1},${col1}) 和 (${row2},${col2}) 可以匹配`);
  }

  // 获取游戏状态
  getGameState() {
    return {
      gameState: this.gameState,
      score: this.scoreSystem.getScoreInfo(),
      level: this.levelSystem.getProgress(),
      board: this.board,
      selectedCards: this.selectedCards
    };
  }
}
```

## 游戏体验优化建议

1. **视觉反馈**：
   - 匹配成功时显示得分动画
   - 连击时显示连击计数
   - 特殊卡片匹配时有特殊效果

2. **音效设计**：
   - 不同类型卡片匹配有不同音效
   - 连击时音效节奏加快
   - 关卡完成时有胜利音效

3. **难度曲线**：
   - 根据玩家表现动态调整难度
   - 连胜时增加特殊卡片出现概率
   - 连败时提供额外提示

4. **成就系统**：
   - 记录最高分、最长连击等
   - 解锁成就如"完美通关"、"速度之星"等
   - 成就影响后续关卡难度
