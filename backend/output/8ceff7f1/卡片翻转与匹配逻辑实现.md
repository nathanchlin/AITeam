### JavaScript逻辑 (game.js)

```javascript
class SheepSheepGame {
    constructor() {
        this.cards = [];
        this.flippedCards = [];
        this.matchedPairs = 0;
        this.moves = 0;
        this.score = 0;
        this.combo = 0;
        this.level = 1;
        this.timer = 0;
        this.timerInterval = null;
        this.soundEnabled = true;
        this.isProcessing = false;
        this.bestScore = localStorage.getItem('sheep-sheep-best-score') || 0;
        
        // 卡片符号
        this.cardSymbols = ['🐑', '🐐', '🐴', '🐄', '🐷', '🐮', '🐵', '🐶', '🐱', '🐭'];
        
        // DOM元素
        this.cardDeck = document.getElementById('card-deck');
        this.levelDisplay = document.getElementById('level');
        this.scoreDisplay = document.getElementById('score');
        this.movesDisplay = document.getElementById('moves');
        this.timerDisplay = document.getElementById('timer');
        this.comboDisplay = document.getElementById('combo');
        this.bestScoreDisplay = document.getElementById('best-score');
        this.gameMessage = document.getElementById('game-message');
        this.startBtn = document.getElementById('start-btn');
        this.resetBtn = document.getElementById('reset-btn');
        this.soundBtn = document.getElementById('sound-btn');
        
        // 初始化
        this.init();
    }
    
    init() {
        // 更新最高分显示
        this.bestScoreDisplay.textContent = this.bestScore;
        
        // 绑定事件
        this.startBtn.addEventListener('click', () => this.startGame());
        this.resetBtn.addEventListener('click', () => this.resetGame());
        this.soundBtn.addEventListener('click', () => this.toggleSound());
        
        // 检测设备类型
        this.detectDevice();
        
        // 初始渲染
        this.renderMessage('点击"开始游戏"开始挑战！');
    }
    
    detectDevice() {
        this.isMobile = window.innerWidth <= 768 || 'ontouchstart' in window;
        
        if (this.isMobile) {
            // 移动设备优化
            this.cardDeck.style.gridTemplateColumns = 'repeat(auto-fill, minmax(70px, 1fr))';
            document.querySelectorAll('.card').forEach(card => {
                card.style.width = '70px';
                card.style.height = '70px';
            });
        }
    }
    
    startGame() {
        // 重置游戏状态
        this.resetGameState();
        
        // 生成卡片
        this.generateCards();
        
        // 开始计时
        this.startTimer();
        
        // 更新UI
        this.startBtn.disabled = true;
        this.renderMessage('游戏开始！找出相同的卡片');
    }
    
    resetGame() {
        this.resetGameState();
        this.clearBoard();
        this.startBtn.disabled = false;
        this.renderMessage('游戏已重置，点击"开始游戏"重新开始');
    }
    
    resetGameState() {
        this.cards = [];
        this.flippedCards = [];
        this.matchedPairs = 0;
        this.moves = 0;
        this.score = 0;
        this.combo = 0;
        this.timer = 0;
        this.isProcessing = false;
        
        if (this.timerInterval) {
            clearInterval(this.timerInterval);
            this.timerInterval = null;
        }
        
        this.updateDisplay();
    }
    
    generateCards() {
        // 根据关卡确定卡片数量
        const cardCount = this.getCardCountForLevel(this.level);
        const pairsNeeded = cardCount / 2;
        
        // 生成卡片对
        const cardPairs = [];
        for (let i = 0; i < pairsNeeded; i++) {
            const symbol = this.cardSymbols[i % this.cardSymbols.length];
            cardPairs.push(symbol, symbol); // 添加一对相同的卡片
        }
        
        // 随机打乱卡片
        this.shuffleArray(cardPairs);
        
        // 创建卡片元素
        cardPairs.forEach((symbol, index) => {
            const card = this.createCard(symbol, index);
            this.cardDeck.appendChild(card);
            this.cards.push({
                element: card,
                symbol: symbol,
                id: index,
                isFlipped: false,
                isMatched: false
            });
        });
    }
    
    getCardCountForLevel(level) {
        // 根据关卡返回卡片数量
        // 基础数量为8对(16张)，每关增加2对(4张)
        return 16 + (level - 1) * 4;
    }
    
    createCard(symbol, id) {
        const card = document.createElement('div');
        card.className = 'card';
        card.dataset.id = id;
        
        // 卡片正面（显示符号）
        const cardFront = document.createElement('div');
        cardFront.className = 'card-face card-front';
        cardFront.textContent = symbol;
        
        // 卡片背面（统一图案）
        const cardBack = document.createElement('div');
        cardBack.className = 'card-face card-back';
        cardBack.textContent = '?';
        
        card.appendChild(cardFront);
        card.appendChild(cardBack);
        
        // 添加点击事件
        card.addEventListener('click', () => this.handleCardClick(id));
        
        return card;
    }
    
    handleCardClick(cardId) {
        // 如果正在处理中或卡片已匹配，则忽略点击
        if (this.isProcessing || this.cards[cardId].isMatched) return;
        
        const card = this.cards[cardId];
        
        // 如果卡片已经翻开，则忽略点击
        if (card.isFlipped) return;
        
        // 如果已经翻开了两张卡片，则忽略点击
        if (this.flippedCards.length >= 2) return;
        
        // 翻开卡片
        this.flipCard(card);
        
        // 添加到已翻开卡片列表
        this.flippedCards.push(card);
        
        // 如果翻开了两张卡片，检查是否匹配
        if (this.flippedCards.length === 2) {
            this.moves++;
            this.updateDisplay();
            
            this.isProcessing = true;
            
            // 延迟检查匹配，让玩家看到卡片
            setTimeout(() => {
                this.checkForMatch();
            }, 800);
        }
    }
    
    flipCard(card) {
        card.isFlipped = true;
        card.element.classList.add('flipped');
        
        // 播放翻牌音效
        if (this.soundEnabled) {
            this.playSound('flip');
        }
    }
    
    unflipCard(card) {
        card.isFlipped = false;
        card.element.classList.remove('flipped');
        card.element.classList.add('shake');
        
        // 移除shake动画类
        setTimeout(() => {
            card.element.classList.remove('shake');
        }, 500);
        
        // 播放错误音效
        if (this.soundEnabled) {
            this.playSound('error');
        }
    }
    
    checkForMatch() {
        const [card1, card2] = this.flippedCards;
        
        if (card1.symbol === card2.symbol) {
            // 匹配成功
            this.handleMatch(card1, card2);
        } else {
            // 匹配失败
            this.handleMismatch(card1, card2);
        }
        
        // 清空已翻开卡片列表
        this.flippedCards = [];
        this.isProcessing = false;
    }
    
    handleMatch(card1, card2) {
        // 标记卡片为已匹配
        card1.isMatched = true;
        card2.isMatched = true;
        card1.element.classList.add('matched');
        card2.element.classList.add('matched');
        
        // 更新分数和连击
        this.combo++;
        const points = 10 * this.combo;
        this.score += points;
        this.matchedPairs++;
        
        // 播放匹配成功音效
        if (this.soundEnabled) {
            this.playSound('match');
        }
        
        // 添加动画效果
        card1.element.classList.add('pulse');
        card2.element.classList.add('pulse');
        
        setTimeout(() => {
            card1.element.classList.remove('pulse');
            card2.element.classList.remove('pulse');
        }, 500);
        
        // 更新显示
        this.updateDisplay();
        
        // 检查是否完成关卡
        if (this.matchedPairs === this.cards.length / 2) {
            this.handleLevelComplete();
        } else {
            this.renderMessage(`匹配成功！+${points}分`);
        }
    }
    
    handleMismatch(card1, card2) {
        // 翻回卡片
        this.unflipCard(card1);
        this.unflipCard(card2);
        
        // 重置连击
        this.combo = 0;
        
        // 更新显示
        this.updateDisplay();
        this.renderMessage('不匹配，请继续尝试');
    }
    
    handleLevelComplete() {
        // 停止计时
        clearInterval(this.timerInterval);
        
        // 计算奖励分数
        const timeBonus = Math.max(0, 300 - this.timer) * 2; // 5分钟内完成有时间奖励
        const moveBonus = Math.max(0, 50 - this.moves) * 5; // 少步数有奖励
        const totalBonus = timeBonus + moveBonus;
        
        this.score += totalBonus;
        
        // 更新最高分
        if (this.score > this.bestScore) {
            this.bestScore = this.score;
            localStorage.setItem('sheep-sheep-best-score', this.bestScore);
            this.bestScoreDisplay.textContent = this.bestScore;
        }
        
        // 显示完成信息
        this.renderMessage(`关卡 ${this.level} 完成！时间奖励: ${timeBonus}步数奖励: ${moveBonus}总分: ${this.score}`);
        
        // 进入下一关
        setTimeout(() => {
            this.level++;
            this.startGame();
        }, 3000);
    }
    
    startTimer() {
        this.timerInterval = setInterval(() => {
            this.timer++;
            this.updateDisplay();
        }, 1000);
    }
    
    updateDisplay() {
        this.levelDisplay.textContent = this.level;
        this.scoreDisplay.textContent = this.score;
        this.movesDisplay.textContent = this.moves;
        this.comboDisplay.textContent = this.combo;
        
        // 格式化时间显示
        const minutes = Math.floor(this.timer / 60);
        const seconds = this.timer % 60;
        this.timerDisplay.textContent = `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`;
    }
    
    clearBoard() {
        this.cardDeck.innerHTML = '';
        this.cards = [];
    }
    
    renderMessage(message) {
        this.gameMessage.textContent = message;
    }
    
    toggleSound() {
        this.soundEnabled = !this.soundEnabled;
        this.soundBtn.textContent = `音效: ${this.soundEnabled ? '开' : '关'}`;
    }
    
    playSound(type) {
        // 在实际应用中，这里会播放真实的音效文件
        // 这里只是模拟音效播放
        console.log(`Playing sound: ${type}`);
    }
    
    shuffleArray(array) {
        // Fisher-Yates 洗牌算法
        for (let i = array.length - 1; i > 0; i--) {
            const j = Math.floor(Math.random() * (i + 1));
            [array[i], array[j]] = [array[j], array[i]];
        }
        return array;
    }
    
    // 处理窗口大小变化
    handleResize() {
        // 重新检测
