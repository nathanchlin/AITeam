    <script>
        // 游戏常量
        const TILE_SIZE = 32;
        const GRID_WIDTH = 20;
        const GRID_HEIGHT = 20;
        const PLAYER_SPEED = 4;
        const ENEMY_SPEED = 2;
        const BULLET_SPEED = 8;
        const ENEMY_BULLET_SPEED = 6;

        // 游戏状态
        let canvas = document.getElementById('gameCanvas');
        let ctx = canvas.getContext('2d');
        let gameRunning = true;
        let score = 0;
        let level = 1;
        let lives = 3;
        let keys = {};

        // 地图数据结构
        const TILE_TYPES = {
            EMPTY: 0,
            WALL: 1,
            BRICK: 2,
            WATER: 3,
            BASE: 4,
            FOREST: 5
        };

        // 地图颜色
        const TILE_COLORS = {
            [TILE_TYPES.EMPTY]: '#000',
            [TILE_TYPES.WALL]: '#444',
            [TILE_TYPES.BRICK]: '#a52a2a',
            [TILE_TYPES.WATER]: '#1e90ff',
            [TILE_TYPES.BASE]: '#00ff00',
            [TILE_TYPES.FOREST]: '#228b22'
        };

        // 初始化地图
        let map = [];
        let playerTank = null;
        let enemyTanks = [];
        let bullets = [];
        let enemyBullets = [];

        // 坦克类
        class Tank {
            constructor(x, y, direction, color, isPlayer = false) {
                this.x = x;
                this.y = y;
                this.direction = direction; // 0: 上, 1: 右, 2: 下, 3: 左
                this.color = color;
                this.isPlayer = isPlayer;
                this.speed = isPlayer ? PLAYER_SPEED : ENEMY_SPEED;
                this.bullets = [];
                this.canShoot = true;
                this.shootCooldown = 0;
            }

            move(dx, dy) {
                // 检查碰撞
                let newX = this.x + dx;
                let newY = this.y + dy;
                
                // 边界检查
                if (newX < 0 || newX >= GRID_WIDTH * TILE_SIZE || 
                    newY < 0 || newY >= GRID_HEIGHT * TILE_SIZE) {
                    return;
                }
                
                // 地图碰撞检查
                let gridX = Math.floor(newX / TILE_SIZE);
                let gridY = Math.floor(newY / TILE_SIZE);
                
                // 检查坦克是否超出边界
                let tankWidth = TILE_SIZE;
                let tankHeight = TILE_SIZE;
                
                // 检查四个角
                let corners = [
                    {x: newX, y: newY},
                    {x: newX + tankWidth - 1, y: newY},
                    {x: newX, y: newY + tankHeight - 1},
                    {x: newX + tankWidth - 1, y: newY + tankHeight - 1}
                ];
                
                for (let corner of corners) {
                    let cornerGridX = Math.floor(corner.x / TILE_SIZE);
                    let cornerGridY = Math.floor(corner.y / TILE_SIZE);
                    
                    if (cornerGridX < 0 || cornerGridX >= GRID_WIDTH || 
                        cornerGridY < 0 || cornerGridY >= GRID_HEIGHT) {
                        return;
                    }
                    
                    let tile = map[cornerGridY][cornerGridX];
                    if (tile === TILE_TYPES.WALL || tile === TILE_TYPES.WATER) {
                        return;
                    }
                    
                    // 检查与其他坦克的碰撞
                    for (let tank of enemyTanks) {
                        if (tank !== this && this.checkCollision(corner, tank)) {
                            return;
                        }
                    }
                    
                    if (this !== playerTank && this.checkCollision(corner, playerTank)) {
                        return;
                    }
                }
                
                this.x = newX;
                this.y = newY;
            }

            checkCollision(point, tank) {
                return point.x >= tank.x && point.x <= tank.x + TILE_SIZE - 1 &&
                       point.y >= tank.y && point.y <= tank.y + TILE_SIZE - 1;
            }

            rotate(direction) {
                this.direction = direction;
            }

            shoot() {
                if (!this.canShoot) return;
                
                let bulletX = this.x + TILE_SIZE / 2 - 2;
                let bulletY = this.y + TILE_SIZE / 2 - 2;
                
                // 根据方向调整子弹初始位置
                switch(this.direction) {
                    case 0: // 上
                        bulletY = this.y - 4;
                        break;
                    case 1: // 右
                        bulletX = this.x + TILE_SIZE;
                        break;
                    case 2: // 下
                        bulletY = this.y + TILE_SIZE;
                        break;
                    case 3: // 左
                        bulletX = this.x - 4;
                        break;
                }
                
                let bullet = new Bullet(bulletX, bulletY, this.direction, this.isPlayer);
                if (this.isPlayer) {
                    bullets.push(bullet);
                } else {
                    enemyBullets.push(bullet);
                }
                
                this.canShoot = false;
                this.shootCooldown = 30; // 30帧冷却时间
            }

            update() {
                if (this.shootCooldown > 0) {
                    this.shootCooldown--;
                    if (this.shootCooldown === 0) {
                        this.canShoot = true;
                    }
                }
            }

            draw() {
                ctx.fillStyle = this.color;
                ctx.fillRect(this.x, this.y, TILE_SIZE, TILE_SIZE);
                
                // 绘制坦克炮管
                ctx.fillStyle = '#000';
                let centerX = this.x + TILE_SIZE / 2;
                let centerY = this.y + TILE_SIZE / 2;
                let barrelLength = TILE_SIZE * 0.6;
                
                ctx.beginPath();
                switch(this.direction) {
                    case 0: // 上
                        ctx.moveTo(centerX, centerY);
                        ctx.lineTo(centerX, centerY - barrelLength);
                        break;
                    case 1: // 右
                        ctx.moveTo(centerX, centerY);
                        ctx.lineTo(centerX + barrelLength, centerY);
                        break;
                    case 2: // 下
                        ctx.moveTo(centerX, centerY);
                        ctx.lineTo(centerX, centerY + barrelLength);
                        break;
                    case 3: // 左
                        ctx.moveTo(centerX, centerY);
                        ctx.lineTo(centerX - barrelLength, centerY);
                        break;
                }
                ctx.lineWidth = 4;
                ctx.stroke();
                
                // 绘制坦克细节
                ctx.fillStyle = '#000';
                ctx.fillRect(this.x + 4, this.y + 4, 8, 8);
                ctx.fillRect(this.x + 20, this.y + 4, 8, 8);
                ctx.fillRect(this.x + 4, this.y + 20, 8, 8);
                ctx.fillRect(this.x + 20, this.y + 20, 8, 8);
            }
        }

        // 子弹类
        class Bullet {
            constructor(x, y, direction, isPlayerBullet) {
                this.x = x;
                this.y = y;
                this.direction = direction;
                this.isPlayerBullet = isPlayerBullet;
                this.speed = isPlayerBullet ? BULLET_SPEED : ENEMY_BULLET_SPEED;
            }

            move() {
                switch(this.direction) {
                    case 0: // 上
                        this.y -= this.speed;
                        break;
                    case 1: // 右
                        this.x += this.speed;
                        break;
                    case 2: // 下
                        this.y += this.speed;
                        break;
                    case 3: // 左
                        this.x -= this.speed;
                        break;
                }
            }

            isOutOfBounds() {
                return this.x < 0 || this.x >= canvas.width || 
                       this.y < 0 || this.y >= canvas.height;
            }

            checkTileCollision() {
                let gridX = Math.floor(this.x / TILE_SIZE);
                let gridY = Math.floor(this.y / TILE_SIZE);
                
                if (gridX < 0 || gridX >= GRID_WIDTH || 
                    gridY < 0 || gridY >= GRID_HEIGHT) {
                    return true;
                }
                
                let tile = map[gridY][gridX];
                
                if (tile === TILE_TYPES.WALL) {
                    return true;
                }
                
                if (tile === TILE_TYPES.BRICK) {
                    // 砖墙可以被摧毁
                    map[gridY][gridX] = TILE_TYPES.EMPTY;
                    score += 10;
                    updateScore();
                    return true;
                }
                
                if (tile === TILE_TYPES.BASE) {
                    // 基地被摧毁，游戏结束
                    gameOver();
                    return true;
                }
                
                return false;
            }

            checkTankCollision(tank) {
                return this.x >= tank.x && this.x <= tank.x + TILE_SIZE - 1 &&
                       this.y >= tank.y && this.y <= tank.y + TILE_SIZE - 1;
            }

            draw() {
                ctx.fillStyle = this.isPlayerBullet ? '#ffff00' : '#ff0000';
                ctx.fillRect(this.x, this.y, 4, 4);
            }
        }

        // 初始化地图
        function initializeMap() {
            map = [];
            for (let y = 0; y < GRID_HEIGHT; y++) {
                map[y] = [];
                for (let x = 0; x < GRID_WIDTH; x++) {
                    // 边界为墙
                    if (x === 0 || x === GRID_WIDTH - 1 || y === 0 || y === GRID_HEIGHT - 1) {
                        map[y][x] = TILE_TYPES.WALL;
                    } else {
                        map[y][x] = TILE_TYPES.EMPTY;
                    }
                }
            }
            
            // 添加一些砖墙
            for (let i = 0; i < 30; i++) {
                let x = Math.floor(Math.random() * (GRID_WIDTH - 2)) + 1;
                let y = Math.floor(Math.random() * (GRID_HEIGHT - 2)) + 1;
                
                // 避免在基地位置生成砖墙
                if (!(x >= GRID_WIDTH - 3 && y >= GRID_HEIGHT - 3)) {
                    map[y][x] = TILE_TYPES.BRICK;
                }
            }
            
            // 添加一些不可摧毁的墙
            for (let i = 0; i < 10; i++) {
                let x = Math.floor(Math.random() * (GRID_WIDTH - 2)) + 1;
                let y = Math.floor(Math.random() * (GRID_HEIGHT - 2)) + 1;
                
                // 避免在基地位置生成墙
                if (!(x >= GRID_WIDTH - 3 && y >= GRID_HEIGHT - 3)) {
                    map[y][x] = TILE_TYPES.WALL;
                }
            }
            
            // 添加一些水域
            for (let i = 0; i < 5; i++) {
                let x = Math.floor(Math.random() * (GRID_WIDTH - 4)) + 2;
                let y = Math.floor(Math.random() * (GRID_HEIGHT - 4)) + 2;
                
                // 创建2x2的水域
                map[y][x] = TILE_TYPES.WATER;
                if (x + 1 < GRID_WIDTH - 1) map[y][x + 1] = TILE_TYPES.WATER;
                if (y + 1 < GRID_HEIGHT - 1) map[y + 1][x] = TILE_TYPES.WATER;
                if (x + 1 < GRID_WIDTH - 1 && y + 1 < GRID_HEIGHT - 1) map[y + 1][x + 1] = TILE_TYPES.WATER;
            }
            
            // 设置基地
            map[GRID_HEIGHT - 2][GRID_WIDTH - 2] = TILE_TYPES.BASE;
            map[GRID_HEIGHT - 2][GRID_WIDTH - 3] = TILE_TYPES.BASE;
            map[GRID_HEIGHT - 3][GRID_WIDTH - 2] = TILE_TYPES.BASE;
            map[GRID_HEIGHT - 3][GRID_WIDTH - 3] = TILE_TYPES.BASE;
        }

        // 初始化玩家坦克
        function initializePlayerTank() {
            playerTank = new Tank(
                TILE_SIZE * 2, 
                TILE_SIZE * 2, 
                0, // 初始方向向上
                '#00ff00',
                true
            );
        }

        // 初始化敌方坦克
        function initializeEnemyTanks() {
            enemyTanks = [];
            let enemyCount = 3 + level; // 随关卡增加敌人数量
            
            for (let i = 0; i < enemyCount; i++) {
                let x, y;
                do {
                    x = Math.floor(Math.random() * (GRID_WIDTH - 4)) + 2;
                    y = Math.floor(Math.random() * (GRID_HEIGHT / 2)) + 2;
                } while (map[y][x] !== TILE_TYPES.EMPTY);
                
                let direction = Math.floor(Math.random() * 4);
                let enemy = new Tank(
                    x * TILE_SIZE,
                    y * TILE_SIZE,
                    direction,
                    '#ff0000',
                    false
                );
                enemyTanks.push(enemy);
            }
        }

        // 绘制地图
        function drawMap() {
            for (let y = 0; y < GRID_HEIGHT; y++) {
                for (let x = 0; x < GRID_WIDTH; x++) {
                    let tile = map[y][x];
                    ctx.fillStyle
