    <script>
        // 游戏配置
        const GAME_CONFIG = {
            CELL_SIZE: 30,
            PLAYER_SPEED: 2,
            ENEMY_SPEED: 1.5,
            BULLET_SPEED: 5,
            ENEMY_BULLET_SPEED: 4,
            PLAYER_COOLDOWN: 300,
            ENEMY_COOLDOWN: 1500,
            MAX_ENEMIES: 5,
            ENEMY_SPAWN_DELAY: 3000,
            LEVEL_UP_SCORE: 1000,
            BASE_DAMAGE: 1,
            TANK_DAMAGE: 1,
            BRICK_DAMAGE: 1
        };

        // 游戏状态
        const GameState = {
            START: 'start',
            PLAYING: 'playing',
            PAUSED: 'paused',
            GAME_OVER: 'game_over',
            LEVEL_COMPLETE: 'level_complete'
        };

        // 方向常量
        const Direction = {
            UP: 0,
            RIGHT: 1,
            DOWN: 2,
            LEFT: 3
        };

        // 地形类型
        const TerrainType = {
            EMPTY: 0,
            BRICK: 1,
            STEEL: 2,
            WATER: 3,
            BASE: 4
        };

        // 游戏类
        class Game {
            constructor() {
                this.canvas = document.getElementById('gameCanvas');
                this.ctx = this.canvas.getContext('2d');
                this.cellSize = GAME_CONFIG.CELL_SIZE;
                
                // 设置画布尺寸
                this.resizeCanvas();
                window.addEventListener('resize', () => this.resizeCanvas());
                
                // 游戏状态
                this.state = GameState.START;
                this.score = 0;
                this.lives = 3;
                this.level = 1;
                this.highScore = localStorage.getItem('tankHighScore') || 0;
                
                // 游戏对象
                this.player = null;
                this.enemies = [];
                this.bullets = [];
                this.enemyBullets = [];
                this.terrain = [];
                this.explosions = [];
                
                // 输入处理
                this.keys = {};
                this.setupInputHandlers();
                
                // 游戏计时器
                this.lastTime = 0;
                this.enemySpawnTimer = 0;
                this.enemyCooldownTimer = 0;
                
                // 更新UI
                this.updateUI();
                
                // 开始游戏循环
                this.gameLoop = this.gameLoop.bind(this);
                requestAnimationFrame(this.gameLoop);
            }
            
            resizeCanvas() {
                const maxWidth = window.innerWidth * 0.9;
                const maxHeight = window.innerHeight * 0.8;
                
                // 保持16:10的宽高比
                const aspectRatio = 16 / 10;
                let width = maxWidth;
                let height = width / aspectRatio;
                
                if (height > maxHeight) {
                    height = maxHeight;
                    width = height * aspectRatio;
                }
                
                this.canvas.width = width;
                this.canvas.height = height;
                
                // 重新计算单元格大小以适应新画布尺寸
                this.cellSize = Math.min(width / 20, height / 13);
            }
            
            setupInputHandlers() {
                // 键盘事件
                window.addEventListener('keydown', (e) => {
                    this.keys[e.key.toLowerCase()] = true;
                    
                    // 处理暂停
                    if (e.key.toLowerCase() === 'p' && this.state === GameState.PLAYING) {
                        this.pauseGame();
                    } else if (e.key.toLowerCase() === 'p' && this.state === GameState.PAUSED) {
                        this.resumeGame();
                    }
                    
                    // 防止空格键滚动页面
                    if (e.key === ' ') {
                        e.preventDefault();
                    }
                });
                
                window.addEventListener('keyup', (e) => {
                    this.keys[e.key.toLowerCase()] = false;
                });
                
                // 按钮事件
                document.getElementById('startBtn').addEventListener('click', () => {
                    this.startGame();
                });
                
                document.getElementById('restartBtn').addEventListener('click', () => {
                    this.restartGame();
                });
            }
            
            startGame() {
                document.getElementById('gameStart').style.display = 'none';
                this.state = GameState.PLAYING;
                this.initializeLevel();
            }
            
            restartGame() {
                this.score = 0;
                this.lives = 3;
                this.level = 1;
                this.updateUI();
                document.getElementById('gameOver').style.display = 'none';
                this.state = GameState.PLAYING;
                this.initializeLevel();
            }
            
            pauseGame() {
                this.state = GameState.PAUSED;
                document.getElementById('pauseIndicator').style.display = 'block';
            }
            
            resumeGame() {
                this.state = GameState.PLAYING;
                document.getElementById('pauseIndicator').style.display = 'none';
            }
            
            initializeLevel() {
                // 清空游戏对象
                this.enemies = [];
                this.bullets = [];
                this.enemyBullets = [];
                this.explosions = [];
                
                // 初始化地形
                this.initializeTerrain();
                
                // 创建玩家
                this.player = new Tank(
                    Math.floor(this.terrain[0].length / 2) * this.cellSize,
                    (this.terrain.length - 1) * this.cellSize,
                    Direction.UP,
                    'player'
                );
                
                // 重置计时器
                this.enemySpawnTimer = 0;
                this.enemyCooldownTimer = 0;
            }
            
            initializeTerrain() {
                const rows = 13;
                const cols = 20;
                this.terrain = [];
                
                // 创建空地形
                for (let i = 0; i < rows; i++) {
                    this.terrain[i] = [];
                    for (let j = 0; j < cols; j++) {
                        this.terrain[i][j] = TerrainType.EMPTY;
                    }
                }
                
                // 添加边界墙
                for (let i = 0; i < rows; i++) {
                    this.terrain[i][0] = TerrainType.STEEL;
                    this.terrain[i][cols - 1] = TerrainType.STEEL;
                }
                for (let j = 0; j < cols; j++) {
                    this.terrain[0][j] = TerrainType.STEEL;
                    this.terrain[rows - 1][j] = TerrainType.STEEL;
                }
                
                // 添加基地
                const baseCol = Math.floor(cols / 2);
                this.terrain[rows - 1][baseCol] = TerrainType.BASE;
                
                // 根据关卡添加地形
                this.addLevelTerrain(rows, cols);
            }
            
            addLevelTerrain(rows, cols) {
                // 关卡1: 简单布局
                if (this.level === 1) {
                    // 添加一些砖墙
                    for (let i = 2; i < rows - 2; i += 2) {
                        for (let j = 2; j < cols - 2; j += 2) {
                            if (Math.random() > 0.3) {
                                this.terrain[i][j] = TerrainType.BRICK;
                            }
                        }
                    }
                }
                
                // 关卡2: 更复杂的布局
                if (this.level === 2) {
                    // 添加砖墙和钢墙
                    for (let i = 2; i < rows - 2; i += 2) {
                        for (let j = 2; j < cols - 2; j += 2) {
                            if (Math.random() > 0.2) {
                                this.terrain[i][j] = Math.random() > 0.7 ? TerrainType.STEEL : TerrainType.BRICK;
                            }
                        }
                    }
                    
                    // 添加一些钢墙保护基地
                    const baseCol = Math.floor(cols / 2);
                    this.terrain[rows - 2][baseCol - 1] = TerrainType.STEEL;
                    this.terrain[rows - 2][baseCol + 1] = TerrainType.STEEL;
                }
                
                // 关卡3+: 更复杂的布局
                if (this.level >= 3) {
                    // 添加砖墙和钢墙
                    for (let i = 2; i < rows - 2; i += 2) {
                        for (let j = 2; j < cols - 2; j += 2) {
                            if (Math.random() > 0.15) {
                                this.terrain[i][j] = Math.random() > 0.6 ? TerrainType.STEEL : TerrainType.BRICK;
                            }
                        }
                    }
                    
                    // 添加一些水
                    for (let i = 3; i < rows - 3; i += 3) {
                        for (let j = 3; j < cols - 3; j += 3) {
                            if (Math.random() > 0.7) {
                                this.terrain[i][j] = TerrainType.WATER;
                            }
                        }
                    }
                    
                    // 添加钢墙保护基地
                    const baseCol = Math.floor(cols / 2);
                    this.terrain[rows - 2][baseCol - 1] = TerrainType.STEEL;
                    this.terrain[rows - 2][baseCol + 1] = TerrainType.STEEL;
                    this.terrain[rows - 3][baseCol] = TerrainType.STEEL;
                }
            }
            
            update(deltaTime) {
                if (this.state !== GameState.PLAYING) return;
                
                // 更新玩家
                if (this.player) {
                    this.updatePlayer(deltaTime);
                }
                
                // 更新敌人
                this.updateEnemies(deltaTime);
                
                // 更新子弹
                this.updateBullets(deltaTime);
                this.updateEnemyBullets(deltaTime);
                
                // 更新爆炸
                this.updateExplosions(deltaTime);
                
                // 检查碰撞
                this.checkCollisions();
                
                // 生成敌人
                this.spawnEnemies(deltaTime);
                
                // 检查游戏结束条件
                this.checkGameEnd();
            }
            
            updatePlayer(deltaTime) {
                const speed = GAME_CONFIG.PLAYER_SPEED;
                let dx = 0;
                let dy = 0;
                
                // 处理输入
                if (this.keys['w']) dy = -speed;
                if (this.keys['s']) dy = speed;
                if (this.keys['a']) dx = -speed;
                if (this.keys['d']) dx = speed;
                
                // 射击
                if (this.keys[' '] && this.player.canShoot()) {
                    this.bullets.push(new Bullet(
                        this.player.x + this.cellSize / 2,
                        this.player.y + this.cellSize / 2,
                        this.player.direction,
                        'player'
                    ));
                    this.player.shoot();
                }
                
                // 移动玩家
                if (dx !== 0 || dy !== 0) {
                    const newX = this.player.x + dx;
                    const newY = this.player.y + dy;
                    
                    // 检查边界
                    if (newX >= 0 && newX <= this.canvas.width - this.cellSize &&
                        newY >= 0 && newY <= this.canvas.height - this.cellSize) {
                        
                        // 检查地形碰撞
                        if (!this.checkTerrainCollision(newX, newY, this.cellSize)) {
                            this.player.x = newX;
                            this.player.y = newY;
                        }
                    }
                }
                
                // 更新玩家方向
                if (this.keys['w']) this.player.direction = Direction.UP;
                else if (this.keys['s']) this.player.direction = Direction.DOWN;
                else if (this.keys['a']) this.player.direction = Direction.LEFT;
                else if (this.keys['d']) this.player.direction = Direction.RIGHT;
                
                // 更新玩家射击冷却
                this.player.updateCooldown(deltaTime);
            }
            
            updateEnemies(deltaTime) {
                // 更新每个敌人
                for (let i = this.enemies.length - 1; i >= 0; i--) {
                    const enemy = this.enemies[i];
                    
                    // 更新AI
                    enemy.updateAI(this.player, this.terrain, this.cellSize);
                    
                    // 移动敌人
                    const speed = GAME_CONFIG.ENEMY_SPEED;
                    const newX = enemy.x + enemy.dx * speed;
                    const newY = enemy.y + enemy.dy * speed;
                    
                    //
