        class SoundSystem {
            constructor() {
                this.sounds = {
                    shoot: this.createSound(200, 0.1, 'square'),

                    hit: this.createSound(150, 0.1, 'triangle'),

            createSound(frequency, duration, type = 'sine') {
                return () => {
                    if (!gameState.soundEnabled) return;
                    
                    const audioContext = new (window.AudioContext || window.webkitAudioContext)();
                    const oscillator = audioContext.createOscillator();
                    const gainNode = audioContext.createGain();
                    
                    oscillator.type = type;
                    oscillator.frequency.value = frequency;
                    gainNode.gain.value = 0.1;
                    
                    oscillator.connect(gainNode);
                    gainNode.connect(audioContext.destination);
                    
                    oscillator.start();
                    
                    gainNode.gain.exponentialRampToValueAtTime(0.001, audioContext.currentTime + duration);
                    oscillator.stop(audioContext.currentTime + duration);
                };
            }
            
            play(soundName) {
                if (this.sounds[soundName]) {
                    this.sounds[soundName]();
                }
            }

        class Tank {
            constructor(x, y, isPlayer = false, color = '#0f0') {
                this.x = x;
                this.y = y;
                this.width = config.cellSize;
                this.height = config.cellSize;

                this.isDestroyed = false;

                this.moveTimer = 0;
                this.moveInterval = Math.random() * 1000 + 500;
            }
            
            update(currentTime) {
                if (this.isDestroyed) return;
                
                // 更新子弹
                this.bullets = this.bullets.filter(bullet => {
                    bullet.update();
                    return !bullet.isOutOfBounds() && !bullet.isDestroyed;
                });
                
                // 移动逻辑
                if (this.isPlayer) {
                    this.handlePlayerMovement();
                } else {
                    this.handleEnemyMovement(currentTime);
                }
                
                // 限制在游戏区域内
                this.x = Math.max(0, Math.min(canvas.width - this.width, this.x));
                this.y = Math.max(0, Math.min(canvas.height - this.height, this.y));
            }
            
            handlePlayerMovement() {

                    this.y += this.speed;

                    this.direction = 'left';
                    this.x -= this.speed;
                }

                    this.direction = 'right';
                    this.x += this.speed;
                }
                
                // 检测碰撞
                if (this.checkCollisionWithTiles()) {
                    this.undoMove();
                }
            }
            
            handleEnemyMovement(currentTime) {
                // 敌人AI逻辑
                this.moveTimer += 16; // 假设60fps，每帧约16ms
                
                // 随机改变方向或朝向玩家
                if (this.moveTimer > this.moveInterval || this.targetDirection === null) {
                    const player = game.player;
                    const distanceToPlayer = Math.sqrt(
                        Math.pow(player.x - this.x, 2) +

                    );
                    
                    // 如果玩家在一定范围内，朝向玩家
                    if (distanceToPlayer < 300) {
                        if (Math.abs(player.x - this.x) > Math.abs(player.y - this.y)) {
                            this.targetDirection = player.x > this.x ? 'right' : 'left';
                        } else {
                            this.targetDirection = player.y > this.y ? 'down' : 'up';
                        }
                    } else {
                        // 随机选择方向
                        const directions = ['up', 'down', 'left', 'right'];
                        this.targetDirection = directions[Math.floor(Math.random() * directions.length)];
                    }
                    
                    this.moveTimer = 0;
                    this.moveInterval = Math.random() * 1000 + 500;
                }
                
                // 尝试朝目标方向移动
                const prevX = this.x;
                const prevY = this.y;
                
                switch (this.targetDirection) {
                    case 'up':
                        this.y -= this.speed;
                        this.direction = 'up';
                        break;
                    case 'down':
                        this.y += this.speed;
                        this.direction = 'down';
                        break;
                    case 'left':
                        this.x -= this.speed;
                        this.direction = 'left';
                        break;
                    case 'right':
                        this.x += this.speed;
                        this.direction = 'right';
                        break;
                }
                
                // 检测碰撞，如果碰撞则尝试其他方向
                if (this.checkCollisionWithTiles()) {
                    this.x = prevX;
                    this.y = prevY;
                    
                    // 随机选择新方向
                    const directions = ['up', 'down', 'left', 'right'];
                    this.targetDirection = directions[Math.floor(Math.random() * directions.length)];
                    this.moveTimer = 0;
                }
                
                // 随机射击
                if (currentTime - this.lastShotTime > this.shotCooldown && Math.random() < 0.01) {
                    this.shoot();
                    this.lastShotTime = currentTime;
                }
            }
            
            undoMove() {
                switch (this.direction) {
                    case 'up':
                        this.y += this.speed;
                        break;
                    case 'down':
                        this.y -= this.speed;
                        break;
                    case 'left':
                        this.x += this.speed;
                        break;
                    case 'right':
                        this.x -= this.speed;
                        break;
                }
            }
            
            checkCollisionWithTiles() {
                const tankLeft = Math.floor(this.x / config.cellSize);
                const tankRight = Math.floor((this.x + this.width - 1) / config.cellSize);
                const tankTop = Math.floor(this.y / config.cellSize);
                const tankBottom = Math.floor((this.y + this.height - 1) / config.cellSize);
                
                // 检查坦克周围的瓦片
                for (let y = tankTop; y <= tankBottom; y++) {
                    for (let x = tankLeft; x <= tankRight; x++) {
                        if (x >= 0 && x < config.gameWidth && y >= 0 && y < config.gameHeight) {
                            const tile = game.map[y][x];
                            if (tile === TILE_TYPES.WALL || tile === TILE_TYPES.BASE) {
                                return true;
                            }
                        }
                    }
                }
                
                // 检查与其他坦克的碰撞
                const allTanks = [game.player, ...game.enemies];
                for (const tank of allTanks) {
                    if (tank !== this && !tank.isDestroyed) {
                        if (this.checkCollision(tank)) {
                            return true;
                        }
                    }
                }
                
                return false;
            }
            
            checkCollision(other) {
                return this.x < other.x + other.width &&
                       this.x + this.width > other.x &&
                       this.y < other.y + other.height &&
                       this.y + this.height > other.y;
            }
