                this.y = y;

                this.moveChangeInterval = 1000;

                if (this.isPlayer) {
                    this.handlePlayerInput();
                } else {
                    this.handleEnemyAI(deltaTime);
                }

                // 边界检查
                this.x = Math.max(0, Math.min(game.width - this.width, this.x));
                this.y = Math.max(0, Math.min(game.height - this.height, this.y));

                    this.y -= this.speed * Math.sin(this.getDirectionAngle());

                    const directions = ['up', 'down', 'left', 'right'];
                    this.moveDirection = directions[Math.floor(Math.random() * directions.length)];
                    this.moveTimer = 0;
                    this.moveChangeInterval = 500 + Math.random() * 1500;
                }

                        this.direction = 'up';
                        break;
                    case 'down':
                        this.y += this.speed;
                        this.direction = 'down';
                        break;
                    case 'left':
                        this.x -= this.speed;
                        this.direction = 'left';
                        break;
                    case 'right':
                        this.x += this.speed;
                        this.direction = 'right';
                        break;

                    this.moveTimer = 0;

                }

                    this.shoot();
                }
            }

            getDirectionAngle() {
                switch (this.direction) {
                    case 'up': return -Math.PI / 2;
                    case 'down': return Math.PI / 2;
                    case 'left': return Math.PI;
                    case 'right': return 0;
                    default: return -Math.PI / 2;
                }
            }

            shoot() {
                const bulletX = this.x + this.width / 2 - 2;
                const bulletY = this.y + this.height / 2 - 2;
                
                if (this.isPlayer) {
                    game.bullets.push(new Bullet(bulletX, bulletY, this.direction, true));
                } else {
                    game.enemyBullets.push(new Bullet(bulletX, bulletY, this.direction, false));
                }
            }

            checkWallCollision() {
                // 创建坦克的边界框
                const tankRect = {
                    x: this.x,
                    y: this.y,
                    width: this.width,
                    height: this.height
                };

                // 检查与所有墙壁的碰撞
                for (const wall of game.walls) {
                    if (this.isColliding(tankRect, wall)) {
                        return true;
                    }
                }

                // 检查与砖块的碰撞
                for (let i = game.bricks.length - 1; i >= 0; i--) {
                    const brick = game.bricks[i];
                    if (this.isColliding(tankRect, brick)) {
                        if (this.isPlayer) {
                            // 玩家可以破坏砖块
                            game.bricks.splice(i, 1);
                            game.score += 10;
                            updateScore();
                        } else {
                            // 敌人不能破坏砖块，返回碰撞
                            return true;
                        }
                    }
                }

                return false;
            }

            isColliding(rect1, rect2) {
                return rect1.x < rect2.x + rect2.width &&
                       rect1.x + rect1.width > rect2.x &&
                       rect1.y < rect2.y + rect2.height &&
                       rect1.y + rect1.height > rect2.y;
            }

                        game.ctx.globalAlpha = 0.5;
                    }
                }
                
                // 绘制坦克主体
                game.ctx.fillStyle = this.color;
                game.ctx.fillRect(this.x, this.y, this.width, this.height);
                
                // 绘制坦克炮管
                game.ctx.fillStyle = this.isPlayer ? '#0a0' : '#a00';
                const barrelLength = this.width * 0.6;
                const barrelWidth = 6;
                const barrelX = this.x + this.width / 2 - barrelWidth / 2;
                const barrelY = this.y + this.height / 2 - barrelWidth / 2;
                
                game.ctx.save();
                game.ctx.translate(barrelX + barrelWidth / 2, barrelY + barrelWidth / 2);
                game.ctx.rotate(this.getDirectionAngle());
                
                if (this.direction === 'up' || this.direction === 'down') {
                    game.ctx.fillRect(-barrelWidth / 2, -barrelLength / 2, barrelWidth, barrelLength);
                } else {
                    game.ctx.fillRect(-barrelLength / 2, -barrelWidth / 2, barrelLength, barrelWidth);
                }
                
                game.ctx.restore();
                
                // 绘制坦克细节

                const centerX = this.x + this.width / 2;
                const centerY = this.y + this.height / 2;
                game.ctx.fillRect(centerX - 4, centerY - 4, 8, 8);
                
                game.ctx.restore();
            }
