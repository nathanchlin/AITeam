## 子弹类设计

```javascript
/**
 * 子弹类
 * @param {number} x - 子弹初始x坐标
 * @param {number} y - 子弹初始y坐标
 * @param {string} direction - 子弹发射方向 ('up', 'down', 'left', 'right')
 * @param {string} owner - 子弹所有者 ('player' 或 'enemy')
 */
class Bullet {
    constructor(x, y, direction, owner = 'player') {
        this.x = x;
        this.y = y;
        this.direction = direction;
        this.owner = owner;
        this.speed = 5; // 子弹移动速度
        this.size = 4; // 子弹大小
        this.active = true; // 子弹是否活跃
        this.damage = 1; // 子弹伤害值
    }

    /**
     * 更新子弹位置
     */
    update() {
        if (!this.active) return;

        switch (this.direction) {
            case 'up':
                this.y -= this.speed;
                break;
            case 'down':
                this.y += this.speed;
                break;
            case 'left':
                this.x -= this.speed;
                break;
            case 'right':
                this.x += this.speed;
                break;
        }

        // 检查子弹是否超出画布边界
        if (this.x < 0 || this.x > canvas.width || 
            this.y < 0 || this.y > canvas.height) {
            this.active = false;
        }
    }

    /**
     * 绘制子弹
     * @param {CanvasRenderingContext2D} ctx - Canvas上下文
     */
    draw(ctx) {
        if (!this.active) return;

        ctx.fillStyle = this.owner === 'player' ? '#00FF00' : '#FF0000';
        ctx.fillRect(
            this.x - this.size / 2, 
            this.y - this.size / 2, 
            this.size, 
            this.size
        );
    }

    /**
     * 获取子弹的边界矩形
     * @returns {Object} 边界矩形对象
     */
    getBounds() {
        return {
            x: this.x - this.size / 2,
            y: this.y - this.size / 2,
            width: this.size,
            height: this.size
        };
    }

    /**
     * 检查子弹与其他对象的碰撞
     * @param {Object} target - 目标对象
     * @returns {boolean} 是否发生碰撞
     */
    checkCollision(target) {
        if (!this.active || !target.active) return false;

        const bulletBounds = this.getBounds();
        const targetBounds = target.getBounds();

        return bulletBounds.x < targetBounds.x + targetBounds.width &&
               bulletBounds.x + bulletBounds.width > targetBounds.x &&
               bulletBounds.y < targetBounds.y + targetBounds.height &&
               bulletBounds.y + bulletBounds.height > targetBounds.y;
    }
}
```

## 子弹管理器

```javascript
/**
 * 子弹管理器 - 负责管理所有子弹
 */
class BulletManager {
    constructor() {
        this.bullets = [];
    }

    /**
     * 发射子弹
     * @param {number} x - 子弹初始x坐标
     * @param {number} y - 子弹初始y坐标
     * @param {string} direction - 子弹发射方向
     * @param {string} owner - 子弹所有者
     */
    fire(x, y, direction, owner = 'player') {
        // 根据方向调整子弹初始位置，使其从坦克中心发射
        let bulletX = x;
        let bulletY = y;
        
        switch (direction) {
            case 'up':
                bulletY -= 20;
                break;
            case 'down':
                bulletY += 20;
                break;
            case 'left':
                bulletX -= 20;
                break;
            case 'right':
                bulletX += 20;
                break;
        }

        this.bullets.push(new Bullet(bulletX, bulletY, direction, owner));
    }

    /**
     * 更新所有子弹
     */
    update() {
        // 更新所有子弹位置
        this.bullets.forEach(bullet => bullet.update());

        // 移除非活跃的子弹
        this.bullets = this.bullets.filter(bullet => bullet.active);
    }

    /**
     * 绘制所有子弹
     * @param {CanvasRenderingContext2D} ctx - Canvas上下文
     */
    draw(ctx) {
        this.bullets.forEach(bullet => bullet.draw(ctx));
    }

    /**
     * 检查所有子弹与目标的碰撞
     * @param {Array} targets - 目标对象数组
     * @returns {Array} 碰撞结果数组
     */
    checkCollisions(targets) {
        const collisions = [];
        
        for (let i = 0; i < this.bullets.length; i++) {
            const bullet = this.bullets[i];
            
            for (let j = 0; j < targets.length; j++) {
                const target = targets[j];
                
                if (bullet.checkCollision(target)) {
                    // 处理碰撞
                    target.hit(bullet.damage);
                    bullet.active = false;
                    collisions.push({
                        bullet: bullet,
                        target: target
                    });
                    
                    // 如果目标被摧毁，从目标数组中移除
                    if (!target.active) {
                        targets.splice(j, 1);
                        j--;
                    }
                    
                    break; // 一个子弹只能击中一个目标
                }
            }
        }
        
        return collisions;
    }

    /**
     * 清空所有子弹
     */
    clear() {
        this.bullets = [];
    }
}
```

```javascript
    bulletManager.fire(
        player.x, 
        player.y, 
        player.direction, 
        'player'
    );
}

// 敌人发射子弹
function enemyShoot(enemy) {
    bulletManager.fire(
        enemy.x, 
        enemy.y, 
        enemy.direction, 
        'enemy'
    );
}

// 设置键盘事件监听
document.addEventListener('keydown', (e) => {
    if (e.key === ' ' && !player.isShooting) { // 空格键发射子弹

        player.isShooting = true;
        setTimeout(() => player.isShooting = false, 300); // 射击冷却时间
    }
```
