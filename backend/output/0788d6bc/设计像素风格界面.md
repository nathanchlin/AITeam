    <script>
        // 游戏状态管理
        const GameState = {
            MENU: 'menu',
            PLAYING: 'playing',
            PAUSED: 'paused',
            GAME_OVER: 'game_over'
        };

        // 游戏类
        class PixelTankGame {
            constructor() {
                this.canvas = document.getElementById('gameCanvas');
                this.ctx = this.canvas.getContext('2d');
                this.ctx.imageSmoothingEnabled = false;
                
                this.gameState = GameState.MENU;
                this.score = 0;
                this.lives = 3;
                this.level = 1;
                
                this.playerTank = null;
                this.enemies = [];
                this.bullets = [];
                this.walls = [];
                this.powerUps = [];
                
                this.keys = {};
                this.lastTime = 0;
                this.enemySpawnTimer = 0;
                this.bulletCooldown = 0;
                
                this.init();
            }
            
            init() {
                // 设置事件监听器
                this.setupEventListeners();
                
                // 初始化游戏对象
                this.resetGame();
                
                // 开始游戏循环
                this.gameLoop(0);
            }
            
            setupEventListeners() {
                // 键盘事件
                window.addEventListener('keydown', (e) => {
                    this.keys[e.key] = true;
                    
                    if (e.key === 'p' || e.key === 'P') {
                        if (this.gameState === GameState.PLAYING) {
                            this.pauseGame();
                        } else if (this.gameState === GameState.PAUSED) {
                            this.resumeGame();
                        }
                    }
                    
                    if (e.key === 'Escape') {
                        if (this.gameState === GameState.PLAYING || this.gameState === GameState.PAUSED) {
                            this.showMenu();
                        }
                    }
                });
                
                window.addEventListener('keyup', (e) => {
                    this.keys[e.key] = false;
                });
                
                // 按钮事件
                document.getElementById('startButton').addEventListener('click', () => {
                    this.startGame();
                });
                
                document.getElementById('instructionsButton').addEventListener('click', () => {
                    document.getElementById('controlsInfo').style.display = 'block';
                });
                
                document.getElementById('backButton').addEventListener('click', () => {
                    document.getElementById('controlsInfo').style.display = 'none';
                });
                
                document.getElementById('resumeButton').addEventListener('click', () => {
                    this.resumeGame();
                });
                
                document.getElementById('restartButton').addEventListener('click', () => {
                    this.restartGame();
                });
                
                document.getElementById('menuButton').addEventListener('click', () => {
                    this.showMenu();
                });
                
                document.getElementById('playAgainButton').addEventListener('click', () => {
                    this.restartGame();
                });
                
                document.getElementById('mainMenuButton').addEventListener('click', () => {
                    this.showMenu();
                });
            }
            
            resetGame() {
                // 初始化玩家坦克
                this.playerTank = {
                    x: 400,
                    y: 500,
                    width: 30,
                    height: 30,
                    speed: 2,
                    direction: 'up',
                    color: '#0f0'
                };
                
                // 清空游戏对象
                this.enemies = [];
                this.bullets = [];
                this.walls = [];
                this.powerUps = [];
                
                // 生成地图
                this.generateMap();
                
                // 重置游戏状态
                this.score = 0;
                this.lives = 3;
                this.level = 1;
                this.enemySpawnTimer = 0;
                this.bulletCooldown = 0;
                
                // 更新UI
                this.updateUI();
            }
            
            generateMap() {
                // 生成砖墙
                const wallPositions = [
                    // 边界墙
                    ...Array.from({length: 26}, (_, i) => ({x: i * 30, y: 0})),
                    ...Array.from({length: 26}, (_, i) => ({x: i * 30, y: 570})),
                    ...Array.from({length: 19}, (_, i) => ({x: 0, y: i * 30 + 30})),
                    ...Array.from({length: 19}, (_, i) => ({x: 770, y: i * 30 + 30})),
                    
                    // 内部障碍物
                    ...Array.from({length: 5}, (_, i) => ({x: 150 + i * 120, y: 150})),
                    ...Array.from({length: 5}, (_, i) => ({x: 150 + i * 120, y: 180})),
                    ...Array.from({length: 5}, (_, i) => ({x: 150 + i * 120, y: 420})),
                    ...Array.from({length: 5}, (_, i) => ({x: 150 + i * 120, y: 450})),
                    
                    // 中心区域
                    {x: 350, y: 250}, {x: 380, y: 250}, {x: 410, y: 250},
                    {x: 350, y: 280}, {x: 380, y: 280}, {x: 410, y: 280},
                    {x: 350, y: 310}, {x: 380, y: 310}, {x: 410, y: 310},
                    
                    {x: 350, y: 320}, {x: 380, y: 320}, {x: 410, y: 320}
                ];
                
                this.walls = wallPositions.map(pos => ({
                    ...pos,
                    width: 30,
                    height: 30,
                    type: 'brick',
                    health: 1
                }));
            }
            
            startGame() {
                document.getElementById('startScreen').style.display = 'none';
                document.getElementById('gameScreen').style.display = 'block';
                this.gameState = GameState.PLAYING;
                this.resetGame();
            }
            
            pauseGame() {
                this.gameState = GameState.PAUSED;
                document.getElementById('pauseScreen').style.display = 'flex';
            }
            
            resumeGame() {
                this.gameState = GameState.PLAYING;
                document.getElementById('pauseScreen').style.display = 'none';
            }
            
            restartGame() {
                document.getElementById('pauseScreen').style.display = 'none';
                document.getElementById('gameOverScreen').style.display = 'none';
                this.gameState = GameState.PLAYING;
                this.resetGame();
            }
            
            showMenu() {
                this.gameState = GameState.MENU;
                document.getElementById('startScreen').style.display = 'flex';
                document.getElementById('gameScreen').style.display = 'none';
                document.getElementById('pauseScreen').style.display = 'none';
                document.getElementById('gameOverScreen').style.display = 'none';
            }
            
            gameOver() {
                this.gameState = GameState.GAME_OVER;
                document.getElementById('gameOverTitle').textContent = '游戏结束';
                document.getElementById('finalScoreValue').textContent = this.score;
                document.getElementById('gameOverScreen').style.display = 'flex';
            }
            
            updateUI() {
                document.getElementById('scoreValue').textContent = this.score;
                document.getElementById('livesValue').textContent = this.lives;
                document.getElementById('levelValue').textContent = this.level;
            }
            
            gameLoop(currentTime) {
                const deltaTime = currentTime - this.lastTime;
                this.lastTime = currentTime;
                
                if (this.gameState === GameState.PLAYING) {
                    this.update(deltaTime);
                }
                
                this.render();
                requestAnimationFrame((time) => this.gameLoop(time));
            }
            
            update(deltaTime) {
                // 更新玩家坦克
                this.updatePlayerTank();
                
                // 更新子弹
                this.updateBullets();
                
                // 更新敌人
                this.updateEnemies(deltaTime);
                
                // 检查碰撞
                this.checkCollisions();
                
                // 检查游戏结束条件
                if (this.lives <= 0) {
                    this.gameOver();
                }
            }
            
            updatePlayerTank() {
                const tank = this.playerTank;
                let moved = false;
                
                // 移动控制
                if (this.keys['ArrowUp'] && tank.y > 30) {
                    tank.y -= tank.speed;
                    tank.direction = 'up';
                    moved = true;
                }
                if (this.keys['ArrowDown'] && tank.y < this.canvas.height - tank.height - 30) {
                    tank.y += tank.speed;
                    tank.direction = 'down';
                    moved = true;
                }
                if (this.keys['ArrowLeft'] && tank.x > 30) {
                    tank.x -= tank.speed;
                    tank.direction = 'left';
                    moved = true;
                }
                if (this.keys['ArrowRight'] && tank.x < this.canvas.width - tank.width - 30) {
                    tank.x += tank.speed;
                    tank.direction = 'right';
                    moved = true;
                }
                
                // 射击控制
                if (this.keys[' '] && this.bulletCooldown <= 0) {
                    this.createBullet(tank, 'player');
                    this.bulletCooldown = 20; // 射击冷却时间
                }
                
                // 减少射击冷却
                if (this.bulletCooldown > 0) {
                    this.bullet
