import pygame
import random
import math
import sys
from enum import Enum
from typing import List, Tuple, Optional

# 初始化Pygame
pygame.init()

# 游戏常量
SCREEN_WIDTH = 800
SCREEN_HEIGHT = 600
FPS = 60

# 颜色定义
BLACK = (0, 0, 0)
WHITE = (255, 255, 255)
RED = (255, 0, 0)
GREEN = (0, 255, 0)
BLUE = (0, 0, 255)
YELLOW = (255, 255, 0)
PURPLE = (128, 0, 128)

# 游戏状态枚举
class GameState(Enum):
    MENU = 1
    PLAYING = 2
    PAUSED = 3
    GAME_OVER = 4

# 玩家飞船类
class Player(pygame.sprite.Sprite):
    def __init__(self):
        super().__init__()
        self.image = pygame.Surface((40, 40), pygame.SRCALPHA)
        # 绘制飞船形状
        pygame.draw.polygon(self.image, BLUE, [(20, 0), (0, 40), (20, 30), (40, 40)])
        self.rect = self.image.get_rect(center=(SCREEN_WIDTH // 2, SCREEN_HEIGHT - 50))
        self.speed = 5
        self.health = 100
        self.max_health = 100
        self.shoot_cooldown = 0
        self.shoot_delay = 10  # 射击冷却时间
        
    def update(self):
        # 获取按键状态
        keys = pygame.key.get_pressed()
        
        # 移动控制
        if keys[pygame.K_LEFT] and self.rect.left > 0:
            self.rect.x -= self.speed
        if keys[pygame.K_RIGHT] and self.rect.right < SCREEN_WIDTH:
            self.rect.x += self.speed
        if keys[pygame.K_UP] and self.rect.top > 0:
            self.rect.y -= self.speed
        if keys[pygame.K_DOWN] and self.rect.bottom < SCREEN_HEIGHT:
            self.rect.y += self.speed
            
        # 射击控制
        if keys[pygame.K_SPACE] and self.shoot_cooldown <= 0:
            self.shoot()
            self.shoot_cooldown = self.shoot_delay
            
        # 更新射击冷却
        if self.shoot_cooldown > 0:
            self.shoot_cooldown -= 1
            
    def shoot(self):
        bullet = Bullet(self.rect.centerx, self.rect.top)
        all_sprites.add(bullet)
        bullets.add(bullet)
        
    def take_damage(self, damage):
        self.health -= damage
        if self.health <= 0:
            self.health = 0
            return True  # 玩家死亡
        return False

# 子弹类
class Bullet(pygame.sprite.Sprite):
    def __init__(self, x, y):
        super().__init__()
        self.image = pygame.Surface((4, 10))
        self.image.fill(YELLOW)
        self.rect = self.image.get_rect(centerx=x, bottom=y)
        self.speed = 10
        
    def update(self):
        self.rect.y -= self.speed
        # 如果子弹超出屏幕，则删除
        if self.rect.bottom < 0:
            self.kill()

# 敌人类
class Enemy(pygame.sprite.Sprite):
    def __init__(self, x, y):
        super().__init__()
        self.image = pygame.Surface((30, 30), pygame.SRCALPHA)
        # 绘制敌人形状
        pygame.draw.polygon(self.image, RED, [(15, 30), (0, 0), (30, 0)])
        self.rect = self.image.get_rect(center=(x, y))
        self.speed = random.uniform(1, 3)
        self.health = 30
        self.score_value = 10
        
    def update(self):
        self.rect.y += self.speed
        # 如果敌人超出屏幕，则删除
        if self.rect.top > SCREEN_HEIGHT:
            self.kill()

# 陨石类
class Asteroid(pygame.sprite.Sprite):
    def __init__(self, x, y):
        super().__init__()
        size = random.randint(20, 50)
        self.image = pygame.Surface((size, size), pygame.SRCALPHA)
        # 绘制陨石形状
        pygame.draw.circle(self.image, (139, 69, 19), (size // 2, size // 2), size // 2)
        self.rect = self.image.get_rect(center=(x, y))
        self.speed = random.uniform(1, 4)
        self.rotation = 0
        self.rotation_speed = random.uniform(-5, 5)
        
    def update(self):
        self.rect.y += self.speed
        self.rotation += self.rotation_speed
        # 如果陨石超出屏幕，则删除
        if self.rect.top > SCREEN_HEIGHT:
            self.kill()

# 爆炸效果类
class Explosion(pygame.sprite.Sprite):
    def __init__(self, center):
        super().__init__()
        self.size = 30
        self.image = pygame.Surface((self.size * 2, self.size * 2), pygame.SRCALPHA)
        self.rect = self.image.get_rect(center=center)
        self.frame = 0
        self.last_frame = 8
        self.update()
        
    def update(self):
        self.frame += 1
        if self.frame > self.last_frame:
            self.kill()
        else:
            # 绘制爆炸动画
            self.image.fill((0, 0, 0, 0))
            size = self.size * (1 - self.frame / self.last_frame)
            pygame.draw.circle(self.image, (255, 165, 0), (self.size, self.size), int(size))
            pygame.draw.circle(self.image, (255, 255, 0), (self.size, self.size), int(size * 0.7))

# 星星背景类
class Star(pygame.sprite.Sprite):
    def __init__(self):
        super().__init__()
        self.image = pygame.Surface((2, 2))
        self.image.fill(WHITE)
        self.rect = self.image.get_rect(
            x=random.randint(0, SCREEN_WIDTH),
            y=random.randint(0, SCREEN_HEIGHT)
        )
        self.speed = random.uniform(0.5, 2)
        
    def update(self):
        self.rect.y += self.speed
        if self.rect.top > SCREEN_HEIGHT:
            self.rect.y = -2
            self.rect.x = random.randint(0, SCREEN_WIDTH)

# 游戏类
class SpaceShooterGame:
    def __init__(self):
        self.screen = pygame.display.set_mode((SCREEN_WIDTH, SCREEN_HEIGHT))
        pygame.display.set_caption("太空射击游戏")
        self.clock = pygame.time.Clock()
        self.font = pygame.font.SysFont(None, 36)
        self.small_font = pygame.font.SysFont(None, 24)
        self.state = GameState.MENU
        self.score = 0
        self.high_score = 0
        self.difficulty = 1
        self.enemy_spawn_timer = 0
        self.asteroid_spawn_timer = 0
        self.enemy_spawn_delay = 60  # 初始敌人生成延迟
        self.asteroid_spawn_delay = 120  # 初始陨石生成延迟
        
        # 初始化精灵组
        global all_sprites, bullets, enemies, asteroids, explosions
        all_sprites = pygame.sprite.Group()
        bullets = pygame.sprite.Group()
        enemies = pygame.sprite.Group()
        asteroids = pygame.sprite.Group()
        explosions = pygame.sprite.Group()
        
        # 创建玩家
        self.player = Player()
        all_sprites.add(self.player)
        
        # 创建星星背景
        for _ in range(50):
            star = Star()
            all_sprites.add(star)
            
    def handle_events(self):
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                pygame.quit()
                sys.exit()
                
            if self.state == GameState.MENU:
                if event.type == pygame.KEYDOWN:
                    if event.key == pygame.K_RETURN:
                        self.start_game()
                    elif event.key == pygame.K_q:
                        pygame.quit()
                        sys.exit()
                        
            elif self.state == GameState.PLAYING:
                if event.type == pygame.KEYDOWN:
                    if event.key == pygame.K_p:
                        self.state = GameState.PAUSED
                        
            elif self.state == GameState.PAUSED:
                if event.type == pygame.KEYDOWN:
                    if event.key == pygame.K_p:
                        self.state = GameState.PLAYING
                        
            elif self.state == GameState.GAME_OVER:
                if event.type == pygame.KEYDOWN:
                    if event.key == pygame.K_RETURN:
                        self.start_game()
                    elif event.key == pygame.K_q:
                        pygame.quit()
                        sys.exit()
    
    def start_game(self):
        self.state = GameState.PLAYING
        self.score = 0
        self.difficulty = 1
        self.enemy_spawn_timer = 0
        self.asteroid_spawn_timer = 0
        self.enemy_spawn_delay = 60
        self.asteroid_spawn_delay = 120
        
        # 重置精灵组
        all_sprites.empty()
        bullets.empty()
        enemies.empty()
        asteroids.empty()
        explosions.empty()
        
        # 重新创建玩家
        self.player = Player()
        all_sprites.add(self.player)
        
        # 重新创建星星背景
        for _ in range(50):
            star = Star()
            all_sprites.add(star)
    
    def update(self):
        if self.state == GameState.PLAYING:
            # 更新所有精灵
            all_sprites.update()
            
            # 生成敌人
            self.enemy_spawn_timer += 1
            if self.enemy_spawn_timer >= self.enemy_spawn_delay:
                self.enemy_spawn_timer = 0
                enemy = Enemy(random.randint(30, SCREEN_WIDTH - 30), -30)
                all_sprites.add(enemy)
                enemies.add(enemy)
                
            # 生成陨石
            self.asteroid_spawn_timer += 1
            if self.asteroid_spawn_timer >= self.asteroid_spawn_delay:
                self.asteroid_spawn_timer = 0
                asteroid = Asteroid(random.randint(20, SCREEN_WIDTH - 20), -20)
                all_sprites.add(asteroid)
                asteroids.add(asteroid)
            
            # 检测子弹与敌人的碰撞
            for bullet in bullets:
                hit_enemies = pygame.sprite.spritecollide(bullet, enemies, False)
                for enemy in hit_enemies:
                    bullet.kill()
                    enemy.health -= 10
                    if enemy.health <= 0:
                        enemy.kill()
                        explosion = Explosion(enemy.rect.center)
                        all_sprites.add(explosion)
                        explosions.add(explosion)
                        self.score += enemy.score_value
                        self.update_difficulty()
            
            # 检测子弹与陨石的碰撞
            for bullet in bullets:
                hit_asteroids = pygame.sprite.spritecollide(bullet, asteroids, False)
                for asteroid in hit_asteroids:
                    bullet.kill()
                    asteroid.kill()
                    explosion = Explosion(asteroid.rect.center)
                    all_sprites.add(explosion)
                    explosions.add(explosion)
                    self.score += 5
                    self.update_difficulty()
            
            # 检测玩家与敌人的碰撞
            player_hit_enemies = pygame.sprite.spritecollide(self.player, enemies, True)
            for enemy in player_hit_enemies:
                explosion = Explosion(enemy.rect.center)
                all_sprites.add(explosion)
                explosions.add(explosion)
                if self.player.take_damage(20):
                    self.game_over()
            
            # 检测玩家与陨石的碰撞
            player_hit_asteroids = pygame.sprite.spritecollide(self.player, asteroids, True)
            for asteroid in player_hit_asteroids:
                explosion = Explosion(asteroid.rect.center)
                all_sprites.add(explosion)
                explosions.add(explosion)
                if self.player.take_damage(30):
                    self.game_over()
    
    def update_difficulty(self):
        # 根据分数更新难度
        new_difficulty = 1 + self.score // 100
        if new_difficulty > self.difficulty:
            self.difficulty = new_difficulty
            # 减少生成延迟，增加难度
            self.enemy_spawn_delay = max(20, 60 - (self.difficulty - 1) * 5)
            self.asteroid_spawn_delay = max(40, 120 - (self.difficulty - 1) * 10)
    
    def draw(self):
        self.screen.fill(BLACK)
        
        if self.state == GameState.MENU:
            self.draw_menu()
        elif self.state == GameState.PLAYING:
            self.draw_game()
        elif self.state == GameState.PAUSED:
            self.draw_game()
            self.draw_pause_screen()
        elif self.state == GameState.GAME_OVER:
            self.draw_game_over()
            
        pygame.display.flip()
    
    def draw_menu(self):
        title_text = self.font.render("太空射击游戏", True, WHITE)
        title_rect = title_text.get_rect(center=(SCREEN_WIDTH // 2, 150))
        self.screen.blit(title_text, title_rect)
        
        start_text = self.font.render("按 Enter 开始游戏", True, WHITE)
        start_rect = start_text.get_rect(center=(SCREEN_WIDTH // 2, 300))
        self.screen.blit(start_text, start_rect)
        
        quit_text = self.font.render("按 Q 退出游戏", True, WHITE)
        quit_rect = quit_text.get_rect(center=(SCREEN_WIDTH // 2, 350))
        self.screen.blit(quit_text, quit_rect)
        
        controls_text = self.small_font.render("控制: 方向键移动, 空格键射击, P键暂停", True, WHITE)
        controls_rect = controls_text.get_rect(center=(SCREEN_WIDTH // 2, 450))
        self.screen.blit(controls_text, controls_rect)
    
    def draw_game(self):
        # 绘制所有精灵
        all_sprites.draw(self.screen)
        
        # 绘制分数
        score_text = self.font.render(f"分数: {self.score}", True, WHITE)
        self.screen.blit(score_text, (10, 10))
        
        # 绘制难度
        difficulty_text = self.small_font.render(f"难度: {self.difficulty}", True, WHITE)
        self.screen.blit(difficulty_text, (10, 50))
        
        # 绘制生命值条
        pygame.draw.rect(self.screen, RED, (10, 80, 200, 20))
        pygame.draw.rect(self.screen, GREEN, (10, 80, 200 * (self.player.health / self.player.max_health), 20))
        
        # 绘制暂停提示
        pause_text = self.small_font.render("按 P 暂停", True, WHITE)
        self.screen.blit(pause_text, (SCREEN_WIDTH - 100, 10))
    
    def draw_pause_screen(self):
        # 绘制半透明覆盖层
        overlay = pygame.Surface((SCREEN_WIDTH, SCREEN_HEIGHT))
        overlay.set_alpha(128)
        overlay.fill(BLACK)
        self.screen.blit(overlay, (0, 0))
        
        # 绘制暂停文本
        pause_text = self.font.render("游戏暂停", True, WHITE)
        pause_rect = pause_text.get_rect(center=(SCREEN_WIDTH // 2, SCREEN_HEIGHT // 2))
        self.screen.blit(pause_text, pause_rect)
        
        resume_text = self.small_font.render("按 P 继续游戏", True, WHITE)
        resume_rect = resume_text.get_rect(center=(SCREEN_WIDTH // 2, SCREEN_HEIGHT // 2 + 50))
        self.screen.blit(resume_text, resume_rect)
    
    def draw_game_over(self):
        # 绘制半透明覆盖层
        overlay = pygame.Surface((SCREEN_WIDTH, SCREEN_HEIGHT))
        overlay.set_alpha(128)
        overlay.fill(BLACK)
        self.screen.blit(overlay, (0, 0))
        
        # 绘制游戏结束文本
        game_over_text = self.font.render("游戏结束", True, WHITE)
        game_over_rect = game_over_text.get_rect(center=(SCREEN_WIDTH // 2, SCREEN_HEIGHT // 2 - 50))
        self.screen.blit(game_over_text, game_over_rect)
        
        # 绘制分数
        score_text = self.font.render(f"最终分数: {self.score}", True, WHITE)
        score_rect = score_text.get_rect(center=(SCREEN_WIDTH // 2, SCREEN_HEIGHT // 2))
        self.screen.blit(score_text, score_rect)
        
        # 绘制最高分
        high_score_text = self.font.render(f"最高分: {self.high_score}", True, WHITE)
        high_score_rect = high_score_text.get_rect(center=(SCREEN_WIDTH // 2, SCREEN_HEIGHT // 2 + 50))
        self.screen.blit(high_score_text, high_score_rect)
        
        # 绘制重新开始提示
        restart_text = self.font.render("按 Enter 重新开始", True, WHITE)
        restart_rect = restart_text.get_rect(center=(SCREEN_WIDTH // 2, SCREEN_HEIGHT // 2 + 100))
        self.screen.blit(restart_text, restart_rect)
        
        # 绘制退出提示
        quit_text = self.font.render("按 Q 退出游戏", True, WHITE)
        quit_rect = quit_text.get_rect(center=(SCREEN_WIDTH // 2, SCREEN_HEIGHT // 2 + 150))
        self.screen.blit(quit_text, quit_rect)
    
    def game_over(self):
        self.state = GameState.GAME_OVER
        if self.score > self.high_score:
            self.high_score = self.score
    
    def run(self):
        while True:
            self.handle_events()
            self.update()
            self.draw()
            self.clock.tick(FPS)

# 全局变量
all_sprites = None
bullets = None
enemies = None

explosions = None

# 主程序入口
if __name__ == "__main__":
    game = SpaceShooterGame()
