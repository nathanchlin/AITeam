## 游戏对象基类

```python
class GameObject:
    """游戏对象基类"""
    def __init__(self, x: float, y: float, width: float, height: float, color: Tuple[int, int, int]):
        self.x = x
        self.y = y
        self.width = width
        self.height = height
        self.color = color
        self.active = True
    
    def update(self):
        """更新游戏对象状态"""
        pass
    
    def draw(self, screen):
        """绘制游戏对象"""
        pygame.draw.rect(screen, self.color, (self.x, self.y, self.width, self.height))
    
    def get_rect(self):
        """获取游戏对象的矩形碰撞区域"""
        return pygame.Rect(self.x, self.y, self.width, self.height)
    
    def is_off_screen(self):
        """检查对象是否离开屏幕"""
        return (self.x < -self.width or self.x > SCREEN_WIDTH or 
                self.y < -self.height or self.y > SCREEN_HEIGHT)
```

class Player(GameObject):
    """玩家飞船类"""
    def __init__(self, x: float, y: float):
        super().__init__(x, y, 40, 40, BLUE)
        self.speed = GameConfig.player_speed
        self.health = 100
        self.max_health = 100
        self.shoot_cooldown = 0
        self.shoot_delay = 10  # 射击冷却时间（帧数）
    
    def update(self):
        """更新玩家状态"""
        # 获取键盘输入
        keys = pygame.key.get_pressed()
        
        # 移动控制
        if keys[pygame.K_LEFT] or keys[pygame.K_a]:
            self.x -= self.speed
        if keys[pygame.K_RIGHT] or keys[pygame.K_d]:
            self.x += self.speed
        if keys[pygame.K_UP] or keys[pygame.K_w]:
            self.y -= self.speed
        if keys[pygame.K_DOWN] or keys[pygame.K_s]:
            self.y += self.speed
        
        # 限制玩家在屏幕范围内
        self.x = max(0, min(self.x, SCREEN_WIDTH - self.width))
        self.y = max(0, min(self.y, SCREEN_HEIGHT - self.height))
        
        # 更新射击冷却
        if self.shoot_cooldown > 0:
            self.shoot_cooldown -= 1
    
    def shoot(self) -> 'Bullet':
        """发射子弹"""
        if self.shoot_cooldown <= 0:
            self.shoot_cooldown = self.shoot_delay
            return Bullet(self.x + self.width // 2 - 2, self.y)
        return None
    
    def take_damage(self, damage: int):
        """受到伤害"""
        self.health -= damage
        if self.health <= 0:
            self.active = False
    
    def draw(self, screen):
        """绘制玩家飞船"""
        # 绘制飞船主体
        pygame.draw.polygon(screen, self.color, [
            (self.x + self.width // 2, self.y),
            (self.x, self.y + self.height),
            (self.x + self.width, self.y + self.height)
        ])
        
        # 绘制生命值条
        bar_width = 40
        bar_height = 5
        bar_x = self.x
        bar_y = self.y - 10
        
        # 背景
        pygame.draw.rect(screen, RED, (bar_x, bar_y, bar_width, bar_height))
        # 当前生命值
        health_width = int(bar_width * (self.health / self.max_health))
        pygame.draw.rect(screen, GREEN, (bar_x, bar_y, health_width, bar_height))
```

## 子弹类

class Bullet(GameObject):
    """子弹类"""
    def __init__(self, x: float, y: float, is_enemy_bullet: bool = False):
        color = RED if is_enemy_bullet else YELLOW
        super().__init__(x, y, 4, 10, color)
        self.speed = GameConfig.bullet_speed
        self.is_enemy_bullet = is_enemy_bullet
    
    def update(self):
        """更新子弹位置"""
        if self.is_enemy_bullet:
            self.y += self.speed
        else:
            self.y -= self.speed
        
        # 如果子弹离开屏幕，标记为非活动状态
        if self.is_off_screen():
            self.active = False
```

class Enemy(GameObject):
    """敌人类"""
    def __init__(self, x: float, y: float):
        super().__init__(x, y, 30, 30, RED)
        self.speed = GameConfig.enemy_speed
        self.shoot_cooldown = 0
        self.shoot_delay = random.randint(60, 120)  # 随机射击间隔
        self.health = 30
        self.score_value = 10
    
    def update(self):
        """更新敌人状态"""
        # 向下移动
        self.y += self.speed
        
        # 更新射击冷却
        if self.shoot_cooldown > 0:
            self.shoot_cooldown -= 1
        
        # 如果敌人离开屏幕，标记为非活动状态
        if self.is_off_screen():
            self.active = False
    
    def shoot(self) -> Optional[Bullet]:
        """发射子弹"""
        if self.shoot_cooldown <= 0 and random.random() < 0.01:  # 1%的概率每帧射击
            self.shoot_cooldown = self.shoot_delay
            return Bullet(self.x + self.width // 2 - 2, self.y + self.height, is_enemy_bullet=True)
        return None
    
    def take_damage(self, damage: int):
        """受到伤害"""
        self.health -= damage
        if self.health <= 0:
            self.active = False

class Asteroid(GameObject):
    """陨石类"""
    def __init__(self, x: float, y: float):
        size = random.randint(20, 50)
        super().__init__(x, y, size, size, (150, 150, 150))
        self.speed = random.uniform(1, 3)
        self.rotation = 0
        self.rotation_speed = random.uniform(-2, 2)
        self.health = size // 10
        self.score_value = size // 5
    
    def update(self):
        """更新陨石状态"""
        # 向下移动
        self.y += self.speed
        
        # 旋转
        self.rotation += self.rotation_speed
        
        # 如果陨石离开屏幕，标记为非活动状态
        if self.is_off_screen():
            self.active = False
    
    def take_damage(self, damage: int):
        """受到伤害"""
        self.health -= damage
        if self.health <= 0:
            self.active = False
    
    def draw(self, screen):
        """绘制陨石"""
        # 创建旋转的陨石形状
        points = []
        num_points = 8
        for i in range(num_points):
            angle = (2 * math.pi * i / num_points) + math.radians(self.rotation)
            radius = self.width / 2 + random.uniform(-5, 5)
            x = self.x + self.width / 2 + radius * math.cos(angle)
            y = self.y + self.height / 2 + radius * math.sin(angle)
            points.append((x, y))
        
        pygame.draw.polygon(screen, self.color, points)
```

class Particle:
    """粒子效果类"""
    def __init__(self, x: float, y: float, color: Tuple[int, int, int]):
        self.x = x
        self.y = y
        self.color = color
        self.size = random.randint(2, 5)
        self.speed_x = random.uniform(-3, 3)
        self.speed_y = random.uniform(-3, 3)
        self.lifetime = random.randint(20, 40)
        self.active = True
    
    def update(self):
        """更新粒子状态"""
        self.x += self.speed_x
        self.y += self.speed_y
        self.lifetime -= 1
        self.size = max(1, self.size - 0.1)
        
        if self.lifetime <= 0:
            self.active = False
    
    def draw(self, screen):
        """绘制粒子"""
        pygame.draw.circle(screen, self.color, (int(self.x), int(self.y)), int(self.size))

## 游戏引擎主类

```python
class SpaceShooterGame:
    """太空射击游戏主类"""
    def __init__(self):
        self.screen = pygame.display.set_mode((SCREEN_WIDTH, SCREEN_HEIGHT))
        pygame.display.set_caption("太空射击游戏")
        self.clock = pygame.time.Clock()
        self.font = pygame.font.SysFont(None, 36)
        self.small_font = pygame.font.SysFont(None, 24)
        
        # 游戏状态
        self.state = GameState.MENU
        self.running = True
        self.score = 0
        self.high_score = 0
        
        # 游戏对象
        self.player = Player(SCREEN_WIDTH // 2 - 20, SCREEN_HEIGHT - 60)
        self.bullets: List[Bullet] = []
        self.enemies: List[Enemy] = []
        self.asteroids: List[Asteroid] = []
        self.particles: List[Particle] = []
        
        # 游戏配置
        self.config = GameConfig()
        self.difficulty = 1.0
        
        # 背景星星
        self.stars = [(random.randint(0, SCREEN_WIDTH), 
                      random.randint(0, SCREEN_HEIGHT)) 
                     for _ in range(100)]
    
    def handle_events(self):
        """处理游戏事件"""
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                self.running = False
            
            if self.state == GameState.MENU:
                if event.type == pygame.KEYDOWN:
                    if event.key == pygame.K_SPACE:
                        self.start_game()
            
            elif self.state == GameState.PLAYING:
                if event.type == pygame.KEYDOWN:
                    if event.key == pygame.K_p:
                        self.state = GameState.PAUSED
                    elif event.key == pygame.K_ESCAPE:
                        self.state = GameState.MENU
            
            elif self.state == GameState.PAUSED:
                if event.type == pygame.KEYDOWN:
                    if event.key == pygame.K_p:
                        self.state = GameState.PLAYING
                    elif event.key == pygame.K_ESCAPE:
                        self.state = GameState.MENU
            
            elif self.state == GameState.GAME_OVER:
                if event.type == pygame.KEYDOWN:
                    if event.key == pygame.K_SPACE:
                        self.start_game()
                    elif event.key == pygame.K_ESCAPE:
                        self.state = GameState.MENU
    
    def start_game(self):
        """开始新游戏"""
        self.state = GameState.PLAYING
        self.score = 0
        self.difficulty = 1.0
        self.player = Player(SCREEN_WIDTH // 2 - 20, SCREEN_HEIGHT - 60)
        self.bullets.clear()
        self.enemies.clear()
        self.asteroids.clear()
        self.particles.clear()
    
    def update(self):
        """更新游戏状态"""
        if self.state == GameState.PLAYING:
            # 更新玩家
            self.player.update()
            
            # 玩家射击
            if pygame.key.get_pressed()[pygame.K_SPACE]:
                bullet = self.player.shoot()
                if bullet:
                    self.bullets.append(bullet)
            
            # 生成敌人
            if random.random() < self.config.enemy_spawn_rate * self.difficulty:
                if len(self.enemies) < self.config.max_enemies:
                    x = random.randint(0, SCREEN_WIDTH - 30)
                    self.enemies.append(Enemy(x, -30))
            
            # 生成陨石
            if random.random() < 0.01 * self.difficulty:
                x = random.randint(0, SCREEN_WIDTH - 50)
                self.asteroids.append(Asteroid(x, -50))
            
            # 更新子弹
            for bullet in self.bullets[:]:
                bullet.update()
                if not bullet.active:
                    self.bullets.remove(bullet)
            
            # 更新敌人
            for enemy in self.enemies[:]:
                enemy.update()
                
                # 敌人射击
                enemy_bullet = enemy.shoot()
                if enemy_bullet:
                    self.bullets.append(enemy_bullet)
                
                if not enemy.active:
                    self.enemies.remove(enemy)
            
            # 更新陨石
            for asteroid in self.asteroids[:]:
                asteroid.update()
                if not asteroid.active:
                    self.asteroids.remove(asteroid)
            
            # 更新粒子
            for particle in self.particles[:]:
                particle.update()
                if not particle.active:
                    self.particles.remove(particle)
            
            # 碰撞检测
            self.check_collisions()
            
            # 增加难度
            self.difficulty += self.config.difficulty_increase_rate
            
            # 更新星星背景
            for i in range(len(self.stars)):
                x, y = self.stars[i]
                y += 1
                if y > SCREEN_HEIGHT:
                    y = 0
                    x = random.randint(0, SCREEN_WIDTH)
                self.stars[i] = (x, y)
            
            # 检查游戏结束
            if not self.player.active:
                self.state = GameState.GAME_OVER
                if self.score > self.high_score:
                    self.high_score = self.score
    
    def check_collisions(self):
        """检测碰撞"""
        # 玩家子弹与敌人/陨石碰撞
        for bullet in self.bullets[:]:
            if not bullet.is_enemy_bullet:
                # 检测与敌人的碰撞
                for enemy in self.enemies:
                    if bullet.get_rect().colliderect(enemy.get_rect()):
                        enemy.take_damage(10)
                        bullet.active = False
                        if not enemy.active:
                            self.score += enemy.score_value
                            self.create_explosion(enemy.x + enemy.width // 2, 
                                                enemy.y + enemy.height // 2, 
                                                enemy.color)
                        break
                
                # 检测与陨石的碰撞
                for asteroid in self.asteroids:
                    if bullet.get_rect().colliderect(asteroid.get_rect()):
                        asteroid.take_damage(10)
                        bullet.active = False
                        if not asteroid.active:
                            self.score += asteroid.score_value
                            self.create_explosion(asteroid.x + asteroid.width // 2, 
                                                asteroid.y + asteroid.height // 2, 
                                                asteroid.color)
                        break
        
        # 敌人子弹与玩家碰撞
        for bullet in self.bullets[:]:
            if bullet.is_enemy_bullet:
                if bullet.get_rect().colliderect(self.player.get_rect()):
                    self.player.take_damage(10)
                    bullet.active = False
                    self.create_explosion(bullet.x, bullet.y, RED)
        
        # 玩家与敌人/陨石碰撞
        for enemy in self.enemies:
            if self.player.get_rect().colliderect(enemy.get_rect()):
                self.player.take_damage(20)
                enemy.active = False
                self.create_explosion(enemy.x + enemy.width // 2, 
                                    enemy.y + enemy.height // 2, 
                                    enemy.color)
        
        for asteroid in self.asteroids:
            if self.player.get_rect().colliderect(asteroid.get_rect()):
                self.player.take_damage(30)
                asteroid.active = False
                self.create_explosion(asteroid.x + asteroid.width // 2, 
                                    asteroid.y + asteroid.height // 2, 
                                    asteroid.color)
    
    def create_explosion(self, x: float, y: float, color: Tuple[int, int, int]):
        """创建爆炸效果"""
        for _ in range(20):
            self.particles.append(Particle(x, y, color))
    
    def draw(self):
        """绘制游戏画面"""
        # 清空屏幕
        self.screen.fill(BLACK)
        
        # 绘制星星背景
        for star in self.stars:
            pygame.draw.circle(self.screen, WHITE, star, 1)
        
        if self.state == GameState.MENU:
            self.draw_menu()
        elif self.state == GameState.PLAYING:
            self.draw_game()
        elif self.state == GameState.PAUSED:
            self.draw_game()
            self.draw_pause_overlay()
        elif self.state == GameState.GAME_OVER:
            self.draw_game_over()
        
        # 更新显示
        pygame.display.flip()
    
    def draw_menu(self):
        """绘制主菜单"""
        title = self.font.render("太空射击游戏", True, WHITE)
        title_rect = title.get_rect(center=(SCREEN_WIDTH // 2, 150))
        self.screen.blit(title, title_rect)
        
        start_text = self.font.render("按空格键开始游戏", True, WHITE)
        start_rect = start_text.get_rect(center=(SCREEN_WIDTH // 2, 300))
        self.screen.blit(start_text, start_rect)
        
        controls = [
            "控制说明:",
            "WASD/方向键 - 移动",
            "空格键 - 射击",
            "P - 暂停游戏",
            "ESC - 返回菜单"
        ]
        
        y = 400
        for line in controls:
            text = self.small_font.render(line, True, WHITE)
            text_rect = text.get_rect(center=(SCREEN_WIDTH // 2, y))
            self.screen.blit(text
