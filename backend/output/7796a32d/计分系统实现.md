    <script>
        // 游戏配置
        const config = {
            type: Phaser.AUTO,
            width: 800,
            height: 600,
            physics: {
                default: 'arcade',
                arcade: {
                    gravity: { y: 600 },
                    debug: false
                }
            },
            scene: {
                preload: preload,
                create: create,
                update: update
            }
        };

        // 创建游戏实例
        const game = new Phaser.Game(config);

        // 游戏变量
        let player;
        let platforms;
        let cursors;
        let score = 0;
        let floor = 1;
        let highScore = localStorage.getItem('highScore') || 0;
        let gameStarted = false;
        let gameOver = false;
        let platformSpeed = 2;
        let platformSpawnTimer = 0;
        let platformSpawnInterval = 2000; // 毫秒
        let lastPlatformY = 0;
        let cameraSpeed = 1;
        let jumpCount = 0;
        let maxJumps = 2; // 允许二段跳
        let isPaused = false;
        let scoreMultiplier = 1;
        let comboTimer = 0;
        let combo = 0;

        // 预加载资源
        function preload() {
            // 创建简单的图形作为游戏元素
            this.add.graphics();
        }

        // 创建游戏场景
        function create() {
            // 设置背景
            this.cameras.main.setBackgroundColor('#16213e');
            
            // 创建玩家
            player = this.physics.add.sprite(400, 100, 'player');
            player.setBounce(0.2);
            player.setCollideWorldBounds(true);
            player.body.setGravityY(600);
            player.setData('isJumping', false);
            
            // 创建平台组
            platforms = this.physics.add.group({
                immovable: true,
                allowGravity: false
            });
            
            // 初始化平台
            generateInitialPlatforms.call(this);
            
            // 设置碰撞检测
            this.physics.add.collider(player, platforms, handlePlatformCollision, null, this);
            
            // 设置键盘控制
            cursors = this.input.keyboard.createCursorKeys();
            this.input.keyboard.on('keydown', handleKeyDown, this);
            this.input.keyboard.on('keyup', handleKeyUp, this);
            
            // 更新最高分显示
            document.getElementById('high-score-display').textContent = highScore;
            
            // 设置按钮事件
            document.getElementById('start-button').addEventListener('click', startGame);
            document.getElementById('restart-button').addEventListener('click', restartGame);
        }

        // 生成初始平台
        function generateInitialPlatforms() {
            // 底部平台
            createPlatform.call(this, 400, 550, 200, 20);
            
            // 初始平台
            for (let i = 1; i <= 10; i++) {
                const x = Phaser.Math.Between(100, 700);
                const y = 550 - i * 60;
                createPlatform.call(this, x, y, 150, 15);
            }
            
            lastPlatformY = 550 - 10 * 60;
        }

        // 创建平台
        function createPlatform(x, y, width, height) {
            const platform = platforms.create(x, y, 'platform');
            platform.displayWidth = width;
            platform.displayHeight = height;
            platform.setData('type', 'normal');
            platform.setData('scored', false);
        }

        // 创建特殊平台
        function createSpecialPlatform(x, y, type) {
            const platform = platforms.create(x, y, 'platform');
            platform.displayWidth = 120;
            platform.displayHeight = 15;
            platform.setData('type', type);
            platform.setData('scored', false);
            
            // 根据类型设置颜色
            if (type === 'moving') {
                platform.setData('moveSpeed', Phaser.Math.Between(50, 150));
                platform.setData('moveDirection', Phaser.Math.Between(0, 1) === 0 ? -1 : 1);
                platform.setData('minX', 50);
                platform.setData('maxX', 750);
            } else if (type === 'breakable') {
                platform.setData('breakTimer', 0);
                platform.setData('isBreaking', false);
            } else if (type === 'bonus') {
                platform.setData('points', 100);
                platform.setTint(0xF39C12); // 金色
            }
        }

        // 游戏更新循环
        function update(time, delta) {
            if (!gameStarted || gameOver || isPaused) return;
            
            // 更新相机位置
            if (player.y > this.cameras.main.scrollY + 400) {
                this.cameras.main.startFollow(player, false, 0.1, 0.1);
            }
            
            // 移动相机
            this.cameras.main.scrollY -= cameraSpeed;
            
            // 更新平台位置
            platforms.children.entries.forEach(platform => {
                // 移动平台
                if (platform.getData('type') === 'moving') {
                    const moveSpeed = platform.getData('moveSpeed');
                    const moveDirection = platform.getData('moveDirection');
                    const minX = platform.getData('minX');
                    const maxX = platform.getData('maxX');
                    
                    platform.x += moveSpeed * moveDirection * delta / 1000;
                    
                    if (platform.x <= minX || platform.x >= maxX) {
                        platform.setData('moveDirection', -moveDirection);
                    }
                }
                
                // 平台掉落效果
                if (platform.getData('type') === 'breakable' && platform.getData('isBreaking')) {
                    platform.y += 5;
                    platform.alpha -= 0.02;
                    
                    if (platform.alpha <= 0) {
                        platform.destroy();
                    }
                }
                
                // 标记已计分平台
                if (!platform.getData('scored') && player.y > platform.y + 50) {
                    platform.setData('scored', true);
                    scorePlatform.call(this, platform);
                }
            });
            
            // 生成新平台
            platformSpawnTimer += delta;
            if (platformSpawnTimer >= platformSpawnInterval) {
                platformSpawnTimer = 0;
                generateNewPlatform.call(this);
                
                // 随着游戏进行，增加难度
                if (floor % 10 === 0) {
                    platformSpeed += 0.2;
                    cameraSpeed += 0.1;
                    platformSpawnInterval = Math.max(1000, platformSpawnInterval - 100);
                }
            }
            
            // 更新连击计时器
            if (comboTimer > 0) {
                comboTimer -= delta;
                if (comboTimer <= 0) {
                    combo = 0;
                    scoreMultiplier = 1;
                }
            }
            
            // 玩家控制
            handlePlayerMovement.call(this, delta);
            
            // 检查游戏结束条件
            if (player.y > this.cameras.main.scrollY + 650) {
                endGame.call(this);
            }
        }

        // 处理平台碰撞
        function handlePlatformCollision(player, platform) {
            const platformType = platform.getData('type');
            
            if (platformType === 'normal' || platformType === 'moving' || platformType === 'bonus') {
                player.setVelocityY(0);
                player.setData('isJumping', false);
                jumpCount = 0;
                
                // 处理奖励平台
                if (platformType === 'bonus' && !platform.getData('scored')) {
                    platform.setData('scored', true);
                    scoreBonus.call(this, platform);
                }
            } else if (platformType === 'breakable') {
                if (!platform.getData('isBreaking')) {
                    platform.setData('isBreaking', true);
                    platform.setData('breakTimer', 1000); // 1秒后开始掉落
                }
            }
        }

        // 处理键盘按下事件
        function handleKeyDown(event) {
            if (!gameStarted || gameOver) return;
            
            // 空格键跳跃
            if (event.code === 'Space') {
                jump.call(this);
            }
            
            // P键暂停
            if (event.code === 'KeyP') {
                togglePause.call(this);
            }
        }

        // 处理键盘释放事件
        function handleKeyUp(event) {
            // 可以在这里添加键盘释放事件处理
        }

        // 跳跃功能
        function jump() {
            if (jumpCount < maxJumps) {
                player.setVelocityY(-400);
                player.setData('isJumping', true);
                jumpCount++;
            }
        }

        // 处理玩家移动
        function handlePlayerMovement(delta) {
            // 左右移动
            if (cursors.left.isDown) {
                player.setVelocityX(-200);
            } else if (cursors.right.isDown) {
                player.setVelocityX(200);
            } else {
                player.setVelocityX(0);
            }
            
            // 限制玩家在屏幕内
            if (player.x < 50) {
                player.x = 50;
            } else if (player.x > 750) {
                player.x = 750;
            }
        }

        // 生成新平台
        function generateNewPlatform() {
            const x = Phaser.Math.Between(100, 700);
            const y = lastPlatformY - Phaser.Math.Between(50, 80);
            
            // 根据楼层生成不同类型的平台
            const rand = Math.random();
            let platformType = 'normal';
            
            if (floor > 20 && rand < 0.2) {
                platformType = 'moving';
            } else if (floor > 30 && rand < 0.15) {
                platformType = 'breakable';
            } else if (floor > 40 && rand < 0.1) {
                platformType = 'bonus';
            }
            
            if (platformType === 'normal') {
                createPlatform.call(this, x, y, 150, 15);
            } else {
                createSpecialPlatform.call(this, x, y, platformType);
            }
            
            lastPlatformY = y;
        }

        // 计分平台
        function scorePlatform(platform) {
            const platformType = platform.getData('type');
            
            // 基础分数
            let points = 10 * scoreMultiplier;
            
            // 根据平台类型给予额外分数
            if (platformType === 'moving') {
                points = 20 * scoreMultiplier;
            } else if (platformType === 'bonus') {
                points = 100 * scoreMultiplier;
            }
            
            // 更新分数
            score += points;
            
            // 更新连击
            combo++;
            comboTimer = 2000; // 2秒内需要再次踩到平台
            
            if (combo >= 5) {
                scoreMultiplier = 2;
            } else if (combo >= 10) {
                scoreMultiplier = 3;
            } else if (combo >= 15) {
                scoreMultiplier = 5;
            }
            
            // 更新楼层
            const newFloor = Math.floor(score / 100) + 1;
            if (newFloor > floor) {
                floor = newFloor;
                // 楼层奖励
                score += 50;
            }
            
            // 更新UI
            updateUI();
        }

        // 奖励平台计分
        function scoreBonus(platform) {
            const points = platform.getData('points') * scoreMultiplier;
            score += points;
            updateUI();
            
            // 视觉效果
            platform.setTint(0
