# 匹配检测算法开发

我将为消消乐游戏开发一个核心的匹配检测算法，用于识别三个及以上相同宝石的水平和垂直匹配。

## 算法设计思路

匹配检测算法需要：
1. 遍历整个游戏网格
2. 检查每个位置的水平和垂直方向是否有连续相同宝石
3. 标记所有匹配的宝石
4. 返回匹配结果供后续处理（消除、计分等）

        // 游戏配置
        const BOARD_SIZE = 8;
        const GEM_TYPES = 6;
        const MIN_MATCH_LENGTH = 3;
        const INITIAL_MOVES = 30;
        
        // 游戏状态
        let board = [];
        let score = 0;
        let moves = INITIAL_MOVES;
        let selectedGem = null;
        let isProcessing = false;
        
        // 宝石图标
        const gemIcons = ['💎', '💚', '🧡', '💜', '🔷', '🔴'];
        
        // 初始化游戏
        function initGame() {
            board = [];
            score = 0;
            moves = INITIAL_MOVES;
            selectedGem = null;
            isProcessing = false;
            
            document.getElementById('score').textContent = score;
            document.getElementById('moves').textContent = moves;
            document.getElementById('gameOver').style.display = 'none';
            
            createBoard();
            renderBoard();
            
            // 确保初始状态没有匹配
            while (checkAllMatches().length > 0) {
                shuffleBoard();
            }
            renderBoard();
        }
        
        // 创建游戏板
        function createBoard() {
            for (let row = 0; row < BOARD_SIZE; row++) {
                board[row] = [];
                for (let col = 0; col < BOARD_SIZE; col++) {
                    board[row][col] = Math.floor(Math.random() * GEM_TYPES) + 1;
                }
            }
        }
        
        // 渲染游戏板
        function renderBoard() {
            const gameBoard = document.getElementById('gameBoard');
            gameBoard.innerHTML = '';
            
            for (let row = 0; row < BOARD_SIZE; row++) {
                for (let col = 0; col < BOARD_SIZE; col++) {
                    const gem = document.createElement('div');
                    gem.className = `gem gem-${board[row][col]}`;
                    gem.textContent = gemIcons[board[row][col] - 1];
                    gem.dataset.row = row;
                    gem.dataset.col = col;
                    gem.addEventListener('click', handleGemClick);
                    gameBoard.appendChild(gem);
                }
            }
        }
        
        // 处理宝石点击
        function handleGemClick(e) {
            if (isProcessing || moves <= 0) return;
            
            const row = parseInt(e.target.dataset.row);
            const col = parseInt(e.target.dataset.col);
            
            if (!selectedGem) {
                selectedGem = { row, col };
                e.target.classList.add('selected');
            } else {
                const prevRow = selectedGem.row;
                const prevCol = selectedGem.col;
                
                // 检查是否是相邻宝石
                if (isAdjacent(row, col, prevRow, prevCol)) {
                    isProcessing = true;
                    
                    // 交换宝石
                    swapGems(row, col, prevRow, prevCol);
                    
                    // 检查匹配
                    const matches = checkAllMatches();
                    
                    if (matches.length > 0) {
                        moves--;
                        document.getElementById('moves').textContent = moves;
                        
                        // 处理匹配
                        processMatches(matches);
                    } else {
                        // 没有匹配，交换回来
                        setTimeout(() => {
                            swapGems(row, col, prevRow, prevCol);
                            selectedGem = null;
                            document.querySelectorAll('.gem.selected').forEach(gem => {
                                gem.classList.remove('selected');
                            });
                            isProcessing = false;
                        }, 300);
                    }
                } else {
                    // 选择新的宝石
                    document.querySelectorAll('.gem.selected').forEach(gem => {
                        gem.classList.remove('selected');
                    });
                    selectedGem = { row, col };
                    e.target.classList.add('selected');
                }
            }
        }
        
        // 检查两个位置是否相邻
        function isAdjacent(row1, col1, row2, col2) {
            const rowDiff = Math.abs(row1 - row2);
            const colDiff = Math.abs(col1 - col2);
            return (rowDiff === 1 && colDiff === 0) || (rowDiff === 0 && colDiff === 1);
        }
        
        // 交换两个宝石
        function swapGems(row1, col1, row2, col2) {
            const temp = board[row1][col1];
            board[row1][col1] = board[row2][col2];
            board[row2][col2] = temp;
            renderBoard();
        }
        
        // 核心匹配检测算法
        function checkAllMatches() {
            const matches = [];
            
            // 检查水平匹配
            for (let row = 0; row < BOARD_SIZE; row++) {
                for (let col = 0; col < BOARD_SIZE - MIN_MATCH_LENGTH + 1; col++) {
                    const gemType = board[row][col];
                    let matchLength = 1;
                    
                    // 检查连续相同宝石
                    for (let i = col + 1; i < BOARD_SIZE; i++) {
                        if (board[row][i] === gemType) {
                            matchLength++;
                        } else {
                            break;
                        }
                    }
                    
                    // 如果有足够长的匹配，添加到结果中
                    if (matchLength >= MIN_MATCH_LENGTH) {
                        const match = [];
                        for (let i = 0; i < matchLength; i++) {
                            match.push({ row, col: col + i });
                        }
                        matches.push(match);
                        col += matchLength - 1; // 跳过已匹配的宝石
                    }
                }
            }
            
            // 检查垂直匹配
            for (let col = 0; col < BOARD_SIZE; col++) {
                for (let row = 0; row < BOARD_SIZE - MIN_MATCH_LENGTH + 1; row++) {
                    const gemType = board[row][col];
                    let matchLength = 1;
                    
                    // 检查连续相同宝石
                    for (let i = row + 1; i < BOARD_SIZE; i++) {
                        if (board[i][col] === gemType) {
                            matchLength++;
                        } else {
                            break;
                        }
                    }
                    
                    // 如果有足够长的匹配，添加到结果中
                    if (matchLength >= MIN_MATCH_LENGTH) {
                        const match = [];
                        for (let i = 0; i < matchLength; i++) {
                            match.push({ row: row + i, col });
                        }
                        matches.push(match);
                        row += matchLength - 1; // 跳过已匹配的宝石
                    }
                }
            }
            
            // 合并重叠的匹配
            return mergeMatches(matches);
        }
        
        // 合并重叠的匹配
        function mergeMatches(matches) {
            const merged = [];
            const used = new Set();
            
            for (let i = 0; i < matches.length; i++) {
                if (used.has(i)) continue;
                
                let currentMatch = [...matches[i]];
                
                // 查找与此匹配重叠的其他匹配
                for (let j = i + 1; j < matches.length; j++) {
                    if (used.has(j)) continue;
                    
                    // 检查是否有重叠
                    if (matchesOverlap(currentMatch, matches[j])) {
                        currentMatch = mergeTwoMatches(currentMatch, matches[j]);
                        used.add(j);
                    }
                }
                
                merged.push(currentMatch);
                used.add(i);
            }
            
            return merged;
        }
        
        // 检查两个匹配是否重叠
        function matchesOverlap(match1, match2) {
            for (const gem1 of match1) {
                for (const gem2 of match2) {
                    if (gem1.row === gem2.row && gem1.col === gem2.col) {
                        return true;
                    }
                }
            }
            return false;
        }
        
        // 合并两个匹配
        function mergeTwoMatches(match1, match2) {
            const merged = [...match1];
            
            for (const gem of match2) {
                if (!match1.some(g => g.row === gem.row && g.col === gem.col)) {
                    merged.push(gem);
                }
            }
            
            return merged;
        }
        
        // 处理匹配
        async function processMatches(matches) {
            // 标记匹配的宝石
            const matchedPositions = new Set();
            matches.forEach(match => {
                match.forEach(gem => {
                    matchedPositions.add(`${gem.row},${gem.col}`);
                });
            });
            
            // 添加匹配动画
            document.querySelectorAll('.gem').forEach(gem => {
                const row = parseInt(gem.dataset.row);
                const col = parseInt(gem.dataset.col);
                if (matchedPositions.has(`${row},${col}`)) {
                    gem.classList.add('matched');
                }
            });
            
            // 更新分数
            const points = matchedPositions.size * 10;
            score += points;
            document.getElementById('score').textContent = score;
            showMessage(`+${points} 分!`);
            
            // 等待动画完成
            await sleep(500);
            
            // 移除匹配的宝石
            matchedPositions.forEach(pos => {
                const [row, col] = pos.split(',').map(Number);
                board[row][col] = 0;
            });
            
            // 宝石下落
            await dropGems();
            
            // 填充新宝石
            await fillBoard();
            
            // 检查是否有新的匹配
            const newMatches = checkAllMatches();
            if (newMatches.length > 0) {
                // 递归处理新的匹配
                await processMatches(newMatches);
            } else {
                // 检查游戏是否结束
                if (moves <= 0) {
                    gameOver();
                } else {
                    // 重置选择状态
                    selectedGem = null;
                    document.querySelectorAll('.gem.selected').forEach(gem => {
                        gem.classList.remove('selected');
                    });
                    isProcessing = false;
                }
            }
        }
        
        // 宝石下落
        async function dropGems() {
            for (let col = 0; col < BOARD_SIZE; col++) {
                let emptyRow = BOARD_SIZE - 1;
                
                // 从底部向上扫描
                for (let row = BOARD_SIZE - 1; row >= 0; row--) {
                    if (board[row][col] !== 0) {
                        // 如果有空位，下落宝石
                        if (row !== emptyRow) {
                            board[emptyRow][col] = board[row][col];
                            board[row][col] = 0;
                        }
                        emptyRow--;
                    }
                }
                
                // 渲染下落动画
                renderBoard();
                
                // 添加下落动画类
                for (let row = emptyRow; row >= 0; row--) {
                    if (board[row][col] !== 0) {
                        const gemIndex = row * BOARD_SIZE + col;
                        const gem = document.querySelectorAll('.gem')[gemIndex];
                        gem.classList.add('falling');
                    }
                }
                
                await sleep(500);
            }
        }
        
        // 填充新宝石
        async function fillBoard() {
            for (let col = 0; col < BOARD_SIZE; col++) {
                for (let row = 0; row < BOARD_SIZE; row++) {
                    if (board[row][col] === 0) {
                        board[row][col] = Math.floor(Math.random() * GEM_TYPES) + 1;
                    }
                }
            }
            renderBoard();
        }
        
        // 洗牌游戏板
        function shuffleBoard() {
            for (let row = 0; row < BOARD_SIZE; row++) {
                for (let col = 0; col < BOARD_SIZE; col++) {
                    board[row][col] = Math.floor(Math.random() * GEM_TYPES) + 1;
                }
            }
        }
        
        // 显示消息
        function showMessage(text) {
            const message = document.getElementById('message');
            message.textContent = text;
            message
