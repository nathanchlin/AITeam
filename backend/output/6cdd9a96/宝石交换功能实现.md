# 宝石交换功能实现

我将实现宝石选择和交换机制，包括相邻宝石交换验证和视觉反馈效果。这部分是消消乐游戏的核心交互功能。

## 实现思路

1. **宝石选择机制**：允许玩家点击选择宝石，高亮显示选中的宝石
2. **交换验证**：检查两个宝石是否相邻（上下左右）
3. **交换动画**：实现平滑的宝石移动动画
4. **视觉反馈**：提供选中状态、交换成功/失败的视觉提示
5. **交换限制**：确保交换后能形成匹配（可选，提高游戏体验）

        document.addEventListener('DOMContentLoaded', () => {

            // 初始化游戏板
            function initializeBoard() {
                board = [];
                gameBoard.innerHTML = '';
                
                for (let row = 0; row < BOARD_SIZE; row++) {
                    board[row] = [];
                    for (let col = 0; col < BOARD_SIZE; col++) {
                        let gemType;
                        do {
                            gemType = GEM_TYPES[Math.floor(Math.random() * GEM_TYPES.length)];
                        } while (wouldCreateMatch(row, col, gemType));
                        
                        board[row][col] = gemType;
                        createGemElement(row, col, gemType);
                    }
                }

            // 检查是否会创建匹配
            function wouldCreateMatch(row, col, gemType) {
                // 检查水平方向
                let horizontalCount = 1;
                
                // 向左检查
                for (let c = col - 1; c >= 0 && board[row][c] === gemType; c--) {
                    horizontalCount++;
                }
                
                // 向右检查
                for (let c = col + 1; c < BOARD_SIZE && board[row] && board[row][c] === gemType; c++) {
                    horizontalCount++;
                }
                
                if (horizontalCount >= 3) return true;
                
                // 检查垂直方向
                let verticalCount = 1;
                
                // 向上检查
                for (let r = row - 1; r >= 0 && board[r][col] === gemType; r--) {
                    verticalCount++;
                }
                
                // 向下检查
                for (let r = row + 1; r < BOARD_SIZE && board[r] && board[r][col] === gemType; r++) {
                    verticalCount++;
                }
                
                return verticalCount >= 3;
            }
            
            // 创建宝石元素
            function createGemElement(row, col, gemType) {
                const gem = document.createElement('div');
                gem.className = `gem ${gemType}`;
                gem.textContent = GEM_SYMBOLS[gemType];
                gem.dataset.row = row;
                gem.dataset.col = col;
                gem.addEventListener('click', () => handleGemClick(row, col));
                gameBoard.appendChild(gem);
            }
            
            // 处理宝石点击
            function handleGemClick(row, col) {
                if (isProcessing) return;
                
                const clickedGem = document.querySelector(`[data-row="${row}"][data-col="${col}"]`);
                
                if (!selectedGem) {
                    // 第一次选择
                    selectedGem = { row, col, element: clickedGem };
                    clickedGem.classList.add('selected');
                } else if (selectedGem.row === row && selectedGem.col === col) {
                    // 取消选择
                    selectedGem.element.classList.remove('selected');
                    selectedGem = null;
                } else {
                    // 尝试交换
                    if (areAdjacent(selectedGem.row, selectedGem.col, row, col)) {
                        swapGems(selectedGem.row, selectedGem.col, row, col);
                    } else {
                        showMessage('只能交换相邻的宝石！', 'error');
                        selectedGem.element.classList.remove('selected');
                        selectedGem = { row, col, element: clickedGem };
                        clickedGem.classList.add('selected');
                    }
                }
            }
            
            // 检查两个宝石是否相邻
            function areAdjacent(row1, col1, row2, col2) {
                const rowDiff = Math.abs(row1 - row2);
                const colDiff = Math.abs(col1 - col2);
                return (rowDiff === 1 && colDiff === 0) || (rowDiff === 0 && colDiff === 1);
            }
            
            // 交换宝石
            async function swapGems(row1, col1, row2, col2) {
                isProcessing = true;
                
                const gem1 = document.querySelector(`[data-row="${row1}"][data-col="${col1}"]`);
                const gem2 = document.querySelector(`[data-row="${row2}"][data-col="${col2}"]`);
                
                // 添加交换动画类
                gem1.classList.add('swapping');
                gem2.classList.add('swapping');
                
                // 交换数据
                [board[row1][col1], board[row2][col2]] = [board[row2][col2], board[row1][col1]];
                
                // 交换位置
                const tempRow = gem1.dataset.row;
                const tempCol = gem1.dataset.col;
                
                gem1.dataset.row = gem2.dataset.row;
                gem1.dataset.col = gem2.dataset.col;
                gem2.dataset.row = tempRow;
                gem2.dataset.col = tempCol;
                
                // 执行交换动画
                const rect1 = gem1.getBoundingClientRect();
                const rect2 = gem2.getBoundingClientRect();
                
                gem1.style.transform = `translate(${rect2.left - rect1.left}px, ${rect2.top - rect1.top}px)`;
                gem2.style.transform = `translate(${rect1.left - rect2.left}px, ${rect1.top - rect2.top}px)`;
                
                // 等待动画完成
                await new Promise(resolve => setTimeout(resolve, 300));
                
                // 重置变换
                gem1.style.transform = '';
                gem2.style.transform = '';
                
                // 移除动画类
                gem1.classList.remove('swapping');
                gem2.classList.remove('swapping');
                
                // 更新选中状态
                gem1.classList.remove('selected');
                gem2.classList.remove('selected');
                
                // 检查是否有匹配
                const matches = findMatches();
                
                if (matches.length > 0) {
                    // 有效交换
                    moves++;
                    updateMoves();
                    await processMatches();
                } else {
                    // 无效交换，交换回来
                    showMessage('无效交换！', 'error');
                    
                    [board[row1][col1], board[row2][col2]] = [board[row2][col2], board[row1][col1]];
                    
                    gem1.dataset.row = tempRow;
                    gem1.dataset.col = tempCol;
                    gem2.dataset.row = row2;
                    gem2.dataset.col = col2;
                    
                    gem1.style.transform = `translate(${rect2.left - rect1.left}px, ${rect2.top - rect1.top}px)`;
                    gem2.style.transform = `translate(${rect1.left - rect2.left}px, ${rect1.top - rect2.top}px)`;
                    
                    await new Promise(resolve => setTimeout(resolve, 300));
                    
                    gem1.style.transform = '';
                    gem2.style.transform = '';
                }
                
                selectedGem = null;
                isProcessing = false;
            }
            
            // 查找所有匹配
            function findMatches() {
                const matches = [];
                const checked = Array(BOARD_SIZE).fill(null).map(() => Array(BOARD_SIZE).fill(false));
                
                // 检查水平匹配
                for (let row = 0; row < BOARD_SIZE; row++) {
                    for (let col = 0; col < BOARD_SIZE - 2; col++) {
                        const gemType = board[row][col];
                        if (gemType && board[row][col + 1] === gemType && board[row][col + 2] === gemType) {
                            const match = [{ row, col }, { row, col: col + 1 }, { row, col: col + 2 }];
                            
                            // 检查是否有更多连续的相同宝石
                            let k = col + 3;
                            while (k < BOARD_SIZE && board[row][k] === gemType) {
                                match.push({ row, col: k });
                                k++;
                            }
                            
                            matches.push(match);
                            col = k - 1; // 跳过已匹配的宝石
                        }
                    }
                }
                
                // 检查垂直匹配
                for (let col = 0; col < BOARD_SIZE; col++) {
                    for (let row = 0; row < BOARD_SIZE - 2; row++) {
                        const gemType = board[row][col];
                        if (gemType && board[row + 1][col] === gemType && board[row + 2][col] === gemType) {
                            const match = [{ row, col }, { row: row + 1, col }, { row: row + 2, col }];
                            
                            // 检查是否有更多连续的相同宝石
                            let k = row + 3;
                            while (k < BOARD_SIZE && board[k][col] === gemType) {
                                match.push({ row: k, col });
                                k++;
                            }
                            
                            matches.push(match);
                            row = k - 1; // 跳过已匹配的宝石
                        }
                    }
                }
                
                return matches;
            }
            
            // 处理匹配
            async function processMatches() {
                let totalMatches = 0;
                
                while
