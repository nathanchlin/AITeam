    <script>
        document.addEventListener('DOMContentLoaded', () => {
            const BOARD_SIZE = 8;
            const GEM_TYPES = ['red', 'blue', 'green', 'yellow', 'purple', 'orange'];
            const GEM_SYMBOLS = {
                'red': '♦',
                'blue': '♠',
                'green': '♣',
                'yellow': '★',
                'purple': '♥',
                'orange': '●'
            };
            
            let board = [];
            let selectedGem = null;
            let score = 0;
            let isProcessing = false;
            
            // 初始化游戏板
            function initBoard() {
                const boardElement = document.getElementById('game-board');
                boardElement.innerHTML = '';
                board = [];
                
                for (let row = 0; row < BOARD_SIZE; row++) {
                    board[row] = [];
                    for (let col = 0; col < BOARD_SIZE; col++) {
                        let gemType;
                        do {
                            gemType = GEM_TYPES[Math.floor(Math.random() * GEM_TYPES.length)];
                        } while (wouldCreateMatch(row, col, gemType));
                        
                        board[row][col] = gemType;
                        createGemElement(row, col, gemType);
                    }
                }
                
                // 确保初始板没有匹配
                while (findMatches().length > 0) {
                    removeMatches();
                    fillBoard();
                }
            }
            
            // 检查是否会在指定位置创建匹配
            function wouldCreateMatch(row, col, gemType) {
                // 检查水平匹配
                let horizontalCount = 1;
                
                // 向左检查
                for (let c = col - 1; c >= 0 && board[row][c] === gemType; c--) {
                    horizontalCount++;
                }
                
                // 向右检查
                for (let c = col + 1; c < BOARD_SIZE && board[row] && board[row][c] === gemType; c++) {
                    horizontalCount++;
                }
                
                if (horizontalCount >= 3) return true;
                
                // 检查垂直匹配
                let verticalCount = 1;
                
                // 向上检查
                for (let r = row - 1; r >= 0 && board[r][col] === gemType; r--) {
                    verticalCount++;
                }
                
                // 向下检查
                for (let r = row + 1; r < BOARD_SIZE && board[r] && board[r][col] === gemType; r++) {
                    verticalCount++;
                }
                
                if (verticalCount >= 3) return true;
                
                return false;
            }
            
            // 创建宝石元素
            function createGemElement(row, col, gemType) {
                const boardElement = document.getElementById('game-board');
                const gemElement = document.createElement('div');
                gemElement.className = `gem ${gemType}`;
                gemElement.textContent = GEM_SYMBOLS[gemType];
                gemElement.dataset.row = row;
                gemElement.dataset.col = col;
                gemElement.addEventListener('click', handleGemClick);
                boardElement.appendChild(gemElement);
            }
            
            // 处理宝石点击
            function handleGemClick(event) {
                if (isProcessing) return;
                
                const clickedGem = event.target;
                const row = parseInt(clickedGem.dataset.row);
                const col = parseInt(clickedGem.dataset.col);
                
                if (!selectedGem) {
                    selectedGem = { row, col, element: clickedGem };
                    clickedGem.classList.add('selected');
                } else {
                    const prevRow = selectedGem.row;
                    const prevCol = selectedGem.col;
                    
                    // 检查是否相邻
                    if (isAdjacent(row, col, prevRow, prevCol)) {
                        isProcessing = true;
                        swapGems(row, col, prevRow, prevCol);
                    } else {
                        // 取消之前的选择
                        selectedGem.element.classList.remove('selected');
                        selectedGem = { row, col, element: clickedGem };
                        clickedGem.classList.add('selected');
                    }
                }
            }
            
            // 检查两个宝石是否相邻
            function isAdjacent(row1, col1, row2, col2) {
                const rowDiff = Math.abs(row1 - row2);
                const colDiff = Math.abs(col1 - col2);
                
                return (rowDiff === 1 && colDiff === 0) || (rowDiff === 0 && colDiff === 1);
            }
            
            // 交换两个宝石
            async function swapGems(row1, col1, row2, col2) {
                // 交换数组中的值
                const temp = board[row1][col1];
                board[row1][col1] = board[row2][col2];
                board[row2][col2] = temp;
                
                // 更新DOM
                updateBoardDisplay();
                
                // 检查是否有匹配
                const matches = findMatches();
                
                if (matches.length > 0) {
                    // 有匹配，处理消除
                    await processMatches();
                } else {
                    // 没有匹配，交换回来
                    const temp = board[row1][col1];
                    board[row1][col1] = board[row2][col2];
                    board[row2][col2] = temp;
                    updateBoardDisplay();
                }
                
                // 清除选择
                document.querySelectorAll('.gem.selected').forEach(gem => {
                    gem.classList.remove('selected');
                });
                selectedGem = null;
                isProcessing = false;
            }
            
            // 更新游戏板显示
            function updateBoardDisplay() {
                const boardElement = document.getElementById('game-board');
                const gems = boardElement.querySelectorAll('.gem');
                
                gems.forEach(gem => {
                    const row = parseInt(gem.dataset.row);
                    const col = parseInt(gem.dataset.col);
                    const gemType = board[row][col];
                    
                    gem.className = `gem ${gemType}`;
                    gem.textContent = GEM_SYMBOLS[gemType];
                });
            }
            
            // 查找所有匹配
            function findMatches() {
                const matches = [];
                
                // 检查水平匹配
                for (let row = 0; row < BOARD_SIZE; row++) {
                    for (let col = 0; col < BOARD_SIZE - 2; col++) {
                        const gemType = board[row][col];
                        if (gemType && board[row][col + 1] === gemType && board[row][col + 2] === gemType) {
                            const match = [{ row, col }, { row, col: col + 1 }, { row, col: col + 2 }];
                            
                            // 检查是否有更多相同宝石
                            for (let c = col + 3; c < BOARD_SIZE && board[row][c] === gemType; c++) {
                                match.push({ row, col: c });
                            }
                            
                            matches.push(match);
                            col += match.length - 1;
                        }
                    }
                }
                
                // 检查垂直匹配
                for (let col = 0; col < BOARD_SIZE; col++) {
                    for (let row = 0; row < BOARD_SIZE - 2; row++) {
                        const gemType = board[row][col];
                        if (gemType && board[row + 1][col] === gemType
