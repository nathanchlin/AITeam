### 1. 动画状态管理

首先，我们需要一个动画状态管理系统来跟踪所有正在进行的动画：

```javascript
class AnimationManager {
  constructor() {
    this.animations = [];
    this.isAnimating = false;
  }

  addAnimation(animation) {
    this.animations.push(animation);
    if (!this.isAnimating) {
      this.isAnimating = true;
      this.animate();
    }
  }

  animate() {
    if (this.animations.length === 0) {
      this.isAnimating = false;
      return;
    }

    const animation = this.animations[0];
    animation.update();
    
    if (animation.isComplete()) {
      this.animations.shift();
      requestAnimationFrame(() => this.animate());
    } else {
      requestAnimationFrame(() => this.animate());
    }
  }

  isAnimating() {
    return this.isAnimating || this.animations.length > 0;
  }
}

// 全局动画管理器
```

### 2. 基础动画类

```javascript
class BaseAnimation {
  constructor(duration, easing = 'linear') {
    this.duration = duration;
    this.easing = easing;
    this.startTime = null;
    this.completed = false;
  }

  update() {
    if (!this.startTime) {
      this.startTime = performance.now();
    }
    
    const elapsed = performance.now() - this.startTime;
    const progress = Math.min(elapsed / this.duration, 1);
    
    // 应用缓动函数
    const easedProgress = this.applyEasing(progress);
    this.apply(easedProgress);
    
    if (progress >= 1) {
      this.completed = true;
    }
  }

  applyEasing(progress) {
    switch (this.easing) {
      case 'ease-in':
        return progress * progress;
      case 'ease-out':
        return 1 - Math.pow(1 - progress, 2);
      case 'ease-in-out':
        return progress < 0.5 
          ? 2 * progress * progress 
          : 1 - Math.pow(-2 * progress + 2, 2) / 2;
      default:
        return progress;
    }
  }

  apply(progress) {
    // 子类实现具体动画逻辑
  }

  isComplete() {
    return this.completed;
  }
}
```

### 3. 宝石交换动画

```javascript
class SwapAnimation extends BaseAnimation {
  constructor(gem1, gem2, onComplete) {
    super(300, 'ease-in-out');
    this.gem1 = gem1;
    this.gem2 = gem2;
    this.onComplete = onComplete;
    this.startPositions = {
      gem1: { x: gem1.x, y: gem1.y },
      gem2: { x: gem2.x, y: gem2.y }
    };
    this.endPositions = {
      gem1: { x: gem2.x, y: gem2.y },
      gem2: { x: gem1.x, y: gem1.y }
    };
  }

  apply(progress) {
    // 使用二次贝塞尔曲线使交换更自然
    const controlPoint = { 
      x: (this.startPositions.gem1.x + this.endPositions.gem1.x) / 2,
      y: Math.min(this.startPositions.gem1.y, this.endPositions.gem1.y) - 50
    };
    
    // 计算贝塞尔曲线上的点
    this.gem1.x = this.quadraticBezier(
      this.startPositions.gem1.x,
      controlPoint.x,
      this.endPositions.gem1.x,
      progress
    );
    
    this.gem1.y = this.quadraticBezier(
      this.startPositions.gem1.y,
      controlPoint.y,
      this.endPositions.gem1.y,
      progress
    );
    
    this.gem2.x = this.quadraticBezier(
      this.startPositions.gem2.x,
      controlPoint.x,
      this.endPositions.gem2.x,
      progress
    );
    
    this.gem2.y = this.quadraticBezier(
      this.startPositions.gem2.y,
      controlPoint.y,
      this.endPositions.gem2.y,
      progress
    );
    
    // 添加缩放效果
    const scale = 1 + Math.sin(progress * Math.PI) * 0.1;
    this.gem1.scale = scale;
    this.gem2.scale = scale;
  }

  quadraticBezier(p0, p1, p2, t) {
    return (1 - t) * (1 - t) * p0 + 2 * (1 - t) * t * p1 + t * t * p2;
  }

  isComplete() {
    if (super.isComplete()) {
      this.gem1.scale = 1;
      this.gem2.scale = 1;
      if (this.onComplete) this.onComplete();
      return true;
    }
    return false;
  }
}
```

```javascript
class MatchAnimation extends BaseAnimation {
  constructor(matchedGems, onComplete) {
    super(400, 'ease-in-out');
    this.matchedGems = matchedGems;
    this.onComplete = onComplete;
    this.originalScales = matchedGems.map(gem => gem.scale || 1);
    this.originalOpacity = matchedGems.map(gem => gem.opacity || 1);
  }

  apply(progress) {
    // 消除动画：缩小并淡出
    const scaleProgress = Math.sin(progress * Math.PI);
    const opacityProgress = 1 - progress;
    
    this.matchedGems.forEach((gem, i) => {
      gem.scale = this.originalScales[i] * scaleProgress;
      gem.opacity = this.originalOpacity[i] * opacityProgress;
      
      // 添加轻微的旋转效果
      gem.rotation = (1 - progress) * 360 * (i % 2 === 0 ? 1 : -1);
    });
    
    // 添加粒子效果
    if (progress > 0.5 && !this.particlesCreated) {
      this.createParticles();
      this.particlesCreated = true;
    }
  }

  createParticles() {
    this.matchedGems.forEach(gem => {
      for (let i = 0; i < 8; i++) {
        const particle = document.createElement('div');
        particle.className = 'particle';
        particle.style.left = `${gem.x * cellSize + cellSize / 2}px`;
        particle.style.top = `${gem.y * cellSize + cellSize / 2}px`;
        particle.style.backgroundColor = gem.color;
        particle.style.width = '8px';
        particle.style.height = '8px';
        particle.style.borderRadius = '50%';
        particle.style.position = 'absolute';
        particle.style.pointerEvents = 'none';
        gameBoard.appendChild(particle);
        
        // 粒子动画
        const angle = (i / 8) * Math.PI * 2;
        const velocity = 100 + Math.random() * 50;
        const lifetime = 1000 + Math.random() * 500;
        
        let startTime = performance.now();
        
        const animateParticle = () => {
          const elapsed = performance.now() - startTime;
          const progress = Math.min(elapsed / lifetime, 1);
          
          if (progress < 1) {
            const distance = velocity * progress;
            particle.style.transform = `translate(${Math.cos(angle) * distance}px, ${Math.sin(angle) * distance}px)`;
            particle.style.opacity = 1 - progress;
            requestAnimationFrame(animateParticle);
          } else {
            particle.remove();
          }
        };
        
        requestAnimationFrame(animateParticle);
      }
    });
  }

  isComplete() {
    if (super.isComplete()) {
      this.matchedGems.forEach(gem => {
        gem.scale = 1;
        gem.opacity = 1;
        gem.rotation = 0;
      });
      if (this.onComplete) this.onComplete();
      return true;
    }
    return false;
  }
}
```

### 5. 宝石下落动画

```javascript
class FallAnimation extends BaseAnimation {
  constructor(gems, onComplete) {
    super(500, 'ease-in-out');
    this.gems = gems;
    this.onComplete = onComplete;
    this.targetPositions = {};
    
    // 计算目标位置
    this.gems.forEach(gem => {
      this.targetPositions[gem.id] = gem.y;
      gem.y = gem.startY;
    });
  }

  apply(progress) {
    // 使用弹性缓动效果
    const bounceProgress = this.easeOutBounce(progress);
    
    this.gems.forEach(gem => {
      const startY = gem.startY;
      const targetY = this.targetPositions[gem.id];
      gem.y = startY + (targetY - startY) * bounceProgress;
      
      // 添加轻微的旋转
      gem.rotation = (1 - progress) * 90 * (gem.id % 2 === 0 ? 1 : -1);
      
      // 缩放效果
      gem.scale = 0.8 + progress * 0.2;
    });
  }

  easeOutBounce(x) {
    const n1 = 7.5625;
    const d1 = 2.75;

    if (x < 1 / d1) {
      return n1 * x * x;
    } else if (x < 2 / d1) {
      return n1 * (x -= 1.5 / d1) * x + 0.75;
    } else if (x < 2.5 / d1) {
      return n1 * (x -= 2.25 / d1) * x + 0.9375;
    } else {
      return n1 * (x -= 2.625 / d1) * x + 0.984375;
    }
  }

  isComplete() {
    if (super.isComplete()) {
      this.gems.forEach(gem => {
        gem.rotation = 0;
        gem.scale = 1;
      });
      if (this.onComplete) this.onComplete();
      return true;
    }
    return false;
  }
}
```

```javascript
class ChainReactionAnimation extends BaseAnimation {
  constructor(matches, onComplete) {
    super(2000, 'ease-in-out');
    this.matches = matches;
    this.onComplete = onComplete;
    this.currentMatchIndex = 0;
    this.matchStartTime = null;
  }

  apply(progress) {
    if (!this.matchStartTime) {
      this.matchStartTime = performance.now();
    }
    
    const matchProgress = (performance.now() - this.matchStartTime) / 400;
    
    if (matchProgress >= 1 && this.currentMatchIndex < this.matches.length - 1) {
      this.currentMatchIndex++;
      this.matchStartTime = performance.now();
    }
    
    // 高亮当前匹配的宝石
    this.matches.forEach((match, index) => {
      if (index < this.currentMatchIndex) {
        match.forEach(gem => {
          gem.highlighted = false;
        });
      } else if (index === this.currentMatchIndex) {
        match.forEach(gem => {
          gem.highlighted = true;
          // 添加脉动效果
          const pulseScale = 1 + Math.sin(matchProgress * Math.PI * 4) * 0.1;
          gem.pulseScale = pulseScale;
        });
      } else {
        match.forEach(gem => {
          gem.highlighted = false;
        });
      }
    });
    
    // 添加分数弹出效果
    if (this.currentMatchIndex < this.matches.length && matchProgress > 0.5 && !this.scorePopupShown) {
      this.showScorePopup();
      this.scorePopupShown = true;
    }
  }

  showScorePopup() {
    const currentMatch = this.matches[this.currentMatchIndex];
    const score = currentMatch.length * 10 * (this.currentMatchIndex + 1);
    
    const popup = document.createElement('div');
    popup.className = 'score-popup';
    popup.textContent = `+${score}`;
    popup.style.position = 'absolute';
    popup.style.left = `${currentMatch[0].x * cellSize + cellSize / 2}px`;
    popup.style.top = `${currentMatch[0].y * cellSize + cellSize / 2}px`;
    popup.style.fontSize = '24px';
    popup.fontWeight = 'bold';
    popup.style.color = '#FFD700';
    popup.style.textShadow = '0 0 10px rgba(255, 215, 0, 0.8)';
    popup.style.pointerEvents = 'none';
    popup.style.zIndex = '1000';
    
    gameBoard.appendChild(popup);
    
    // 动画效果
    let startTime = performance.now();
    const animatePopup = () => {
      const elapsed = performance.now() - startTime;
      const progress = Math.min(elapsed / 1000, 1);
      
      if (progress < 1) {
        popup.style.transform = `translateY(-${progress * 50}px) scale(${1 + progress * 0.5})`;
        popup.style.opacity = 1 - progress;
        requestAnimationFrame(animatePopup);
      } else {
        popup.remove();
      }
    };
    
    requestAnimationFrame(animatePopup);
  }

  isComplete() {
    if (super.isComplete()) {
      this.matches.forEach(match => {
        match.forEach(gem => {
          gem.highlighted = false;
          gem.pulseScale = 1;
        });
      });
      if (this.onComplete) this.onComplete();
      return true;
    }
    return false;
  }
}
```

class Game {
  constructor() {
    // 初始化游戏
    this.initializeBoard();
    this.score = 0;
    this.isProcessing = false;
    
    // 绑定事件
    this.setupEventListeners();
  }

  initializeBoard() {
    // 创建8x8网格
    this.board = [];
    for (let y = 0; y < 8; y++) {
      this.board[y] = [];
      for (let x = 0; x < 8; x++) {
        const gem = this.createGem(x, y);
        this.board[y][x] = gem;
      }
    }
    
    // 确保初始板面没有匹配
    this.removeInitialMatches();
    this.renderBoard();
  }

  createGem(x, y) {
    const types = ['red', 'blue', 'green', 'yellow', 'purple'];
    const type = types[Math.floor(Math.random() * types.length)];
    
    return {
      id: `${x}-${y}`,
      x: x,
      y: y,
      type: type,
      color: this.getGemColor(type),
      scale: 1,
      opacity: 1,
      rotation: 0,
      highlighted: false,
      pulseScale: 1
    };
  }

  getGemColor(type) {
    const colors = {
      red: '#FF5252',
      blue: '#448AFF',
      green: '#69F0AE',
      yellow: '#FFD740',
      purple: '#E040FB'
    };
    return colors[type];
  }

  removeInitialMatches() {
    let hasMatches = true;
    while (hasMatches) {
      hasMatches = false;
      for (let y = 0; y < 8; y++) {
        for (let x = 0; x < 8; x++) {
          if (this.checkMatchAt(x, y).length >= 3) {
            this.board[y][x] = this.createGem(x, y);
            hasMatches = true;
          }
        }
      }
    }
  }

  renderBoard() {
    const gameBoard = document.getElementById('gameBoard');
    gameBoard.innerHTML = '';
    
    for (let y = 0; y < 8; y++) {
      for (let x = 0; x < 8; x++) {
        const gem = this.board[y][x];
        const gemElement = document.createElement('div');
        gemElement.className = 'gem';
        gemElement.style.width = `${cellSize}px`;
        gemElement.style.height = `${cellSize}px`;
        gemElement.style.left = `${gem.x * cellSize}px`;
        gemElement.style.top = `${gem.y * cellSize}px`;
        gemElement.style.backgroundColor = gem.color;
        gemElement.style.borderRadius = '50%';
        gemElement.style.position = 'absolute';
        gemElement.style.transition = 'transform 0.1s, opacity 0.1s';
        gemElement.style.cursor = 'pointer';
        gemElement.dataset.x = x;
        gemElement.dataset.y = y;
        
        // 应用动画变换
        gemElement.style.transform = `
          scale(${gem.scale * (gem.pulseScale || 1)})
          rotate(${gem.rotation}deg)
        `;
        gemElement.style.opacity = gem.opacity;
        
        if (gem.highlighted) {
          gemElement.style.boxShadow = '0 0 20px rgba(255, 255, 255, 0.8)';
        }
        
        gameBoard.appendChild(gemElement);
      }
    }
  }

  setupEventListeners() {
    const gameBoard = document.getElementById('gameBoard');
    let selectedGem = null;
    
    gameBoard.addEventListener('click', (e) => {
      if (animationManager.isAnimating() || this.isProcessing) return;
      
      const x = parseInt(e.target.dataset.x);
      const y = parseInt(e.target.dataset.y);
      
      if (isNaN(x) || isNaN(y)) return;
      
      const clickedGem = this.board[y][x];
      
      if (!selectedGem) {
        selectedGem = clickedGem;
        clickedGem.highlighted = true;
        this.renderBoard();
      } else {
        // 检查是否相邻
        const dx = Math.abs(selectedGem.x - clickedGem.x);
        const dy = Math.abs(selectedGem.y - clickedGem.y);

          selectedGem = clickedGem;
          clickedGem.highlighted = true;
          this.renderBoard();
        }
      }
    });
  }
