    <script>
        // 游戏配置
        const BOARD_SIZE = 8;
        const GEM_TYPES = ['red', 'blue', 'green', 'yellow', 'purple', 'orange'];
        const GEM_SYMBOLS = {
            'red': '♦',
            'blue': '♠',
            'green': '♣',
            'yellow': '★',
            'purple': '♥',
            'orange': '●'
        };
        const INITIAL_MOVES = 30;
        const MATCH_THRESHOLD = 3;
        
        // 游戏状态
        let board = [];
        let score = 0;
        let movesLeft = INITIAL_MOVES;
        let selectedGem = null;
        let isProcessing = false;
        
        // DOM 元素
        const gameBoard = document.getElementById('gameBoard');
        const scoreElement = document.getElementById('score');
        const movesElement = document.getElementById('moves');
        const messageElement = document.getElementById('message');
        const resetButton = document.getElementById('resetButton');
        
        // 初始化游戏
        function initGame() {
            board = [];
            score = 0;
            movesLeft = INITIAL_MOVES;
            selectedGem = null;
            isProcessing = false;
            
            updateScore();
            updateMoves();
            createBoard();
            renderBoard();
            
            // 确保初始板面上没有匹配
            while (findMatches().length > 0) {
                createBoard();
            }
        }
        
        // 创建游戏板
        function createBoard() {
            for (let row = 0; row < BOARD_SIZE; row++) {
                board[row] = [];
                for (let col = 0; col < BOARD_SIZE; col++) {
                    board[row][col] = getRandomGemType();
                }
            }
        }
        
        // 获取随机宝石类型
        function getRandomGemType() {
            return GEM_TYPES[Math.floor(Math.random() * GEM_TYPES.length)];
        }
        
        // 渲染游戏板
        function renderBoard() {
            gameBoard.innerHTML = '';
            
            for (let row = 0; row < BOARD_SIZE; row++) {
                for (let col = 0; col < BOARD_SIZE; col++) {
                    const gem = document.createElement('div');
                    gem.className = `gem ${board[row][col]}`;
                    gem.textContent = GEM_SYMBOLS[board[row][col]];
                    gem.dataset.row = row;
                    gem.dataset.col = col;
                    gem.addEventListener('click', handleGemClick);
                    gameBoard.appendChild(gem);
                }
            }
        }
        
        // 处理宝石点击
        function handleGemClick(event) {
            if (isProcessing) return;
            
            const row = parseInt(event.target.dataset.row);
            const col = parseInt(event.target.dataset.col);
            
            if (selectedGem) {
                const selectedRow = parseInt(selectedGem.dataset.row);
                const selectedCol = parseInt(selectedGem.dataset.col);
                
                // 检查是否是相邻的宝石
                if (isAdjacent(row, col, selectedRow, selectedCol)) {
                    // 交换宝石
                    swapGems(row, col, selectedRow, selectedCol);
                    
                    // 检查是否有匹配
                    const matches = findMatches();
                    
                    if (matches.length > 0) {
                        // 有效移动
                        movesLeft--;
                        updateMoves();
                        processMatches();
                    } else {
                        // 无效移动，交换回来
                        swapGems(row, col, selectedRow, selectedCol);
                        showMessage('无效移动！', 1000);
                    }
                    
                    // 取消选中状态
                    selectedGem.classList.remove('selected');
                    selectedGem = null;
                } else {
                    // 选择新的宝石
                    selectedGem.classList.remove('selected');
                    selectedGem = event.target;
                    selectedGem.classList.add('selected');
                }
            } else {
                // 选择第一个宝石
                selectedGem = event.target;
                selectedGem.classList.add('selected');
            }
        }
        
        // 检查两个位置是否相邻
        function isAdjacent(row1, col1, row2, col2) {
            const rowDiff = Math.abs(row1 - row2);
            const colDiff = Math.abs(col1 - col2);
            return (rowDiff === 1 && colDiff === 0) || (rowDiff === 0 && colDiff === 1);
        }
        
        // 交换两个宝石
        function swapGems(row1, col1, row2, col2) {
            const temp = board[row1][col1];
            board[row1][col1] = board[row2][col2];
            board[row2][col2] = temp;
            renderBoard();
        }
        
        // 查找匹配的宝石
        function findMatches() {
            const matches = [];
            
            // 检查水平匹配
            for (let row = 0; row < BOARD_SIZE; row++) {
                for (let col = 0; col < BOARD_SIZE - 2; col++) {
                    const gemType = board[row][col];
                    if (gemType && gemType === board[row][col + 1] && gemType === board[row][col + 2]) {
                        const match = [{row, col}, {row, col: col + 1}, {row, col: col + 2}];
                        
                        // 检查是否有更多相同的宝石
                        let k = col + 3;
                        while (k < BOARD_SIZE && board[row][k] === gemType) {
                            match.push({row, col: k});
                            k++;
                        }
                        
                        matches.push(match);
                        col = k - 1; // 跳过已匹配的宝石
                    }
                }
            }
            
            // 检查垂直匹配
            for (let col = 0; col < BOARD_SIZE; col++) {
                for (let row = 0; row < BOARD_SIZE - 2; row++) {
                    const gemType = board[row][col];
                    if (gemType && gemType === board[row + 1][col] && gemType === board[row + 2][col]) {
                        const match = [{row, col}, {row: row + 1, col}, {row: row + 2, col}];
                        
                        // 检查是否有更多相同的宝石
                        let k = row + 3;
                        while (k < BOARD_SIZE && board[k][col] === gemType) {
                            match.push({row: k, col});
                            k++;
                        }
                        
                        matches.push(match);
                        row = k - 1; // 跳过已匹配的宝石
                    }
                }
            }
            
            return matches;
        }
        
        // 处理匹配
        function processMatches() {
            isProcessing = true;
            
            const matches = findMatches();
            
            if (matches.length > 0) {
                // 计算得分
                let points = 0;
                matches.forEach(match => {
                    points += match.length * 10;
                });
                score += points;
                updateScore();
                
                // 移除匹配的宝石
                removeMatches(matches);
                
                // 延迟后让宝石下落
                setTimeout(() => {
                    dropGems();
                    
                    // 填充空位
                    setTimeout(() => {
                        fillEmptySpaces();
                        
                        // 重新检查是否有新的匹配
                        setTimeout(() => {
                            const newMatches = findMatches();
                            if (newMatches.length > 0) {
                                processMatches();
                            } else {
                                isProcessing = false;
                                checkGameEnd();
                            }
                        }, 300);
                    }, 300);
                }, 300);
            } else {
                isProcessing = false;
                checkGameEnd();
            }
        }
        
        // 移除匹配的宝石
        function removeMatches(matches) {
            matches.forEach(match => {
                match.forEach(({row, col}) => {
                    board[row][col] = null;
                });
            });
            
            // 添加动画效果
            const gems = document.querySelectorAll('.gem');
            gems.forEach(gem => {
                const row = parseInt(gem.dataset.row);
                const col = parseInt(gem.dataset.col);
                
                if (matches.some(match => match.some(pos => pos.row === row && pos.col === col))) {
                    gem.style.opacity = '0';
                }
            });
        }
        
        // 宝石下落
        function dropGems() {
            for (let col = 0; col < BOARD_SIZE; col++) {
                let emptyRow = BOARD_SIZE - 1;
                
                for (let row = BOARD_SIZE - 1; row >= 0; row--) {
                    if (board[row][col] !== null) {
                        if (row !== emptyRow) {
                            board[emptyRow][col] = board[row][col];
                            board[row][col] = null;
                        }
                        emptyRow--;
                    }
                }
            }
            
            renderBoard();
        }
        
        // 填充空位
        function fillEmptySpaces() {
            for (let col = 0; col < BOARD_SIZE; col++) {
                for (let row = 0; row < BOARD_SIZE; row++) {
                    if (board[row][col] === null) {
                        board[row][col] = getRandomGemType();
                    }
                }
            }
            
            renderBoard();
        }
        
        // 更新分数显示
        function updateScore() {
            scoreElement.textContent = score;
        }
        
        // 更新剩余步数显示
        function updateMoves() {
            movesElement.textContent = movesLeft;
        }
        
        // 显示消息
        function showMessage(text, duration = 2000) {
            messageElement.textContent = text;
            messageElement.style.display = 'block';
            
            setTimeout(() => {
                messageElement.style.display = 'none';
            }, duration);
        }
        
        // 检查游戏是否结束
        function checkGameEnd() {
            if (movesLeft <= 0) {
                showMessage(`游戏结束！最终得分: ${score}`);
            } else {
                // 检查是否还有可能的移动
                if (!hasPossibleMoves()) {
                    showMessage('没有可能的移动了！重新开始游戏。');
                    setTimeout(initGame, 2000);
                }
            }
        }
        
        // 检查是否还有可能的移动
        function hasPossibleMoves() {
            // 简化版检查，实际实现应该更全面
            for (let row = 0; row < BOARD_SIZE; row++) {
                for (let col = 0; col < BOARD_SIZE; col++) {
                    // 检查水平交换
                    if (col < BOARD_SIZE - 1) {
                        swapGems(row, col, row, col + 1);
                        if (findMatches().length > 0) {
                            swapGems(row, col, row, col + 1); // 交换回来
                            return true;
                        }
                        swapGems(row, col, row, col + 1); // 交换回来
                    }
                    
                    // 检查垂直交换
                    if (row < BOARD_SIZE - 1) {
                        swapGems(row, col, row + 1, col);
                        if (findMatches().length > 0) {
                            swapGems(row, col, row + 1, col); // 交换回来
                            return true;
                        }
                        swapGems(row, col, row + 1, col); // 交换回来
                    }
                }
            }
            
            return false;
        }
        
        // 重置游戏
        resetButton.addEventListener('click', initGame);
        
        // 初始化游戏
        initGame();
    </script>
</body>
</html>
