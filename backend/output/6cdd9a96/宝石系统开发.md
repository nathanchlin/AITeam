```javascript
const GEM_TYPES = {
  RED: { id: 0, color: '#FF6B6B', shape: 'circle' },
  BLUE: { id: 1, color: '#4ECDC4', shape: 'square' },
  GREEN: { id: 2, color: '#45B7D1', shape: 'triangle' },
  YELLOW: { id: 3, color: '#FFD166', shape: 'diamond' },

};

const GEM_TYPE_COUNT = Object.keys(GEM_TYPES).length;
const BOARD_SIZE = 8;
```

### 宝石类

```javascript
class Gem {
  constructor(type, row, col) {
    this.type = type;
    this.row = row;
    this.col = col;
    this.element = null;
    this.isMatched = false;
    this.isFalling = false;
  }

  createElement() {
    const gem = document.createElement('div');
    gem.className = 'gem';
    gem.style.backgroundColor = this.type.color;
    gem.dataset.row = this.row;
    gem.dataset.col = this.col;
    
    // 根据形状应用不同的CSS类
    switch(this.type.shape) {
      case 'circle':
        gem.style.borderRadius = '50%';
        break;
      case 'square':
        gem.style.borderRadius = '0';
        break;
      case 'triangle':
        gem.style.width = '0';
        gem.style.height = '0';
        gem.style.backgroundColor = 'transparent';
        gem.style.borderLeft = '25px solid transparent';
        gem.style.borderRight = '25px solid transparent';
        gem.style.borderBottom = `50px solid ${this.type.color}`;
        gem.style.borderRadius = '0';
        break;
      case 'diamond':
        gem.style.transform = 'rotate(45deg)';
        break;
      case 'hexagon':
        gem.style.clipPath = 'polygon(50% 0%, 100% 25%, 100% 75%, 50% 100%, 0% 75%, 0% 25%)';
        break;
      case 'star':
        gem.style.clipPath = 'polygon(50% 0%, 61% 35%, 98% 35%, 68% 57%, 79% 91%, 50% 70%, 21% 91%, 32% 57%, 2% 35%, 39% 35%)';
        break;
    }
    
    this.element = gem;
    return gem;
  }
}
```

### 宝石板类

```javascript
class GemBoard {
  constructor(containerId) {
    this.container = document.getElementById(containerId);
    this.board = [];
    this.score = 0;
    this.selectedGem = null;
    this.isAnimating = false;
    
    this.initBoard();
    this.renderBoard();
    this.setupEventListeners();
  }

  initBoard() {
    // 初始化8x8网格
    for (let row = 0; row < BOARD_SIZE; row++) {
      this.board[row] = [];
      for (let col = 0; col < BOARD_SIZE; col++) {
        let gemType;
        let isValidPosition = false;
        let attempts = 0;
        
        // 确保初始布局不产生匹配
        while (!isValidPosition && attempts < 100) {
          gemType = Math.floor(Math.random() * GEM_TYPE_COUNT);
          
          // 检查水平方向是否会有匹配
          if (col >= 2 && 
              gemType === this.board[row][col - 1].type && 
              gemType === this.board[row][col - 2].type) {
            attempts++;
            continue;
          }
          
          // 检查垂直方向是否会有匹配
          if (row >= 2 && 
              gemType === this.board[row - 1][col].type && 
              gemType === this.board[row - 2][col].type) {
            attempts++;
            continue;
          }
          
          isValidPosition = true;
        }
        
        this.board[row][col] = new Gem(GEM_TYPES[Object.keys(GEM_TYPES)[gemType]], row, col);
      }
    }
  }

  renderBoard() {
    // 清空容器
    this.container.innerHTML = '';
    
    // 创建网格
    const grid = document.createElement('div');
    grid.className = 'game-board';
    grid.style.gridTemplateColumns = `repeat(${BOARD_SIZE}, 1fr)`;
    
    // 添加宝石到网格
    for (let row = 0; row < BOARD_SIZE; row++) {
      for (let col = 0; col < BOARD_SIZE; col++) {
        const gemElement = this.board[row][col].createElement();
        grid.appendChild(gemElement);
      }
    }
    
    this.container.appendChild(grid);
  }

  setupEventListeners() {
    this.container.addEventListener('click', (e) => {
      if (this.isAnimating) return;
      
      const gemElement = e.target.closest('.gem');
      if (!gemElement) return;
      
      const row = parseInt(gemElement.dataset.row);
      const col = parseInt(gemElement.dataset.col);
      
      if (!this.selectedGem) {
        // 选择第一个宝石
        this.selectedGem = { row, col };
        gemElement.classList.add('selected');
      } else {
        // 尝试交换宝石
        const prevRow = this.selectedGem.row;
        const prevCol = this.selectedGem.col;
        
        // 检查是否相邻
        if (this.areAdjacent(prevRow, prevCol, row, col)) {
          this.swapGems(prevRow, prevCol, row, col);
        } else {
          // 取消前一个选择，选择新的
          document.querySelector('.gem.selected')?.classList.remove('selected');
          this.selectedGem = { row, col };
          gemElement.classList.add('selected');
        }
      }
    });
  }

  areAdjacent(row1, col1, row2, col2) {
    const rowDiff = Math.abs(row1 - row2);
    const colDiff = Math.abs(col1 - col2);
    return (rowDiff === 1 && colDiff === 0) || (rowDiff === 0 && colDiff === 1);
  }

  async swapGems(row1, col1, row2, col2) {
    this.isAnimating = true;
    
    // 交换数组中的宝石
    [this.board[row1][col1], this.board[row2][col2]] = 
    [this.board[row2][col2], this.board[row1][col1]];
    
    // 更新位置
    this.board[row1][col1].row = row1;
    this.board[row1][col1].col = col1;
    this.board[row2][col2].row = row2;
    this.board[row2][col2].col = col2;
    
    // 重新渲染
    this.renderBoard();
    
    // 检查是否有匹配
    const matches = this.findMatches();
    
    if (matches.length > 0) {
      // 有匹配，处理消除
      await this.handleMatches(matches);
    } else {
      // 没有匹配，交换回来
      await this.delay(300);
      [this.board[row1][col1], this.board[row2][col2]] = 
      [this.board[row2][col2], this.board[row1][col1]];
      
      this.board[row1][col1].row = row1;
      this.board[row1][col1].col = col1;
      this.board[row2][col2].row = row2;
      this.board[row2][col2].col = col2;
      
      this.renderBoard();
    }
    
    this.isAnimating = false;
    document.querySelector('.gem.selected')?.classList.remove('selected');
    this.selectedGem = null;
  }

  findMatches() {
    const matches = [];
    
    // 检查水平匹配
    for (let row = 0; row < BOARD_SIZE; row++) {
      for (let col = 0; col < BOARD_SIZE - 2; col++) {
        const type = this.board[row][col].type.id;
        if (type === this.board[row][col + 1].type.id && 
            type === this.board[row][col + 2].type.id) {
          
          let matchLength = 3;
          while (col + matchLength < BOARD_SIZE && 
                 this.board[row][col + matchLength].type.id === type) {
            matchLength++;
          }
          
          for (let i = 0; i < matchLength; i++) {
            matches.push({ row, col: col + i });
          }
          
          col += matchLength - 1;
        }
      }
    }
    
    // 检查垂直匹配
    for (let col = 0; col < BOARD_SIZE; col++) {
      for (let row = 0; row < BOARD_SIZE - 2; row++) {
        const type = this.board[row][col].type.id;
        if (type === this.board[row + 1][col].type.id && 
            type === this.board[row + 2][col].type.id) {
          
          let matchLength = 3;
          while (row + matchLength < BOARD_SIZE && 
                 this.board[row + matchLength][col].type.id === type) {
            matchLength++;
          }
          
          for (let i = 0; i < matchLength; i++) {
            matches.push({ row: row + i, col });
          }
          
          row += matchLength - 1;
        }
      }
    }
    
    // 去重
    const uniqueMatches = [];
    const matchSet = new Set();
    
    for (const match of matches) {
      const key = `${match.row},${match.col}`;
      if (!matchSet.has(key)) {
        matchSet.add(key);
        uniqueMatches.push(match);
      }
    }
    
    return uniqueMatches;
  }

  async handleMatches(matches) {
    // 标记匹配的宝石
    for (const match of matches) {
      this.board[match.row][match.col].isMatched = true;
    }
    
    // 更新分数
    this.score += matches.length * 10;
    document.getElementById('score').textContent = this.score;
    
    // 移除匹配的宝石
    await this.delay(300);
    for (const match of matches) {
      this.board[match.row][match.col] = null;
    }
    
    // 宝石下落
    await this.dropGems();
    
    // 填充新宝石
    await this.fillBoard();
    
    // 检查新的匹配
    const newMatches = this.findMatches();
    if (newMatches.length > 0) {
      await this.handleMatches(newMatches);
    }
  }

  async dropGems() {
    // 从下往上检查每一列
    for (let col = 0; col < BOARD_SIZE; col++) {
      let emptyRow = BOARD_SIZE - 1;
      
      // 从下往上移动宝石
      for (let row = BOARD_SIZE - 1; row >= 0; row--) {
        if (this.board[row][col] !== null) {
          if (row !== emptyRow) {
            this.board[emptyRow][col] = this.board[row][col];
            this.board[emptyRow][col].row = emptyRow;
            this.board[row][col] = null;
          }
          emptyRow--;
        }
      }
    }
    
    this.renderBoard();
    await this.delay(300);
  }

  async fillBoard() {
    // 从上往下填充新宝石
    for (let col = 0; col < BOARD_SIZE; col++) {
      for (let row = 0; row < BOARD_SIZE; row++) {
        if (this.board[row][col] === null) {
          const type = Math.floor(Math.random() * GEM_TYPE_COUNT);
          this.board[row][col] = new Gem(GEM_TYPES[Object.keys(GEM_TYPES)[type]], row, col);
        }
      }
    }
    
    this.renderBoard();
    await this.delay(300);
  }

  delay(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}
```
