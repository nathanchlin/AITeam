# 宝石下落系统实现

我将为您实现宝石消除后的下落填充机制，包括下落动画效果和连锁反应处理。这部分代码将作为完整消消乐游戏的一个模块。

## 实现思路

1. **下落机制**：当宝石被消除后，上方的宝石会向下填充空位
2. **动画效果**：使用CSS过渡和JavaScript动画实现平滑的下落效果
3. **连锁反应**：检测新形成的匹配组合并继续消除
4. **填充新宝石**：当所有下落完成后，从顶部生成新宝石

## 代码实现

```javascript
class GemFallSystem {
  constructor(gameBoard) {
    this.gameBoard = gameBoard;
    this.board = gameBoard.board;
    this.boardElement = gameBoard.boardElement;
    this.scoreElement = gameBoard.scoreElement;
    this.score = 0;
    this.isAnimating = false;
    this.fallSpeed = 300; // 下落动画时间(毫秒)
  }

  /**
   * 处理宝石消除后的下落
   * @param {Array} matchedGems - 需要消除的宝石位置数组
   */
  async handleGemFall(matchedGems) {
    if (this.isAnimating || matchedGems.length === 0) return;
    
    this.isAnimating = true;
    
    // 标记要消除的宝石
    this.markGemsForRemoval(matchedGems);
    
    // 更新分数
    this.updateScore(matchedGems.length);
    
    // 移除匹配的宝石并让上方宝石下落
    await this.removeMatchedGemsAndFall();
    
    // 填充顶部空位
    this.fillEmptySpaces();
    
    // 检查新的匹配
    await this.checkForNewMatches();
    
    this.isAnimating = false;
  }

  /**
   * 标记要消除的宝石
   */
  markGemsForRemoval(matchedGems) {
    matchedGem.forEach(gem => {
      const gemElement = this.boardElement.children[gem.row].children[gem.col];
      gemElement.classList.add('removing');
    });
  }

  /**
   * 移除匹配的宝石并让上方宝石下落
   */
  async removeMatchedGemsAndFall() {
    // 移除匹配的宝石
    for (let col = 0; col < this.board.length; col++) {
      // 从下往上查找空位
      let emptyRow = this.board.length - 1;
      
      for (let row = this.board.length - 1; row >= 0; row--) {
        if (this.board[row][col] === null) {
          // 找到空位，上方的宝石下落
          for (let aboveRow = row - 1; aboveRow >= 0; aboveRow--) {
            if (this.board[aboveRow][col] !== null) {
              // 移动宝石数据
              this.board[emptyRow][col] = this.board[aboveRow][col];
              this.board[aboveRow][col] = null;
              
              // 更新DOM
              this.moveGemToPosition(aboveRow, col, emptyRow, col);
              
              emptyRow--;
            }
          }
        }
      }
    }
    
    // 等待动画完成
    await this.wait(this.fallSpeed);
  }

  /**
   * 移动宝石到新位置并添加动画
   */
  moveGemToPosition(fromRow, fromCol, toRow, toCol) {
    const gemElement = this.boardElement.children[fromRow].children[fromCol];
    gemElement.style.transition = `transform ${this.fallSpeed}ms ease-in-out`;
    
    // 计算移动距离
    const tileSize = this.gameBoard.tileSize;
    const translateY = (toRow - fromRow) * tileSize;
    
    gemElement.style.transform = `translateY(${translateY}px)`;
    
    // 更新数据属性
    gemElement.dataset.row = toRow;
    gemElement.dataset.col = toCol;
    
    // 动画结束后重置样式
    setTimeout(() => {
      gemElement.style.transition = '';
      gemElement.style.transform = '';
      this.boardElement.children[toRow].children[toCol] = gemElement;
    }, this.fallSpeed);
  }

  /**
   * 填充顶部空位
   */
  fillEmptySpaces() {
    for (let col = 0; col < this.board.length; col++) {
      for (let row = 0; row < this.board.length; row++) {
        if (this.board[row][col] === null) {
          // 创建新宝石
          const gemType = this.gameBoard.getRandomGemType();
          this.board[row][col] = gemType;
          
          // 创建DOM元素
          const gemElement = document.createElement('div');
          gemElement.className = 'gem';
          gemElement.dataset.row = row;
          gemElement.dataset.col = col;
          gemElement.dataset.type = gemType;
          
          // 设置初始位置（从顶部开始）
          const tileSize = this.gameBoard.tileSize;
          gemElement.style.position = 'absolute';
          gemElement.style.top = `-${(this.board.length - row) * tileSize}px`;
          gemElement.style.left = `${col * tileSize}px`;
          gemElement.style.width = `${tileSize}px`;
          gemElement.style.height = `${tileSize}px`;
          
          // 添加动画
          setTimeout(() => {
            gemElement.style.transition = `transform ${this.fallSpeed}ms ease-in-out`;
            gemElement.style.transform = `translateY(${(this.board.length - row) * tileSize}px)`;
          }, 10);
          
          // 添加到游戏板
          this.boardElement.children[row].children[col] = gemElement;
          this.boardElement.appendChild(gemElement);
        }
      }
    }
    
    // 等待填充动画完成
    return this.wait(this.fallSpeed);
  }

  /**
   * 检查新的匹配
   */
  async checkForNewMatches() {
    const newMatches = this.gameBoard.findMatches();
    if (newMatches.length > 0) {
      // 递归处理新的匹配
      await this.handleGemFall(newMatches);
    }
  }

  /**
   * 更新分数
   */
  updateScore(matchedCount) {
    // 每个宝石10分，连锁反应有额外加分
    const baseScore = matchedCount * 10;
    const comboBonus = matchedCount > 3 ? (matchedCount - 3) * 5 : 0;
    this.score += baseScore + comboBonus;
    
    if (this.scoreElement) {
      this.scoreElement.textContent = this.score;
    }
  }

  /**
   * 工具函数：等待指定毫秒
   */
  wait(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}
```

  <script>
    class Match3Game {
      constructor() {
        this.boardSize = 8;
        this.tileSize = 50;
        this.board = [];
        this.boardElement = document.getElementById('gameBoard');
        this.scoreElement = document.getElementById('score');
        this.messageElement = document.getElementById('message');
        this.selectedGem = null;
        this.isAnimating = false;
        
        // 宝石类型
        this.gemTypes = 6;
        this.gemEmojis = ['💎', '💚', '💛', '💜', '🧡', '💙'];
        
        // 初始化游戏
        this.init();
      }
      
      init() {
        // 创建游戏板
        this.createBoard();
        
        // 初始化宝石
        this.initializeGems();
        
        // 添加点击事件
        this.addClickEvents();
        
        // 初始化下落系统
        this.fallSystem = new GemFallSystem(this);
      }
      
      createBoard() {
        // 设置游戏板样式
        this.boardElement.style.width = `${this.boardSize * this.tileSize}px`;
        this.boardElement.style.height = `${this.boardSize * this.tileSize}px`;
        
        // 创建网格
        for (let row = 0; row < this.boardSize; row++) {
          this.board[row] = [];
          for (let col = 0; col < this.boardSize; col++) {
            this.board[row][col] = null;
            
            // 创建单元格
            const cell = document.createElement('div');
            cell.style.position = 'absolute';
            cell.style.width = `${this.tileSize}px`;
            cell.style.height = `${this.tileSize}px`;
            cell.style.top = `${row * this.tileSize}px`;
            cell.style.left = `${col * this.tileSize}px`;
            
            this.boardElement.appendChild(cell);
          }
        }
      }
      
      initializeGems() {
        // 随机填充宝石
        for (let row = 0; row < this.boardSize; row++) {
          for (let col = 0; col < this.boardSize; col++) {
            const gemType = this.getRandomGemType();
            this.board[row][col] = gemType;
            this.createGemElement(row, col, gemType);
          }
        }
        
        // 确保初始状态没有匹配
        this.removeInitialMatches();
      }
      
      getRandomGemType() {
        return Math.floor(Math.random() * this.gemTypes) + 1;
      }
      
      createGemElement(row, col, type) {
        const gem = document.createElement('div');
        gem.className = `gem gem-type-${type}`;
        gem.dataset.row = row;
        gem.dataset.col = col;
        gem.dataset.type = type;
        gem.textContent = this.gemEmojis[type - 1];
        
        gem.style.position = 'absolute';
        gem.style.width = `${this.tileSize - 4}px`;
        gem.style.height = `${this.tileSize - 4}px`;
        gem.style.top = `${row * this.tileSize + 2}px`;
        gem.style.left = `${col * this.tileSize + 2}px`;
        
        this.boardElement.appendChild(gem);
        return gem;
      }
      
      removeInitialMatches() {
        let hasMatches = true;
        while (hasMatches) {
          const matches = this.findMatches();
          if (matches.length === 0) {
            hasMatches = false;
          } else {
            // 重新生成匹配的宝石
            matches.forEach(match => {
              const gemType = this.getRandomGemType();
              this.board[match.row][match.col] = gemType;
              const gemElement = this.boardElement.children[match.row].children[match.col];
              gemElement.className = `gem gem-type-${gemType}`;
              gemElement.dataset.type = gemType;
              gemElement.textContent = this.gemEmojis[gemType - 1];
            });
          }
        }
      }
      
      addClickEvents() {
        this.boardElement.addEventListener('click', (e) => {
          if (this.isAnimating) return;
          
          const gem = e.target.closest('.gem');
          if (!gem) return;
          
          const row = parseInt(gem.dataset.row);
          const col = parseInt(gem.dataset.col);
          
          if (!this.selectedGem) {
            // 选择第一个宝石
            this.selectedGem = { row, col, element: gem };
            gem.classList.add('selected');
          } else {
            // 尝试交换宝石
            const prevRow = this.selectedGem.row;
            const prevCol = this.selectedGem.col;
            
            // 检查是否相邻
            if (this.areAdjacent(prevRow, prevCol, row, col)) {
              this.swapGems(prevRow, prevCol, row, col);
            } else {
              // 不相邻，选择新的宝石
              this.selectedGem.element.classList.remove('selected');
              this.selectedGem = { row, col, element: gem };
              gem.classList.add('selected');
            }
          }
        });
      }
      
      areAdjacent(row1, col1, row2, col2) {
        const rowDiff = Math.abs(row1 - row2);
        const colDiff = Math.abs(col1 - col2);
        return (rowDiff === 1 && colDiff === 0) || (rowDiff === 0 && colDiff === 1);
      }
      
      async swapGems(row1, col1, row2, col2) {
        this.isAnimating = true;
        
        // 交换数据
        const temp = this.board[row1][col1];
        this.board[row1][col1] = this.board[row2][col2];
        this.board[row2][col2] = temp;
        
        // 获取宝石元素
        const gem1 = this.boardElement.children[row1].children[col1];
        const gem2 = this.boardElement.children[row2].children[col2];
        
        // 交换位置
        const tempRow = gem1.dataset.row;
        const tempCol = gem1.dataset.col;
        
        gem1.dataset.row = gem2.dataset.row;
        gem1.dataset.col = gem2.dataset.col;
        gem2.dataset.row = tempRow;
        gem2.dataset.col = tempCol;
        
        // 添加动画
        gem1.style.transition = 'transform 0.3s ease';
        gem2.style.transition = 'transform 0.3s ease';
        
        gem1.style.transform = `translateX(${(col2 - col1) * this.tileSize}px) translateY(${(row2 - row1) * this.tileSize}px)`;
        gem2.style.transform = `translateX(${(col1 - col2) * this.tileSize}px) translateY(${(row1 - row2) * this.tileSize}px)`;
        
        // 等待动画完成
        await this.wait(300);
        
        // 重置样式
        gem1.style.transition = '';
        gem2.style.transition = '';
        gem1.style.transform = '';
        gem2.style.transform = '';
        
        // 检查是否有匹配
        const matches = this.findMatches();
        if (matches.length > 0) {
          // 有匹配，处理消除
          this.selectedGem.element.classList.remove('selected');
          this.selectedGem = null;
          
          await this.fallSystem.handleGemFall(matches);
        } else {
          // 没有匹配，交换回来
          const temp = this.board[row1][col1];
          this.board[row1][col1] = this.board[row2][col2];
          this.board[row2][col2] = temp;
          
          gem1.dataset.row = row1;
          gem1.dataset.col = col1;
          gem2.dataset.row = row
