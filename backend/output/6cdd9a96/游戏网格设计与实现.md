<body>

    <script>
        document.addEventListener('DOMContentLoaded', () => {
            // 游戏配置
            const BOARD_SIZE = 8;
            const GEM_TYPES = ['red', 'blue', 'green', 'yellow', 'purple', 'orange'];
            const GEM_SYMBOLS = {
                'red': '💎',
                'blue': '🔷',
                'green': '💚',
                'yellow': '⭐',
                'purple': '🔮',
                'orange': '🧡'
            };
            
            // 游戏状态
            let board = [];
            let selectedGem = null;
            let score = 0;
            let moves = 0;
            let isProcessing = false;

            // DOM 元素
            const gameBoard = document.getElementById('gameBoard');
            const scoreElement = document.getElementById('score');
            const movesElement = document.getElementById('moves');
            const messageElement = document.getElementById('message');

            // 初始化游戏板
            function initializeBoard() {
                board = [];
                gameBoard.innerHTML = '';
                
                for (let row = 0; row < BOARD_SIZE; row++) {
                    board[row] = [];
                    for (let col = 0; col < BOARD_SIZE; col++) {
                        let gemType;
                        do {
                            gemType = GEM_TYPES[Math.floor(Math.random() * GEM_TYPES.length)];
                        } while (wouldCreateMatch(row, col, gemType));
                        
                        board[row][col] = gemType;
                        createGemElement(row, col, gemType);
                    }
                }
            }

            // 检查是否会创建匹配
            function wouldCreateMatch(row, col, gemType) {
                // 检查水平方向
                let horizontalCount = 1;
                
                // 向左检查
                for (let c = col - 1; c >= 0 && board[row][c] === gemType; c--) {
                    horizontalCount++;
                }
                
                // 如果水平方向已经有2个或更多相同宝石，放置这个会创建匹配
                if (horizontalCount >= 2) return true;
                
                // 检查垂直方向
                let verticalCount = 1;
                
                // 向上检查
                for (let r = row - 1; r >= 0 && board[r] && board[r][col] === gemType; r--) {
                    verticalCount++;
                }
                
                // 如果垂直方向已经有2个或更多相同宝石，放置这个会创建匹配
                if (verticalCount >= 2) return true;
                
                return false;
            }

            // 创建宝石元素
            function createGemElement(row, col, gemType) {
                const gem = document.createElement('div');
                gem.className = `gem ${gemType}`;
                gem.dataset.row = row;
                gem.dataset.col = col;
                gem.textContent = GEM_SYMBOLS[gemType];
                
                gem.addEventListener('click', () => handleGemClick(row, col));
                
                gameBoard.appendChild(gem);
            }

            // 处理宝石点击
            function handleGemClick(row, col) {
                if (isProcessing) return;
                
                const clickedGem = document.querySelector(`.gem[data-row="${row}"][data-col="${col}"]`);
                
                if (!selectedGem) {
                    // 选择第一个宝石
                    selectedGem = { row, col };
                    clickedGem.classList.add('selected');
                } else {
                    // 尝试交换宝石
                    const { row: selectedRow, col: selectedCol } = selectedGem;
                    
                    // 检查是否相邻
                    if (isAdjacent(selectedRow, selectedCol, row, col)) {
                        swapGems(selectedRow, selectedCol, row, col);
                    } else {
                        // 选择新的宝石
                        document.querySelector('.gem.selected').classList.remove('selected');
                        selectedGem = { row, col };
                        clickedGem.classList.add('selected');
                    }
                }
            }

            // 检查两个位置是否相邻
            function isAdjacent(row1, col1, row2, col2) {
                const rowDiff = Math.abs(row1 - row2);
                const colDiff = Math.abs(col1 - col2);
                
                return (rowDiff === 1 && colDiff === 0) || (rowDiff === 0 && colDiff === 1);
            }

            // 交换宝石
            async function swapGems(row1, col1, row2, col2) {
                isProcessing = true;
                
                // 交换数据
                const temp = board[row1][col1];
                board[row1][col1] = board[row2][col2];
                board[row2][col2] = temp;
                
                // 更新UI
                updateGem(row1, col1);
                updateGem(row2, col2);
                
                // 清除选择
                document.querySelector('.gem.selected').classList.remove('selected');
                selectedGem = null;
                
                // 增加步数
                moves++;
                movesElement.textContent = moves;
                
                // 检查匹配
                await sleep(300);
                const matches = findMatches();
                
                if (matches.length > 0) {
                    await processMatches(matches);
                } else {
                    // 没有匹配，交换回来
                    const temp = board[row1][col1];
                    board[row1][col1] = board[row2][col2];
                    board[row2][col2] = temp;
                    
                    updateGem(row1, col1);
                    updateGem(row2, col2);
                    
                    showMessage('没有找到匹配！');
                }
                
                isProcessing = false;
            }

            // 更新宝石UI
            function updateGem(row, col) {
                const gem = document.querySelector(`.gem[data-row="${row}"][data-col="${col}"]`);
                const gemType = board[row][col];
                
                gem.className = `gem ${gemType}`;
                gem.textContent = GEM_SYMBOLS[gemType];
            }

            // 查找匹配
            function findMatches() {
                const matches = [];
                
                // 检查水平匹配
                for (let row = 0; row < BOARD_SIZE; row++) {
                    for (let col = 0; col < BOARD_SIZE - 2; col++) {
                        const gemType = board[row][col];
                        if (gemType && gemType === board[row][col + 1] && gemType === board[row][col + 2]) {
                            const match = [{ row, col }, { row, col: col + 1 }, { row, col: col + 2 }];
                            
                            // 检查是否有更多相同的
                            for (let c = col + 3; c < BOARD_SIZE && board[row][c] === gemType; c++) {
                                match.push({ row, col: c });
                            }
                            
                            matches.push(match);
                            col += match.length - 1; // 跳过已匹配的宝石
                        }
                    }
                }
                
                // 检查垂直匹配
                for (let col = 0; col < BOARD_SIZE; col++) {
                    for (let row = 0; row < BOARD_SIZE - 2; row++) {
                        const gemType = board[row][col];
                        if (gemType && gemType === board[row + 1][col] && gemType === board[row + 2][col]) {
                            const match = [{ row, col }, { row: row + 1, col }, { row: row + 2, col }];
                            
                            // 检查是否有更多相同的
                            for (let r = row + 3; r < BOARD_SIZE && board[r][col] === gemType; r++) {
                                match.push({ row: r, col });
                            }
                            
                            matches.push(match);
                            row += match.length - 1; // 跳过已匹配的宝石
                        }
                    }
                }
                
                return matches;
            }

            // 处理匹配
            async function processMatches(matches) {
                // 标记要移除的宝石
                const gemsToRemove = new Set();
                matches.forEach(match => {
                    match.forEach(gem => {
                        gemsToRemove.add(`${gem.row},${gem.col}`);
                    });
                });
                
                // 添加移除动画
                gemsToRemove.forEach(key => {
                    const [row, col] = key.split(',').map(Number);
                    const gem = document.querySelector(`.gem[data-row="${row}"][data-col="${col}"]`);
                    gem.classList.add('removing');
                });
                
                // 更新分数
                const points = gemsToRemove.size * 10;
                score += points;
                scoreElement.textContent = score;
                
                // 显示得分信息
                if (gemsToRemove.size >= 3) {
                    showMessage(`消除了 ${gemsToRemove.size} 个宝石！+${points} 分`);
                }
                
                // 等待动画完成
                await sleep(300);
                
                // 移除宝石并下落
                await removeGemsAndDrop(gemsToRemove);
                
                // 检查新的匹配
                await sleep(300);
                const newMatches = findMatches();
                
                if (newMatches.length > 0) {
                    await processMatches(newMatches);
                }
            }

            // 移除宝石并下落
            async function removeGemsAndDrop(gemsToRemove) {
                // 移除宝石并创建空位
                gemsToRemove.forEach(key => {
                    const [row, col] = key.split(',').map(Number);
                    board[row][col] = null;
                });
                
                // 让宝石下落
                for (let col = 0; col < BOARD_SIZE; col++) {
                    let emptyRow = BOARD_SIZE - 1;
                    
                    for (let row = BOARD_SIZE - 1; row >= 0; row--) {
                        if (board[row][col] !== null) {
                            if (row !== emptyRow) {
                                board[emptyRow][col] = board[row][col];
                                board[row][col] = null;
                            }
                            emptyRow--;
                        }
                    }
                    
                    // 填充新宝石
                    for (let row = emptyRow; row >= 0; row--) {
                        const gemType = GEM_TYPES[Math.floor(Math.random() * GEM_TYPES.length)];
                        board[row][col] = gemType;
                    }
                }
                
                // 更新UI
                for (let row = 0; row < BOARD_SIZE; row++) {
                    for (let col = 0; col < BOARD_SIZE; col++) {
                        const gem = document.querySelector(`.gem[data-row="${row}"][data-col="${col}"]`);
                        
                        if (board[row][col]) {
                            const gemType = board[row][col];
                            gem.className = `gem ${gemType} falling`;
                            gem.textContent = GEM_SYMBOLS[gemType];
                            
                            // 移除falling类
                            setTimeout(() => {
                                gem.classList.remove('falling');
                            }, 500);
                        }
                    }
                }
                
                // 等待动画完成
                await sleep(500);
            }

            // 显示消息
            function showMessage(text) {
                messageElement.textContent = text;
                messageElement.classList.add('show');
                
                setTimeout(() => {
                    messageElement.classList.remove('show');
                }, 2000);
            }

            // 延迟函数
            function sleep(ms) {
                return new Promise(resolve => setTimeout(resolve, ms));
            }

            // 初始化游戏
            initializeBoard();
        });
    </script>
</body>
</html>
