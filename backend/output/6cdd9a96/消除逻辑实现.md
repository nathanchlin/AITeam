    <script>
        // 游戏配置
        const BOARD_SIZE = 8;
        const GEM_TYPES = ['💎', '💚', '💙', '💛', '❤️', '🧡', '💜'];
        const MIN_MATCH = 3;
        const COMBO_MULTIPLIER = 1.5;
        
        // 游戏状态
        let board = [];
        let score = 0;
        let combo = 0;
        let highScore = localStorage.getItem('gemHighScore') || 0;
        let selectedGem = null;
        let isProcessing = false;
        let isAnimating = false;
        
        // DOM元素
        const gameBoard = document.getElementById('gameBoard');
        const scoreElement = document.getElementById('score');
        const comboElement = document.getElementById('combo');
        const highScoreElement = document.getElementById('highScore');
        const comboIndicator = document.getElementById('comboIndicator');
        const messageElement = document.getElementById('message');
        const restartBtn = document.getElementById('restartBtn');
        
        // 初始化游戏
        function initGame() {
            board = [];
            score = 0;
            combo = 0;
            selectedGem = null;
            isProcessing = false;
            isAnimating = false;
            
            updateScore();
            updateCombo();
            highScoreElement.textContent = highScore;
            
            // 初始化游戏板
            for (let row = 0; row < BOARD_SIZE; row++) {
                board[row] = [];
                for (let col = 0; col < BOARD_SIZE; col++) {
                    let gemType;
                    do {
                        gemType = getRandomGemType();
                    } while (hasInitialMatch(row, col, gemType));
                    
                    board[row][col] = gemType;
                    createGemElement(row, col, gemType);
                }
            }
            
            // 确保初始板面没有匹配
            while (findMatches().length > 0) {
                reshuffleBoard();
            }
        }
        
        // 获取随机宝石类型
        function getRandomGemType() {
            return GEM_TYPES[Math.floor(Math.random() * GEM_TYPES.length)];
        }
        
        // 检查初始位置是否会形成匹配
        function hasInitialMatch(row, col, gemType) {
            // 检查水平方向
            let horizontalCount = 1;
            for (let c = col - 1; c >= 0 && board[row][c] === gemType; c--) {
                horizontalCount++;
            }
            for (let c = col + 1; c < BOARD_SIZE && board[row] && board[row][c] === gemType; c++) {
                horizontalCount++;
            }
            
            // 检查垂直方向
            let verticalCount = 1;
            for (let r = row - 1; r >= 0 && board[r][col] === gemType; r--) {
                verticalCount++;
            }
            for (let r = row + 1; r < BOARD_SIZE && board[r] && board[r][col] === gemType; r++) {
                verticalCount++;
            }
            
            return horizontalCount >= MIN_MATCH || verticalCount >= MIN_MATCH;
        }
        
        // 创建宝石元素
        function createGemElement(row, col, gemType) {
            const gem = document.createElement('div');
            gem.className = 'gem';
            gem.textContent = gemType;
            gem.dataset.row = row;
            gem.dataset.col = col;
            gem.addEventListener('click', handleGemClick);
            gameBoard.appendChild(gem);
        }
        
        // 处理宝石点击
        function handleGemClick(e) {
            if (isProcessing || isAnimating) return;
            
            const clickedGem = e.target;
            const row = parseInt(clickedGem.dataset.row);
            const col = parseInt(clickedGem.dataset.col);
            
            if (!selectedGem) {
                // 选择第一个宝石
                selectedGem = { row, col, element: clickedGem };
                clickedGem.classList.add('selected');
            } else {
                // 尝试交换宝石
                const prevRow = selectedGem.row;
                const prevCol = selectedGem.col;
                
                // 检查是否相邻
                if (isAdjacent(row, col, prevRow, prevCol)) {
                    swapGems(row, col, prevRow, prevCol);
                } else {
                    // 选择新的宝石
                    selectedGem.element.classList.remove('selected');
                    selectedGem = { row, col, element: clickedGem };
                    clickedGem.classList.add('selected');
                }
            }
        }
        
        // 检查两个位置是否相邻
        function isAdjacent(row1, col1, row2, col2) {
            const rowDiff = Math.abs(row1 - row2);
            const colDiff = Math.abs(col1 - col2);
            return (rowDiff === 1 && colDiff === 0) || (rowDiff === 0 && colDiff === 1);
        }
        
        // 交换宝石
        function swapGems(row1, col1, row2, col2) {
            isProcessing = true;
            
            // 交换数据
            const temp = board[row1][col1];
            board[row1][col1] = board[row2][col2];
            board[row2][col2] = temp;
            
            // 更新UI
            updateGemElement(row1, col1, board[row1][col1]);
            updateGemElement(row2, col2, board[row2][col2]);
            
            // 移除选中状态
            selectedGem.element.classList.remove('selected');
            selectedGem = null;
            
            // 检查是否有匹配
            setTimeout(() => {
                const matches = findMatches();
                if (matches.length > 0) {
                    processMatches(matches);
                } else {
                    // 没有匹配，交换回来
                    const temp = board[row1][col1];
                    board[row1][col1] = board[row2][col2];
                    board[row2][col2] = temp;
                    
                    updateGemElement(row1, col1, board[row1][col1]);
                    updateGemElement(row2, col2, board[row2][col2]);
                    
                    isProcessing = false;
                }
            }, 300);
        }
        
        // 更新宝石元素
        function updateGemElement(row, col, gemType) {
            const gem = document.querySelector(`.gem[data-row="${row}"][data-col="${col}"]`);
            if (gem) {
                gem.textContent = gemType;
            }
        }
        
        // 查找所有匹配
        function findMatches() {
            const matches = [];
            
            // 检查水平匹配
            for (let row = 0; row < BOARD_SIZE; row++) {
                for (let col = 0; col < BOARD_SIZE - MIN_MATCH + 1; col++) {
                    const gemType = board[row][col];
                    if (!gemType) continue;
                    
                    let matchLength = 1;
                    for (let c = col + 1; c < BOARD_SIZE && board[row][c] === gemType; c++) {
                        matchLength++;
                    }
                    
                    if (matchLength >= MIN_MATCH) {
                        const match = [];
                        for (let c = col; c < col + matchLength; c++) {
                            match.push({ row, col: c });
                        }
                        matches.push(match);
                        col += matchLength - 1;
                    }
                }
            }
            
            // 检查垂直匹配
            for (let col = 0; col < BOARD_SIZE; col++) {
                for (let row = 0; row < BOARD_SIZE - MIN_MATCH + 1; row++) {
                    const gemType = board[row][col];
                    if (!gemType) continue;
                    
                    let matchLength = 1;
                    for (let r = row + 1; r < BOARD_SIZE && board[r][col] === gemType; r++) {
                        matchLength++;
                    }
                    
                    if (matchLength >= MIN_MATCH) {
                        const match = [];
                        for (let r = row; r < row + matchLength; r++) {
                            match.push({ row: r, col });
                        }
                        matches.push(match);
                        row += matchLength - 1;
                    }
                }
            }
            
            return matches;
        }
        
        // 处理匹配
        function processMatches(matches) {
            isAnimating = true;
            
            // 高亮显示匹配的宝石
            highlightMatches(matches);
            
            // 计算得分
            const totalMatched = matches.reduce((sum, match) => sum + match.length, 0);
            const points = calculatePoints(totalMatched);
            score += points;
            updateScore();
            
            // 显示连击
            if (matches.length > 1) {
                combo++;
                updateCombo();
                showComboIndicator(combo);
            } else {
                combo = 0;
                updateCombo();
            }
            
            // 延迟后消除宝石
            setTimeout(() => {
                removeMatches(matches);
                
                // 宝石下落
                setTimeout(() => {
                    dropGems();
                    
                    // 填充新宝石
                    setTimeout(() => {
                        fillBoard();
                        
                        // 检查是否有新的匹配
                        setTimeout(() => {
                            const newMatches = findMatches();
                            if (newMatches.length > 0) {
                                processMatches(newMatches);
                            } else {
                                isAnimating = false;
                                isProcessing = false;
                            }
                        }, 300);
                    }, 300);
                }, 300);
            }, 500);
        }
        
        // 高亮显示匹配的宝石
        function highlightMatches(matches) {
            matches.forEach(match => {
                match.forEach(({ row, col }) => {
                    const gem = document.querySelector(`.gem[data-row="${row}"][data-col="${col}"]`);
                    if (gem) {
                        gem.classList.add('matched');
                    }
                });
            });
        }
        
        // 计算得分
        function calculatePoints(matchedCount) {
            let points = matchedCount * 10;
            
            // 连击加成
            if (combo > 0) {
                points *= Math.pow(COMBO_MULTIPLIER, combo);
            }
            
            // 大匹配额外加分
            if (matchedCount > 5) {
                points += 50;
            } else if (matchedCount > 3) {
                points += 20;
            }
            
            return Math.floor(points);
        }
        
        // 更新分数显示
        function updateScore() {
            scoreElement.textContent = score;
            
            // 更新最高分
            if (score > highScore) {
                highScore = score;
                highScoreElement.textContent = highScore;
                localStorage.setItem('gemHighScore', highScore);
            }
        }
        
        // 更新连击显示
        function updateCombo() {
            comboElement.textContent = combo;
        }
        
        // 显示连击指示器
        function showComboIndicator(comboCount) {
            comboIndicator.textContent = `${comboCount}x 连击!`;
            comboIndicator.classList.add('show');
            
            setTimeout(() => {
                comboIndicator.classList.remove('show');
            }, 1000);
        }
        
        // 移除匹配的宝石
        function removeMatches(matches) {
            matches.forEach(match => {
                match.forEach(({ row, col }) => {
                    board[row][col] = null;
                });
            });
            
            // 从DOM中移除匹配的宝石
            setTimeout(() => {
                matches.forEach(match => {
                    match.forEach(({ row, col }) => {
                        const gem = document.querySelector(`.gem[data-row="${row}"][data-col="${col}"]`);
                        if (gem) {
                            gem.remove();
                        }
                    });
                });
            }, 500);
        }
        
        // 宝石下落
        function dropGems() {
            for (let col = 0; col < BOARD_SIZE; col++) {
                let
