        // 游戏配置
        const BOARD_SIZE = 8;
        const GEM_TYPES = ['♦', '♠', '♣', '♥', '★', '◆'];
        const GEM_CLASSES = ['gem-red', 'gem-blue', 'gem-green', 'gem-yellow', 'gem-purple', 'gem-orange'];
        const INITIAL_MOVES = 30;
        const HINT_COUNT = 3;
        
        // 游戏状态
        let board = [];
        let score = 0;
        let moves = INITIAL_MOVES;
        let level = 1;
        let selectedGem = null;
        let isProcessing = false;
        let hintsRemaining = HINT_COUNT;
        let isPaused = false;
        
        // DOM元素
        const gameBoard = document.getElementById('gameBoard');
        const scoreElement = document.getElementById('score');
        const movesElement = document.getElementById('moves');
        const levelElement = document.getElementById('level');
        const messageElement = document.getElementById('message');
        const gameOverModal = document.getElementById('gameOverModal');
        const modalTitle = document.getElementById('modalTitle');
        const modalMessage = document.getElementById('modalMessage');
        const finalScoreElement = document.getElementById('finalScore');
        
        // 初始化游戏
        function initGame() {
            board = [];
            score = 0;
            moves = INITIAL_MOVES;
            level = 1;
            selectedGem = null;
            isProcessing = false;
            hintsRemaining = HINT_COUNT;
            isPaused = false;
            
            updateUI();
            createBoard();
            renderBoard();
            
            // 确保初始板面没有匹配
            while (checkForMatches().length > 0) {
                createBoard();
            }
            
            renderBoard();
            showMessage("游戏开始! 交换相邻宝石进行匹配");
        }
        
        // 创建游戏板
        function createBoard() {
            for (let row = 0; row < BOARD_SIZE; row++) {
                board[row] = [];
                for (let col = 0; col < BOARD_SIZE; col++) {
                    board[row][col] = Math.floor(Math.random() * GEM_TYPES.length);
                }
            }
        }
        
        // 渲染游戏板
        function renderBoard() {
            gameBoard.innerHTML = '';
            
            for (let row = 0; row < BOARD_SIZE; row++) {
                for (let col = 0; col < BOARD_SIZE; col++) {
                    const gem = document.createElement('div');
                    gem.className = `gem ${GEM_CLASSES[board[row][col]]}`;
                    gem.textContent = GEM_TYPES[board[row][col]];
                    gem.dataset.row = row;
                    gem.dataset.col = col;
                    gem.addEventListener('click', handleGemClick);
                    gameBoard.appendChild(gem);
                }
            }
        }
        
        // 处理宝石点击
        function handleGemClick(e) {
            if (isProcessing || isPaused) return;
            
            const row = parseInt(e.target.dataset.row);
            const col = parseInt(e.target.dataset.col);
            
            if (!selectedGem) {
                selectedGem = { row, col };
                e.target.classList.add('selected');
            } else {
                const prevSelected = document.querySelector('.gem.selected');
                if (prevSelected) prevSelected.classList.remove('selected');
                
                // 检查是否是相邻宝石
                const isAdjacent = 
                    (Math.abs(selectedGem.row - row) === 1 && selectedGem.col === col) ||
                    (Math.abs(selectedGem.col - col) === 1 && selectedGem.row === row);
                
                if (isAdjacent) {
                    // 交换宝石
                    swapGems(selectedGem.row, selectedGem.col, row, col);
                    
                    // 检查是否有匹配
                    const matches = checkForMatches();
                    if (matches.length > 0) {
                        moves--;
                        updateUI();
                        processMatches();
                    } else {
                        // 没有匹配，交换回来
                        swapGems(selectedGem.row, selectedGem.col, row, col);
                        showMessage("没有找到匹配，请尝试其他交换");
                    }
                }
                
                selectedGem = null;
            }
        }
        
        // 交换宝石
        function swapGems(row1, col1, row2, col2) {
            const temp = board[row1][col1];
            board[row1][col1] = board[row2][col2];
            board[row2][col2] = temp;
            renderBoard();
        }
        
        // 检查匹配
        function checkForMatches() {
            const matches = [];
            
            // 检查水平匹配
            for (let row = 0; row < BOARD_SIZE; row++) {
                for (let col = 0; col < BOARD_SIZE - 2; col++) {
                    const gemType = board[row][col];
                    if (gemType !== null && 
                        gemType === board[row][col + 1] && 
                        gemType === board[row][col + 2]) {
                        let matchLength = 3;
                        while (col + matchLength < BOARD_SIZE && board[row][col + matchLength] === gemType) {
                            matchLength++;
                        }
                        
                        for (let i = 0; i < matchLength; i++) {
                            matches.push({ row, col: col + i });
                        }
                        
                        col += matchLength - 1;
                    }
                }
            }
            
            // 检查垂直匹配
            for (let col = 0; col < BOARD_SIZE; col++) {
                for (let row = 0; row < BOARD_SIZE - 2; row++) {
                    const gemType = board[row][col];
                    if (gemType !== null && 
                        gemType === board[row + 1][col] && 
                        gemType === board[row + 2][col]) {
                        let matchLength = 3;
                        while (row + matchLength < BOARD_SIZE && board[row + matchLength][col] === gemType) {
                            matchLength++;
                        }
                        
                        for (let i = 0; i < matchLength; i++) {
                            matches.push({ row: row + i, col });
                        }
                        
                        row += matchLength - 1;
                    }
                }
            }
            
            // 去重
            const uniqueMatches = [];
            const matchSet = new Set();
            
            for (const match of matches) {
                const key = `${match.row},${match.col}`;
                if (!matchSet.has(key)) {
                    matchSet.add(key);
                    uniqueMatches.push(match);
                }
            }
            
            return uniqueMatches;
        }
        
        // 处理匹配
        function processMatches() {
            isProcessing = true;
            const matches = checkForMatches();
            
            if (matches.length > 0) {
                // 计算得分
                const points = matches.length * 10 * level;
                score += points;
                updateUI();
                
                // 移除匹配的宝石
                for (const match of matches) {
                    board[match.row][match.col] = null;
                }
                
                // 显示得分动画
                showScoreAnimation(points);
                
                // 下落宝石
                setTimeout(() => {
                    dropGems();
                    fillEmptySpaces();
                    
                    // 检查是否有新的匹配
                    setTimeout(() => {
                        const newMatches = checkForMatches();
                        if (newMatches.length > 0) {
                            processMatches();
                        } else {
                            isProcessing = false;
                            checkGameState();
                        }
                    }, 300);
                }, 300);
            } else {
                isProcessing = false;
                checkGameState();
            }
        }
        
        // 宝石下落
        function dropGems() {
            for (let col = 0; col < BOARD_SIZE; col++) {
                let emptyRow = BOARD_SIZE - 1;
                
                for (let row = BOARD_SIZE - 1; row >= 0; row--) {
                    if (board[row][col] !== null) {
                        if (row !== emptyRow) {
                            board[emptyRow][col] = board[row][col];
                            board[row][col] = null;
                        }
                        emptyRow--;
                    }
                }
            }
            
            renderBoard();
        }
        
        // 填充空位
        function fillEmptySpaces() {
            for (let col = 0; col < BOARD_SIZE; col++) {
                for (let row = 0; row < BOARD_SIZE; row++) {
                    if (board[row][col] === null) {
                        board[row][col] = Math.floor(Math.random() * GEM_TYPES.length);
                    }
                }
            }
            
            renderBoard();
        }
        
        // 检查游戏状态
        function checkGameState() {
            if (moves <= 0) {
                endGame(false);
            } else if (score >= level * 1000) {
                levelUp();
            }
        }
        
        // 升级
        function levelUp() {
            level++;
            moves += 10;
            hintsRemaining += 2;
            updateUI();
            showMessage(`恭喜升级到第 ${level} 级! 获得10步额外步数和2个提示`);
            
            // 增加游戏难度
            if (level % 3 === 0) {
