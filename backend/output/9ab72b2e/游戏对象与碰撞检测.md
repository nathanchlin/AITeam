extends Node2D

# 定义一些常量
const GRAVITY = 10.0
const JUMP_SPEED = -15.0
const GROUND_HEIGHT = 620.0
const SCREEN_WIDTH = 720.0
const SCREEN_HEIGHT = 1280.0

var velocity = Vector2()
var is_on_ground = false
var is_touching = false

func _ready():
    # 设置屏幕尺寸
    get_tree().root().get_node("Camera2D").size = Vector2(1, 1.78)
    # 初始化玩家位置
    position = Vector2(360, 570)

func _process(delta):
    # 处理触摸输入
    if Input.is_action_pressed("ui_accept"):
        if is_on_ground:
            velocity.y = JUMP_SPEED
            is_on_ground = false
        elif not is_touching:
            # 确定触摸位置
            var touch_position = Input.get_touch_position(0)
            # 如果玩家在触摸位置上方，则认为是跳跃
            if touch_position.y < position.y:
                velocity.y = JUMP_SPEED
                is_on_ground = false
    else:
        # 如果没有触摸，重置is_touching标志
        is_touching = false
    
    # 更新玩家位置
    velocity.x = 0
    velocity.y += GRAVITY * delta
    position += velocity * delta

    # 检查是否在地面
    if position.y + get_node("Player").size.y / 2 >= GROUND_HEIGHT:
        position.y = GROUND_HEIGHT - get_node("Player").size.y / 2
        velocity.y = 0
        is_on_ground = true

    # 碰撞检测
    if _check_collision_with("Obstacle"):
        position = Vector2(360, 570)  # 重置玩家位置到初始位置
        velocity = Vector2()          # 重置玩家速度

func _draw():
    # 绘制背景
    draw_rect(Rect2(0, 0, SCREEN_WIDTH, SCREEN_HEIGHT), Color(0, 0, 0, 1))

    # 绘制玩家
    draw_rect(Rect2(position.x - get_node("Player").size.x / 2,
                     position.y - get_node("Player").size.y / 2,
                     get_node("Player").size.x,
                     get_node("Player").size.y), Color(255, 255, 255, 255))

    # 绘制地面
    draw_rect(Rect2(0, GROUND_HEIGHT - 50, SCREEN_WIDTH, 50), Color(0, 255, 0, 255))

    # 绘制平台
    draw_rect(Rect2(360 - 100, 400 - 50, 200, 50), Color(0, 0, 255, 255))

    # 绘制障碍物
    draw_rect(Rect2(600 - 25, 500 - 50, 50, 100), Color(255, 0, 255, 255))

func _check_collision_with(node_name):
    # 检查玩家是否与名为node_name的节点碰撞
    var player_rect = Rect2(position.x - get_node("Player").size.x / 2,
                            position.y - get_node("Player").size.y / 2,
                            get_node("Player").size.x,
                            get_node("Player").size.y)
    var obstacle_rect = get_node(node_name).get_rect()
    return player_rect.intersects(obstacle_rect)
