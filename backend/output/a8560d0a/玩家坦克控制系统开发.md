        class Bullet {
            constructor(x, y, angle, color) {
                this.x = x;
                this.y = y;
                this.angle = angle;
                this.speed = BULLET_SPEED;
                this.size = BULLET_SIZE;
                this.color = color;
                this.life = 100; // 子弹生命周期
            }
            
            update() {
                this.x += Math.cos(this.angle) * this.speed;
                this.y += Math.sin(this.angle) * this.speed;
                this.life--;
            }
            
            draw() {
                ctx.fillStyle = this.color;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                ctx.fill();
            }
            
            isAlive() {
                return this.life > 0 && 
                       this.x > 0 && 
                       this.x < CANVAS_WIDTH && 
                       this.y > 0 && 
                       this.y < CANVAS_HEIGHT;
            }

        // 检测碰撞
        function checkCollisions() {
            // 玩家子弹与敌方坦克碰撞
            playerTank.bullets.forEach((bullet, bulletIndex) => {
                enemyTanks.forEach((enemy, enemyIndex) => {
                    if (checkCollision(bullet, enemy)) {
                        // 创建爆炸效果
                        explosions.push(new Explosion(bullet.x, bullet.y));
                        
                        // 移除子弹
                        playerTank.bullets.splice(bulletIndex, 1);
                        
                        // 对敌人造成伤害
                        if (enemy.takeDamage(25)) {
                            enemyTanks.splice(enemyIndex, 1);
                        }
                    }
                });
                
                // 玩家子弹与敌方总部碰撞
                if (checkCollision(bullet, enemyHeadquarter)) {
                    explosions.push(new Explosion(bullet.x, bullet.y));
                    playerTank.bullets.splice(bulletIndex, 1);
                    enemyHeadquarter.takeDamage(10);
                }
            });
            
            // 敌方子弹与玩家坦克碰撞
            enemyTanks.forEach(enemy => {
                enemy.bullets.forEach((bullet, bulletIndex) => {
                    if (checkCollision(bullet, playerTank)) {
                        explosions.push(new Explosion(bullet.x, bullet.y));
                        enemy.bullets.splice(bulletIndex, 1);
                        playerTank.takeDamage(10);
                    }
                });
                
                // 敌方子弹与玩家总部碰撞
                enemy.bullets.forEach((bullet, bulletIndex) => {
                    if (checkCollision(bullet, playerHeadquarter)) {
                        explosions.push(new Explosion(bullet.x, bullet.y));
                        enemy.bullets.splice(bulletIndex, 1);
                        playerHeadquarter.takeDamage(10);
                    }
                });
            });
            
            // 坦克与总部碰撞
            enemyTanks.forEach(enemy => {
                if (checkCollision(enemy, playerHeadquarter)) {
                    playerHeadquarter.takeDamage(1);
                }
            });
            
            // 检查游戏结束条件
            if (playerHeadquarter.health <= 0) {
                gameRunning = false;
            }
        }
        
        // 碰撞检测函数
        function checkCollision(obj1, obj2) {
            const dx = obj1.x - obj2.x;
            const dy = obj1.y - obj2.y;
            const distance = Math.sqrt(dx * dx + dy * dy);
            
            // 对于坦克，使用宽度的一半作为碰撞半径
            const tankRadius = obj1.width ? obj1.width / 2 : 0;
            const obj2Radius = obj2.width ? obj2.width / 2 : obj2.size / 2;
            
            return distance < tankRadius + obj2Radius;
        }
        
        // 生成敌方坦克

                lastEnemySpawnTime = now;
            }
        }
