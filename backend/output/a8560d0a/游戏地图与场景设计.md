    <script>
        // 游戏常量
        const CANVAS_WIDTH = 800;
        const CANVAS_HEIGHT = 600;
        const TANK_SIZE = 30;
        const BULLET_SIZE = 5;
        const OBSTACLE_SIZE = 40;
        const HEADQUARTER_SIZE = 60;
        const TANK_SPEED = 3;
        const BULLET_SPEED = 5;
        const ENEMY_TANK_SPEED = 1.5;
        
        // 游戏状态
        let canvas, ctx;
        let gameRunning = true;
        let playerHealth = 100;
        let score = 0;
        let keys = {};
        
        // 游戏对象
        let playerTank;
        let enemyTanks = [];
        let bullets = [];
        let obstacles = [];
        let headquarters = [];
        
        // 地图设计
        const mapLayout = {
            playerHeadquarter: { x: 100, y: CANVAS_HEIGHT / 2 - HEADQUARTER_SIZE / 2 },
            enemyHeadquarter: { x: CANVAS_WIDTH - 100 - HEADQUARTER_SIZE, y: CANVAS_HEIGHT / 2 - HEADQUARTER_SIZE / 2 },
            obstacles: [
                // 横向障碍物
                { x: 200, y: 150, width: 100, height: OBSTACLE_SIZE },
                { x: 500, y: 150, width: 100, height: OBSTACLE_SIZE },
                { x: 200, y: 450, width: 100, height: OBSTACLE_SIZE },
                { x: 500, y: 450, width: 100, height: OBSTACLE_SIZE },
                
                // 纵向障碍物
                { x: 350, y: 100, width: OBSTACLE_SIZE, height: 100 },
                { x: 350, y: 400, width: OBSTACLE_SIZE, height: 100 },
                
                // 中间障碍物
                { x: 400, y: 250, width: OBSTACLE_SIZE, height: 100 },
                { x: 400, y: 300, width: OBSTACLE_SIZE, height: 100 }
            ]
        };
        
        // 坦克类
        class Tank {
            constructor(x, y, color, isPlayer = false) {
                this.x = x;
                this.y = y;
                this.width = TANK_SIZE;
                this.height = TANK_SIZE;
                this.color = color;
                this.direction = isPlayer ? 'right' : 'left';
                this.speed = isPlayer ? TANK_SPEED : ENEMY_TANK_SPEED;
                this.isPlayer = isPlayer;
                this.lastShot = 0;
                this.shootCooldown = isPlayer ? 500 : 1000;
            }
            
            draw() {
                ctx.save();
                ctx.translate(this.x + this.width / 2, this.y + this.height / 2);
                
                // 根据方向旋转坦克
                let rotation = 0;
                if (this.direction === 'up') rotation = -Math.PI / 2;
                else if (this.direction === 'down') rotation = Math.PI / 2;
                else if (this.direction === 'left') rotation = Math.PI;
                ctx.rotate(rotation);
                
                // 绘制坦克主体
                ctx.fillStyle = this.color;
                ctx.fillRect(-this.width / 2, -this.height / 2, this.width, this.height);
                
                // 绘制坦克炮管
                ctx.fillStyle = '#2c3e50';
                ctx.fillRect(this.width / 2 - 5, -3, 15, 6);
                
                // 绘制坦克履带
                ctx.fillStyle = '#1a252f';
                ctx.fillRect(-this.width / 2, -this.height / 2 + 2, this.width, 5);
                ctx.fillRect(-this.width / 2, this.height / 2 - 7, this.width, 5);
                
                ctx.restore();
            }
            
            move(direction) {
                this.direction = direction;
                let newX = this.x;
                let newY = this.y;
                
                if (direction === 'up') newY -= this.speed;
                else if (direction === 'down') newY += this.speed;
                else if (direction === 'left') newX -= this.speed;
                else if (direction === 'right') newX += this.speed;
                
                // 边界检查
                if (newX >= 0 && newX <= CANVAS_WIDTH - this.width &&
                    newY >= 0 && newY <= CANVAS_HEIGHT - this.height) {
                    
                    // 碰撞检测
                    if (!this.checkCollisionWithObstacles(newX, newY) &&
                        !this.checkCollisionWithTanks(newX, newY)) {
                        this.x = newX;
                        this.y = newY;
                    }
                }
            }
            
            checkCollisionWithObstacles(x, y) {
                for (let obstacle of obstacles) {
                    if (x < obstacle.x + obstacle.width &&
                        x + this.width > obstacle.x &&
                        y < obstacle.y + obstacle.height &&
                        y + this.height > obstacle.y) {
                        return true;
                    }
                }
                return false;
            }
            
            checkCollisionWithTanks(x, y) {
                // 检查与其他坦克的碰撞
                for (let tank of enemyTanks) {
                    if (tank !== this && 
                        x < tank.x + tank.width &&
                        x + this.width > tank.x &&
                        y < tank.y + tank.height &&
                        y + this.height > tank.y) {
                        return true;
                    }
                }
                
                // 检查与玩家坦克的碰撞
                if (!this.isPlayer && 
                    x < playerTank.x + playerTank.width &&
                    x + this.width > playerTank.x &&
                    y < playerTank.y + playerTank.height &&
                    y + this.height > playerTank.y) {
                    return true;
                }
                
                return false;
            }
            
            shoot() {
                const now = Date.now();
                if (now - this.lastShot > this.shootCooldown) {
                    this.lastShot = now;
                    
                    let bulletX = this.x + this.width / 2 - BULLET_SIZE / 2;
                    let bulletY = this.y + this.height / 2 - BULLET_SIZE / 2;
                    
                    // 根据方向调整子弹位置
                    if (this.direction === 'up') {
                        bulletY = this.y - BULLET_SIZE;
                    } else if (this.direction === 'down') {
                        bulletY = this.y + this.height;
                    } else if (this.direction === 'left') {
                        bulletX = this.x - BULLET_SIZE;
                    } else if (this.direction === 'right') {
                        bulletX = this.x + this.width;
                    }
                    
                    bullets.push(new Bullet(bulletX, bulletY, this.direction, this.isPlayer));
                }
            }
        }
        
        // 子弹类
        class Bullet {
            constructor(x, y, direction, isPlayerBullet) {
                this.x = x;
                this.y = y;
                this.width = BULLET_SIZE;
                this.height = BULLET_SIZE;
                this.direction = direction;
                this.speed = BULLET_SPEED;
                this.isPlayerBullet = isPlayerBullet;
            }
            
            draw() {
                ctx.fillStyle = this.isPlayerBullet ? '#f1c40f' : '#e74c3c';
                ctx.fillRect(this.x, this.y, this.width, this.height);
            }
            
            move() {
                if (this.direction === 'up') this.y -= this.speed;
                else if (this.direction === 'down') this.y += this.speed;
                else if (this.direction === 'left') this.x -= this.speed;
                else if (this.direction === 'right') this.x += this.speed;
            }
            
            isOutOfBounds() {
                return this.x < 0 || this.x > CANVAS_WIDTH || 
                       this.y < 0 || this.y > CANVAS_HEIGHT;
            }
        }
        
        // 总部类
        class Headquarter {
            constructor(x, y, isPlayer) {
                this.x = x;
                this.y = y;
                this.width = HEADQUARTER_SIZE;
                this.height = HEADQUARTER_SIZE;
                this.isPlayer = isPlayer;
                this.health = 100;
            }
            
            draw() {
                // 绘制总部建筑
                ctx.fillStyle = this.isPlayer ? '#3498db' : '#e74c3c';
                ctx.fillRect(this.x, this.y, this.width, this.height);
                
                // 绘制总部旗帜
                ctx.fillStyle = this.isPlayer ? '#f1c40f' : '#ecf0f1';
                ctx.beginPath();
                ctx.moveTo(this.x + this.width / 2, this.y);
                ctx.lineTo(this.x + this.width / 2 + 20, this.y + 10);
                ctx.lineTo(this.x + this.width / 2, this.y + 20);
                ctx.closePath();
                ctx.fill();
                
                // 绘制总部标识
                ctx.fillStyle = 'white';
                ctx.font = '16px Arial';
                ctx.textAlign = 'center';
                ctx.fillText(this.isPlayer ? 'P' : 'E', this.x + this.width / 2, this.y + this.height / 2 + 5);
            }
            
            takeDamage() {
                this.health -= 20;
                return this.health <= 0;
            }
        }
        
        // 初始化游戏
        function initGame() {
            canvas = document.getElementById('gameCanvas');
            ctx = canvas.getContext('2d');
            
            // 创建玩家坦克
            playerTank = new Tank(
                mapLayout.playerHeadquarter.x + HEADQUARTER_SIZE / 2 - TANK_SIZE / 2,
                mapLayout.playerHeadquarter.y + HEADQUARTER_SIZE / 2 - TANK_SIZE / 2,
                '#2ecc71',
                true
            );
            
            // 创建总部
            headquarters.push(new Headquarter(
                mapLayout.playerHeadquarter.x,
                mapLayout.playerHeadquarter.y,
                true
            ));
            
            headquarters.push(new Headquarter(
                mapLayout.enemyHeadquarter.x,
                mapLayout.enemyHeadquarter.y,
                false
            ));
            
            // 创建障碍物
            for (let obstacleData of mapLayout.obstacles) {
                obstacles.push({
                    x: obstacleData.x,
                    y: obstacleData.y,
                    width: obstacleData.width,
                    height: obstacleData.height
                });
            }
            
            // 创建初始敌人坦克
            spawnEnemyTanks();
            
            // 添加键盘事件监听
            document.addEventListener('keydown', handleKeyDown);
            document.addEventListener('keyup', handleKeyUp);
            
            // 开始游戏循环
            gameLoop();
        }
        
        // 生成敌人坦克
        function spawnEnemyTanks() {
            // 生成3个敌人坦克
            for (let i = 0; i < 3; i++) {
                let x = mapLayout.enemyHeadquarter.x + HEADQUARTER_SIZE / 2 - TANK_SIZE / 2;
                let y = mapLayout.enemyHeadquarter.y + HEADQUARTER_SIZE / 2 - TANK_SIZE / 2 + (i - 1) * 50;
                
                enemyTanks.push(new Tank(x, y, '#e74c3c', false));
            }
        }
        
        // 键盘事件处理
        function handleKeyDown(e) {
            keys[e.key] = true;
            
            // 空格键射击
            if (e.key === ' ') {
                playerTank.shoot();
            }
        }
        
        function handleKeyUp(e) {
            keys[e.key] = false;
        }
        
        // 更新游戏状态
        function update() {
            if (!gameRunning) return;
            
            // 玩家坦克移动
            if (keys['ArrowUp']) playerTank.move('up');
            if (keys['ArrowDown']) playerTank.move('down');
            if (keys['ArrowLeft']) playerTank.move('left');
            if (keys['ArrowRight']) playerTank.move('right');
            
            // 更新子弹
            for (let i = bullets.length - 1; i >= 0; i--) {
                bullets[i].move();
                
                // 检查子弹是否出界
                if (bullets[i].isOutOfBounds()) {
                    bullets.splice(i, 1);
                    continue;
                }
                
                // 检查子弹与障碍物的碰撞
                for (let obstacle of obstacles) {
                    if (bullets[i].x < obstacle.x + obstacle.width &&
                        bullets[i].x + bullets[i].width > obstacle.x &&
                        bullets[i].y < obstacle.y + obstacle.height &&
                        bullets[i].y + bullets[i].height > obstacle.y) {
                        bullets.splice(i, 1);
                        break;
                    }
                }
                
                // 检查子弹与坦克的碰撞
                if (i >= 0 && bullets[i].isPlayerBullet) {
                    // 玩家子弹击中敌人坦克
                    for (let j = enemyTanks.length - 1; j >= 0; j--) {
                        if (bullets[i].x < enemyTanks[j].x + enemyTanks[j].width &&
                            bullets[i].x + bullets[i].width > enemyTanks[j].x &&
                            bullets[i].y < enemyTanks[j].y + enemyTanks[j].height &&
                            bullets[i].y + bullets[i].height > enemyTanks[j].y) {
                            
                            // 移除子弹
                            bullets.splice(i, 1);
                            
                            // 移除敌人坦克并增加分数
                            enemyTanks.splice(j, 1);
                            score += 100;
                            document.getElementById('score').textContent = score;
                            
                            // 如果所有敌人坦克被消灭，生成新的
                            if (enemyTanks.length === 0) {
                                spawnEnemyTanks();
                            }
                            
                            break;
                        }
                    }
                } else if (i >= 0 && !bullets[i].isPlayerBullet) {
                    // 敌人子弹击中玩家
                    if (bullets[i].x < playerTank.x + playerTank.width &&
                        bullets[i].x + bullets[i].width > playerTank.x &&
                        bullets[i].y < playerTank.y + playerTank.height &&
                        bullets[i].y + bullets[i].height > playerTank.y) {
                        
                        // 移除子弹
                        bullets.splice(i, 1);
                        
                        // 减少玩家生命值
                        playerHealth -= 20;
                        document.getElementById('playerHealth').textContent = playerHealth;
                        
                        // 检查游戏是否结束
                        if (playerHealth <= 0) {
                            gameOver(false);
                        }
                    }
                }
                
                // 检查子弹与总部的碰撞
                if (i >= 0) {
                    for (let j = 0; j < headquarters.length; j++) {
                        if (bullets[i].x < headquarters[j].x + headquarters[j].width &&
                            bullets[i].x + bullets[i].width > headquarters[j].x &&
                            bullets[i].y < headquarters[j].y + headquarters[j].height &&
                            bullets[i].y + bullets[i].height > headquarters[j].y) {
                            
                            // 移除子弹
                            bullets.splice(i, 1);
                            
                            // 总部受到伤害
                            if (headquarters[j].takeDamage()) {
                                // 总部被摧毁
                                if (headquarters[j].isPlayer) {
                                    gameOver(false);
                                } else {
                                    score += 500;
                                    document.getElementById('score').textContent = score;
                                }
                            }
                            
                            break;
                        }
                    }
                }
            }
            
            // 更新敌人坦克行为
            for (let enemy of enemyTanks) {
                // 简单的AI：随机移动和射击
                if (Math.random() < 0.02) {
                    const directions = ['up', 'down', 'left', 'right'];
                    enemy.direction = directions[Math.floor(Math.random() * directions.length)];
                }
                
                enemy.move(enemy.direction);
                
                // 敌人射击
                if (Math.random() < 0.01) {
