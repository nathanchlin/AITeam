    <script>
        // 游戏状态枚举
        const GameState = {
            START: 'start',
            PLAYING: 'playing',
            GAME_OVER: 'gameOver'
        };

        // 坦克类
        class Tank {
            constructor(x, y, width, height, color, isPlayer = false) {
                this.x = x;
                this.y = y;
                this.width = width;
                this.height = height;
                this.color = color;
                this.speed = isPlayer ? 3 : 1.5;
                this.direction = 0; // 0: 上, 1: 右, 2: 下, 3: 左
                this.isPlayer = isPlayer;
                this.health = 100;
                this.maxHealth = 100;
                this.bullets = [];
                this.lastShotTime = 0;
                this.shotCooldown = isPlayer ? 300 : 1000; // 玩家射击冷却300ms，敌方1000ms
                this.movePattern = isPlayer ? null : this.generateMovePattern();
                this.moveTimer = 0;
                this.moveInterval = 60; // 敌方移动间隔
            }

            generateMovePattern() {
                const patterns = ['random', 'horizontal', 'vertical', 'diagonal'];
                return patterns[Math.floor(Math.random() * patterns.length)];
            }

            update(canvasWidth, canvasHeight) {
                if (this.isPlayer) {
                    // 玩家坦克由键盘控制，在update方法中不处理移动
                } else {
                    // 敌方AI移动
                    this.moveTimer++;
                    if (this.moveTimer >= this.moveInterval) {
                        this.moveTimer = 0;
                        this.aiMove(canvasWidth, canvasHeight);
                    }
                    
                    // 随机射击
                    if (Math.random() < 0.02) {
                        this.shoot();
                    }
                }

                // 更新子弹
                this.bullets = this.bullets.filter(bullet => {
                    bullet.update();
                    return bullet.isActive();
                });
            }

            aiMove(canvasWidth, canvasHeight) {
                const oldDirection = this.direction;
                
                switch (this.movePattern) {
                    case 'random':
                        this.direction = Math.floor(Math.random() * 4);
                        break;
                    case 'horizontal':
                        this.direction = Math.random() < 0.5 ? 1 : 3; // 左右移动
                        break;
                    case 'vertical':
                        this.direction = Math.random() < 0.5 ? 0 : 2; // 上下移动
                        break;
                    case 'diagonal':
                        this.direction = Math.floor(Math.random() * 2) * 2 + Math.floor(Math.random() * 2);
                        break;
                }
                
                // 边界检查
                let newX = this.x;
                let newY = this.y;
                
                switch (this.direction) {
                    case 0: newY -= this.speed; break;
                    case 1: newX += this.speed; break;
                    case 2: newY += this.speed; break;
                    case 3: newX -= this.speed; break;
                }
                
                if (newX >= 0 && newX <= canvasWidth - this.width &&
                    newY >= 0 && newY <= canvasHeight - this.height) {
                    this.x = newX;
                    this.y = newY;
                } else {
                    // 如果碰到边界，改变方向
                    this.direction = Math.floor(Math.random() * 4);
                }
            }

            move(direction, canvasWidth, canvasHeight) {
                if (!this.isPlayer) return;
                
                this.direction = direction;
                let newX = this.x;
                let newY = this.y;
                
                switch (direction) {
                    case 0: newY -= this.speed; break;
                    case 1: newX += this.speed; break;
                    case 2: newY += this.speed; break;
                    case 3: newX -= this.speed; break;
                }
                
                // 边界检查
                if (newX >= 0 && newX <= canvasWidth - this.width &&
                    newY >= 0 && newY <= canvasHeight - this.height) {
                    this.x = newX;
                    this.y = newY;
                }
            }

            shoot() {
                const now = Date.now();
                if (now - this.lastShotTime < this.shotCooldown) return;
                
                this.lastShotTime = now;
                
                let bulletX, bulletY;
                const bulletSize = 5;
                const bulletSpeed = this.isPlayer ? 8 : -5; // 玩家子弹向右，敌方子弹向左
                
                // 根据坦克方向计算子弹初始位置
                switch (this.direction) {
                    case 0: // 上
                        bulletX = this.x + this.width / 2 - bulletSize / 2;
                        bulletY = this.y - bulletSize;
                        break;
                    case 1: // 右
                        bulletX = this.x + this.width;
                        bulletY = this.y + this.height / 2 - bulletSize / 2;
                        break;
                    case 2: // 下
                        bulletX = this.x + this.width / 2 - bulletSize / 2;
                        bulletY = this.y + this.height;
                        break;
                    case 3: // 左
                        bulletX = this.x - bulletSize;
                        bulletY = this.y + this.height / 2 - bulletSize / 2;
                        break;
                }
                
                this.bullets.push(new Bullet(bulletX, bulletY, bulletSize, bulletSpeed, this.isPlayer));
            }

            draw(ctx) {
                // 绘制坦克主体
                ctx.fillStyle = this.color;
                ctx.fillRect(this.x, this.y, this.width, this.height);
                
                // 绘制坦克炮管
                ctx.fillStyle = this.color;
                const barrelLength = 20;
                const barrelWidth = 8;
                
                switch (this.direction) {
                    case 0: // 上
                        ctx.fillRect(this.x + this.width / 2 - barrelWidth / 2, this.y - barrelLength, barrelWidth, barrelLength);
                        break;
                    case 1: // 右
                        ctx.fillRect(this.x + this.width, this.y + this.height / 2 - barrelWidth / 2, barrelLength, barrelWidth);
                        break;
                    case 2: // 下
                        ctx.fillRect(this.x + this.width / 2 - barrelWidth / 2, this.y + this.height, barrelWidth, barrelLength);
                        break;
                    case 3: // 左
                        ctx.fillRect(this.x - barrelLength, this.y + this.height / 2 - barrelWidth / 2, barrelLength, barrelWidth);
                        break;
                }
                
                // 绘制生命值条
                const healthBarWidth = this.width;
                const healthBarHeight = 5;
                const healthPercentage = this.health / this.maxHealth;
                
                // 生命值条背景
                ctx.fillStyle = 'rgba(255, 0, 0, 0.5)';
                ctx.fillRect(this.x, this.y - 10, healthBarWidth, healthBarHeight);
                
                // 生命值条前景
                ctx.fillStyle = healthPercentage > 0.5 ? 'green' : healthPercentage > 0.25 ? 'yellow' : 'red';
                ctx.fillRect(this.x, this.y - 10, healthBarWidth * healthPercentage, healthBarHeight);
                
                // 绘制子弹
                this.bullets.forEach(bullet => bullet.draw(ctx));
            }

            takeDamage(amount) {
                this.health -= amount;
                if (this.health < 0) this.health = 0;
            }

            isDestroyed() {
                return this.health <= 0;
            }

            getBounds() {
                return {
                    x: this.x,
                    y: this.y,
                    width: this.width,
                    height: this.height
                };
            }
        }

        // 子弹类
        class Bullet {
            constructor(x, y, size, speed, isPlayerBullet) {
                this.x = x;
                this.y = y;
                this.size = size;
                this.speed = speed;
                this.isPlayerBullet = isPlayerBullet;
            }

            update() {
                this.x += this.speed;
            }

            draw(ctx) {
                ctx.fillStyle = this.isPlayerBullet ? '#f1c40f' : '#e74c3c';
                ctx.fillRect(this.x, this.y, this.size, this.size);
            }

            isActive() {
                return this.x > -10 && this.x < 810; // 在画布范围内
            }

            getBounds() {
                return {
                    x: this.x,
                    y: this.y,
                    width: this.size,
                    height: this.size
                };
            }
        }

        // 游戏类
        class Game {
            constructor() {
                this.canvas = document.getElementById('gameCanvas');
                this.ctx = this.canvas.getContext('2d');
                this.gameState = GameState.START;
                
                // 初始化玩家坦克
                this.playerTank = new Tank(100, 300, 40, 40, '#3498db', true);
                
                // 初始化敌方坦克
                this.enemyTank = new Tank(600, 300, 40, 40, '#e74c3c', false);
                
                // 总部
                this.headquarters = {
                    player: { x: 50, y: 250, width: 60, height: 100, health: 100 },
                    enemy: { x: 690, y: 250, width: 60, height: 100, health: 100 }
                };
                
                // 键盘状态
                this.keys = {};
                
                // 绑定事件
                this.bindEvents();
                
                // 开始游戏循环
                this.gameLoop();
            }

            bindEvents() {
                // 键盘事件
                window.addEventListener('keydown', (e) => {
                    this.keys[e.key] = true;
                    
                    // 空格键射击
                    if (e.key === ' ' && this.gameState === GameState.PLAYING) {
                        e.preventDefault();
                        this.playerTank.shoot();
                    }
                });
                
                window.addEventListener('keyup', (e) => {
                    this.keys[e.key] = false;
                });
                
                // 重新开始按钮
                document.getElementById('restartBtn').addEventListener('click', () => {
                    this.restart();
                });
            }

            update() {
                if (this.gameState !== GameState.PLAYING) return;
                
                // 处理玩家输入
                if (this.keys['ArrowUp'] || this.keys['w'] || this.keys['W']) {
                    this.playerTank.move(0, this.canvas.width, this.canvas.height);
                }
                if (this.keys['ArrowRight'] || this.keys['d'] || this.keys['D']) {
                    this.playerTank.move(1, this.canvas.width, this.canvas.height);
                }
                if (this.keys['ArrowDown'] || this.keys['s'] || this.keys['S']) {
                    this.playerTank.move(2, this.canvas.width, this.canvas.height);
                }
                if (this.keys['ArrowLeft'] || this.keys['a'] || this.keys['A']) {
                    this.playerTank.move(3, this.canvas.width, this.canvas.height);
                }
                
                // 更新坦克
                this.playerTank.update(this.canvas.width, this.canvas.height);
                this.enemyTank.update(this.canvas.width, this.canvas.height);
                
                // 检测子弹碰撞
                this.checkCollisions();
                
                // 检查游戏结束条件
                this.checkGameOver();
                
                // 更新UI
                this.updateUI();
            }

            checkCollisions() {
                // 玩家子弹与敌方坦克和敌方总部碰撞
                this.playerTank.bullets = this.playerTank.bullets.filter(bullet => {
                    // 检查与敌方坦克碰撞
                    if (this.checkCollision(bullet, this.enemyTank)) {
                        this.enemyTank.takeDamage(20);
                        return false;
                    }
                    
                    // 检查与敌方总部碰撞
                    if (this.checkCollision(bullet, this.headquarters.enemy)) {
                        this.headquarters.enemy.health -= 10;
                        return false;
                    }
                    
                    return true;
                });
                
                // 敌方子弹与玩家坦克和玩家总部碰撞
                this.enemyTank.bullets = this.enemyTank.bullets.filter(bullet => {
                    // 检查与玩家坦克碰撞
                    if (this.checkCollision(bullet, this.playerTank)) {
                        this.playerTank.takeDamage(20);
                        return false;
                    }
                    
                    // 检查与玩家总部碰撞
                    if (this.checkCollision(bullet, this.headquarters.player)) {
                        this.headquarters.player.health -= 10;
                        return false;
                    }
                    
                    return true;
                });
            }

            checkCollision(obj1, obj2) {
                const bounds1 = obj1.getBounds();
                const bounds2 = obj2.getBounds();
                
                return bounds1.x < bounds2.x + bounds2.width &&
                       bounds1.x + bounds1.width > bounds2.x &&
                       bounds1.y < bounds2.y + bounds2.height &&
                       bounds1.y + bounds1.height > bounds2.y;
            }

            checkGameOver() {
                if (this.playerTank.isDestroyed() || this.headquarters.player.health <= 0) {
                    this.gameState = GameState.GAME_OVER;
                    document.getElementById('gameStateText').textContent = '游戏结束 - 你输了！';
                    document.getElementById('gameState').style.display = 'block';
                } else if (this.enemyTank.isDestroyed() || this.headquarters.enemy.health <= 0) {
                    this.gameState = GameState.GAME_OVER;
                    document.getElementById('gameStateText').textContent = '恭喜你赢了！';
                    document.getElementById('gameState').style.display = 'block';
                }
            }

            updateUI() {
                document.getElementById('playerHealth').textContent = this.playerTank.health;
                document.getElementById('enemyHealth').textContent = this.enemyTank.health;
            }

            draw() {
                // 清空画布
                this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);
                
                // 绘制背景网格
                this.drawGrid();
                
                // 绘制总部
                this.drawHeadquarters();
                
                // 绘制坦克
                this.playerTank.draw(this.ctx);
                this.enemyTank.draw(this.ctx);
                
                // 绘制游戏状态
                if (this.gameState === GameState.START) {
                    this.ctx.fillStyle = 'rgba(0, 0, 0, 0.7)';
                    this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);
                    
                    this.ctx.fillStyle = 'white';
                    this.ctx.font = '36px Arial';
                    this.ctx.textAlign = 'center';
                    this.ctx.fillText('坦克大战', this.canvas.width / 2, this.canvas.height / 2 - 40);
                    
                    this.ctx.font = '20px Arial';
                    this.ctx.fillText('按任意键开始游戏', this.canvas.width / 2, this.canvas.height / 2 + 20);
                }
            }

            drawGrid() {
                this.ctx.strokeStyle = 'rgba(255, 255, 255, 0.1)';
                this.ctx.lineWidth = 1;
                
                // 绘制水平线
                for (let y = 0; y < this.canvas.height; y += 20) {
                    this.ctx.beginPath();
                    this.ctx.moveTo(0, y);
                    this.ctx.lineTo(this.canvas.width, y);
                    this.ctx.stroke();
                }
                
                // 绘制垂直线
                for (let x = 0; x < this.canvas.width; x += 20) {
                    this.ctx.beginPath();
                    this.ctx.moveTo(x, 0);
                    this.ctx.lineTo(x, this.canvas.height);
                    this
