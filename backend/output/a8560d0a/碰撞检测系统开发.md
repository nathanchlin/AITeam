    <script>
        // 游戏配置
        const GAME_CONFIG = {
            canvasWidth: 800,
            canvasHeight: 600,
            tankSize: 30,
            bulletSpeed: 5,
            bulletSize: 4,
            tankSpeed: 3,
            enemyTankSpeed: 2,
            headquartersSize: 50,
            obstacleSize: 40,
            maxEnemies: 5,
            enemySpawnInterval: 3000,
            bulletCooldown: 300,
            enemyShootCooldown: 1500
        };

        // 游戏状态
        const gameState = {
            playerLives: 3,
            enemies: [],
            bullets: [],
            obstacles: [],
            playerTank: null,
            headquarters: null,
            keys: {},
            lastTime: 0,
            gameOver: false,
            lastEnemySpawn: 0,
            lastPlayerShoot: 0,
            lastEnemyShoot: 0
        };

        // 获取Canvas上下文
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');

        // 碰撞检测工具函数
        function checkCollision(rect1, rect2) {
            return rect1.x < rect2.x + rect2.width &&
                   rect1.x + rect1.width > rect2.x &&
                   rect1.y < rect2.y + rect2.height &&
                   rect1.y + rect1.height > rect2.y;
        }

        // 坦克类
        class Tank {
            constructor(x, y, color, isPlayer = false) {
                this.x = x;
                this.y = y;
                this.width = GAME_CONFIG.tankSize;
                this.height = GAME_CONFIG.tankSize;
                this.color = color;
                this.direction = 'up'; // up, down, left, right
                this.isPlayer = isPlayer;
                this.speed = isPlayer ? GAME_CONFIG.tankSpeed : GAME_CONFIG.enemyTankSpeed;
                this.lastShoot = 0;
                this.health = 1;
            }

            update() {
                if (this.isPlayer) {
                    // 玩家坦克控制
                    if (gameState.keys['w'] || gameState.keys['W']) {
                        this.direction = 'up';
                        this.y = Math.max(0, this.y - this.speed);
                    }
                    if (gameState.keys['s'] || gameState.keys['S']) {
                        this.direction = 'down';
                        this.y = Math.min(canvas.height - this.height, this.y + this.speed);
                    }
                    if (gameState.keys['a'] || gameState.keys['A']) {
                        this.direction = 'left';
                        this.x = Math.max(0, this.x - this.speed);
                    }
                    if (gameState.keys['d'] || gameState.keys['D']) {
                        this.direction = 'right';
                        this.x = Math.min(canvas.width - this.width, this.x + this.speed);
                    }
                    
                    // 射击
                    if (gameState.keys[' '] && Date.now() - this.lastShoot > GAME_CONFIG.bulletCooldown) {
                        this.shoot();
                        this.lastShoot = Date.now();
                    }
                } else {
                    // 敌人坦克AI
                    this.aiUpdate();
                }
                
                // 检查与障碍物的碰撞
                for (const obstacle of gameState.obstacles) {
                    if (checkCollision(this, obstacle)) {
                        // 简单的碰撞响应 - 将坦克推回
                        if (this.direction === 'up') this.y += this.speed;
                        if (this.direction === 'down') this.y -= this.speed;
                        if (this.direction === 'left') this.x += this.speed;
                        if (this.direction === 'right') this.x -= this.speed;
                    }
                }
                
                // 检查与总部的碰撞
                if (gameState.headquarters && checkCollision(this, gameState.headquarters)) {
                    if (this.isPlayer) {
                        // 玩家坦克不能进入总部
                        if (this.direction === 'up') this.y += this.speed;
                        if (this.direction === 'down') this.y -= this.speed;
                        if (this.direction === 'left') this.x += this.speed;
                        if (this.direction === 'right') this.x -= this.speed;
                    } else {
                        // 敌人坦克进入总部导致游戏结束
                        endGame();
                    }
                }
            }

            aiUpdate() {
                // 简单的AI行为
                if (Math.random() < 0.02) {
                    // 随机改变方向
                    const directions = ['up', 'down', 'left', 'right'];
                    this.direction = directions[Math.floor(Math.random() * directions.length)];
                }
                
                // 移动
                if (this.direction === 'up') this.y = Math.max(0, this.y - this.speed);
                if (this.direction === 'down') this.y = Math.min(canvas.height - this.height, this.y + this.speed);
                if (this.direction === 'left') this.x = Math.max(0, this.x - this.speed);
                if (this.direction === 'right') this.x = Math.min(canvas.width - this.width, this.x + this.speed);
                
                // 射击
                if (Math.random() < 0.01 && Date.now() - this.lastShoot > GAME_CONFIG.enemyShootCooldown) {
                    this.shoot();
                    this.lastShoot = Date.now();
                }
            }

            shoot() {
                let bulletX = this.x + this.width / 2;
                let bulletY = this.y + this.height / 2;
                
                // 根据坦克方向调整子弹初始位置
                if (this.direction === 'up') bulletY -= this.height / 2;
                if (this.direction === 'down') bulletY += this.height / 2;
                if (this.direction === 'left') bulletX -= this.width / 2;
                if (this.direction === 'right') bulletX += this.width / 2;
                
                const bullet = new Bullet(
                    bulletX,
                    bulletY,
                    this.direction,
                    this.isPlayer
                );
                gameState.bullets.push(bullet);
            }

            draw() {
                ctx.fillStyle = this.color;
                
                // 绘制坦克主体
                ctx.fillRect(this.x, this.y, this.width, this.height);
                
                // 绘制坦克炮管
                ctx.fillStyle = this.isPlayer ? '#0ff' : '#f0f';
                
                const centerX = this.x + this.width / 2;
                const centerY = this.y + this.height / 2;
                const barrelLength = this.width * 0.8;
                
                ctx.beginPath();
                if (this.direction === 'up') {
                    ctx.moveTo(centerX, this.y);
                    ctx.lineTo(centerX, this.y - barrelLength);
                    ctx.lineWidth = 4;
                    ctx.stroke();
                } else if (this.direction === 'down') {
                    ctx.moveTo(centerX, this.y + this.height);
                    ctx.lineTo(centerX, this.y + this.height + barrelLength);
                    ctx.lineWidth = 4;
                    ctx.stroke();
                } else if (this.direction === 'left') {
                    ctx.moveTo(this.x, centerY);
                    ctx.lineTo(this.x - barrelLength, centerY);
                    ctx.lineWidth = 4;
                    ctx.stroke();
                } else if (this.direction === 'right') {
                    ctx.moveTo(this.x + this.width, centerY);
                    ctx.lineTo(this.x + this.width + barrelLength, centerY);
                    ctx.lineWidth = 4;
                    ctx.stroke();
                }
                
                // 绘制坦克细节
                ctx.fillStyle = this.isPlayer ? '#0ff' : '#f0f';
                ctx.beginPath();
                ctx.arc(centerX, centerY, this.width / 4, 0, Math.PI * 2);
                ctx.fill();
            }
        }

        // 子弹类
        class Bullet {
            constructor(x, y, direction, isPlayerBullet) {
                this.x = x;
                this.y = y;
                this.width = GAME_CONFIG.bulletSize;
                this.height = GAME_CONFIG.bulletSize;
                this.direction = direction;
                this.speed = GAME_CONFIG.bulletSpeed;
                this.isPlayerBullet = isPlayerBullet;
                this.active = true;
            }

            update() {
                if (this.direction === 'up') this.y -= this.speed;
                if (this.direction === 'down') this.y += this.speed;
                if (this.direction === 'left') this.x -= this.speed;
                if (this.direction === 'right') this.x += this.speed;
                
                // 检查是否超出边界
                if (this.x < 0 || this.x > canvas.width || 
                    this.y < 0 || this.y > canvas.height) {
                    this.active = false;
                }
                
                // 碰撞检测
                this.checkCollisions();
            }

            checkCollisions() {
                // 与障碍物的碰撞
                for (const obstacle of gameState.obstacles) {
                    if (checkCollision(this, obstacle)) {
                        this.active = false;
                        break;
                    }
                }
                
                // 与坦克的碰撞
                if (this.isPlayerBullet) {
                    // 玩家子弹与敌人坦克
                    for (let i = 0; i < gameState.enemies.length; i++) {
                        if (checkCollision(this, gameState.enemies[i])) {
                            this.active = false;
                            gameState.enemies.splice(i, 1);
                            updateUI();
                            break;
                        }
                    }
                } else {
                    // 敌人子弹与玩家坦克
                    if (gameState.playerTank && checkCollision(this, gameState.playerTank)) {
                        this.active = false;
                        gameState.playerLives--;
                        updateUI();
                        
                        if (gameState.playerLives <= 0) {
                            endGame();
                        }
                    }
                }
                
                // 与总部的碰撞
                if (gameState.headquarters && checkCollision(this, gameState.headquarters)) {
                    this.active = false;
                    endGame();
                }
            }

            draw() {
                ctx.fillStyle = this.isPlayerBullet ? '#0ff' : '#f0f';
                ctx.fillRect(this.x - this.width / 2, this.y - this.height / 2, this.width, this.height);
            }
        }

        // 总部类
        class Headquarters {
            constructor(x, y) {
                this.x = x;
                this.y = y;
                this.width = GAME_CONFIG.headquartersSize;
                this.height = GAME_CONFIG.headquartersSize;
            }

            draw() {
                // 绘制总部建筑
                ctx.fillStyle = '#090';
                ctx.fillRect(this.x, this.y, this.width, this.height);
                
                // 绘制总部旗帜
                ctx.fillStyle = '#f00';
                ctx.beginPath();
                ctx.moveTo(this.x + this.width / 2, this.y);
                ctx.lineTo(this.x + this.width / 2, this.y - 20);
                ctx.lineTo(this.x + this.width / 2 + 15, this.y - 10);
                ctx.closePath();
                ctx.fill();
                
                // 绘制总部细节
                ctx.strokeStyle = '#0f0';
                ctx.lineWidth = 2;
                ctx.strokeRect(this.x + 5, this.y + 5, this.width - 10, this.height - 10);
            }
        }

        // 障碍物类
        class Obstacle {
            constructor(x, y) {
                this.x = x;
                this.y = y;
                this.width = GAME_CONFIG.obstacleSize;
                this.height = GAME_CONFIG.obstacleSize;
            }

            draw() {
                // 绘制障碍物
                ctx.fillStyle = '#666';
                ctx.fillRect(this.x, this.y, this.width, this.height);
                
                // 绘制障碍物纹理
                ctx.strokeStyle = '#999';
                ctx.lineWidth = 1;
                ctx.strokeRect(this.x + 5, this.y + 5, this.width - 10, this.height - 10);
                
                // 添加一些线条纹理
                ctx.beginPath();
                ctx.moveTo(this.x + 5, this.y + 5);
                ctx.lineTo(this.x + this.width - 5, this.y + this.height - 5);
                ctx.moveTo(this.x + this.width - 5, this.y + 5);
                ctx.lineTo(this.x + 5, this.y + this.height - 5);
                ctx.stroke();
            }
        }

        // 初始化游戏
        function initGame() {
            // 重置游戏状态
            gameState.playerLives = 3;
            gameState.enemies = [];
            gameState.bullets = [];
            gameState.obstacles = [];
            gameState.gameOver = false;
            gameState.lastEnemySpawn = 0;
            gameState.lastPlayerShoot = 0;
            gameState.lastEnemyShoot = 0;
            
            // 创建玩家坦克
            gameState.playerTank = new Tank(
                canvas.width / 2 - GAME_CONFIG.tankSize / 2,
                canvas.height - GAME_CONFIG.tankSize - 10,
                '#0f0',
                true
            );
            
            // 创建总部
            gameState.headquarters = new Headquarters(
                canvas.width / 2 - GAME_CONFIG.headquartersSize / 2,
                10
            );
            
            // 创建障碍物
            const obstaclePositions = [
                {x: 100, y: 150},
                {x: 300, y: 200},
                {x: 500, y: 150},
                {x: 200, y: 350},
                {x: 400, y: 400},
                {x: 600, y: 350},
                {x: 150, y: 450},
                {x: 650, y: 450}
            ];
            
            for (const pos of obstaclePositions) {
                gameState.obstacles.push(new Obstacle(pos.x, pos.y));
            }
            
            // 生成初始敌人
            for (let i = 0; i < 3; i++) {
                spawnEnemy();
            }
            
            updateUI();
            document.getElementById('gameOver').style.display = 'none';
        }

        // 生成敌人
        function spawnEnemy() {
            const x = Math.random() * (canvas.width - GAME_CONFIG.tankSize);
            const y = canvas.height - GAME_CONFIG.tankSize - 10;
            
            gameState.enemies.push(new Tank(x, y, '#f00', false));
            updateUI();
        }

        // 更新UI
        function updateUI() {
            document.getElementById('playerLives').textContent = gameState.playerLives;
            document.getElementById('enemyCount').textContent = gameState.enemies.length;
        }

        // 游戏结束
        function endGame() {
            gameState.gameOver = true;
            document.getElementById('gameOver').style.display = 'block';
        }

        // 游戏主循环
        function gameLoop(timestamp) {
            if (!gameState.gameOver) {
                // 清除画布
                ctx.clearRect(0, 0, canvas.width, canvas.height);
                
                // 更新游戏状态
                if (gameState.playerTank) {
                    gameState.playerTank.update();
                }
                
                // 更新敌人
                for (const enemy of gameState.enemies) {
                    enemy.update();
                }
                
                // 更新子弹
                for (let i = gameState.bullets.length - 1; i >= 0; i--) {
                    const bullet = gameState.bullets[i];
                    bullet.update();
                    
                    if (!bullet.active) {
                        gameState.bullets.splice(i, 1);
                    }
                }
                
                // 生成新敌人
                if (timestamp - gameState.lastEnemySpawn > GAME_CONFIG.enemySpawnInterval && 
                    gameState.enemies.length < GAME_CONFIG.maxEnemies) {
                    spawnEnemy();
                    gameState.lastEnemySpawn = timestamp;
                }
                
                // 绘制游戏元素
                if (gameState.headquarters) {
                    gameState.headquarters.draw();
                }
                
                for (const obstacle of gameState.obstacles) {
                    obstacle.draw();
                }
                
                if (gameState.playerTank) {
                    gameState.playerTank.draw();
                }
                
                for (const enemy of gameState.enemies) {
                    enemy.draw();
                }
                
                for (const bullet of gameState.bullets) {
                    bullet.draw();
                }
            }
            
            requestAnimationFrame(gameLoop);
        }

        window.addEventListener('keydown', (e) => {
            gameState.keys[e.key] = true;
            
            // 重新开始游戏
            if (e.key === 'r' || e.key === 'R') {
                initGame();
            }
        });
