        const GAME_CONFIG = {

        // 玩家类
        class Player {
            constructor(id, name, x, y, color) {
                this.id = id;
                this.name = name;
                this.x = x;
                this.y = y;
                this.color = color;
                this.radius = GAME_CONFIG.MIN_PLAYER_RADIUS;
                this.score = 0;
                this.targetX = x;
                this.targetY = y;
                this.speed = GAME_CONFIG.PLAYER_SPEED;
                this.isAlive = true;
                this.ejectedMass = [];
                this.cells = [{
                    x: x,
                    y: y,
                    radius: this.radius
                }];
            }

            update() {
                if (!this.isAlive) return;

                // 移动逻辑
                const dx = this.targetX - this.x;
                const dy = this.targetY - this.y;
                const distance = Math.sqrt(dx * dx + dy * dy);

                if (distance > 1) {
                    const moveX = (dx / distance) * this.speed * (1 - this.radius / 100);
                    const moveY = (dy / distance) * this.speed * (1 - this.radius / 100);
                    this.x += moveX;
                    this.y += moveY;
                }

                // 更新所有细胞位置
                this.cells[0].x = this.x;
                this.cells[0].y = this.y;
                this.cells[0].radius = this.radius;

                // 更新喷射出的质量
                this.ejectedMass = this.ejectedMass.filter(mass => {
                    mass.x += mass.vx;
                    mass.y += mass.vy;
                    mass.radius -= 0.1;
                    return mass.radius > 0;
                });
            }

            grow(amount) {
                this.radius = Math.min(this.radius + amount, GAME_CONFIG.MAX_PLAYER_RADIUS);
                this.score += Math.floor(amount);
                this.cells[0].radius = this.radius;
            }

            split() {
                if (this.cells.length >= 8 || this.radius < GAME_CONFIG.MIN_PLAYER_RADIUS * 2) return;
                
                const newCell = {
                    x: this.x + this.radius,
                    y: this.y,
                    radius: this.radius / 2
                };
                
                this.cells.push(newCell);
                this.radius = this.radius / 2;
                this.score = Math.floor(this.score / 2);
            }

            ejectMass() {
                if (this.radius < GAME_CONFIG.MIN_PLAYER_RADIUS * 1.5) return;
                
                const angle = Math.atan2(this.targetY - this.y, this.targetX - this.x);
                const mass = {
                    x: this.x + Math.cos(angle) * (this.radius + 10),
                    y: this.y + Math.sin(angle) * (this.radius + 10),
                    radius: GAME_CONFIG.EJECT_SIZE,
                    vx: Math.cos(angle) * GAME_CONFIG.EJECT_SPEED,
                    vy: Math.sin(angle) * GAME_CONFIG.EJECT_SPEED
                };
                
                this.ejectedMass.push(mass);
                this.radius -= GAME_CONFIG.EJECT_SIZE;
                this.score = Math.floor(this.score * 0.9);
            }

            draw(ctx) {
                if (!this.isAlive) return;

                // 绘制细胞
                this.cells.forEach(cell => {
                    ctx.beginPath();
                    ctx.arc(cell.x, cell.y, cell.radius, 0, Math.PI * 2);
                    ctx.fillStyle = this.color;
                    ctx.fill();
                    ctx.strokeStyle = '#fff';
                    ctx.lineWidth = 2;
                    ctx.stroke();
                });

                // 绘制喷射出的质量
                this.ejectedMass.forEach(mass => {
                    ctx.beginPath();
                    ctx.arc(mass.x, mass.y, mass.radius, 0, Math.PI * 2);
                    ctx.fillStyle = this.color;
                    ctx.fill();
                });

                // 绘制玩家名称
                ctx.fillStyle = '#fff';
                ctx.font = '14px Arial';
                ctx.textAlign = 'center';
                ctx.fillText(this.name, this.x, this.y - this.radius - 10);
            }
        }

            init() {

            addBots(count) {
                for (let i = 0; i < count; i++) {
                    const botId = `bot_${Date.now()}_${i}`;
                    const bot = new Player(
                        botId,
                        `Bot_${i + 1}`,
                        Math.random() * this.canvas.width,
                        Math.random() * this.canvas.height,
                        this.getRandomColor()
                    );
                    
                    // 设置AI行为
                    bot.isAI = true;
                    bot.updateTarget = () => {
                        if (!bot.isAlive) return;
                        
                        // 随机移动
                        if (Math.random() < 0.02) {
                            bot.targetX = Math.random() * this.canvas.width;
                            bot.targetY = Math.random() * this.canvas.height;
                        }
                        
                        // 寻找食物
                        const nearestFood = this.findNearestFood(bot);
                        if (nearestFood) {
                            bot.targetX = nearestFood.x;
                            bot.targetY = nearestFood.y;
                        }
                        
                        // 逃离大玩家
                        const nearestThreat = this.findNearestLargerPlayer(bot);
                        if (nearestThreat) {
                            const dx = bot.x - nearestThreat.x;
                            const dy = bot.y - nearestThreat.y;
                            const distance = Math.sqrt(dx * dx + dy * dy);
                            
                            if (distance < 200 && bot.radius < nearestThreat.radius * 0.8) {
                                bot.targetX = bot.x + (dx / distance) * 100;
                                bot.targetY = bot.y + (dy / distance) * 100;
                            }
                        }
                    };
                    
                    this.players.set(botId, bot);
                }
                
                this.updateOnlineCount();
            }

            findNearestFood(player) {
                let nearest = null;
                let minDistance = Infinity;
                
                this.foods.forEach(food => {
                    const dx = food.x - player.x;
                    const dy = food.y - player.y;
                    const distance = Math.sqrt(dx * dx + dy * dy);
                    
                    if (distance < minDistance) {
                        minDistance = distance;
                        nearest = food;
                    }
                });
                
                return nearest;
            }

            findNearestLargerPlayer(player) {
                let nearest = null;
                let minDistance = Infinity;
                
                this.players.forEach(other => {
                    if (other.id === player.id || !other.isAlive) return;
                    
                    if (other.radius > player.radius) {
                        const dx = other.x - player.x;
                        const dy = other.y - player.y;
                        const distance = Math.sqrt(dx * dx + dy * dy);
                        
                        if (distance < minDistance) {
                            minDistance = distance;
                            nearest = other;
                        }
                    }
                });
                
                return nearest;
            }
