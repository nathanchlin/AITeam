        // 游戏配置
        const GAME_CONFIG = {
            MAP_WIDTH: 3000,
            MAP_HEIGHT: 3000,
            CANVAS_WIDTH: window.innerWidth,
            CANVAS_HEIGHT: window.innerHeight,
            PLAYER_SIZE_MIN: 10,

            FOOD_SIZE: 5,
            FOOD_COUNT: 500,

            EJECT_SPEED: 20,
            UPDATE_INTERVAL: 50, // 网络更新间隔(ms)
            PING_INTERVAL: 1000, // 心跳间隔(ms)
            LERP_FACTOR: 0.1, // 位置插值因子

        // 游戏状态
        class GameState {
            constructor() {
                this.players = new Map();
                this.foods = [];
                this.ejectedBalls = [];
                this.playerId = null;
                this.camera = { x: 0, y: 0 };
                this.lastUpdateTime = 0;

                this.players.delete(playerId);
            }

            updatePlayer(playerId, updates) {
                const player = this.players.get(playerId);
                if (player) {
                    Object.assign(player, updates);
                }
            }

            generateFood() {
                const food = {
                    id: Math.random().toString(36).substr(2, 9),
                    x: Math.random() * GAME_CONFIG.MAP_WIDTH,
                    y: Math.random() * GAME_CONFIG.MAP_HEIGHT,
                    color: this.getRandomColor(),
                    size: GAME_CONFIG.FOOD_SIZE
                };
                this.foods.push(food);
                return food;
            }

            generateInitialFood() {
                this.foods = [];
                for (let i = 0; i < GAME_CONFIG.FOOD_COUNT; i++) {
                    this.generateFood();
                }
            }

            getRandomColor() {
                const colors = [
                    '#FF5252', '#FF4081', '#E040FB', '#7C4DFF', '#536DFE',
                    '#448AFF', '#40C4FF', '#18FFFF', '#64FFDA', '#69F0AE',
                    '#B2FF59', '#EEFF41', '#FFFF00', '#FFD740', '#FFAB40'
                ];
                return colors[Math.floor(Math.random() * colors.length)];
            }

            removeFood(foodId) {
                this.foods = this.foods.filter(food => food.id !== foodId);
            }

            addEjectedBall(ball) {
                this.ejectedBalls.push(ball);
            }

            updateEjectedBalls() {
                this.ejectedBalls = this.ejectedBalls.filter(ball => {
                    ball.x += ball.vx;
                    ball.y += ball.vy;
                    ball.size *= 0.98; // 逐渐缩小
                    
                    // 检查是否超出边界
                    if (ball.x < 0 || ball.x > GAME_CONFIG.MAP_WIDTH || 
                        ball.y < 0 || ball.y > GAME_CONFIG.MAP_HEIGHT || 
                        ball.size < 1) {
                        return false;
                    }
                    
                    // 检查是否被其他玩家吃掉
                    for (const [playerId, player] of this.players) {
                        const dx = ball.x - player.x;
                        const dy = ball.y - player.y;
                        const distance = Math.sqrt(dx * dx + dy * dy);
                        
                        if (distance < player.size + ball.size) {
                            player.size += ball.size * 0.5;
                            return false;
                        }
                    }
                    
                    return true;
                });
            }

            updateCamera() {

                if (player) {
                    this.camera.x = player.x - GAME_CONFIG.CANVAS_WIDTH / 2;
                    this.camera.y = player.y - GAME_CONFIG.CANVAS_HEIGHT / 2;

                    this.camera.x = Math.max(0, Math.min(GAME_CONFIG.MAP_WIDTH - GAME_CONFIG.CANVAS_WIDTH, this.camera.x));
                    this.camera.y = Math.max(0, Math.min(GAME_CONFIG.MAP_HEIGHT - GAME_CONFIG.CANVAS_HEIGHT, this.camera.y));

        // 玩家类
        class Player {
            constructor(id, x, y, name, isLocal = false) {
                this.id = id;
                this.x = x;
                this.y = y;
                this.name = name;
                this.size = GAME_CONFIG.PLAYER_SIZE_MIN;
                this.color = this.getRandomColor();
                this.isLocal = isLocal;
                this.targetX = x;
                this.targetY = y;
                this.vx = 0;
                this.vy = 0;
                this.speed = GAME_CONFIG.PLAYER_SPEED_MAX;
                this.cells = [{ x, y, size: this.size }]; // 多细胞支持
            }

            getRandomColor() {
                const colors = [
                    '#FF5252', '#FF4081', '#E040FB', '#7C4DFF', '#536DFE',
                    '#448AFF', '#40C4FF', '#18FFFF', '#64FFDA', '#69F0AE',
                    '#B2FF59', '#EEFF41', '#FFFF00', '#FFD740', '#FFAB40'
                ];
                return colors[Math.floor(Math.random() * colors.length)];
            }

            update(targetX, targetY) {
                if (!this.isLocal) {
                    // 对于远程玩家，使用插值平滑移动
                    this.x += (this.targetX - this.x) * GAME_CONFIG.LERP_FACTOR;
                    this.y += (this.targetY - this.y) * GAME_CONFIG.LERP_FACTOR;
                    return;
                }

                // 本地玩家移动逻辑
                const dx = targetX - this.x;
                const dy = targetY - this.y;
                const distance = Math.sqrt(dx * dx + dy * dy);
                
                if (distance > 1) {
                    // 根据大小计算速度
                    this.speed = Math.max(
                        GAME_CONFIG.PLAYER_SPEED_MIN, 
                        GAME_CONFIG.PLAYER_SPEED_MAX * (GAME_CONFIG.PLAYER_SIZE_MIN / this.size)
                    );
                    
                    // 计算方向
                    const dirX = dx / distance;
                    const dirY = dy / distance;
                    
                    // 应用速度平滑
                    this.vx = this.vx * GAME_CONFIG.SMOOTHING_FACTOR + dirX * this.speed * (1 - GAME_CONFIG.SMOOTHING_FACTOR);
                    this.vy = this.vy * GAME_CONFIG.SMOOTHING_FACTOR + dirY * this.speed * (1 - GAME_CONFIG.SMOOTHING_FACTOR);
                    
                    // 更新位置
                    this.x += this.vx;
                    this.y += this.vy;
                    
                    // 限制在地图内
                    this.x = Math.max(this.size, Math.min(GAME_CONFIG.MAP_WIDTH - this.size, this.x));
                    this.y = Math.max(this.size, Math.min(GAME_CONFIG.MAP_HEIGHT - this.size, this.y));
                    
                    // 更新目标位置（用于网络同步）
                    this.targetX = this.x;
                    this.targetY = this.y;
                }
            }

            eat(food) {
                // 增加大小
                this.size += food.size * 0.1;
                this.size = Math.min(this.size, GAME_CONFIG.PLAYER_SIZE_MAX);
                
                // 更新细胞大小
                this.cells[0].size = this.size;
            }

            split() {
                if (this.size > GAME_CONFIG.PLAYER_SIZE_MIN * 2) {
                    // 创建新细胞
                    const newCell = {
                        x: this.x + this.size,
                        y: this.y,
                        size: this.size / 2
                    };
                    
                    // 减少当前细胞大小
                    this.size /= 2;
                    this.cells[0].size = this.size;
                    
                    // 添加新细胞
                    this.cells.push(newCell);
                    
                    // 返回新创建的细胞
                    return newCell;
                }
                return null;
            }

            eject() {
                if (this.size > GAME_CONFIG.PLAYER_SIZE_MIN * 1.5) {
                    // 创建喷射球
                    const angle = Math.atan2(this.vy, this.vx);
                    const ejectedBall = {
                        x: this.x + Math.cos(angle) * this.size,
                        y: this.y + Math.sin(angle) * this.size,
                        vx: Math.cos(angle) * GAME_CONFIG.EJECT_SPEED,
                        vy: Math.sin(angle) * GAME_CONFIG.EJECT_SPEED,
                        size: GAME_CONFIG.EJECT_SIZE,
                        color: this.color
                    };
                    
                    // 减少玩家大小
                    this.size *= 0.95;
                    this.cells[0].size = this.size;
                    
                    return ejectedBall;
                }
                return null;
            }

            checkCollisions(gameState) {
                // 检查食物碰撞
                for (let i = 0; i < gameState.foods.length; i++) {
                    const food = gameState.foods[i];
                    const dx = this.x - food.x;
                    const dy = this.y - food.y;
                    const distance = Math.sqrt(dx * dx + dy * dy);
                    
                    if (distance < this.size + food.size) {
                        this.eat(food);
                        gameState.removeFood(food.id);
                        gameState.generateFood(); // 生成新食物
                    }
                }
                
                // 检查与其他玩家的碰撞
                for (const [playerId, player] of gameState.players) {
                    if (playerId === this.id) continue;
                    
                    const dx = this.x - player.x;
                    const dy = this.y - player.y;
                    const distance = Math.sqrt(dx * dx + dy * dy);
                    
                    if (distance < this.size - player.size && this.size > player.size) {
                        // 吃掉其他玩家
                        this.size += player.size * 0.8;
                        this.cells[0].size = this.size;
                        gameState.removePlayer(playerId);
                        
                        // 生成新食物
                        for (let i = 0; i < Math.floor(player.size / 10); i++) {
                            gameState.generateFood();
                        }
                    }
                }
            }
        }
