        class Particle {
            constructor(x, y, color) {
                this.x = x;
                this.y = y;
                this.color = color;
                this.size = Math.random() * 3 + 1;
                this.speedX = (Math.random() - 0.5) * 2;
                this.speedY = (Math.random() - 0.5) * 2;
                this.life = 1;
                this.decay = Math.random() * 0.02 + 0.01;
            }

            update() {
                this.x += this.speedX;
                this.y += this.speedY;
                this.life -= this.decay;
                this.size *= 0.98;
            }

            draw() {
                ctx.save();
                ctx.globalAlpha = this.life;
                ctx.fillStyle = this.color;
                ctx.shadowBlur = 10;
                ctx.shadowColor = this.color;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                ctx.fill();
                ctx.restore();
            }
        }

                this.targetX = x;

            }

            update() {
                // 计算移动方向
                const dx = this.targetX - this.x;
                const dy = this.targetY - this.y;
                const distance = Math.sqrt(dx * dx + dy * dy);
                
                if (distance > 1) {
                    // 速度与大小成反比
                    this.speed = config.speedFactor / (this.size / 20);
                    const moveX = (dx / distance) * this.speed;
                    const moveY = (dy / distance) * this.speed;
                    
                    this.x += moveX;
                    this.y += moveY;
                    
                    // 添加轨迹点
                    this.trail.push({ x: this.x, y: this.y, size: this.size });
                    if (this.trail.length > this.trailLength) {
                        this.trail.shift();
                    }
                }
                
                // 边界检查
                this.x = Math.max(this.size, Math.min(worldSize.width - this.size, this.x));
                this.y = Math.max(this.size, Math.min(worldSize.height - this.size, this.y));
                
                // 更新相机位置
                camera.x = this.x - canvas.width / 2;
                camera.y = this.y - canvas.height / 2;
                camera.zoom = Math.max(0.5, Math.min(1.5, 50 / this.size));
            }

            draw() {
                ctx.save();
                ctx.translate(canvas.width / 2, canvas.height / 2);
                ctx.scale(camera.zoom, camera.zoom);
                ctx.translate(-this.x, -this.y);
                
                // 绘制轨迹
                for (let i = 0; i < this.trail.length; i++) {
                    const point = this.trail[i];
                    const alpha = i / this.trail.length * 0.5;
                    ctx.globalAlpha = alpha;
                    ctx.fillStyle = this.color;
                    ctx.beginPath();
                    ctx.arc(point.x, point.y, point.size * 0.8, 0, Math.PI * 2);
                    ctx.fill();
                }
                
                // 绘制光晕
                ctx.globalAlpha = 0.3;
                ctx.fillStyle = this.color;
                ctx.shadowBlur = config.glowIntensity;
                ctx.shadowColor = this.color;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.size * 1.5, 0, Math.PI * 2);
                ctx.fill();
                
                // 绘制主体

                ctx.fillStyle = this.color;
                ctx.shadowBlur = config.glowIntensity / 2;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                ctx.fill();
                
                // 绘制边框
                ctx.strokeStyle = '#fff';

                ctx.textAlign = 'center';
                ctx.textBaseline = 'middle';
                ctx.shadowBlur = 5;
                ctx.shadowColor = '#000';
                ctx.fillText(this.name, this.x, this.y);
                
                ctx.restore();
            }

            eat(target) {
                // 创建粒子效果
                for (let i = 0; i < config.particleCount; i++) {

                }
                
                // 更新大小和分数
                const areaIncrease = Math.PI * target.size * target.size;
                const currentArea = Math.PI * this.size * this.size;
                const newArea = currentArea + areaIncrease * 0.8;
                this.size = Math.sqrt(newArea / Math.PI);
                this.size = Math.min(this.size, config.maxPlayerSize);
                
                this.score += Math.floor(target.size);
                this.eaten.push(target);
            }
        }

        // 敌人类
        class Enemy {
            constructor() {
                this.x = Math.random() * worldSize.width;
                this.y = Math.random() * worldSize.height;
                this.size = config.minPlayerSize + Math.random() * 20;
                this.color = this.getRandomColor();
                this.speed = config.speedFactor / (this.size / 20);
                this.targetX = this.x;
                this.targetY = this.y;
                this.name = this.generateName();
                this.aiUpdateCounter = 0;
            }

            generateName() {
                const names = ['AI机器人', '智能球', '自动球', '电脑球', '机器球', 
                              '电子球', '虚拟球', '数字球', '智能体', '自动体'];
                return names[Math.floor(Math.random() * names.length)] + Math.floor(Math.random() * 100);
            }

            getRandomColor() {
                const hue = Math.floor(Math.random() * 360);
                return `hsl(${hue}, 100%, 50%)`;
            }

            update() {
                // AI行为
                this.aiUpdateCounter++;
                if (this.aiUpdateCounter % 30 === 0) {
                    // 寻找最近的食物或较小的球
                    let nearestTarget = null;
                    let nearestDistance = Infinity;
                    
                    // 检查食物
                    foods.forEach(food => {
                        const distance = this.getDistance(food);
                        if (distance < nearestDistance && distance < 200) {
                            nearestDistance = distance;
                            nearestTarget = food;
                        }
                    });
                    
                    // 检查较小的球
                    [...enemies, player].forEach(ball => {
                        if (ball !== this && ball.size < this.size * 0.8) {
                            const distance = this.getDistance(ball);
                            if (distance < nearestDistance && distance < 300) {
                                nearestDistance = distance;
                                nearestTarget = ball;
                            }
                        }
                    });
                    
                    // 随机移动
                    if (!nearestTarget || Math.random() < 0.3) {
                        this.targetX = Math.random() * worldSize.width;
                        this.targetY = Math.random() * worldSize.height;
                    } else {
                        this.targetX = nearestTarget.x;
                        this.targetY = nearestTarget.y;
                    }
                }
                
                // 移动
                const dx = this.targetX - this.x;
                const dy = this.targetY - this.y;
                const distance = Math.sqrt(dx * dx + dy * dy);
                
                if (distance > 1) {
                    const moveX = (dx / distance) * this.speed;
                    const moveY = (dy / distance) * this.speed;
                    
                    this.x += moveX;
                    this.y += moveY;
                }
                
                // 边界检查
                this.x = Math.max(this.size, Math.min(worldSize.width - this.size, this.x));
                this.y = Math.max(this.size, Math.min(worldSize.height - this.size, this.y));
            }

            getDistance(target) {
                const dx = this.x - target.x;
                const dy = this.y - target.y;
                return Math.sqrt(dx * dx + dy * dy);
            }

                ctx.translate(canvas.width / 2, canvas.height / 2);
                ctx.scale(camera.zoom, camera.zoom);
