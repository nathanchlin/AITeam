        // 游戏引擎核心类
        class GameEngine {
            constructor(canvasId) {
                this.canvas = document.getElementById(canvasId);
                this.ctx = this.canvas.getContext('2d');
                this.width = window.innerWidth;
                this.height = window.innerHeight;
                this.canvas.width = this.width;
                this.canvas.height = this.height;
                
                // 游戏状态
                this.isRunning = false;
                this.gameStarted = false;
                
                // 游戏对象
                this.player = null;
                this.enemies = [];
                this.foods = [];
                this.particles = [];
                
                // 游戏配置
                this.enemyCount = 10;
                this.foodCount = 100;
                this.particleCount = 20;
                
                // 游戏数据
                this.score = 0;
                this.rank = 1;
                
                // 输入处理
                this.mouseX = this.width / 2;
                this.mouseY = this.height / 2;
                
                // 初始化
                this.init();
            }
            
            init() {
                // 创建玩家
                this.player = new Player(this.width / 2, this.height / 2, 20, '#4CAF50');
                
                // 创建敌人
                for (let i = 0; i < this.enemyCount; i++) {
                    const x = Math.random() * this.width;
                    const y = Math.random() * this.height;
                    const radius = Math.random() * 30 + 10;
                    const color = this.getRandomColor();
                    this.enemies.push(new Enemy(x, y, radius, color));
                }
                
                // 创建食物
                for (let i = 0; i < this.foodCount; i++) {
                    const x = Math.random() * this.width;
                    const y = Math.random() * this.height;
                    const radius = Math.random() * 5 + 2;
                    const color = this.getRandomColor();
                    this.foods.push(new Food(x, y, radius, color));
                }
                
                // 创建粒子效果
                for (let i = 0; i < this.particleCount; i++) {
                    const x = Math.random() * this.width;
                    const y = Math.random() * this.height;
                    const radius = Math.random() * 2 + 1;
                    const color = this.getRandomColor();
                    this.particles.push(new Particle(x, y, radius, color));
                }
                
                // 绑定事件
                this.bindEvents();
            }
            
            bindEvents() {
                // 鼠标移动事件
                this.canvas.addEventListener('mousemove', (e) => {
                    const rect = this.canvas.getBoundingClientRect();
                    this.mouseX = e.clientX - rect.left;
                    this.mouseY = e.clientY - rect.top;
                });
                
                // 窗口大小调整
                window.addEventListener('resize', () => {
                    this.width = window.innerWidth;
                    this.height = window.innerHeight;
                    this.canvas.width = this.width;
                    this.canvas.height = this.height;
                });
                
                // 开始按钮
                document.getElementById('startButton').addEventListener('click', () => {
                    this.start();
                });
            }
            
            start() {
                this.isRunning = true;
                this.gameStarted = true;
                document.getElementById('startScreen').style.display = 'none';
                this.gameLoop();
            }
            
            gameLoop() {
                if (!this.isRunning) return;
                
                // 更新游戏状态
                this.update();
                
                // 渲染游戏
                this.render();
                
                // 继续游戏循环
                requestAnimationFrame(() => this.gameLoop());
            }
            
            update() {
                // 更新玩家
                this.player.update(this.mouseX, this.mouseY);
                
                // 边界检查
                this.checkBoundary(this.player);
                
                // 更新敌人
                for (let i = this.enemies.length - 1; i >= 0; i--) {
                    const enemy = this.enemies[i];
                    enemy.updateAI(this.player, this.width, this.height);
                    this.checkBoundary(enemy);
                    
                    // 玩家吃掉敌人
                    if (this.player.radius > enemy.radius && this.checkCollision(this.player, enemy)) {
                        this.eat(enemy, this.player);
                        this.enemies.splice(i, 1);
                        this.addScore(enemy.radius * 10);
                        
                        // 生成新敌人
                        const x = Math.random() * this.width;
                        const y = Math.random() * this.height;
                        const radius = Math.random() * 30 + 10;
                        const color = this.getRandomColor();
                        this.enemies.push(new Enemy(x, y, radius, color));
                    }
                    
                    // 敌人吃掉玩家
                    if (enemy.radius > this.player.radius && this.checkCollision(this.player, enemy)) {
                        this.gameOver();
                    }
                }
                
                // 更新食物
                for (let i = this.foods.length - 1; i >= 0; i--) {
                    const food = this.foods[i];
                    
                    // 玩家吃掉食物
                    if (this.checkCollision(this.player, food)) {
                        this.eat(food, this.player);
                        this.foods.splice(i, 1);
                        this.addScore(5);
                        
                        // 生成新食物
                        const x = Math.random() * this.width;
                        const y = Math.random() * this.height;
                        const radius = Math.random() * 5 + 2;
                        const color = this.getRandomColor();
                        this.foods.push(new Food(x, y, radius, color));
                    }
                }
                
                // 更新粒子
                for (let i = this.particles.length - 1; i >= 0; i--) {
                    const particle = this.particles[i];
                    particle.update();
                    
                    // 玩家吃掉粒子
                    if (this.checkCollision(this.player, particle)) {
                        this.eat(particle, this.player);
                        this.particles.splice(i, 1);
                        
                        // 生成新粒子
                        const x = Math.random() * this.width;
                        const y = Math.random() * this.height;
                        const radius = Math.random() * 2 + 1;
                        const color = this.getRandomColor();
                        this.particles.push(new Particle(x, y, radius, color));
                    }
                }
                
                // 更新排名
                this.updateRank();
            }
            
            render() {
                // 清空画布
                this.ctx.clearRect(0, 0, this.width, this.height);
                
                // 绘制背景网格
                this.drawGrid();
                
                // 绘制食物
                this.foods.forEach(food => food.draw(this.ctx));
                
                // 绘制粒子
                this.particles.forEach(particle => particle.draw(this.ctx));
                
                // 绘制敌人
                this.enemies.forEach(enemy => enemy.draw(this.ctx));
                
                // 绘制玩家
                this.player.draw(this.ctx);
                
                // 绘制玩家名称
                this.ctx.fillStyle = '#333';
                this.ctx.font = '16px Arial';
                this.ctx.textAlign = 'center';
                this.ctx.fillText('我', this.player.x, this.player.y - this.player.radius - 10);
            }
            
            drawGrid() {
                this.ctx.strokeStyle = '#ddd';
                this.ctx.lineWidth = 0.5;
                
                // 垂直线
                for (let x = 0; x < this.width; x += 50) {
                    this.ctx.beginPath();
                    this.ctx.moveTo(x, 0);
                    this.ctx.lineTo(x, this.height);
                    this.ctx.stroke();
                }
                
                // 水平线
                for (let y = 0; y < this.height; y += 50) {
                    this.ctx.beginPath();
                    this.ctx.moveTo(0, y);
                    this.ctx.lineTo(this.width, y);
                    this.ctx.stroke();
                }
            }
            
            checkBoundary(object) {
                // 左右边界
                if (object.x - object.radius < 0) {
                    object.x = object.radius;
                } else if (object.x + object.radius > this.width) {
                    object.x = this.width - object.radius;
                }
                
                // 上下边界
                if (object.y - object.radius < 0) {
                    object.y = object.radius;
                } else if (object.y + object.radius > this.height) {
                    object.y = this.height - object.radius;
                }
            }
            
            checkCollision(obj1, obj2) {
                const dx = obj1.x - obj2.x;
                const dy = obj1.y - obj2.y;
                const distance = Math.sqrt(dx * dx + dy * dy);
                
                return distance < obj1.radius + obj2.radius;
            }
            
            eat(eaten, eater) {
                // 增加半径
                eater.radius += eaten.radius * 0.1;
                
                // 创建吃掉效果
                this.createEatEffect(eaten.x, eaten.y, eaten.color);
            }
            
            createEatEffect(x, y, color) {
                for (let i = 0; i < 5; i++) {
                    const angle = Math.random() * Math.PI * 2;
                    const speed = Math.random() * 2 + 1;
                    const vx = Math.cos(angle) * speed;
                    const vy = Math.sin(angle) * speed;
                    const radius = Math.random() * 3 + 1;
                    
                    this.particles.push(new EffectParticle(x, y, vx, vy, radius, color));
                }
            }
            
            addScore(points) {
                this.score += points;
                document.getElementById('score').textContent = this.score;
            }
            
            updateRank() {
                // 计算排名
                const allBalls = [this.player, ...this.enemies];
                allBalls.sort((a, b) => b.radius - a.radius);
                
                const playerRank = allBalls.findIndex(ball => ball === this.player) + 1;
                this.rank = playerRank;
                document.getElementById('rank').textContent = this.rank;
            }
            
            gameOver() {
                this.isRunning = false;

            getRandomColor() {
                const colors = ['#FF5252', '#FF4081', '#E040FB', '#7C4DFF', '#536DFE', 
                               '#448AFF', '#40C4FF', '#18FFFF', '#64FFDA', '#69F0AE',

                return colors[Math.floor(Math.random() * colors.length)];
            }

        // 玩家类
        class Player {
            constructor(x, y, radius, color) {
                this.x = x;
                this.y = y;
                this.radius = radius;
                this.color = color;
                this.speed = 5;
                this.targetX = x;
                this.targetY = y;
            }
            
            update(mouseX, mouseY) {
                // 平滑移动到鼠标位置
                const dx = mouseX - this.x;
                const dy = mouseY - this.y;
                const distance = Math.sqrt(dx * dx + dy * dy);
                
                if (distance > 5) {
                    // 根据球的大小调整速度，球越大速度越慢
                    const speedFactor = Math.max(0.5, 1 - (this.radius - 20) / 100);
                    const moveX = (dx / distance) * this.speed * speedFactor;
                    const moveY = (dy / distance) * this.speed * speedFactor;
                    
                    this.x += moveX;
                    this.y += moveY;
                }
            }
            
            draw(ctx) {
                // 绘制球体
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
                ctx.fillStyle = this.color;
                ctx.fill();
                
                // 绘制高光效果
                ctx.beginPath();
                ctx.arc(this.x - this.radius * 0.3, this.y - this.radius * 0.3, this.radius * 0.2, 0, Math.PI * 2);
                ctx.fillStyle = 'rgba(255, 255, 255, 0.5)';
                ctx.fill();
            }
        }

        class Enemy {
            constructor(x, y, radius, color) {
                this.x = x;
                this.y = y;

                this.speed = 2;

                const dx = player.x - this.x;

                const distance = Math.sqrt(dx * dx + dy * dy);
                
                if (distance < 300) {
                    // 如果比玩家小，逃跑
                    if (this.radius < player.radius) {
                        this.targetX = this.x - dx;
                        this.targetY = this.y - dy;
                    } 
                    // 如果比玩家大，追逐
                    else {
                        this.targetX = player.x;
                        this.targetY = player.y;
                    }

                    // 随机移动
                    this.directionChangeTimer--;

                        this.direction = Math.random() * Math.PI * 2;
                        this.directionChangeTimer = Math.random() * 60 + 30;

                        const margin = this.radius + 50;
                        this.targetX = Math.max(margin, Math.min(width - margin, 
                            this.x + Math.cos(this.direction) * 100));
                        this.targetY = Math.max(margin, Math.min(height - margin, 
                            this.y + Math.sin(this.direction) * 100));
                    }
                }

                const dy2 = this.targetY - this.y;

                    const moveY = (dy2 / distance2) * this.speed * speedFactor;
                    
                    this.x += moveX;
                    this.y += moveY;
                }
            }
