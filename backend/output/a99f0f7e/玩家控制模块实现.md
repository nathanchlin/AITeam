        // 游戏状态

        // 玩家类
        class Player {
            constructor(id, name, x, y, color, isAI = false) {
                this.id = id;
                this.name = name;
                this.x = x;
                this.y = y;
                this.color = color;
                this.isAI = isAI;
                this.cells = [new Cell(this, x, y, 10, color)];
                this.score = 0;
                this.targetX = x;
                this.targetY = y;
            }

            update() {
                if (this.isAI) {
                    // AI行为
                    this.updateAI();
                } else {
                    // 玩家控制
                    this.updatePlayerControl();
                }

                // 更新所有细胞
                for (let i = 0; i < this.cells.length; i++) {
                    this.cells[i].update();
                }

                // 合并相近的细胞
                this.mergeCells();
            }

            updatePlayerControl() {
                const dx = game.mouseX - game.width / 2;
                const dy = game.mouseY - game.height / 2;
                const distance = Math.sqrt(dx * dx + dy * dy);
                
                if (distance > 5) {
                    const angle = Math.atan2(dy, dx);
                    
                    for (let cell of this.cells) {
                        const targetX = cell.x + Math.cos(angle) * cellSpeed(cell);
                        const targetY = cell.y + Math.sin(angle) * cellSpeed(cell);
                        
                        cell.moveTo(targetX, targetY);
                    }
                }
            }

            updateAI() {
                // 简单的AI行为：寻找最近的食物
                let nearestFood = null;
                let minDistance = Infinity;
                
                for (let food of game.foods) {
                    const dx = food.x - this.cells[0].x;
                    const dy = food.y - this.cells[0].y;
                    const distance = Math.sqrt(dx * dx + dy * dy);
                    
                    if (distance < minDistance) {
                        minDistance = distance;
                        nearestFood = food;
                    }
                }
                
                if (nearestFood) {
                    const dx = nearestFood.x - this.cells[0].x;
                    const dy = nearestFood.y - this.cells[0].y;
                    const angle = Math.atan2(dy, dx);
                    
                    for (let cell of this.cells) {
                        const targetX = cell.x + Math.cos(angle) * cellSpeed(cell);
                        const targetY = cell.y + Math.sin(angle) * cellSpeed(cell);
                        
                        cell.moveTo(targetX, targetY);
                    }
                }
            }

            split() {
                const now = Date.now();
                if (now - game.lastSplitTime < config.splitCooldown) return;
                
                const cellsToSplit = [];
                for (let cell of this.cells) {
                    if (cell.mass > config.minCellSize * 2) {
                        cellsToSplit.push(cell);
                    }
                }
                
                for (let cell of cellsToSplit) {
                    if (this.cells.length >= 16) break; // 最多16个细胞
                    
                    const angle = Math.random() * Math.PI * 2;
                    const newX = cell.x + Math.cos(angle) * cell.radius * 2;
                    const newY = cell.y + Math.sin(angle) * cell.radius * 2;
                    
                    const newMass = cell.mass / 2;
                    cell.mass = newMass;
                    cell.radius = Math.sqrt(newMass / Math.PI);
                    
                    this.cells.push(new Cell(this, newX, newY, newMass, this.color));
                }
                
                game.lastSplitTime = now;
            }

            eject() {
                const now = Date.now();
                if (now - game.lastEjectTime < 500) return; // 0.5秒冷却时间
                
                if (this.cells.length === 0) return;
                
                const mainCell = this.cells[0];
                if (mainCell.mass < config.ejectMass + config.minCellSize) return;
                
                const angle = Math.atan2(
                    game.mouseY - game.height / 2,
                    game.mouseX - game.width / 2
                );
                
                const newX = mainCell.x + Math.cos(angle) * (mainCell.radius + 10);
                const newY = mainCell.y + Math.sin(angle) * (mainCell.radius + 10);
                
                mainCell.mass -= config.ejectMass;
                mainCell.radius = Math.sqrt(mainCell.mass / Math.PI);
                
                const ejectedFood = new Food(newX, newY, config.ejectMass, this.color);
                ejectedFood.ejected = true;
                ejectedFood.ejectedBy = this;
                game.foods.push(ejectedFood);
                
                game.lastEjectTime = now;
            }

            mergeCells() {
                for (let i = 0; i < this.cells.length; i++) {
                    for (let j = i + 1; j < this.cells.length; j++) {
                        const cell1 = this.cells[i];
                        const cell2 = this.cells[j];
                        
                        const dx = cell1.x - cell2.x;
                        const dy = cell1.y - cell2.y;
                        const distance = Math.sqrt(dx * dx + dy * dy);
                        
                        if (distance < cell1.radius + cell2.radius) {
                            // 合并细胞
                            const totalMass = cell1.mass + cell2.mass;
                            cell1.mass = totalMass;
                            cell1.radius = Math.sqrt(totalMass / Math.PI);
                            
                            // 移除被合并的细胞
                            this.cells.splice(j, 1);
                            j--;
                            
                            // 更新分数
                            this.score += Math.floor(cell2.mass);
                        }
                    }
                }
            }

            draw() {
                for (let cell of this.cells) {
                    cell.draw();
                }
                
                // 绘制玩家名称
                if (this.cells.length > 0) {
                    const mainCell = this.cells[0];
                    game.ctx.fillStyle = '#000';
                    game.ctx.font = '16px Arial';
                    game.ctx.textAlign = 'center';
                    game.ctx.fillText(
                        this.name,
                        mainCell.x - game.camera.x,
                        mainCell.y - game.camera.y - mainCell.radius - 10
                    );
                }
            }
        }

        // 细胞类
        class Cell {
            constructor(player, x, y, mass, color) {
                this.player = player;
                this.x = x;
                this.y = y;
                this.mass = mass;
                this.color = color;
                this.radius = Math.sqrt(mass / Math.PI);
                this.vx = 0;
                this.vy = 0;
                this.targetX = x;
                this.targetY = y;
            }

            update() {
                // 应用速度
                this.x += this.vx;
                this.y += this.vy;
                
                // 应用摩擦力
                this.vx *= (1 - config.cellSpeedDecay);
                this.vy *= (1 - config.cellSpeedDecay);
                
                // 边界检查
                this.x = Math.max(this.radius, Math.min(game.width - this.radius, this.x));
                this.y = Math.max(this.radius, Math.min(game.height - this.radius, this.y));
            }

            moveTo(x, y) {
                this.targetX = x;
                this.targetY = y;
                
                const dx = x - this.x;
                const dy = y - this.y;
                const distance = Math.sqrt(dx * dx + dy * dy);
                
                if (distance > 0) {
                    const speed = cellSpeed(this);
                    this.vx = (dx / distance) * speed;
                    this.vy = (dy / distance) * speed;
                }
            }

            draw() {
                game.ctx.fillStyle = this.color;
                game.ctx.beginPath();
                game.ctx.arc(
                    this.x - game.camera.x,
                    this.y - game.camera.y,
                    this.radius,
                    0,
                    Math.PI * 2
                );
                game.ctx.fill();
                
                // 绘制边框
                game.ctx.strokeStyle = '#000';
                game.ctx.lineWidth = 2;
                game.ctx.stroke();
            }
        }

        // 食物类
        class Food {
            constructor(x, y, mass, color) {
                this.x = x;
                this.y = y;
                this.mass = mass;
                this.radius = Math.sqrt(mass / Math.PI);
                this.color = color;
                this.ejected = false;
                this.ejectedBy = null;

            draw() {
                game.ctx.fillStyle = this.color;
                game.ctx.beginPath();

                    this.x - game.camera.x,
                    this.y - game.camera.y,
                    this.radius,
                    0,

                game.ctx.fill();
            }
        }

        // 病毒类
        class Virus {
            constructor(x, y) {
                this.x = x;
                this.y = y;
                this.radius = config.virusSize;
                this.color = '#00FF00';
                this.angle = 0;
            }

            update() {
                this.angle += 0.01;
            }

            draw() {
                game.ctx.save();
                game.ctx.translate(this.x - game.camera.x, this.y - game.camera.y);
                game.ctx.rotate(this.angle);
                
                game.ctx.fillStyle = this.color;
                game.ctx.beginPath();
                
                // 绘制星形病毒
                const spikes = 8;
                const outerRadius = this.radius;
                const innerRadius = this.radius * 0.5;
                
                for (let i = 0; i < spikes * 2; i++) {
                    const radius = i % 2 === 0 ? outerRadius : innerRadius;
                    const angle = (i * Math.PI) / spikes;
                    const x = Math.cos(angle) * radius;
                    const y = Math.sin(angle) * radius;
                    
                    if (i === 0) {
                        game.ctx.moveTo(x, y);
                    } else {
                        game.ctx.lineTo(x, y);
                    }
                }
                
                game.ctx.closePath();
                game.ctx.fill();
                
                game.ctx.restore();
            }
        }

        // 计算细胞速度
        function cellSpeed(cell) {
            return Math.max(1, config.cellSpeed * Math.pow(cell.mass / 100, -0.5));
        }

        // 创建食物
        function createFood() {
            const mass = config.foodMinSize + Math.random() * (config.foodMaxSize - config.foodMinSize);
            const x = Math.random() * game.width;
            const y = Math.random() * game.height;
            const color = config.playerColors[Math.floor(Math.random() * config.playerColors.length)];
            
            game.foods.push(new Food(x, y, mass, color));
        }

        // 创建病毒
        function createVirus() {
            const x = Math.random() * game.width;
            const y = Math.random() * game.height;
            
            game.viruses.push(new Virus(x, y));
        }

        // 绑定事件
        function bindEvents() {
            // 鼠标移动
            game.canvas.addEventListener('mousemove', (e) => {
                game.mouseX = e.clientX;
                game.mouseY = e.clientY;
            });
            
            // 键盘事件
            window.addEventListener('keydown', (e) => {
                game.keys[e.key] = true;
                
                if (e.key === ' ') {
                    e.preventDefault();
                    game.players[0].split();
                }
                
                if (e.key === 'w' || e.key === 'W') {
                    // 加速
                    for (let cell of game.playerCells) {
                        cell.vx *= 1.5;
                        cell.vy *= 1.5;
                    }
                }
            });
            
            window.addEventListener('keyup', (e) => {
                game.keys[e.key] = false;
            });
            
            // 窗口大小调整
            window.addEventListener('resize', () => {
                game.width = window.innerWidth;
                game.height = window.innerHeight;
                resizeCanvas();
            });
        }

        // 更新游戏状态
        function update() {
            // 更新玩家
            for (let player of game.players) {
                player.update();
            }
            
            // 更新病毒
            for (let virus of game.viruses) {
                virus.update();
            }
            
            // 检查碰撞
            checkCollisions();
            
            // 补充食物
            while (game.foods.length < config.foodCount) {
                createFood();
            }
            
            // 更新相机位置（跟随玩家）
            if (game.playerCells.length > 0) {
                const mainCell = game.playerCells[0];
                game.camera.x = mainCell.x - game.width / 2;
                game.camera.y = mainCell.y - game.height / 2;
            }
            
            // 更新UI
            updateUI();
        }

        // 检查碰撞
        function checkCollisions() {
            // 玩家与食物碰撞
            for (let player of game.players) {
                for (let i = 0; i < player.cells.length; i++) {
                    const cell = player.cells[i];
                    
                    // 检查与食物的碰撞
                    for (let j = game.foods.length - 1; j >= 0; j--) {
                        const food = game.foods[j];
                        
                        const dx = cell.x - food.x;
                        const dy = cell.y - food.y;
                        const distance = Math.sqrt(dx * dx + dy * dy);

                            cell.mass += food.mass;

                            if (food.ejected &&
