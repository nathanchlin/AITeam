        class GameStateManager {
            constructor() {
                this.state = {
                    // 时间系统
                    currentDate: new Date(2024, 3, 1, 8, 0), // 2024年4月1日 8:00
                    timePhase: 'morning', // morning, afternoon, evening, night
                    
                    // 主角属性
                    stats: {
                        knowledge: 30,
                        charm: 30,
                        stamina: 80,
                        social: 30,
                        art: 20,
                        fitness: 30,
                        money: 5000
                    },
                    
                    // 角色好感度
                    affection: {
                        sakura: 0,
                        yuki: 0,
                        haru: 0
                    },
                    
                    // 游戏标志
                    flags: {
                        tutorialComplete: false,
                        metSakura: false,
                        metYuki: false,
                        metHaru: false
                    },
                    
                    // 日程事件
                    scheduledEvents: {},
                    
                    // 当前场景
                    currentScene: 'classroom',
                    
                    // 剧情进度
                    currentNode: 'start'
                };
            }
            
            // 获取当前时间段名称
            getTimePhaseName() {
                const phases = {
                    morning: '早晨',
                    noon: '午间',
                    afternoon: '下午',
                    evening: '傍晚',
                    night: '夜晚'
                };
                return phases[this.state.timePhase] || '早晨';
            }
            
            // 推进时间
            advanceTime(hours) {
                const currentHour = this.state.currentDate.getHours();
                let newHour = currentHour + hours;
                let dayAdvanced = false;
                
                // 处理跨天
                while (newHour >= 24) {
                    newHour -= 24;
                    this.state.currentDate.setDate(this.state.currentDate.getDate() + 1);
                    dayAdvanced = true;
                }
                
                this.state.currentDate.setHours(newHour);
                this.updateTimePhase();
                
                // 每天恢复部分体力
                if (dayAdvanced) {
                    this.state.stats.stamina = Math.min(
                        PlayerStats.stamina.max,
                        this.state.stats.stamina + 30
                    );
                }
                
                return dayAdvanced;
            }
            
            // 更新时间段
            updateTimePhase() {
                const hour = this.state.currentDate.getHours();
                if (hour >= 5 && hour < 12) {
                    this.state.timePhase = 'morning';
                } else if (hour >= 12 && hour < 14) {
                    this.state.timePhase = 'noon';
                } else if (hour >= 14 && hour < 18) {
                    this.state.timePhase = 'afternoon';
                } else if (hour >= 18 && hour < 21) {
                    this.state.timePhase = 'evening';
                } else {
                    this.state.timePhase = 'night';
                }
            }
            
            // 修改属性
            modifyStat(statName, amount) {
                const stat = PlayerStats[statName];
                if (!stat) return;
                
                const oldValue = this.state.stats[statName];
                let newValue = oldValue + amount;
                
                // 限制范围
                if (statName === 'money') {
                    newValue = Math.max(0, Math.min(stat.max, newValue));
                } else {
                    newValue = Math.max(0, Math.min(stat.max, newValue));
                }
                
                this.state.stats[statName] = newValue;
                return { oldValue, newValue, change: amount };
            }
            
            // 修改好感度
            modifyAffection(characterId, amount) {
                const maxAffection = 100;
                const oldValue = this.state.affection[characterId] || 0;
                const newValue = Math.max(0, Math.min(maxAffection, oldValue + amount));
                this.state.affection[characterId] = newValue;
                return { oldValue, newValue, change: amount };
            }
            
            // 执行行动
            executeAction(actionId) {
                let action = null;
                let category = null;
                
                // 查找行动
                for (const cat in ActionsData) {
                    const found = ActionsData[cat].find(a => a.id === actionId);
                    if (found) {
                        action = found;
                        category = cat;
                        break;
                    }
                }
                
                if (!action) return null;
                
                // 检查体力是否足够
                const staminaCost = Math.abs(action.effects.stamina || 0);
                if (action.effects.stamina < 0 && this.state.stats.stamina < staminaCost) {
                    return { success: false, message: '体力不足！' };
                }
                
                // 检查金钱是否足够
                const moneyCost = Math.abs(action.effects.money || 0);
                if (action.effects.money < 0 && this.state.stats.money < moneyCost) {
                    return { success: false, message: '金钱不足！' };
                }
                
                // 应用效果
                const changes = [];
                for (const stat in action.effects) {
                    const amount = action.effects[stat];
                    const change = this.modifyStat(stat, amount);
                    if (change && amount !== 0) {
                        changes.push({
                            stat,
                            statName: PlayerStats[stat].name,
                            icon: PlayerStats[stat].icon,
                            ...change
                        });
                    }
                }
                
                // 推进时间
                const dayAdvanced = this.advanceTime(action.timeCost || 2);
                
                return {
                    success: true,
                    action,
                    changes,
                    dayAdvanced
                };
            }
            
            // 执行约会
            executeDate(characterId) {
                const character = CharacterData[characterId];
                if (!character) return null;
                
                // 检查体力
                if (this.state.stats.stamina < 20) {
                    return { success: false, message: '体力不足！' };
                }
                
                // 消耗体力并增加好感度
                this.modifyStat('stamina', -15);
                const affectionChange = this.modifyAffection(characterId, 5 + Math.floor(Math.random() * 5));
                
                // 根据角色喜好增加额外属性
                const bonusEffects = [];
                if (characterId === 'sakura') {
                    this.modifyStat('charm', 2);
                    bonusEffects.push({ stat: 'charm', change: 2 });
                } else if (characterId === 'yuki') {
                    this.modifyStat('knowledge', 2);
                    bonusEffects.push({ stat: 'knowledge', change: 2 });
                } else if (characterId === 'haru') {
                    this.modifyStat('fitness', 2);
                    bonusEffects.push({ stat: 'fitness', change: 2 });
                }
                
                this.advanceTime(3);
                
                return {
                    success: true,
                    character,
                    affectionChange,
                    bonusEffects
                };
            }
            
            // 获取好感度等级（心数）
            getAffectionHearts(characterId) {
                const affection = this.state.affection[characterId] || 0;
                const hearts = [];
                const fullHearts = Math.floor(affection / 20);
                const halfHeart = (affection % 20) >= 10;
                
                for (let i = 0; i < 5; i++) {
                    if (i < fullHearts) {
                        hearts.push('full');
                    } else if (i === fullHearts && halfHeart) {
                        hearts.push('half');
                    } else {
                        hearts.push('empty');
                    }
                }
                
                return hearts;
            }
            
            // 保存游戏
            saveGame() {
                localStorage.setItem('tokimemo_save', JSON.stringify(this.state));
            }
            
            // 加载游戏
            loadGame() {
                const saved = localStorage.getItem('tokimemo_save');
                if (saved) {
                    this.state = JSON.parse(saved);
                    this.state.currentDate = new Date(this.state.currentDate);
                    return true;
                }
                return false;
            }
        }

            // 渲染日历
            renderCalendar() {
                const date = this.gameState.state.currentDate;
                const year = date.getFullYear();
                const month = date.getMonth();
                
                document.getElementById('calendarMonthYear').textContent = `${year}年${month + 1}月`;
                
                const container = document.getElementById('calendarDays');
                let html = '';
                
                // 获取当月第一天和最后一天
                const firstDay = new Date(year, month, 1);
                const lastDay = new Date(year, month + 1, 0);
                const daysInMonth = lastDay.getDate();
                const startWeekday = firstDay.getDay();
                
                // 上月日期填充
                const prevMonth = new Date(year, month, 0);
                const prevDays = prevMonth.getDate();
                for (let i = startWeekday - 1; i >= 0; i--) {
                    html += `<div class="calendar-day other-month">${prevDays - i}</div>`;
                }
                
                // 当月日期
                const today = date.getDate();
                for (let day = 1; day <= daysInMonth; day++) {
                    const isToday = day === today;
                    const weekday = new Date(year, month, day).getDay();
                    const isWeekend = weekday === 0 || weekday === 6;
                    
                    let classes = 'calendar-day';
                    if (isToday) classes += ' today';
                    if (isWeekend) classes += ' weekend';
                    
                    // 检查是否有事件
                    const dateKey = `${year}-${month + 1}-${day}`;
                    if (this.gameState.state.scheduledEvents[dateKey]) {
                        classes += ' has-event';
                    }
                    
                    html += `<div class="${classes}">${day}</div>`;
                }
                
                // 下月日期填充
                const totalCells = startWeekday + daysInMonth;
                const remaining = totalCells % 7 === 0 ? 0 : 7 - (totalCells % 7);
                for (let i = 1; i <= remaining; i++) {
                    html += `<div class="calendar-day other-month">${i}</div>`;
                }
                
                container.innerHTML = html;
            }

        class SceneRenderer {
            constructor(canvas, ctx) {
                this.canvas = canvas;
                this.ctx = ctx;
                this.particles = [];
                this.stars = [];
                this.currentScene = 'classroom';
                this.timeOfDay = 0;
                
                this.initParticles();
                this.initStars();
            }
            
            initParticles() {
                for (let i = 0; i < CONFIG.PARTICLE_COUNT; i++) {
                    this.particles.push({
                        x: Math.random() * this.canvas.width,
                        y: Math.random() * this.canvas.height,
                        size: Math.random() * 4 + 2,
                        speedX: (Math.random() - 0.5) * 0.5,
                        speedY: Math.random() * 0.5 + 0.2,
                        opacity: Math.random() * 0.5 + 0.3,
                        rotation: Math.random() * Math.PI * 2
                    });
                }
            }
            
            initStars() {
                for (let i = 0; i < CONFIG.STAR_COUNT; i++) {
                    this.stars.push({
                        x: Math.random() * this.canvas.width,
                        y: Math.random() * this.canvas.height * 0.6,
                        size: Math.random() * 2 + 0.5,
                        twinkle: Math.random() * Math.PI * 2,
                        speed: Math.random() * 0.02 + 0.01
                    });
                }
            }
            
            setTimeOfDay(hour) {
                this.timeOfDay = hour / 24;
            }
            
            render(scene, time) {
                this.currentScene = scene;
                
                // 清空画布
                this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);
                
                // 渲染背景
                this.renderBackground(time);
                
                // 渲染场景元素
                this.renderScene(scene, time);
                
                // 渲染粒子
                this.renderParticles(time);
                
                // 渲染氛围效果
                this.renderAtmosphere(time);
            }
            
            renderBackground(time) {
                const gradient = this.ctx.createLinearGradient(0, 0, 0, this.canvas.height);
                
                // 根据时间调整颜色
                const t = this.timeOfDay;
                
                if (t < 0.25) { // 夜晚到黎明
                    const p = t / 0.25;
                    gradient.addColorStop(0, this.lerpColor('#0a0a1a', '#1a1a3a', p));
                    gradient.addColorStop(1, this.lerpColor('#1a1a2e', '#2a2a4e', p));
                } else if (t < 0.5) { // 黎明到中午
                    const p = (t - 0.25) / 0.25;
                    gradient.addColorStop(0, this.lerpColor('#1a1a3a', '#4a6fa5', p));
                    gradient.addColorStop(1, this.lerpColor('#2a2a4e', '#7eb8da', p));
                } else if (t < 0.75) { // 中午到傍晚
                    const p = (t - 0.5) / 0.25;
                    gradient.addColorStop(0, this.lerpColor('#4a6fa5', '#ff7e5f', p));
                    gradient.addColorStop(1, this.lerpColor('#7eb8da', '#ffb347', p));
                } else { // 傍晚到夜晚
                    const p = (t - 0.75) / 0.25;
                    gradient.addColorStop(0, this.lerpColor('#ff7e5f', '#0a0a1a', p));
                    gradient.addColorStop(1, this.lerpColor('#ffb347', '#1a1a2e', p));
                }
                
                this.ctx.fillStyle = gradient;
                this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);
                
                // 渲染星星（夜晚）
                if (t < 0.3 || t > 0.8) {
                    this.renderStars(time);
                }
            }
            
            renderStars(time) {
                this.stars.forEach(star => {
                    star.twinkle += star.speed;
                    const opacity = 0.3 + Math.sin(star.twinkle) * 0.3;
                    
                    this.ctx.beginPath();
                    this.ctx.arc(star.x, star.y, star.size, 0, Math.PI * 2);
                    this.ctx.fillStyle = `rgba(255, 255, 255, ${opacity})`;
                    this.ctx.fill();
                });
            }
            
            renderScene(scene, time) {
                switch(scene) {
                    case 'classroom':
                        this.renderClassroom(time);
                        break;
                    case 'garden':
                        this.renderGarden(time);
                        break;
                    case 'rooftop':
                        this.renderRooftop(time);
                        break;
                    default:
                        this.renderClassroom(time);
                }
            }
            
            renderClassroom(time) {
                // 地板
                const floorGradient = this.ctx.createLinearGradient(0, 500, 0, 700);
                floorGradient.addColorStop(0, '#8b7355');
                floorGradient.addColorStop(1, '#6b5344');
                this.ctx.fillStyle = floorGradient;
                this.ctx.fillRect(0, 500, this.canvas.width, 200);
                
                // 窗户
                this.ctx.fillStyle = 'rgba(135, 206, 235, 0.3)';
                this.ctx.fillRect(100, 100, 200, 300);
                this.ctx.fillRect(350, 100, 200, 300);
                this.ctx.fillRect(600, 100, 200, 300);
                
                // 窗框
                this.ctx.strokeStyle = '#4a4a4a';
                this.ctx.lineWidth = 4;
                [100, 350, 600].forEach(x => {
                    this.ctx.strokeRect(x, 100, 200, 300);
                    this.ctx.beginPath();
                    this.ctx.moveTo(x + 100, 100);
                    this.ctx.lineTo(x + 100, 400);
                    this.ctx.stroke();
                    this.ctx.beginPath();
                    this.ctx.moveTo(x, 250);
                    this.ctx.lineTo(x + 200, 250);
                    this.ctx.stroke();
                });
                
                // 黑板
                this.ctx.fillStyle = '#2d4a3e';
                this.ctx.fillRect(850, 120, 300, 200);
                this.ctx.strokeStyle = '#8b7355';
                this.ctx.lineWidth = 8;
                this.ctx.strokeRect(850, 120, 300, 200);
                
                // 讲台
                this.ctx.fillStyle = '#5a4a3a';
                this.ctx.fillRect(880, 400, 250, 80);
            }
            
            renderGarden(time) {
                // 草地
                const grassGradient = this.ctx.createLinearGradient(0, 400, 0, 700);
                grassGradient.addColorStop(0, '#4a7c59');
                grassGradient.addColorStop(1, '#2d5a3d');
                this.ctx.fillStyle = grassGradient;
                this.ctx.fillRect(0, 400, this.canvas.width, 300);
                
                // 樱花树
                this.renderSakuraTree(200, 350, time);
                this.renderSakuraTree(900, 380, time);
                
                // 长椅
                this.ctx.fillStyle = '#8b5a2b';
                this.ctx.fillRect(500, 500, 200, 20);
                this.ctx.fillRect(520, 520, 20, 50);
                this.ctx.fillRect(660, 520, 20, 50);
                
                // 小路
                this.ctx.fillStyle = '#c9b896';
                this.ctx.beginPath();
                this.ctx.moveTo(0, 600);
                this.ctx.quadraticCurveTo(300, 550, 600, 600);
                this.ctx.quadraticCurveTo(900, 650, 1200, 600);
                this.ctx.lineTo(1200, 700);
                this.ctx.lineTo(0, 700);
                this.ctx.fill();
            }
            
            renderSakuraTree(x, y, time) {
                // 树干
                this.ctx.fillStyle = '#5a3d2b';
                this.ctx.fillRect(x - 20, y, 40, 150);
                
                // 树枝
                this.ctx.strokeStyle = '#5a3d2b';
                this.ctx.lineWidth = 8;
                this.ctx.beginPath();
                this.ctx.moveTo(x, y);
                this.ctx.lineTo(x - 80, y - 50);
                this.ctx.moveTo(x, y);
                this.ctx.lineTo(x + 80, y - 30);
                this.ctx.stroke();
                
                // 樱花
                const petalCount = 30;
                for (let i = 0; i < petalCount; i++) {
                    const angle = (i / petalCount) * Math.PI * 2 + time * 0.0002;
                    const radius = 60 + Math.sin(time * 0.001 + i) * 10;
                    const px = x + Math.cos(angle) * radius;
                    const py = y - 30 + Math.sin(angle) * radius * 0.6;
                    
                    this.ctx.beginPath();
                    this.ctx.arc(px, py, 15 + Math.random() * 10, 0, Math.PI * 2);
                    this.ctx.fillStyle = `rgba(255, 183, 197, ${0.6 + Math.random() * 0.3})`;
                    this.ctx.fill();
                }
            }
            
            renderRooftop(time) {
                // 地面
                this.ctx.fillStyle = '#4a4a4a';
                this.ctx.fillRect(0, 450, this.canvas.width, 250);
                
                // 栏杆
                this.ctx.strokeStyle = '#7a7a7a';
                this.ctx.lineWidth = 6;
                this.ctx.beginPath();
                this.ctx.moveTo(0, 420);
                this.ctx.lineTo(this.canvas.width, 420);
                this.ctx.stroke();
                
                // 栏杆柱子
                for (let x = 50; x < this.canvas.width; x += 100) {
                    this.ctx.fillStyle = '#6a6a6a';
                    this.ctx.fillRect(x - 5, 350, 10, 70);
                }
                
                // 水箱
                this.ctx.fillStyle = '#5a5a5a';
                this.ctx.fillRect(100, 200, 150, 200);
                this.ctx.fillRect(950, 250, 120, 150);
                
                // 栏杆网格
                this.ctx.strokeStyle = 'rgba(200, 200, 200, 0.3)';
                this.ctx.lineWidth = 2;
                for (let x = 50; x < this.canvas.width; x += 20) {
                    this.ctx.beginPath();
                    this.ctx.moveTo(x, 350);
                    this.ctx.lineTo(x, 420);
                    this.ctx.stroke();
                }
            }
            
            renderParticles(time) {
                this.particles.forEach(p => {
                    p.x += p.speedX;
                    p.y += p.speedY;
                    p.rotation += 0.01;
                    
                    // 循环
                    if (p.y > this.canvas.height + 10) {
                        p.y = -10;
                        p.x = Math.random() * this.canvas.width;
                    }
                    if (p.x < -10) p.x = this.canvas.width + 10;
                    if (p.x > this.canvas.width + 10) p.x = -10;
                    
                    // 渲染花瓣形状
                    this.ctx.save();
                    this.ctx.translate(p.x, p.y);
                    this.ctx.rotate(p.rotation);
                    
                    this.ctx.beginPath();
                    this.ctx.ellipse(0, 0, p.size, p.size * 0.6, 0, 0, Math.PI * 2);
                    this.ctx.fillStyle = `rgba(255, 183, 197, ${p.opacity})`;
                    this.ctx.fill();
                    
                    this.ctx.restore();
                });
            }
            
            renderAtmosphere(time) {
                // 添加光晕效果
                const glowGradient = this.ctx.createRadialGradient(
                    this.canvas.width / 2, this.canvas.height / 2, 0,
                    this.canvas.width / 2, this.canvas.height / 2, 500
                );
                glowGradient.addColorStop(0, 'rgba(255, 183, 197, 0.05)');
                glowGradient.addColorStop(1, 'rgba(255, 183, 197, 0)');
                this.ctx.fillStyle = glowGradient;
                this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);
            }
            
            lerpColor(color1, color2, t) {
                const c1 = this.hexToRgb(color1);
                const c2 = this.hexToRgb(color2);
                
                const r = Math.round(c1.r + (c2.r - c1.r) * t);
                const g = Math.round(c1.g + (c2.g - c1.g) * t);
                const b = Math.round(c1.b + (c2.b - c1.b) * t);
                
                return `rgb(${r}, ${g}, ${b})`;
            }
            
            hexToRgb(hex) {
                const result = /^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i.exec(hex);
                return result ? {
                    r: parseInt(result[1], 16),
                    g: parseInt(result[2], 16),
                    b: parseInt(result[3], 16)
                } : { r: 0, g: 0, b: 0 };
            }
        }

        class CharacterRenderer {
            constructor(ctx) {
                this.ctx = ctx;
                this.expressionState = ExpressionState.NORMAL;
                this.animationTime = 0;
                this.blinkTimer = 0;
                this.isBlinking = false;
            }
            
            renderCharacter(characterData, x, y, scale, expression, time) {
                this.animationTime = time;
                this.expressionState = expression || ExpressionState.NORMAL;
                
                // 呼吸动画
                const breathOffset = Math.sin(time * 0.002) * 3;
                
                this.ctx.save();
                this.ctx.translate(x, y + breathOffset);
                this.ctx.scale(scale, scale);
                
                // 渲染各个部分
                this.renderHair(characterData, 0, 0);
                this.renderFace(characterData, 0, 0);
                this.renderEyes(characterData, 0, 0, time);
                this.renderMouth(characterData, 0, 0);
                this.renderBody(characterData, 0, 80);
                this.renderAccessories(characterData, 0, 0);
                
                // 渲染表情特效
                this.renderExpressionEffects(expression, 0, -50);
                
                this.ctx.restore();
            }
            
            renderHair(data, x, y) {
                // 后发
                this.ctx.fillStyle = data.hairColor;
                this.ctx.beginPath();

                // Canvas点击事件

                    this.handleCanvasClick(e);
                });
                
                // 键盘事件
                document.addEventListener('keydown', (e) => {

                });
                
                // 窗口关闭前保存
                window.addEventListener('beforeunload', () => {
                    this.gameState.saveGame();
                });
            }
            
            handleCanvasClick(e) {
                const rect = this.canvas.getBoundingClientRect();
                const x = e.clientX - rect.left;
                const y = e.clientY - rect.top;
                
                // 简单的场景切换测试
                if (y < 100) {
                    const scenes = ['classroom', 'garden', 'rooftop'];
                    const currentIndex = scenes.indexOf(this.currentScene);
                    this.currentScene = scenes[(currentIndex + 1) % scenes.length];
                    this.gameState.state.currentScene = this.currentScene;
                }
                
                // 切换角色显示
                if (x > 400 && x < 800 && y > 200 && y < 500) {
                    if (!this.showCharacter) {
                        this.showCharacter = 'sakura';
                        this.characterExpression = ExpressionState.HAPPY;
                    } else {
                        const characters = ['sakura', 'yuki', 'haru'];
                        const currentIndex = characters.indexOf(this.showCharacter);
                        this.showCharacter = characters[(currentIndex + 1) % characters.length];
                        
                        // 随机表情
                        const expressions = Object.values(ExpressionState);
                        this.characterExpression = expressions[Math.floor(Math.random() * expressions.length)];
                    }
                }
            }
            
            handleKeyDown(e) {
                switch(e.key) {
                    case 's':
                    case 'S':
                        this.gameState.saveGame();
                        this.uiManager.showToast('💾 保存成功', '<p>游戏进度已保存</p>');
                        break;
                    case 'Escape':
                        // 关闭所有面板
                        document.querySelectorAll('.schedule-panel, .calendar-panel').forEach(panel => {
                            panel.classList.remove('visible');
                        });
                        break;
                }
            }

                if (!this.isRunning) return;
                
                const currentTime = performance.now();
                const deltaTime = currentTime - this.lastTime;
                this.lastTime = currentTime;
                
                // 更新时间显示
                this.uiManager.updateTimeDisplay();
                
                // 更新场景时间
                const hour = this.gameState.state.currentDate.getHours();
                this.sceneRenderer.setTimeOfDay(hour);
                
                // 渲染
                this.render(currentTime);
                
                requestAnimationFrame(() => this.gameLoop());
            }

                            time

                    }
                }
            }
