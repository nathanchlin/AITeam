// 存档管理系统
// ============================================
class SaveManager {
    constructor() {
        this.SAVE_KEY_PREFIX = 'tokimemo_save_';
        this.AUTO_SAVE_KEY = 'tokimemo_autosave';
        this.CONFIG_KEY = 'tokimemo_config';
        this.VERSION = '1.0.0';
        this.CHECKSUM_SALT = 'tokimemo_2024';
    }

    // 计算校验和
    calculateChecksum(data) {
        const str = JSON.stringify(data) + this.CHECKSUM_SALT;
        let hash = 0;
        for (let i = 0; i < str.length; i++) {
            const char = str.charCodeAt(i);
            hash = ((hash << 5) - hash) + char;
            hash = hash & hash;
        }
        return hash.toString(16);
    }

    // 序列化游戏状态
    serializeGameState() {
        const saveData = {
            version: this.VERSION,
            timestamp: Date.now(),
            playtime: GameState.stats.totalPlayTime || 0,
            scene: GameState.currentScene || 'game',
            player: JSON.parse(JSON.stringify(GameState.player)),
            time: JSON.parse(JSON.stringify(GameState.time)),
            affections: JSON.parse(JSON.stringify(GameState.affections)),
            flags: JSON.parse(JSON.stringify(GameState.flags)),
            dialogue: {
                current: GameState.dialogue.current,
                currentIndex: GameState.dialogue.currentIndex
            },
            settings: JSON.parse(JSON.stringify(GameState.settings)),
            stats: JSON.parse(JSON.stringify(GameState.stats))
        };
        
        // 计算校验和
        saveData.checksum = this.calculateChecksum({
            player: saveData.player,
            time: saveData.time,
            affections: saveData.affections
        });
        
        return JSON.stringify(saveData);
    }

    // 反序列化游戏状态
    deserializeGameState(jsonString) {
        try {
            const data = JSON.parse(jsonString);
            if (!this.validateSaveData(data)) {
                console.error('存档数据校验失败');
                return null;
            }
            return data;
        } catch (e) {
            console.error('存档解析失败:', e);
            return null;
        }
    }

    // 验证存档数据完整性
    validateSaveData(data) {
        // 基本结构检查
        if (!data || typeof data !== 'object') return false;
        if (!data.version || !data.timestamp) return false;
        if (!data.player || !data.time || !data.affections) return false;
        
        // 校验和验证
        if (data.checksum) {
            const expectedChecksum = this.calculateChecksum({
                player: data.player,
                time: data.time,
                affections: data.affections
            });
            if (data.checksum !== expectedChecksum) {
                console.warn('存档校验和不匹配，数据可能已被篡改');
                // 可以选择仍然加载数据，但标记为不可信
            }
        }
        
        // 玩家数据检查
        if (!data.player.stats || typeof data.player.stats !== 'object') return false;
        
        // 时间数据检查
        if (typeof data.time.year !== 'number' || 
            typeof data.time.month !== 'number' ||
            typeof data.time.day !== 'number') return false;
        
        return true;
    }

    // 保存到指定槽位
    saveToSlot(slotIndex) {
        if (slotIndex < 0 || slotIndex >= CONFIG.MAX_SAVE_SLOTS) {
            return { success: false, message: '无效的存档槽位' };
        }
        
        try {
            const key = this.SAVE_KEY_PREFIX + slotIndex;
            const data = this.serializeGameState();
            localStorage.setItem(key, data);
            
            GameState.stats.saveCount++;
            
            return { success: true, message: '保存成功' };
        } catch (e) {
            console.error('保存失败:', e);
            if (e.name === 'QuotaExceededError') {
                return { success: false, message: '存储空间不足' };
            }
            return { success: false, message: '保存失败: ' + e.message };
        }
    }

    // 从槽位读取
    loadFromSlot(slotIndex) {
        const key = this.SAVE_KEY_PREFIX + slotIndex;
        const data = localStorage.getItem(key);
        
        if (!data) {
            return { success: false, message: '存档不存在' };
        }
        
        const parsed = this.deserializeGameState(data);
        if (!parsed) {
            return { success: false, message: '存档数据损坏' };
        }
        
        // 应用存档数据到游戏状态
        this.applySaveData(parsed);
        
        return { success: true, message: '读取成功', data: parsed };
    }

    // 应用存档数据
    applySaveData(data) {
        GameState.currentScene = data.scene || 'game';
        GameState.player = JSON.parse(JSON.stringify(data.player));
        GameState.time = JSON.parse(JSON.stringify(data.time));
        GameState.affections = JSON.parse(JSON.stringify(data.affections));
        GameState.flags = data.flags ? JSON.parse(JSON.stringify(data.flags)) : {};
        GameState.dialogue = data.dialogue ? JSON.parse(JSON.stringify(data.dialogue)) : { current: null, currentIndex: 0 };
        GameState.settings = data.settings ? JSON.parse(JSON.stringify(data.settings)) : GameState.settings;
        GameState.stats = data.stats ? JSON.parse(JSON.stringify(data.stats)) : GameState.stats;
    }

    // 删除存档
    deleteSlot(slotIndex) {
        const key = this.SAVE_KEY_PREFIX + slotIndex;
        localStorage.removeItem(key);
        return { success: true, message: '删除成功' };
    }

    // 获取存档预览信息
    getSlotPreview(slotIndex) {
        const key = this.SAVE_KEY_PREFIX + slotIndex;
        const data = localStorage.getItem(key);
        
        if (!data) return null;
        
        const parsed = this.deserializeGameState(data);
        if (!parsed) return null;
        
        return {
            index: slotIndex,
            exists: true,
            timestamp: parsed.timestamp,
            scene: parsed.scene,
            sceneName: SceneNames[parsed.scene] || parsed.scene,
            player: parsed.player,
            time: parsed.time,
            playtime: parsed.playtime || 0
        };
    }

    // 获取所有存档槽位信息
    getAllSaveSlots() {
        const slots = [];
        for (let i = 0; i < CONFIG.MAX_SAVE_SLOTS; i++) {
            const preview = this.getSlotPreview(i);
            if (preview) {
                slots.push(preview);
            } else {
                slots.push({ index: i, exists: false });
            }
        }
        return slots;
    }

    // 自动存档
    autoSave() {
        try {
            const data = this.serializeGameState();
            localStorage.setItem(this.AUTO_SAVE_KEY, data);
            return { success: true };
        } catch (e) {
            console.error('自动存档失败:', e);
            return { success: false };
        }
    }

    // 获取自动存档
    getAutoSave() {
        const data = localStorage.getItem(this.AUTO_SAVE_KEY);
        if (!data) return null;
        
        return this.deserializeGameState(data);
    }

    // 格式化时间戳
    formatTimestamp(timestamp) {
        const date = new Date(timestamp);
        const year = date.getFullYear();
        const month = String(date.getMonth() + 1).padStart(2, '0');
        const day = String(date.getDate()).padStart(2, '0');
        const hour = String(date.getHours()).padStart(2, '0');
        const minute = String(date.getMinutes()).padStart(2, '0');
        return `${year}/${month}/${day} ${hour}:${minute}`;
    }

    // 格式化游戏时间
    formatPlaytime(seconds) {
        if (!seconds || seconds < 0) return '0分钟';
        const hours = Math.floor(seconds / 3600);
        const mins = Math.floor((seconds % 3600) / 60);
        if (hours > 0) {
            return `${hours}小时${mins}分钟`;
        }
        return `${mins}分钟`;
    }

    // 保存设置
    saveSettings() {
        try {
            localStorage.setItem(this.CONFIG_KEY, JSON.stringify(GameState.settings));
            return true;
        } catch (e) {
            return false;
        }
    }

    // 加载设置
    loadSettings() {
        try {
            const data = localStorage.getItem(this.CONFIG_KEY);
            if (data) {
                GameState.settings = { ...GameState.settings, ...JSON.parse(data) };
            }
            return true;
        } catch (e) {
            return false;
        }
    }

    // 检查存储空间
    getStorageInfo() {
        let totalSize = 0;
        for (let key in localStorage) {
            if (localStorage.hasOwnProperty(key)) {
                totalSize += localStorage[key].length * 2; // UTF-16
            }
        }
        return {
            used: totalSize,
            usedMB: (totalSize / 1024 / 1024).toFixed(2)
        };
    }
}

// 自动存档控制器

class AutoSaveController {
    constructor() {
        this.interval = null;
        this.lastSaveTime = 0;
        this.isEnabled = true;
    }

    start() {
        if (this.interval) return;
        
        this.interval = setInterval(() => {
            if (this.isEnabled && GameState.currentScreen === 'game') {
                this.performAutoSave();
            }
        }, CONFIG.AUTO_SAVE_INTERVAL);
        
        console.log('自动存档已启动');
    }

    stop() {
        if (this.interval) {
            clearInterval(this.interval);
            this.interval = null;
        }
    }

    performAutoSave() {
        const indicator = document.getElementById('autoSaveIndicator');
        if (indicator) {
            indicator.classList.add('visible');
        }
        
        setTimeout(() => {
            const result = saveManager.autoSave();
            this.lastSaveTime = Date.now();
            
            if (indicator) {
                indicator.classList.remove('visible');
            }
            
            if (result.success) {
                console.log('自动存档完成');
            }
        }, 500);
    }

    // 场景切换时触发自动存档
    onSceneChange() {
        if (this.isEnabled && GameState.currentScreen === 'game') {
            this.performAutoSave();
        }
    }
}

// 执行读取
function performLoad(index, isAutoSave) {
    if (isAutoSave) {
        const autoData = saveManager.getAutoSave();
        if (autoData) {
            saveManager.applySaveData(autoData);
            hideSaveScreen();
            enterGame();
            showNotification('已读取自动存档', 'success');
        } else {
            showNotification('自动存档不存在', 'error');
        }
    } else {
        const result = saveManager.loadFromSlot(index);
        
        if (result.success) {
            hideSaveScreen();
            enterGame();
            showNotification('读取成功！', 'success');
        } else {
            showNotification(result.message, 'error');
        }
    }
}

    if (periodDisplay) {

        let periodIndex = 0;
        if (hour >= 0 && hour < 5) periodIndex = 0;
        else if (hour >= 5 && hour < 7) periodIndex = 1;
        else if (hour >= 7 && hour < 9) periodIndex = 2;
        else if (hour >= 9 && hour < 12) periodIndex = 3;
        else if (hour >= 12 && hour < 14) periodIndex = 4;
        else if (hour >= 14 && hour < 17) periodIndex = 5;
        else if (hour >= 17 && hour < 19) periodIndex = 6;
        else periodIndex = 7;
        
        periodDisplay.textContent = periods[periodIndex];
    }
}

function createParticle() {
    const colors = ['#ffb7c5', '#ffd4de', '#ffffff', '#ffe4e8'];
    return {
        x: Math.random() * canvas.width,
        y: Math.random() * canvas.height - canvas.height,
        size: Math.random() * 8 + 4,
        speedX: Math.random() * 2 - 1,
        speedY: Math.random() * 1.5 + 0.5,
        rotation: Math.random() * Math.PI * 2,
        rotationSpeed: (Math.random() - 0.5) * 0.05,
        opacity: Math.random() * 0.6 + 0.4,
        color: colors[Math.floor(Math.random() * colors.length)],
        swing: Math.random() * Math.PI * 2,
        swingSpeed: Math.random() * 0.02 + 0.01
    };
}

function updateParticles() {
    particles.forEach(p => {
        p.swing += p.swingSpeed;
        p.x += p.speedX + Math.sin(p.swing) * 0.5;
        p.y += p.speedY;
        p.rotation += p.rotationSpeed;

        if (p.y > canvas.height + 20) {
            p.y = -20;
            p.x = Math.random() * canvas.width;
        }
        if (p.x < -20) p.x = canvas.width + 20;
        if (p.x > canvas.width + 20) p.x = -20;
    });
}

function drawBackground() {
    // 创建渐变背景
    const gradient = ctx.createLinearGradient(0, 0, 0, canvas.height);
    gradient.addColorStop(0, '#1a1a2e');
    gradient.addColorStop(0.5, '#16213e');
    gradient.addColorStop(1, '#1a1a2e');
    ctx.fillStyle = gradient;
    ctx.fillRect(0, 0, canvas.width, canvas.height);
    
    // 绘制装饰性光斑
    const time = Date.now() * 0.001;
    for (let i = 0; i < 3; i++) {
        const x = canvas.width * (0.2 + i * 0.3) + Math.sin(time + i) * 50;
        const y = canvas.height * 0.3 + Math.cos(time * 0.7 + i) * 30;
        const gradient = ctx.createRadialGradient(x, y, 0, x, y, 150);
        gradient.addColorStop(0, 'rgba(255, 183, 197, 0.15)');
        gradient.addColorStop(1, 'transparent');
        ctx.fillStyle = gradient;
        ctx.beginPath();
        ctx.arc(x, y, 150, 0, Math.PI * 2);
        ctx.fill();
    }
}

function drawParticles() {
    particles.forEach(p => {
        ctx.save();
        ctx.translate(p.x, p.y);
        ctx.rotate(p.rotation);
        ctx.globalAlpha = p.opacity;
        ctx.fillStyle = p.color;
        
        // 绘制花瓣形状
        ctx.beginPath();
        ctx.moveTo(0, -p.size);
        ctx.bezierCurveTo(p.size * 0.5, -p.size * 0.5, p.size * 0.5, p.size * 0.5, 0, p.size);
        ctx.bezierCurveTo(-p.size * 0.5, p.size * 0.5, -p.size * 0.5, -p.size * 0.5, 0, -p.size);
        ctx.fill();
        
        ctx.restore();
    });
}

function gameLoop() {
    // 清除画布
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    
    // 绘制背景
    drawBackground();
    
    // 更新和绘制粒子
    if (GameState.currentScreen !== 'menu') {
        updateParticles();
        drawParticles();
    }
    
    // 继续动画循环
    animationId = requestAnimationFrame(gameLoop);
}
