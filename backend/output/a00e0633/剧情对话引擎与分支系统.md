    class TypewriterEffect {
        constructor() {
            this.text = '';
            this.displayedText = '';
            this.currentIndex = 0;
            this.isTyping = false;
            this.isComplete = false;
            this.speed = CONFIG.TYPE_SPEED;
            this.lastTypeTime = 0;
            this.onComplete = null;
            this.skipRequested = false;
        }

        start(text, speed = CONFIG.TYPE_SPEED) {
            this.text = text;
            this.displayedText = '';
            this.currentIndex = 0;
            this.isTyping = true;
            this.isComplete = false;
            this.speed = speed;
            this.skipRequested = false;
        }

        update(currentTime) {
            if (!this.isTyping || this.isComplete) return;

            if (this.skipRequested) {
                this.displayedText = this.text;
                this.complete();
                return;
            }

            if (currentTime - this.lastTypeTime >= this.speed) {
                if (this.currentIndex < this.text.length) {
                    // 处理富文本标签（简单处理）
                    const char = this.text[this.currentIndex];
                    this.displayedText += char;
                    this.currentIndex++;
                    this.lastTypeTime = currentTime;
                } else {
                    this.complete();
                }
            }
        }

        complete() {
            this.isTyping = false;
            this.isComplete = true;
            if (this.onComplete) {
                this.onComplete();
            }
        }

        skip() {
            if (this.isTyping && !this.isComplete) {
                this.skipRequested = true;
            }
        }

        getDisplayText() {
            return this.displayedText;
        }

        getCursorHTML() {
            return this.isTyping ? '<span class="typing-cursor"></span>' : '';
        }
    }

    class DialogEngine {
        constructor(game) {
            this.game = game;
            this.currentNode = null;
            this.currentNodeId = null;
            this.typewriter = new TypewriterEffect();
            this.isAnimating = false;
            this.scriptData = ScriptData;
            this.waitingForInput = false;
            this.waitingForChoice = false;
            this.currentChoices = [];
            
            this.dialogContainer = document.getElementById('dialogContainer');
            this.dialogSpeaker = document.getElementById('dialogSpeaker');
            this.dialogText = document.getElementById('dialogText');
            this.dialogContinue = document.getElementById('dialogContinue');
            this.choiceContainer = document.getElementById('choiceContainer');
            
            this.init();
        }

        init() {
            // 点击对话框继续
            this.dialogContainer.addEventListener('click', (e) => {
                if (this.waitingForInput) {
                    this.advance();
                } else if (this.typewriter.isTyping) {
                    this.typewriter.skip();
                }
            });

            // 键盘事件
            document.addEventListener('keydown', (e) => {
                if (e.code === 'Space' || e.code === 'Enter') {
                    e.preventDefault();
                    if (this.waitingForInput) {
                        this.advance();
                    } else if (this.typewriter.isTyping) {
                        this.typewriter.skip();
                    }
                }
                
                // 数字键选择选项
                if (this.waitingForChoice && e.key >= '1' && e.key <= '9') {
                    const index = parseInt(e.key) - 1;
                    if (index < this.currentChoices.length) {
                        this.selectChoice(index);
                    }
                }
            });
        }

        startScript(startNodeId = 'start') {
            this.game.gameState.flags = {};
            this.game.gameState.affection = { ...ScriptData.initialAffection };
            this.goToNode(startNodeId);
        }

        goToNode(nodeId) {
            this.currentNodeId = nodeId;
            this.currentNode = this.scriptData.nodes[nodeId];

            if (!this.currentNode) {
                console.error(`Node not found: ${nodeId}`);
                return;
            }

            this.processNode();
        }

        processNode() {
            const node = this.currentNode;

            switch(node.type) {
                case 'dialogue':
                    this.processDialogue(node);
                    break;
                case 'choice':
                    this.processDialogue(node);
                    break;
                case 'scene':
                    this.processScene(node);
                    break;
                case 'transition':
                    this.processTransition(node);
                    break;
                case 'end':
                    this.processEnd(node);
                    break;
                default:
                    console.warn(`Unknown node type: ${node.type}`);
            }
        }

        processDialogue(node) {
            // 显示对话框
            this.dialogContainer.classList.add('active');
            
            // 设置说话者名称
            if (node.speakerName) {
                this.dialogSpeaker.textContent = node.speakerName;
                this.dialogSpeaker.style.display = 'block';
            } else {
                this.dialogSpeaker.style.display = 'none';
            }

            // 处理角色效果
            if (node.speaker && node.speaker !== 'narrator') {
                this.game.showCharacter(node.speaker, node.expression || 'normal');
                
                if (node.characterEffect === 'enter') {
                    this.game.characterEnter(node.speaker);
                }
            } else {
                this.game.hideAllCharacters();
            }

            // 应用效果
            if (node.effects) {
                this.applyEffects(node.effects);
            }

            // 开始打字机效果
            this.typewriter.start(node.text);
            this.waitingForInput = false;
            this.dialogContinue.style.display = 'none';

            this.typewriter.onComplete = () => {
                if (node.type === 'choice') {
                    this.showChoices(node.choices);
                } else {
                    this.waitingForInput = true;
                    this.dialogContinue.style.display = 'flex';
                }
            };
        }

        processScene(node) {
            this.game.changeScene(node.scene);
            setTimeout(() => {
                this.goToNode(node.next);
            }, 100);
        }

        processTransition(node) {
            this.game.transition(node.transitionType, node.duration, () => {
                this.goToNode(node.next);
            });
        }

        processEnd(node) {
            this.dialogContainer.classList.remove('active');
            this.game.showEnding(node.message);
        }

        showChoices(choices) {
            this.currentChoices = choices;
            this.waitingForChoice = true;
            this.choiceContainer.innerHTML = '';

            choices.forEach((choice, index) => {
                const btn = document.createElement('button');
                btn.className = 'choice-btn';
                
                // 检查条件
                let isLocked = false;
                if (choice.condition) {
                    isLocked = !this.checkCondition(choice.condition);
                }

                if (isLocked) {
                    btn.classList.add('choice-locked');
                    btn.innerHTML = `
                        <span>${choice.text}</span>
                        <div class="choice-requirement">[需要: 好感度 ${choice.condition.value}+]</div>
                    `;
                } else {
                    btn.innerHTML = `
                        <span>${index + 1}. ${choice.text}</span>
                        <span class="choice-indicator">▶</span>
                    `;
                    btn.addEventListener('click', () => this.selectChoice(index));
                }

                this.choiceContainer.appendChild(btn);
            });

            this.choiceContainer.classList.add('active');
        }

        selectChoice(index) {
            if (!this.waitingForChoice) return;

            const choice = this.currentChoices[index];
            this.waitingForChoice = false;
            this.choiceContainer.classList.remove('active');

            // 应用效果
            if (choice.effects) {
                this.applyEffects(choice.effects);
            }

            // 跳转到下一个节点
            this.goToNode(choice.next);
        }

        checkCondition(condition) {
            const gameState = this.game.gameState;
            
            switch(condition.type) {
                case 'affection':
                    return ScriptData.conditions.checkAffection(
                        gameState, 
                        condition.character, 
                        condition.operator, 
                        condition.value
                    );
                case 'flag':
                    return ScriptData.conditions.checkFlag(gameState, condition.flagName);
                case 'stat':
                    return ScriptData.conditions.checkStat(
                        gameState, 
                        condition.statName, 
                        condition.operator, 
                        condition.value
                    );
                default:
                    return true;
            }
        }

        applyEffects(effects) {
            if (effects.affection) {
                Object.keys(effects.affection).forEach(character => {
                    const change = effects.affection[character];
                    this.game.changeAffection(character, change);
                });
            }

            if (effects.flags) {
                Object.keys(effects.flags).forEach(flag => {
                    this.game.gameState.flags[flag] = effects.flags[flag];
                });
            }

            if (effects.stats) {
                Object.keys(effects.stats).forEach(stat => {
                    this.game.changeStat(stat, effects.stats[stat]);
                });
            }
        }

        advance() {
            if (!this.waitingForInput || this.waitingForChoice) return;
            
            this.waitingForInput = false;
            this.dialogContinue.style.display = 'none';

            if (this.currentNode.next) {
                this.goToNode(this.currentNode.next);
            }
        }

        update(currentTime) {
            if (this.typewriter.isTyping) {
                this.typewriter.update(currentTime);
                this.dialogText.innerHTML = this.typewriter.getDisplayText() + this.typewriter.getCursorHTML();
            }
        }

        hide() {
            this.dialogContainer.classList.remove('active');
            this.choiceContainer.classList.remove('active');
        }

        show() {
            this.dialogContainer.classList.add('active');
        }
    }

    class DayNightCycle {
        constructor() {
            this.time = 6 * 60 * 60 * 1000;
            this.speed = CONFIG.TIME_SPEED;
            this.paused = false;
        }

        update(deltaTime) {
            if (!this.paused) {
                this.time += deltaTime * this.speed * CONFIG.DAY_LENGTH;
                if (this.time >= CONFIG.DAY_LENGTH) {
                    this.time -= CONFIG.DAY_LENGTH;
                }
            }
        }

        getHours() {
            return Math.floor(this.time / (60 * 60 * 1000));
        }

        getMinutes() {
            return Math.floor((this.time % (60 * 60 * 1000)) / (60 * 1000));
        }

        getTimeString() {
            const hours = this.getHours().toString().padStart(2, '0');
            const minutes = this.getMinutes().toString().padStart(2, '0');
            return `${hours}:${minutes}`;
        }

        getPeriod() {
            const hours = this.getHours();
            if (hours >= 5 && hours < 8) return { name: '清晨', id: 'dawn' };
            if (hours >= 8 && hours < 12) return { name: '上午', id: 'morning' };
            if (hours >= 12 && hours < 14) return { name: '午间', id: 'noon' };
            if (hours >= 14 && hours < 17) return { name: '下午', id: 'afternoon' };
            if (hours >= 17 && hours < 20) return { name: '黄昏', id: 'evening' };
            if (hours >= 20 && hours < 23) return { name: '夜晚', id: 'night' };
            return { name: '深夜', id: 'midnight' };
        }

        getSkyGradient(ctx, width, height) {
            const hours = this.getHours() + this.getMinutes() / 60;
            let topColor, bottomColor;

            if (hours >= 5 && hours < 7) {
                topColor = '#1a1a3e';
                bottomColor = '#ff9966';
            } else if (hours >= 7 && hours < 10) {
                topColor = '#4a90b8';
                bottomColor = '#87ceeb';
            } else if (hours >= 10 && hours < 16) {
                topColor = '#5c9fd4';
                bottomColor = '#a8d8ea';
            } else if (hours >= 16 && hours < 18) {
                topColor = '#4a6fa5';
                bottomColor = '#ffb347';
            } else if (hours >= 18 && hours < 20) {
                topColor = '#2d1b4e';
                bottomColor = '#ff6b6b';
            } else if (hours >= 20 && hours < 22) {
                topColor = '#0f0f23';
                bottomColor = '#1a1a4e';
            } else {
                topColor = '#0a0a1a';
                bottomColor = '#151530';
            }

            const gradient = ctx.createLinearGradient(0, 0, 0, height);
            gradient.addColorStop(0, topColor);
            gradient.addColorStop(1, bottomColor);
            return gradient;
        }

        isNight() {
            const hours = this.getHours();
            return hours >= 20 || hours < 6;
        }

        getSunPosition(width, height) {
            const hours = this.getHours() + this.getMinutes() / 60;
            const dayProgress = (hours - 6) / 12;
            
            if (dayProgress < 0 || dayProgress > 1) {
                return null;
            }

            const x = width * 0.1 + width * 0.8 * dayProgress;
            const y = height * 0.8 - Math.sin(dayProgress * Math.PI) * height * 0.7;
            
            return { x, y };
        }

        getMoonPosition(width, height) {
            const hours = this.getHours() + this.getMinutes() / 60;
            const nightProgress = hours >= 20 ? (hours - 20) / 8 : (hours + 4) / 8;
            
            const x = width * 0.1 + width * 0.8 * nightProgress;
            const y = height * 0.7 - Math.sin(nightProgress * Math.PI) * height * 0.5;
            
            return { x, y };
        }
    }

    class Particle {
        constructor(x, y, type) {
            this.x = x;
            this.y = y;
            this.type = type;
            this.reset();
        }

        reset() {
            this.vx = (Math.random() - 0.5) * 2;
            this.vy = Math.random() * 0.5 + 0.3;
            this.size = Math.random() * 4 + 2;
            this.rotation = Math.random() * Math.PI * 2;
            this.rotationSpeed = (Math.random() - 0.5) * 0.05;
            this.opacity = Math.random() * 0.5 + 0.5;
            this.wobble = Math.random() * Math.PI * 2;
            this.wobbleSpeed = Math.random() * 0.05 + 0.02;
        }

        update(width, height) {
            this.wobble += this.wobbleSpeed;
            this.x += this.vx + Math.sin(this.wobble) * 0.5;
            this.y += this.vy;
            this.rotation += this.rotationSpeed;

            if (this.y > height + 20 || this.x < -20 || this.x > width + 20) {
                this.x = Math.random() * width;
                this.y = -20;
                this.reset();
            }
        }

        draw(ctx) {
            ctx.save();
            ctx.translate(this.x, this.y);
            ctx.rotate(this.rotation);
            ctx.globalAlpha = this.opacity;

            if (this.type === 'sakura') {
                this.drawSakura(ctx);
            } else if (this.type === 'leaf') {
                this.drawLeaf(ctx);
            } else if (this.type === 'dust') {
                this.drawDust(ctx);
            }

            ctx.restore();
        }

        drawSakura(ctx) {
            ctx.fillStyle = '#ffb7c5';
            for (let i = 0; i < 5; i++) {
                ctx.save();
                ctx.rotate((i * Math.PI * 2) / 5);
                ctx.beginPath();
                ctx.ellipse(0, -this.size * 0.8, this.size * 0.4, this.size, 0, 0, Math.PI * 2);
                ctx.fill();
                ctx.restore();
            }
            ctx.fillStyle = '#ffe4e9';
            ctx.beginPath();
            ctx.arc(0, 0, this.size * 0.3, 0, Math.PI * 2);
            ctx.fill();
        }

        drawLeaf(ctx) {
            ctx.fillStyle = `rgba(76, 175, 80, ${this.opacity})`;
            ctx.beginPath();
            ctx.ellipse(0, 0, this.size, this.size * 0.5, 0, 0, Math.PI * 2);
            ctx.fill();
        }

        drawDust(ctx) {
            ctx.fillStyle = `rgba(255, 255, 255, ${this.opacity * 0.5})`;
            ctx.beginPath();
            ctx.arc(0, 0, this.size * 0.5, 0, Math.PI * 2);
            ctx.fill();
        }
    }

    class ParticleSystem {
        constructor(width, height) {
            this.width = width;
            this.height = height;
            this.particles = [];
            this.active = true;
            this.type = 'sakura';

        init(count, type) {
            this.particles = [];
            this.type = type;
            
            for (let i = 0; i < count; i++) {

                this.particles.push(new Particle(x, y, type));

        }

        update() {
            if (!this.active) return;
            this.particles.forEach(p => p.update(this.width, this.height));
        }

        draw(ctx) {
            if (!this.active) return;
            this.particles.forEach(p => p.draw(ctx));
        }

        resize(width, height) {
            this.width = width;
            this.height = height;
        }
    }

                ctx.beginPath();
                ctx.arc(star.x, star.y, star.size, 0, Math.PI * 2);

            });

    class Character {
        constructor(id, data, position) {
            this.id = id;
            this.data = data;
            this.position = position;
            this.targetPosition = position;
            this.visible = true;
            this.opacity = 1;
            
            // 表情状态
            this.expression = ExpressionState.NORMAL;
            this.targetExpression = ExpressionState.NORMAL;
            this.transitionProgress = 1;
            this.transitionTime = 300;
            
            // 动画状态
            this.blinkTimer = Math.random() * 3000;
            this.isBlinking = false;
            this.blinkDuration = 150;
            this.blinkStartTime = 0;
            
            // 呼吸动画
            this.breathPhase = Math.random() * Math.PI * 2;
            this.breathSpeed = 0.002;
            
            // 脸红效果
            this.blushAmount = 0;
            this.targetBlush = 0;
            
            // 瞳孔动画
            this.pupilScale = 1;
            this.targetPupilScale = 1;
            
            // 进入/退出动画
            this.enterProgress = 1;
            this.isEntering = false;
            this.exitProgress = 0;
            this.isExiting = false;
        }

        setExpression(expression, immediate = false) {
            this.targetExpression = expression;
            if (immediate) {
                this.expression = expression;
                this.transitionProgress = 1;
            } else {
                this.transitionProgress = 0;
            }
            
            // 根据表情设置脸红程度
            switch(expression) {
                case ExpressionState.SHY:
                    this.targetBlush = 0.6;
                    break;
                case ExpressionState.LOVE:
                    this.targetBlush = 0.8;
                    break;
                case ExpressionState.HAPPY:
                    this.targetBlush = 0.2;
                    break;
                case ExpressionState.SAD:
                    this.targetPupilScale = 1.2;
                    this.targetBlush = 0;
                    break;
                case ExpressionState.SURPRISED:
                    this.targetPupilScale = 1.4;
                    this.targetBlush = 0;
                    break;
                case ExpressionState.ANGRY:
                    this.targetBlush = 0.3;
                    this.targetPupilScale = 0.8;
                    break;
                default:
                    this.targetBlush = 0;
                    this.targetPupilScale = 1;
            }
        }

        startEnter() {
            this.isEntering = true;
            this.enterProgress = 0;
            this.position = { x: this.targetPosition.x - 100, y: this.targetPosition.y };
        }

        startExit() {
            this.isExiting = true;
            this.exitProgress = 0;
        }

        update(deltaTime) {
            // 表情过渡
            if (this.transitionProgress < 1) {
                this.transitionProgress += deltaTime / this.transitionTime;
                if (this.transitionProgress >= 1) {
                    this.transitionProgress = 1;
                    this.expression = this.targetExpression;
                }
            }
            
            // 眨眼计时
            this.blinkTimer -= deltaTime;
            if (this.blinkTimer <= 0) {
                this.isBlinking = true;
                this.blinkStartTime = performance.now();
                this.blinkTimer = 2000 + Math.random() * 4000;
            }
            
            if (this.isBlinking && performance.now() - this.blinkStartTime > this.blinkDuration) {
                this.isBlinking = false;
            }
            
            // 呼吸动画
            this.breathPhase += this.breathSpeed * deltaTime;
            
            // 脸红渐变
            this.blushAmount += (this.targetBlush - this.blushAmount) * 0.05;
            
            // 瞳孔缩放
            this.pupilScale += (this.targetPupilScale - this.pupilScale) * 0.1;
            
            // 进入动画
            if (this.isEntering) {
                this.enterProgress += deltaTime * 0.003;
                if (this.enterProgress >= 1) {
                    this.enterProgress = 1;
                    this.isEntering = false;
                }
                const eased = this.easeOutCubic(this.enterProgress);
                this.position.x = this.targetPosition.x - 100 + 100 * eased;
                this.opacity = eased;
            }
            
            // 退出动画
            if (this.isExiting) {
                this.exitProgress += deltaTime * 0.003;
                if (this.exitProgress >= 1) {
                    this.exitProgress = 1;
                    this.isExiting = false;
                    this.visible = false;
                }
                const eased = this.easeInCubic(this.exitProgress);
                this.position.x = this.targetPosition.x + 100 * eased;
                this.opacity = 1 - eased;
            }
        }

        easeOutCubic(t) {
            return 1 - Math.pow(1 - t, 3);
        }

        easeInCubic(t) {
            return t * t * t;
        }

        draw(ctx) {
            if (!this.visible) return;
            
            ctx.save();
            ctx.globalAlpha = this.opacity;
            
            const breathOffset = Math.sin(this.breathPhase) * 3;
            const x = this.position.x;
            const y = this.position.y + breathOffset;
            
            // 绘制角色（简化版立绘）
            this.drawCharacter(ctx, x, y);
            
            ctx.restore();
        }

        drawCharacter(ctx, x, y) {
            const scale = 1.5;
            
            // 身体阴影
            ctx.fillStyle = 'rgba(0, 0, 0, 0.1)';
            ctx.beginPath();
            ctx.ellipse(x, y + 200 * scale, 60 * scale, 15 * scale, 0, 0, Math.PI * 2);
            ctx.fill();
            
            // 头发（后层）
            ctx.fillStyle = this.data.hairColor;
            ctx.beginPath();
            ctx.ellipse(x, y - 20 * scale, 75 * scale, 85 * scale, 0, 0, Math.PI * 2);
            ctx.fill();
            
            // 脸部
            ctx.fillStyle = this.data.skinColor;
            ctx.beginPath();
            ctx.ellipse(x, y, 50 * scale, 55 * scale, 0, 0, Math.PI * 2);
            ctx.fill();
            
            // 脸红效果
            if (this.blushAmount > 0) {
                ctx.fillStyle = `rgba(255, 150, 150, ${this.blushAmount * 0.5})`;
                ctx.beginPath();
                ctx.ellipse(x - 35 * scale, y + 15 * scale, 15 * scale, 8 * scale, 0, 0, Math.PI * 2);
                ctx.fill();
                ctx.beginPath();
                ctx.ellipse(x + 35 * scale, y + 15 * scale, 15 * scale, 8 * scale, 0, 0, Math.PI * 2);
                ctx.fill();
            }
            
            // 眼睛
            const eyeY = y - 5 * scale;
            const eyeSpacing = 25 * scale;
            
            // 眼白
            ctx.fillStyle = '#ffffff';
            ctx.beginPath();
            ctx.ellipse(x - eyeSpacing, eyeY, 15 * scale, this.isBlinking ? 2 * scale : 12 * scale, 0, 0, Math.PI * 2);
            ctx.fill();
            ctx.beginPath();
            ctx.ellipse(x + eyeSpacing, eyeY, 15 * scale, this.isBlinking ? 2 * scale : 12 * scale, 0, 0, Math.PI * 2);
            ctx.fill();
            
            if (!this.isBlinking) {
                // 瞳孔
                const pupilSize = 8 * scale * this.pupilScale;
                ctx.fillStyle = this.data.eyeColor;
                ctx.beginPath();
                ctx.arc(x - eyeSpacing, eyeY + 2 * scale, pupilSize, 0, Math.PI * 2);
                ctx.fill();
                ctx.beginPath();
                ctx.arc(x + eyeSpacing, eyeY + 2 * scale, pupilSize, 0, Math.PI * 2);
                ctx.fill();
                
                // 高光
                ctx.fillStyle = '#ffffff';
                ctx.beginPath();
                ctx.arc(x - eyeSpacing + 3 * scale, eyeY - 2 * scale, 3 * scale, 0, Math.PI * 2);
                ctx.fill();
                ctx.beginPath();
                ctx.arc(x + eyeSpacing + 3 * scale, eyeY - 2 * scale, 3 * scale, 0, Math.PI * 2);
                ctx.fill();
                
                // 表情特殊绘制
                this.drawExpression(ctx, x, y, scale);
            }
            
            // 嘴巴
            this.drawMouth(ctx, x, y + 25 * scale, scale);
            
            // 头发（前层/刘海）
            ctx.fillStyle = this.data.hairColor;
            this.drawBangs(ctx, x, y - 50 * scale, scale);
            
            // 发饰
            ctx.fillStyle = this.data.accessoryColor;
            ctx.beginPath();
            ctx.arc(x + 45 * scale, y - 40 * scale, 8 * scale, 0, Math.PI * 2);
            ctx.fill();
        }

        drawExpression(ctx, x, y, scale) {
            switch(this.expression) {
                case ExpressionState.HAPPY:
                case ExpressionState.LOVE:
                    // 开心的弯眼睛
                    ctx.strokeStyle = this.data.eyeColor;
                    ctx.lineWidth = 3;
                    ctx.beginPath();
                    ctx.arc(x - 25 * scale, y - 3 * scale, 10 * scale, 0.2 * Math.PI, 0.8 * Math.PI);
                    ctx.stroke();
                    ctx.beginPath();
                    ctx.arc(x + 25 * scale, y - 3 * scale, 10 * scale, 0.2 * Math.PI, 0.8 * Math.PI);
                    ctx.stroke();
                    break;
                case ExpressionState.SAD:
                    // 悲伤的眼睛（下垂）
                    ctx.strokeStyle = this.data.eyeColor;
                    ctx.lineWidth = 2;
                    ctx.beginPath();
                    ctx.moveTo(x - 35 * scale, y - 10 * scale);
                    ctx.lineTo(x - 15 * scale, y - 5 * scale);
                    ctx.stroke();
                    ctx.beginPath();
                    ctx.moveTo(x + 35 * scale, y - 10 * scale);
                    ctx.lineTo(x + 15 * scale, y - 5 * scale);
                    ctx.stroke();
                    break;
            }
        }

        drawMouth(ctx, x, y, scale) {
            ctx.fillStyle = '#d4738a';
            
            switch(this.expression) {
                case ExpressionState.HAPPY:
                case ExpressionState.LOVE:
                    ctx.beginPath();
                    ctx.arc(x, y, 8 * scale, 0, Math.PI);
                    ctx.fill();
                    break;
                case ExpressionState.SHY:
                    ctx.beginPath();
                    ctx.ellipse(x, y, 5 * scale, 3 * scale, 0, 0, Math.PI * 2);
                    ctx.fill();
                    break;
                case ExpressionState.SAD:
                    ctx.beginPath();
                    ctx.arc(x, y + 10 * scale, 8 * scale, Math.PI, 2 * Math.PI);
                    ctx.fill();
                    break;
                case ExpressionState.SURPRISED:
                    ctx.beginPath();
                    ctx.ellipse(x, y, 5 * scale, 8 * scale, 0, 0, Math.PI * 2);
                    ctx.fill();
                    break;
                default:
                    ctx.beginPath();
                    ctx.ellipse(x, y, 6 * scale, 3 * scale, 0, 0, Math.PI * 2);
                    ctx.fill();
            }
        }

        drawBangs(ctx, x, y, scale) {
            ctx.fillStyle = this.data.hairColor;
            
            // 简化的刘海
            for (let i = -3; i <= 3; i++) {
                ctx.beginPath();
                ctx.moveTo(x + i * 15 * scale, y - 30 * scale);
                ctx.quadraticCurveTo(
                    x + i * 15 * scale, 
                    y + 20 * scale, 
                    x + (i + 0.5) * 10 * scale, 
                    y + 25 * scale
                );
                ctx.quadraticCurveTo(
                    x + (i + 1) * 15 * scale, 
                    y + 20 * scale, 
                    x + (i + 1) * 15 * scale, 
                    y - 30 * scale
                );
                ctx.fill();
            }
        }
    }

    class SceneRenderer {
        constructor(width, height) {
            this.width = width;
            this.height = height;
            this.currentScene = 'school';
        }

        draw(ctx, dayNight) {
            // 绘制天空
            ctx.fillStyle = dayNight.getSkyGradient(ctx, this.width, this.height);
            ctx.fillRect(0, 0, this.width, this.height);
            
            // 绘制星星（夜晚）
            if (dayNight.isNight()) {
                // 星空在Game类中处理
            }
            
            // 绘制太阳/月亮
            this.drawCelestialBody(ctx, dayNight);
            
            // 绘制场景
            switch(this.currentScene) {
                case 'school':
                    this.drawSchoolScene(ctx);
                    break;
                case 'garden':
                    this.drawGardenScene(ctx);
                    break;
                case 'rooftop':
                    this.drawRooftopScene(ctx);
                    break;
                case 'library':
                    this.drawLibraryScene(ctx);
                    break;
            }
        }

        drawCelestialBody(ctx, dayNight) {
            if (dayNight.isNight()) {
                const moonPos = dayNight.getMoonPosition(this.width, this.height);
                if (moonPos) {
                    // 月亮光晕
                    const glowGradient = ctx.createRadialGradient(
                        moonPos.x, moonPos.y, 0,
                        moonPos.x, moonPos.y, 80
                    );
                    glowGradient.addColorStop(0, 'rgba(255, 255, 255, 0.3)');
                    glowGradient.addColorStop(1, 'rgba(255, 255, 255, 0)');
                    ctx.fillStyle = glowGradient;
                    ctx.beginPath();
                    ctx.arc(moonPos.x, moonPos.y, 80, 0, Math.PI * 2);
                    ctx.fill();
                    
                    // 月亮本体
                    ctx.fillStyle = '#f5f5dc';
                    ctx.beginPath();
                    ctx.arc(moonPos.x, moonPos.y, 30, 0, Math.PI * 2);
                    ctx.fill();
                }
            } else {
                const sunPos = dayNight.getSunPosition(this.width, this.height);
                if (sunPos) {
                    // 太阳光晕
                    const glowGradient = ctx.createRadialGradient(
                        sunPos.x, sunPos.y, 0,
                        sunPos.x, sunPos.y, 100
                    );
                    glowGradient.addColorStop(0, 'rgba(255, 200, 100, 0.4)');
                    glowGradient.addColorStop(1, 'rgba(255, 200, 100, 0)');
                    ctx.fillStyle = glowGradient;
                    ctx.beginPath();
                    ctx.arc(sunPos.x, sunPos.y, 100, 0, Math.PI * 2);
                    ctx.fill();
                    
                    // 太阳本体
                    ctx.fillStyle = '#ffd700';
                    ctx.beginPath();
                    ctx.arc(sunPos.x, sunPos.y, 35, 0, Math.PI * 2);
                    ctx.fill();
                }
            }
        }

    class Game {
        constructor() {
            this.canvas = document.getElementById('gameCanvas');
            this.ctx = this.canvas.getContext('2d');
            
            this.canvas.width = CONFIG.CANVAS_WIDTH;
            this.canvas.height = CONFIG.CANVAS_HEIGHT;
            
            this.lastTime = 0;
            this.running = true;
            
            // 游戏状态
            this.gameState = {
                affection: { ...ScriptData.initialAffection },
                stats: {
                    energy: 80,
                    mood: 70
                },
                flags: {}
            };
            
            // 初始化系统
            this.dayNight = new DayNightCycle();
            this.starField = new StarField(this.canvas.width, this.canvas.height, CONFIG.STAR_COUNT);
            this.particleSystem = new ParticleSystem(this.canvas.width, this.canvas.height);
            this.sceneRenderer = new SceneRenderer(this.canvas.width, this.canvas.height);
            
            // 角色系统
            this.characters = {};
            this.activeCharacter = null;
            this.initCharacters();
            
            // 对话引擎
            this.dialogEngine = new DialogEngine(this);
            
            // 粒子类型
            this.particleSystem.init(CONFIG.PARTICLE_COUNT, 'sakura');
            
            // 绑定事件
            this.bindEvents();
            
            // 更新UI
            this.updateStatBars();
            
            // 开始剧情
            setTimeout(() => {
                this.dialogEngine.startScript('start');
            }, 500);
            
            // 开始游戏循环
            requestAnimationFrame((time) => this.gameLoop(time));
        }

        initCharacters() {
            const positions = {
                sakura: { x: 600, y: 400 },
                yuki: { x: 550, y: 400 },
                haru: { x: 650, y: 400 }
            };

            Object.keys(CharacterData).forEach(id => {
                this.characters[id] = new Character(
                    id, 
                    CharacterData[id], 
                    positions[id]
                );
                this.characters[id].visible = false;
            });
        }

        bindEvents() {
            // 场景切换按钮
            document.querySelectorAll('.nav-btn').forEach(btn => {
                btn.addEventListener('click', () => {
                    document.querySelectorAll('.nav-btn').forEach(b => b.classList.remove('active'));
                    btn.classList.add('active');
                    this.changeScene(btn.dataset.scene);
                });
            });
        }

        showCharacter(id, expression = 'normal') {
            if (this.activeCharacter && this.activeCharacter !== id) {
                this.characters[this.activeCharacter].visible = false;
            }
            
            if (this.characters[id]) {
                this.characters[id].visible = true;
                this.characters[id].setExpression(expression, true);
                this.activeCharacter = id;
                this.updateCharacterInfoPanel(id);
            }
        }

        hideAllCharacters() {
            Object.values(this.characters).forEach(char => {
                char.visible = false;
            });
            this.activeCharacter = null;
            document.getElementById('characterInfoPanel').classList.remove('active');
        }

        characterEnter(id) {
            if (this.characters[id]) {
                this.characters[id].startEnter();
            }
        }

        updateCharacterInfoPanel(id) {
            const panel = document.getElementById('characterInfoPanel');
            const data = CharacterData[id];
            const affection = this.gameState.affection[id] || 0;
            
            document.getElementById('infoPanelName').textContent = data.name;
            
            // 更新心形图标
            const heartIcons = document.getElementById('heartIcons');
            heartIcons.innerHTML = '';
            const heartCount = Math.floor(affection / 20);
            
            for (let i = 0; i < 5; i++) {
                const heart = document.createElementNS('http://www.w3.org/2000/svg', 'svg');
                heart.setAttribute('viewBox', '0 0 24 24');
                heart.classList.add('heart-icon');
                if (i < heartCount) {
                    heart.classList.add('filled');
                }
                heart.innerHTML = '<path d="M12 21.35l-1.45-1.32C5.4 15.36 2 12.28 2 8.5 2 5.42 4.42 3 7.5 3c1.74 0 3.41.81 4.5 2.09C13.09 3.81 14.76 3 16.5 3 19.58 3 22 5.42 22 8.5c0 3.78-3.4 6.86-8.55 11.54L12 21.35z"/>';
                heartIcons.appendChild(heart);
            }
            
            panel.classList.add('active');
        }

        changeAffection(character, amount) {
            if (!this.gameState.affection[character]) {
                this.gameState.affection[character] = 0;
            }
            
            this.gameState.affection[character] += amount;
            this.gameState.affection[character] = Math.max(0, Math.min(100, this.gameState.affection[character]));
            
            // 显示通知
            if (amount !== 0) {
                this.showNotification(
                    `${amount > 0 ? '+' : ''}${amount} ${CharacterData[character].name} 好感度`,
                    amount > 0
                );
            }
            
            // 更新UI
            this.updateStatBars();
            
            if (this.activeCharacter === character) {
                this.updateCharacterInfoPanel(character);
            }
        }

        changeStat(stat, amount) {
            if (this.gameState.stats[stat] !== undefined) {
                this.gameState.stats[stat] += amount;
                this.gameState.stats[stat] = Math.max(0, Math.min(100, this.gameState.stats[stat]));
                this.updateStatBars();
            }
        }

        updateStatBars() {
            // 更新好感度条（显示当前角色的好感度）
            if (this.activeCharacter) {
                const lovePercent = this.gameState.affection[this.activeCharacter] || 0;
                document.getElementById('loveBar').style.width = `${lovePercent}%`;
            }
            
            document.getElementById('energyBar').style.width = `${this.gameState.stats.energy}%`;
            document.getElementById('moodBar').style.width = `${this.gameState.stats.mood}%`;
        }

        showNotification(message, isPositive = true) {
            const notification = document.getElementById('notification');
            notification.innerHTML = `<span class="notification-heart">♥</span>${message}`;
            notification.style.borderColor = isPositive ? 'rgba(255, 107, 138, 0.5)' : 'rgba(100, 100, 100, 0.5)';
            notification.classList.add('show');
            
            setTimeout(() => {
                notification.classList.remove('show');
            }, 2000);
        }

        changeScene(sceneName) {
            this.sceneRenderer.setScene(sceneName);
            
            // 根据场景更改粒子类型
            switch(sceneName) {
                case 'garden':
                    this.particleSystem.init(CONFIG.PARTICLE_COUNT, 'sakura');
                    break;
                case 'school':
                    this.particleSystem.init(30, 'leaf');
                    break;
                default:
                    this.particleSystem.init(20, 'dust');
            }
        }

        transition(type, duration, callback) {
            // 简单的淡入淡出过渡
            const overlay = document.createElement('div');
            overlay.style.cssText = `
                position: absolute;
                top: 0;
                left: 0;
                right: 0;
                bottom: 0;
                background: black;
                opacity: 0;
                transition: opacity ${duration / 2}ms ease;
                z-index: 1000;
                pointer-events: none;
            `;
            
            document.querySelector('.game-container').appendChild(overlay);
            
            requestAnimationFrame(() => {
                overlay.style.opacity = '1';
            });
            
            setTimeout(() => {
                if (callback) callback();
                overlay.style.opacity = '0';
                setTimeout(() => overlay.remove(), duration / 2);
            }, duration / 2);
        }

        showEnding(message) {
            this.hideAllCharacters();
            
            const overlay = document.createElement('div');
            overlay.style.cssText = `
                position: absolute;
                top: 0;
                left: 0;
                right: 0;
                bottom: 0;
                background: rgba(0, 0, 0, 0.8);
                display: flex;
                align-items: center;
                justify-content: center;
                flex-direction: column;
                z-index: 1000;
                opacity: 0;
                transition: opacity 1s ease;
            `;
            
            overlay.innerHTML = `
                <div style="color: #ffb7c5; font-size: 32px; margin-bottom: 20px;">${message}</div>
                <button style="
                    background: linear-gradient(135deg, rgba(255, 183, 197, 0.3), rgba(255, 215, 0, 0.2));
                    border: 2px solid #ffb7c5;
                    border-radius: 10px;
                    padding: 15px 30px;
                    color: #f0e6d3;
                    font-size: 16px;
                    cursor: pointer;
                " onclick="location.reload()">重新开始</button>
            `;
            
            document.querySelector('.game-container').appendChild(overlay);
            
            requestAnimationFrame(() => {
                overlay.style.opacity = '1';
            });
        }

        gameLoop(currentTime) {
            if (!this.running) return;
            
            const deltaTime = currentTime - this.lastTime;
            this.lastTime = currentTime;
            
            this.update(deltaTime, currentTime);
            this.render();
            
            requestAnimationFrame((time) => this.gameLoop(time));
        }

        update(deltaTime, currentTime) {
            // 更新昼夜循环
            this.dayNight.update(deltaTime);
            
            // 更新时间显示
            document.getElementById('timeDisplay').textContent = this.dayNight.getTimeString();
            document.getElementById('periodDisplay').textContent = this.dayNight.getPeriod().name;
            
            // 更新星空
            this.starField.update(deltaTime);
            
            // 更新粒子
            this.particleSystem.update();
            
            // 更新角色
            Object.values(this.characters).forEach(char => {
                if (char.visible) {
                    char.update(deltaTime);
                }
            });
            
            // 更新对话引擎
            this.dialogEngine.update(currentTime);
        }

        render() {
            const ctx = this.ctx;
            
            // 清空画布
            ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);
            
            // 绘制场景
            this.sceneRenderer.draw(ctx, this.dayNight);
            
            // 绘制星空（夜晚）
            if (this.dayNight.isNight()) {
                this.starField.draw(ctx);
            }
            
            // 绘制粒子
            this.particleSystem.draw(ctx);
            
            // 绘制角色
            Object.values(this.characters).forEach(char => {
                if (char.visible) {
                    char.draw(ctx);
                }
            });
        }
    }

    // 启动游戏
    window.addEventListener('DOMContentLoaded', () => {
        new Game();
    });
    </script>
</body>

## 代码说明

### 剧情对话引擎核心功能

1. **打字机效果**
   - `TypewriterEffect` 类实现了逐字显示文本的效果
   - 支持跳过功能（点击或按空格键）
   - 带有闪烁光标动画

2. **对话框UI**
   - 美观的玻璃态对话框设计
   - 显示角色名称（带粉色标签）
   - 点击继续提示动画

3. **选项分支系统**
   - 支持多选项分支
   - 每个选项可配置效果（好感度变化、状态标志等）
   - 条件判断支持（基于好感度限制选项）

4. **剧本数据结构**
   - JSON格式的节点系统
   - 支持类型：dialogue、choice、scene、transition、end
   - 可扩展的条件检查系统

5. **角色联动**
   - 对话时自动显示对应角色立绘
   - 根据对话内容自动切换表情
   - 支持角色进场动画
