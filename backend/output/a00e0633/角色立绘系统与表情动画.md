    // ============================================
    // 昼夜循环系统

    class DayNightCycle {
        constructor() {
            this.time = 6 * 60 * 60 * 1000;
            this.speed = CONFIG.TIME_SPEED;
            this.paused = false;
        }

        update(deltaTime) {
            if (!this.paused) {
                this.time += deltaTime * this.speed * CONFIG.DAY_LENGTH;
                if (this.time >= CONFIG.DAY_LENGTH) {
                    this.time -= CONFIG.DAY_LENGTH;
                }
            }
        }

        getHours() {
            return Math.floor(this.time / (60 * 60 * 1000));
        }

        getMinutes() {
            return Math.floor((this.time % (60 * 60 * 1000)) / (60 * 1000));
        }

        getTimeString() {
            const hours = this.getHours().toString().padStart(2, '0');
            const minutes = this.getMinutes().toString().padStart(2, '0');
            return `${hours}:${minutes}`;
        }

        getPeriod() {
            const hours = this.getHours();
            if (hours >= 5 && hours < 8) return { name: '清晨', id: 'dawn' };
            if (hours >= 8 && hours < 12) return { name: '上午', id: 'morning' };
            if (hours >= 12 && hours < 14) return { name: '午间', id: 'noon' };
            if (hours >= 14 && hours < 17) return { name: '下午', id: 'afternoon' };
            if (hours >= 17 && hours < 20) return { name: '黄昏', id: 'evening' };
            if (hours >= 20 && hours < 23) return { name: '夜晚', id: 'night' };
            return { name: '深夜', id: 'midnight' };
        }

        getSkyGradient(ctx, width, height) {
            const hours = this.getHours() + this.getMinutes() / 60;
            let topColor, bottomColor;

            if (hours >= 5 && hours < 7) {
                topColor = '#1a1a3e';
                bottomColor = '#ff9966';
            } else if (hours >= 7 && hours < 10) {
                topColor = '#4a90b8';
                bottomColor = '#87ceeb';
            } else if (hours >= 10 && hours < 16) {
                topColor = '#5c9fd4';
                bottomColor = '#a8d8ea';
            } else if (hours >= 16 && hours < 18) {
                topColor = '#4a6fa5';
                bottomColor = '#ffb347';
            } else if (hours >= 18 && hours < 20) {
                topColor = '#2d1b4e';
                bottomColor = '#ff6b6b';
            } else if (hours >= 20 && hours < 22) {
                topColor = '#0f0f23';
                bottomColor = '#1a1a4e';
            } else {
                topColor = '#0a0a1a';
                bottomColor = '#151530';
            }

            const gradient = ctx.createLinearGradient(0, 0, 0, height);
            gradient.addColorStop(0, topColor);
            gradient.addColorStop(1, bottomColor);
            return gradient;
        }

        isNight() {
            const hours = this.getHours();
            return hours >= 20 || hours < 6;
        }

        getSunPosition(width, height) {
            const hours = this.getHours() + this.getMinutes() / 60;
            const dayProgress = (hours - 6) / 12;
            
            if (dayProgress < 0 || dayProgress > 1) {
                return null;
            }

            const x = width * 0.1 + width * 0.8 * dayProgress;
            const y = height * 0.8 - Math.sin(dayProgress * Math.PI) * height * 0.7;
            
            return { x, y };
        }

        getMoonPosition(width, height) {
            const hours = this.getHours() + this.getMinutes() / 60;
            const nightProgress = hours >= 20 ? (hours - 20) / 8 : (hours + 4) / 8;
            
            const x = width * 0.1 + width * 0.8 * nightProgress;
            const y = height * 0.7 - Math.sin(nightProgress * Math.PI) * height * 0.5;
            
            return { x, y };
        }
    }

    // 粒子系统

    class Particle {
        constructor(x, y, type) {
            this.x = x;
            this.y = y;
            this.type = type;
            this.reset();
        }

        reset() {
            this.vx = (Math.random() - 0.5) * 2;
            this.vy = Math.random() * 0.5 + 0.3;
            this.size = Math.random() * 4 + 2;
            this.rotation = Math.random() * Math.PI * 2;
            this.rotationSpeed = (Math.random() - 0.5) * 0.05;
            this.opacity = Math.random() * 0.5 + 0.5;
            this.wobble = Math.random() * Math.PI * 2;
            this.wobbleSpeed = Math.random() * 0.05 + 0.02;
        }

        update(width, height) {
            this.wobble += this.wobbleSpeed;
            this.x += this.vx + Math.sin(this.wobble) * 0.5;
            this.y += this.vy;
            this.rotation += this.rotationSpeed;

            if (this.y > height + 20 || this.x < -20 || this.x > width + 20) {
                this.x = Math.random() * width;
                this.y = -20;
                this.reset();
            }
        }

        draw(ctx) {
            ctx.save();
            ctx.translate(this.x, this.y);
            ctx.rotate(this.rotation);
            ctx.globalAlpha = this.opacity;

            if (this.type === 'sakura') {
                this.drawSakura(ctx);
            } else if (this.type === 'leaf') {
                this.drawLeaf(ctx);
            } else if (this.type === 'dust') {
                this.drawDust(ctx);
            }

            ctx.restore();
        }

        drawSakura(ctx) {
            ctx.fillStyle = '#ffb7c5';
            for (let i = 0; i < 5; i++) {
                ctx.save();
                ctx.rotate((i * Math.PI * 2) / 5);
                ctx.beginPath();
                ctx.ellipse(0, -this.size * 0.8, this.size * 0.4, this.size, 0, 0, Math.PI * 2);
                ctx.fill();
                ctx.restore();
            }
            ctx.fillStyle = '#ffe4e9';
            ctx.beginPath();
            ctx.arc(0, 0, this.size * 0.3, 0, Math.PI * 2);
            ctx.fill();
        }

        drawLeaf(ctx) {
            ctx.fillStyle = `rgba(76, 175, 80, ${this.opacity})`;
            ctx.beginPath();
            ctx.ellipse(0, 0, this.size, this.size * 0.5, 0, 0, Math.PI * 2);
            ctx.fill();
        }

        drawDust(ctx) {
            ctx.fillStyle = `rgba(255, 255, 255, ${this.opacity * 0.5})`;
            ctx.beginPath();
            ctx.arc(0, 0, this.size * 0.5, 0, Math.PI * 2);
            ctx.fill();
        }
    }

    class ParticleSystem {
        constructor(width, height) {
            this.width = width;
            this.height = height;
            this.particles = [];
            this.active = true;
            this.type = 'sakura';
        }

        init(count, type) {
            this.particles = [];
            this.type = type;
            
            for (let i = 0; i < count; i++) {
                const x = Math.random() * this.width;
                const y = Math.random() * this.height;
                this.particles.push(new Particle(x, y, type));
            }
        }

        update() {
            if (!this.active) return;
            this.particles.forEach(p => p.update(this.width, this.height));
        }

        draw(ctx) {
            if (!this.active) return;
            this.particles.forEach(p => p.draw(ctx));
        }

        resize(width, height) {
            this.width = width;
            this.height = height;
        }
    }

    class StarField {
        constructor(width, height, count) {
            this.width = width;
            this.height = height;
            this.stars = [];
            this.init(count);
        }

        init(count) {
            this.stars = [];
            for (let i = 0; i < count; i++) {
                this.stars.push({
                    x: Math.random() * this.width,
                    y: Math.random() * this.height * 0.6,
                    size: Math.random() * 2 + 0.5,
                    brightness: Math.random(),
                    twinkleSpeed: Math.random() * 0.02 + 0.01,
                    twinklePhase: Math.random() * Math.PI * 2
                });
            }
        }

        update(deltaTime) {
            this.stars.forEach(star => {
                star.twinklePhase += star.twinkleSpeed;
            });
        }

        draw(ctx) {
            this.stars.forEach(star => {
                const brightness = 0.3 + Math.sin(star.twinklePhase) * 0.4 + 0.3;
                ctx.fillStyle = `rgba(255, 255, 255, ${brightness})`;
                ctx.beginPath();
                ctx.arc(star.x, star.y, star.size, 0, Math.PI * 2);
                ctx.fill();
            });
        }
    }

            this.mouthOpenness = 0.3;
            this.targetMouthOpenness = 0.3;
        }

        show() {
            this.visible = true;
            this.slideOffset = -100;
            this.fadeAlpha = 0;
        }

        hide() {
            this.visible = false;
        }

        setExpression(expression, transitionTime = 300) {
            if (this.expression === expression) return;
            
            this.targetExpression = expression;
            this.transitionProgress = 0;
            this.transitionTime = transitionTime;
            
            // 根据表情设置相关参数
            this.applyExpressionParams(expression);
        }

        applyExpressionParams(expression) {
            switch (expression) {
                case ExpressionState.HAPPY:
                    this.targetBlush = 0.2;
                    this.targetPupilScale = 1.1;
                    this.targetMouthOpenness = 0.5;
                    this.targetEyebrowOffset = -3;
                    break;
                case ExpressionState.SHY:
                    this.targetBlush = 0.7;
                    this.targetPupilScale = 0.9;
                    this.targetMouthOpenness = 0.2;
                    this.targetEyebrowOffset = 2;
                    this.targetHeadTilt = 10;
                    break;
                case ExpressionState.SAD:
                    this.targetBlush = 0;
                    this.targetPupilScale = 0.95;
                    this.targetMouthOpenness = 0.2;
                    this.targetEyebrowOffset = 5;
                    this.targetHeadTilt = -5;
                    break;
                case ExpressionState.SURPRISED:
                    this.targetBlush = 0.1;
                    this.targetPupilScale = 1.4;
                    this.targetMouthOpenness = 0.7;
                    this.targetEyebrowOffset = -5;
                    break;
                case ExpressionState.LOVE:
                    this.targetBlush = 0.5;
                    this.targetPupilScale = 1.2;
                    this.targetMouthOpenness = 0.3;
                    this.targetEyebrowOffset = -2;
                    this.targetHeadTilt = 5;
                    break;
                case ExpressionState.ANGRY:
                    this.targetBlush = 0.3;
                    this.targetPupilScale = 0.85;
                    this.targetMouthOpenness = 0.3;
                    this.targetEyebrowOffset = 8;
                    break;
                default:
                    this.targetBlush = 0;
                    this.targetPupilScale = 1;
                    this.targetMouthOpenness = 0.3;
                    this.targetEyebrowOffset = 0;
                    this.targetHeadTilt = 0;
            }
        }

        update(deltaTime) {
            if (!this.visible) return;
            
            // 出场动画
            if (this.slideOffset !== this.targetSlideOffset) {
                this.slideOffset += (this.targetSlideOffset - this.slideOffset) * 0.1;
                if (Math.abs(this.slideOffset - this.targetSlideOffset) < 0.1) {
                    this.slideOffset = this.targetSlideOffset;
                }
            }
            if (this.fadeAlpha < 1) {
                this.fadeAlpha += deltaTime * 0.003;
                if (this.fadeAlpha > 1) this.fadeAlpha = 1;
            }
            
            // 眨眼计时
            this.blinkTimer -= deltaTime;
            if (this.blinkTimer <= 0) {
                this.isBlinking = true;
                this.blinkState = 1;
                this.blinkTimer = 2500 + Math.random() * 3000;
            }
            
            // 眨眼动画
            if (this.isBlinking) {
                this.blinkState += deltaTime / (this.blinkDuration / 2);
                if (this.blinkState >= 4) {
                    this.isBlinking = false;
                    this.blinkState = 0;
                }
            }
            
            // 呼吸动画
            this.breathePhase += this.breatheSpeed * deltaTime;
            
            // 表情过渡
            if (this.transitionProgress < 1) {
                this.transitionProgress += deltaTime / this.transitionTime;
                if (this.transitionProgress >= 1) {
                    this.transitionProgress = 1;
                    this.expression = this.targetExpression;
                }
            }
            
            // 平滑过渡各种参数
            this.smoothTransition('blushIntensity', this.targetBlush, deltaTime, this.blushTransitionTime);
            this.smoothTransition('pupilScale', this.targetPupilScale, deltaTime, 200);
            this.smoothTransition('headTilt', this.targetHeadTilt, deltaTime, 400);
            this.smoothTransition('eyebrowOffset', this.targetEyebrowOffset, deltaTime, 300);
            this.smoothTransition('mouthOpenness', this.targetMouthOpenness, deltaTime, 200);
        }

        smoothTransition(property, target, deltaTime, transitionTime) {
            const diff = target - this[property];
            if (Math.abs(diff) > 0.01) {
                this[property] += diff * Math.min(1, deltaTime / transitionTime);
            } else {
                this[property] = target;
            }
        }

        getEyeOpenness() {
            if (!this.isBlinking) return 1;
            if (this.blinkState < 1) return 1;
            if (this.blinkState < 2) return 1 - (this.blinkState - 1);
            if (this.blinkState < 3) return 0;
            return this.blinkState - 3;
        }

        draw(ctx, canvasWidth, canvasHeight) {
            if (!this.visible) return;
            
            const baseX = this.position === 'left' ? canvasWidth * 0.3 :
                         this.position === 'right' ? canvasWidth * 0.7 :
                         canvasWidth * 0.5;
            const baseY = canvasHeight * 0.55;
            const scale = 1.2;
            
            // 呼吸偏移
            const breatheY = Math.sin(this.breathePhase) * this.breatheAmount;
            
            ctx.save();
            ctx.globalAlpha = this.fadeAlpha;
            ctx.translate(baseX + this.slideOffset, baseY + breatheY);
            ctx.scale(scale, scale);
            
            // 头部旋转
            ctx.save();
            ctx.rotate(this.headTilt * Math.PI / 180);
            
            // 绘制角色各部分
            this.drawHairBack(ctx);
            this.drawBody(ctx);
            this.drawNeck(ctx);
            this.drawHead(ctx);
            this.drawFace(ctx);
            this.drawHairFront(ctx);
            this.drawAccessories(ctx);
            
            ctx.restore();
            ctx.restore();
        }

        drawEye(ctx, x, y, openness) {
            const eyeHeight = 18 * Math.max(0.1, openness);
            
            // 眼白
            ctx.fillStyle = '#ffffff';
            ctx.beginPath();

            ctx.beginPath();
            ctx.ellipse(x, y + eyeHeight * 0.3, 12, eyeHeight * 0.4, 0, 0, Math.PI);
            ctx.fill();
            
            if (openness > 0.2) {
                // 虹膜
                const irisGradient = ctx.createRadialGradient(x, y, 0, x, y, 10);
                irisGradient.addColorStop(0, this.lightenColor(this.data.eyeColor, 20));
                irisGradient.addColorStop(0.7, this.data.eyeColor);
                irisGradient.addColorStop(1, this.darkenColor(this.data.eyeColor, 20));

                ctx.beginPath();
                ctx.arc(x, y, 10 * this.pupilScale, 0, Math.PI * 2);

                ctx.fillStyle = '#1a1a2e';
                ctx.beginPath();
                ctx.arc(x, y, 5 * this.pupilScale, 0, Math.PI * 2);
                ctx.fill();

                ctx.fillStyle = '#ffffff';
                ctx.beginPath();
                ctx.ellipse(x - 4, y - 4, 4, 3, -0.5, 0, Math.PI * 2);

                ctx.beginPath();
                ctx.arc(x + 3, y + 3, 2, 0, Math.PI * 2);

                    this.drawHeartPupil(ctx, x, y);
                }
            }

            ctx.fillStyle = this.data.hairColor;
            ctx.beginPath();
            ctx.ellipse(x, y - eyeHeight + 2, 15, 5, 0, 0, Math.PI * 2);

            ctx.fillStyle = this.darkenColor(this.data.hairColor, 20);
            for (let i = -2; i <= 2; i++) {
                ctx.beginPath();
                ctx.ellipse(x + i * 5, y - eyeHeight, 2, 4, i * 0.1, 0, Math.PI * 2);
                ctx.fill();
            }
        }

        drawHeartPupil(ctx, x, y) {
            const progress = this.transitionProgress;
            if (progress <= 0) return;
            
            ctx.save();
            ctx.globalAlpha = progress * 0.8;
            ctx.fillStyle = '#ff6b8a';
            
            const size = 3 * this.pupilScale;
            
            // 小心形在瞳孔内
            ctx.beginPath();
            ctx.moveTo(x, y + size * 0.3);
            ctx.bezierCurveTo(x, y - size * 0.3, x - size, y - size * 0.3, x - size, y + size * 0.1);
            ctx.bezierCurveTo(x - size, y + size * 0.5, x, y + size, x, y + size);
            ctx.bezierCurveTo(x, y + size, x + size, y + size * 0.5, x + size, y + size * 0.1);
            ctx.bezierCurveTo(x + size, y - size * 0.3, x, y - size * 0.3, x, y + size * 0.3);
            ctx.fill();
            
            ctx.restore();
        }

        drawMouth(ctx) {

                    ctx.beginPath();
                    ctx.moveTo(-10, 55);

        drawBlush(ctx) {

        darkenColor(hex, percent) {
            const num = parseInt(hex.replace('#', ''), 16);
            const amt = Math.round(2.55 * percent);
            const R = Math.max(0, (num >> 16) - amt);
            const G = Math.max(0, ((num >> 8) & 0x00FF) - amt);
            const B = Math.max(0, (num & 0x0000FF) - amt);
            return `#${(0x1000000 + R * 0x10000 + G * 0x100 + B).toString(16).slice(1)}`;
        }

        lightenColor(hex, percent) {
            const num = parseInt(hex.replace('#', ''), 16);
            const amt = Math.round(2.55 * percent);
            const R = Math.min(255, (num >> 16) + amt);
            const G = Math.min(255, ((num >> 8) & 0x00FF) + amt);
            const B = Math.min(255, (num & 0x0000FF) + amt);
            return `#${(0x1000000 + R * 0x10000 + G * 0x100 + B).toString(16).slice(1)}`;
        }
    }

    class CharacterManager {
        constructor() {
            this.characters = new Map();
            this.activeCharacter = null;
        }

        createCharacter(id, position = 'center') {
            const data = CharacterData[id];
            if (!data) {
                console.warn(`角色数据不存在: ${id}`);
                return null;
            }
            
            const character = new Character(id, data, position);
            this.characters.set(id, character);
            return character;
        }

        getCharacter(id) {
            return this.characters.get(id);
        }

        showCharacter(id, position = 'center') {
            let character = this.characters.get(id);
            if (!character) {
                character = this.createCharacter(id, position);
            }
            
            if (character) {
                character.position = position;
                character.show();
                this.activeCharacter = character;
            }
            
            return character;
        }

        hideCharacter(id) {
            const character = this.characters.get(id);
            if (character) {
                character.hide();
            }
        }

        setExpression(id, expression) {
            const character = this.characters.get(id);
            if (character) {
                character.setExpression(expression);
            }
        }

        update(deltaTime) {
            this.characters.forEach(char => char.update(deltaTime));
        }

        draw(ctx, width, height) {
            this.characters.forEach(char => {
                if (char.visible) {
                    char.draw(ctx, width, height);
                }
            });
        }
    }

    class SceneRenderer {
        constructor(ctx, width, height) {
            this.ctx = ctx;
            this.width = width;
            this.height = height;
            this.currentScene = 'school';
        }

        setScene(sceneId) {
            this.currentScene = sceneId;
        }

        draw(dayNight) {
            const ctx = this.ctx;
            
            // 绘制天空
            ctx.fillStyle = dayNight.getSkyGradient(ctx, this.width, this.height);
            ctx.fillRect(0, 0, this.width, this.height);
            
            // 绘制场景
            switch (this.currentScene) {
                case 'school':
                    this.drawSchoolScene(ctx, dayNight);
                    break;
                case 'courtyard':
                    this.drawCourtyardScene(ctx, dayNight);
                    break;
                case 'rooftop':
                    this.drawRooftopScene(ctx, dayNight);
                    break;
                case 'library':
                    this.drawLibraryScene(ctx, dayNight);
                    break;
            }
        }

        drawSchoolScene(ctx, dayNight) {
            // 地面
            ctx.fillStyle = '#4a5568';
            ctx.fillRect(0, this.height * 0.7, this.width, this.height * 0.3);
            
            // 教学楼
            const buildingX = this.width * 0.3;
            const buildingY = this.height * 0.3;
            
            ctx.fillStyle = '#e2e8f0';
            ctx.fillRect(buildingX, buildingY, 300, 250);
            
            // 窗户
            ctx.fillStyle = dayNight.isNight() ? '#ffd700' : '#87ceeb';
            for (let row = 0; row < 3; row++) {
                for (let col = 0; col < 4; col++) {
                    ctx.fillRect(
                        buildingX + 20 + col * 70,
                        buildingY + 20 + row * 75,
                        50, 55
                    );
                }
            }
            
            // 屋顶
            ctx.fillStyle = '#718096';
            ctx.beginPath();
            ctx.moveTo(buildingX - 20, buildingY);
            ctx.lineTo(buildingX + 150, buildingY - 50);
            ctx.lineTo(buildingX + 320, buildingY);
            ctx.closePath();
            ctx.fill();

        drawRooftopScene(ctx, dayNight) {

    // ============================================

    class Game {
        constructor() {

        updateDialog(charId, expression) {
            const data = CharacterData[charId];
            if (!data) return;

        typeText(text) {
            const dialogText = document.getElementById('dialogText');
            dialogText.innerHTML = '';

            const cursor = document.createElement('span');
            cursor.className = 'typing-cursor';
            
            const type = () => {
                if (index < text.length) {
                    dialogText.innerHTML = text.substring(0, index + 1);
                    dialogText.appendChild(cursor);
                    index++;
                    setTimeout(type, 50);
                } else {
                    cursor.remove();
                }
            };
            
            type();

            this.stats[type] = Math.min(100, Math.max(0, this.stats[type] + delta));

            if (bar) {
                bar.style.width = `${this.stats[type]}%`;

            for (let i = 0; i < 5; i++) {
                setTimeout(() => {
                    const heart = document.createElement('div');
                    heart.className = 'heart-float';
                    heart.innerHTML = `
                        <svg width="30" height="30" viewBox="0 0 30 30" fill="#ff6b8a">
                            <path d="M15 26 C5 18, 2 10, 8 5 C12 2, 15 5, 15 8 C15 5, 18 2, 22 5 C28 10, 25 18, 15 26Z"/>
                        </svg>
                    `;
                    heart.style.left = `${300 + Math.random() * 600}px`;
                    heart.style.top = `${400 + Math.random() * 100}px`;
                    container.appendChild(heart);
                    
                    setTimeout(() => heart.remove(), 2000);
                }, i * 200);
            }

        spawnClickEffect(x, y) {
            // 简单的点击粒子效果
            const particle = {
                x, y,
                life: 1,
                particles: []
            };
            
            for (let i = 0; i < 8; i++) {
                const angle = (i / 8) * Math.PI * 2;
                particle.particles.push({
                    x: 0,
                    y: 0,
                    vx: Math.cos(angle) * 3,
                    vy: Math.sin(angle) * 3,
                    size: 3 + Math.random() * 3
                });
            }
            
            this.clickEffects = this.clickEffects || [];
            this.clickEffects.push(particle);
        }

        gameLoop(currentTime) {
            const deltaTime = currentTime - this.lastTime;
            this.lastTime = currentTime;
            
            this.update(deltaTime);
            this.render();
            
            if (this.running) {
                requestAnimationFrame((time) => this.gameLoop(time));
            }
        }

        update(deltaTime) {
            // 更新昼夜系统
            this.dayNight.update(deltaTime);
            
            // 更新时间显示
            document.getElementById('timeDisplay').textContent = this.dayNight.getTimeString();
            document.getElementById('periodDisplay').textContent = this.dayNight.getPeriod().name;
            
            // 更新粒子系统
            this.particleSystem.update();
            
            // 更新星空（夜间显示）
            if (this.dayNight.isNight()) {
                this.starField.update(deltaTime);
            }
            
            // 更新角色
            this.characterManager.update(deltaTime);
            
            // 更新点击效果
            if (this.clickEffects) {
                this.clickEffects = this.clickEffects.filter(effect => {
                    effect.life -= 0.02;
                    effect.particles.forEach(p => {
                        p.x += p.vx;
                        p.y += p.vy;
                        p.vy += 0.1;
                    });
                    return effect.life > 0;
                });
            }
        }

        render() {
            const ctx = this.ctx;
            
            // 清空画布
            ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);
            
            // 绘制场景
            this.sceneRenderer.draw(this.dayNight);
            
            // 绘制星空（夜间）
            if (this.dayNight.isNight()) {
                ctx.globalAlpha = 0.8;
                this.starField.draw(ctx);
                ctx.globalAlpha = 1;
            }
            
            // 绘制粒子
            this.particleSystem.draw(ctx);
            
            // 绘制角色
            this.characterManager.draw(ctx, this.canvas.width, this.canvas.height);
            
            // 绘制点击效果
            if (this.clickEffects) {
                this.clickEffects.forEach(effect => {
                    ctx.globalAlpha = effect.life;
                    ctx.fillStyle = '#ffb7c5';
                    effect.particles.forEach(p => {
                        ctx.beginPath();
                        ctx.arc(effect.x + p.x, effect.y + p.y, p.size * effect.life, 0, Math.PI * 2);
                        ctx.fill();
                    });
                });
                ctx.globalAlpha = 1;
            }
            
            // 绘制环境光效果
            this.drawAmbientLight(ctx);
        }

        drawAmbientLight(ctx) {
            const ambient = this.dayNight.getSkyGradient(ctx, this.canvas.width, this.canvas.height * 0.3);
            
            // 顶部渐变遮罩
            const gradient = ctx.createLinearGradient(0, 0, 0, 100);
            gradient.addColorStop(0, 'rgba(0,0,0,0.3)');
            gradient.addColorStop(1, 'rgba(0,0,0,0)');
            ctx.fillStyle = gradient;
            ctx.fillRect(0, 0, this.canvas.width, 100);
        }
    }

## 实现说明

### 角色立绘系统功能

1. **多角色支持**：预定义了三个角色（小樱、雪乃、春菜），每个角色有独特的外观配色和对话文本

2. **表情系统**：支持6种表情状态
   - `normal` - 普通表情
   - `happy` - 开心（微笑、眉毛上扬）
   - `shy` - 害羞（脸红、低头）
   - `surprised` - 惊讶（瞳孔放大、O形嘴）
   - `sad` - 难过（眉毛下垂、嘴角下撇）
   - `love` - 心动（心形瞳孔、脸红）

3. **微动画系统**
   - **眨眼动画**：随机间隔自动眨眼，平滑过渡
   - **呼吸动画**：身体轻微上下浮动
   - **表情过渡**：表情之间平滑过渡，而非突变

4. **情感表达效果**
   - **脸红效果**：脸颊红晕渐变，害羞时显示经典的红晕线条
   - **瞳孔缩放**：惊讶时放大，害羞时缩小
   - **心形瞳孔**：心动表情时瞳孔变为心形
   - **眉毛动画**：根据表情调整眉毛位置

5. **交互功能**
   - 点击角色面板切换不同角色
   - 点击表情面板切换表情
   - 对话框打字机效果
   - 心动表情触发爱心飘浮动画
