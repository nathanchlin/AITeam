        // ============================================
        // 粒子系统（樱花/光斑）
        // ============================================
        class ParticleSystem {
            constructor(canvas) {
                this.canvas = canvas;
                this.ctx = canvas.getContext('2d');
                this.particles = [];
                this.lightSpots = [];
                this.init();
            }

            init() {
                // 创建樱花粒子
                for (let i = 0; i < CONFIG.PARTICLE_COUNT; i++) {
                    this.particles.push(this.createSakuraParticle());
                }
                // 创建光斑
                for (let i = 0; i < 20; i++) {
                    this.lightSpots.push(this.createLightSpot());
                }
            }

            createSakuraParticle() {
                const colors = ['#ffb7c5', '#ffd4de', '#ffffff', '#ffe4e8'];
                return {
                    x: Math.random() * this.canvas.width,
                    y: Math.random() * this.canvas.height - this.canvas.height,
                    size: Math.random() * 8 + 4,
                    speedX: Math.random() * 2 - 1,
                    speedY: Math.random() * 1.5 + 0.5,
                    rotation: Math.random() * Math.PI * 2,
                    rotationSpeed: (Math.random() - 0.5) * 0.05,
                    opacity: Math.random() * 0.6 + 0.4,
                    color: colors[Math.floor(Math.random() * colors.length)],
                    swing: Math.random() * Math.PI * 2,
                    swingSpeed: Math.random() * 0.02 + 0.01
                };
            }

            createLightSpot() {
                return {
                    x: Math.random() * this.canvas.width,
                    y: Math.random() * this.canvas.height,
                    size: Math.random() * 100 + 50,
                    opacity: Math.random() * 0.15 + 0.05,
                    speedX: (Math.random() - 0.5) * 0.3,
                    speedY: (Math.random() - 0.5) * 0.3,
                    hue: Math.random() * 30 + 330 // 粉色系
                };
            }

            update(timeOfDay) {
                // 更新樱花粒子
                this.particles.forEach(p => {
                    p.swing += p.swingSpeed;
                    p.x += p.speedX + Math.sin(p.swing) * 0.5;
                    p.y += p.speedY;
                    p.rotation += p.rotationSpeed;

                    if (p.y > this.canvas.height + 20) {
                        p.y = -20;
                        p.x = Math.random() * this.canvas.width;
                    }
                    if (p.x < -20) p.x = this.canvas.width + 20;
                    if (p.x > this.canvas.width + 20) p.x = -20;
                });

                // 更新光斑
                this.lightSpots.forEach(spot => {
                    spot.x += spot.speedX;
                    spot.y += spot.speedY;
                    
                    if (spot.x < -spot.size) spot.x = this.canvas.width + spot.size;
                    if (spot.x > this.canvas.width + spot.size) spot.x = -spot.size;
                    if (spot.y < -spot.size) spot.y = this.canvas.height + spot.size;
                    if (spot.y > this.canvas.height + spot.size) spot.y = -spot.size;
                });
            }

            draw(ctx, timeOfDay) {
                // 绘制光斑
                this.lightSpots.forEach(spot => {
                    const gradient = ctx.createRadialGradient(spot.x, spot.y, 0, spot.x, spot.y, spot.size);
                    gradient.addColorStop(0, `hsla(${spot.hue}, 80%, 90%, ${spot.opacity})`);
                    gradient.addColorStop(1, 'transparent');
                    ctx.fillStyle = gradient;
                    ctx.beginPath();
                    ctx.arc(spot.x, spot.y, spot.size, 0, Math.PI * 2);
                    ctx.fill();
                });

                // 绘制樱花粒子
                this.particles.forEach(p => {
                    ctx.save();
                    ctx.translate(p.x, p.y);
                    ctx.rotate(p.rotation);
                    ctx.globalAlpha = p.opacity;
                    
                    // 绘制花瓣形状
                    ctx.beginPath();
                    ctx.fillStyle = p.color;
                    
                    // 五瓣花形状
                    for (let i = 0; i < 5; i++) {
                        const angle = (i / 5) * Math.PI * 2;
                        const x = Math.cos(angle) * p.size * 0.5;
                        const y = Math.sin(angle) * p.size * 0.3;
                        if (i === 0) {
                            ctx.moveTo(x, y);
                        } else {
                            ctx.quadraticCurveTo(
                                Math.cos(angle - 0.3) * p.size * 0.7,
                                Math.sin(angle - 0.3) * p.size * 0.5,
                                x, y
                            );
                        }
                    }
                    ctx.closePath();
                    ctx.fill();
                    
                    ctx.restore();
                });
            }
        }

            getTimePhaseName() {
                const phases = {
                    morning: '早晨',
                    noon: '午间',
                    afternoon: '下午',
                    evening: '傍晚',
                    night: '夜晚'
                };
                return phases[this.state.timePhase] || '早晨';
            }

            advanceTime(hours) {
                const currentHour = this.state.currentDate.getHours();
                let newHour = currentHour + hours;
                let dayAdvanced = false;

                while (newHour >= 24) {
                    newHour -= 24;
                    this.state.currentDate.setDate(this.state.currentDate.getDate() + 1);
                    dayAdvanced = true;
                }

                this.state.currentDate.setHours(newHour);
                this.updateTimePhase();

                if (dayAdvanced) {
                    this.state.stats.stamina = Math.min(100, this.state.stats.stamina + 30);
                }

                return dayAdvanced;
            }

            updateTimePhase() {
                const hour = this.state.currentDate.getHours();
                if (hour >= 5 && hour < 12) {
                    this.state.timePhase = 'morning';
                } else if (hour >= 12 && hour < 14) {
                    this.state.timePhase = 'noon';
                } else if (hour >= 14 && hour < 18) {
                    this.state.timePhase = 'afternoon';
                } else if (hour >= 18 && hour < 21) {
                    this.state.timePhase = 'evening';
                } else {
                    this.state.timePhase = 'night';
                }
            }

            modifyStat(stat, value) {
                const max = PlayerStats[stat]?.max || 100;
                const oldValue = this.state.stats[stat];
                this.state.stats[stat] = Math.max(0, Math.min(max, oldValue + value));
                return this.state.stats[stat] - oldValue;
            }

            modifyAffection(character, value) {
                const oldValue = this.state.affection[character];
                this.state.affection[character] = Math.max(0, Math.min(100, oldValue + value));
                return this.state.affection[character] - oldValue;
            }
        }

            drawScene(timePhase) {
                const colors = this.getSceneColors(this.currentScene, timePhase);
                const ctx = this.ctx;
                const w = this.canvas.width;
                const h = this.canvas.height;

                // 绘制天空渐变
                const skyGradient = ctx.createLinearGradient(0, 0, 0, h * 0.7);
                skyGradient.addColorStop(0, colors.sky[0]);
                skyGradient.addColorStop(1, colors.sky[1]);
                ctx.fillStyle = skyGradient;
                ctx.fillRect(0, 0, w, h);

                // 绘制云朵（白天）
                if (timePhase !== 'night') {
                    this.drawClouds(ctx, timePhase);
                } else {
                    this.drawStars(ctx);
                }

                // 根据场景绘制不同元素
                switch(this.currentScene) {
                    case 'classroom':
                        this.drawClassroom(ctx, colors, timePhase);
                        break;
                    case 'courtyard':
                        this.drawCourtyard(ctx, colors, timePhase);
                        break;
                    case 'rooftop':
                        this.drawRooftop(ctx, colors, timePhase);
                        break;
                    case 'library':
                        this.drawLibrary(ctx, colors, timePhase);
                        break;
                }
            }

            drawClouds(ctx, timePhase) {
                const cloudColor = timePhase === 'evening' ? 'rgba(255, 200, 180, 0.8)' : 'rgba(255, 255, 255, 0.8)';
                ctx.fillStyle = cloudColor;
                
                const cloudPositions = [
                    { x: 100, y: 80, scale: 1 },
                    { x: 400, y: 50, scale: 0.8 },
                    { x: 700, y: 100, scale: 1.2 },
                    { x: 1000, y: 60, scale: 0.9 }
                ];

                cloudPositions.forEach(cloud => {
                    this.drawCloud(ctx, cloud.x, cloud.y, cloud.scale);
                });
            }

            drawCloud(ctx, x, y, scale) {
                ctx.save();
                ctx.translate(x, y);
                ctx.scale(scale, scale);
                ctx.beginPath();
                ctx.arc(0, 0, 30, 0, Math.PI * 2);
                ctx.arc(35, -10, 25, 0, Math.PI * 2);
                ctx.arc(70, 0, 30, 0, Math.PI * 2);
                ctx.arc(35, 10, 20, 0, Math.PI * 2);
                ctx.fill();
                ctx.restore();
            }

            drawStars(ctx) {
                ctx.fillStyle = '#ffffff';
                for (let i = 0; i < 80; i++) {
                    const x = (i * 73 + 17) % this.canvas.width;
                    const y = (i * 47 + 23) % (this.canvas.height * 0.5);
                    const size = (i % 3) + 1;
                    const twinkle = Math.sin(Date.now() * 0.003 + i) * 0.5 + 0.5;
                    ctx.globalAlpha = twinkle * 0.8;
                    ctx.beginPath();
                    ctx.arc(x, y, size, 0, Math.PI * 2);
                    ctx.fill();
                }
                ctx.globalAlpha = 1;
                
                // 月亮
                ctx.fillStyle = '#fffacd';
                ctx.beginPath();
                ctx.arc(900, 80, 40, 0, Math.PI * 2);
                ctx.fill();
                ctx.fillStyle = '#1a1a2e';
                ctx.beginPath();
                ctx.arc(915, 70, 35, 0, Math.PI * 2);
                ctx.fill();
            }

            drawClassroom(ctx, colors, timePhase) {
                const w = this.canvas.width;
                const h = this.canvas.height;

                    ctx.lineWidth = 8;
                    ctx.strokeRect(wx, 100, 200, 250);
                    ctx.beginPath();
                    ctx.moveTo(wx + 100, 100);
                    ctx.lineTo(wx + 100, 350);
                    ctx.moveTo(wx, 225);
                    ctx.lineTo(wx + 200, 225);
                    ctx.stroke();
                }

                // 黑板
                ctx.fillStyle = '#2e7d32';
                ctx.fillRect(50, 400, 300, 150);
                ctx.strokeStyle = '#5d4037';
                ctx.lineWidth = 10;
                ctx.strokeRect(50, 400, 300, 150);

            drawCourtyard(ctx, colors, timePhase) {

        class CharacterRenderer {
            constructor(ctx) {
                this.ctx = ctx;
                this.characters = {};
                this.currentCharacter = null;
                this.expression = ExpressionState.NORMAL;
                this.position = { x: 0, y: 0 };
                this.blinkTimer = 0;
                this.isBlinking = false;
                this.breathOffset = 0;
                this.blushAmount = 0;
            }

            drawCharacter(characterKey, expression, x, y) {
                const data = CharacterData[characterKey];
                if (!data) return;

                this.currentCharacter = characterKey;
                this.expression = expression;
                this.position = { x, y };

                const ctx = this.ctx;
                ctx.save();
                ctx.translate(x, y);

                // 呼吸动画
                this.breathOffset = Math.sin(Date.now() * 0.002) * 3;
                ctx.translate(0, this.breathOffset);

                // 绘制角色
                this.drawFullBody(data, expression);

                ctx.restore();
            }

            drawFullBody(data, expression) {
                const ctx = this.ctx;
                
                // 身体/服装
                ctx.fillStyle = data.outfitColor;
                ctx.beginPath();
                ctx.ellipse(0, 180, 60, 100, 0, 0, Math.PI * 2);
                ctx.fill();

                // 脖子
                ctx.fillStyle = data.skinColor;
                ctx.fillRect(-12, -20, 24, 40);

                // 头部
                ctx.fillStyle = data.skinColor;
                ctx.beginPath();
                ctx.ellipse(0, -60, 50, 55, 0, 0, Math.PI * 2);
                ctx.fill();

                // 头发
                this.drawHair(data);

                // 脸红效果
                if (expression === ExpressionState.SHY || expression === ExpressionState.LOVE) {
                    this.blushAmount = Math.min(1, this.blushAmount + 0.05);
                } else {
                    this.blushAmount = Math.max(0, this.blushAmount - 0.03);
                }

                if (this.blushAmount > 0) {
                    ctx.fillStyle = `rgba(255, 150, 150, ${this.blushAmount * 0.5})`;
                    ctx.beginPath();
                    ctx.ellipse(-30, -45, 15, 8, 0, 0, Math.PI * 2);
                    ctx.fill();
                    ctx.beginPath();
                    ctx.ellipse(30, -45, 15, 8, 0, 0, Math.PI * 2);
                    ctx.fill();
                }

                // 眼睛
                this.drawEyes(data, expression);

                // 嘴巴
                this.drawMouth(expression);

                // 腮红装饰
                ctx.fillStyle = data.accessoryColor;
                ctx.globalAlpha = 0.3;
                ctx.beginPath();
                ctx.ellipse(-35, -40, 12, 6, 0, 0, Math.PI * 2);
                ctx.fill();
                ctx.beginPath();
                ctx.ellipse(35, -40, 12, 6, 0, 0, Math.PI * 2);
                ctx.fill();
                ctx.globalAlpha = 1;
            }

            drawHair(data) {
                const ctx = this.ctx;
                
                // 后发
                ctx.fillStyle = data.hairColor;
                ctx.beginPath();
                ctx.ellipse(0, -50, 55, 60, 0, 0, Math.PI * 2);
                ctx.fill();

                // 刘海
                ctx.fillStyle = data.hairHighlight;
                for (let i = -3; i <= 3; i++) {
                    ctx.beginPath();
                    ctx.moveTo(i * 15 - 10, -110);
                    ctx.quadraticCurveTo(i * 15 - 5, -70, i * 15, -50);
                    ctx.quadraticCurveTo(i * 15 + 5, -70, i * 15 + 10, -110);
                    ctx.fill();
                }

                // 侧发
                ctx.fillStyle = data.hairColor;
                ctx.beginPath();
                ctx.moveTo(-50, -80);
                ctx.quadraticCurveTo(-70, 0, -55, 80);
                ctx.lineTo(-45, 80);
                ctx.quadraticCurveTo(-55, 0, -40, -80);
                ctx.fill();

                ctx.beginPath();
                ctx.moveTo(50, -80);
                ctx.quadraticCurveTo(70, 0, 55, 80);
                ctx.lineTo(45, 80);
                ctx.quadraticCurveTo(55, 0, 40, -80);
                ctx.fill();
            }

            drawEyes(data, expression) {
                const ctx = this.ctx;
                
                // 眨眼逻辑
                this.blinkTimer++;
                if (this.blinkTimer > 180) {
                    this.isBlinking = true;
                    if (this.blinkTimer > 190) {
                        this.isBlinking = false;
                        this.blinkTimer = 0;
                    }
                }

                const eyeScale = this.isBlinking ? 0.1 : 1;
                const pupilScale = expression === ExpressionState.SURPRISED ? 1.3 : 
                                   expression === ExpressionState.LOVE ? 1.2 : 1;

                [-25, 25].forEach(ex => {
                    // 眼白
                    ctx.fillStyle = '#ffffff';
                    ctx.beginPath();
                    ctx.ellipse(ex, -55, 14, 10 * eyeScale, 0, 0, Math.PI * 2);
                    ctx.fill();

                    if (!this.isBlinking) {
                        // 虹膜
                        ctx.fillStyle = data.eyeColor;
                        ctx.beginPath();
                        ctx.ellipse(ex, -55, 10 * pupilScale, 8 * pupilScale, 0, 0, Math.PI * 2);
                        ctx.fill();

                        // 瞳孔
                        ctx.fillStyle = '#000000';
                        ctx.beginPath();
                        ctx.ellipse(ex, -55, 4 * pupilScale, 5 * pupilScale, 0, 0, Math.PI * 2);
                        ctx.fill();

                        // 高光
                        ctx.fillStyle = '#ffffff';
                        ctx.beginPath();
                        ctx.arc(ex + 3, -58, 3, 0, Math.PI * 2);
                        ctx.fill();
                        ctx.beginPath();
                        ctx.arc(ex - 2, -52, 1.5, 0, Math.PI * 2);
                        ctx.fill();
                    }
                });

                // 眉毛
                ctx.strokeStyle = data.hairColor;
                ctx.lineWidth = 3;
                const browOffset = expression === ExpressionState.SAD ? 5 : 
                                   expression === ExpressionState.ANGRY ? -5 : 0;
                ctx.beginPath();
                ctx.moveTo(-35, -80 + browOffset);
                ctx.quadraticCurveTo(-25, -85, -15, -80 + browOffset);
                ctx.stroke();
                ctx.beginPath();
                ctx.moveTo(35, -80 + browOffset);
                ctx.quadraticCurveTo(25, -85, 15, -80 + browOffset);
                ctx.stroke();
            }

            drawMouth(expression) {
                const ctx = this.ctx;
                ctx.strokeStyle = '#d4738a';
                ctx.fillStyle = '#ffb6c1';
                ctx.lineWidth = 2;

                switch(expression) {
                    case ExpressionState.HAPPY:
                    case ExpressionState.LOVE:
                        ctx.beginPath();
                        ctx.arc(0, -30, 10, 0, Math.PI);
                        ctx.fill();
                        ctx.stroke();
                        break;
                    case ExpressionState.SAD:
                        ctx.beginPath();
                        ctx.arc(0, -20, 8, Math.PI, 0);
                        ctx.stroke();
                        break;
                    case ExpressionState.SURPRISED:
                        ctx.beginPath();
                        ctx.ellipse(0, -28, 6, 10, 0, 0, Math.PI * 2);
                        ctx.fill();
                        ctx.stroke();
                        break;
                    case ExpressionState.SHY:
                        ctx.beginPath();
                        ctx.moveTo(-8, -30);
                        ctx.quadraticCurveTo(0, -25, 8, -30);
                        ctx.stroke();
                        break;
                    default:
                        ctx.beginPath();
                        ctx.moveTo(-8, -30);
                        ctx.lineTo(8, -30);
                        ctx.stroke();
                }
            }
        }

        class DialogueSystem {
            constructor(game) {
                this.game = game;
                this.dialogueBox = document.getElementById('dialogueBox');
                this.speakerAvatar = document.getElementById('speakerAvatar');
                this.speakerName = document.getElementById('speakerName');
                this.dialogueText = document.getElementById('dialogueText');
                this.choicesContainer = document.getElementById('choicesContainer');
                this.currentText = '';
                this.targetText = '';
                this.charIndex = 0;
                this.isTyping = false;
                this.typeSpeed = CONFIG.TYPE_SPEED;
                this.currentDialogue = null;
                this.onComplete = null;
            }

            async showDialogue(dialogueData, onComplete) {
                this.currentDialogue = dialogueData;
                this.onComplete = onComplete;
                
                const character = CharacterData[dialogueData.character];
                this.speakerAvatar.textContent = character.avatar;
                this.speakerName.textContent = character.name;
                this.dialogueBox.style.display = 'block';
                
                // 设置情绪模糊效果
                this.game.effects.setEmotionBlur(dialogueData.expression, true);
                
                await this.typeText(dialogueData.text, dialogueData.expression);
            }

            async typeText(text, expression) {
                this.targetText = text;
                this.currentText = '';
                this.charIndex = 0;
                this.isTyping = true;

                return new Promise(resolve => {
                    const type = () => {
                        if (this.charIndex < this.targetText.length) {
                            this.currentText += this.targetText[this.charIndex];
                            this.dialogueText.innerHTML = this.currentText + '<span class="typing-cursor"></span>';
                            this.charIndex++;
                            setTimeout(type, this.typeSpeed);
                        } else {
                            this.dialogueText.innerHTML = this.currentText;
                            this.isTyping = false;
                            if (this.currentDialogue?.choices) {
                                this.showChoices(this.currentDialogue.choices);
                            }
                            resolve();
                        }
                    };
                    type();
                });
            }

            showChoices(choices) {
                this.choicesContainer.innerHTML = '';
                choices.forEach((choice, index) => {
                    const btn = document.createElement('button');
                    btn.className = 'choice-btn';
                    btn.innerHTML = `<span class="choice-indicator">❀</span>${choice.text}`;
                    btn.onclick = () => this.selectChoice(choice);
                    this.choicesContainer.appendChild(btn);
                    
                    // 动画延迟
                    btn.style.opacity = '0';
                    btn.style.transform = 'translateX(-20px)';
                    setTimeout(() => {
                        btn.style.transition = 'all 0.3s ease';
                        btn.style.opacity = '1';
                        btn.style.transform = 'translateX(0)';
                    }, index * 100);
                });
            }

            async selectChoice(choice) {
                this.choicesContainer.innerHTML = '';
                
                // 好感度变化
                if (choice.affection) {
                    Object.entries(choice.affection).forEach(([char, value]) => {
                        const actualChange = this.game.state.modifyAffection(char, value);
                        if (actualChange !== 0) {
                            this.game.effects.triggerHeartbeat();
                            this.game.effects.triggerBloom(1500);
                            this.game.effects.showFloatingNumber(
                                actualChange,
                                CONFIG.CANVAS_WIDTH / 2,
                                CONFIG.CANVAS_HEIGHT / 2,
                                'affection'
                            );
                        }
                    });
                    this.game.updateAffectionUI();
                }

                this.hideDialogue();
                
                if (this.onComplete) {
                    this.onComplete(choice);
                }
            }

            hideDialogue() {
                this.dialogueBox.style.display = 'none';
                this.choicesContainer.innerHTML = '';
                this.game.effects.setEmotionBlur(null, false);
            }

            skipTyping() {
                if (this.isTyping) {
                    this.currentText = this.targetText;
                    this.dialogueText.innerHTML = this.currentText;
                    this.isTyping = false;
                    if (this.currentDialogue?.choices) {
                        this.showChoices(this.currentDialogue.choices);
                    }
                }
            }
        }

            initEventListeners() {
                // 场景导航按钮
                document.querySelectorAll('.nav-btn').forEach(btn => {
                    btn.addEventListener('click', async (e) => {
                        const scene = e.target.dataset.scene;
                        if (scene && scene !== this.game.state.state.currentScene) {
                            this.game.effects.flashButton(e.target);
                            await this.game.renderer.changeScene(scene, this.game.effects);
                            this.game.state.state.currentScene = scene;
                            
                            document.querySelectorAll('.nav-btn').forEach(b => b.classList.remove('active'));
                            e.target.classList.add('active');
                        }
                    });
                });

                // 属性面板折叠
                document.getElementById('toggleStats').addEventListener('click', () => {
                    const panel = document.getElementById('statsPanel');
                    const btn = document.getElementById('toggleStats');
                    panel.classList.toggle('collapsed');
                    btn.textContent = panel.classList.contains('collapsed') ? '▶' : '◀';
                });

                // 行动分类标签
                document.querySelectorAll('.action-tab').forEach(tab => {
                    tab.addEventListener('click', (e) => {
                        document.querySelectorAll('.action-tab').forEach(t => t.classList.remove('active'));
                        e.target.classList.add('active');
                        this.currentActionCategory = e.target.dataset.category;
                        this.renderActionList();
                    });
                });

                // 画布点击
                this.game.canvas.addEventListener('click', () => {
                    if (this.game.dialogue.isTyping) {
                        this.game.dialogue.skipTyping();
                    } else if (!this.game.state.state.inDialogue && !this.game.state.state.showingActions) {
                        this.showActionPanel();
                    }
                });
            }

            updateStatsUI() {
                this.statsContent.innerHTML = '';
                Object.entries(this.game.state.state.stats).forEach(([key, value]) => {
                    const stat = PlayerStats[key];
                    const item = document.createElement('div');
                    item.className = 'stat-item';
                    item.innerHTML = `
                        <span class="stat-icon">${stat.icon}</span>
                        <div class="stat-info">
                            <div class="stat-name">${stat.name}</div>
                            <div class="stat-bar-container">
                                <div class="stat-bar" style="width: ${(value / stat.max) * 100}%; background: ${stat.color}"></div>
                            </div>
                        </div>
                        <span class="stat-value">${value}</span>
                    `;

                });

                        <div class="character-info">
                            <div class="character-name">${char.name}</div>
                            <div class="hearts-container">${heartsHtml}</div>
                        </div>
                    `;
                    
                    item.addEventListener('click', () => {
                        this.triggerCharacterInteraction(key);
                    });
                    
                    this.affectionContent.appendChild(item);
                });
            }

            highlightAffection(characterKey) {
                const item = document.querySelector(`.affection-item[data-character="${characterKey}"]`);
                if (item) {
                    item.classList.remove('highlight');
                    void item.offsetWidth;
                    item.classList.add('highlight');
                    
                    // 更新心形动画
                    const hearts = item.querySelectorAll('.heart');
                    const value = this.game.state.state.affection[characterKey];
                    const filledHearts = Math.floor(value / 10);
                    hearts.forEach((heart, i) => {
                        if (i === filledHearts - 1 && !heart.classList.contains('filled')) {
                            heart.classList.add('newly-filled');
                        }
                        if (i < filledHearts) {
                            heart.classList.add('filled');
                        }
                    });
                }
            }

            showActionPanel() {
                this.actionPanel.classList.add('active');
                this.game.state.state.showingActions = true;
                this.renderActionList();
            }

            hideActionPanel() {
                this.actionPanel.classList.remove('active');
                this.game.state.state.showingActions = false;
            }

            renderActionList() {
                const actions = ActionsData[this.currentActionCategory];
                this.actionList.innerHTML = '';
                
                actions.forEach(action => {
                    const item = document.createElement('div');
                    item.className = 'action-item';
                    if (this.game.state.state.stats.stamina < 10 && action.effects.stamina < 0) {
                        item.classList.add('disabled');
                    }
                    
                    let effectsHtml = '';
                    Object.entries(action.effects).forEach(([stat, value]) => {
                        const isPositive = value > 0;
                        effectsHtml += `<span class="effect-tag ${isPositive ? 'positive' : 'negative'}">${PlayerStats[stat]?.icon || '💰'} ${value > 0 ? '+' : ''}${value}</span>`;
                    });
                    
                    item.innerHTML = `
                        <div class="action-name">${action.icon} ${action.name}</div>
                        <div class="action-effects">${effectsHtml}</div>
                    `;
                    
                    item.addEventListener('click', () => {
                        if (!item.classList.contains('disabled')) {
                            this.executeAction(action);
                        }
                    });
                    
                    this.actionList.appendChild(item);
                });
            }

            executeAction(action) {
                this.hideActionPanel();
                
                // 应用效果
                const changes = [];
                Object.entries(action.effects).forEach(([stat, value]) => {
                    const actualChange = this.game.state.modifyStat(stat, value);
                    if (actualChange !== 0) {
                        changes.push({ stat, value: actualChange });
                        this.game.effects.showFloatingNumber(
                            actualChange,
                            CONFIG.CANVAS_WIDTH / 2 + Math.random() * 100 - 50,
                            CONFIG.CANVAS_HEIGHT / 2,
                            actualChange > 0 ? 'positive' : 'negative'
                        );
                    }
                });
                
                // 推进时间
                const dayAdvanced = this.game.state.advanceTime(action.timeCost);
                
                // 触发Bloom效果
                this.game.effects.triggerBloom(1000);
                
                // 更新UI
                this.game.updateStatsUI();
                this.game.updateTimeUI();
                
                // 随机事件 - 遇到角色
                if (Math.random() < 0.3) {
                    setTimeout(() => {
                        this.triggerRandomEncounter();
                    }, 800);
                }
            }

            async triggerCharacterInteraction(characterKey) {
                const char = CharacterData[characterKey];
                const expressions = Object.values(ExpressionState);
                const randomExpression = expressions[Math.floor(Math.random() * expressions.length)];
                
                this.game.state.state.inDialogue = true;
                
                // 显示角色
                this.game.currentCharacter = characterKey;
                this.game.characterExpression = randomExpression;
                
                await this.game.dialogue.showDialogue({
                    character: characterKey,
                    text: char.dialogues[randomExpression] || char.dialogues.normal,
                    expression: randomExpression,
                    choices: [
                        { text: '很开心和你聊天', affection: { [characterKey]: 5 } },
                        { text: '希望下次再见面', affection: { [characterKey]: 3 } },
                        { text: '我还有事，先走了', affection: { [characterKey]: -2 } }
                    ]
                }, () => {
                    this.game.state.state.inDialogue = false;
                    this.game.currentCharacter = null;
                    this.highlightAffection(characterKey);
                });
            }

            triggerRandomEncounter() {
                const characters = Object.keys(CharacterData);
                const randomChar = characters[Math.floor(Math.random() * characters.length)];
                this.triggerCharacterInteraction(randomChar);
            }

            updateTimeUI() {
                const date = this.game.state.state.currentDate;
                const months = ['1月', '2月', '3月', '4月', '5月', '6月', '7月', '8月', '9月', '10月', '11月', '12月'];
                const days = ['星期日', '星期一', '星期二', '星期三', '星期四', '星期五', '星期六'];
                
                document.getElementById('dateDisplay').textContent = 
                    `${months[date.getMonth()]}${date.getDate()}日 ${days[date.getDay()]}`;
                document.getElementById('timeDisplay').textContent = 
                    `${String(date.getHours()).padStart(2, '0')}:${String(date.getMinutes()).padStart(2, '0')}`;
                document.getElementById('periodDisplay').textContent = 
                    this.game.state.getTimePhaseName();
            }
        }

        // ============================================
        // 主游戏类
        // ============================================
        class Game {
            constructor() {
                this.canvas = document.getElementById('gameCanvas');
                this.canvas.width = CONFIG.CANVAS_WIDTH;
                this.canvas.height = CONFIG.CANVAS_HEIGHT;
                this.ctx = this.canvas.getContext('2d');
                
                this.state = new GameStateManager();
                this.renderer = new SceneRenderer(this.canvas);
                this.effects = new EffectsManager();
                this.particles = new ParticleSystem(this.canvas);
                this.characterRenderer = new CharacterRenderer(this.ctx);
                this.dialogue = new DialogueSystem(this);
                this.ui = new UIManager(this);
                
                this.currentCharacter = null;
                this.characterExpression = ExpressionState.NORMAL;
                
                this.init();
            }

            init() {
                this.updateStatsUI = () => this.ui.updateStatsUI();
                this.updateAffectionUI = () => this.ui.updateAffectionUI();
                this.updateTimeUI = () => this.ui.updateTimeUI();
                
                this.ui.updateStatsUI();
                this.ui.updateAffectionUI();
                this.ui.updateTimeUI();
                
                this.gameLoop();
            }

            gameLoop() {
                const timePhase = this.state.state.timePhase;
                
                // 清除画布
                this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);
                
                // 绘制场景
                this.renderer.drawScene(timePhase);
                
                // 更新和绘制粒子
                this.particles.update(timePhase);
                this.particles.draw(this.ctx, timePhase);
                
                // 绘制角色
                if (this.currentCharacter) {
                    this.characterRenderer.drawCharacter(
                        this.currentCharacter,
                        this.characterExpression,
                        this.canvas.width / 2,
                        this.canvas.height - 50
                    );
                }
                
                requestAnimationFrame(() => this.gameLoop());
            }
        }

        // 启动游戏
        window.addEventListener('DOMContentLoaded', () => {
            const game = new Game();
            
            // 演示：自动触发一个心跳效果
            setTimeout(() => {
                game.effects.triggerHeartbeat();
                game.effects.triggerBloom(2000);
            }, 2000);
        });
