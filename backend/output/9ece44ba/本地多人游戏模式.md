# 本地多人游戏模式

**Agent**: coder

**Time**: 2026-02-23T10:54:15.208304

---

为了实现本地多人游戏模式，我们需要对之前的代码进行一些修改，以便支持多个玩家在同一设备上同时游戏。以下是一个简单的实现方案：

1. 增加一个玩家对象数组，每个玩家都有自己的位置、速度和颜色。
2. 修改游戏循环，以便同时更新和绘制所有玩家。
3. 添加玩家控制逻辑，允许玩家使用不同的键来控制自己的角色。
4. 修改碰撞检测逻辑，以便检测所有玩家之间的碰撞。

以下是修改后的代码示例：

```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>Multiplayer Bullet Dodging Game</title>
<style>
  body {
    margin: 0;
    overflow: hidden;
  }
  canvas {
    display: block;
  }
</style>
</head>
<body>
<canvas id="gameCanvas"></canvas>
<script>
  const canvas = document.getElementById('gameCanvas');
  const ctx = canvas.getContext('2d');
  const width = canvas.width = window.innerWidth;
  const height = canvas.height = window.innerHeight;

  let players = [
    {
      x: width / 4,
      y: height - 50,
      width: 30,
      height: 50,
      color: 'blue',
      dx: 0,
      dy: 0
    },
    {
      x: width / 2,
      y: height - 50,
      width: 30,
      height: 50,
      color: 'red',
      dx: 0,
      dy: 0
    },
    // 可以添加更多玩家
  ];

  let bullets = [];
  let score = 0;

  function drawPlayer(player) {
    ctx.fillStyle = player.color;
    ctx.fillRect(player.x, player.y, player.width, player.height);
  }

  function drawBullet() {
    ctx.fillStyle = 'red';
    for (let bullet of bullets) {
      ctx.fillRect(bullet.x, bullet.y, 5, 10);
      bullet.x += 5;
    }
    bullets = bullets.filter(bullet => bullet.x < width);
  }

  function drawScore() {
    ctx.fillStyle = 'black';
    ctx.fillText(`Score: ${score}`, 10, 30);
  }

  function collisionCheck() {
    for (let i = 0; i < players.length; i++) {
      for (let j = i + 1; j < players.length; j++) {
        if (players[i].x < players[j].x + players[j].width &&
            players[i].x + players[i].width > players[j].x &&
            players[i].y < players[j].y + players[j].height &&
            players[i].y + players[i].height > players[j].y) {
          return true;
        }
      }
    }
    return false;
  }

  function gameLoop() {
    ctx.fillStyle = 'rgba(0, 0, 0, 0.2)';
    ctx.fillRect(0, 0, width, height);
    players.forEach(drawPlayer);
    drawBullet();
    drawScore();

    if (collisionCheck()) {
      alert('Game Over! Score: ' + score);
      score = 0;
      bullets = [];
      players.forEach(player => {
        player.x = width / 4 * (player.y % 2);
        player.y = height - 50;
      });
    }

    requestAnimationFrame(gameLoop);
  }

  window.onload = function() {
    gameLoop();
    document.addEventListener('keydown', function(event) {
      let player = players[event.key === 'ArrowLeft' || event.key === 'ArrowRight' ? 0 : 1];
      if (event.key === 'ArrowLeft') {
        player.dx = -10;
      } else if (event.key === 'ArrowRight') {
        player.dx = 10;
      } else if (event.key === 'ArrowUp') {
        player.dy = -10;
      } else if (event.key === 'ArrowDown') {
        player.dy = 10;
      }
    });

    document.addEventListener('keyup', function(event) {
      let player = players[event.key === 'ArrowLeft' || event.key === 'ArrowRight' ? 0 : 1];
      if (event.key === 'ArrowLeft' || event.key === 'ArrowRight') {
        player.dx = 0;
      } else if (event.key === 'ArrowUp' || event.key === 'ArrowDown') {
        player.dy = 0;
      }
    });

    setInterval(function() {
      bullets.push({
        x: Math.random() * (width - 5),
        y: 0
      });
    }, 500);
  };
</script>
</body>
</html>
```

在这个代码中，我们创建了一个`players`数组来存储所有玩家的信息。每个玩家都有自己的位置、大小、颜色、速度和方向。我们还修改了`drawPlayer`函数来绘制所有玩家，并添加了`gameLoop`函数中的碰撞检测逻辑来检测所有玩家之间的碰撞。

此外，我们添加了键盘事件监听器来控制每个玩家的移动。每个玩家可以使用箭头键来控制自己的角色。当游戏结束时，所有玩家的位置会重置到初始位置。

请注意，这个示例代码仅用于演示目的，它没有实现更复杂的游戏逻辑，例如玩家得分、边界碰撞检测或更复杂的碰撞检测逻辑。这些功能可以根据具体需求进行添加。
