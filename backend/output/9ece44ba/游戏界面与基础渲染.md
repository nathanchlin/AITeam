# 游戏界面与基础渲染

以下是一个简单的躲避弹幕游戏的HTML文件，其中包含了内联CSS和JavaScript代码。这个游戏使用了Canvas API来绘制角色、弹幕和得分板。

```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>Bullet Dodging Game</title>
<style>
  body {
    margin: 0;
    overflow: hidden;
  }
  canvas {
    display: block;
  }
</style>
</head>
<body>
<canvas id="gameCanvas"></canvas>
<script>
  const canvas = document.getElementById('gameCanvas');
  const ctx = canvas.getContext('2d');
  const width = canvas.width = window.innerWidth;
  const height = canvas.height = window.innerHeight;
  let player = {
    x: width / 2,
    y: height - 50,
    width: 30,
    height: 50,
    color: 'blue'
  };
  let bullets = [];
  let score = 0;

  function drawPlayer() {
    ctx.fillStyle = player.color;
    ctx.fillRect(player.x, player.y, player.width, player.height);
  }

  function drawBullet() {
    ctx.fillStyle = 'red';
    for (let bullet of bullets) {
      ctx.fillRect(bullet.x, bullet.y, 5, 10);
      bullet.x += 5;
    }
    bullets = bullets.filter(bullet => bullet.x < width);
  }

  function drawScore() {
    ctx.fillStyle = 'black';
    ctx.fillText(`Score: ${score}`, 10, 30);
  }

  function collisionCheck() {
    for (let bullet of bullets) {
      if (player.x < bullet.x + 5 && player.x + player.width > bullet.x &&
          player.y < bullet.y + 10 && player.y + player.height > bullet.y) {
        return true;
      }
    }
    return false;
  }

  function gameLoop() {
    ctx.fillStyle = 'rgba(0, 0, 0, 0.2)';
    ctx.fillRect(0, 0, width, height);
    drawPlayer();
    drawBullet();
    drawScore();

    if (collisionCheck()) {
      alert('Game Over! Score: ' + score);
      score = 0;
      bullets = [];
    }

    requestAnimationFrame(gameLoop);
  }

  window.onload = function() {
    gameLoop();
    document.addEventListener('keydown', function(event) {
      if (event.key === 'ArrowLeft') {
        player.x -= 10;
      } else if (event.key === 'ArrowRight') {
        player.x += 10;
      }
      if (player.x < 0) player.x = 0;
      if (player.x > width - player.width) player.x = width - player.width;
    });

    setInterval(function() {
      bullets.push({
        x: Math.random() * (width - 5),
        y: 0
      });
    }, 500);
  };
</script>
</body>
</html>
```

这段代码创建了一个简单的躲避弹幕游戏，其中包含以下功能：

- `<canvas>` 元素用于绘制游戏画面。
- `drawPlayer` 函数绘制玩家角色。
- `drawBullet` 函数绘制弹幕。
- `drawScore` 函数显示得分。
- `collisionCheck` 函数检查玩家是否与弹幕发生碰撞。
- `gameLoop` 函数是游戏的主循环，使用 `requestAnimationFrame` 来连续绘制画面。
- `window.onload` 函数初始化游戏，包括设置画布大小、添加键盘事件监听器以及定时生成弹幕。
- 键盘事件监听器允许玩家使用左右箭头键控制玩家角色的移动。

请注意，这个游戏非常简单，没有考虑边界碰撞检测和复杂的游戏逻辑。它仅用于演示如何使用Canvas API创建一个基础的游戏界面。
