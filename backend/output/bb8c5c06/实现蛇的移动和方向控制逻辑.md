    <script>
        // 游戏配置
        const GRID_SIZE = 20;
        const CELL_COUNT = GRID_SIZE * GRID_SIZE;
        
        // 蛇的方向
        const DIRECTIONS = {
            UP: { x: 0, y: -1 },
            DOWN: { x: 0, y: 1 },
            LEFT: { x: -1, y: 0 },
            RIGHT: { x: 1, y: 0 }
        };
        
        // 蛇类
        class Snake {
            constructor() {
                this.reset();
            }
            
            reset() {
                // 初始位置：蛇在中间，长度为3
                const middle = Math.floor(GRID_SIZE / 2);
                this.body = [
                    { x: middle, y: middle },
                    { x: middle - 1, y: middle },
                    { x: middle - 2, y: middle }
                ];
                this.direction = DIRECTIONS.RIGHT;
                this.nextDirection = DIRECTIONS.RIGHT;
                this.growing = false;
            }
            
            // 改变方向
            changeDirection(newDirection) {
                // 防止蛇直接向相反方向移动
                if (
                    (this.direction === DIRECTIONS.UP && newDirection === DIRECTIONS.DOWN) ||
                    (this.direction === DIRECTIONS.DOWN && newDirection === DIRECTIONS.UP) ||
                    (this.direction === DIRECTIONS.LEFT && newDirection === DIRECTIONS.RIGHT) ||
                    (this.direction === DIRECTIONS.RIGHT && newDirection === DIRECTIONS.LEFT)
                ) {
                    return;
                }
                this.nextDirection = newDirection;
            }
            
            // 移动蛇
            move() {
                // 更新方向
                this.direction = this.nextDirection;
                
                // 计算新的头部位置
                const head = { ...this.body[0] };
                head.x += this.direction.x;
                head.y += this.direction.y;
                
                // 添加新头部
                this.body.unshift(head);
                
                // 如果不需要增长，移除尾部
                if (!this.growing) {
                    this.body.pop();
                } else {
                    this.growing = false;
                }
            }
            
            // 增长蛇
            grow() {
                this.growing = true;
            }
            
            // 检查是否撞到自己
            checkSelfCollision() {
                const head = this.body[0];
                for (let i = 1; i < this.body.length; i++) {
                    if (head.x === this.body[i].x && head.y === this.body[i].y) {
                        return true;
                    }
                }
                return false;
            }
            
            // 检查是否撞墙
            checkWallCollision() {
                const head = this.body[0];
                return head.x < 0 || head.x >= GRID_SIZE || head.y < 0 || head.y >= GRID_SIZE;
            }
        }
        
        // 食物类
        class Food {
            constructor() {
                this.position = this.generatePosition();
            }
            
            // 生成食物位置
            generatePosition() {
                let newPosition;
                do {
                    newPosition = {
                        x: Math.floor(Math.random() * GRID_SIZE),
                        y: Math.floor(Math.random() * GRID_SIZE)
                    };
                } while (this.isPositionOnSnake(newPosition));
                return newPosition;
            }
            
            // 检查位置是否在蛇身上
            isPositionOnSnake(position) {
                // 这个方法会在游戏实例中实现
                return false;
            }
            
            // 重新生成食物
            respawn(snake) {
                this.position = this.generatePosition();
            }
        }
        
        // 游戏类
        class Game {
            constructor() {
                this.snake = new Snake();
                this.food = new Food();
                this.score = 0;
                this.highScore = parseInt(localStorage.getItem('snakeHighScore')) || 0;
                this.gameRunning = false;
                this.gamePaused = false;
                this.gameSpeed = 150;
                
                this.initializeDOM();
                this.setupEventListeners();
                this.updateHighScoreDisplay();
            }
            
            // 初始化DOM
            initializeDOM() {
                this.gameBoard = document.getElementById('gameBoard');
                this.scoreDisplay = document.getElementById('score');
                this.highScoreDisplay = document.getElementById('highScore');
                this.gameOverElement = document.getElementById('gameOver');
                this.finalScoreDisplay = document.getElementById('finalScore');
                
                // 创建游戏格子
                for (let i = 0; i < CELL_COUNT; i++) {
                    const cell = document.createElement('div');
                    cell.classList.add('cell');
                    this.gameBoard.appendChild(cell);
                }
                
                this.cells = document.querySelectorAll('.cell');
            }
            
            // 设置事件监听器
            setupEventListeners() {
                // 键盘事件
                document.addEventListener('keydown', (e) => this.handleKeyPress(e));
                
                // 按钮事件
                document.getElementById('startButton').addEventListener('click', () => this.startGame());
                document.getElementById('pauseButton').addEventListener('click', () => this.togglePause());
                document.getElementById('restartButton').addEventListener('click', () => this.restartGame());
                
                // 速度选择
                document.getElementById('speedSelect').addEventListener('change', (e) => {
                    this.gameSpeed = parseInt(e.target.value);
                    if (this.gameRunning && !this.gamePaused) {
                        this.stopGameLoop();
                        this.startGameLoop();
                    }
                });
            }
            
            // 处理键盘输入
            handleKeyPress(e) {
                if (!this.gameRunning || this.gamePaused) return;
                
                switch (e.key) {
                    case 'ArrowUp':
                        this.snake.changeDirection(DIRECTIONS.UP);
                        break;
                    case 'ArrowDown':
                        this.snake.changeDirection(DIRECTIONS.DOWN);
                        break;
                    case 'ArrowLeft':
                        this.snake.changeDirection(DIRECTIONS.LEFT);
                        break;
                    case 'ArrowRight':
                        this.snake.changeDirection(DIRECTIONS.RIGHT);
                        break;
                }
            }
            
            // 开始游戏
            startGame() {
                if (this.gameRunning && !this.gamePaused) return;
                
                if (!this.gameRunning) {
                    this.resetGame();
                }
                
                this.gameRunning = true;
                this.gamePaused = false;
                this.gameOverElement.style.display = 'none';
                this.startGameLoop();
            }
            
            // 暂停/继续游戏
            togglePause() {
                if (!this.gameRunning) return;
                
                this.gamePaused = !this.gamePaused;
                
                if (this.gamePaused) {
                    this.stopGameLoop();
                    document.getElementById('pauseButton').textContent = '继续';
                } else {
                    this.startGameLoop();
                    document.getElementById('pauseButton').textContent = '暂停';
                }
            }
            
            // 重新开始游戏
            restartGame() {
                this.resetGame();
                this.startGame();
            }
            
            // 重置游戏
            resetGame() {
                this.snake.reset();
                this.food = new Food();
                this.score = 0;
                this.updateScoreDisplay();
                this.render();
            }
            
            // 游戏主循环
            gameLoop() {
                if (this.gamePaused) return;
                
                // 移动蛇
                this.snake.move();
                
                // 检查碰撞
                if (this.snake.checkWallCollision() || this.snake.checkSelfCollision()) {
                    this.gameOver();
                    return;
                }
                
                // 检查是否吃到食物
                if (this.snake.body[0].x === this.food.position.x && 
                    this.snake.body[0].y === this.food.position.y) {
                    this.snake.grow();
                    this.score += 10;
                    this.updateScoreDisplay();
                    this.food.respawn(this.snake);
                }
                
                // 渲染游戏
                this.render();
            }
            
            // 开始游戏循环
            startGameLoop() {
                this.gameInterval = setInterval(() => this.gameLoop(), this.gameSpeed);
            }
            
            // 停止游戏循环
            stopGameLoop() {
                clearInterval(this.gameInterval);
            }
            
            // 游戏结束
            gameOver() {
                this.gameRunning = false;
                this.stopGameLoop();
                
                // 更新最高分
                if (this.score > this.highScore) {
                    this.highScore = this.score;
                    localStorage.setItem('snakeHighScore', this.highScore);
                    this.updateHighScoreDisplay();
                }
                
                // 显示游戏结束界面
                this.finalScoreDisplay.textContent = this.score;
                this.gameOverElement.style.display = 'block';
                document.getElementById('pauseButton').textContent = '暂停';
            }
            
            // 渲染游戏
            render() {
                // 清除所有单元格
                this.cells.forEach(cell => {
                    cell.classList.remove('snake', 'snake-head', 'food');
                });
                
                // 渲染蛇
                this.snake.body.forEach((segment, index) => {
                    const cellIndex = segment.y * GRID_SIZE + segment.x;
                    if (cellIndex >= 0 && cellIndex < CELL_COUNT) {
                        if (index === 0) {
                            this.cells[cellIndex].classList.add('snake-head');
                        } else {
                            this.cells[cellIndex].classList.add('snake');
                        }
                    }
                });
                
                // 渲染食物
                const foodCellIndex = this.food.position.y * GRID_SIZE + this.food.position.x;
                if (foodCellIndex >= 0 && foodCellIndex < CELL_COUNT) {
                    this.cells[foodCellIndex].classList.add('food');
                }
            }
            
            // 更新分数显示
            updateScoreDisplay() {
                this.scoreDisplay.textContent = this.score;
            }
            
            // 更新最高分显示
            updateHighScoreDisplay() {
                this.highScoreDisplay.textContent = this.highScore;
            }
        }
        
        // 重写Food类的isPositionOnSnake方法，使其能够访问蛇的实例
        Food.prototype.isPositionOnSnake = function(position) {
            return this.game.snake.body.some(segment => 
                segment.x === position.x && segment.y === position.y
            );
        };
        
        // 初始化游戏
        const game = new Game();
    </script>
