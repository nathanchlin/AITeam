    <script>
        // 游戏配置
        const GRID_SIZE = 20;
        const INITIAL_SPEED = 150; // 毫秒
        const SPEED_INCREMENT = 5; // 每吃一个食物速度增加的毫秒数
        
        // 游戏状态
        let gameBoard;
        let scoreElement;
        let highScoreElement;

        const DIRECTIONS = {
            UP: { x: 0, y: -1 },
            DOWN: { x: 0, y: 1 },
            LEFT: { x: -1, y: 0 },
            RIGHT: { x: 1, y: 0 }
        };
        
        // 难度配置
        const DIFFICULTY_SETTINGS = {
            easy: { speed: 150, name: '简单' },
            medium: { speed: 100, name: '中等' },

        // 蛇类
        class Snake {
            constructor() {
                this.reset();
            }
            
            reset() {
                this.body = [
                    { x: Math.floor(GRID_SIZE / 2), y: Math.floor(GRID_SIZE / 2) }
                ];
                this.direction = DIRECTIONS.RIGHT;
                this.nextDirection = DIRECTIONS.RIGHT;
            }
            
            move() {
                // 更新方向
                this.direction = this.nextDirection;
                
                // 计算新的头部位置
                const head = { ...this.body[0] };
                head.x += this.direction.x;
                head.y += this.direction.y;
                
                // 添加新头部
                this.body.unshift(head);
                
                // 如果没有吃到食物，移除尾部
                if (!this.hasEaten) {
                    this.body.pop();
                } else {
                    this.hasEaten = false;
                }
            }
            
            grow() {
                this.hasEaten = true;
            }
            
            changeDirection(newDirection) {
                // 防止蛇直接掉头
                const oppositeDirection = {
                    [DIRECTIONS.UP]: DIRECTIONS.DOWN,
                    [DIRECTIONS.DOWN]: DIRECTIONS.UP,
                    [DIRECTIONS.LEFT]: DIRECTIONS.RIGHT,
                    [DIRECTIONS.RIGHT]: DIRECTIONS.LEFT
                };
                
                if (newDirection !== oppositeDirection[this.direction]) {
                    this.nextDirection = newDirection;
                }
            }
            
            // 碰撞检测 - 撞墙
            checkWallCollision() {
                const head = this.body[0];
                return head.x < 0 || head.x >= GRID_SIZE || head.y < 0 || head.y >= GRID_SIZE;
            }
            
            // 碰撞检测 - 撞自身
            checkSelfCollision() {
                const head = this.body[0];
                for (let i = 1; i < this.body.length; i++) {
                    if (head.x === this.body[i].x && head.y === this.body[i].y) {
                        return true;
                    }
                }
                return false;
            }
            
            // 获取蛇头位置
            getHeadPosition() {
                return { ...this.body[0] };
            }
            
            // 检查是否吃到食物
            checkFoodCollision(foodPosition) {
                const head = this.body[0];
                return head.x === foodPosition.x && head.y === foodPosition.y;
            }
        }
        
        // 食物类
        class Food {
            constructor() {
                this.position = this.generateRandomPosition();
            }
            
            generateRandomPosition() {
                let newPosition;
                do {
                    newPosition = {
                        x: Math.floor(Math.random() * GRID_SIZE),
                        y: Math.floor(Math.random() * GRID_SIZE)
                    };
                } while (this.isPositionOnSnake(newPosition));
                
                return newPosition;
            }
            
            isPositionOnSnake(position) {
                return snake.body.some(segment => 
                    segment.x === position.x && segment.y === position.y
                );
            }
            
            respawn() {
                this.position = this.generateRandomPosition();
            }
        }

            createGameBoard();
            
            // 初始化游戏对象
            snake = new Snake();

                        // 更新按钮样式
                        difficultyButtons.forEach(btn => btn.classList.remove('active'));
                        button.classList.add('active');

                });

            // 键盘控制
            document.addEventListener('keydown', handleKeyPress);

        // 创建游戏网格
        function createGameBoard() {
            gameBoard.innerHTML = '';
            for (let i = 0; i < GRID_SIZE * GRID_SIZE; i++) {
                const cell = document.createElement('div');
                cell.classList.add('cell');
                gameBoard.appendChild(cell);
            }

        // 开始游戏
        function startGame() {
            if (gameInterval) return;
            
            // 重置游戏状态
            snake.reset();
            food = new Food();
            score = 0;
            scoreElement.textContent = score;
            direction = DIRECTIONS.RIGHT;
            
            // 隐藏游戏结束界面
            gameOverElement.style.display = 'none';
            
            // 开始游戏循环
            gameInterval = setInterval(gameLoop, gameSpeed);
        }
        
        // 游戏主循环
        function gameLoop() {
            // 移动蛇
            snake.move();
            
            // 检查碰撞
            if (snake.checkWallCollision() || snake.checkSelfCollision()) {
                endGame();
                return;
            }
            
            // 检查是否吃到食物
            if (snake.checkFoodCollision(food.position)) {
                snake.grow();
                score += 10;
                scoreElement.textContent = score;
                
                // 更新最高分
                if (score > highScore) {
                    highScore = score;
                    highScoreElement.textContent = highScore;
                    localStorage.setItem('snakeHighScore', highScore);
                }
                
                // 生成新食物
                food.respawn();
                
                // 增加游戏速度
                if (gameSpeed > 50) {
                    gameSpeed -= SPEED_INCREMENT;
                    clearInterval(gameInterval);
                    gameInterval = setInterval(gameLoop, gameSpeed);
                }
            }
            
            // 更新显示
            updateDisplay();
        }
        
        // 更新游戏显示
        function updateDisplay() {
            // 清除所有单元格
            const cells = gameBoard.querySelectorAll('.cell');
            cells.forEach(cell => {
                cell.classList.remove('snake', 'snake-head', 'food');
            });
            
            // 绘制蛇
            snake.body.forEach((segment, index) => {
                const cellIndex = segment.y * GRID_SIZE + segment.x;
                if (cells[cellIndex]) {
                    if (index === 0) {
                        cells[cellIndex].classList.add('snake-head');
                    } else {
                        cells[cellIndex].classList.add('snake');
                    }
                }
            });
            
            // 绘制食物
            const foodCellIndex = food.position.y * GRID_SIZE + food.position.x;
            if (cells[foodCellIndex]) {
                cells[foodCellIndex].classList.add('food');
            }
        }
        
        // 结束游戏
        function endGame() {
            clearInterval(gameInterval);
            gameInterval = null;
            
            // 显示游戏结束界面
            finalScoreElement.textContent = score;
            gameOverElement.style.display = 'block';
        }
        
        // 重新开始游戏
        function restartGame() {
            startGame();
        }
        
        // 处理键盘输入
        function handleKeyPress(event) {
            if (!gameInterval) return;
            
            switch (event.key) {
                case 'ArrowUp':
                    event.preventDefault();
                    snake.changeDirection(DIRECTIONS.UP);
                    break;
                case 'ArrowDown':
                    event.preventDefault();
                    snake.changeDirection(DIRECTIONS.DOWN);
                    break;
                case 'ArrowLeft':
                    event.preventDefault();
                    snake.changeDirection(DIRECTIONS.LEFT);
                    break;
                case 'ArrowRight':
                    event.preventDefault();
                    snake.changeDirection(DIRECTIONS.RIGHT);
                    break;
            }
        }
        
        // 页面加载完成后初始化游戏
        document.addEventListener('DOMContentLoaded', initGame);
    </script>

## 碰撞检测模块详解

### 1. 撞墙检测

在 `Snake` 类中的 `checkWallCollision` 方法实现：

```javascript
checkWallCollision() {
    const head = this.body[0];
    return head.x < 0 || head.x >= GRID_SIZE || head.y < 0 || head.y >= GRID_SIZE;
}
```

**工作原理**：
- 检查蛇头的 x 和 y 坐标是否超出游戏网格边界
- 如果蛇头的 x 坐标小于 0 或大于等于 GRID_SIZE，或者 y 坐标小于 0 或大于等于 GRID_SIZE，则判定为撞墙
- 返回布尔值，表示是否发生碰撞

### 2. 撞自身检测

在 `Snake` 类中的 `checkSelfCollision` 方法实现：

```javascript
checkSelfCollision() {
    const head = this.body[0];
    for (let i = 1; i < this.body.length; i++) {
        if (head.x === this.body[i].x && head.y === this.body[i].y) {
            return true;
        }
    }
    return false;
}
```

**工作原理**：
- 获取蛇头的位置
- 遍历蛇身体的其余部分（从索引 1 开始）
- 检查蛇头的位置是否与蛇身体的任何部分重叠
- 如果发现重叠，则判定为撞自身，返回 true
- 如果遍历完所有部分都没有发现重叠，返回 false

### 3. 碰撞检测在游戏循环中的应用

在游戏主循环中，碰撞检测被整合：

```javascript
function gameLoop() {
    // 移动蛇
    snake.move();
    
    // 检查碰撞
    if (snake.checkWallCollision() || snake.checkSelfCollision()) {
        endGame();
        return;
    }
    
    // 其他游戏逻辑...
}
```

**流程**：
1. 每次游戏循环开始时，蛇会移动
2. 移动后立即检查是否撞墙或撞自身
3. 如果发生碰撞，调用 `endGame()` 函数结束游戏
4. 如果没有碰撞，继续执行其他游戏逻辑（如检查是否吃到食物）

### 4. 其他碰撞检测

除了撞墙和撞自身，游戏还实现了食物碰撞检测：

```javascript
checkFoodCollision(foodPosition) {
    const head = this.body[0];
    return head.x === foodPosition.x && head.y === foodPosition.y;
}
```

**工作原理**：
- 比较蛇头位置与食物位置
- 如果两者坐标相同，则判定为吃到食物
- 返回布尔值，表示是否吃到食物
