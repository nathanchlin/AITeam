```html
    <script>
        // 游戏配置
        const GRID_SIZE = 20;
        const CELL_SIZE = 20;
        const INITIAL_SPEED = 150;
        
        // 游戏状态
        let gameRunning = false;
        let score = 0;
        let gameSpeed = INITIAL_SPEED;
        let gameInterval;
        
        // 获取DOM元素
        const gameBoard = document.getElementById('game-board');
        const scoreElement = document.getElementById('score');
        const gameOverElement = document.getElementById('game-over');
        const finalScoreElement = document.getElementById('final-score');
        const restartBtn = document.getElementById('restart-btn');
        const speedSelect = document.getElementById('speed');
        
        // 蛇类定义
        class Snake {
            constructor() {
                this.reset();
            }
            
            reset() {
                this.body = [
                    { x: Math.floor(GRID_SIZE / 2), y: Math.floor(GRID_SIZE / 2) }
                ];
                this.direction = { x: 1, y: 0 };
                this.nextDirection = { x: 1, y: 0 };
                this.growing = false;
            }
            
            move() {
                // 更新方向
                this.direction = { ...this.nextDirection };
                
                // 计算新的头部位置
                const head = { ...this.body[0] };
                head.x += this.direction.x;
                head.y += this.direction.y;
                
                // 添加新头部
                this.body.unshift(head);
                
                // 如果不需要增长，移除尾部
                if (!this.growing) {
                    this.body.pop();
                } else {
                    this.growing = false;
                }
            }
            
            grow() {
                this.growing = true;
            }
            
            changeDirection(newDirection) {
                // 防止蛇直接掉头
                if (this.direction.x + newDirection.x !== 0 || 
                    this.direction.y + newDirection.y !== 0) {
                    this.nextDirection = newDirection;
                }
            }
            
            checkCollision() {
                const head = this.body[0];
                
                // 检查是否撞墙
                if (head.x < 0 || head.x >= GRID_SIZE || 
                    head.y < 0 || head.y >= GRID_SIZE) {
                    return true;
                }
                
                // 检查是否撞到自己（从第二个身体部分开始检查）
                for (let i = 1; i < this.body.length; i++) {
                    if (head.x === this.body[i].x && head.y === this.body[i].y) {
                        return true;
                    }
                }
                
                return false;
            }
            
            render() {
                // 清除之前的蛇
                document.querySelectorAll('.snake').forEach(cell => {
                    cell.classList.remove('snake');
                });
                
                // 渲染新的蛇
                this.body.forEach(segment => {
                    const cell = document.querySelector(`.cell[data-x="${segment.x}"][data-y="${segment.y}"]`);
                    if (cell) {
                        cell.classList.add('snake');
                    }
                });
            }
        }
        
        // 食物类
        class Food {
            constructor() {
                this.position = this.generatePosition();
            }
            
            generatePosition() {
                let newPosition;
                do {
                    newPosition = {
                        x: Math.floor(Math.random() * GRID_SIZE),
                        y: Math.floor(Math.random() * GRID_SIZE)
                    };
                } while (isPositionOnSnake(newPosition));
                
                return newPosition;
            }
            
            respawn() {
                this.position = this.generatePosition();
            }
            
            render() {
                // 清除之前食物
                document.querySelectorAll('.food').forEach(cell => {
                    cell.classList.remove('food');
                });
                
                // 渲染新食物
                const cell = document.querySelector(`.cell[data-x="${this.position.x}"][data-y="${this.position.y}"]`);
                if (cell) {
                    cell.classList.add('food');
                }
            }
        }
        
        // 检查位置是否在蛇身上
        function isPositionOnSnake(position) {
            return snake.body.some(segment => 
                segment.x === position.x && segment.y === position.y
            );
        }
        
        // 初始化游戏
        function initGame() {
            // 创建游戏网格
            gameBoard.innerHTML = '';
            for (let y = 0; y < GRID_SIZE; y++) {
                for (let x = 0; x < GRID_SIZE; x++) {
                    const cell = document.createElement('div');
                    cell.className = 'cell';
                    cell.setAttribute('data-x', x);
                    cell.setAttribute('data-y', y);
                    gameBoard.appendChild(cell);
                }
            }
            
            // 初始化蛇和食物
            snake = new Snake();
            food = new Food();
            
            // 重置游戏状态
            score = 0;
            scoreElement.textContent = score;
            gameRunning = true;
            gameSpeed = parseInt(speedSelect.value);
            
            // 隐藏游戏结束界面
            gameOverElement.style.display = 'none';
            
            // 渲染初始状态
            snake.render();
            food.render();
            
            // 开始游戏循环
            if (gameInterval) clearInterval(gameInterval);
            gameInterval = setInterval(gameLoop, gameSpeed);
        }
        
        // 游戏主循环
        function gameLoop() {
            if (!gameRunning) return;
            
            // 移动蛇
            snake.move();
            
            // 检查碰撞
            if (snake.checkCollision()) {
                endGame();
                return;
            }
            
            // 检查是否吃到食物
            const head = snake.body[0];
            if (head.x === food.position.x && head.y === food.position.y) {
                // 增加分数
                score += 10;
                scoreElement.textContent = score;
                
                // 蛇增长
                snake.grow();
                
                // 生成新食物
                food.respawn();
                food.render();
            }
            
            // 渲染蛇
            snake.render();
        }
        
        // 结束游戏
        function endGame() {
            gameRunning = false;
            clearInterval(gameInterval);
            
            // 显示游戏结束界面
            finalScoreElement.textContent = score;
            gameOverElement.style.display = 'block';
        }
        
        // 键盘事件处理
        document.addEventListener('keydown', (e) => {
            if (!gameRunning) return;
            
            switch (e.key) {
                case 'ArrowUp':
                    snake.changeDirection({ x: 0, y: -1 });
                    break;
                case 'ArrowDown':
                    snake.changeDirection({ x: 0, y: 1 });
                    break;
                case 'ArrowLeft':
                    snake.changeDirection({ x: -1, y: 0 });
                    break;
                case 'ArrowRight':
                    snake.changeDirection({ x: 1, y: 0 });
                    break;
            }
        });
        
        // 重新开始按钮事件
        restartBtn.addEventListener('click', initGame);
        
        // 速度选择事件
        speedSelect.addEventListener('change', (e) => {
            gameSpeed = parseInt(e.target.value);
            if (gameRunning) {
                clearInterval(gameInterval);
                gameInterval = setInterval(gameLoop, gameSpeed);
            }
        });
        
        // 初始化游戏
        let snake, food;
        initGame();
    </script>
</body>
</html>
```
