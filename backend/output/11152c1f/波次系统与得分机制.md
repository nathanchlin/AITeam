        lerp(a, b, t) {
            return a + (b - a) * t;
        },
        
        clamp(val, min, max) {
            return Math.max(min, Math.min(max, val));
        },
        
        distance(x1, y1, x2, y2) {
            const dx = x2 - x1;
            const dy = y2 - y1;
            return Math.sqrt(dx * dx + dy * dy);
        },
        
        angle(x1, y1, x2, y2) {
            return Math.atan2(y2 - y1, x2 - x1);
        },
        
        randomRange(min, max) {
            return min + Math.random() * (max - min);
        },

            return Math.floor(Math.random() * (max - min + 1)) + min;
        },
        
        weightedRandom(items) {
            const totalWeight = items.reduce((sum, item) => sum + item.weight, 0);
            let random = Math.random() * totalWeight;
            for (const item of items) {
                random -= item.weight;
                if (random <= 0) return item;
            }
            return items[items.length - 1];
        },

        circleCollision(x1, y1, r1, x2, y2, r2) {
            const dx = x2 - x1;
            const dy = y2 - y1;
            const dist = Math.sqrt(dx * dx + dy * dy);
            return dist < r1 + r2;
        },

        pointInCircle(px, py, cx, cy, r) {
            const dx = px - cx;
            const dy = py - cy;
            return dx * dx + dy * dy < r * r;
        },

        formatTime(seconds) {
            const mins = Math.floor(seconds / 60);
            const secs = Math.floor(seconds % 60);
            return `${mins}:${secs.toString().padStart(2, '0')}`;
        },

        formatScore(score) {
            return score.toString().replace(/\B(?=(\d{3})+(?!\d))/g, ',');
        }
    };

    class Vector2 {
        constructor(x = 0, y = 0) {
            this.x = x;
            this.y = y;
        }
        
        set(x, y) {
            this.x = x;
            this.y = y;
            return this;
        }
        
        copy(v) {
            this.x = v.x;
            this.y = v.y;
            return this;
        }
        
        clone() {
            return new Vector2(this.x, this.y);
        }
        
        add(v) {
            this.x += v.x;
            this.y += v.y;
            return this;
        }
        
        sub(v) {
            this.x -= v.x;
            this.y -= v.y;
            return this;
        }
        
        scale(s) {
            this.x *= s;
            this.y *= s;
            return this;
        }
        
        normalize() {
            const len = this.length();
            if (len > 0) {
                this.x /= len;
                this.y /= len;
            }
            return this;
        }
        
        length() {
            return Math.sqrt(this.x * this.x + this.y * this.y);
        }
        
        dot(v) {
            return this.x * v.x + this.y * v.y;
        }
    }

    class ScoreSystem {
        constructor() {
            this.currentScore = 0;
            this.highScore = this.loadHighScore();

            this.comboTimer = 0;
            this.totalKills = 0;
            this.totalHeadshots = 0;
            this.waveKills = 0;
            this.waveHeadshots = 0;
            this.waveStartTime = 0;
            this.scorePopups = [];
        }

        loadHighScore() {
            try {
                const saved = localStorage.getItem(CONFIG.HIGH_SCORE_KEY);
                return saved ? parseInt(saved, 10) : 0;
            } catch (e) {
                return 0;
            }
        }

        saveHighScore() {
            try {
                localStorage.setItem(CONFIG.HIGH_SCORE_KEY, this.highScore.toString());
            } catch (e) {
                console.warn('Could not save high score');
            }
        }

        addKill(zombieType, isHeadshot = false) {
            let points = CONFIG.SCORE_BASE_KILL * zombieType.scoreMultiplier;
            
            if (isHeadshot) {
                points += CONFIG.SCORE_HEADSHOT_BONUS;
                this.totalHeadshots++;
                this.waveHeadshots++;
            }
            
            this.combo++;
            this.comboTimer = CONFIG.COMBO_TIMEOUT;
            
            if (this.combo > CONFIG.COMBO_MAX) {
                this.combo = CONFIG.COMBO_MAX;
            }
            
            const comboMultiplier = 1 + (this.combo - 1) * 0.1;
            points = Math.round(points * comboMultiplier);
            
            this.currentScore += points;
            this.totalKills++;
            this.waveKills++;
            
            this.scorePopups.push({
                value: points,
                x: 0,
                y: 0,
                isHeadshot,
                combo: this.combo
            });
            
            return points;
        }

        completeWave(waveNumber, waveTime) {
            let bonus = CONFIG.SCORE_WAVE_COMPLETION_BONUS * waveNumber;
            const timeBonus = Math.max(0, Math.round((30 - waveTime) * CONFIG.SCORE_WAVE_TIME_BONUS_MULTIPLIER));
            bonus += timeBonus;
            
            this.currentScore += bonus;
            
            if (this.currentScore > this.highScore) {
                this.highScore = this.currentScore;
                this.saveHighScore();
                return { bonus, timeBonus, isNewRecord: true };
            }
            
            return { bonus, timeBonus, isNewRecord: false };
        }

        update(dt) {
            if (this.comboTimer > 0) {
                this.comboTimer -= dt;
                if (this.comboTimer <= 0) {
                    this.combo = 0;
                }
            }
        }

        startWave() {
            this.waveKills = 0;
            this.waveHeadshots = 0;
            this.waveStartTime = 0;
        }

        getComboMultiplier() {
            return 1 + (this.combo - 1) * 0.1;
        }

        reset() {
            this.currentScore = 0;
            this.combo = 0;
            this.comboTimer = 0;
            this.totalKills = 0;
            this.totalHeadshots = 0;
            this.waveKills = 0;
            this.waveHeadshots = 0;
            this.waveStartTime = 0;
            this.scorePopups = [];
        }
    }

    class WaveManager {
        constructor() {
            this.currentWave = 0;
            this.waveState = 'idle'; // idle, countdown, spawning, active, complete
            this.zombiesToSpawn = 0;
            this.zombiesRemaining = 0;
            this.spawnTimer = 0;
            this.countdownTimer = 0;
            this.waveTimer = 0;
            this.spawnQueue = [];
            this.announced = false;
        }

        startNextWave() {
            this.currentWave++;
            this.waveState = 'countdown';
            this.countdownTimer = CONFIG.WAVE_DELAY;
            this.zombiesToSpawn = this.calculateZombieCount();
            this.zombiesRemaining = this.zombiesToSpawn;
            this.spawnQueue = this.generateSpawnQueue();
            this.announced = false;
            
            return {
                wave: this.currentWave,
                zombieCount: this.zombiesToSpawn,
                countdown: CONFIG.WAVE_DELAY
            };
        }

        calculateZombieCount() {
            const baseCount = CONFIG.WAVE_BASE_ZOMBIES;
            const increment = CONFIG.WAVE_ZOMBIE_INCREMENT * (this.currentWave - 1);
            const exponentialGrowth = Math.floor(this.currentWave * this.currentWave * 0.1);
            return Math.min(baseCount + increment + exponentialGrowth, CONFIG.MAX_ZOMBIES);
        }

        generateSpawnQueue() {
            const queue = [];
            const types = Object.values(ZOMBIE_TYPES);
            
            // 根据波次调整权重
            const adjustedTypes = types.map(type => {
                let weight = type.weight;
                
                // 高波次时增加特殊僵尸
                if (this.currentWave >= 3 && type.name === 'FAST') {
                    weight += (this.currentWave - 2) * 3;
                }
                if (this.currentWave >= 5 && type.name === 'TANK') {
                    weight += (this.currentWave - 4) * 2;
                }
                
                return { ...type, weight };
            });
            
            for (let i = 0; i < this.zombiesToSpawn; i++) {
                const type = Utils.weightedRandom(adjustedTypes);
                queue.push(type);
            }
            
            // 打乱顺序
            for (let i = queue.length - 1; i > 0; i--) {
                const j = Math.floor(Math.random() * (i + 1));
                [queue[i], queue[j]] = [queue[j], queue[i]];
            }
            
            return queue;
        }

        getDifficultyMultiplier() {
            return 1 + (this.currentWave - 1) * CONFIG.WAVE_DIFFICULTY_SCALE;
        }

        getHealthMultiplier() {
            return 1 + (this.currentWave - 1) * CONFIG.WAVE_HEALTH_SCALE;
        }

        getSpeedMultiplier() {
            return 1 + Math.min((this.currentWave - 1) * CONFIG.WAVE_SPEED_SCALE, 0.5);
        }

        update(dt) {
            this.waveTimer += dt;
            
            switch (this.waveState) {
                case 'countdown':
                    this.countdownTimer -= dt;
                    if (this.countdownTimer <= 0) {
                        this.waveState = 'spawning';
                        this.spawnTimer = 0;
                    }
                    break;
                    
                case 'spawning':
                    this.spawnTimer -= dt;
                    if (this.spawnTimer <= 0 && this.spawnQueue.length > 0) {
                        this.spawnTimer = CONFIG.SPAWN_INTERVAL / Math.min(1 + this.currentWave * 0.1, 2);
                        return this.spawnQueue.shift();
                    }
                    
                    if (this.spawnQueue.length === 0) {
                        this.waveState = 'active';
                    }
                    break;
                    
                case 'active':
                    // 等待所有僵尸被消灭
                    break;
            }
            
            return null;
        }

        zombieKilled() {
            this.zombiesRemaining--;
            if (this.zombiesRemaining <= 0 && this.waveState === 'active') {
                this.waveState = 'complete';
                return true;
            }
            return false;
        }

        getWaveTime() {
            return this.waveTimer;
        }

        isInProgress() {
            return this.waveState === 'spawning' || this.waveState === 'active';
        }

        isCountdown() {
            return this.waveState === 'countdown';
        }

        reset() {
            this.currentWave = 0;
            this.waveState = 'idle';
            this.zombiesToSpawn = 0;
            this.zombiesRemaining = 0;
            this.spawnTimer = 0;
            this.countdownTimer = 0;
            this.waveTimer = 0;
            this.spawnQueue = [];
            this.announced = false;
        }
    }

    class Viewport {
        constructor(canvas, worldWidth, worldHeight) {
            this.canvas = canvas;
            this.worldWidth = worldWidth;
            this.worldHeight = worldHeight;
            this.x = 0;
            this.y = 0;
            this.targetX = 0;
            this.targetY = 0;
            this.smoothing = 0.08;
            this.shakeAmount = 0;
            this.shakeDecay = 0.9;
        }
        
        follow(target, dt) {
            const targetCenterX = target.x - this.canvas.width / 2 + target.width / 2;
            const targetCenterY = target.y - this.canvas.height / 2 + target.height / 2;
            
            this.targetX = Utils.clamp(targetCenterX, 0, this.worldWidth - this.canvas.width);
            this.targetY = Utils.clamp(targetCenterY, 0, this.worldHeight - this.canvas.height);
            
            this.x = Utils.lerp(this.x, this.targetX, this.smoothing);
            this.y = Utils.lerp(this.y, this.targetY, this.smoothing);
            
            if (this.shakeAmount > 0.5) {
                this.x += Utils.randomRange(-this.shakeAmount, this.shakeAmount);
                this.y += Utils.randomRange(-this.shakeAmount, this.shakeAmount);
                this.shakeAmount *= this.shakeDecay;
            } else {
                this.shakeAmount = 0;
            }
        }
        
        shake(amount) {
            this.shakeAmount = Math.max(this.shakeAmount, amount);
        }
        
        worldToScreen(worldX, worldY) {
            return {
                x: worldX - this.x,
                y: worldY - this.y
            };
        }
        
        screenToWorld(screenX, screenY) {
            return {
                x: screenX + this.x,
                y: screenY + this.y
            };
        }
        
        isVisible(x, y, width, height) {
            const margin = 100;
            return x + width + margin > this.x && 
                   x - margin < this.x + this.canvas.width &&
                   y + height + margin > this.y &&
                   y - margin < this.y + this.canvas.height;
        }
    }

                ctx.shadowColor = '#ff6b00';
                ctx.shadowBlur = 10;
            } else {
                ctx.fillStyle = '#ffffff';
            }
            
            ctx.fillText(text, screen.x, screen.y);
            ctx.restore();
        }

            this.trail.unshift({ x: this.centerX, y: this.centerY });
            if (this.trail.length > CONFIG.BULLET_TRAIL_LENGTH) {
                this.trail.pop();
            }
            
            this.x += this.vx * dt;
            this.y += this.vy * dt;
            
            // 检查边界
            if (this.x < -50 || this.x > CONFIG.WORLD_WIDTH + 50 ||
                this.y < -50 || this.y > CONFIG.WORLD_HEIGHT + 50) {
                this.active = false;
            }
        }

            const screen = viewport.worldToScreen(this.centerX, this.centerY);
            
            // 绘制轨迹

                ctx.strokeStyle = '#ffd93d';
                ctx.lineWidth = 3;
                ctx.lineCap = 'round';
                ctx.globalAlpha = 0.5;
                
                ctx.beginPath();
                const startScreen = viewport.worldToScreen(this.trail[0].x, this.trail[0].y);
                ctx.moveTo(startScreen.x, startScreen.y);
                
                for (let i = 1; i < this.trail.length; i++) {
                    const ts = viewport.worldToScreen(this.trail[i].x, this.trail[i].y);
                    ctx.lineTo(ts.x, ts.y);
                    ctx.globalAlpha = 0.5 * (1 - i / this.trail.length);
                }
                ctx.stroke();
                ctx.restore();
            }
            
            // 绘制子弹
            ctx.save();
            ctx.translate(screen.x, screen.y);
            ctx.rotate(this.rotation);
            
            ctx.fillStyle = '#ffd93d';
            ctx.shadowColor = '#ffd93d';
            ctx.shadowBlur = 10;
            
            ctx.beginPath();
            ctx.ellipse(0, 0, 8, 3, 0, 0, Math.PI * 2);
            ctx.fill();
            
            ctx.restore();
        }
    }

    class Zombie extends Entity {
        constructor(x, y, type, waveManager) {
            const size = CONFIG.ZOMBIE_BASE_SIZE * type.size;
            super(x - size, y - size, size * 2, size * 2);
            
            this.type = type;
            this.state = ZOMBIE_STATE.SPAWNING;
            
            // 根据波次调整属性
            const healthMult = waveManager.getHealthMultiplier();
            const speedMult = waveManager.getSpeedMultiplier();
            
            this.maxHealth = CONFIG.ZOMBIE_BASE_HEALTH * type.health * healthMult;
            this.health = this.maxHealth;
            this.speed = CONFIG.ZOMBIE_BASE_SPEED * type.speed * speedMult;
            this.damage = CONFIG.ZOMBIE_ATTACK_DAMAGE * type.damage;
            
            this.attackCooldown = 0;
            this.attackRange = CONFIG.ZOMBIE_ATTACK_RANGE;
            this.stunTimer = 0;
            
            this.spawnTimer = 0.3;
            this.spawnScale = 0;
            
            this.hitFlashTimer = 0;
            this.deathTimer = 0;
            
            // 视觉效果
            this.bobOffset = Math.random() * Math.PI * 2;
            this.targetRotation = 0;
        }

        update(dt, game) {
            switch (this.state) {
                case ZOMBIE_STATE.SPAWNING:
                    this.spawnTimer -= dt;
                    this.spawnScale = 1 - (this.spawnTimer / 0.3);
                    if (this.spawnTimer <= 0) {
                        this.state = ZOMBIE_STATE.CHASE;
                        this.spawnScale = 1;
                    }
                    break;
                    
                case ZOMBIE_STATE.STUNNED:
                    this.stunTimer -= dt;
                    this.vx *= 0.9;
                    this.vy *= 0.9;
                    this.x += this.vx * dt;
                    this.y += this.vy * dt;
                    if (this.stunTimer <= 0) {
                        this.state = ZOMBIE_STATE.CHASE;
                    }
                    break;
                    
                case ZOMBIE_STATE.CHASE:
                    this.chasePlayer(dt, game.player);
                    break;
                    
                case ZOMBIE_STATE.ATTACK:
                    this.attackCooldown -= dt;
                    if (this.attackCooldown <= 0) {
                        this.state = ZOMBIE_STATE.CHASE;
                    }
                    break;
                    
                case ZOMBIE_STATE.DYING:
                    this.deathTimer -= dt;
                    this.x += this.vx * dt;
                    this.y += this.vy * dt;
                    this.vx *= 0.95;
                    this.vy *= 0.95;
                    if (this.deathTimer <= 0) {
                        this.state = ZOMBIE_STATE.DEAD;
                        this.active = false;
                    }
                    break;
            }
            
            // 更新闪烁效果
            if (this.hitFlashTimer > 0) {
                this.hitFlashTimer -= dt;
            }
            
            // 边界约束
            this.x = Utils.clamp(this.x, 0, CONFIG.WORLD_WIDTH - this.width);
            this.y = Utils.clamp(this.y, 0, CONFIG.WORLD_HEIGHT - this.height);
        }

        chasePlayer(dt, player) {
            const dx = player.centerX - this.centerX;
            const dy = player.centerY - this.centerY;
            const dist = Math.sqrt(dx * dx + dy * dy);
            
            if (dist > 0) {
                this.targetRotation = Math.atan2(dy, dx);
            }
            
            // 平滑旋转
            let angleDiff = this.targetRotation - this.rotation;
            while (angleDiff > Math.PI) angleDiff -= Math.PI * 2;
            while (angleDiff < -Math.PI) angleDiff += Math.PI * 2;
            this.rotation += angleDiff * 8 * dt;
            
            if (dist > this.attackRange) {
                const moveX = (dx / dist) * this.speed * dt;
                const moveY = (dy / dist) * this.speed * dt;
                
                this.x += moveX;
                this.y += moveY;
            } else {
                // 攻击
                if (this.attackCooldown <= 0) {
                    this.state = ZOMBIE_STATE.ATTACK;
                    this.attackCooldown = CONFIG.ZOMBIE_ATTACK_COOLDOWN;
                    this.tryAttack(player);
                }
            }
        }

        tryAttack(player) {
            if (!player.isInvincible()) {
                player.takeDamage(this.damage, this);
            }
        }

        takeDamage(damage, knockbackX, knockbackY, isHeadshot) {
            this.health -= damage;
            this.hitFlashTimer = 0.1;
            
            // 击退
            this.vx = knockbackX;
            this.vy = knockbackY;
            
            if (this.health <= 0) {
                this.die();
                return true;
            } else {
                // 击晕
                this.state = ZOMBIE_STATE.STUNNED;
                this.stunTimer = CONFIG.ZOMBIE_HIT_STUN_DURATION;
                return false;
            }
        }

        die() {
            this.state = ZOMBIE_STATE.DYING;
            this.deathTimer = 0.3;
            this.vx *= 0.3;
            this.vy *= 0.3;
        }

        isInHeadshotZone(bulletX, bulletY) {
            const headRadius = this.width * CONFIG.ZOMBIE_HEADSHOT_ZONE_RATIO;
            const headX = this.centerX + Math.cos(this.rotation) * this.width * 0.3;
            const headY = this.centerY + Math.sin(this.rotation) * this.width * 0.3;
            
            const dx = bulletX - headX;
            const dy = bulletY - headY;
            return dx * dx + dy * dy < headRadius * headRadius;
        }

        render(ctx, viewport) {
            if (!viewport.isVisible(this.x, this.y, this.width, this.height)) return;
            
            const screen = viewport.worldToScreen(this.centerX, this.centerY);
            const scale = this.state === ZOMBIE_STATE.SPAWNING ? this.spawnScale : 
                          this.state === ZOMBIE_STATE.DYING ? (1 - this.deathTimer / 0.3) : 1;
            
            ctx.save();
            ctx.translate(screen.x, screen.y);
            ctx.scale(scale, scale);
            ctx.rotate(this.rotation);
            
            // 光晕
            ctx.shadowColor = this.type.glowColor;
            ctx.shadowBlur = 15;
            
            // 身体颜色
            let bodyColor = this.type.color;
            if (this.hitFlashTimer > 0) {
                bodyColor = '#ffffff';
            }
            
            // 绘制僵尸身体
            ctx.fillStyle = bodyColor;
            
            // 身体
            ctx.beginPath();
            ctx.ellipse(0, 0, this.width / 2, this.height / 2.5, 0, 0, Math.PI * 2);
            ctx.fill();
            
            // 头部
            const headSize = this.width * 0.35;
            ctx.beginPath();
            ctx.arc(this.width * 0.25, 0, headSize, 0, Math.PI * 2);
            ctx.fill();
            
            // 眼睛
            ctx.fillStyle = '#ff3d3d';
            ctx.shadowColor = '#ff3d3d';
            ctx.shadowBlur = 8;
            ctx.beginPath();
            ctx.arc(this.width * 0.3, -headSize * 0.3, 3, 0, Math.PI * 2);
            ctx.arc(this.width * 0.3, headSize * 0.3, 3, 0, Math.PI * 2);
            ctx.fill();
            
            ctx.restore();
            
            // 生命条（受伤时显示）
            if (this.health < this.maxHealth && this.state !== ZOMBIE_STATE.DYING) {
                const barWidth = this.width;
                const barHeight = 4;
                const healthPercent = this.health / this.maxHealth;
                
                ctx.fillStyle = 'rgba(0,0,0,0.5)';
                ctx.fillRect(screen.x - barWidth / 2, screen.y - this.height / 2 - 10, barWidth, barHeight);
                
                ctx.fillStyle = healthPercent > 0.5 ? '#4ecdc4' : healthPercent > 0.25 ? '#ffd93d' : '#ff3d3d';
                ctx.fillRect(screen.x - barWidth / 2, screen.y - this.height / 2 - 10, barWidth * healthPercent, barHeight);
            }
        }
    }

    class Player extends Entity {
        constructor(x, y) {
            const size = CONFIG.PLAYER_SIZE;
            super(x - size, y - size, size * 2, size * 2);
            
            this.maxHealth = CONFIG.PLAYER_MAX_HEALTH;
            this.health = this.maxHealth;
            this.invincibleTimer = 0;
            
            this.velocity = new Vector2();
            this.dashCooldown = 0;
            this.isDashing = false;
            this.dashTimer = 0;
            this.dashDirection = new Vector2();
            
            // 武器系统
            this.currentAmmo = CONFIG.MAGAZINE_SIZE;
            this.reserveAmmo = CONFIG.RESERVE_AMMO;
            this.isReloading = false;
            this.reloadTimer = 0;
            this.fireCooldown = 0;
            this.muzzleFlashTimer = 0;
            
            this.targetRotation = 0;
        }

        update(dt, game) {
            this.handleMovement(dt, game.input);
            this.handleShooting(dt, game);
            this.handleDash(dt, game.input);
            this.handleReload(dt);
            
            // 更新无敌时间
            if (this.invincibleTimer > 0) {
                this.invincibleTimer -= dt;
            }
            
            // 更新枪口闪光
            if (this.muzzleFlashTimer > 0) {
                this.muzzleFlashTimer -= dt;
            }
            
            // 边界约束
            this.x = Utils.clamp(this.x, 0, CONFIG.WORLD_WIDTH - this.width);
            this.y = Utils.clamp(this.y, 0, CONFIG.WORLD_HEIGHT - this.height);
        }

        handleMovement(dt, input) {
            const acceleration = new Vector2();
            
            if (input.isKeyDown('KeyW') || input.isKeyDown('ArrowUp')) acceleration.y -= 1;
            if (input.isKeyDown('KeyS') || input.isKeyDown('ArrowDown')) acceleration.y += 1;
            if (input.isKeyDown('KeyA') || input.isKeyDown('ArrowLeft')) acceleration.x -= 1;
            if (input.isKeyDown('KeyD') || input.isKeyDown('ArrowRight')) acceleration.x += 1;
            
            if (acceleration.length() > 0) {
                acceleration.normalize();
            }
            
            const speed = this.isDashing ? CONFIG.PLAYER_DASH_SPEED : CONFIG.PLAYER_SPEED;
            acceleration.scale(speed * CONFIG.PLAYER_ACCELERATION * dt);
            
            this.velocity.add(acceleration);
            
            // 摩擦力
            const friction = Math.pow(0.01, CONFIG.PLAYER_FRICTION * dt);
            this.velocity.scale(friction);
            
            // 限制最大速度
            const currentSpeed = this.velocity.length();
            if (currentSpeed > speed) {
                this.velocity.scale(speed / currentSpeed);
            }
            
            this.x += this.velocity.x * dt;
            this.y += this.velocity.y * dt;
        }

        handleShooting(dt, game) {
            if (this.fireCooldown > 0) {
                this.fireCooldown -= dt;
            }
            
            // 计算朝向鼠标的角度
            const dx = game.input.mouse.worldX - this.centerX;
            const dy = game.input.mouse.worldY - this.centerY;
            this.targetRotation = Math.atan2(dy, dx);
            
            // 平滑旋转
            let angleDiff = this.targetRotation - this.rotation;
            while (angleDiff > Math.PI) angleDiff -= Math.PI * 2;
            while (angleDiff < -Math.PI) angleDiff += Math.PI * 2;
            this.rotation += angleDiff * 15 * dt;
            
            // 射击
            if (game.input.mouse.leftDown && !this.isReloading && this.fireCooldown <= 0 && this.currentAmmo > 0) {
                this.shoot(game);
            }
        }

        shoot(game) {
            this.currentAmmo--;
            this.fireCooldown = CONFIG.FIRE_RATE / 1000;
            this.muzzleFlashTimer = CONFIG.MUZZLE_FLASH_DURATION;
            
            const bulletX = this.centerX + Math.cos(this.rotation) * this.width;
            const bulletY = this.centerY + Math.sin(this.rotation) * this.height;
            
            const bullet = new Bullet(bulletX, bulletY, this.rotation, CONFIG.BULLET_DAMAGE);
            game.bullets.push(bullet);
            
            game.viewport.shake(3);
        }

        handleDash(dt, input) {
            if (this.dashCooldown > 0) {
                this.dashCooldown -= dt;
            }
            
            if (this.isDashing) {
                this.dashTimer -= dt;
                if (this.dashTimer <= 0) {
                    this.isDashing = false;
                }
                
                this.velocity.copy(this.dashDirection);
                this.velocity.scale(CONFIG.PLAYER_DASH_SPEED);
            }
            
            if (input.isKeyDown('Space') && !this.isDashing && this.dashCooldown <= 0) {
                this.isDashing = true;
                this.dashTimer = 0.15;
                this.dashCooldown = 1.0;
                
                this.dashDirection.set(
                    input.isKeyDown('KeyD') - input.isKeyDown('KeyA'),
                    input.isKeyDown('KeyS') - input.isKeyDown('KeyW')
                );
                
                if (this.dashDirection.length() === 0) {
                    this.dashDirection.set(Math.cos(this.rotation), Math.sin(this.rotation));
                }
                
                this.dashDirection.normalize();
                this.invincibleTimer = 0.15;
            }
        }

        handleReload(dt) {
            if (this.isReloading) {
                this.reloadTimer -= dt;
                if (this.reloadTimer <= 0) {
                    const ammoNeeded = CONFIG.MAGAZINE_SIZE - this.currentAmmo;
                    const ammoToAdd = Math.min(ammoNeeded, this.reserveAmmo);
                    this.currentAmmo += ammoToAdd;
                    this.reserveAmmo -= ammoToAdd;
                    this.isReloading = false;
                }
            }
            
            // 自动装弹
            if (!this.isReloading && this.currentAmmo === 0 && this.reserveAmmo > 0) {
                this.startReload();
            }
        }

        startReload() {
            if (!this.isReloading && this.currentAmmo < CONFIG.MAGAZINE_SIZE && this.reserveAmmo > 0) {
                this.isReloading = true;
                this.reloadTimer = CONFIG.RELOAD_TIME;
            }
        }

        isInvincible() {
            return this.invincibleTimer > 0 || this.isDashing;
        }

        takeDamage(damage, source) {
            if (this.isInvincible()) return;
            
            this.health -= damage;
            this.invincibleTimer = CONFIG.PLAYER_INVINCIBLE_TIME;
            
            // 击退
            const dx = this.centerX - source.centerX;
            const dy = this.centerY - source.centerY;
            const dist = Math.sqrt(dx * dx + dy * dy) || 1;
            
            const knockback = CONFIG.ZOMBIE_KNOCKBACK_FORCE * (1 - CONFIG.PLAYER_KNOCKBACK_RESIST);
            this.velocity.x += (dx / dist) * knockback;
            this.velocity.y += (dy / dist) * knockback;
        }

        render(ctx, viewport) {
            const screen = viewport.worldToScreen(this.centerX, this.centerY);
            
            ctx.save();
            ctx.translate(screen.x, screen.y);
            ctx.rotate(this.rotation);
            
            // 闪烁效果（无敌时）
            if (this.invincibleTimer > 0 && Math.floor(this.invincibleTimer * 20) % 2 === 0) {
                ctx.globalAlpha = 0.5;
            }
            
            // 光晕
            ctx.shadowColor = 'rgba(78, 205, 196, 0.5)';
            ctx.shadowBlur = 20;
            
            // 身体
            ctx.fillStyle = '#4ecdc4';
            ctx.beginPath();
            ctx.arc(0, 0, this.width / 2, 0, Math.PI * 2);
            ctx.fill();
            
            // 枪
            ctx.fillStyle = '#2a3a3a';
            ctx.shadowBlur = 0;
            ctx.fillRect(this.width / 2 - 5, -3, 20, 6);
            
            // 枪口闪光
            if (this.muzzleFlashTimer > 0) {
                ctx.fillStyle = '#ff9500';
                ctx.shadowColor = '#ff9500';
                ctx.shadowBlur = 15;
                ctx.beginPath();
                ctx.arc(this.width / 2 + 18, 0, 6, 0, Math.PI * 2);
                ctx.fill();
            }
            
            ctx.restore();
        }
    }

                gameOverScreen: document.getElementById('gameOverScreen'),
                finalWave: document.getElementById('finalWave'),
                finalKills: document.getElementById('finalKills'),
                finalHeadshots: document.getElementById('finalHeadshots'),
                finalScore: document.getElementById('finalScore'),
                newRecordStat: document.getElementById('newRecordStat')
            };
        }

        resize() {
            this.canvas.width = window.innerWidth;
            this.canvas.height = window.innerHeight;
        }

        startGame() {
            this.gameState = 'playing';
            
            this.player = new Player(CONFIG.WORLD_WIDTH / 2, CONFIG.WORLD_HEIGHT / 2);
            this.bullets = [];
            this.zombies = [];
            this.damageNumbers = [];
            this.particles = [];
            
            this.scoreSystem.reset();
            this.waveManager.reset();
            
            // 启动第一波
            this.startNextWave();
            
            this.updateUI();
        }

        restartGame() {
            this.uiElements.gameOverScreen.classList.remove('active');
            this.startGame();
        }

        startNextWave() {
            const waveInfo = this.waveManager.startNextWave();
            this.scoreSystem.startWave();
            
            // 显示波次公告
            this.showWaveAnnouncement(waveInfo.wave);
        }

        showWaveAnnouncement(wave) {
            const elem = this.uiElements.waveAnnouncement;
            this.uiElements.announcementWave.textContent = wave;
            
            // 根据波次设置副标题
            const subtitles = [
                'Prepare yourself...',
                'Here they come!',
                'More incoming!',
                'Stay sharp!',
                'The horde grows!',
                'No rest for the wicked!',
                'They keep coming!',
                'Endless horde!',
                'Can you survive?',
                'Nightmare mode!'
            ];
            this.uiElements.announcementSubtitle.textContent = 
                subtitles[Math.min(wave - 1, subtitles.length - 1)];
            
            elem.classList.remove('active');
            void elem.offsetWidth; // 强制重绘
            elem.classList.add('active');
        }

        showWaveComplete(waveTime, bonus, timeBonus) {
            const elem = this.uiElements.waveComplete;
            this.uiElements.waveCompleteTime.textContent = Utils.formatTime(waveTime);
            this.uiElements.waveCompleteScore.textContent = `+${Utils.formatScore(bonus)}`;
            
            elem.classList.remove('active');
            void elem.offsetWidth;
            elem.classList.add('active');
        }

        showGameOver() {
            this.gameState = 'gameover';
            
            const isNewRecord = this.scoreSystem.currentScore >= this.scoreSystem.highScore;
            
            this.uiElements.finalWave.textContent = this.waveManager.currentWave;
            this.uiElements.finalKills.textContent = this.scoreSystem.totalKills;
            this.uiElements.finalHeadshots.textContent = this.scoreSystem.totalHeadshots;
            this.uiElements.finalScore.textContent = Utils.formatScore(this.scoreSystem.currentScore);
            
            if (isNewRecord) {
                this.uiElements.newRecordStat.style.display = 'block';
            } else {
                this.uiElements.newRecordStat.style.display = 'none';
            }
            
            this.uiElements.gameOverScreen.classList.add('active');
        }

        loop(currentTime) {
            const dt = Math.min((currentTime - this.lastTime) / 1000, 0.1);
            this.lastTime = currentTime;
            
            // FPS计算
            this.fpsUpdateTimer += dt;
            if (this.fpsUpdateTimer >= 0.5) {
                this.fps = Math.round(1 / dt);
                this.fpsUpdateTimer = 0;
            }
            
            if (this.gameState === 'playing') {
                this.update(dt);
            }
            
            this.render();
            this.updateUI();
            
            requestAnimationFrame((t) => this.loop(t));
        }

        update(dt) {
            this.input.updateWorldMouse(this.viewport);
            
            // 更新玩家
            this.player.update(dt, this);
            
            // 视口跟随
            this.viewport.follow(this.player, dt);
            
            // 更新波次
            const zombieType = this.waveManager.update(dt);
            if (zombieType) {
                this.spawnZombie(zombieType);
            }
            
            // 检查波次完成
            if (this.waveManager.waveState === 'complete') {
                const waveTime = this.waveManager.getWaveTime();
                const result = this.scoreSystem.completeWave(this.waveManager.currentWave, waveTime);
                this.showWaveComplete(waveTime, result.bonus, result.timeBonus);
                
                // 短暂延迟后开始下一波
                setTimeout(() => {
                    if (this.gameState === 'playing') {
                        this.startNextWave();
                    }
                }, 2500);
                
                this.waveManager.waveState = 'idle';
            }
            
            // 装弹
            if (this.input.isKeyDown('KeyR') && !this.player.isReloading) {
                this.player.startReload();
            }
            
            // 更新子弹
            this.bullets.forEach(bullet => bullet.update(dt, this));
            this.bullets = this.bullets.filter(b => b.active);
            
            // 更新僵尸
            this.zombies.forEach(zombie => zombie.update(dt, this));
            this.zombies = this.zombies.filter(z => z.active);
            
            // 碰撞检测
            this.checkCollisions();
            
            // 更新伤害数字
            this.damageNumbers.forEach(dn => dn.update(dt));
            this.damageNumbers = this.damageNumbers.filter(dn => dn.active);
            
            // 更新粒子
            this.particles.forEach(p => p.update(dt));
            this.particles = this.particles.filter(p => p.active);
            
            // 更新得分系统
            this.scoreSystem.update(dt);
            
            // 检查游戏结束
            if (this.player.health <= 0) {
                this.showGameOver();
            }
        }

        spawnZombie(type) {
            // 在视口外生成
            let x, y;
            const side = Math.floor(Math.random() * 4);
            const margin = 50;
            
            switch (side) {
                case 0: // 上
                    x = Utils.randomRange(this.viewport.x, this.viewport.x + this.canvas.width);
                    y = this.viewport.y - margin;
                    break;
                case 1: // 右
                    x = this.viewport.x + this.canvas.width + margin;
                    y = Utils.randomRange(this.viewport.y, this.viewport.y + this.canvas.height);
                    break;
                case 2: // 下
                    x = Utils.randomRange(this.viewport.x, this.viewport.x + this.canvas.width);
                    y = this.viewport.y + this.canvas.height + margin;
                    break;
                case 3: // 左
                    x = this.viewport.x - margin;
                    y = Utils.randomRange(this.viewport.y, this.viewport.y + this.canvas.height);
                    break;
            }
            
            const zombie = new Zombie(x, y, type, this.waveManager);
            this.zombies.push(zombie);
        }

        checkCollisions() {
            // 子弹与僵尸
            for (const bullet of this.bullets) {
                for (const zombie of this.zombies) {
                    if (!bullet.active || zombie.state === ZOMBIE_STATE.DYING) continue;
                    
                    if (Utils.circleCollision(
                        bullet.centerX, bullet.centerY, CONFIG.BULLET_SIZE,
                        zombie.centerX, zombie.centerY, zombie.width / 2
                    )) {
                        // 检查爆头
                        const isHeadshot = zombie.isInHeadshotZone(bullet.centerX, bullet.centerY);
                        const damage = isHeadshot ? 
                            CONFIG.BULLET_DAMAGE * CONFIG.BULLET_HEADSHOT_MULTIPLIER : 
                            CONFIG.BULLET_DAMAGE;
                        
                        // 计算击退方向
                        const dx = zombie.centerX - bullet.centerX;
                        const dy = zombie.centerY - bullet.centerY;
                        const dist = Math.sqrt(dx * dx + dy * dy) || 1;
                        const knockbackX = (dx / dist) * CONFIG.ZOMBIE_KNOCKBACK_FORCE;
                        const knockbackY = (dy / dist) * CONFIG.ZOMBIE_KNOCKBACK_FORCE;
                        
                        const killed = zombie.takeDamage(damage, knockbackX, knockbackY, isHeadshot);
                        bullet.active = false;
                        
                        // 显示伤害数字
                        this.damageNumbers.push(new DamageNumber(
                            zombie.centerX,
                            zombie.centerY - zombie.height / 2,
                            damage,
                            isHeadshot
                        ));
                        
                        // 生成血液粒子
                        this.spawnBloodParticles(zombie.centerX, zombie.centerY, dx, dy);
                        
                        if (killed) {
                            // 计分
                            this.scoreSystem.addKill(zombie.type, isHeadshot);
                            this.waveManager.zombieKilled();
                        }
                        
                        break;
                    }
                }
            }
        }

        spawnBloodParticles(x, y, dx, dy) {
            const dist = Math.sqrt(dx * dx + dy * dy) || 1;
            const baseVx = (dx / dist) * 100;
            const baseVy = (dy / dist) * 100;
            
            for (let i = 0; i < CONFIG.BLOOD_PARTICLE_COUNT; i++) {
                const angle = Math.random() * Math.PI * 2;
                const speed = Utils.randomRange(50, 150);
                const vx = baseVx + Math.cos(angle) * speed;
                const vy = baseVy + Math.sin(angle) * speed;
                
                const colors = ['#8b0000', '#a00000', '#c00000', '#600000'];
                const color = colors[Math.floor(Math.random() * colors.length)];
                
                this.particles.push(new Particle(
                    x, y, vx, vy,
                    color,
                    Utils.randomRange(2, 5),
                    Utils.randomRange(0.3, 0.6)
                ));
            }
        }

        render() {
            this.ctx.fillStyle = '#050607';
            this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);
            
            this.renderBackground();
            
            // 按层级排序实体
            const entities = [
                ...this.particles,
                ...this.bullets,
                ...this.zombies,
                this.player,
                ...this.damageNumbers
            ].filter(e => e && e.active);
            
            entities.sort((a, b) => a.layer - b.layer);
            
            entities.forEach(entity => entity.render(this.ctx, this.viewport));
        }

        renderBackground() {
            const gridSize = CONFIG.GRID_SIZE;
            const startX = Math.floor(this.viewport.x / gridSize) * gridSize;
            const startY = Math.floor(this.viewport.y / gridSize) * gridSize;
            
            this.ctx.strokeStyle = 'rgba(255, 255, 255, 0.03)';
            this.ctx.lineWidth = 1;
            
            for (let x = startX; x < this.viewport.x + this.canvas.width + gridSize; x += gridSize) {
                const screen = this.viewport.worldToScreen(x, 0);
                this.ctx.beginPath();
                this.ctx.moveTo(screen.x, 0);
                this.ctx.lineTo(screen.x, this.canvas.height);
                this.ctx.stroke();
            }
            
            for (let y = startY; y < this.viewport.y + this.canvas.height + gridSize; y += gridSize) {
                const screen = this.viewport.worldToScreen(0, y);
                this.ctx.beginPath();
                this.ctx.moveTo(0, screen.y);
                this.ctx.lineTo(this.canvas.width, screen.y);
                this.ctx.stroke();
            }
            
            // 世界边界
            const corners = [
                this.viewport.worldToScreen(0, 0),
                this.viewport.worldToScreen(CONFIG.WORLD_WIDTH, 0),
                this.viewport.worldToScreen(CONFIG.WORLD_WIDTH, CONFIG.WORLD_HEIGHT),
                this.viewport.worldToScreen(0, CONFIG.WORLD_HEIGHT)
            ];
            
            this.ctx.strokeStyle = '#ff3d3d';
            this.ctx.lineWidth = 3;
            this.ctx.shadowColor = '#ff3d3d';
            this.ctx.shadowBlur = 10;
            
            this.ctx.beginPath();
            this.ctx.moveTo(corners[0].x, corners[0].y);
            for (let i = 1; i < corners.length; i++) {
                this.ctx.lineTo(corners[i].x, corners[i].y);
            }
            this.ctx.closePath();
            this.ctx.stroke();
            
            this.ctx.shadowBlur = 0;
        }

        updateUI() {
            // 生命值
            const healthPercent = (this.player.health / this.player.maxHealth) * 100;
            this.uiElements.healthFill.style.width = `${Math.max(0, healthPercent)}%`;

                    <span class="ammo-separator">/</span>
                    <span class="ammo-reserve">${this.player.reserveAmmo}</span>
                `;
            }
            
            // 波次
            this.uiElements.waveNumber.textContent = this.waveManager.currentWave;
            this.uiElements.enemyCount.textContent = this.zombies.length;
            
            // 波次倒计时
            if (this.waveManager.isCountdown()) {
                this.uiElements.waveTimer.classList.add('active');
                this.uiElements.waveTimerValue.textContent = this.waveManager.countdownTimer.toFixed(1);
            } else {
                this.uiElements.waveTimer.classList.remove('active');
            }
            
            // 得分
            this.uiElements.scoreValue.textContent = Utils.formatScore(this.scoreSystem.currentScore);
            this.uiElements.highScore.textContent = Utils.formatScore(this.scoreSystem.highScore);
            this.uiElements.killCount.textContent = this.scoreSystem.totalKills;
            this.uiElements.headshotCount.textContent = this.scoreSystem.totalHeadshots;
            
            // 连击
            if (this.scoreSystem.combo > 1) {
                this.uiElements.comboDisplay.classList.add('active');
                this.uiElements.comboValue.textContent = `x${this.scoreSystem.combo}`;
                this.uiElements.comboMultiplier.textContent = `+${Math.round((this.scoreSystem.getComboMultiplier() - 1) * 100)}%`;
            } else {
                this.uiElements.comboDisplay.classList.remove('active');
            }
            
            // FPS
            this.uiElements.fpsValue.textContent = this.fps;
        }
    }

## 实现说明

### 波次系统
1. **WaveManager 类** - 管理波次状态机：`idle` → `countdown` → `spawning` → `active` → `complete`
2. **难度曲线** - 敌人数量随波次递增（基础+线性增长+指数增长），高波次增加特殊僵尸权重
3. **属性缩放** - 僵尸生命值、速度根据波次动态调整

### 得分机制
1. **ScoreSystem 类** - 追踪当前得分、最高分、连击、击杀统计
2. **连击系统** - 2秒内击杀累积连击，最高10连，提供额外加成（每级+10%）
3. **爆头奖励** - 额外50分，不同僵尸类型有不同的得分倍率

### 数据持久化
- 使用 `localStorage` 存储最高分
- 游戏结束时检测新纪录并显示特殊标记

### UI 反馈
1. **波次公告** - 大字体动画显示新波次
2. **波次完成** - 显示通关时间和奖励分数
3. **连击显示** - 实时显示当前连击数和加成百分比
4. **准备倒计时** - 波次开始前显示倒计时
