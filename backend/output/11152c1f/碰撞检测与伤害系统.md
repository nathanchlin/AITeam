    const Utils = {
        lerp(a, b, t) {
            return a + (b - a) * t;
        },
        
        clamp(val, min, max) {
            return Math.max(min, Math.min(max, val));
        },
        
        distance(x1, y1, x2, y2) {
            const dx = x2 - x1;
            const dy = y2 - y1;
            return Math.sqrt(dx * dx + dy * dy);
        },
        
        angle(x1, y1, x2, y2) {
            return Math.atan2(y2 - y1, x2 - x1);
        },
        
        randomRange(min, max) {
            return min + Math.random() * (max - min);
        },
        
        randomInt(min, max) {
            return Math.floor(Math.random() * (max - min + 1)) + min;
        },
        
        weightedRandom(items) {
            const totalWeight = items.reduce((sum, item) => sum + item.weight, 0);
            let random = Math.random() * totalWeight;
            for (const item of items) {
                random -= item.weight;
                if (random <= 0) return item;
            }
            return items[items.length - 1];
        },

        circleCollision(x1, y1, r1, x2, y2, r2) {
            const dx = x2 - x1;
            const dy = y2 - y1;
            const dist = Math.sqrt(dx * dx + dy * dy);
            return dist < r1 + r2;
        },

        pointInCircle(px, py, cx, cy, r) {
            const dx = px - cx;
            const dy = py - cy;
            return dx * dx + dy * dy < r * r;
        }
    };

    class Vector2 {
        constructor(x = 0, y = 0) {
            this.x = x;
            this.y = y;
        }
        
        set(x, y) {
            this.x = x;
            this.y = y;
            return this;
        }
        
        copy(v) {
            this.x = v.x;
            this.y = v.y;
            return this;
        }
        
        clone() {
            return new Vector2(this.x, this.y);
        }
        
        add(v) {
            this.x += v.x;
            this.y += v.y;
            return this;
        }
        
        sub(v) {
            this.x -= v.x;
            this.y -= v.y;
            return this;
        }
        
        scale(s) {
            this.x *= s;
            this.y *= s;
            return this;
        }
        
        normalize() {
            const len = this.length();
            if (len > 0) {
                this.x /= len;
                this.y /= len;
            }
            return this;
        }
        
        length() {
            return Math.sqrt(this.x * this.x + this.y * this.y);
        }
        
        dot(v) {
            return this.x * v.x + this.y * v.y;
        }
    }

            });

                if (e.button === 0) this.mouse.leftDown = true;
                if (e.button === 2) this.mouse.rightDown = true;
            });
            
            this.canvas.addEventListener('mouseup', (e) => {
                if (e.button === 0) this.mouse.leftDown = false;

            });
            
            this.canvas.addEventListener('contextmenu', (e) => e.preventDefault());
        }

            return this.keys[code] === true;
        }

    class Viewport {
        constructor(canvas, worldWidth, worldHeight) {
            this.canvas = canvas;
            this.worldWidth = worldWidth;
            this.worldHeight = worldHeight;
            this.x = 0;
            this.y = 0;
            this.targetX = 0;
            this.targetY = 0;
            this.smoothing = 0.08;
            this.shakeAmount = 0;
            this.shakeDecay = 0.9;
        }
        
        follow(target, dt) {
            const targetCenterX = target.x - this.canvas.width / 2 + target.width / 2;
            const targetCenterY = target.y - this.canvas.height / 2 + target.height / 2;
            
            this.targetX = Utils.clamp(targetCenterX, 0, this.worldWidth - this.canvas.width);
            this.targetY = Utils.clamp(targetCenterY, 0, this.worldHeight - this.canvas.height);
            
            this.x = Utils.lerp(this.x, this.targetX, this.smoothing);
            this.y = Utils.lerp(this.y, this.targetY, this.smoothing);
            
            if (this.shakeAmount > 0.5) {
                this.x += Utils.randomRange(-this.shakeAmount, this.shakeAmount);
                this.y += Utils.randomRange(-this.shakeAmount, this.shakeAmount);
                this.shakeAmount *= this.shakeDecay;
            } else {
                this.shakeAmount = 0;
            }
        }
        
        shake(amount) {
            this.shakeAmount = Math.max(this.shakeAmount, amount);
        }
        
        worldToScreen(worldX, worldY) {
            return {
                x: worldX - this.x,
                y: worldY - this.y
            };
        }
        
        screenToWorld(screenX, screenY) {
            return {
                x: screenX + this.x,
                y: screenY + this.y
            };
        }
        
        isVisible(x, y, width, height) {
            const margin = 100;
            return x + width + margin > this.x && 
                   x - margin < this.x + this.canvas.width &&
                   y + height + margin > this.y &&
                   y - margin < this.y + this.canvas.height;
        }
    }

            const combinedRadius = (this.width + other.width) / 2;
            return dist < combinedRadius;
        }
    }

    class DamageNumber {
        constructor(x, y, damage, isHeadshot = false) {
            this.x = x;
            this.y = y;
            this.damage = Math.round(damage);
            this.isHeadshot = isHeadshot;
            this.lifeTime = CONFIG.DAMAGE_NUMBER_LIFETIME;
            this.maxLifeTime = CONFIG.DAMAGE_NUMBER_LIFETIME;
            this.vy = -80;
            this.vx = Utils.randomRange(-30, 30);
            this.active = true;
            this.scale = isHeadshot ? 1.5 : 1;
        }

        update(dt) {
            this.y += this.vy * dt;
            this.x += this.vx * dt;
            this.vy *= 0.95;
            this.lifeTime -= dt;
            
            if (this.lifeTime <= 0) {
                this.active = false;
            }
        }

        render(ctx, viewport) {
            const screen = viewport.worldToScreen(this.x, this.y);
            const alpha = this.lifeTime / this.maxLifeTime;
            const scale = this.scale * (1 + (1 - alpha) * 0.3);
            
            ctx.save();
            ctx.globalAlpha = alpha;
            ctx.font = `bold ${Math.round(16 * scale)}px Orbitron`;
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            
            if (this.isHeadshot) {

            } else {
                ctx.fillStyle = '#ffffff';
                ctx.shadowColor = '#000000';
                ctx.shadowBlur = 4;
                ctx.fillText(this.damage, screen.x, screen.y);
            }
            
            ctx.restore();
        }
    }

    class BloodParticle {
        constructor(x, y, angle) {
            this.x = x;
            this.y = y;
            const speed = Utils.randomRange(50, 150);
            this.vx = Math.cos(angle) * speed + Utils.randomRange(-50, 50);
            this.vy = Math.sin(angle) * speed + Utils.randomRange(-50, 50);

            this.lifeTime = Utils.randomRange(0.3, 0.6);
            this.maxLifeTime = this.lifeTime;
            this.active = true;

        }

            this.x += this.vx * dt;
            this.y += this.vy * dt;
            this.vx *= 0.95;
            this.vy *= 0.95;

            this.lifeTime -= dt;
            
            if (this.lifeTime <= 0) {
                this.active = false;
            }
        }

        render(ctx, viewport) {
            const screen = viewport.worldToScreen(this.x, this.y);

            ctx.fillStyle = this.color;
            ctx.beginPath();
            ctx.arc(screen.x, screen.y, this.size * alpha, 0, Math.PI * 2);
            ctx.fill();
            ctx.restore();
        }
    }

    class Bullet extends Entity {
        constructor(x, y, angle, damage = CONFIG.BULLET_DAMAGE) {
            super(x - CONFIG.BULLET_SIZE / 2, y - CONFIG.BULLET_SIZE / 2, 
                  CONFIG.BULLET_SIZE, CONFIG.BULLET_SIZE);
            this.angle = angle;
            this.speed = CONFIG.BULLET_SPEED;
            this.damage = damage;
            this.vx = Math.cos(angle) * this.speed;
            this.vy = Math.sin(angle) * this.speed;
            this.lifeTime = 2;
            this.trail = [];
            this.maxTrailLength = CONFIG.BULLET_TRAIL_LENGTH;
            this.hitEntities = new Set();
            this.piercedCount = 0;
            this.maxPierce = 1;
        }
        
        update(dt, game) {
            this.trail.unshift({ x: this.centerX, y: this.centerY });
            if (this.trail.length > this.maxTrailLength) {
                this.trail.pop();
            }
            
            this.x += this.vx * dt;
            this.y += this.vy * dt;
            
            this.lifeTime -= dt;
            if (this.lifeTime <= 0) {
                this.active = false;
            }
            
            if (this.x < -50 || this.x > CONFIG.WORLD_WIDTH + 50 ||
                this.y < -50 || this.y > CONFIG.WORLD_HEIGHT + 50) {
                this.active = false;
            }
        }
        
        render(ctx, viewport) {
            const screenPos = viewport.worldToScreen(this.x, this.y);
            const screenCenterX = screenPos.x + this.width / 2;
            const screenCenterY = screenPos.y + this.height / 2;
            
            if (this.trail.length > 1) {
                ctx.save();
                ctx.lineCap = 'round';
                
                for (let i = 0; i < this.trail.length - 1; i++) {
                    const t = this.trail[i];
                    const t2 = this.trail[i + 1];
                    const screenT = viewport.worldToScreen(t.x, t.y);
                    const screenT2 = viewport.worldToScreen(t2.x, t2.y);
                    const alpha = 1 - (i / this.trail.length);
                    const width = (1 - i / this.trail.length) * 4;
                    
                    ctx.strokeStyle = `rgba(255, 217, 61, ${alpha * 0.6})`;
                    ctx.lineWidth = width;
                    ctx.beginPath();
                    ctx.moveTo(screenT.x, screenT.y);
                    ctx.lineTo(screenT2.x, screenT2.y);
                    ctx.stroke();
                }
                
                ctx.restore();
            }
            
            ctx.save();
            ctx.fillStyle = CONFIG.bullet || '#ffd93d';
            ctx.shadowColor = 'rgba(255, 217, 61, 0.8)';
            ctx.shadowBlur = 8;
            ctx.beginPath();
            ctx.arc(screenCenterX, screenCenterY, CONFIG.BULLET_SIZE, 0, Math.PI * 2);
            ctx.fill();
            ctx.restore();
        }

        checkHeadshot(zombie) {
            const headZoneRadius = zombie.width * CONFIG.ZOMBIE_HEADSHOT_ZONE_RATIO;
            const headCenterX = zombie.centerX + Math.cos(zombie.rotation) * zombie.width * 0.3;
            const headCenterY = zombie.centerY + Math.sin(zombie.rotation) * zombie.width * 0.3;
            
            return Utils.pointInCircle(this.centerX, this.centerY, headCenterX, headCenterY, headZoneRadius);
        }
    }

    class Player extends Entity {
        constructor(x, y) {
            super(x - CONFIG.PLAYER_SIZE / 2, y - CONFIG.PLAYER_SIZE / 2, 
                  CONFIG.PLAYER_SIZE, CONFIG.PLAYER_SIZE);
            
            this.health = CONFIG.PLAYER_MAX_HEALTH;
            this.maxHealth = CONFIG.PLAYER_MAX_HEALTH;
            this.speed = CONFIG.PLAYER_SPEED;
            
            this.vx = 0;
            this.vy = 0;
            
            this.dashCooldown = 0;
            this.dashDuration = 0;
            this.dashDirection = new Vector2();
            this.isDashing = false;
            this.dashCooldownTime = 1.5;
            this.dashDurationTime = 0.15;
            
            this.isInvincible = false;
            this.invincibleTimer = 0;
            this.invincibleFlashTimer = 0;
            
            this.muzzleFlashTimer = 0;
            this.lastFireTime = 0;
            
            this.currentAmmo = CONFIG.MAGAZINE_SIZE;
            this.reserveAmmo = CONFIG.RESERVE_AMMO;
            this.isReloading = false;
            this.reloadTimer = 0;
            
            this.knockbackVx = 0;
            this.knockbackVy = 0;
        }
        
        update(dt, game) {
            this.updateMovement(dt, game.input);
            this.updateDash(dt);
            this.updateShooting(dt, game);
            this.updateReload(dt);
            this.updateInvincibility(dt);
            this.updateKnockback(dt);
            this.updatePosition(dt);
            this.constrainToWorld();
            
            game.input.updateWorldMouse(game.viewport);
        }
        
        updateMovement(dt, input) {
            if (this.isDashing) return;
            
            const moveDir = new Vector2();
            
            if (input.isKeyDown('KeyW') || input.isKeyDown('ArrowUp')) moveDir.y -= 1;
            if (input.isKeyDown('KeyS') || input.isKeyDown('ArrowDown')) moveDir.y += 1;
            if (input.isKeyDown('KeyA') || input.isKeyDown('ArrowLeft')) moveDir.x -= 1;
            if (input.isKeyDown('KeyD') || input.isKeyDown('ArrowRight')) moveDir.x += 1;
            
            if (moveDir.length() > 0) {
                moveDir.normalize();
                this.vx += moveDir.x * CONFIG.PLAYER_ACCELERATION * dt;
                this.vy += moveDir.y * CONFIG.PLAYER_ACCELERATION * dt;
            }
            
            this.vx *= (1 - CONFIG.PLAYER_FRICTION * dt);
            this.vy *= (1 - CONFIG.PLAYER_FRICTION * dt);
            
            const currentSpeed = Math.sqrt(this.vx * this.vx + this.vy * this.vy);
            if (currentSpeed > this.speed) {
                const scale = this.speed / currentSpeed;
                this.vx *= scale;
                this.vy *= scale;
            }
        }
        
        updateDash(dt, input) {
            this.dashCooldown = Math.max(0, this.dashCooldown - dt);
            
            if (input.isKeyDown('Space') && this.dashCooldown <= 0 && !this.isDashing) {
                const moveDir = new Vector2();
                if (input.isKeyDown('KeyW') || input.isKeyDown('ArrowUp')) moveDir.y -= 1;
                if (input.isKeyDown('KeyS') || input.isKeyDown('ArrowDown')) moveDir.y += 1;
                if (input.isKeyDown('KeyA') || input.isKeyDown('ArrowLeft')) moveDir.x -= 1;
                if (input.isKeyDown('KeyD') || input.isKeyDown('ArrowRight')) moveDir.x += 1;
                
                if (moveDir.length() > 0) {
                    moveDir.normalize();
                } else {
                    moveDir.set(Math.cos(this.rotation), Math.sin(this.rotation));
                }
                
                this.dashDirection.copy(moveDir);
                this.isDashing = true;
                this.dashDuration = this.dashDurationTime;
                this.dashCooldown = this.dashCooldownTime;
                this.isInvincible = true;
                this.invincibleTimer = this.dashDurationTime;
            }
            
            if (this.isDashing) {
                this.dashDuration -= dt;
                this.vx = this.dashDirection.x * CONFIG.PLAYER_DASH_SPEED;
                this.vy = this.dashDirection.y * CONFIG.PLAYER_DASH_SPEED;
                
                if (this.dashDuration <= 0) {
                    this.isDashing = false;
                }
            }
        }
        
        updateShooting(dt, game) {
            const dx = game.input.mouse.worldX - this.centerX;
            const dy = game.input.mouse.worldY - this.centerY;
            this.rotation = Math.atan2(dy, dx);
            
            this.muzzleFlashTimer = Math.max(0, this.muzzleFlashTimer - dt);
            
            if (game.input.mouse.leftDown && !this.isReloading && this.currentAmmo > 0) {
                const now = performance.now();
                if (now - this.lastFireTime >= CONFIG.FIRE_RATE) {
                    this.fire(game);
                    this.lastFireTime = now;
                }
            }
            
            if (game.input.isKeyDown('KeyR') && !this.isReloading && 
                this.currentAmmo < CONFIG.MAGAZINE_SIZE && this.reserveAmmo > 0) {
                this.startReload();
            }
        }
        
        fire(game) {
            const bulletX = this.centerX + Math.cos(this.rotation) * this.width * 0.6;
            const bulletY = this.centerY + Math.sin(this.rotation) * this.height * 0.6;
            
            const spread = (Math.random() - 0.5) * 0.05;
            const bullet = new Bullet(bulletX, bulletY, this.rotation + spread);
            game.bullets.push(bullet);
            
            this.currentAmmo--;
            this.muzzleFlashTimer = CONFIG.MUZZLE_FLASH_DURATION;
            game.viewport.shake(3);
        }
        
        startReload() {
            this.isReloading = true;
            this.reloadTimer = CONFIG.RELOAD_TIME;
        }
        
        updateReload(dt) {
            if (this.isReloading) {
                this.reloadTimer -= dt;
                if (this.reloadTimer <= 0) {
                    const needed = CONFIG.MAGAZINE_SIZE - this.currentAmmo;
                    const toReload = Math.min(needed, this.reserveAmmo);
                    this.currentAmmo += toReload;
                    this.reserveAmmo -= toReload;
                    this.isReloading = false;
                }
            }
        }
        
        updateInvincibility(dt) {
            if (this.isInvincible) {
                this.invincibleTimer -= dt;
                this.invincibleFlashTimer += dt * 10;
                
                if (this.invincibleTimer <= 0) {
                    this.isInvincible = false;
                    this.invincibleFlashTimer = 0;
                }
            }
        }

        updateKnockback(dt) {
            this.x += this.knockbackVx * dt;
            this.y += this.knockbackVy * dt;
            this.knockbackVx *= 0.9;
            this.knockbackVy *= 0.9;
        }
        
        updatePosition(dt) {
            this.x += this.vx * dt;
            this.y += this.vy * dt;
        }
        
        constrainToWorld() {
            this.x = Utils.clamp(this.x, 0, CONFIG.WORLD_WIDTH - this.width);
            this.y = Utils.clamp(this.y, 0, CONFIG.WORLD_HEIGHT - this.height);
        }

        takeDamage(damage, knockbackDir, knockbackForce) {
            if (this.isInvincible) return false;
            
            this.health -= damage;
            this.isInvincible = true;
            this.invincibleTimer = CONFIG.PLAYER_INVINCIBLE_TIME;
            
            const resistance = CONFIG.PLAYER_KNOCKBACK_RESIST;
            this.knockbackVx = knockbackDir.x * knockbackForce * resistance;
            this.knockbackVy = knockbackDir.y * knockbackForce * resistance;
            
            if (this.health <= 0) {
                this.health = 0;
                return true;
            }
            
            return false;
        }
        
        render(ctx, viewport) {
            const screenPos = viewport.worldToScreen(this.x, this.y);
            const screenCenterX = screenPos.x + this.width / 2;
            const screenCenterY = screenPos.y + this.height / 2;
            
            ctx.save();
            ctx.translate(screenCenterX, screenCenterY);
            ctx.rotate(this.rotation);
            
            if (this.isInvincible && Math.sin(this.invincibleFlashTimer) > 0) {
                ctx.globalAlpha = 0.5;
            }
            
            // 玩家身体
            ctx.fillStyle = '#4ecdc4';
            ctx.shadowColor = 'rgba(78, 205, 196, 0.5)';
            ctx.shadowBlur = 15;
            ctx.beginPath();
            ctx.arc(0, 0, this.width / 2, 0, Math.PI * 2);
            ctx.fill();
            
            ctx.shadowBlur = 0;
            ctx.fillStyle = '#3dbdb5';
            ctx.beginPath();
            ctx.arc(0, 0, this.width / 2 - 4, 0, Math.PI * 2);
            ctx.fill();
            
            // 武器
            ctx.fillStyle = '#2a2a2a';
            ctx.fillRect(this.width / 2 - 5, -3, 15, 6);
            ctx.fillStyle = '#1a1a1a';
            ctx.fillRect(this.width / 2 + 5, -2, 8, 4);
            
            ctx.restore();
            
            // 枪口闪光
            if (this.muzzleFlashTimer > 0) {
                const muzzleX = screenCenterX + Math.cos(this.rotation) * this.width * 0.9;
                const muzzleY = screenCenterY + Math.sin(this.rotation) * this.height * 0.9;
                
                ctx.save();
                const flashSize = 15 * (this.muzzleFlashTimer / CONFIG.MUZZLE_FLASH_DURATION);
                const gradient = ctx.createRadialGradient(muzzleX, muzzleY, 0, muzzleX, muzzleY, flashSize);
                gradient.addColorStop(0, 'rgba(255, 200, 100, 1)');
                gradient.addColorStop(0.5, 'rgba(255, 149, 0, 0.8)');
                gradient.addColorStop(1, 'rgba(255, 100, 0, 0)');
                
                ctx.fillStyle = gradient;
                ctx.beginPath();
                ctx.arc(muzzleX, muzzleY, flashSize, 0, Math.PI * 2);
                ctx.fill();
                ctx.restore();
            }
        }
    }

    class Zombie extends Entity {
        constructor(x, y, type = ZOMBIE_TYPES.NORMAL) {
            const size = CONFIG.ZOMBIE_BASE_SIZE * type.size;
            super(x - size / 2, y - size / 2, size, size);
            
            this.type = type;
            this.state = ZOMBIE_STATE.SPAWNING;
            this.health = CONFIG.ZOMBIE_BASE_HEALTH * type.health;
            this.maxHealth = this.health;
            this.speed = CONFIG.ZOMBIE_BASE_SPEED * type.speed;
            this.damage = CONFIG.ZOMBIE_ATTACK_DAMAGE * type.damage;
            
            this.attackCooldown = 0;
            this.attackRange = CONFIG.ZOMBIE_ATTACK_RANGE;
            
            this.spawnTimer = 0.5;
            this.stunTimer = 0;
            this.deathTimer = 0;
            
            this.knockbackVx = 0;
            this.knockbackVy = 0;
            
            this.targetOffset = new Vector2();
            this.wanderTimer = 0;
            this.animOffset = Math.random() * Math.PI * 2;
            
            this.hitFlashTimer = 0;
        }
        
        update(dt, game) {
            switch (this.state) {
                case ZOMBIE_STATE.SPAWNING:
                    this.updateSpawning(dt);
                    break;
                case ZOMBIE_STATE.CHASE:
                case ZOMBIE_STATE.IDLE:
                    this.updateChase(dt, game);
                    break;
                case ZOMBIE_STATE.STUNNED:
                    this.updateStunned(dt);
                    break;
                case ZOMBIE_STATE.ATTACK:
                    this.updateAttack(dt, game);
                    break;
                case ZOMBIE_STATE.DYING:
                    this.updateDying(dt);
                    break;
            }
            
            this.hitFlashTimer = Math.max(0, this.hitFlashTimer - dt);
            this.applyKnockback(dt);
            this.constrainToWorld();
        }
        
        updateSpawning(dt) {
            this.spawnTimer -= dt;
            if (this.spawnTimer <= 0) {
                this.state = ZOMBIE_STATE.CHASE;
            }
        }
        
        updateChase(dt, game) {
            if (!game.player) return;
            
            const dx = game.player.centerX - this.centerX;
            const dy = game.player.centerY - this.centerY;
            const dist = Math.sqrt(dx * dx + dy * dy);
            
            this.rotation = Math.atan2(dy, dx);
            
            if (dist > this.attackRange) {
                const moveX = (dx / dist) * this.speed;
                const moveY = (dy / dist) * this.speed;
                
                // 群体避让
                const separation = this.calculateSeparation(game.zombies);
                this.x += (moveX + separation.x * 50) * dt;
                this.y += (moveY + separation.y * 50) * dt;
            } else {
                this.state = ZOMBIE_STATE.ATTACK;
                this.attackCooldown = 0;
            }
        }

        updateStunned(dt) {
            this.stunTimer -= dt;
            if (this.stunTimer <= 0) {
                this.state = ZOMBIE_STATE.CHASE;
            }
        }
        
        updateAttack(dt, game) {
            this.attackCooldown -= dt;
            
            if (!game.player) return;
            
            const dx = game.player.centerX - this.centerX;
            const dy = game.player.centerY - this.centerY;
            const dist = Math.sqrt(dx * dx + dy * dy);
            
            this.rotation = Math.atan2(dy, dx);
            
            if (dist > this.attackRange * 1.5) {
                this.state = ZOMBIE_STATE.CHASE;
                return;
            }
            
            if (this.attackCooldown <= 0) {
                this.performAttack(game);
                this.attackCooldown = CONFIG.ZOMBIE_ATTACK_COOLDOWN;
            }
        }
        
        performAttack(game) {
            if (!game.player) return;
            
            const dx = game.player.centerX - this.centerX;
            const dy = game.player.centerY - this.centerY;
            const dist = Math.sqrt(dx * dx + dy * dy);
            
            if (dist < this.attackRange + game.player.width / 2) {
                const knockbackDir = new Vector2(dx, dy).normalize();
                const playerDead = game.player.takeDamage(
                    this.damage, 
                    knockbackDir, 
                    CONFIG.ZOMBIE_KNOCKBACK_FORCE
                );
                
                game.triggerDamageFlash();
                game.viewport.shake(8);
                
                if (playerDead) {
                    game.gameOver();
                }
            }
        }
        
        updateDying(dt) {
            this.deathTimer -= dt;
            if (this.deathTimer <= 0) {
                this.state = ZOMBIE_STATE.DEAD;
                this.active = false;
            }
        }
        
        calculateSeparation(zombies) {
            const separation = new Vector2();
            let count = 0;
            
            for (const other of zombies) {
                if (other === this || !other.active) continue;
                
                const dx = this.centerX - other.centerX;
                const dy = this.centerY - other.centerY;
                const dist = Math.sqrt(dx * dx + dy * dy);
                const minDist = (this.width + other.width) / 2;
                
                if (dist < minDist && dist > 0) {
                    separation.x += dx / dist;
                    separation.y += dy / dist;
                    count++;
                }
            }
            
            if (count > 0) {
                separation.scale(1 / count);
            }
            
            return separation;
        }

        applyKnockback(dt) {
            this.x += this.knockbackVx * dt;
            this.y += this.knockbackVy * dt;
            this.knockbackVx *= 0.85;
            this.knockbackVy *= 0.85;
        }
        
        constrainToWorld() {
            this.x = Utils.clamp(this.x, 0, CONFIG.WORLD_WIDTH - this.width);
            this.y = Utils.clamp(this.y, 0, CONFIG.WORLD_HEIGHT - this.height);
        }

        takeDamage(damage, bullet, game) {
            if (this.state === ZOMBIE_STATE.DYING || this.state === ZOMBIE_STATE.DEAD) {
                return false;
            }
            
            let actualDamage = damage;
            let isHeadshot = false;

            // 爆头检测
            if (bullet && bullet.checkHeadshot(this)) {
                actualDamage *= CONFIG.BULLET_HEADSHOT_MULTIPLIER;
                isHeadshot = true;
                game.headshots++;
            }
            
            this.health -= actualDamage;
            this.hitFlashTimer = 0.1;
            
            // 击退效果
            const knockbackDir = new Vector2(
                this.centerX - bullet.centerX,
                this.centerY - bullet.centerY
            ).normalize();
            
            this.knockbackVx = knockbackDir.x * CONFIG.ZOMBIE_KNOCKBACK_FORCE;
            this.knockbackVy = knockbackDir.y * CONFIG.ZOMBIE_KNOCKBACK_FORCE;
            
            // 停顿效果
            this.stunTimer = CONFIG.ZOMBIE_HIT_STUN_DURATION;
            this.state = ZOMBIE_STATE.STUNNED;
            
            // 生成伤害数字
            game.damageNumbers.push(new DamageNumber(
                this.centerX,
                this.y - 10,
                actualDamage,
                isHeadshot
            ));
            
            // 生成血液粒子
            this.spawnBloodParticles(game, bullet);
            
            game.viewport.shake(isHeadshot ? 6 : 3);
            
            if (this.health <= 0) {
                this.die(game);
                return true;
            }
            
            return false;
        }

        spawnBloodParticles(game, bullet) {
            const angle = Math.atan2(this.centerY - bullet.centerY, this.centerX - bullet.centerX);
            
            for (let i = 0; i < CONFIG.BLOOD_PARTICLE_COUNT; i++) {
                game.bloodParticles.push(new BloodParticle(
                    this.centerX,
                    this.centerY,
                    angle + Utils.randomRange(-0.5, 0.5)
                ));
            }
        }
        
        die(game) {
            this.state = ZOMBIE_STATE.DYING;
            this.deathTimer = 0.3;
            game.kills++;
            game.score += Math.round(this.maxHealth * 10);
        }
        
        render(ctx, viewport) {
            if (!viewport.isVisible(this.x, this.y, this.width, this.height)) return;
            
            const screenPos = viewport.worldToScreen(this.x, this.y);
            const screenCenterX = screenPos.x + this.width / 2;
            const screenCenterY = screenPos.y + this.height / 2;
            
            ctx.save();
            ctx.translate(screenCenterX, screenCenterY);
            
            // 生成动画
            if (this.state === ZOMBIE_STATE.SPAWNING) {
                const scale = 1 - (this.spawnTimer / 0.5);
                ctx.scale(scale, scale);
                ctx.globalAlpha = scale;
            }
            
            // 死亡动画
            if (this.state === ZOMBIE_STATE.DYING) {
                const progress = 1 - (this.deathTimer / 0.3);
                ctx.globalAlpha = 1 - progress;
                ctx.rotate(progress * Math.PI / 4);
                ctx.scale(1, 1 - progress * 0.5);
            }
            
            ctx.rotate(this.rotation);
            
            // 受击闪烁
            if (this.hitFlashTimer > 0) {
                ctx.fillStyle = '#ffffff';
            } else {
                ctx.fillStyle = this.type.color;
            }
            
            // 僵尸身体
            ctx.shadowColor = this.type.glowColor;
            ctx.shadowBlur = 10;
            ctx.beginPath();
            ctx.arc(0, 0, this.width / 2, 0, Math.PI * 2);
            ctx.fill();
            
            ctx.shadowBlur = 0;
            
            // 僵尸细节
            ctx.fillStyle = 'rgba(0, 0, 0, 0.3)';
            ctx.beginPath();
            ctx.arc(0, 0, this.width / 2 - 5, 0, Math.PI * 2);
            ctx.fill();
            
            // 眼睛
            ctx.fillStyle = '#ff3333';
            ctx.shadowColor = '#ff0000';
            ctx.shadowBlur = 5;
            ctx.beginPath();
            ctx.arc(this.width * 0.2, -this.width * 0.15, 3, 0, Math.PI * 2);
            ctx.arc(this.width * 0.2, this.width * 0.15, 3, 0, Math.PI * 2);
            ctx.fill();
            
            ctx.restore();
            
            // 血量条
            if (this.state !== ZOMBIE_STATE.SPAWNING && 
                this.state !== ZOMBIE_STATE.DYING && 
                this.health < this.maxHealth) {
                this.renderHealthBar(ctx, screenPos);
            }
        }
        
        renderHealthBar(ctx, screenPos) {
            const barWidth = this.width;
            const barHeight = 4;
            const barY = screenPos.y - 10;
            const barX = screenPos.x;
            
            ctx.fillStyle = 'rgba(0, 0, 0, 0.5)';
            ctx.fillRect(barX, barY, barWidth, barHeight);
            
            const healthPercent = this.health / this.maxHealth;
            ctx.fillStyle = healthPercent > 0.5 ? '#4ecdc4' : 
                           healthPercent > 0.25 ? '#ffd93d' : '#ff3d3d';
            ctx.fillRect(barX, barY, barWidth * healthPercent, barHeight);
        }
    }

    // ============================================
    // 波次管理器
    // ============================================
    class WaveManager {
        constructor() {
            this.currentWave = 0;
            this.zombiesRemaining = 0;
            this.zombiesToSpawn = 0;
            this.spawnTimer = 0;
            this.waveTimer = 0;
            this.isWaveActive = false;
            this.isBetweenWaves = true;
        }
        
        startNextWave(game) {
            this.currentWave++;
            this.zombiesToSpawn = CONFIG.WAVE_BASE_ZOMBIES + 
                                 (this.currentWave - 1) * CONFIG.WAVE_ZOMBIE_INCREMENT;
            this.zombiesRemaining = this.zombiesToSpawn;
            this.isWaveActive = true;
            this.isBetweenWaves = false;
            this.spawnTimer = 0;
        }
        
        update(dt, game) {
            if (this.isBetweenWaves) {
                this.waveTimer -= dt;
                if (this.waveTimer <= 0) {
                    this.startNextWave(game);
                }
                return;
            }
            
            if (this.zombiesToSpawn > 0) {
                this.spawnTimer -= dt;
                if (this.spawnTimer <= 0) {
                    this.spawnZombie(game);
                    this.spawnTimer = CONFIG.SPAWN_INTERVAL;
                }
            }
            
            const aliveZombies = game.zombies.filter(z => 
                z.active && z.state !== ZOMBIE_STATE.DEAD
            ).length;
            
            this.zombiesRemaining = aliveZombies + this.zombiesToSpawn;
            
            if (this.zombiesToSpawn <= 0 && aliveZombies === 0) {
                this.isWaveActive = false;
                this.isBetweenWaves = true;
                this.waveTimer = CONFIG.WAVE_DELAY;
            }
        }
        
        spawnZombie(game) {
            if (game.zombies.length >= CONFIG.MAX_ZOMBIES) return;
            
            const type = Utils.weightedRandom(Object.values(ZOMBIE_TYPES));
            const size = CONFIG.ZOMBIE_BASE_SIZE * type.size;
            
            let x, y;
            const side = Utils.randomInt(0, 3);
            const margin = 50;
            
            switch (side) {
                case 0:
                    x = Utils.randomRange(0, CONFIG.WORLD_WIDTH);
                    y = -margin;
                    break;
                case 1:
                    x = CONFIG.WORLD_WIDTH + margin;
                    y = Utils.randomRange(0, CONFIG.WORLD_HEIGHT);
                    break;
                case 2:
                    x = Utils.randomRange(0, CONFIG.WORLD_WIDTH);
                    y = CONFIG.WORLD_HEIGHT + margin;
                    break;
                case 3:
                    x = -margin;
                    y = Utils.randomRange(0, CONFIG.WORLD_HEIGHT);
                    break;
            }
            
            const zombie = new Zombie(x, y, type);
            game.zombies.push(zombie);
            this.zombiesToSpawn--;
        }
    }

    class Game {
        constructor() {
            this.canvas = document.getElementById('gameCanvas');
            this.ctx = this.canvas.getContext('2d');
            
            this.resize();
            window.addEventListener('resize', () => this.resize());
            
            this.input = new InputManager(this.canvas);
            this.viewport = new Viewport(this.canvas, CONFIG.WORLD_WIDTH, CONFIG.WORLD_HEIGHT);
            this.waveManager = new WaveManager();
            
            this.player = null;
            this.bullets = [];
            this.zombies = [];
            this.damageNumbers = [];
            this.bloodParticles = [];
            
            this.kills = 0;
            this.headshots = 0;
            this.score = 0;
            this.isGameOver = false;
            
            this.lastTime = 0;
            this.gridPattern = null;
            
            this.init();
        }
        
        resize() {
            this.canvas.width = window.innerWidth;
            this.canvas.height = window.innerHeight;
            this.createGridPattern();
        }
        
        createGridPattern() {
            const gridCanvas = document.createElement('canvas');
            gridCanvas.width = CONFIG.GRID_SIZE;
            gridCanvas.height = CONFIG.GRID_SIZE;
            const gridCtx = gridCanvas.getContext('2d');
            
            gridCtx.strokeStyle = 'rgba(255, 255, 255, 0.03)';
            gridCtx.lineWidth = 1;
            gridCtx.beginPath();
            gridCtx.moveTo(CONFIG.GRID_SIZE, 0);
            gridCtx.lineTo(CONFIG.GRID_SIZE, CONFIG.GRID_SIZE);
            gridCtx.moveTo(0, CONFIG.GRID_SIZE);
            gridCtx.lineTo(CONFIG.GRID_SIZE, CONFIG.GRID_SIZE);
            gridCtx.stroke();
            
            this.gridPattern = this.ctx.createPattern(gridCanvas, 'repeat');
        }
        
        init() {
            this.player = new Player(CONFIG.WORLD_WIDTH / 2, CONFIG.WORLD_HEIGHT / 2);
            this.bullets = [];
            this.zombies = [];
            this.damageNumbers = [];
            this.bloodParticles = [];
            this.kills = 0;
            this.headshots = 0;
            this.score = 0;
            this.isGameOver = false;
            
            this.waveManager.currentWave = 0;
            this.waveManager.waveTimer = 1;
            this.waveManager.isBetweenWaves = true;
            
            document.getElementById('gameOverScreen').classList.remove('active');
            
            this.lastTime = performance.now();
            requestAnimationFrame((t) => this.gameLoop(t));
        }
        
        gameLoop(currentTime) {
            const dt = Math.min((currentTime - this.lastTime) / 1000, 0.05);
            this.lastTime = currentTime;
            
            if (!this.isGameOver) {
                this.update(dt);
            }
            
            this.render();
            this.updateUI();
            
            requestAnimationFrame((t) => this.gameLoop(t));
        }
        
        update(dt) {
            this.player.update(dt, this);
            this.viewport.follow(this.player, dt);
            
            this.waveManager.update(dt, this);
            
            // 更新子弹并检测碰撞
            for (let i = this.bullets.length - 1; i >= 0; i--) {
                const bullet = this.bullets[i];
                bullet.update(dt, this);
                
                if (!bullet.active) {
                    this.bullets.splice(i, 1);
                    continue;
                }
                
                // 子弹与僵尸碰撞检测
                for (const zombie of this.zombies) {
                    if (!zombie.active || zombie.state === ZOMBIE_STATE.DEAD ||
                        zombie.state === ZOMBIE_STATE.DYING) continue;
                    
                    if (bullet.hitEntities.has(zombie)) continue;
                    
                    const bulletRadius = CONFIG.BULLET_SIZE;
                    const zombieRadius = zombie.width / 2;
                    
                    if (Utils.circleCollision(
                        bullet.centerX, bullet.centerY, bulletRadius,
                        zombie.centerX, zombie.centerY, zombieRadius
                    )) {
                        const killed = zombie.takeDamage(bullet.damage, bullet, this);
                        bullet.hitEntities.add(zombie);
                        bullet.piercedCount++;
                        
                        if (bullet.piercedCount >= bullet.maxPierce) {
                            bullet.active = false;
                        }
                        break;
                    }
                }
            }
            
            // 更新僵尸
            for (let i = this.zombies.length - 1; i >= 0; i--) {
                const zombie = this.zombies[i];
                zombie.update(dt, this);
                
                if (!zombie.active) {
                    this.zombies.splice(i, 1);
                }
            }
            
            // 更新伤害数字
            for (let i = this.damageNumbers.length - 1; i >= 0; i--) {
                const dn = this.damageNumbers[i];
                dn.update(dt);
                
                if (!dn.active) {
                    this.damageNumbers.splice(i, 1);
                }
            }
            
            // 更新血液粒子
            for (let i = this.bloodParticles.length - 1; i >= 0; i--) {
                const bp = this.bloodParticles[i];
                bp.update(dt);
                
                if (!bp.active) {
                    this.bloodParticles.splice(i, 1);
                }
            }
        }
        
        render() {
            this.ctx.fillStyle = '#050607';
            this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);
            
            this.renderBackground();
            this.renderBloodParticles();
            this.renderZombies();
            this.renderPlayer();
            this.renderBullets();
            this.renderDamageNumbers();
        }
        
        renderBackground() {
            this.ctx.save();
            this.ctx.translate(-this.viewport.x % CONFIG.GRID_SIZE, 
                              -this.viewport.y % CONFIG.GRID_SIZE);
            
            if (this.gridPattern) {
                this.ctx.fillStyle = this.gridPattern;
                this.ctx.fillRect(
                    -CONFIG.GRID_SIZE, 
                    -CONFIG.GRID_SIZE, 
                    this.canvas.width + CONFIG.GRID_SIZE * 2, 
                    this.canvas.height + CONFIG.GRID_SIZE * 2
                );
            }
            this.ctx.restore();
            
            // 边界指示器
            const border = 5;
            this.ctx.strokeStyle = '#ff3d3d';
            this.ctx.lineWidth = border;
            this.ctx.strokeRect(
                -this.viewport.x + border / 2, 
                -this.viewport.y + border / 2, 
                CONFIG.WORLD_WIDTH - border, 
                CONFIG.WORLD_HEIGHT - border
            );
        }
        
        renderPlayer() {
            if (this.player) {
                this.player.render(this.ctx, this.viewport);
            }
        }
        
        renderZombies() {
            for (const zombie of this.zombies) {
                zombie.render(this.ctx, this.viewport);
            }
        }
        
        renderBullets() {
            for (const bullet of this.bullets) {
                bullet.render(this.ctx, this.viewport);
            }
        }

        renderBloodParticles() {
            for (const bp of this.bloodParticles) {
                bp.render(this.ctx, this.viewport);
            }
        }

        renderDamageNumbers() {
            for (const dn of this.damageNumbers) {
                dn.render(this.ctx, this.viewport);
            }
        }
        
        updateUI() {
            const healthPercent = (this.player.health / this.player.maxHealth) * 100;
            document.getElementById('healthFill').style.width = `${healthPercent}%`;
            
            const dashPercent = Math.max(0, (1 - this.player.dashCooldown / this.player.dashCooldownTime)) * 100;
            document.getElementById('dashFill').style.width = `${dashPercent}%`;
            
            const ammoDisplay = document.getElementById('ammoCount');
            if (this.player.isReloading) {
                ammoDisplay.innerHTML = '<span class="ammo-reloading">RELOADING</span>';
            } else {
                ammoDisplay.innerHTML = `
                    <span class="ammo-current">${this.player.currentAmmo}</span>
                    <span class="ammo-separator">/</span>
                    <span class="ammo-reserve">${this.player.reserveAmmo}</span>

            }
            
            document.getElementById('scoreDisplay').textContent = this.score;

## 实现说明

### 碰撞检测系统

1. **圆形碰撞检测**：使用 `Utils.circleCollision()` 和 `Utils.pointInCircle()` 实现精确的圆形碰撞判定

2. **爆头判定**：子弹类中的 `checkHeadshot()` 方法检测子弹是否击中僵尸头部区域（头部位于僵尸朝向的前方30%位置）

### 伤害系统

1. **基础伤害**：子弹默认造成25点伤害
2. **爆头加成**：爆头造成2.5倍伤害，并显示特殊的"HEADSHOT!"文字
3. **伤害数字**：击中时浮动显示伤害数值，爆头伤害显示为橙色

### 受击反馈

1. **僵尸击退**：被击中时向子弹相反方向弹出，击退力度200
2. **僵尸停顿**：被击中时进入眩晕状态0.15秒
3. **血液粒子**：击中时喷射8个血液粒子
4. **受击闪烁**：僵尸被击中时短暂变白

### 玩家受伤系统

1. **无敌帧**：受伤后1.5秒无敌时间，期间角色闪烁
2. **玩家击退**：受伤时被击退，但有30%抗性
3. **伤害闪烁**：屏幕边缘红色闪烁效果
4. **Game Over**：血量归零时显示结算界面

### 操作说明

- **WASD/方向键**：移动
- **鼠标**：瞄准和射击
- **空格**：闪避（期间无敌）
- **R**：换弹
