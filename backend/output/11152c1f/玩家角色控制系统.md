    const CONFIG = {

    // ============================================
    // 工具函数
    // ============================================
    const Utils = {
        lerp(a, b, t) {
            return a + (b - a) * t;
        },
        
        clamp(val, min, max) {
            return Math.max(min, Math.min(max, val));
        },
        
        distance(x1, y1, x2, y2) {
            const dx = x2 - x1;
            const dy = y2 - y1;
            return Math.sqrt(dx * dx + dy * dy);
        },
        
        angle(x1, y1, x2, y2) {
            return Math.atan2(y2 - y1, x2 - x1);
        },
        
        randomRange(min, max) {
            return min + Math.random() * (max - min);
        },
        
        randomInt(min, max) {
            return Math.floor(Math.random() * (max - min + 1)) + min;
        }
    };

    // ============================================
    // 向量类
    // ============================================
    class Vector2 {
        constructor(x = 0, y = 0) {
            this.x = x;
            this.y = y;
        }
        
        set(x, y) {
            this.x = x;
            this.y = y;
            return this;
        }
        
        copy(v) {
            this.x = v.x;
            this.y = v.y;
            return this;
        }
        
        clone() {
            return new Vector2(this.x, this.y);
        }
        
        add(v) {
            this.x += v.x;
            this.y += v.y;
            return this;
        }
        
        sub(v) {
            this.x -= v.x;
            this.y -= v.y;
            return this;
        }
        
        scale(s) {
            this.x *= s;
            this.y *= s;
            return this;
        }
        
        normalize() {
            const len = this.length();
            if (len > 0) {
                this.x /= len;
                this.y /= len;
            }
            return this;
        }
        
        length() {
            return Math.sqrt(this.x * this.x + this.y * this.y);
        }
        
        dot(v) {
            return this.x * v.x + this.y * v.y;
        }
    }

    // ============================================
    // 输入管理器
    // ============================================
    class InputManager {
        constructor(canvas) {
            this.canvas = canvas;
            this.keys = {};
            this.mouse = {
                x: 0,
                y: 0,
                worldX: 0,
                worldY: 0,
                leftDown: false,
                rightDown: false
            };
            
            this.bindEvents();
        }
        
        bindEvents() {
            window.addEventListener('keydown', (e) => {
                this.keys[e.code] = true;
                if (['Space', 'KeyW', 'KeyA', 'KeyS', 'KeyD', 'ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight'].includes(e.code)) {
                    e.preventDefault();
                }
            });
            
            window.addEventListener('keyup', (e) => {
                this.keys[e.code] = false;
            });
            
            this.canvas.addEventListener('mousemove', (e) => {
                const rect = this.canvas.getBoundingClientRect();
                this.mouse.x = e.clientX - rect.left;
                this.mouse.y = e.clientY - rect.top;
            });
            
            this.canvas.addEventListener('mousedown', (e) => {
                if (e.button === 0) this.mouse.leftDown = true;
                if (e.button === 2) this.mouse.rightDown = true;
            });
            
            this.canvas.addEventListener('mouseup', (e) => {
                if (e.button === 0) this.mouse.leftDown = false;
                if (e.button === 2) this.mouse.rightDown = false;
            });
            
            this.canvas.addEventListener('contextmenu', (e) => e.preventDefault());
        }
        
        isKeyDown(code) {
            return this.keys[code] === true;
        }
        
        updateWorldMouse(viewport) {
            this.mouse.worldX = this.mouse.x + viewport.x;
            this.mouse.worldY = this.mouse.y + viewport.y;
        }
    }

    // ============================================
    // 视口管理器
    // ============================================
    class Viewport {
        constructor(canvas, worldWidth, worldHeight) {
            this.canvas = canvas;
            this.worldWidth = worldWidth;
            this.worldHeight = worldHeight;
            this.x = 0;
            this.y = 0;
            this.targetX = 0;
            this.targetY = 0;
            this.smoothing = 0.08;
            this.shakeAmount = 0;
            this.shakeDecay = 0.9;
        }
        
        follow(target, dt) {
            const targetCenterX = target.x - this.canvas.width / 2 + target.width / 2;
            const targetCenterY = target.y - this.canvas.height / 2 + target.height / 2;
            
            this.targetX = Utils.clamp(targetCenterX, 0, this.worldWidth - this.canvas.width);
            this.targetY = Utils.clamp(targetCenterY, 0, this.worldHeight - this.canvas.height);
            
            this.x = Utils.lerp(this.x, this.targetX, this.smoothing);
            this.y = Utils.lerp(this.y, this.targetY, this.smoothing);
            
            // 屏幕震动
            if (this.shakeAmount > 0.5) {
                this.x += Utils.randomRange(-this.shakeAmount, this.shakeAmount);
                this.y += Utils.randomRange(-this.shakeAmount, this.shakeAmount);
                this.shakeAmount *= this.shakeDecay;
            } else {
                this.shakeAmount = 0;
            }
        }
        
        shake(amount) {
            this.shakeAmount = amount;
        }
        
        worldToScreen(worldX, worldY) {
            return {
                x: worldX - this.x,
                y: worldY - this.y
            };
        }
        
        screenToWorld(screenX, screenY) {
            return {
                x: screenX + this.x,
                y: screenY + this.y
            };
        }
        
        isVisible(x, y, width, height) {
            const margin = 100;
            return x + width + margin > this.x && 
                   x - margin < this.x + this.canvas.width &&
                   y + height + margin > this.y &&
                   y - margin < this.y + this.canvas.height;
        }
    }

    class Entity {
        constructor(x, y, width, height) {
            this.x = x;
            this.y = y;
            this.width = width;
            this.height = height;
            this.vx = 0;
            this.vy = 0;
            this.active = true;
            this.rotation = 0;
            this.layer = 0;
        }
        
        get centerX() {
            return this.x + this.width / 2;
        }
        
        get centerY() {
            return this.y + this.height / 2;
        }
        
        update(dt, game) {
            // 子类实现
        }
        
        render(ctx, viewport) {
            // 子类实现
        }
        
        collidesWith(other) {
            const dx = this.centerX - other.centerX;
            const dy = this.centerY - other.centerY;
            const dist = Math.sqrt(dx * dx + dy * dy);
            const combinedRadius = (this.width + other.width) / 2;
            return dist < combinedRadius;
        }
    }

    // ============================================
    // 玩家类 - 带物理加速度系统
    // ============================================
    class Player extends Entity {
        constructor(x, y) {
            super(x, y, CONFIG.PLAYER_SIZE, CONFIG.PLAYER_SIZE);
            this.speed = CONFIG.PLAYER_SPEED;
            this.health = 100;
            this.maxHealth = 100;
            this.score = 0;
            this.dashCooldown = 0;
            this.dashDuration = 0;
            this.isDashing = false;
            this.dashDir = new Vector2();
            this.shootCooldown = 0;
            this.invincibleTime = 0;
            this.animTime = 0;
            
            // 物理参数
            this.acceleration = CONFIG.PLAYER_ACCELERATION;
            this.friction = CONFIG.PLAYER_FRICTION;
            this.targetVx = 0;
            this.targetVy = 0;
        }
        
        update(dt, game) {
            this.animTime += dt;
            
            // 移动输入
            const moveDir = new Vector2();
            if (game.input.isKeyDown('KeyW') || game.input.isKeyDown('ArrowUp')) moveDir.y -= 1;
            if (game.input.isKeyDown('KeyS') || game.input.isKeyDown('ArrowDown')) moveDir.y += 1;
            if (game.input.isKeyDown('KeyA') || game.input.isKeyDown('ArrowLeft')) moveDir.x -= 1;
            if (game.input.isKeyDown('KeyD') || game.input.isKeyDown('ArrowRight')) moveDir.x += 1;
            
            // 冲刺冷却
            if (this.dashCooldown > 0) this.dashCooldown -= dt;
            
            // 冲刺状态
            if (this.dashDuration > 0) {
                this.dashDuration -= dt;
                this.vx = this.dashDir.x * CONFIG.PLAYER_DASH_SPEED;
                this.vy = this.dashDir.y * CONFIG.PLAYER_DASH_SPEED;
                if (this.dashDuration <= 0) {
                    this.isDashing = false;
                }
            } else {
                // 正常移动 - 带加速度
                if (moveDir.length() > 0) {
                    moveDir.normalize();
                    this.targetVx = moveDir.x * this.speed;
                    this.targetVy = moveDir.y * this.speed;
                } else {
                    this.targetVx = 0;
                    this.targetVy = 0;
                }
                
                // 应用加速度/摩擦力
                if (Math.abs(this.vx - this.targetVx) > 0.1) {
                    this.vx = Utils.lerp(this.vx, this.targetVx, this.acceleration * dt / 1000);
                } else {
                    this.vx *= (1 - this.friction * dt);
                }
                
                if (Math.abs(this.vy - this.targetVy) > 0.1) {
                    this.vy = Utils.lerp(this.vy, this.targetVy, this.acceleration * dt / 1000);
                } else {
                    this.vy *= (1 - this.friction * dt);
                }
                
                // 冲刺触发
                if (game.input.isKeyDown('Space') && this.dashCooldown <= 0 && moveDir.length() > 0) {
                    this.isDashing = true;
                    this.dashDuration = 0.15;
                    this.dashCooldown = 1.2;
                    this.dashDir.copy(moveDir);
                    this.invincibleTime = 0.25;
                    game.viewport.shake(5);
                }
            }
            
            // 更新位置
            this.x += this.vx * dt;
            this.y += this.vy * dt;
            
            // 世界边界碰撞
            const margin = 5;
            if (this.x < margin) {
                this.x = margin;
                this.vx = 0;
            }
            if (this.x > CONFIG.WORLD_WIDTH - this.width - margin) {
                this.x = CONFIG.WORLD_WIDTH - this.width - margin;
                this.vx = 0;
            }
            if (this.y < margin) {
                this.y = margin;
                this.vy = 0;
            }
            if (this.y > CONFIG.WORLD_HEIGHT - this.height - margin) {
                this.y = CONFIG.WORLD_HEIGHT - this.height - margin;
                this.vy = 0;
            }
            
            // 朝向鼠标
            this.rotation = Utils.angle(
                this.centerX, this.centerY,
                game.input.mouse.worldX, game.input.mouse.worldY
            );
            
            // 射击
            if (this.shootCooldown > 0) this.shootCooldown -= dt;
            if (game.input.mouse.leftDown && this.shootCooldown <= 0) {
                this.shoot(game);
            }
            
            // 无敌时间
            if (this.invincibleTime > 0) this.invincibleTime -= dt;
        }
        
        shoot(game) {
            this.shootCooldown = 0.15;
            
            const bulletX = this.centerX + Math.cos(this.rotation) * 24;
            const bulletY = this.centerY + Math.sin(this.rotation) * 24;
            
            game.addBullet(bulletX, bulletY, this.rotation);
            game.viewport.shake(4);
        }
        
        takeDamage(amount, game) {
            if (this.invincibleTime > 0 || this.isDashing) return;
            
            this.health -= amount;
            this.invincibleTime = 0.6;
            game.viewport.shake(10);
            
            if (this.health <= 0) {
                this.health = 0;
                this.active = false;
            }
        }
        
        render(ctx, viewport) {
            const screenPos = viewport.worldToScreen(this.x, this.y);
            const centerX = screenPos.x + this.width / 2;
            const centerY = screenPos.y + this.height / 2;
            
            ctx.save();
            ctx.translate(centerX, centerY);
            ctx.rotate(this.rotation);
            
            // 冲刺残影效果
            if (this.isDashing) {
                ctx.globalAlpha = 0.3;
                for (let i = 1; i <= 3; i++) {
                    const offsetX = -this.dashDir.x * i * 12;
                    const offsetY = -this.dashDir.y * i * 12;
                    ctx.save();
                    ctx.translate(offsetX, offsetY);
                    this.drawBody(ctx, 0.2);
                    ctx.restore();
                }
                ctx.globalAlpha = 1;
            }
            
            // 无敌闪烁
            if (this.invincibleTime > 0 && !this.isDashing) {
                ctx.globalAlpha = 0.5 + Math.sin(this.animTime * 20) * 0.3;
            }
            
            // 绘制玩家主体
            this.drawBody(ctx, 1);
            
            // 绘制瞄准线
            this.drawAimLine(ctx);
            
            ctx.restore();
        }
        
        drawBody(ctx, alpha) {
            ctx.globalAlpha = alpha;
            
            // 外发光
            const gradient = ctx.createRadialGradient(0, 0, 0, 0, 0, 30);
            gradient.addColorStop(0, 'rgba(78, 205, 196, 0.3)');
            gradient.addColorStop(1, 'rgba(78, 205, 196, 0)');
            ctx.fillStyle = gradient;
            ctx.beginPath();
            ctx.arc(0, 0, 30, 0, Math.PI * 2);
            ctx.fill();
            
            // 主体圆形
            ctx.fillStyle = '#1a2f3f';
            ctx.strokeStyle = '#4ecdc4';
            ctx.lineWidth = 3;
            ctx.beginPath();
            ctx.arc(0, 0, 12, 0, Math.PI * 2);
            ctx.fill();
            ctx.stroke();
            
            // 内部装饰
            ctx.fillStyle = '#4ecdc4';
            ctx.beginPath();
            ctx.arc(0, 0, 4, 0, Math.PI * 2);
            ctx.fill();
            
            // 武器指向
            ctx.fillStyle = '#2d4a5e';
            ctx.strokeStyle = '#4ecdc4';
            ctx.lineWidth = 2;
            ctx.beginPath();
            ctx.roundRect(8, -4, 18, 8, 2);
            ctx.fill();
            ctx.stroke();
            
            // 武器发光点
            ctx.fillStyle = '#ffd93d';
            ctx.beginPath();
            ctx.arc(22, 0, 2, 0, Math.PI * 2);
            ctx.fill();
            
            ctx.globalAlpha = 1;
        }
        
        drawAimLine(ctx) {
            ctx.strokeStyle = 'rgba(78, 205, 196, 0.2)';
            ctx.lineWidth = 1;
            ctx.setLineDash([5, 5]);
            ctx.beginPath();
            ctx.moveTo(26, 0);
            ctx.lineTo(80, 0);
            ctx.stroke();
            ctx.setLineDash([]);
        }
        
        getDashProgress() {
            if (this.dashCooldown <= 0) return 1;
            return 1 - (this.dashCooldown / 1.2);
        }
    }

    // ============================================
    // 子弹类
    // ============================================
    class Bullet extends Entity {
        constructor(x, y, angle) {
            super(x - CONFIG.BULLET_SIZE / 2, y - CONFIG.BULLET_SIZE / 2, CONFIG.BULLET_SIZE, CONFIG.BULLET_SIZE);
            this.vx = Math.cos(angle) * CONFIG.BULLET_SPEED;
            this.vy = Math.sin(angle) * CONFIG.BULLET_SPEED;
            this.rotation = angle;
            this.damage = 25;
            this.trail = [];
            this.maxTrailLength = 8;
        }
        
        update(dt, game) {
            // 保存轨迹
            this.trail.push({ x: this.centerX, y: this.centerY });
            if (this.trail.length > this.maxTrailLength) {
                this.trail.shift();
            }
            
            this.x += this.vx * dt;
            this.y += this.vy * dt;
            
            // 超出世界边界或静止则失效
            if (this.x < -50 || this.x > CONFIG.WORLD_WIDTH + 50 ||
                this.y < -50 || this.y > CONFIG.WORLD_HEIGHT + 50) {
                this.active = false;
            }
        }
        
        render(ctx, viewport) {
            const screenPos = viewport.worldToScreen(this.x, this.y);
            const centerX = screenPos.x + this.width / 2;
            const centerY = screenPos.y + this.height / 2;
            
            // 绘制轨迹
            if (this.trail.length > 1) {
                ctx.beginPath();
                for (let i = 0; i < this.trail.length; i++) {
                    const pos = viewport.worldToScreen(this.trail[i].x, this.trail[i].y);
                    if (i === 0) {
                        ctx.moveTo(pos.x, pos.y);
                    } else {
                        ctx.lineTo(pos.x, pos.y);
                    }
                }
                ctx.strokeStyle = 'rgba(255, 217, 61, 0.4)';
                ctx.lineWidth = 3;
                ctx.lineCap = 'round';
                ctx.stroke();
            }
            
            // 外发光
            const gradient = ctx.createRadialGradient(centerX, centerY, 0, centerX, centerY, 12);
            gradient.addColorStop(0, 'rgba(255, 217, 61, 0.6)');
            gradient.addColorStop(1, 'rgba(255, 217, 61, 0)');
            ctx.fillStyle = gradient;
            ctx.beginPath();
            ctx.arc(centerX, centerY, 12, 0, Math.PI * 2);
            ctx.fill();
            
            // 子弹主体
            ctx.fillStyle = '#ffd93d';
            ctx.beginPath();
            ctx.arc(centerX, centerY, 4, 0, Math.PI * 2);
            ctx.fill();
            
            // 高光
            ctx.fillStyle = '#fff';
            ctx.beginPath();
            ctx.arc(centerX - 1, centerY - 1, 1.5, 0, Math.PI * 2);
            ctx.fill();
        }
    }

    // ============================================
    // 僵尸类
    // ============================================
    class Zombie extends Entity {
        constructor(x, y) {
            super(x, y, CONFIG.ZOMBIE_SIZE, CONFIG.ZOMBIE_SIZE);
            this.speed = Utils.randomRange(CONFIG.ZOMBIE_SPEED * 0.7, CONFIG.ZOMBIE_SPEED * 1.3);
            this.health = 50;
            this.maxHealth = 50;
            this.damage = 15;
            this.attackCooldown = 0;
            this.animTime = Math.random() * Math.PI * 2;
            this.wobble = 0;
        }
        
        update(dt, game) {
            if (!this.active) return;
            
            this.animTime += dt * 3;
            this.wobble = Math.sin(this.animTime) * 0.1;
            
            // 追踪玩家
            if (game.player && game.player.active) {
                const dx = game.player.centerX - this.centerX;
                const dy = game.player.centerY - this.centerY;
                const dist = Math.sqrt(dx * dx + dy * dy);
                
                if (dist > 0) {
                    this.vx = (dx / dist) * this.speed;
                    this.vy = (dy / dist) * this.speed;
                    this.rotation = Math.atan2(dy, dx);
                }
                
                // 攻击判定
                if (this.attackCooldown > 0) {
                    this.attackCooldown -= dt;
                }
                
                if (dist < this.width && this.attackCooldown <= 0) {
                    game.player.takeDamage(this.damage, game);
                    this.attackCooldown = 1.0;
                }
            }
            
            this.x += this.vx * dt;
            this.y += this.vy * dt;
            
            // 边界限制
            this.x = Utils.clamp(this.x, 0, CONFIG.WORLD_WIDTH - this.width);
            this.y = Utils.clamp(this.y, 0, CONFIG.WORLD_HEIGHT - this.height);
        }
        
        takeDamage(amount) {
            this.health -= amount;
            if (this.health <= 0) {
                this.active = false;
                return true;
            }
            return false;
        }
        
        render(ctx, viewport) {
            if (!viewport.isVisible(this.x, this.y, this.width, this.height)) return;
            
            const screenPos = viewport.worldToScreen(this.x, this.y);
            const centerX = screenPos.x + this.width / 2;
            const centerY = screenPos.y + this.height / 2;
            
            ctx.save();
            ctx.translate(centerX, centerY);
            ctx.rotate(this.rotation + this.wobble);
            
            // 外发光
            const gradient = ctx.createRadialGradient(0, 0, 0, 0, 0, 25);
            gradient.addColorStop(0, 'rgba(122, 158, 90, 0.2)');
            gradient.addColorStop(1, 'rgba(122, 158, 90, 0)');
            ctx.fillStyle = gradient;
            ctx.beginPath();
            ctx.arc(0, 0, 25, 0, Math.PI * 2);
            ctx.fill();
            
            // 僵尸主体
            ctx.fillStyle = '#2d3a2a';
            ctx.strokeStyle = '#7a9e5a';
            ctx.lineWidth = 2;
            ctx.beginPath();
            ctx.arc(0, 0, 14, 0, Math.PI * 2);
            ctx.fill();
            ctx.stroke();
            
            // 眼睛
            ctx.fillStyle = '#ff3d3d';
            ctx.beginPath();
            ctx.arc(-4, -4, 3, 0, Math.PI * 2);
            ctx.fill();
            ctx.beginPath();
            ctx.arc(4, -4, 3, 0, Math.PI * 2);
            ctx.fill();
            
            // 血条
            if (this.health < this.maxHealth) {
                const barWidth = 24;
                const barHeight = 4;
                const healthPercent = this.health / this.maxHealth;
                
                ctx.fillStyle = 'rgba(0,0,0,0.5)';
                ctx.fillRect(-barWidth/2, -22, barWidth, barHeight);
                
                ctx.fillStyle = '#ff3d3d';
                ctx.fillRect(-barWidth/2, -22, barWidth * healthPercent, barHeight);
            }
            
            ctx.restore();
        }
    }

    // ============================================
    // 粒子效果类
    // ============================================
    class Particle extends Entity {
        constructor(x, y, vx, vy, color, size, life) {
            super(x, y, size, size);
            this.vx = vx;
            this.vy = vy;
            this.color = color;
            this.life = life;
            this.maxLife = life;
            this.gravity = 200;
        }
        
        update(dt, game) {
            this.life -= dt;
            if (this.life <= 0) {
                this.active = false;
                return;
            }
            
            this.vy += this.gravity * dt;
            this.x += this.vx * dt;
            this.y += this.vy * dt;
        }
        
        render(ctx, viewport) {
            const alpha = this.life / this.maxLife;
            const screenPos = viewport.worldToScreen(this.x, this.y);
            
            ctx.globalAlpha = alpha;
            ctx.fillStyle = this.color;
            ctx.beginPath();
            ctx.arc(screenPos.x + this.width/2, screenPos.y + this.height/2, this.width/2 * alpha, 0, Math.PI * 2);
            ctx.fill();
            ctx.globalAlpha = 1;
        }
    }

    // ============================================
    // 游戏主类
    // ============================================
    class Game {
        constructor(canvas) {
            this.canvas = canvas;
            this.ctx = canvas.getContext('2d');
            
            // 初始化尺寸
            this.resize();
            window.addEventListener('resize', () => this.resize());
            
            // 核心系统
            this.input = new InputManager(canvas);
            this.viewport = new Viewport(canvas, CONFIG.WORLD_WIDTH, CONFIG.WORLD_HEIGHT);
            
            // 游戏实体
            this.player = null;
            this.bullets = [];
            this.zombies = [];
            this.particles = [];
            
            // 游戏状态
            this.score = 0;
            this.wave = 1;
            this.spawnTimer = 0;
            this.isGameOver = false;
            this.isPaused = false;
            
            // 性能监控
            this.lastTime = 0;
            this.fps = 60;
            this.fpsBuffer = [];
            
            // 初始化游戏
            this.init();
        }
        
        resize() {
            this.canvas.width = window.innerWidth;
            this.canvas.height = window.innerHeight;
        }
        
        init() {
            // 创建玩家在世界中心
            this.player = new Player(
                CONFIG.WORLD_WIDTH / 2,
                CONFIG.WORLD_HEIGHT / 2
            );
            
            this.bullets = [];
            this.zombies = [];
            this.particles = [];
            this.score = 0;
            this.wave = 1;
            this.spawnTimer = 0;
            this.isGameOver = false;
            
            // 初始僵尸
            for (let i = 0; i < 5; i++) {
                this.spawnZombie();
            }
            
            // 隐藏游戏结束界面
            document.getElementById('gameOver').style.display = 'none';
        }
        
        addBullet(x, y, angle) {
            this.bullets.push(new Bullet(x, y, angle));
        }
        
        addParticle(x, y, vx, vy, color, size, life) {
            this.particles.push(new Particle(x, y, vx, vy, color, size, life));
        }
        
        spawnZombie() {
            // 在玩家周围生成，但不在视野内
            let x, y;
            const minDist = 400;
            const maxDist = 800;
            
            do {
                const angle = Math.random() * Math.PI * 2;
                const dist = Utils.randomRange(minDist, maxDist);
                x = this.player.centerX + Math.cos(angle) * dist;
                y = this.player.centerY + Math.sin(angle) * dist;
            } while (
                x < 50 || x > CONFIG.WORLD_WIDTH - 50 ||
                y < 50 || y > CONFIG.WORLD_HEIGHT - 50
            );
            
            this.zombies.push(new Zombie(x, y));
        }
        
        spawnParticles(x, y, count, color) {
            for (let i = 0; i < count; i++) {
                const angle = Math.random() * Math.PI * 2;
                const speed = Utils.randomRange(50, 200);
                const vx = Math.cos(angle) * speed;
                const vy = Math.sin(angle) * speed - 50;
                const size = Utils.randomRange(2, 5);
                const life = Utils.randomRange(0.3, 0.6);
                this.addParticle(x, y, vx, vy, color, size, life);
            }
        }
        
        update(dt) {
            if (this.isGameOver || this.isPaused) return;
            
            // 更新输入的世界坐标
            this.input.updateWorldMouse(this.viewport);
            
            // 更新玩家
            if (this.player && this.player.active) {
                this.player.update(dt, this);
            } else if (this.player && !this.player.active && !this.isGameOver) {
                this.gameOver();
            }
            
            // 更新子弹
            this.bullets.forEach(bullet => bullet.update(dt, this));
            this.bullets = this.bullets.filter(b => b.active);
            
            // 更新僵尸
            this.zombies.forEach(zombie => zombie.update(dt, this));
            this.zombies = this.zombies.filter(z => z.active);
            
            // 更新粒子
            this.particles.forEach(particle => particle.update(dt, this));
            this.particles = this.particles.filter(p => p.active);
            
            // 子弹与僵尸碰撞检测
            this.checkBulletCollisions();
            
            // 僵尸生成
            this.spawnTimer += dt * 1000;
            const spawnRate = Math.max(500, CONFIG.SPAWN_RATE - this.wave * 150);
            if (this.spawnTimer >= spawnRate && this.zombies.length < CONFIG.MAX_ZOMBIES) {
                this.spawnZombie();
                this.spawnTimer = 0;
            }
            
            // 波次递增
            if (this.score > 0 && this.score % 500 === 0 && this.zombies.length === 0) {
                this.wave++;
            }
            
            // 更新视口
            if (this.player && this.player.active) {
                this.viewport.follow(this.player, dt);
            }
            
            // 更新UI
            this.updateUI();
        }
        
        checkBulletCollisions() {
            for (const bullet of this.bullets) {
                if (!bullet.active) continue;
                
                for (const zombie of this.zombies) {
                    if (!zombie.active) continue;
                    
                    if (bullet.collidesWith(zombie)) {
                        bullet.active = false;
                        
                        if (zombie.takeDamage(bullet.damage)) {
                            // 僵尸死亡
                            this.score += 100;
                            this.spawnParticles(zombie.centerX, zombie.centerY, 10, '#7a9e5a');
                            this.viewport.shake(4);
                        } else {
                            // 僵尸受伤
                            this.spawnParticles(bullet.centerX, bullet.centerY, 5, '#ffd93d');
                        }
                        break;
                    }
                }
            }
        }
        
        render() {
            const ctx = this.ctx;
            
            // 清空画布
            ctx.fillStyle = '#050607';
            ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);
            
            // 绘制背景网格
            this.renderGrid(ctx);
            
            // 绘制世界边界
            this.renderWorldBorder(ctx);
            
            // 绘制粒子（底层）
            this.particles.forEach(p => p.render(ctx, this.viewport));
            
            // 绘制子弹
            this.bullets.forEach(b => b.render(ctx, this.viewport));
            
            // 绘制僵尸
            this.zombies.forEach(z => z.render(ctx, this.viewport));
            
            // 绘制玩家
            if (this.player && this.player.active) {
                this.player.render(ctx, this.viewport);
            }
        }
        
        renderGrid(ctx) {
            const gridSize = CONFIG.GRID_SIZE;
            ctx.strokeStyle = 'rgba(255,255,255,0.03)';
            ctx.lineWidth = 1;
            
            const startX = Math.floor(this.viewport.x / gridSize) * gridSize;
            const startY = Math.floor(this.viewport.y / gridSize) * gridSize;
            const endX = startX + this.canvas.width + gridSize;
            const endY = startY + this.canvas.height + gridSize;
            
            ctx.beginPath();
            for (let x = startX; x <= endX; x += gridSize) {
                const screenX = x - this.viewport.x;
                ctx.moveTo(screenX, 0);
                ctx.lineTo(screenX, this.canvas.height);
            }
            for (let y = startY; y <= endY; y += gridSize) {
                const screenY = y - this.viewport.y;
                ctx.moveTo(0, screenY);
                ctx.lineTo(this.canvas.width, screenY);
            }
            ctx.stroke();
        }
        
        renderWorldBorder(ctx) {
            const topLeft = this.viewport.worldToScreen(0, 0);
            const bottomRight = this.viewport.worldToScreen(CONFIG.WORLD_WIDTH, CONFIG.WORLD_HEIGHT);
            
            ctx.strokeStyle = '#ff3d3d';
            ctx.lineWidth = 4;
            ctx.setLineDash([20, 10]);
            ctx.strokeRect(topLeft.x, topLeft.y, bottomRight.x - topLeft.x, bottomRight.y - topLeft.y);
            ctx.setLineDash([]);
        }
        
        updateUI() {
            document.getElementById('score').textContent = this.score;
            document.getElementById('wave').textContent = this.wave;
            document.getElementById('healthFill').style.width = `${this.player ? this.player.health : 0}%`;
            document.getElementById('dashFill').style.width = `${this.player ? this.player.getDashProgress() * 100 : 0}%`;
            document.getElementById('objectCount').textContent = this.bullets.length + this.zombies.length + this.particles.length;
            
            // 调试面板
            if (this.player) {
                document.getElementById('playerVel').textContent = 
                    `${Math.round(this.player.vx)}, ${Math.round(this.player.vy)}`;
                document.getElementById('playerPos').textContent = 
                    `${Math.round(this.player.x)}, ${Math.round(this.player.y)}`;
                document.getElementById('aimAngle').textContent = 
                    `${(this.player.rotation * 180 / Math.PI).toFixed(1)}°`;
            }
            document.getElementById('zombieCount').textContent = this.zombies.length;
        }
        
        gameOver() {
            this.isGameOver = true;
            document.getElementById('finalScore').textContent = this.score;
            document.getElementById('gameOver').style.display = 'block';
        }
        
        restart() {
            this.init();
        }
        
        loop(currentTime) {
            // 计算帧时间
            const dt = Math.min((currentTime - this.lastTime) / 1000, 0.1);
            this.lastTime = currentTime;
            
            // FPS计算
            this.fpsBuffer.push(1 / dt);
            if (this.fpsBuffer.length > 60) this.fpsBuffer.shift();
            this.fps = Math.round(this.fpsBuffer.reduce((a, b) => a + b) / this.fpsBuffer.length);
            document.getElementById('fpsValue').textContent = this.fps;
            
            // 游戏循环
            this.update(dt);
            this.render();
            
            requestAnimationFrame((t) => this.loop(t));
        }
        
        start() {
            this.lastTime = performance.now();
            requestAnimationFrame((t) => this.loop(t));
        }
    }

    // ============================================
    // 启动游戏
    // ============================================
    window.addEventListener('DOMContentLoaded', () => {
        const canvas = document.getElementById('gameCanvas');

        game.start();
        
        // 重新开始按钮
        document.getElementById('restartBtn').addEventListener('click', () => {
            game.restart();
        });
    });
</script>
