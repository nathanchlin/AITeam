            const dx = x2 - x1;
            const dy = y2 - y1;
            return Math.sqrt(dx * dx + dy * dy);

        weightedRandom(items) {
            const totalWeight = items.reduce((sum, item) => sum + item.weight, 0);
            let random = Math.random() * totalWeight;
            for (const item of items) {
                random -= item.weight;
                if (random <= 0) return item;
            }
            return items[items.length - 1];
        },

        circleCollision(x1, y1, r1, x2, y2, r2) {
            const dx = x2 - x1;
            const dy = y2 - y1;
            const dist = Math.sqrt(dx * dx + dy * dy);
            return dist < r1 + r2;
        },

        pointInCircle(px, py, cx, cy, r) {
            const dx = px - cx;
            const dy = py - cy;
            return dx * dx + dy * dy < r * r;

            this.y *= s;
            return this;
        }
        
        normalize() {
            const len = this.length();
            if (len > 0) {
                this.x /= len;
                this.y /= len;
            }
            return this;
        }
        
        length() {
            return Math.sqrt(this.x * this.x + this.y * this.y);
        }
        
        dot(v) {
            return this.x * v.x + this.y * v.y;
        }
    }

    class ScreenShake {
        constructor() {
            this.intensity = 0;
            this.duration = 0;
            this.timer = 0;
            this.offsetX = 0;
            this.offsetY = 0;
        }

        shake(intensity, duration = CONFIG.SCREEN_SHAKE_DURATION) {
            this.intensity = Math.max(this.intensity, intensity);
            this.duration = Math.max(this.duration, duration);
            this.timer = this.duration;
        }

        update(dt) {
            if (this.timer > 0) {
                this.timer -= dt;
                const progress = this.timer / this.duration;
                const currentIntensity = this.intensity * progress;
                this.offsetX = (Math.random() * 2 - 1) * currentIntensity;
                this.offsetY = (Math.random() * 2 - 1) * currentIntensity;
                
                if (this.timer <= 0) {
                    this.intensity = 0;
                    this.offsetX = 0;
                    this.offsetY = 0;
                }
            }
        }

        apply(ctx) {
            ctx.save();
            ctx.translate(this.offsetX, this.offsetY);
        }

        reset(ctx) {
            ctx.restore();
        }
    }

        constructor(x, y, angle, speed, color, size) {
            this.x = x;
            this.y = y;
            this.vx = Math.cos(angle) * speed;
            this.vy = Math.sin(angle) * speed;
            this.color = color;
            this.size = size;

            this.gravity = 400;
            this.ground = CONFIG.WORLD_HEIGHT;
        }

        update(dt) {
            this.x += this.vx * dt;
            this.y += this.vy * dt;
            this.vy += this.gravity * dt;
            this.life -= this.decay * dt;
            
            // 地面碰撞
            if (this.y > this.ground) {
                this.y = this.ground;
                this.vy *= -0.3;
                this.vx *= 0.5;
            }
        }

        render(ctx, camera) {
            if (this.life <= 0) return;
            const screenX = this.x - camera.x;
            const screenY = this.y - camera.y;
            
            ctx.globalAlpha = this.life;
            ctx.fillStyle = this.color;
            ctx.beginPath();
            ctx.arc(screenX, screenY, this.size * this.life, 0, Math.PI * 2);
            ctx.fill();
            ctx.globalAlpha = 1;
        }

        isDead() {
            return this.life <= 0;
        }
    }

    }

    class WaveManager {
        constructor() {
            this.currentWave = 0;
            this.waveState = 'idle';
            this.zombiesToSpawn = 0;
            this.zombiesRemaining = 0;
            this.spawnTimer = 0;
            this.countdownTimer = 0;
            this.waveTimer = 0;
            this.spawnQueue = [];
            this.announced = false;
        }

        startNextWave() {
            this.currentWave++;
            this.waveState = 'countdown';
            this.countdownTimer = CONFIG.WAVE_DELAY;
            this.zombiesToSpawn = this.calculateZombieCount();
            this.zombiesRemaining = this.zombiesToSpawn;
            this.spawnQueue = this.generateSpawnQueue();
            this.announced = false;
            
            return {
                wave: this.currentWave,
                zombieCount: this.zombiesToSpawn,
                countdown: CONFIG.WAVE_DELAY
            };
        }

        calculateZombieCount() {
            const baseCount = CONFIG.WAVE_BASE_ZOMBIES;
            const increment = CONFIG.WAVE_ZOMBIE_INCREMENT * (this.currentWave - 1);
            const exponentialGrowth = Math.floor(this.currentWave * this.currentWave * 0.1);
            return Math.min(baseCount + increment + exponentialGrowth, CONFIG.MAX_ZOMBIES);
        }

        generateSpawnQueue() {
            const queue = [];
            const types = Object.values(ZOMBIE_TYPES);
            
            const adjustedTypes = types.map(type => {
                let weight = type.weight;
                
                if (this.currentWave >= 3 && type.name === 'FAST') {
                    weight += (this.currentWave - 2) * 3;
                }
                if (this.currentWave >= 5 && type.name === 'TANK') {
                    weight += (this.currentWave - 4) * 2;
                }
                
                return { ...type, weight };
            });
            
            for (let i = 0; i < this.zombiesToSpawn; i++) {
                const type = Utils.weightedRandom(adjustedTypes);
                queue.push(type);
            }
            
            return queue;
        }

        update(dt, player, zombies) {
            this.waveTimer += dt;
            
            switch (this.waveState) {
                case 'countdown':
                    this.countdownTimer -= dt;
                    if (this.countdownTimer <= 0) {
                        this.waveState = 'spawning';
                        this.waveTimer = 0;
                    }
                    break;
                    
                case 'spawning':
                    if (this.spawnQueue.length > 0) {
                        this.spawnTimer -= dt;
                        if (this.spawnTimer <= 0) {
                            const zombieType = this.spawnQueue.shift();
                            zombies.spawnZombie(player, zombieType, this.currentWave);
                            this.spawnTimer = CONFIG.SPAWN_INTERVAL;
                        }
                    } else {
                        this.waveState = 'active';
                    }
                    break;
                    
                case 'active':
                    if (zombies.getActiveCount() === 0 && this.spawnQueue.length === 0) {
                        this.waveState = 'complete';
                        return { completed: true, waveTime: this.waveTimer };
                    }
                    break;
            }
            
            return null;
        }

        zombieKilled() {
            this.zombiesRemaining--;
        }

        isCountdown() {
            return this.waveState === 'countdown';
        }

        isActive() {
            return this.waveState === 'active' || this.waveState === 'spawning';
        }
    }

    class AmmoSystem {
        constructor() {
            this.magazineSize = CONFIG.MAGAZINE_SIZE;
            this.currentAmmo = this.magazineSize;
            this.reserveAmmo = CONFIG.RESERVE_AMMO;
            this.isReloading = false;
            this.reloadTimer = 0;
            this.reloadTime = CONFIG.RELOAD_TIME;
        }

        canShoot() {
            return !this.isReloading && this.currentAmmo > 0;
        }

        shoot() {
            if (this.canShoot()) {
                this.currentAmmo--;
                if (this.currentAmmo === 0 && this.reserveAmmo > 0) {
                    this.startReload();
                }
                return true;
            }
            return false;
        }

        startReload() {
            if (this.isReloading || this.currentAmmo === this.magazineSize || this.reserveAmmo === 0) {
                return false;
            }
            this.isReloading = true;
            this.reloadTimer = this.reloadTime;
            return true;
        }

        update(dt) {
            if (this.isReloading) {
                this.reloadTimer -= dt;
                if (this.reloadTimer <= 0) {
                    this.finishReload();
                }
            }
        }

        finishReload() {
            const ammoNeeded = this.magazineSize - this.currentAmmo;
            const ammoToLoad = Math.min(ammoNeeded, this.reserveAmmo);
            this.currentAmmo += ammoToLoad;
            this.reserveAmmo -= ammoToLoad;
            this.isReloading = false;
        }

        addAmmo(amount) {
            this.reserveAmmo += amount;
        }
    }

    class Bullet {
        constructor(x, y, angle, damage) {
            this.x = x;
            this.y = y;
            this.angle = angle;
            this.speed = CONFIG.BULLET_SPEED;
            this.damage = damage;
            this.size = CONFIG.BULLET_SIZE;
            this.trail = [];
            this.trailLength = CONFIG.BULLET_TRAIL_LENGTH;
            this.active = true;
            this.lifetime = 3;
        }

        update(dt) {
            // 保存轨迹
            this.trail.unshift({ x: this.x, y: this.y });
            if (this.trail.length > this.trailLength) {
                this.trail.pop();
            }
            
            const vx = Math.cos(this.angle) * this.speed;
            const vy = Math.sin(this.angle) * this.speed;
            
            this.x += vx * dt;
            this.y += vy * dt;
            
            this.lifetime -= dt;
            if (this.lifetime <= 0) {
                this.active = false;
            }
            
            // 边界检测
            if (this.x < 0 || this.x > CONFIG.WORLD_WIDTH ||
                this.y < 0 || this.y > CONFIG.WORLD_HEIGHT) {
                this.active = false;
            }
        }

        render(ctx, camera) {
            const screenX = this.x - camera.x;
            const screenY = this.y - camera.y;
            
            // 渲染轨迹
            for (let i = 0; i < this.trail.length; i++) {
                const point = this.trail[i];
                const alpha = 1 - (i / this.trail.length);
                const size = this.size * (1 - i / this.trail.length * 0.5);
                
                ctx.globalAlpha = alpha * 0.6;
                ctx.fillStyle = CONFIG.BULLET_COLOR || '#ffd93d';
                ctx.beginPath();
                ctx.arc(point.x - camera.x, point.y - camera.y, size, 0, Math.PI * 2);
                ctx.fill();
            }
            
            // 渲染子弹主体
            ctx.globalAlpha = 1;
            
            // 发光效果
            const gradient = ctx.createRadialGradient(screenX, screenY, 0, screenX, screenY, this.size * 3);
            gradient.addColorStop(0, 'rgba(255, 217, 61, 0.8)');
            gradient.addColorStop(0.5, 'rgba(255, 217, 61, 0.3)');
            gradient.addColorStop(1, 'rgba(255, 217, 61, 0)');
            ctx.fillStyle = gradient;
            ctx.beginPath();
            ctx.arc(screenX, screenY, this.size * 3, 0, Math.PI * 2);
            ctx.fill();
            
            // 子弹核心
            ctx.fillStyle = '#ffffff';
            ctx.beginPath();
            ctx.arc(screenX, screenY, this.size, 0, Math.PI * 2);
            ctx.fill();
        }
    }

    class Player {
        constructor(x, y) {
            this.x = x;
            this.y = y;
            this.vx = 0;
            this.vy = 0;
            this.size = CONFIG.PLAYER_SIZE;
            this.health = CONFIG.PLAYER_MAX_HEALTH;
            this.maxHealth = CONFIG.PLAYER_MAX_HEALTH;
            this.angle = 0;
            this.isDashing = false;
            this.dashTimer = 0;
            this.dashCooldown = 0;
            this.dashDuration = 0.15;
            this.dashCooldownTime = 1.0;
            this.dashDirection = new Vector2();
            this.isInvincible = false;
            this.invincibleTimer = 0;
            this.invincibleBlink = false;
            this.muzzleFlash = new MuzzleFlash();
            this.ammo = new AmmoSystem();
            this.fireTimer = 0;
        }

        update(dt, input, camera) {
            // 冲刺
            if (this.isDashing) {
                this.dashTimer -= dt;
                if (this.dashTimer <= 0) {
                    this.isDashing = false;
                }
                this.vx = this.dashDirection.x * CONFIG.PLAYER_DASH_SPEED;
                this.vy = this.dashDirection.y * CONFIG.PLAYER_DASH_SPEED;
            } else {
                // 正常移动
                const moveX = (input.keys.d || input.keys.arrowright ? 1 : 0) - (input.keys.a || input.keys.arrowleft ? 1 : 0);
                const moveY = (input.keys.s || input.keys.arrowdown ? 1 : 0) - (input.keys.w || input.keys.arrowup ? 1 : 0);
                
                const moveDir = new Vector2(moveX, moveY);
                if (moveDir.length() > 0) {
                    moveDir.normalize();
                    this.vx += moveDir.x * CONFIG.PLAYER_ACCELERATION * dt;
                    this.vy += moveDir.y * CONFIG.PLAYER_ACCELERATION * dt;
                }
                
                // 摩擦力
                this.vx /= (1 + CONFIG.PLAYER_FRICTION * dt);
                this.vy /= (1 + CONFIG.PLAYER_FRICTION * dt);
                
                // 速度限制
                const speed = Math.sqrt(this.vx * this.vx + this.vy * this.vy);
                if (speed > CONFIG.PLAYER_SPEED) {
                    this.vx = (this.vx / speed) * CONFIG.PLAYER_SPEED;
                    this.vy = (this.vy / speed) * CONFIG.PLAYER_SPEED;
                }
            }
            
            // 冲刺冷却
            if (this.dashCooldown > 0) {
                this.dashCooldown -= dt;
            }
            
            // 无敌时间
            if (this.isInvincible) {
                this.invincibleTimer -= dt;
                this.invincibleBlink = Math.floor(this.invincibleTimer * 10) % 2 === 0;
                if (this.invincibleTimer <= 0) {
                    this.isInvincible = false;
                    this.invincibleBlink = false;
                }
            }
            
            // 射击冷却
            if (this.fireTimer > 0) {
                this.fireTimer -= dt;
            }
            
            // 弹药系统更新
            this.ammo.update(dt);
            this.muzzleFlash.update(dt);
            
            // 更新位置
            this.x += this.vx * dt;
            this.y += this.vy * dt;
            
            // 边界限制
            this.x = Utils.clamp(this.x, this.size, CONFIG.WORLD_WIDTH - this.size);
            this.y = Utils.clamp(this.y, this.size, CONFIG.WORLD_HEIGHT - this.size);
            
            // 朝向鼠标
            const mouseX = input.mouseX + camera.x;
            const mouseY = input.mouseY + camera.y;
            this.angle = Math.atan2(mouseY - this.y, mouseX - this.x);
        }

        dash(input) {
            if (this.dashCooldown > 0 || this.isDashing) return false;
            
            const moveX = (input.keys.d || input.keys.arrowright ? 1 : 0) - (input.keys.a || input.keys.arrowleft ? 1 : 0);
            const moveY = (input.keys.s || input.keys.arrowdown ? 1 : 0) - (input.keys.w || input.keys.arrowup ? 1 : 0);
            
            this.dashDirection.set(moveX, moveY);
            if (this.dashDirection.length() === 0) {
                this.dashDirection.set(Math.cos(this.angle), Math.sin(this.angle));
            }
            this.dashDirection.normalize();
            
            this.isDashing = true;
            this.dashTimer = this.dashDuration;
            this.dashCooldown = this.dashCooldownTime;
            
            return true;
        }

        shoot(bullets) {
            if (!this.ammo.canShoot() || this.fireTimer > 0) return false;
            
            this.ammo.shoot();
            this.fireTimer = CONFIG.FIRE_RATE / 1000;
            
            const bulletX = this.x + Math.cos(this.angle) * (this.size + 10);
            const bulletY = this.y + Math.sin(this.angle) * (this.size + 10);
            
            bullets.push(new Bullet(bulletX, bulletY, this.angle, CONFIG.BULLET_DAMAGE));
            
            this.muzzleFlash.fire(bulletX, bulletY, this.angle);
            
            return true;
        }

        takeDamage(amount, knockbackX = 0, knockbackY = 0) {
            if (this.isInvincible || this.isDashing) return false;
            
            this.health -= amount;
            this.isInvincible = true;
            this.invincibleTimer = CONFIG.PLAYER_INVINCIBLE_TIME;
            
            // 击退
            const knockbackScale = 1 - CONFIG.PLAYER_KNOCKBACK_RESIST;
            this.vx += knockbackX * knockbackScale;
            this.vy += knockbackY * knockbackScale;
            
            return true;
        }

        render(ctx, camera) {
            const screenX = this.x - camera.x;
            const screenY = this.y - camera.y;
            
            if (this.invincibleBlink) return;
            
            ctx.save();
            ctx.translate(screenX, screenY);
            
            // 冲刺时的拖影效果
            if (this.isDashing) {
                ctx.globalAlpha = 0.3;
                ctx.fillStyle = '#4ecdc4';
                ctx.beginPath();
                ctx.arc(-this.vx * 0.02, -this.vy * 0.02, this.size, 0, Math.PI * 2);
                ctx.fill();
                ctx.globalAlpha = 1;
            }
            
            // 玩家发光
            const glowGradient = ctx.createRadialGradient(0, 0, 0, 0, 0, this.size * 2);
            glowGradient.addColorStop(0, 'rgba(78, 205, 196, 0.4)');
            glowGradient.addColorStop(1, 'rgba(78, 205, 196, 0)');
            ctx.fillStyle = glowGradient;
            ctx.beginPath();
            ctx.arc(0, 0, this.size * 2, 0, Math.PI * 2);
            ctx.fill();
            
            // 玩家身体
            ctx.rotate(this.angle);
            
            ctx.fillStyle = '#4ecdc4';
            ctx.beginPath();
            ctx.arc(0, 0, this.size, 0, Math.PI * 2);
            ctx.fill();
            
            // 玩家边框
            ctx.strokeStyle = '#3dbdb5';
            ctx.lineWidth = 3;
            ctx.stroke();
            
            // 武器指示
            ctx.fillStyle = '#2a9d8f';
            ctx.fillRect(this.size - 5, -4, 20, 8);
            
            ctx.restore();
            
            // 枪口闪光
            this.muzzleFlash.render(ctx, camera);
        }

        isDead() {
            return this.health <= 0;
        }
    }

    // ============================================
    // 僵尸类（带溶解效果）
    // ============================================
    class Zombie {
        constructor(x, y, type, wave) {
            this.x = x;
            this.y = y;
            this.type = type;
            this.wave = wave;
            
            // 根据类型和波次计算属性
            const healthScale = 1 + (wave - 1) * CONFIG.WAVE_HEALTH_SCALE;
            const speedScale = 1 + (wave - 1) * CONFIG.WAVE_SPEED_SCALE;
            
            this.size = CONFIG.ZOMBIE_BASE_SIZE * type.size;
            this.health = CONFIG.ZOMBIE_BASE_HEALTH * type.health * healthScale;
            this.maxHealth = this.health;
            this.speed = CONFIG.ZOMBIE_BASE_SPEED * type.speed * speedScale;
            this.damage = CONFIG.ZOMBIE_ATTACK_DAMAGE * type.damage;
            
            this.state = ZOMBIE_STATE.SPAWNING;
            this.spawnTimer = 0.5;
            this.angle = 0;
            this.vx = 0;
            this.vy = 0;
            
            this.attackCooldown = 0;
            this.stunTimer = 0;
            
            // 动画
            this.bobOffset = Math.random() * Math.PI * 2;
            this.bobSpeed = 3 + Math.random() * 2;
            
            // 溶解效果
            this.dissolveTimer = 0;
            this.dissolveDuration = CONFIG.ZOMBIE_DISSOLVE_DURATION;
            
            // 生成位置渐变
            this.spawnAlpha = 0;
        }

        update(dt, player) {
            switch (this.state) {
                case ZOMBIE_STATE.SPAWNING:
                    this.spawnTimer -= dt;
                    this.spawnAlpha = 1 - (this.spawnTimer / 0.5);
                    if (this.spawnTimer <= 0) {
                        this.state = ZOMBIE_STATE.CHASE;
                        this.spawnAlpha = 1;
                    }
                    break;
                    
                case ZOMBIE_STATE.STUNNED:
                    this.stunTimer -= dt;
                    if (this.stunTimer <= 0) {
                        this.state = ZOMBIE_STATE.CHASE;
                    }
                    // 击退减速
                    this.vx *= 0.9;
                    this.vy *= 0.9;
                    break;
                    
                case ZOMBIE_STATE.CHASE:
                    this.chasePlayer(dt, player);
                    break;
                    
                case ZOMBIE_STATE.ATTACK:
                    this.attackCooldown -= dt;
                    if (this.attackCooldown <= 0) {
                        this.state = ZOMBIE_STATE.CHASE;
                    }
                    break;
                    
                case ZOMBIE_STATE.DYING:
                    this.dissolveTimer += dt;
                    if (this.dissolveTimer >= this.dissolveDuration) {
                        this.state = ZOMBIE_STATE.DEAD;
                    }
                    break;
            }
            
            // 应用速度
            this.x += this.vx * dt;
            this.y += this.vy * dt;
            
            // 边界限制
            this.x = Utils.clamp(this.x, this.size, CONFIG.WORLD_WIDTH - this.size);
            this.y = Utils.clamp(this.y, this.size, CONFIG.WORLD_HEIGHT - this.size);
        }

        chasePlayer(dt, player) {
            const dx = player.x - this.x;
            const dy = player.y - this.y;
            const dist = Math.sqrt(dx * dx + dy * dy);
            
            if (dist > 0) {
                this.angle = Math.atan2(dy, dx);
                
                if (dist > CONFIG.ZOMBIE_ATTACK_RANGE) {
                    this.vx = (dx / dist) * this.speed;
                    this.vy = (dy / dist) * this.speed;
                } else {
                    this.vx *= 0.8;
                    this.vy *= 0.8;
                }
            }
        }

        takeDamage(damage, angle, isHeadshot = false) {
            const finalDamage = isHeadshot ? damage * CONFIG.BULLET_HEADSHOT_MULTIPLIER : damage;
            this.health -= finalDamage;
            
            // 击退
            const knockback = CONFIG.ZOMBIE_KNOCKBACK_FORCE;
            this.vx = Math.cos(angle) * knockback;
            this.vy = Math.sin(angle) * knockback;
            
            // 硬直
            this.stunTimer = CONFIG.ZOMBIE_HIT_STUN_DURATION;
            this.state = ZOMBIE_STATE.STUNNED;
            
            if (this.health <= 0) {
                this.startDying();
            }
            
            return { damage: finalDamage, isHeadshot, killed: this.health <= 0 };
        }

        startDying() {
            this.state = ZOMBIE_STATE.DYING;
            this.dissolveTimer = 0;
        }

        tryAttack(player) {
            if (this.state !== ZOMBIE_STATE.CHASE) return null;
            if (this.attackCooldown > 0) return null;
            
            const dist = Utils.distance(this.x, this.y, player.x, player.y);
            
            if (dist < CONFIG.ZOMBIE_ATTACK_RANGE + this.size + player.size) {
                this.state = ZOMBIE_STATE.ATTACK;
                this.attackCooldown = CONFIG.ZOMBIE_ATTACK_COOLDOWN;
                
                const angle = Math.atan2(player.y - this.y, player.x - this.x);
                const knockbackX = Math.cos(angle) * 300;
                const knockbackY = Math.sin(angle) * 300;
                
                return { damage: this.damage, knockbackX, knockbackY };
            }
            
            return null;
        }

        render(ctx, camera, time) {
            if (this.state === ZOMBIE_STATE.DEAD) return;
            
            const screenX = this.x - camera.x;
            const screenY = this.y - camera.y;
            
            ctx.save();
            ctx.translate(screenX, screenY);
            
            // 生成和溶解效果
            if (this.state === ZOMBIE_STATE.SPAWNING) {
                ctx.globalAlpha = this.spawnAlpha;
            } else if (this.state === ZOMBIE_STATE.DYING) {
                const dissolveProgress = this.dissolveTimer / this.dissolveDuration;
                ctx.globalAlpha = 1 - dissolveProgress;
                
                // 溶解时的抖动
                const shake = Math.sin(this.dissolveTimer * 30) * (5 * dissolveProgress);
                ctx.translate(shake, 0);
            }
            
            // 硬直闪烁
            if (this.state === ZOMBIE_STATE.STUNNED) {
                ctx.fillStyle = '#ffffff';
            }
            
            // 浮动动画
            const bob = Math.sin(time * this.bobSpeed + this.bobOffset) * 3;
            ctx.translate(0, bob);
            
            // 僵尸发光
            const glowGradient = ctx.createRadialGradient(0, 0, 0, 0, 0, this.size * 2);
            glowGradient.addColorStop(0, this.type.glowColor);
            glowGradient.addColorStop(1, 'rgba(0, 0, 0, 0)');
            ctx.fillStyle = glowGradient;
            ctx.beginPath();
            ctx.arc(0, 0, this.size * 2, 0, Math.PI * 2);
            ctx.fill();
            
            // 僵尸身体
            ctx.rotate(this.angle);
            
            // 溶解效果 - 使用噪声
            if (this.state === ZOMBIE_STATE.DYING) {
                const dissolveProgress = this.dissolveTimer / this.dissolveDuration;
                ctx.globalCompositeOperation = 'source-over';
                
                // 绘制破碎效果
                ctx.beginPath();
                const segments = 8;
                for (let i = 0; i < segments; i++) {
                    const a = (i / segments) * Math.PI * 2;
                    const nextA = ((i + 1) / segments) * Math.PI * 2;
                    
                    // 随机溶解边缘
                    const dissolve = Math.random() < dissolveProgress ? 0 : 1;
                    const r = this.size * (dissolve ? Utils.randomRange(0.3, 1) : 1);
                    
                    ctx.lineTo(Math.cos(a) * r, Math.sin(a) * r);
                }
                ctx.closePath();
            } else {
                ctx.beginPath();
                ctx.arc(0, 0, this.size, 0, Math.PI * 2);
            }
            
            ctx.fillStyle = this.state === ZOMBIE_STATE.STUNNED ? '#ffffff' : this.type.color;
            ctx.fill();
            
            ctx.strokeStyle = 'rgba(0, 0, 0, 0.3)';
            ctx.lineWidth = 2;
            ctx.stroke();
            
            // 眼睛
            ctx.fillStyle = '#ff3333';
            ctx.beginPath();
            ctx.arc(this.size * 0.4, -this.size * 0.3, 3, 0, Math.PI * 2);
            ctx.arc(this.size * 0.4, this.size * 0.3, 3, 0, Math.PI * 2);
            ctx.fill();
            
            ctx.restore();
            
            // 生命条
            if (this.health < this.maxHealth && this.state !== ZOMBIE_STATE.DYING) {
                const barWidth = this.size * 2;
                const barHeight = 4;
                const barY = -this.size - 15 + bob;
                
                ctx.fillStyle = 'rgba(0, 0, 0, 0.5)';
                ctx.fillRect(screenX - barWidth / 2, barY, barWidth, barHeight);
                
                ctx.fillStyle = '#ff3d3d';
                ctx.fillRect(screenX - barWidth / 2, barY, barWidth * (this.health / this.maxHealth), barHeight);
            }
        }

        isDead() {
            return this.state === ZOMBIE_STATE.DEAD;
        }

        canBeHit() {
            return this.state !== ZOMBIE_STATE.DYING && 
                   this.state !== ZOMBIE_STATE.DEAD &&
                   this.state !== ZOMBIE_STATE.SPAWNING;
        }
    }

    class ZombieManager {
        constructor() {
            this.zombies = [];
        }

        spawnZombie(player, type, wave) {
            // 在玩家周围生成，但不在视野内
            const minDist = 500;
            const maxDist = 800;
            const angle = Math.random() * Math.PI * 2;
            const dist = Utils.randomRange(minDist, maxDist);
            
            let x = player.x + Math.cos(angle) * dist;
            let y = player.y + Math.sin(angle) * dist;
            
            // 确保在世界范围内
            x = Utils.clamp(x, 50, CONFIG.WORLD_WIDTH - 50);
            y = Utils.clamp(y, 50, CONFIG.WORLD_HEIGHT - 50);
            
            this.zombies.push(new Zombie(x, y, type, wave));
        }

        update(dt, player, bloodSystem, screenShake, scoreSystem, killPopupManager, camera) {
            for (let i = this.zombies.length - 1; i >= 0; i--) {
                const zombie = this.zombies[i];
                zombie.update(dt, player);
                
                // 攻击检测
                const attack = zombie.tryAttack(player);
                if (attack && player.takeDamage(attack.damage, attack.knockbackX, attack.knockbackY)) {
                    screenShake.shake(CONFIG.SCREEN_SHAKE_INTENSITY_DAMAGE, 0.25);
                    document.getElementById('damageFlash').classList.add('active');
                    setTimeout(() => {
                        document.getElementById('damageFlash').classList.remove('active');
                    }, 300);
                }
                
                // 清理死亡僵尸
                if (zombie.isDead()) {
                    this.zombies.splice(i, 1);
                }
            }
        }

        checkBulletCollisions(bullets, bloodSystem, screenShake, scoreSystem, killPopupManager, camera, waveManager) {
            for (let i = bullets.length - 1; i >= 0; i--) {
                const bullet = bullets[i];
                if (!bullet.active) continue;
                
                for (const zombie of this.zombies) {
                    if (!zombie.canBeHit()) continue;
                    
                    const dist = Utils.distance(bullet.x, bullet.y, zombie.x, zombie.y);
                    
                    if (dist < zombie.size + bullet.size) {
                        // 爆头检测
                        const bulletAngle = Math.atan2(bullet.vy || Math.sin(bullet.angle) * bullet.speed, 
                                                        bullet.vx || Math.cos(bullet.angle) * bullet.speed);
                        const zombieAngle = Math.atan2(zombie.y - bullet.y, zombie.x - bullet.x);
                        const angleDiff = Math.abs(bulletAngle - zombieAngle);
                        
                        const isHeadshot = angleDiff > Math.PI - CONFIG.ZOMBIE_HEADSHOT_ZONE_RATIO || 
                                          angleDiff < CONFIG.ZOMBIE_HEADSHOT_ZONE_RATIO;
                        
                        const result = zombie.takeDamage(bullet.damage, bullet.angle, isHeadshot);
                        bullet.active = false;
                        
                        // 血液效果
                        const bloodCount = isHeadshot ? CONFIG.BLOOD_PARTICLE_COUNT * 2 : CONFIG.BLOOD_PARTICLE_COUNT;
                        bloodSystem.spawn(bullet.x, bullet.y, bullet.angle, bloodCount, isHeadshot);
                        
                        // 屏幕震动
                        if (result.killed) {
                            screenShake.shake(CONFIG.SCREEN_SHAKE_INTENSITY_HEADSHOT, 0.2);
                            const points = scoreSystem.addKill(zombie.type, isHeadshot);
                            waveManager.zombieKilled();
                            
                            // 击杀弹出
                            killPopupManager.show(zombie.x, zombie.y - zombie.size, points, isHeadshot, scoreSystem.combo, camera);
                        } else {
                            screenShake.shake(CONFIG.SCREEN_SHAKE_INTENSITY_HIT, 0.1);
                        }
                        
                        break;
                    }
                }
            }
        }

        render(ctx, camera, time) {
            for (const zombie of this.zombies) {
                zombie.render(ctx, camera, time);
            }
        }

        getActiveCount() {
            return this.zombies.filter(z => z.state !== ZOMBIE_STATE.DEAD && z.state !== ZOMBIE_STATE.DYING).length;
        }
    }

    class Camera {
        constructor(width, height) {
            this.x = 0;
            this.y = 0;
            this.width = width;
            this.height = height;
            this.targetX = 0;
            this.targetY = 0;
            this.smoothing = 5;
        }

        follow(target, dt) {
            this.targetX = target.x - this.width / 2;
            this.targetY = target.y - this.height / 2;
            
            this.x = Utils.lerp(this.x, this.targetX, this.smoothing * dt);
            this.y = Utils.lerp(this.y, this.targetY, this.smoothing * dt);
            
            // 边界限制
            this.x = Utils.clamp(this.x, 0, CONFIG.WORLD_WIDTH - this.width);
            this.y = Utils.clamp(this.y, 0, CONFIG.WORLD_HEIGHT - this.height);
        }

        resize(width, height) {
            this.width = width;
            this.height = height;
        }
    }

    class InputManager {
        constructor(canvas) {
            this.canvas = canvas;
            this.keys = {};
            this.mouseX = 0;
            this.mouseY = 0;
            this.mouseDown = false;
            this.mouseClicked = false;
            
            this.setupListeners();
        }

        setupListeners() {
            window.addEventListener('keydown', (e) => {
                this.keys[e.key.toLowerCase()] = true;
                
                if (e.key === ' ') {
                    e.preventDefault();
                }
            });
            
            window.addEventListener('keyup', (e) => {
                this.keys[e.key.toLowerCase()] = false;
            });
            
            this.canvas.addEventListener('mousemove', (e) => {
                const rect = this.canvas.getBoundingClientRect();
                this.mouseX = e.clientX - rect.left;
                this.mouseY = e.clientY - rect.top;
            });
            
            this.canvas.addEventListener('mousedown', (e) => {
                if (e.button === 0) {
                    this.mouseDown = true;
                    this.mouseClicked = true;
                }
            });
            
            this.canvas.addEventListener('mouseup', (e) => {
                if (e.button === 0) {
                    this.mouseDown = false;
                }
            });
            
            this.canvas.addEventListener('contextmenu', (e) => {
                e.preventDefault();
            });
        }

        consumeClick() {
            const clicked = this.mouseClicked;
            this.mouseClicked = false;
            return clicked;
        }
    }

    class Renderer {
        constructor(canvas) {
            this.canvas = canvas;
            this.ctx = canvas.getContext('2d');
        }

        clear() {
            this.ctx.fillStyle = '#050607';
            this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);
        }

        renderBackground(camera) {
            // 网格
            this.ctx.strokeStyle = 'rgba(255, 255, 255, 0.03)';
            this.ctx.lineWidth = 1;
            
            const startX = Math.floor(camera.x / CONFIG.GRID_SIZE) * CONFIG.GRID_SIZE - camera.x;
            const startY = Math.floor(camera.y / CONFIG.GRID_SIZE) * CONFIG.GRID_SIZE - camera.y;
            
            for (let x = startX; x < this.canvas.width + CONFIG.GRID_SIZE; x += CONFIG.GRID_SIZE) {
                this.ctx.beginPath();
                this.ctx.moveTo(x, 0);
                this.ctx.lineTo(x, this.canvas.height);
                this.ctx.stroke();
            }
            
            for (let y = startY; y < this.canvas.height + CONFIG.GRID_SIZE; y += CONFIG.GRID_SIZE) {
                this.ctx.beginPath();
                this.ctx.moveTo(0, y);
                this.ctx.lineTo(this.canvas.width, y);
                this.ctx.stroke();
            }
            
            // 世界边界
            this.ctx.strokeStyle = 'rgba(255, 61, 61, 0.3)';
            this.ctx.lineWidth = 4;
            this.ctx.strokeRect(-camera.x, -camera.y, CONFIG.WORLD_WIDTH, CONFIG.WORLD_HEIGHT);
        }

        renderFlashlight(ctx, player, camera) {
            const screenX = player.x - camera.x;
            const screenY = player.y - camera.y;
            
            // 创建手电筒光照效果
            const gradient = ctx.createRadialGradient(
                screenX, screenY, CONFIG.FLASHLIGHT_INNER_RADIUS,
                screenX, screenY, CONFIG.FLASHLIGHT_RADIUS
            );
            
            gradient.addColorStop(0, 'rgba(0, 0, 0, 0)');
            gradient.addColorStop(0.5, 'rgba(0, 0, 0, 0.3)');
            gradient.addColorStop(1, 'rgba(0, 0, 0, 0.7)');
            
            // 绘制圆形遮罩
            ctx.save();
            ctx.globalCompositeOperation = 'source-over';
            
            // 创建反向遮罩
            ctx.fillStyle = 'rgba(0, 0, 0, 0.6)';
            ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);
            
            // 挖出光照区域
            ctx.globalCompositeOperation = 'destination-out';
            ctx.fillStyle = gradient;
            ctx.beginPath();
            ctx.arc(screenX, screenY, CONFIG.FLASHLIGHT_RADIUS, 0, Math.PI * 2);
            ctx.fill();
            
            ctx.restore();
            
            // 方向性光照（锥形）
            ctx.save();
            ctx.globalCompositeOperation = 'lighter';
            
            const coneAngle = Math.PI * 0.4; // 72度锥角
            const coneLength = CONFIG.FLASHLIGHT_RADIUS * 1.2;
            
            const coneGradient = ctx.createRadialGradient(
                screenX, screenY, 0,
                screenX, screenY, coneLength
            );
            coneGradient.addColorStop(0, 'rgba(78, 205, 196, 0.15)');
            coneGradient.addColorStop(0.5, 'rgba(78, 205, 196, 0.05)');
            coneGradient.addColorStop(1, 'rgba(78, 205, 196, 0)');
            
            ctx.fillStyle = coneGradient;
            ctx.beginPath();
            ctx.moveTo(screenX, screenY);
            ctx.arc(screenX, screenY, coneLength, player.angle - coneAngle / 2, player.angle + coneAngle / 2);
            ctx.closePath();
            ctx.fill();
            
            ctx.restore();
        }

        resize() {
            this.canvas.width = window.innerWidth;
            this.canvas.height = window.innerHeight;
        }
    }

            this.highScore.textContent = Utils.formatScore(highScore);
        }

        updateCombo(combo, comboProgress) {

                this.comboDisplay.classList.add('active');
                this.comboMultiplier.textContent = `x${combo}`;
                this.comboBarFill.style.width = (comboProgress * 100) + '%';
            } else {
                this.comboDisplay.classList.remove('active');
            }

        updateStats(kills, headshots) {
            this.totalKills.textContent = kills;
            this.totalHeadshots.textContent = headshots;

            this.announcementTitle.textContent = `WAVE ${wave}`;
            this.announcementSubtitle.textContent = `${zombieCount} zombies incoming...`;
            this.waveAnnouncement.classList.remove('show');
            void this.waveAnnouncement.offsetWidth; // 强制重绘
            this.waveAnnouncement.classList.add('show');

        showGameOver(wave, kills, headshots, score, isNewRecord) {
            this.finalWave.textContent = wave;
            this.finalKills.textContent = kills;
            this.finalHeadshots.textContent = headshots;
            this.finalScore.textContent = Utils.formatScore(score);
            this.newRecord.style.display = isNewRecord ? 'block' : 'none';
            this.gameOverScreen.classList.add('show');

            this.gameOverScreen.classList.remove('show');

    // ============================================
    // 游戏主类
    // ============================================
    class Game {
        constructor() {
            this.canvas = document.getElementById('gameCanvas');
            this.renderer = new Renderer(this.canvas);
            this.camera = new Camera(this.canvas.width, this.canvas.height);
            this.input = new InputManager(this.canvas);
            this.ui = new UIManager();
            
            // 特效系统
            this.screenShake = new ScreenShake();
            this.bloodSystem = new BloodSystem();
            this.dustSystem = new DustSystem();
            this.killPopupManager = new KillPopupManager();
            
            this.scoreSystem = new ScoreSystem();
            this.waveManager = new WaveManager();
            
            this.player = null;
            this.bullets = [];
            this.zombieManager = new ZombieManager();
            
            this.time = 0;
            this.isGameOver = false;
            this.isPaused = false;
            
            this.init();
            this.setupEventListeners();
        }

        init() {
            this.renderer.resize();
            this.camera.resize(this.canvas.width, this.canvas.height);
            
            // 初始化玩家
            this.player = new Player(CONFIG.WORLD_WIDTH / 2, CONFIG.WORLD_HEIGHT / 2);
            
            // 重置系统
            this.bullets = [];
            this.scoreSystem.reset();
            this.waveManager = new WaveManager();
            this.zombieManager = new ZombieManager();
            this.bloodSystem = new BloodSystem();
            this.screenShake = new ScreenShake();
            this.isGameOver = false;
            this.time = 0;
            
            // 更新UI
            this.ui.updateHighScore(this.scoreSystem.highScore);
            this.ui.hideGameOver();
            
            // 开始第一波
            this.startNextWave();
        }

        setupEventListeners() {
            window.addEventListener('resize', () => {
                this.renderer.resize();
                this.camera.resize(this.canvas.width, this.canvas.height);
            });
            
            // 重新开始按钮
            document.getElementById('restartBtn').addEventListener('click', () => {
                this.init();
            });
            
            // 射击
            this.canvas.addEventListener('mousedown', (e) => {
                if (e.button === 0 && !this.isGameOver) {
                    this.tryShoot();
                }
            });
            
            // 按键处理
            window.addEventListener('keydown', (e) => {
                if (this.isGameOver) return;
                
                if (e.key.toLowerCase() === 'r') {
                    this.player.ammo.startReload();
                }
                
                if (e.key === ' ') {
                    e.preventDefault();
                    this.player.dash(this.input);
                }
            });
        }

        tryShoot() {
            if (this.player.shoot(this.bullets)) {
                // 射击成功
            }
        }

        startNextWave() {
            const waveInfo = this.waveManager.startNextWave();
            this.scoreSystem.startWave();
            this.ui.showWaveAnnouncement(waveInfo.wave, waveInfo.zombieCount);
        }

        update(dt) {
            if (this.isGameOver) return;
            
            this.time += dt;
            
            // 更新玩家
            this.player.update(dt, this.input, this.camera);
            
            // 持续射击
            if (this.input.mouseDown) {
                this.tryShoot();
            }
            
            // 更新相机
            this.camera.follow(this.player, dt);
            
            // 更新子弹
            for (let i = this.bullets.length - 1; i >= 0; i--) {
                this.bullets[i].update(dt);
                if (!this.bullets[i].active) {
                    this.bullets.splice(i, 1);
                }
            }
            
            // 更新僵尸
            this.zombieManager.update(dt, this.player, this.bloodSystem, this.screenShake, 
                                      this.scoreSystem, this.killPopupManager, this.camera);
            
            // 子弹碰撞检测
            this.zombieManager.checkBulletCollisions(this.bullets, this.bloodSystem, this.screenShake,
                                                     this.scoreSystem, this.killPopupManager, this.camera, this.waveManager);
            
            // 更新波次
            const waveResult = this.waveManager.update(dt, this.player, this.zombieManager);
            if (waveResult && waveResult.completed) {
                const bonusResult = this.scoreSystem.completeWave(this.waveManager.currentWave, waveResult.waveTime);
                this.startNextWave();
            }
            
            // 更新特效系统
            this.bloodSystem.update(dt);
            this.dustSystem.update(dt, this.time);
            this.screenShake.update(dt);
            this.scoreSystem.update(dt);
            
            // 检查游戏结束
            if (this.player.isDead()) {
                this.gameOver();
            }
            
            // 更新UI
            this.updateUI();
        }

        updateUI() {
            this.ui.updateHealth(this.player.health, this.player.maxHealth);
            this.ui.updateDash(this.player.dashCooldown, this.player.dashCooldownTime);
            this.ui.updateAmmo(this.player.ammo);
            this.ui.updateScore(this.scoreSystem.currentScore, this.scoreSystem.highScore);
            this.ui.updateCombo(this.scoreSystem.combo, this.scoreSystem.comboTimer / CONFIG.COMBO_TIMEOUT);
            this.ui.updateStats(this.scoreSystem.totalKills, this.scoreSystem.totalHeadshots);
            
            // 波次状态
            let status = '';
            if (this.waveManager.isCountdown()) {
                status = `Starting in ${Math.ceil(this.waveManager.countdownTimer)}s`;
            } else if (this.waveManager.isActive()) {
                status = `ZOMBIES: ${this.zombieManager.getActiveCount()}`;
            }
            this.ui.updateWave(this.waveManager.currentWave, this.zombieManager.getActiveCount(), status);
        }

        gameOver() {
            this.isGameOver = true;
            this.ui.showGameOver(
                this.waveManager.currentWave,
                this.scoreSystem.totalKills,
                this.scoreSystem.totalHeadshots,
                this.scoreSystem.currentScore,
                this.scoreSystem.currentScore >= this.scoreSystem.highScore
            );
        }

        render() {
            const ctx = this.renderer.ctx;
            
            this.renderer.clear();
            
            // 应用屏幕震动
            this.screenShake.apply(ctx);
            
            // 渲染背景
            this.renderer.renderBackground(this.camera);
            
            // 渲染尘埃粒子
            this.dustSystem.render(ctx, this.camera);
            
            // 渲染血液
            this.bloodSystem.render(ctx, this.camera);
            
            // 渲染子弹
            for (const bullet of this.bullets) {
                bullet.render(ctx, this.camera);
            }
            
            // 渲染僵尸
            this.zombieManager.render(ctx, this.camera, this.time);
            
            // 渲染玩家
            this.player.render(ctx, this.camera);
            
            // 重置屏幕震动变换
            this.screenShake.reset(ctx);
            
            // 渲染手电筒效果
            this.renderer.renderFlashlight(ctx, this.player, this.camera);
        }

        loop(timestamp) {
            const dt = Math.min((timestamp - this.lastTime) / 1000, 0.1);
            this.lastTime = timestamp;
            
            this.update(dt);
            this.render();
            
            requestAnimationFrame((t) => this.loop(t));
        }

        start() {
            this.lastTime = performance.now();
            requestAnimationFrame((t) => this.loop(t));
        }
    }

    // ============================================
    // 启动游戏
    // ============================================
    window.addEventListener('load', () => {
        const game = new Game();
        game.start();
    });
    </script>
</body>
</html>
