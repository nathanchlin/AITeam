    const Utils = {
        lerp(a, b, t) {
            return a + (b - a) * t;
        },
        
        clamp(val, min, max) {
            return Math.max(min, Math.min(max, val));
        },
        
        distance(x1, y1, x2, y2) {
            const dx = x2 - x1;
            const dy = y2 - y1;
            return Math.sqrt(dx * dx + dy * dy);
        },
        
        angle(x1, y1, x2, y2) {
            return Math.atan2(y2 - y1, x2 - x1);
        },
        
        randomRange(min, max) {
            return min + Math.random() * (max - min);
        },
        
        randomInt(min, max) {
            return Math.floor(Math.random() * (max - min + 1)) + min;
        }
    };

    // ============================================

    class Vector2 {
        constructor(x = 0, y = 0) {
            this.x = x;
            this.y = y;
        }
        
        set(x, y) {
            this.x = x;
            this.y = y;
            return this;
        }
        
        copy(v) {
            this.x = v.x;
            this.y = v.y;
            return this;
        }
        
        clone() {
            return new Vector2(this.x, this.y);
        }
        
        add(v) {
            this.x += v.x;
            this.y += v.y;
            return this;
        }
        
        sub(v) {
            this.x -= v.x;
            this.y -= v.y;
            return this;
        }
        
        scale(s) {
            this.x *= s;
            this.y *= s;
            return this;
        }
        
        normalize() {
            const len = this.length();
            if (len > 0) {
                this.x /= len;
                this.y /= len;
            }
            return this;
        }
        
        length() {
            return Math.sqrt(this.x * this.x + this.y * this.y);
        }
        
        dot(v) {
            return this.x * v.x + this.y * v.y;
        }
    }

    class InputManager {
        constructor(canvas) {
            this.canvas = canvas;
            this.keys = {};
            this.mouse = {
                x: 0,
                y: 0,
                worldX: 0,
                worldY: 0,
                leftDown: false,
                rightDown: false
            };
            
            this.bindEvents();
        }
        
        bindEvents() {
            window.addEventListener('keydown', (e) => {
                this.keys[e.code] = true;
                if (['Space', 'KeyW', 'KeyA', 'KeyS', 'KeyD', 'KeyR', 
                     'ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight'].includes(e.code)) {
                    e.preventDefault();
                }
            });
            
            window.addEventListener('keyup', (e) => {
                this.keys[e.code] = false;
            });
            
            this.canvas.addEventListener('mousemove', (e) => {
                const rect = this.canvas.getBoundingClientRect();
                this.mouse.x = e.clientX - rect.left;
                this.mouse.y = e.clientY - rect.top;
            });
            
            this.canvas.addEventListener('mousedown', (e) => {
                if (e.button === 0) this.mouse.leftDown = true;
                if (e.button === 2) this.mouse.rightDown = true;
            });
            
            this.canvas.addEventListener('mouseup', (e) => {
                if (e.button === 0) this.mouse.leftDown = false;
                if (e.button === 2) this.mouse.rightDown = false;
            });
            
            this.canvas.addEventListener('contextmenu', (e) => e.preventDefault());
        }
        
        isKeyDown(code) {
            return this.keys[code] === true;
        }
        
        updateWorldMouse(viewport) {
            this.mouse.worldX = this.mouse.x + viewport.x;
            this.mouse.worldY = this.mouse.y + viewport.y;
        }
    }

    class Viewport {
        constructor(canvas, worldWidth, worldHeight) {
            this.canvas = canvas;
            this.worldWidth = worldWidth;
            this.worldHeight = worldHeight;
            this.x = 0;
            this.y = 0;
            this.targetX = 0;
            this.targetY = 0;
            this.smoothing = 0.08;
            this.shakeAmount = 0;
            this.shakeDecay = 0.9;
        }
        
        follow(target, dt) {
            const targetCenterX = target.x - this.canvas.width / 2 + target.width / 2;
            const targetCenterY = target.y - this.canvas.height / 2 + target.height / 2;
            
            this.targetX = Utils.clamp(targetCenterX, 0, this.worldWidth - this.canvas.width);
            this.targetY = Utils.clamp(targetCenterY, 0, this.worldHeight - this.canvas.height);
            
            this.x = Utils.lerp(this.x, this.targetX, this.smoothing);
            this.y = Utils.lerp(this.y, this.targetY, this.smoothing);
            
            // 屏幕震动
            if (this.shakeAmount > 0.5) {
                this.x += Utils.randomRange(-this.shakeAmount, this.shakeAmount);
                this.y += Utils.randomRange(-this.shakeAmount, this.shakeAmount);
                this.shakeAmount *= this.shakeDecay;
            } else {
                this.shakeAmount = 0;
            }
        }
        
        shake(amount) {
            this.shakeAmount = amount;
        }
        
        worldToScreen(worldX, worldY) {
            return {
                x: worldX - this.x,
                y: worldY - this.y
            };
        }
        
        screenToWorld(screenX, screenY) {
            return {
                x: screenX + this.x,
                y: screenY + this.y
            };
        }
        
        isVisible(x, y, width, height) {
            const margin = 100;
            return x + width + margin > this.x && 
                   x - margin < this.x + this.canvas.width &&
                   y + height + margin > this.y &&
                   y - margin < this.y + this.canvas.height;
        }
    }

    class Bullet extends Entity {
        constructor(x, y, angle, damage = 25) {
            super(x - CONFIG.BULLET_SIZE / 2, y - CONFIG.BULLET_SIZE / 2, 
                  CONFIG.BULLET_SIZE, CONFIG.BULLET_SIZE);
            this.angle = angle;
            this.speed = CONFIG.BULLET_SPEED;
            this.damage = damage;
            this.vx = Math.cos(angle) * this.speed;
            this.vy = Math.sin(angle) * this.speed;
            this.lifeTime = 2; // 秒
            this.trail = [];
            this.maxTrailLength = CONFIG.BULLET_TRAIL_LENGTH;
            this.pierce = 0; // 穿透次数
        }
        
        update(dt, game) {
            // 记录轨迹
            this.trail.unshift({ x: this.centerX, y: this.centerY });
            if (this.trail.length > this.maxTrailLength) {
                this.trail.pop();
            }
            
            // 移动
            this.x += this.vx * dt;
            this.y += this.vy * dt;
            
            // 生命周期
            this.lifeTime -= dt;
            if (this.lifeTime <= 0) {
                this.active = false;
            }
            
            // 世界边界检测
            if (this.x < -50 || this.x > CONFIG.WORLD_WIDTH + 50 ||
                this.y < -50 || this.y > CONFIG.WORLD_HEIGHT + 50) {
                this.active = false;
            }
        }
        
        render(ctx, viewport) {
            const screenPos = viewport.worldToScreen(this.x, this.y);
            const screenCenterX = screenPos.x + this.width / 2;
            const screenCenterY = screenPos.y + this.height / 2;
            
            // 渲染轨迹
            if (this.trail.length > 1) {
                ctx.save();
                ctx.lineCap = 'round';
                
                for (let i = 0; i < this.trail.length - 1; i++) {
                    const t = this.trail[i];
                    const t2 = this.trail[i + 1];
                    const screenT = viewport.worldToScreen(t.x - this.width / 2, t.y - this.height / 2);
                    const screenT2 = viewport.worldToScreen(t2.x - this.width / 2, t2.y - this.height / 2);
                    
                    const alpha = 1 - (i / this.trail.length);
                    const width = Utils.lerp(1, CONFIG.BULLET_SIZE * 0.8, alpha);
                    
                    ctx.beginPath();
                    ctx.moveTo(screenT.x + this.width / 2, screenT.y + this.height / 2);
                    ctx.lineTo(screenT2.x + this.width / 2, screenT2.y + this.height / 2);
                    ctx.strokeStyle = `rgba(255, 217, 61, ${alpha * 0.6})`;
                    ctx.lineWidth = width;
                    ctx.stroke();
                }
                
                ctx.restore();
            }
            
            // 渲染子弹主体
            ctx.save();
            ctx.translate(screenCenterX, screenCenterY);
            
            // 发光效果
            const gradient = ctx.createRadialGradient(0, 0, 0, 0, 0, CONFIG.BULLET_SIZE * 2);
            gradient.addColorStop(0, 'rgba(255, 217, 61, 0.8)');
            gradient.addColorStop(0.5, 'rgba(255, 180, 50, 0.3)');
            gradient.addColorStop(1, 'rgba(255, 150, 50, 0)');
            ctx.fillStyle = gradient;
            ctx.beginPath();
            ctx.arc(0, 0, CONFIG.BULLET_SIZE * 2, 0, Math.PI * 2);
            ctx.fill();
            
            // 子弹核心
            ctx.fillStyle = '#ffd93d';
            ctx.beginPath();
            ctx.arc(0, 0, CONFIG.BULLET_SIZE / 2, 0, Math.PI * 2);
            ctx.fill();
            
            // 高光
            ctx.fillStyle = '#ffffff';
            ctx.beginPath();
            ctx.arc(-CONFIG.BULLET_SIZE / 4, -CONFIG.BULLET_SIZE / 4, CONFIG.BULLET_SIZE / 4, 0, Math.PI * 2);
            ctx.fill();
            
            ctx.restore();
        }
    }

    // 枪口闪光效果

    class MuzzleFlash {
        constructor(x, y, angle) {
            this.x = x;
            this.y = y;
            this.angle = angle;
            this.lifeTime = CONFIG.MUZZLE_FLASH_DURATION;
            this.maxLifeTime = CONFIG.MUZZLE_FLASH_DURATION;
            this.size = Utils.randomRange(15, 25);
            this.active = true;
        }
        
        update(dt) {
            this.lifeTime -= dt;
            if (this.lifeTime <= 0) {
                this.active = false;
            }
        }
        
        render(ctx, viewport) {
            const screenPos = viewport.worldToScreen(this.x, this.y);
            const progress = this.lifeTime / this.maxLifeTime;
            const currentSize = this.size * progress;
            
            ctx.save();
            ctx.translate(screenPos.x, screenPos.y);
            ctx.rotate(this.angle);
            
            // 主闪光
            const gradient = ctx.createRadialGradient(0, 0, 0, 0, 0, currentSize * 2);
            gradient.addColorStop(0, `rgba(255, 200, 100, ${progress})`);
            gradient.addColorStop(0.3, `rgba(255, 150, 50, ${progress * 0.8})`);
            gradient.addColorStop(0.6, `rgba(255, 100, 30, ${progress * 0.4})`);
            gradient.addColorStop(1, 'rgba(255, 50, 0, 0)');
            
            ctx.fillStyle = gradient;
            ctx.beginPath();
            ctx.arc(0, 0, currentSize * 2, 0, Math.PI * 2);
            ctx.fill();
            
            // 火焰尖端
            ctx.fillStyle = `rgba(255, 255, 200, ${progress})`;
            ctx.beginPath();
            ctx.moveTo(currentSize, 0);
            ctx.lineTo(-currentSize * 0.3, -currentSize * 0.5);
            ctx.lineTo(-currentSize * 0.3, currentSize * 0.5);
            ctx.closePath();
            ctx.fill();
            
            // 辉光射线
            const rayCount = 6;
            ctx.strokeStyle = `rgba(255, 220, 100, ${progress * 0.6})`;
            ctx.lineWidth = 2;
            for (let i = 0; i < rayCount; i++) {
                const rayAngle = (i / rayCount) * Math.PI * 2;
                const rayLength = currentSize * (0.8 + Math.random() * 0.4);
                ctx.beginPath();
                ctx.moveTo(0, 0);
                ctx.lineTo(Math.cos(rayAngle) * rayLength, Math.sin(rayAngle) * rayLength);
                ctx.stroke();
            }
            
            ctx.restore();
        }
    }

                                        game.input.mouse.worldX, game.input.mouse.worldY);
            
            // 移动输入

            if (game.input.isKeyDown('KeyW') || game.input.isKeyDown('ArrowUp')) moveDir.y -= 1;
            if (game.input.isKeyDown('KeyS') || game.input.isKeyDown('ArrowDown')) moveDir.y += 1;
            if (game.input.isKeyDown('KeyA') || game.input.isKeyDown('ArrowLeft')) moveDir.x -= 1;
            if (game.input.isKeyDown('KeyD') || game.input.isKeyDown('ArrowRight')) moveDir.x += 1;
            
            // 冲刺冷却
            if (this.dashCooldown > 0) this.dashCooldown -= dt;
            
            // 无敌时间
            if (this.invincibleTime > 0) this.invincibleTime -= dt;
            
            // 装弹逻辑
            this.updateReload(dt, game);
            
            // 冲刺状态
            if (this.dashDuration > 0) {
                this.dashDuration -= dt;
                this.vx = this.dashDir.x * CONFIG.PLAYER_DASH_SPEED;
                this.vy = this.dashDir.y * CONFIG.PLAYER_DASH_SPEED;
                if (this.dashDuration <= 0) {
                    this.isDashing = false;
                }
            } else {
                // 正常移动 - 带加速度
                if (moveDir.length() > 0) {
                    moveDir.normalize();
                    this.targetVx = moveDir.x * this.speed;
                    this.targetVy = moveDir.y * this.speed;
                } else {
                    this.targetVx = 0;
                    this.targetVy = 0;
                }
                
                // 应用加速度/摩擦力
                if (Math.abs(this.vx - this.targetVx) > 0.1) {
                    this.vx = Utils.lerp(this.vx, this.targetVx, this.acceleration * dt / 1000);
                } else {
                    this.vx *= (1 - this.friction * dt);
                }
                
                if (Math.abs(this.vy - this.targetVy) > 0.1) {
                    this.vy = Utils.lerp(this.vy, this.targetVy, this.acceleration * dt / 1000);
                } else {
                    this.vy *= (1 - this.friction * dt);
                }
                
                // 冲刺触发
                if (game.input.isKeyDown('Space') && this.dashCooldown <= 0 && moveDir.length() > 0) {
                    this.isDashing = true;
                    this.dashDuration = 0.15;
                    this.dashCooldown = 1.2;
                    this.dashDir.copy(moveDir);
                    this.invincibleTime = 0.25;
                    game.viewport.shake(5);
                }
            }
            
            // 射击
            this.updateShooting(dt, game);
            
            // 更新位置
            this.x += this.vx * dt;
            this.y += this.vy * dt;
            
            // 世界边界碰撞
            const margin = 5;
            if (this.x < margin) {
                this.x = margin;
                this.vx = 0;
            }
            if (this.x > CONFIG.WORLD_WIDTH - this.width - margin) {
                this.x = CONFIG.WORLD_WIDTH - this.width - margin;
                this.vx = 0;
            }
            if (this.y < margin) {
                this.y = margin;
                this.vy = 0;
            }
            if (this.y > CONFIG.WORLD_HEIGHT - this.height - margin) {
                this.y = CONFIG.WORLD_HEIGHT - this.height - margin;
                this.vy = 0;
            }
        }
        
        updateReload(dt, game) {
            // R键装弹
            if (game.input.isKeyDown('KeyR') && !this.isReloading && 
                this.currentAmmo < this.maxAmmo && this.reserveAmmo > 0) {
                this.startReload();
            }
            
            // 弹药耗尽自动装弹
            if (this.currentAmmo <= 0 && !this.isReloading && this.reserveAmmo > 0) {
                this.startReload();
            }
            
            // 装弹进度
            if (this.isReloading) {
                this.reloadProgress += dt;
                if (this.reloadProgress >= this.reloadTime) {
                    this.finishReload();
                }
            }
        }
        
        startReload() {
            this.isReloading = true;
            this.reloadProgress = 0;
        }
        
        finishReload() {
            const ammoNeeded = this.maxAmmo - this.currentAmmo;
            const ammoToAdd = Math.min(ammoNeeded, this.reserveAmmo);
            this.currentAmmo += ammoToAdd;
            this.reserveAmmo -= ammoToAdd;
            this.isReloading = false;
            this.reloadProgress = 0;
        }
        
        updateShooting(dt, game) {
            const now = performance.now();
            
            // 射击冷却
            if (this.shootCooldown > 0) this.shootCooldown -= dt;
            
            // 不能射击的情况
            if (this.isReloading) return;
            if (this.currentAmmo <= 0) return;
            if (now - this.lastFireTime < this.fireRate) return;
            
            // 鼠标左键射击
            if (game.input.mouse.leftDown) {
                this.fire(game);
            }
        }
        
        fire(game) {
            const now = performance.now();
            this.lastFireTime = now;
            this.currentAmmo--;
            
            // 计算枪口位置
            const gunOffset = CONFIG.PLAYER_SIZE * 0.8;
            const muzzleX = this.centerX + Math.cos(this.aimAngle) * gunOffset;
            const muzzleY = this.centerY + Math.sin(this.aimAngle) * gunOffset;
            
            // 添加轻微散布
            const spread = Utils.randomRange(-0.03, 0.03);
            const bulletAngle = this.aimAngle + spread;
            
            // 创建子弹
            const bullet = new Bullet(muzzleX, muzzleY, bulletAngle);
            game.addBullet(bullet);
            
            // 创建枪口闪光
            const flash = new MuzzleFlash(muzzleX, muzzleY, bulletAngle);
            game.addMuzzleFlash(flash);
            
            // 屏幕震动
            game.viewport.shake(3);
        }
        
        render(ctx, viewport) {
            const screenPos = viewport.worldToScreen(this.x, this.y);
            const screenCenterX = screenPos.x + this.width / 2;
            const screenCenterY = screenPos.y + this.height / 2;
            
            ctx.save();
            ctx.translate(screenCenterX, screenCenterY);
            
            // 闪烁效果（无敌时）
            if (this.invincibleTime > 0 && Math.floor(this.animTime * 10) % 2 === 0) {
                ctx.globalAlpha = 0.5;
            }
            
            // 冲刺拖尾
            if (this.isDashing) {
                ctx.globalAlpha = 0.3;
                ctx.fillStyle = 'rgba(78, 205, 196, 0.5)';
                ctx.beginPath();
                ctx.arc(-this.vx * 0.02, -this.vy * 0.02, this.width / 2, 0, Math.PI * 2);
                ctx.fill();
                ctx.globalAlpha = 1;
            }
            
            // 玩家身体发光
            const glowGradient = ctx.createRadialGradient(0, 0, 0, 0, 0, this.width);
            glowGradient.addColorStop(0, 'rgba(78, 205, 196, 0.4)');
            glowGradient.addColorStop(0.5, 'rgba(78, 205, 196, 0.1)');
            glowGradient.addColorStop(1, 'rgba(78, 205, 196, 0)');
            ctx.fillStyle = glowGradient;
            ctx.beginPath();
            ctx.arc(0, 0, this.width, 0, Math.PI * 2);
            ctx.fill();
            
            // 玩家身体
            ctx.fillStyle = '#4ecdc4';
            ctx.beginPath();
            ctx.arc(0, 0, this.width / 2, 0, Math.PI * 2);
            ctx.fill();
            
            // 内圈
            ctx.fillStyle = '#3db8b0';
            ctx.beginPath();
            ctx.arc(0, 0, this.width / 3, 0, Math.PI * 2);
            ctx.fill();
            
            // 武器指示器
            ctx.rotate(this.aimAngle);
            ctx.fillStyle = '#2a8a84';
            ctx.fillRect(this.width / 3, -3, this.width / 2, 6);
            
            // 枪口
            ctx.fillStyle = '#1a5a56';
            ctx.fillRect(this.width / 2 + this.width / 3 - 2, -4, 8, 8);
            
            ctx.restore();
        }
        
        getDashCooldownPercent() {
            return Math.max(0, 1 - this.dashCooldown / 1.2);
        }
        
        getReloadProgressPercent() {
            return this.reloadProgress / this.reloadTime;
        }
    }

    // ============================================
    // 游戏主类
    // ============================================
    class Game {
        constructor() {
            this.canvas = document.getElementById('gameCanvas');
            this.ctx = this.canvas.getContext('2d');
            
            this.resize();
            window.addEventListener('resize', () => this.resize());
            
            this.input = new InputManager(this.canvas);
            this.viewport = new Viewport(this.canvas, CONFIG.WORLD_WIDTH, CONFIG.WORLD_HEIGHT);
            
            // 游戏实体
            this.player = null;
            this.bullets = [];
            this.muzzleFlashes = [];
            
            // 游戏状态
            this.isRunning = true;
            this.lastTime = 0;
            this.frameCount = 0;
            this.fps = 60;
            this.fpsUpdateTime = 0;
            
            this.init();
        }
        
        init() {
            // 创建玩家
            this.player = new Player(CONFIG.WORLD_WIDTH / 2, CONFIG.WORLD_HEIGHT / 2);
            
            // 开始游戏循环
            requestAnimationFrame((time) => this.gameLoop(time));
        }
        
        resize() {
            this.canvas.width = window.innerWidth;
            this.canvas.height = window.innerHeight;
        }
        
        gameLoop(currentTime) {
            if (!this.isRunning) return;
            
            const dt = Math.min((currentTime - this.lastTime) / 1000, 0.1);
            this.lastTime = currentTime;
            
            // FPS计算
            this.frameCount++;
            this.fpsUpdateTime += dt;
            if (this.fpsUpdateTime >= 0.5) {
                this.fps = Math.round(this.frameCount / this.fpsUpdateTime);
                this.frameCount = 0;
                this.fpsUpdateTime = 0;
            }
            
            this.update(dt);
            this.render();
            this.updateUI();
            
            requestAnimationFrame((time) => this.gameLoop(time));
        }
        
        update(dt) {
            // 更新世界鼠标位置
            this.input.updateWorldMouse(this.viewport);
            
            // 更新玩家
            if (this.player) {
                this.player.update(dt, this);
            }
            
            // 更新视口跟随
            this.viewport.follow(this.player, dt);
            
            // 更新子弹
            for (let i = this.bullets.length - 1; i >= 0; i--) {
                this.bullets[i].update(dt, this);
                if (!this.bullets[i].active) {
                    this.bullets.splice(i, 1);
                }
            }
            
            // 更新枪口闪光
            for (let i = this.muzzleFlashes.length - 1; i >= 0; i--) {
                this.muzzleFlashes[i].update(dt);
                if (!this.muzzleFlashes[i].active) {
                    this.muzzleFlashes.splice(i, 1);
                }
            }
        }
        
        render() {
            const ctx = this.ctx;
            
            // 清空画布
            ctx.fillStyle = '#050607';
            ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);
            
            // 渲染网格背景
            this.renderGrid();
            
            // 渲染子弹
            for (const bullet of this.bullets) {
                if (this.viewport.isVisible(bullet.x, bullet.y, bullet.width, bullet.height)) {
                    bullet.render(ctx, this.viewport);
                }
            }
            
            // 渲染枪口闪光
            for (const flash of this.muzzleFlashes) {
                flash.render(ctx, this.viewport);
            }
            
            // 渲染玩家
            if (this.player) {
                this.player.render(ctx, this.viewport);
            }
            
            // 渲染准星
            this.renderCrosshair();
        }
        
        renderGrid() {
            const ctx = this.ctx;
            const gridSize = CONFIG.GRID_SIZE;
            
            ctx.strokeStyle = 'rgba(255, 255, 255, 0.03)';
            ctx.lineWidth = 1;
            
            const startX = Math.floor(this.viewport.x / gridSize) * gridSize;
            const startY = Math.floor(this.viewport.y / gridSize) * gridSize;
            const endX = this.viewport.x + this.canvas.width;
            const endY = this.viewport.y + this.canvas.height;
            
            ctx.beginPath();
            for (let x = startX; x <= endX; x += gridSize) {
                const screenX = x - this.viewport.x;
                ctx.moveTo(screenX, 0);
                ctx.lineTo(screenX, this.canvas.height);
            }
            for (let y = startY; y <= endY; y += gridSize) {
                const screenY = y - this.viewport.y;
                ctx.moveTo(0, screenY);
                ctx.lineTo(this.canvas.width, screenY);
            }
            ctx.stroke();
        }
        
        renderCrosshair() {
            const ctx = this.ctx;
            const x = this.input.mouse.x;
            const y = this.input.mouse.y;
            const size = 12;
            const gap = 4;
            
            ctx.save();
            ctx.strokeStyle = this.player && this.player.isReloading ? '#ff3d3d' : '#4ecdc4';
            ctx.lineWidth = 2;
            ctx.lineCap = 'round';
            
            // 十字准星
            ctx.beginPath();
            // 上
            ctx.moveTo(x, y - gap);
            ctx.lineTo(x, y - size);
            // 下
            ctx.moveTo(x, y + gap);
            ctx.lineTo(x, y + size);
            // 左
            ctx.moveTo(x - gap, y);
            ctx.lineTo(x - size, y);
            // 右
            ctx.moveTo(x + gap, y);
            ctx.lineTo(x + size, y);
            ctx.stroke();
            
            // 中心点
            ctx.fillStyle = ctx.strokeStyle;
            ctx.beginPath();
            ctx.arc(x, y, 2, 0, Math.PI * 2);
            ctx.fill();
            
            // 弹药耗尽时显示X
            if (this.player && this.player.currentAmmo <= 0 && !this.player.isReloading) {
                ctx.strokeStyle = '#ff3d3d';
                ctx.lineWidth = 3;
                ctx.beginPath();
                ctx.moveTo(x - size, y - size);
                ctx.lineTo(x + size, y + size);
                ctx.moveTo(x + size, y - size);
                ctx.lineTo(x - size, y + size);
                ctx.stroke();
            }
            
            ctx.restore();
        }
        
        updateUI() {
            if (!this.player) return;
            
            // 分数
            document.getElementById('score').textContent = this.player.score;
            
            // 血量
            const healthPercent = (this.player.health / this.player.maxHealth) * 100;
            document.getElementById('healthFill').style.width = healthPercent + '%';
            
            // 冲刺冷却
            const dashPercent = this.player.getDashCooldownPercent() * 100;
            document.getElementById('dashFill').style.width = dashPercent + '%';
            
            // 弹药
            document.getElementById('ammoCurrent').textContent = this.player.currentAmmo;
            document.getElementById('ammoReserve').textContent = this.player.reserveAmmo;
            
            // 装弹状态
            const reloadingText = document.getElementById('reloadingText');
            const ammoCount = document.getElementById('ammoCount');
            
            if (this.player.isReloading) {
                reloadingText.style.display = 'block';
                reloadingText.textContent = `RELOADING... ${Math.floor(this.player.getReloadProgressPercent() * 100)}%`;
                ammoCount.style.opacity = '0.5';
            } else {
                reloadingText.style.display = 'none';
                ammoCount.style.opacity = '1';
            }
            
            // FPS
            document.getElementById('fps').textContent = this.fps;
        }
        
        // 添加子弹
        addBullet(bullet) {
            this.bullets.push(bullet);
        }
        
        // 添加枪口闪光
        addMuzzleFlash(flash) {
            this.muzzleFlashes.push(flash);
        }
    }

    // 启动游戏
    window.addEventListener('DOMContentLoaded', () => {
        new Game();
    });
    </script>

   - 使用 `Utils.angle()` 计算玩家到鼠标的角度
   - 添加轻微散布（±0.03弧度）增加真实感

3. **子弹轨迹渲染与飞行物理**
   - `Bullet` 类带有 `trail[]` 数组记录轨迹点
   - 渲染时绘制渐变透明度的轨迹线
   - 子弹有发光效果和2秒生命周期

   - 射速：150ms/发
   - 装弹期间无法射击

   - `MuzzleFlash` 类实现动态闪光
   - 包含径向渐变、火焰尖端、射线效果
   - 持续时间80ms自然衰减
