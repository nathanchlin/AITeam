# 数据存储与持久化

**Time**: 2026-02-25T19:00:35.681801

---

```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>躲避弹幕游戏</title>
<style>
  body { margin: 0; overflow: hidden; }
  canvas { display: block; }
  #scoreboard { position: absolute; top: 10px; left: 10px; color: white; font-size: 20px; }
  #gameOver { position: absolute; top: 50%; left: 50%; transform: translate(-50%, -50%); color: white; font-size: 30px; display: none; }
  #player1, #player2 { position: absolute; }
  #player1 { background-color: blue; width: 50px; height: 50px; }
  #player2 { background-color: red; width: 50px; height: 50px; }
</style>
</head>
<body>
<canvas id="gameCanvas"></canvas>
<div id="scoreboard">Score: 0</div>
<div id="gameOver">Game Over</div>
<div id="player1"></div>
<div id="player2"></div>
<script>
// 游戏变量
const canvas = document.getElementById('gameCanvas');
const ctx = canvas.getContext('2d');
const playerSize = 50;
const bulletSize = 5;
const bulletColor = 'red';
let score = 0;
let gameOver = false;
let player1 = { x: canvas.width / 2 - playerSize, y: canvas.height - playerSize, dx: 0, dy: 0 };
let player2 = { x: canvas.width / 2 + playerSize, y: canvas.height - playerSize, dx: 0, dy: 0 };
let bullets = [];

// 存储游戏状态
function saveGameState() {
  const gameState = {
    score: score,
    gameOver: gameOver,
    player1: { x: player1.x, y: player1.y, dx: player1.dx, dy: player1.dy },
    player2: { x: player2.x, y: player2.y, dx: player2.dx, dy: player2.dy },
    bullets: bullets
  };
  localStorage.setItem('gameState', JSON.stringify(gameState));
}

// 加载游戏状态
function loadGameState() {
  const savedState = localStorage.getItem('gameState');
  if (savedState) {
    const gameState = JSON.parse(savedState);
    score = gameState.score;
    gameOver = gameState.gameOver;
    player1.x = gameState.player1.x;
    player1.y = gameState.player1.y;
    player1.dx = gameState.player1.dx;
    player1.dy = gameState.player1.dy;
    player2.x = gameState.player2.x;
    player2.y = gameState.player2.y;
    player2.dx = gameState.player2.dx;
    player2.dy = gameState.player2.dy;
    bullets = gameState.bullets;
    updateScoreboard();
    if (gameOver) {
      document.getElementById('gameOver').style.display = 'block';
    }
  }
}

// 初始化游戏
function init() {
  canvas.width = window.innerWidth;
  canvas.height = window.innerHeight;
  window.addEventListener('keydown', handleKeyDown);
  canvas.addEventListener('click', shoot);
  requestAnimationFrame(gameLoop);
  updateScoreboard();
  loadGameState();
}

// 绘制玩家
function drawPlayer(player) {
  ctx.fillStyle = player.x < canvas.width / 2 ? 'blue' : 'red';
  ctx.fillRect(player.x, player.y, playerSize, playerSize);
}

// 绘制弹幕
function drawBullets() {
  for (let i = 0; i < bullets.length; i++) {
    ctx.fillStyle = bulletColor;
    ctx.beginPath();
    ctx.arc(bullets[i].x, bullets[i].y, bulletSize, 0, Math.PI * 2);
    ctx.fill();
  }
}

// 更新游戏状态
function update() {
  if (gameOver) return;

  player1.x += player1.dx;
  player1.y += player1.dy;
  player1.x = Math.max(0, Math.min(player1.x, canvas.width - playerSize));
  player1.y = Math.max(0, Math.min(player1.y, canvas.height - playerSize));
  player2.x += player2.dx;
  player2.y += player2.dy;
  player2.x = Math.max(0, Math.min(player2.x, canvas.width - playerSize));
  player2.y = Math.max(0, Math.min(player2.y, canvas.height - playerSize));
  for (let i = 0; i < bullets.length; i++) {
    bullets[i].x += bullets[i].dx;
    // 碰撞检测
    if (bullets[i].x > player1.x && bullets[i].x < player1.x + playerSize &&
        bullets[i].y > player1.y && bullets[i].y < player1.y + playerSize) {
      score += 10;
      bullets.splice(i, 1);
      i--;
    } else if (bullets[i].x > player2.x && bullets[i].x < player2.x + playerSize &&
        bullets[i].y > player2.y && bullets[i].y < player2.y + playerSize) {
      score += 10;
      bullets.splice(i, 1);
      i--;
    } else if (bullets[i].x > canvas.width) {
      bullets.splice(i, 1);
      i--;
    }
  }
  saveGameState();
}

// 游戏循环
function gameLoop() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  update();
  drawPlayer(player1);
  drawPlayer(player2);
  drawBullets();
  requestAnimationFrame(gameLoop);
}

// 处理键盘按键
function handleKeyDown(event) {
  if (gameOver) return;

  switch (event.keyCode) {
    case 65: // A
    case 37: // 左箭头
      player1.dx = -5;
      break;
    case 87: // W
    case 38: // 上箭头
      player1.dy = -5;
      break;
    case 68: // D
    case 39: // 右箭头
      player1.dx = 5;
      break;
    case 83: // S
    case 40: // 下箭头
      player1.dy = 5;
      break;
    case 74: // J
    case 65: // A
      player2.dx = -5;
      break;
    case 75: // K
    case 87: // W
      player2.dy = -5;
      break;
    case 76: // L
    case 69: // E
      player2.dx = 5;
      break;
    case 82: // R
    case 81: // Q
      player2.dy = 5;
      break;
    default:
      player1.dx = 0;
      player1.dy = 0;
      player2.dx = 0;
      player2.dy = 0;
      break;
  }
}

// 开火
function shoot(event) {
  if (gameOver) return;

  const player = event.target.id === 'player1' ? player1 : player2;
  bullets.push({
    x: player.x + playerSize / 2,
    y: player.y,
    dx: 0,
    dy: -5
  });
}

// 更新得分板
function updateScoreboard() {
  document.getElementById('scoreboard').innerText = 'Score: ' + score;
}

// 重新开始游戏
function restartGame() {
  gameOver = false;
  player1.x = canvas.width / 2 - playerSize;
  player1.y = canvas.height - playerSize;
  player1.dx = 0;
  player1.dy = 0;
  player2.x = canvas.width / 2 + playerSize;
  player2.y = canvas.height - playerSize;
  player2.dx = 0;
  player2.dy = 0;
  bullets = [];
  score = 0;
  updateScoreboard();
  document.getElementById('gameOver').style.display = 'none';
}

// 监听重新开始游戏事件
document.getElementById('gameOver').addEventListener('click', restartGame);

// 加载游戏
window.onload = init;
</script>
</body>
</html>
```

在这个修改后的代码中，我添加了`saveGameState`和`loadGameState`函数来处理游戏状态的存储和加载。游戏状态包括得分、游戏是否结束、玩家的位置和速度以及弹幕数组。这些状态被存储在`localStorage`中，以便在页面重新加载时恢复游戏状态。此外，我还确保在每次更新游戏状态时调用`saveGameState`函数来保存状态。
