```html

            display: block;
        }
        
        #score {

        #gameOver h2 {

<body>
    <div id="gameContainer">
        <canvas id="gameCanvas" width="800" height="400"></canvas>
        <div id="score">得分: 0</div>
        <div id="gameOver">
            <h2>游戏结束!</h2>
            <p>最终得分: <span id="finalScore">0</span></p>
            <button id="restartBtn">重新开始</button>
        </div>
    </div>

    <script>
        // Matter.js 物理引擎核心模块
        const Engine = Matter.Engine,
              Render = Matter.Render,
              Runner = Matter.Runner,
              Bodies = Matter.Bodies,
              Composite = Matter.Composite,
              World = Matter.World,
              Body = Matter.Body,
              Events = Matter.Events,
              Mouse = Matter.Mouse,
              MouseConstraint = Matter.MouseConstraint;

        // 游戏状态
        let engine, runner;
        let player, ground, obstacles = [], platforms = [];
        let gameRunning = true;
        let score = 0;
        let gameSpeed = 3;
        let lastObstacleTime = 0;
        let canvas, ctx;
        
        // 游戏常量
        const PLAYER_SIZE = 20;
        const OBSTACLE_MIN_HEIGHT = 30;
        const OBSTACLE_MAX_HEIGHT = 80;
        const PLATFORM_WIDTH = 100;
        const PLATFORM_HEIGHT = 15;
        const GRAVITY = 0.5;
        const JUMP_FORCE = -10;
        const RUN_SPEED = 5;
        
        // 初始化游戏
        function initGame() {
            canvas = document.getElementById('gameCanvas');
            ctx = canvas.getContext('2d');
            
            // 创建物理引擎
            engine = Engine.create();
            engine.world.gravity.y = GRAVITY;
            
            // 创建渲染器
            const render = Render.create({
                canvas: canvas,
                engine: engine,
                options: {
                    width: 800,
                    height: 400,
                    wireframes: false,
                    background: 'transparent'
                }
            });
            
            // 创建地面
            ground = Bodies.rectangle(400, 390, 810, 20, { 
                isStatic: true,
                render: { fillStyle: '#8BC34A' }
            });
            
            // 创建玩家
            player = Bodies.rectangle(100, 300, PLAYER_SIZE, PLAYER_SIZE * 2, {
                frictionAir: 0.01,
                render: {
                    fillStyle: '#FF5722'
                }
            });
            
            // 将物体添加到世界
            World.add(engine.world, [ground, player]);
            
            // 初始化障碍物和平台
            createInitialPlatforms();
            
            // 设置事件监听
            setupEventListeners();
            
            // 开始游戏循环
            gameLoop();
        }
        
        // 创建初始平台
        function createInitialPlatforms() {
            // 起始平台
            platforms.push(Bodies.rectangle(200, 350, PLATFORM_WIDTH, PLATFORM_HEIGHT, {
                isStatic: true,
                render: { fillStyle: '#795548' }
            }));
            
            // 添加一些初始障碍物
            for (let i = 0; i < 3; i++) {
                createObstacle(300 + i * 200, 350);
            }
            
            World.add(engine.world, [...platforms, ...obstacles]);
        }
        
        // 创建障碍物
        function createObstacle(x, y) {
            const height = Math.random() * (OBSTACLE_MAX_HEIGHT - OBSTACLE_MIN_HEIGHT) + OBSTACLE_MIN_HEIGHT;
            const obstacle = Bodies.rectangle(x, y - height/2, 20, height, {
                isStatic: true,
                render: { fillStyle: '#F44336' }
            });
            obstacles.push(obstacle);
            return obstacle;
        }
        
        // 创建平台
        function createPlatform(x, y) {
            const platform = Bodies.rectangle(x, y, PLATFORM_WIDTH, PLATFORM_HEIGHT, {
                isStatic: true,
                render: { fillStyle: '#795548' }
            });
            platforms.push(platform);
            return platform;
        }
        
        // 设置事件监听
        function setupEventListeners() {
            // 键盘事件
            document.addEventListener('keydown', (e) => {
                if (e.key === ' ' || e.key === 'ArrowUp') {
                    jump();
                }
            });
            
            // 触摸事件（移动设备）
            canvas.addEventListener('touchstart', (e) => {
                e.preventDefault();
                jump();
            });
            
            // 重新开始按钮
            document.getElementById('restartBtn').addEventListener('click', restartGame);
        }
        
        // 跳跃函数
        function jump() {
            if (gameRunning && isOnGround()) {
                Body.applyForce(player, player.position, { x: 0, y: JUMP_FORCE });
            }
        }
        
        // 检查是否在地面上
        function isOnGround() {
            const playerBottom = player.position.y + PLAYER_SIZE;
            return playerBottom >= ground.bounds.min.y - 5;
        }
        
        // 游戏主循环
        function gameLoop() {
            if (!gameRunning) return;
            
            // 更新物理引擎
            Engine.update(engine);
            
            // 移动障碍物和平台
            moveObstacles();
            
            // 生成新的障碍物和平台
            generateObstacles();
            
            // 更新分数
            updateScore();
            
            // 检查碰撞
            checkCollisions();
            
            // 渲染游戏
            render();
            
            // 增加游戏难度
            increaseDifficulty();
            
            requestAnimationFrame(gameLoop);
        }
        
        // 移动障碍物和平台
        function moveObstacles() {
            const currentTime = Date.now();
            
            // 移动障碍物
            for (let i = obstacles.length - 1; i >= 0; i--) {
                const obstacle = obstacles[i];
                Body.setPosition(obstacle, {
                    x: obstacle.position.x - gameSpeed,
                    y: obstacle.position.y
                });
                
                // 移除屏幕外的障碍物
                if (obstacle.position.x < -50) {
                    World.remove(engine.world, obstacle);
                    obstacles.splice(i, 1);
                }
            }
            
            // 移动平台
            for (let i = platforms.length - 1; i >= 0; i--) {
                const platform = platforms[i];
                Body.setPosition(platform, {
                    x: platform.position.x - gameSpeed,
                    y: platform.position.y
                });
                
                // 移除屏幕外的平台
                if (platform.position.x < -50) {
                    World.remove(engine.world, platform);
                    platforms.splice(i, 1);
                }
            }
        }
        
        // 生成新的障碍物和平台
        function generateObstacles() {
            const currentTime = Date.now();
            
            // 每2秒生成一个新的障碍物
            if (currentTime - lastObstacleTime > 2000) {
                const lastPlatformX = platforms.length > 0 ? 
                    platforms[platforms.length - 1].position.x : 200;
                
                createObstacle(lastPlatformX + 300, 350);
                createPlatform(lastPlatformX + 500, 350);
                
                World.add(engine.world, [obstacles[obstacles.length - 1], platforms[platforms.length - 1]]);
                
                lastObstacleTime = currentTime;
            }
        }
        
        // 更新分数
        function updateScore() {
            score += Math.floor(gameSpeed / 3);
            document.getElementById('score').textContent = `得分: ${score}`;
        }
        
        // 检查碰撞
        function checkCollisions() {
            // 检查玩家是否与障碍物碰撞
            for (const obstacle of obstacles) {
                const collision = Matter.Query.collides(player, [obstacle]);
                if (collision.length > 0) {
                    gameOver();
                    return;
                }
            }
            
            // 检查玩家是否掉出屏幕
            if (player.position.y > canvas.height) {
                gameOver();
            }
        }
        
        // 增加游戏难度
        function increaseDifficulty() {
            // 每500分增加游戏速度
            if (score > 0 && score % 500 === 0) {
                gameSpeed += 0.1;
            }
        }
        
        // 渲染游戏
        function render() {
            // 清空画布
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            // 绘制云朵背景
            drawClouds();
            
            // 绘制玩家（火柴人）
            drawStickman(player.position.x, player.position.y);
            
            // 绘制障碍物
            for (const obstacle of obstacles) {
                ctx.fillStyle = obstacle.render.fillStyle;
                ctx.fillRect(
                    obstacle.position.x - obstacle.bounds.max.x + obstacle.bounds.min.x,
                    obstacle.position.y - obstacle.bounds.max.y + obstacle.bounds.min.y,
                    obstacle.bounds.max.x - obstacle.bounds.min.x,
                    obstacle.bounds.max.y - obstacle.bounds.min.y
                );
            }
            
            // 绘制平台
            for (const platform of platforms) {
                ctx.fillStyle = platform.render.fillStyle;
                ctx.fillRect(
                    platform.position.x - platform.bounds.max.x + platform.bounds.min.x,
                    platform.position.y - platform.bounds.max.y + platform.bounds.min.y,
                    platform.bounds.max.x - platform.bounds.min.x,
                    platform.bounds.max.y - platform.bounds.min.y
                );
            }
        }
        
        // 绘制火柴人
        function drawStickman(x, y) {
            ctx.strokeStyle = '#FF5722';
            ctx.lineWidth = 3;
            ctx.lineCap = 'round';
            
            // 头部
            ctx.beginPath();
            ctx.arc(x, y - PLAYER_SIZE, PLAYER_SIZE / 2, 0, Math.PI * 2);
            ctx.stroke();
            
            // 身体
            ctx.beginPath();
            ctx.moveTo(x, y - PLAYER_SIZE / 2);
            ctx.lineTo(x, y + PLAYER_SIZE / 2);
            ctx.stroke();
            
            // 左臂
            ctx.beginPath();
            ctx.moveTo(x, y - PLAYER_SIZE / 4);
            ctx.lineTo(x - PLAYER_SIZE, y);
            ctx.stroke();
            
            // 右臂
            ctx.beginPath();
            ctx.moveTo(x, y - PLAYER_SIZE / 4);
            ctx.lineTo(x + PLAYER_SIZE, y);
            ctx.stroke();
            
            // 左腿
            ctx.beginPath();
            ctx.moveTo(x, y + PLAYER_SIZE / 2);
            ctx.lineTo(x - PLAYER_SIZE / 2, y + PLAYER_SIZE);
            ctx.stroke();
            
            // 右腿
            ctx.beginPath();
            ctx.moveTo(x, y + PLAYER_SIZE / 2);
            ctx.lineTo(x + PLAYER_SIZE / 2, y + PLAYER_SIZE);
            ctx.stroke();
        }
        
        // 绘制云朵
        function drawClouds() {
            ctx.fillStyle = 'rgba(255, 255, 255, 0.8)';
            
            // 静态云朵
            drawCloud(100, 50, 30);
            drawCloud(300, 80, 40);
            drawCloud(500, 60, 35);
            drawCloud(700, 90, 45);
        }
        
        // 绘制单个云朵
        function drawCloud(x, y, size) {
            ctx.beginPath();
            ctx.arc(x, y, size, 0, Math.PI * 2);
            ctx.arc(x + size, y, size * 0.8, 0, Math.PI * 2);
            ctx.arc(x - size, y, size * 0.8, 0, Math.PI * 2);
            ctx.fill();
        }
        
        // 游戏结束
        function gameOver() {
            gameRunning = false;
            document.getElementById('finalScore').textContent = score;
            document.getElementById('gameOver').style.display = 'block';
        }
        
        // 重新开始游戏
        function restartGame() {
            // 重置游戏状态
            gameRunning = true;
            score = 0;
            gameSpeed = 3;
            lastObstacleTime = 0;
            
            // 清除所有障碍物和平台
            for (const obstacle of obstacles) {
                World.remove(engine.world, obstacle);
            }
            obstacles = [];
            
            for (const platform of platforms) {
                World.remove(engine.world, platform);
            }
            platforms = [];
            
            // 重置玩家位置
            Body.setPosition(player, { x: 100, y: 300 });
            Body.setVelocity(player, { x: 0, y: 0 });
            
            // 隐藏游戏结束界面
            document.getElementById('gameOver').style.display = 'none';
            
            // 重新创建初始平台
            createInitialPlatforms();
            
            // 重新开始游戏循环
            gameLoop();
        }
        
        // 页面加载完成后初始化游戏
        window.onload = function() {
            initGame();
        };
    </script>
</body>
</html>
```
