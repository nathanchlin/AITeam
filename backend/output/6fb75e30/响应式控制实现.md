    <script>
        // 游戏配置
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        const scoreElement = document.getElementById('scoreValue');
        const gameOverElement = document.getElementById('gameOver');
        const finalScoreElement = document.getElementById('finalScore');
        const restartBtn = document.getElementById('restartBtn');
        
        // 设置画布尺寸
        function resizeCanvas() {
            const container = document.querySelector('.game-container');
            canvas.width = container.clientWidth;
            canvas.height = container.clientHeight;
        }
        
        window.addEventListener('resize', resizeCanvas);
        resizeCanvas();
        
        // 游戏状态
        let gameRunning = true;
        let score = 0;
        let gameSpeed = 5;
        let gravity = 0.5;
        let friction = 0.9;
        
        // 火柴人对象
        const stickman = {
            x: 100,
            y: canvas.height - 100,
            width: 30,
            height: 60,
            velocityY: 0,
            jumping: false,
            movingLeft: false,
            movingRight: false,
            color: '#fff'
        };
        
        // 地面
        const ground = {
            x: 0,
            y: canvas.height - 40,
            width: canvas.width,
            height: 40,
            color: '#555'
        };
        
        // 障碍物数组
        let obstacles = [];
        
        // 云朵数组（背景装饰）
        let clouds = [];
        
        // 粒子效果数组
        let particles = [];
        
        // 初始化云朵
        function initClouds() {
            clouds = [];
            for (let i = 0; i < 5; i++) {
                clouds.push({
                    x: Math.random() * canvas.width,
                    y: Math.random() * 150,
                    width: 60 + Math.random() * 40,
                    height: 30 + Math.random() * 20,
                    speed: 0.5 + Math.random() * 0.5
                });
            }
        }
        
        // 创建障碍物
        function createObstacle() {
            const height = 30 + Math.random() * 40;
            obstacles.push({
                x: canvas.width,
                y: ground.y - height,
                width: 20 + Math.random() * 30,
                height: height,
                color: '#e74c3c'
            });
        }
        
        // 创建粒子效果
        function createParticles(x, y, count) {
            for (let i = 0; i < count; i++) {
                particles.push({
                    x: x,
                    y: y,
                    velocityX: -2 + Math.random() * 4,
                    velocityY: -2 + Math.random() * 4,
                    size: 2 + Math.random() * 3,
                    life: 1.0
                });
            }
        }
        
        // 绘制火柴人
        function drawStickman() {
            ctx.fillStyle = stickman.color;
            ctx.strokeStyle = stickman.color;
            ctx.lineWidth = 2;
            
            // 头部
            ctx.beginPath();
            ctx.arc(stickman.x + stickman.width / 2, stickman.y + 10, 8, 0, Math.PI * 2);
            ctx.stroke();
            
            // 身体
            ctx.beginPath();
            ctx.moveTo(stickman.x + stickman.width / 2, stickman.y + 18);
            ctx.lineTo(stickman.x + stickman.width / 2, stickman.y + stickman.height - 10);
            ctx.stroke();
            
            // 手臂
            ctx.beginPath();
            ctx.moveTo(stickman.x + stickman.width / 2, stickman.y + 25);
            ctx.lineTo(stickman.x + 5, stickman.y + 35);
            ctx.moveTo(stickman.x + stickman.width / 2, stickman.y + 25);
            ctx.lineTo(stickman.x + stickman.width - 5, stickman.y + 35);
            ctx.stroke();
            
            // 腿部
            const legOffset = stickman.jumping ? 5 : 0;
            ctx.beginPath();
            ctx.moveTo(stickman.x + stickman.width / 2, stickman.y + stickman.height - 10);
            ctx.lineTo(stickman.x + 5 + legOffset, stickman.y + stickman.height);
            ctx.moveTo(stickman.x + stickman.width / 2, stickman.y + stickman.height - 10);
            ctx.lineTo(stickman.x + stickman.width - 5 - legOffset, stickman.y + stickman.height);
            ctx.stroke();
        }
        
        // 绘制地面
        function drawGround() {
            ctx.fillStyle = ground.color;
            ctx.fillRect(ground.x, ground.y, ground.width, ground.height);
            
            // 地面纹理
            ctx.strokeStyle = '#444';
            ctx.lineWidth = 1;
            for (let i = 0; i < ground.width; i += 20) {
                ctx.beginPath();
                ctx.moveTo(i, ground.y);
                ctx.lineTo(i + 10, ground.y + 10);
                ctx.stroke();
            }
        }
        
        // 绘制云朵
        function drawClouds() {
            ctx.fillStyle = 'rgba(255, 255, 255, 0.8)';
            clouds.forEach(cloud => {
                ctx.beginPath();
                ctx.arc(cloud.x, cloud.y, cloud.height / 2, 0, Math.PI * 2);
                ctx.arc(cloud.x + cloud.width * 0.25, cloud.y - cloud.height * 0.1, cloud.height / 2, 0, Math.PI * 2);
                ctx.arc(cloud.x + cloud.width * 0.5, cloud.y, cloud.height / 2, 0, Math.PI * 2);
                ctx.arc(cloud.x + cloud.width * 0.75, cloud.y - cloud.height * 0.05, cloud.height / 2, 0, Math.PI * 2);
                ctx.fill();
            });
        }
        
        // 绘制障碍物
        function drawObstacles() {
            obstacles.forEach(obstacle => {
                ctx.fillStyle = obstacle.color;
                ctx.fillRect(obstacle.x, obstacle.y, obstacle.width, obstacle.height);
                
                // 障碍物高光
                ctx.fillStyle = 'rgba(255, 255, 255, 0.3)';
                ctx.fillRect(obstacle.x, obstacle.y, obstacle.width, 5);
            });
        }
        
        // 绘制粒子
        function drawParticles() {
            particles.forEach((particle, index) => {
                ctx.fillStyle = `rgba(255, 255, 255, ${particle.life})`;
                ctx.beginPath();
                ctx.arc(particle.x, particle.y, particle.size, 0, Math.PI * 2);
                ctx.fill();
                
                // 更新粒子
                particle.x += particle.velocityX;
                particle.y += particle.velocityY;
                particle.life -= 0.02;
                
                // 移除消失的粒子
                if (particle.life <= 0) {
                    particles.splice(index, 1);
                }
            });
        }
        
        // 绘制背景
        function drawBackground() {
            // 天空渐变
            const gradient = ctx.createLinearGradient(0, 0, 0, canvas.height);
            gradient.addColorStop(0, '#87CEEB');
            gradient.addColorStop(1, '#E0F7FA');
            ctx.fillStyle = gradient;
            ctx.fillRect(0, 0, canvas.width, canvas.height);
        }
        
        // 更新游戏状态
        function update() {
            if (!gameRunning) return;
            
            // 更新火柴人位置
            if (stickman.movingLeft) {
                stickman.x -= 5;
            }
            if (stickman.movingRight) {
                stickman.x += 5;
            }
            
            // 应用重力
            stickman.velocityY += gravity;
            stickman.y += stickman.velocityY;
            
            // 地面碰撞检测
            if (stickman.y + stickman.height > ground.y) {
                stickman.y = ground.y - stickman.height;
                stickman.velocityY = 0;
                stickman.jumping = false;
            }
            
            // 边界检测
            if (stickman.x < 0) stickman.x = 0;
            if (stickman.x + stickman.width > canvas.width) stickman.x = canvas.width - stickman.width;
            
            // 更新云朵
            clouds.forEach(cloud => {
                cloud.x -= cloud.speed;
                if (cloud.x + cloud.width < 0) {
                    cloud.x = canvas.width + Math.random() * 100;
                    cloud.y = Math.random() * 150;
                }
            });
            
            // 更新障碍物
            obstacles.forEach((obstacle, index) => {
                obstacle.x -= gameSpeed;
                
                // 移除屏幕外的障碍物
                if (obstacle.x + obstacle.width < 0) {
                    obstacles.splice(index, 1);
                    score += 10;
                    scoreElement.textContent = score;
                    
                    // 增加游戏难度
                    if (score % 100 === 0) {
                        gameSpeed += 0.5;
                    }
                }
                
                // 碰撞检测
                if (
                    stickman.x < obstacle.x + obstacle.width &&
                    stickman.x + stickman.width > obstacle.x &&
                    stickman.y < obstacle.y + obstacle.height &&
                    stickman.y + stickman.height > obstacle.y
                ) {
                    gameOver();
                }
            });
            
            // 创建新障碍物
            if (obstacles.length === 0 || obstacles[obstacles.length - 1].x < canvas.width - 200 - Math.random() * 200) {
                createObstacle();
            }
            
            // 创建跳跃粒子效果
            if (stickman.jumping && stickman.velocityY > 0) {
                if (Math.random() < 0.3) {
                    createParticles(
                        stickman.x + stickman.width / 2,
                        stickman.y + stickman.height,
                        2
                    );
                }
            }
        }
        
        // 渲染游戏
        function render() {
            // 清空画布
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            // 绘制背景
            drawBackground();
            
            // 绘制云朵
            drawClouds();
            
            // 绘制地面
            drawGround();
            
            // 绘制障碍物
            drawObstacles();
            
            // 绘制火柴人
            drawStickman();
            
            // 绘制粒子
            drawParticles();
        }
        
        // 游戏循环
        function gameLoop() {
            update();
            render();
            
            if (gameRunning) {
                requestAnimationFrame(gameLoop);
            }
        }
        
        // 跳跃功能
        function jump() {
            if (!stickman.jumping && gameRunning) {
                stickman.velocityY = -12;
                stickman.jumping = true;
                createParticles(stickman.x + stickman.width / 2, stickman.y + stickman.height, 5);
            }
        }
        
        // 游戏结束
        function gameOver() {
            gameRunning = false;
            finalScoreElement.textContent = score;
            gameOverElement.style.display = 'block';
        }
        
        // 重新开始游戏
        function restartGame() {
            gameRunning = true;
            score = 0;
            gameSpeed = 5;
            scoreElement.textContent = score;
            gameOverElement.style.display = 'none';
            
            // 重置火柴人位置
            stickman.x = 100;
            stickman.y = canvas.height - 100;
            stickman.velocityY = 0;
            stickman.jumping = false;
            stickman.movingLeft = false;
            stickman.movingRight = false;
            
            // 清空障碍物
            obstacles = [];
            
            // 重新初始化
            initClouds();
            createObstacle();
            
            // 重新开始游戏循环
            gameLoop();
        }
        
        // 键盘控制
        document.addEventListener('keydown', (e) => {
            if (!gameRunning) return;
            
            switch(e.key) {
                case 'ArrowLeft':
                case 'a':
                case 'A':
                    stickman.movingLeft = true;
                    break;
                case 'ArrowRight':
                case 'd':
                case 'D':
                    stickman.movingRight = true;
                    break;
                case 'ArrowUp':
                case ' ':
                case 'w':
                case 'W':
                    jump();
                    break;
            }
        });
        
        document.addEventListener('keyup', (e) => {
            switch(e.key) {
                case 'ArrowLeft':
                case 'a':
                case 'A':
                    stickman.movingLeft = false;
                    break;
                case 'ArrowRight':
                case 'd':
                case 'D':
                    stickman.movingRight = false;
                    break;
            }
        });
        
        // 触屏控制
        let touchStartX = null;
        
        document.addEventListener('touchstart', (e) => {
            if (!gameRunning) return;
            
            const touch = e.touches[0];
            touchStartX = touch.clientX;
            
            // 检查是否点击了控制按钮
            if (e.target.id === 'leftBtn') {
                stickman.movingLeft = true;
            } else if (e.target.id === 'rightBtn') {
                stickman.movingRight = true;
            } else if (e.target.id === 'jumpBtn') {
                jump();
            }
        });
        
        document.addEventListener('touchmove', (e) => {
            if (!gameRunning || touchStartX === null) return;
            
            const touch = e.touches[0];
            const deltaX = touch.clientX - touchStartX;
            
            // 根据滑动方向控制移动
            if (deltaX < -30) {
                stickman.movingLeft = true;
                stickman.movingRight = false;
            } else if (deltaX
