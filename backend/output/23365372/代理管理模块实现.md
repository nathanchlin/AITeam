    <script>
        // 代理管理模块
        class AgentManager {
            constructor() {
                this.agents = this.loadAgents();
                this.init();
            }
            
            // 从本地存储加载代理数据
            loadAgents() {
                const storedAgents = localStorage.getItem('agents');
                if (storedAgents) {
                    return JSON.parse(storedAgents);
                }
                // 默认代理数据
                return [
                    {
                        id: 'agent-001',
                        name: '北京代理-01',
                        ip: '192.168.1.101',
                        port: 8080,
                        region: 'north',
                        capacity: 100,
                        status: 'online',
                        description: '北京地区主要代理',
                        registeredAt: new Date().toISOString(),
                        lastUpdated: new Date().toISOString()
                    },
                    {
                        id: 'agent-002',
                        name: '上海代理-01',
                        ip: '192.168.1.102',
                        port: 8080,
                        region: 'east',
                        capacity: 150,
                        status: 'online',
                        description: '上海地区主要代理',
                        registeredAt: new Date().toISOString(),
                        lastUpdated: new Date().toISOString()
                    },
                    {
                        id: 'agent-003',
                        name: '广州代理-01',
                        ip: '192.168.1.103',
                        port: 8080,
                        region: 'south',
                        capacity: 120,
                        status: 'maintenance',
                        description: '广州地区代理，维护中',
                        registeredAt: new Date().toISOString(),
                        lastUpdated: new Date().toISOString()
                    }
                ];
            }
            
            // 保存代理数据到本地存储
            saveAgents() {
                localStorage.setItem('agents', JSON.stringify(this.agents));
            }
            
            // 初始化
            init() {
                this.setupEventListeners();
                this.renderAgentList();
                this.updateStats();
                this.renderBatchAgents();
            }
            
            // 设置事件监听器
            setupEventListeners() {
                // Tab切换
                document.querySelectorAll('.tab').forEach(tab => {
                    tab.addEventListener('click', (e) => {
                        this.switchTab(e.target.dataset.tab);
                    });
                });
                
                // 注册表单提交
                document.getElementById('register-form').addEventListener('submit', (e) => {
                    e.preventDefault();
                    this.registerAgent();
                });
                
                // 批量操作表单提交
                document.getElementById('batch-form').addEventListener('submit', (e) => {
                    e.preventDefault();
                    this.executeBatchAction();
                });
            }
            
            // 切换Tab
            switchTab(tabId) {
                document.querySelectorAll('.tab').forEach(tab => {
                    tab.classList.remove('active');
                });
                document.querySelectorAll('.tab-content').forEach(content => {
                    content.classList.remove('active');
                });
                
                document.querySelector(`[data-tab="${tabId}"]`).classList.add('active');
                document.getElementById(`${tabId}-tab`).classList.add('active');
            }
            
            // 显示消息
            showMessage(message, type = 'info') {
                const messageContainer = document.getElementById('message-container');
                messageContainer.innerHTML = `<div class="message ${type}">${message}</div>`;
                
                setTimeout(() => {
                    messageContainer.innerHTML = '';
                }, 3000);
            }
            
            // 注册代理
            registerAgent() {
                const name = document.getElementById('agent-name').value;
                const ip = document.getElementById('agent-ip').value;
                const port = parseInt(document.getElementById('agent-port').value);
                const region = document.getElementById('agent-region').value;
                const capacity = parseInt(document.getElementById('agent-capacity').value);
                const description = document.getElementById('agent-description').value;
                
                // 检查代理名称是否已存在
                if (this.agents.some(agent => agent.name === name)) {
                    this.showMessage('代理名称已存在，请使用其他名称', 'error');
                    return;
                }
                
                // 创建新代理
                const newAgent = {
                    id: `agent-${Date.now()}`,
                    name,
                    ip,
                    port,
                    region,
                    capacity,
                    status: 'offline',
                    description,
                    registeredAt: new Date().toISOString(),
                    lastUpdated: new Date().toISOString()
                };
                
                this.agents.push(newAgent);
                this.saveAgents();
                this.renderAgentList();
                this.updateStats();
                this.renderBatchAgents();
                
                // 重置表单
                document.getElementById('register-form').reset();
                
                this.showMessage(`代理 ${name} 注册成功`, 'success');
            }
            
            // 更新代理
            updateAgent(agentId) {
                const agent = this.agents.find(a => a.id === agentId);
                if (!agent) return;
                
                const name = prompt('代理名称:', agent.name);
                if (!name || name.trim() === '') return;
                
                // 检查代理名称是否已被其他代理使用
                if (this.agents.some(a => a.id !== agentId && a.name === name)) {
                    this.showMessage('代理名称已存在，请使用其他名称', 'error');
                    return;
                }
                
                const ip = prompt('IP地址:', agent.ip);
                if (!ip || ip.trim() === '') return;
                
                const port = prompt('端口:', agent.port);
                if (!port || isNaN(port)) return;
                
                const capacity = prompt('容量:', agent.capacity);
                if (!capacity || isNaN(capacity)) return;
                
                const description = prompt('描述:', agent.description);
                
                // 更新代理信息
                agent.name = name;
                agent.ip = ip;
                agent.port = parseInt(port);
                agent.capacity = parseInt(capacity);
                agent.description = description || '';
                agent.lastUpdated = new Date().toISOString();
                
                this.saveAgents();
                this.renderAgentList();
                this.updateStats();
                this.renderBatchAgents();
                
                this.showMessage(`代理 ${name} 更新成功`, 'success');
            }
            
            // 下线代理
            offlineAgent(agentId) {
                const agent = this.agents.find(a => a.id === agentId);
                if (!agent) return;
                
                if (confirm(`确定要将代理 ${agent.name} 下线吗？`)) {
                    agent.status = 'offline';
                    agent.lastUpdated = new Date().toISOString();
                    this.saveAgents();
                    this.renderAgentList();
                    this.updateStats();
                    this.renderBatchAgents();
                    
                    this.showMessage(`代理 ${agent.name} 已下线`, 'info');
                }
            }
            
            // 上线代理
            onlineAgent(agentId) {
                const agent = this.agents.find(a => a.id === agentId);
                if (!agent) return;
                
                if (confirm(`确定要将代理 ${agent.name} 上线吗？`)) {
                    agent.status = 'online';
                    agent.lastUpdated = new Date().toISOString();
                    this.saveAgents();
                    this.renderAgentList();
                    this.updateStats();
                    this.renderBatchAgents();
                    
                    this.showMessage(`代理 ${agent.name} 已上线`, 'success');
                }
            }
            
            // 维护代理
            maintenanceAgent(agentId) {
                const agent = this.agents.find(a => a.id === agentId);
                if (!agent) return;
                
                if (confirm(`确定要将代理 ${agent.name} 设置为维护状态吗？`)) {
                    agent.status = 'maintenance';
                    agent.lastUpdated = new Date().toISOString();
                    this.saveAgents();
                    this.renderAgentList();
                    this.updateStats();
                    this.renderBatchAgents();
                    
                    this.showMessage(`代理 ${agent.name} 已设置为维护状态`, 'info');
                }
            }
            
            // 删除代理
            deleteAgent(agentId) {
                const agent = this.agents.find(a => a.id === agentId);
                if (!agent) return;
                
                if (confirm(`确定要删除代理 ${agent.name} 吗？此操作不可恢复。`)) {
                    this.agents = this.agents.filter(a => a.id !== agentId);
                    this.saveAgents();
                    this.renderAgentList();
                    this.updateStats();
                    this.renderBatchAgents();
                    
                    this.showMessage(`代理 ${agent.name} 已删除`, 'info');
                }
            }
            
            // 渲染代理列表
            renderAgentList() {
                const agentList = document.getElementById('agent-list');
                
                if (this.agents.length === 0) {
                    agentList.innerHTML = '<p>暂无代理数据</p>';
                    return
