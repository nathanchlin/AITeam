    <script>
        // 测试框架实现
        (function() {
            // 测试框架状态
            const testFramework = {
                testSuites: [],
                testResults: [],
                currentSuite: null,
                currentTest: null,
                running: false,
                passedCount: 0,
                failedCount: 0,
                totalCount: 0,
                startTime: null,
                endTime: null
            };
            
            // 断言方法
            const assertions = {
                assertTrue: function(value, message) {
                    if (!value) {
                        throw new AssertionError(message || '预期值为真，但实际为假');
                    }
                },
                
                assertFalse: function(value, message) {
                    if (value) {
                        throw new AssertionError(message || '预期值为假，但实际为真');
                    }
                },
                
                assertEquals: function(expected, actual, message) {
                    if (expected !== actual) {
                        throw new AssertionError(
                            message || `期望值 ${expected} 与实际值 ${actual} 不相等`
                        );
                    }
                },
                
                assertNotEquals: function(expected, actual, message) {
                    if (expected === actual) {
                        throw new AssertionError(
                            message || `期望值 ${expected} 与实际值 ${actual} 相等，但不应该相等`
                        );
                    }
                },
                
                assertNull: function(value, message) {
                    if (value !== null) {
                        throw new AssertionError(message || `期望值为 null，但实际为 ${value}`);
                    }
                },
                
                assertNotNull: function(value, message) {
                    if (value === null) {
                        throw new AssertionError(message || '期望值不为 null，但实际为 null');
                    }
                },
                
                assertUndefined: function(value, message) {
                    if (value !== undefined) {
                        throw new AssertionError(message || `期望值为 undefined，但实际为 ${value}`);
                    }
                },
                
                assertNotUndefined: function(value, message) {
                    if (value === undefined) {
                        throw new AssertionError(message || '期望值不为 undefined，但实际为 undefined');
                    }
                },
                
                assertArray: function(value, message) {
                    if (!Array.isArray(value)) {
                        throw new AssertionError(message || `期望值为数组，但实际为 ${typeof value}`);
                    }
                },
                
                assertObject: function(value, message) {
                    if (typeof value !== 'object' || value === null || Array.isArray(value)) {
                        throw new AssertionError(message || `期望值为对象，但实际为 ${typeof value}`);
                    }
                },
                
                assertString: function(value, message) {
                    if (typeof value !== 'string') {
                        throw new AssertionError(message || `期望值为字符串，但实际为 ${typeof value}`);
                    }
                },
                
                assertNumber: function(value, message) {
                    if (typeof value !== 'number' || isNaN(value)) {
                        throw new AssertionError(message || `期望值为数字，但实际为 ${typeof value}`);
                    }
                },
                
                assertFunction: function(value, message) {
                    if (typeof value !== 'function') {
                        throw new AssertionError(message || `期望值为函数，但实际为 ${typeof value}`);
                    }
                },
                
                assertThrows: function(func, message) {
                    try {
                        func();
                        throw new AssertionError(message || '函数应该抛出异常，但没有抛出');
                    } catch (e) {
                        if (!(e instanceof AssertionError)) {
                            // 如果是其他类型的异常，我们认为测试通过
                        }
                    }
                },
                
                assertContains: function(container, item, message) {
                    let contains = false;
                    
                    if (Array.isArray(container)) {
                        contains = container.includes(item);
                    } else if (typeof container === 'string') {
                        contains = container.includes(item);
                    } else if (typeof container === 'object') {
                        contains = Object.prototype.hasOwnProperty.call(container, item);
                    }
                    
                    if (!contains) {
                        throw new AssertionError(message || `容器不包含项目: ${item}`);
                    }
                },
                
                assertGreater: function(a, b, message) {
                    if (!(a > b)) {
                        throw new AssertionError(message || `${a} 不大于 ${b}`);
                    }
                },
                
                assertLess: function(a, b, message) {
                    if (!(a < b)) {
                        throw new AssertionError(message || `${a} 不小于 ${b}`);
                    }
                },
                
                assertContainsText: function(string, substring, message) {
                    if (typeof string !== 'string') {
                        throw new AssertionError(message || '第一个参数应该是字符串');
                    }
                    
                    if (typeof substring !== 'string') {
                        throw new AssertionError(message || '第二个参数应该是字符串');
                    }
                    
                    if (!string.includes(substring)) {
                        throw new AssertionError(message || `字符串 "${string}" 不包含 "${substring}"`);
                    }
                }
            };
            
            // 自定义断言错误类
            function AssertionError(message) {
                this.name = 'AssertionError';
                this.message = message;
                this.stack = (new Error()).stack;
            }
            
            AssertionError.prototype = Object.create(Error.prototype);
            AssertionError.prototype.constructor = AssertionError;
            
            // 添加测试套件
            function addTestSuite(name, testFunction) {
                const suite = {
                    name: name,
                    tests: [],
                    passed: 0,
                    failed: 0
                };
                
                testFramework.testSuites.push(suite);
                testFramework.currentSuite = suite;
                
                try {
                    testFunction();
                } catch (e) {
                    console.error(`测试套件 "${name}" 执行出错:`, e);
                }
                
                testFramework.currentSuite = null;
            }
            
            // 添加测试用例
            function addTestCase(name, testFunction) {
                if (!testFramework.currentSuite) {
                    console.error('无法添加测试用例: 没有活动的测试套件');
                    return;
                }
                
                const test = {
                    name: name,
                    function: testFunction,
                    passed: false,
                    error: null
                };
                
                testFramework.currentSuite.tests.push(test);
                testFramework.testSuites[testFramework.testSuites.length - 1] = testFramework.currentSuite;
            }
            
            // 运行单个测试
            function runTest(test, suite) {
                testFramework.currentTest = test;
                testFramework.currentSuite = suite;
                
                try {
                    test.function();
                    test.passed = true;
                    suite.passed++;
                    testFramework.passedCount++;
                } catch (e) {
                    test.passed = false;
                    test.error = e;
                    suite.failed++;
                    testFramework.failedCount++;
                } finally {
                    testFramework.currentTest = null;
                    testFramework.currentSuite = null;
                }
            }
            
            // 运行所有测试
            function runAllTests() {
                if (testFramework.running) {
                    console.warn('测试正在运行中...');
                    return;
                }
                
                testFramework.running = true;
                testFramework.testResults = [];
                testFramework.passedCount = 0;
                testFramework.failedCount = 0;
                testFramework.totalCount = 0;
                testFramework.startTime = new Date();
                
                // 计算总测试数
                testFramework.testSuites.forEach(suite => {
                    testFramework.totalCount += suite.tests.length;
                });
                
                // 重置进度条
                updateProgressBar(0);
                
                // 按顺序运行所有测试
                runTestsSequentially(0);
            }
            
            // 按顺序运行测试
            function runTestsSequentially(index) {
                if (index >= testFramework.testSuites.length) {
                    // 所有测试套件都已完成
                    finishTests();
                    return;
